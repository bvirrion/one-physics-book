\section*{Hoofdstuk \ref{ch:b1:point-kinematics} --- Kinematica van een punt}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{12}{m/s} op \qty{3.0}{s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ op $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). Eenparige cirkelbeweging: constante baansnelheid,
\omterm{cor:b1:point-kinematics:circular}{centripetale versnelling}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, oftewel $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Hoogste punt op $t = 7.71/9.81 = \qty{0.79}{s}$, hoogte $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Dracht $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
Landing: $(9.19, -7.71)$, \qty{12}{m/s} onder $40^\circ$ met de
horizontaal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. Na \qty{5}{s}: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. De term $2\dot r\dot\theta$ bestaat uit twee gelijke helften: de
richting van de radiale \omterm{def:b1:point-kinematics:velocity}{snelheid} draait met \omterm{def:b1:point-kinematics:velocity}{snelheid} $\dot\theta$ (wat
$\dot r\dot\theta\,\vect e_\theta$ geeft), en naar buiten bewegen vergroot de
orthoradiale \omterm{def:b1:point-kinematics:velocity}{snelheid} $r\dot\theta$ (nog eens $\dot r\dot\theta$).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ naar de as toe; spoed $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ loopt een kwart periode voor op $x$, en $a = -\omega^2x$ is tegengesteld aan $x$.
$\abs v$ is het grootst in $x = 0$, $\abs a$ in de uitersten $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Vervolgens is
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, en nog een keer afleiden
met de productregel geeft de \omterm{def:b1:point-kinematics:velocity}{versnelling} uit
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Hoogste punt: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$.
Vertrekpunt: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$.
De straal in het hoogste punt is een factor $\cos^3\alpha$ kleiner: de parabool buigt het sterkst
waar zij het traagst is en waar $\vect g$ volledig normaal staat.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. Aan de grond ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ en $\vect a = (0, v^2/R)$, omhoog. Bovenin
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: \omterm{def:b1:point-kinematics:velocity}{snelheid} $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. Op $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). De eerste is de centripetale term
van de draaiing; de tweede, $2\dot r\dot\theta$, bestaat alleen omdat de
radiale richting zelf draait --- op de grond geeft rechtuit lopen met
constante \omterm{def:b1:point-kinematics:velocity}{snelheid} helemaal geen \omterm{def:b1:point-kinematics:velocity}{versnelling}.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (beweging met toenemende $\theta$),
$\vect N = -\vect e_r$ (naar $C$), $\rho = R$.

\textbf{3.} Bovenin: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$. Zijkanten:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ en $v = R\dot\theta$:
$\dot v = -g\sin\theta$. Negatief voor $0 < \theta < \pi$ (vertragen op de
weg omhoog), positief op de weg omlaag.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: onderin
\qty{62.5}{m/s^2}; zijkanten $62.5 - 19.6 = \qty{42.9}{m/s^2}$; bovenin $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} Onderin: \qty{62.5}{m/s^2} (geen tangentieel deel); zijkanten:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; bovenin: \qty{23.3}{m/s^2},
recht omlaag naar $C$ gericht: hoek $0$ met de verticaal.

\textbf{7.} Loodrecht onderin en bovenin ($a_T = 0$); nooit
evenwijdig ($a_N > 0$ overal).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
onderin \qty{2.50}{rad/s}, bovenin \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: de vergelijking van een
wiskundige slinger met lengte $R$ --- het wagentje op zijn \omterm{def:b1:point-kinematics:frame}{baan} is een slingerbol
die over de top gaat.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Gemiddelde van $25$, $20.7$, $15.3$, $20.7$: ongeveer \qty{20}{m/s};
lengte $2\pi R = \qty{63}{m}$: ruwweg \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: ja. Minimaal: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} Onderin: $\vect a = 62.5$ omhoog, $\vect g$ omlaag:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} Bovenin: beide omlaag, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
wat de inzittende in de stoel drukt (naar het middelpunt toe, dus omhoog voor
de ondersteboven hangende inzittende).

\textbf{15.} Zijkanten: $\vect a = 42.9$ horizontaal naar $C$ plus $9.8$
omlaag (tangentieel); $\vect a - \vect g$ is $42.9$ horizontaal: $4.4g$
zijwaarts, naar het middelpunt toe.

\textbf{16.} Niet gehaald: $7.4g$ onderin (bovenin en opzij is het in orde).

\textbf{17.} Schijnbare gewichtloosheid bovenin: $v_{\text{boven}}^2 = gR$;
dan is $v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ onderin en is de belasting
$5g + g = 6g$, onafhankelijk van $R$.

\textbf{18.} $a_N = v^2/\rho$: $v$ ligt vast door de hoogte (energie), $\rho$
door de meetkunde van de \omterm{def:b1:point-kinematics:frame}{baan}; een andere $\rho$ op dezelfde hoogte verandert
alleen $a_N$.

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{boven}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: zij groeit lineair
vanaf nul --- een constante verandering van de \omterm{def:b1:point-kinematics:velocity}{versnelling} --- in plaats van
te springen van $0$ naar $v^2/R$ bij het inrijden van een cirkel, wat de nek
als een klap voelt.

\textbf{22.} Een druppel: wijd en flauw gekromd onderin waar het
wagentje snel gaat, krap bovenin waar het traag gaat. Zij houdt de belasting rond
$3g$ over het hele traject en brengt haar geleidelijk op.

\textbf{23.} Kijklijn: horizontale afstand \qty{50}{m}, hoogte
\qty{20}{m}: $r = \qty{53.9}{m}$, elevatie $\beta = 21.8^\circ$. De
transversale component van de horizontale \omterm{def:b1:point-kinematics:velocity}{snelheid} is $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Vanuit een punt buiten het middelpunt is de hoeksnelheid van de
kijklijn niet constant (snel als het wagentje dichtbij is, traag als het ver is):
een \omterm{def:b1:optical-instruments:camera}{camera} met constante draaisnelheid loopt van het wagentje af.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Poolcoördinaten} in $C$ voor \omterm{def:b1:point-kinematics:velocity}{snelheid} en \omterm{def:b1:point-kinematics:velocity}{versnelling} op
de cirkel; de frenetsplitsing voor ``versnellen'' tegenover ``draaien'' en
voor de clothoïde; \omterm{def:b1:point-kinematics:cylindrical}{poolcoördinaten} bij de \omterm{def:b1:optical-instruments:camera}{camera} voor de volgsnelheid;
energie (volgende hoofdstukken) voor de snelheidswet.
\end{solution}
