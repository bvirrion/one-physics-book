\section*{Hoofdstuk \ref{ch:b1:mirrors-thin-lenses} --- Spiegels en dunne lenzen}

\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$; $1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. Op \qty{60}{cm}: $\overline{SA'} = \qty{-30}{cm}$,
reëel, $\gamma = -0.5$. Op \qty{30}{cm}: $\overline{SA'} = \qty{-60}{cm}$,
reëel, $\gamma = -2$. Op \qty{10}{cm}: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$, $\overline{SA'} = \qty{+20}{cm}$ achter de spiegel, virtueel,
$\gamma = +2$ (rechtop, vergroot).
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:2}
$\overline{OA} = \qty{-20}{cm}$: $1/\overline{OA'} = 1/5 - 1/20 = 3/20$,
$\overline{OA'} = \qty{+6.7}{cm}$, reëel en omgekeerd, $\gamma = -1/3$.
$\overline{OA} = \qty{-3.0}{cm}$: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$,
$\overline{OA'} = \qty{-7.5}{cm}$, virtueel en rechtop, $\gamma = +2.5$.
\end{solution}

$V = 1/0.25 = \qty{+4.0}{\delta}$; $f' = 1/(-2.0) = \qty{-50}{cm}$;
$V = 3.0 - 1.0 = \qty{+2.0}{\delta}$, $f' = \qty{50}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:4}
$\overline{FA} = \qty{-15}{cm}$: $\overline{F'A'} = -f'^2/\overline{FA}
= +100/15 = \qty{6.7}{cm}$ voorbij $F'$, dus $\overline{OA'} = \qty{16.7}{cm}$;
$\gamma = f'/\overline{FA} = -0.67$. Descartes: $\overline{OA} =
\qty{-25}{cm}$, $1/\overline{OA'} = 1/10 - 1/25 = 3/50$,
$\overline{OA'} = \qty{16.7}{cm}$, $\gamma = 16.7/(-25) = -0.67$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}
$\overline{SF} = \qty{+1.0}{m}$, $\overline{SA} = \qty{-20}{m}$:
$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ achter
de spiegel (virtueel, rechtop); $\gamma = -0.95/(-20) = +0.048$, beeldhoogte
\qty{7.1}{cm}. Het brein leidt afstand af uit schijnbare grootte: een
auto die twintig keer kleiner op \qty{0.95}{m} wordt getoond, wordt gelezen
als een auto ver weg --- verder dan \qty{20}{m}.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$,
$\overline{OA'} = \qty{51.3}{mm}$: de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} beweegt \qty{1.3}{mm} van
de sensor af. $\gamma = 51.3/(-2000) = -0.026$. Voor $\abs\gamma = 24/1800
= 1/75$: $\overline{OA} = f'(1/\gamma - 1) = 50 \times (-76) =
\qty{-3.8}{m}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$. Het
\omterm{def:b1:mirrors-thin-lenses:image}{beeld} is omgekeerd ($\gamma < 0$), boven--onder en links--rechts: de dia
gaat er over $180^\circ$ gedraaid in.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
Voorwerp in het brandvlak $\Rightarrow$ de stralen uit elk punt verlaten de
\omterm{def:b1:mirrors-thin-lenses:lens}{lens} evenwijdig; de \omterm{prop:b1:rays-reflection-refraction:mirror}{vlakke spiegel} stuurt ze evenwijdig terug (symmetrische
richting); de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} brengt een evenwijdige bundel samen in haar brandvlak, in
het punt dat symmetrisch met het bronpunt ten opzichte van de as ligt: \omterm{def:b1:mirrors-thin-lenses:image}{beeld} in
het voorwerpsvlak, omgekeerd, $\gamma = -1$. De spiegel over $\alpha$ kantelen
draait de terugkerende bundel over $2\alpha$ en verschuift het \omterm{def:b1:mirrors-thin-lenses:image}{beeld} zijdelings
over $2\alpha f'$, nog altijd scherp in hetzelfde vlak.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:9}
$f' = (100^2 - 40^2)/(4 \times 100) = \qty{21}{cm}$. Kleinste afstand
$4f' = \qty{84}{cm}$, met $\gamma = -1$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ beeldt oneindig af in $F_1'$, \qty{20}{cm} voorbij $L_1$, dus \qty{5}{cm}
voorbij $L_2$: een virtueel voorwerp, $\overline{O_2A_1} = \qty{+5}{cm}$.
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: eindbeeld \qty{10}{cm}
voorbij $L_2$. Hoogte van het tussenbeeld $h_1 = f_1'\alpha = 20\alpha$ (cm);
$\gamma_2 = 10/5 = 2$, dus $h = 40\alpha$: gelijkwaardige \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand}
\qty{40}{cm}, in een huls van $15 + 10 = \qty{25}{cm}$ lang --- een telelens.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ geeft $\overline{SA'} =
3\overline{SA}$; beide reëel, ervóór: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, en de afstand scherm--lamp $2x = \qty{80}{cm}$:
$x = \qty{40}{cm}$. Spiegel \qty{40}{cm} van de lamp, \qty{120}{cm}
van het scherm; $1/\overline{SF} = -1/40 - 1/120 = -1/30$:
$R = 2 \times 30 = \qty{60}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$; beelddiameter $f'\theta =
\qty{4.6}{mm}$. Opgevangen vermogen $10^3 \times \pi(0.05)^2 = \qty{7.9}{W}$;
beeldoppervlak $\pi(2.3\times10^{-3})^2 = \qty{1.7e-5}{m^2}$; bestralingssterkte
$\qty{4.7e5}{W/m^2}$, $470$ keer het directe zonlicht: papier verkoolt en

\end{solution}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$; as georiënteerd met het licht mee,
afstanden algebraïsch vanaf $O$.

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$, dus $p^2 - Dp + Df_1' = 0$; reële wortels precies als
$D^2 - 4Df_1' \geq 0$, dus $D \geq 4f_1'$.

\textbf{3.} De wortels tellen op tot $D$: $p_1 + p_2 = D$, dus $p_2 = D - p_1$ ---
voorwerps- en beeldafstand wisselen om, de standen liggen symmetrisch om het
midden. $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$, dus $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} Alleen de verplaatsing $d$ van de huls tussen de twee scherpe
standen en de afstand voorwerp--scherm $D$ komen erin voor: geen van beide
vereist te weten waar $O$ binnen de huls zit.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$; dus
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ en $u(f_1') = \qty{0.14}{cm}$:
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ of \qty{60}{cm}: $\gamma =
-60/20 = -3$ (dia dicht bij de \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, vergroot: de projector) en
$\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: het reële \omterm{def:b1:mirrors-thin-lenses:image}{beeld} is over $180^\circ$ om de as
gedraaid, dus wordt de dia over $180^\circ$ gedraaid ingestoken (ondersteboven
en spiegelbeeldig).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$: de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} beweegt \qty{0.7}{mm}
naar de dia toe. $\gamma = 1000/(-15.23) = -65.7$: \omterm{def:b1:mirrors-thin-lenses:image}{beeld}
\qty{2.36}{m} breed.

\textbf{13.} Descartes afleiden: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, dus $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: het \omterm{def:b1:mirrors-thin-lenses:image}{beeld}
loopt uit focus zodra de film opwarmt --- vandaar dia's tussen glas of een
scherpstelling die de film volgt.

\textbf{15.} Gelijkvormige driehoeken van de opening naar het beeldpunt:
vlek $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$; vanaf \qty{5}{m} beslaat die
$\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$: niet gemerkt.

\textbf{16.} Oppervlak van de dia $8.64\times10^{-4}\ \unit{m^2}$: $E_{\text{dia}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$; het scherm is $\gamma^2$
keer groter, $\qty{2.16}{m^2}$: $E_{\text{scherm}} = \qty{0.93}{W/m^2}$;
hetzelfde vermogen, oppervlakken in verhouding $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$: het
tussenbeeld ligt \qty{29.3}{cm} \emph{voorbij} $L_2$ --- een virtueel
voorwerp voor $L_2$ (de stralen uit $L_1$ convergeren ernaartoe).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$; $\gamma_2 = 1200/29.3 =
41.0$; $\gamma = -60.9$.

\textbf{20.} Breedte $60.9 \times 36 = \qty{2.19}{m}$. $L_1$ alleen:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, breedte
\qty{2.86}{m}.

\textbf{21.} Bundel onder $\alpha$: $L_1$ brengt haar samen op hoogte $h_1 = f_1'\alpha$
in haar brandvlak, dat $\overline{O_2A_1} = f_1' - e$ van $L_2$ ligt;
$L_2$ beeldt dit punt af in $\overline{O_2A'}$ met $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ en vergroot met $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$, dus $h = \gamma_2 f_1'\alpha$ en $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, waarvan het omgekeerde de gegeven
formule is. $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ en $15/19.6 = 0.77$. Slechts bij benadering,
omdat de dia op eindige afstand staat en de twee werpafstanden vanaf
verschillende lenzen worden gemeten; voor een voorwerp op oneindig zou de
verhouding exact zijn.

\textbf{23.} Virtueel voorwerp op $p > 0$ voorbij een \omterm{def:b1:mirrors-thin-lenses:lens}{divergerende lens}:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, positief (\omterm{def:b1:mirrors-thin-lenses:image}{reëel
beeld}) precies als $p < \abs{f_2'}$.

\textbf{24.} Hetzelfde vermogen op een \omterm{def:b1:mirrors-thin-lenses:image}{beeld} met een oppervlak $(2.19/2.86)^2 = 0.59$
keer kleiner: bestralingssterkte $\times 1.7$.

\textbf{25.} De voorzetlens vermenigvuldigt de \omterm{def:b1:mirrors-thin-lenses:lens}{brandpuntsafstand} met
$f_{\mathrm{eq}}/f_1' = 1.3$; bij een gegeven werpafstand is het \omterm{def:b1:mirrors-thin-lenses:image}{beeld} $1.3$
keer kleiner in elke richting en $1.7$ keer helderder --- een \omterm{def:b1:mirrors-thin-lenses:lens}{lens} voor
grote werpafstand.
\end{solution}
