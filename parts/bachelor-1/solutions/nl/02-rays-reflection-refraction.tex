\section*{Hoofdstuk \ref{ch:b1:rays-reflection-refraction} --- Geometrische optica: stralen, reflectie, breking}

\begin{solution}{exo:b1:rays-reflection-refraction:1}
$v = c/n$: water \qty{2.26e8}{m/s}, glas \qty{2.00e8}{m/s}, diamant
\qty{1.24e8}{m/s}. Vezel: $t = nL/c = 1.48 \times 10^3/3.00\times10^8 =
\qty{4.9}{\micro s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:2}
$\sin i_2 = \sin i_1/1.33$: $30^\circ \to 22.1^\circ$; $60^\circ \to
40.6^\circ$. Vanuit water onder $50^\circ$: $1.33\sin 50^\circ = 1.02 > 1$,
geen gebroken straal --- totale reflectie ($i_c = 48.8^\circ$).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}
$i_c = \arcsin(1/n)$: glas $41.8^\circ$, diamant $24.4^\circ$, water
$48.8^\circ$. De hemelschijf beslaat $2 \times 48.8^\circ = 97.5^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
De ogen op \qty{1.60}{m}. De straal vanaf de voeten bereikt het oog na
reflectie op de halve ooghoogte, \qty{0.80}{m}; de straal vanaf de kruin
kaatst halverwege kruin en ogen, op \qty{1.65}{m}.
Spiegel van \qty{0.80}{m} tot \qty{1.65}{m}: hoogte \qty{0.85}{m}, de helft
van de lichaamslengte --- onafhankelijk van de afstand (de middenregel geldt
op elke afstand).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
Kleine hoeken: de schijnbare plaats is $h' = h/n$ bij kijken vanuit het
ijlere medium en $h' = nh$ vanuit het dichtere. Munt: $1.2/1.33 =
\qty{0.90}{m}$. Visser: $1.33 \times 1.8 = \qty{2.4}{m}$. Verantwoording:
voor een oppervlaktepunt op horizontale afstand $x$ is $\tan i_2 \approx i_2 =
x/h$ en $\tan i_1 \approx i_1 = x/h'$ met $i_1 = n i_2$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$, dus
$\frac{A + D_{\min}}{2} = 49.5^\circ$, $D_{\min} = 38.9^\circ$.
Uittreden vereist $r' \leq i_c$; ook geeft $\sin r = \sin i/n \leq 1/n$ dat
$r \leq i_c$; dus $A = r + r' \leq 2i_c$. Hier is $i_c = \arcsin(1/1.52) =
41.1^\circ$, $2i_c = 82.3^\circ > 60^\circ$: in orde.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$,
$\theta_a = 14.1^\circ$. $\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$. Over \qty{1}{km}:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}
$D \approx (n - 1)A$: rood $0.510 \times 10^\circ = 5.10^\circ$, blauw
$5.30^\circ$; spectrumbreedte $0.20^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
Intrede: $\sin r = \sin 45^\circ/1.5 = 0.471$, $r = 28.1^\circ$. De twee
vlakken zijn evenwijdig, dus is de inval op het tweede vlak $r$ en
$\sin i' = n\sin r = \sin i$: $i' = i$, de uittredende straal is evenwijdig
aan de invallende. Langs de inwendige weg met lengte $e/\cos r$ is de
loodrechte verschuiving $(e/\cos r)\sin(i - r)$:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
Twee brekingen ($2(i - r)$) en twee reflecties ($2(\pi - 2r)$):
$D = 2\pi + 2i - 6r$. $\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ geeft
$3\cos i = n\cos r$, $9\cos^2 i = n^2 - 1 + \cos^2 i$, $\cos^2 i =
(n^2 - 1)/8$. Voor $n = 1.333$: $\cos^2 i = 0.0971$, $i = 71.8^\circ$,
$\sin r = 0.950/1.333$, $r = 45.5^\circ$, $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$: de boog ligt op $230.9^\circ - 180^\circ =
51^\circ$ vanaf het punt tegenover de zon, buiten de primaire ($42^\circ$).
Rood ($n = 1.331$) op $50.4^\circ$, violet ($1.343$) op $53.6^\circ$:
de kleuren staan omgekeerd, violet buiten.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$, $i_c = 89.56^\circ$: stralen
binnen $0.44^\circ = \qty{7.7}{mrad}$ van het wegdek worden volledig
gereflecteerd (in het algemeen is de scherende hoek $\approx \sqrt{2\Delta n/n}$).
Vanaf \qty{1.2}{m} hoogte scheert de kijklijn onder die hoek voorbij
$1.2/0.0077 \approx \qty{160}{m}$: de weg weerspiegelt daar de hemel ---
de ``plas''.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$;
$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$, precies nul wanneer
$\sin i_1/v_1 = \sin i_2/v_2$, dat wil zeggen $n_1\sin i_1 = n_2\sin i_2$ met
$n = c/v$.
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$, onveranderd in glas;
$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$.

\textbf{4.} Een blote kern geleidt alleen zolang haar oppervlak volmaakt
schoon en onaangeraakt is: stof, water of een greep vervangt de lucht
plaatselijk door een hogere index en het licht ontsnapt, en krassen
verstrooien het. De mantel is een afgesloten, beheerst grensvlak met een
opzettelijk gekozen indexsprong.

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$; geleid precies dan als $\cos\theta' \geq
n_2/n_1$, dus $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$, dus
$\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$:
$\theta_a = 12.1^\circ$.

\textbf{8.} $\mathrm{NA} = 0.210$. Een grotere sprong verwijdt de kegel,
zodat meer licht van een divergerende bron (een led) wordt opgevangen.

\textbf{9.} De straal raakt de cirkel met straal $R_b$ (binnenrand) en
treft de cirkel met straal $R_b + 2a$ (buitenrand); in de rechthoekige
driehoek middelpunt--raakpunt--trefpunt voldoet de hoek in het trefpunt
tussen de straal en de radius (de normaal) aan
$\sin i = R_b/(R_b + 2a)$. Geleid dan en slechts dan als $\sin i \geq n_2/n_1$:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$. Strakkere bochten lekken.

\textbf{10.} $t_0 = Ln_1/c$; weg van de steilste straal $L/\cos\theta'_{\max} =
Ln_1/n_2$, dus $t_1 = Ln_1^2/(n_2 c)$.

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$; over \qty{2}{km}: \qty{101}{ns}.

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} Stralen ver van de as leggen langere wegen af, maar in een
lagere index en dus sneller; met een goed gekozen profiel heffen de twee
effecten elkaar op en komen alle stralen tegelijk aan.

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$: de kern is niet groot
vergeleken met de golflengte, zodat diffractie overheerst en stralen geen
betekenis hebben. De golfbehandeling toont één enkel geleid patroon: helemaal
geen intermodale spreiding; alleen chromatische dispersie en demping
blijven over.

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$:
$P/P_0 = 10^{-2}$, één procent.

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$, dus $L_{\max} =
30/0.20 = \qty{150}{km}$.

\textbf{18.} \qty{100}{km} kost \qty{20}{dB}: marge \qty{10}{dB}, een
factor $10$ in vermogen.

\textbf{19.} Led: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$, $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$. Laser:
$\Delta t_c = \qty{170}{ps}$, $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$.

\textbf{20.} $30/2.0 = \qty{15}{km}$, tien keer minder. Kwartsglas is het
doorzichtigst rond \qty{1550}{nm} (zijn dempingsminimum), en daar bestaan
optische versterkers.

\textbf{21.} $30/30 = \qty{1}{km}$ tussen versterkers, tegen
\qty{150}{km}: de vezel wint met een factor $150$.

\textbf{22.} Intermodaal: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s});
chromatisch (laser): $17 \times 0.1 \times 2 = \qty{3.4}{ps}$;
demping \qty{0.4}{dB}, verwaarloosbaar. De intermodale spreiding begrenst:
\qty{5}{Mbit/s}.

\textbf{23.} Geen intermodale spreiding; chromatisch \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}; demping \qty{20}{dB}, binnen het budget. De chromatische
dispersie begrenst, bij ongeveer \qty{3}{Gbit/s}.

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} Stapindex: \qty{10}{Mbit.s^{-1}.km}; monomodekanaal:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- een factor van ongeveer
$3\times10^4$. Eén feit: één enkele geleide weg, en dus geen meetkundige
(intermodale) spreiding van de aankomsttijden.
\end{solution}
