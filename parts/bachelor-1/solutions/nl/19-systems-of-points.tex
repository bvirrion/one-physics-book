\section*{Hoofdstuk \ref{ch:b1:systems-of-points} --- Systemen van punten; inleiding tot vaste lichamen}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$. Aarde--maan:
$384000/82 = \qty{4700}{km}$ van het middelpunt van de aarde, binnen de aarde
(\qty{6370}{km}): de aarde wiebelt om een punt \qty{1700}{km} onder
haar oppervlak.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: $v = \qty{2.0}{m/s}$. Kogel $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$, geweer \qty{8}{J}: uit de chemische energie
van het kruit; het lichte lichaam neemt er bijna alles van ($E_k = p^2/2m$
bij gelijke $p$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$; $E_k$: \qty{200}{kJ} ervoor, $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$ erna: $60\%$ verloren.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
Uiteinde: $\int_0^L x^2(M/L)\dd x = ML^2/3$; midden: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$; Huygens met $d = L/2$: $ML^2/12 + ML^2/4 = ML^2/3$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: cilinder $1.5$, hoepel $2$ maal $\tfrac12 Mv^2$.
Bij dezelfde snelheid meer energie voor de hoepel: hij rolt verder op een vlakke vloer
(dezelfde wrijving) en klimt hoger ($h = v^2(1 + k)/2g$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
HCl: $\mu = 35m_H/36 = 0.972m_H$; atoom: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$. Frequenties $\propto 1/\sqrt\mu$: $+1.4\%$ voor HCl, $+0.027\%$
voor waterstof (de isotoopverschuiving tussen de spectra van H en D berust erop).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$, $d = L/2$: $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$; equivalente lengte $2L/3 = \qty{0.67}{m}$. Een klap in het
slagmiddelpunt zet de staaf om het draaipunt aan het draaien zonder daar een
stootreactie te geven: raak de bal op de ``zoete plek'' van het slaghout en
de handen voelen geen naschok.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$; $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$, $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$;
$\Gamma = J\omega/t = \qty{314}{N.m}$; omwentelingen: $\tfrac12\omega t/2\pi = 250$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: bol ($k = 2/5$) $0.714$, cilinder ($1/2$)
$0.667$, hoepel ($1$) $0.500$ maal $g\sin\alpha$: de bol eerst, de hoepel laatst;
nergens een massa of een straal.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
Impuls: $0.010 \times 400 = 2.01\,v$: $v = \qty{2.0}{m/s}$; dan energie:
$h = v^2/2g = \qty{0.20}{m}$. $E_k$: \qty{800}{J} ervoor, $\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$ erna: $99.5\%$ verloren aan vervorming en
warmte. De inslag is niet-elastisch --- de energie blijft er niet behouden,
de impuls wel (de spankracht in het koord is verticaal en de inslag kort).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ en $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ geven
$v_1 + v_1' = v_2'$ (deel de energievergelijking door die van de impuls),
waaruit de formules volgen. Deel van de energie naar het doel $4m_1m_2/(m_1 + m_2)^2$:
proton $100\%$; koolstof $4 \times 12/169 = 28\%$; lood $4 \times 207/208^2 =
1.9\%$. Lichte kernen nemen de energie van het neutron in een paar botsingen over
(water, grafiet); lood zou er honderden nodig hebben.
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} Op de as, in het middelpunt van de schijf, uit symmetrie.

\textbf{2.} Koenig: $E_k = \tfrac12 Mv_G^2 + E_k^*$, en in het \omterm{def:b1:systems-of-points:barycentric}{barycentrische
stelsel} draait de jojo om zijn as: $E_k^* = \tfrac12 J\omega^2$.

\textbf{3.} Ring met straal $\rho$, massa $2M\rho\,\dd\rho/R^2$: $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$.

\textbf{4.} Het touwtje ligt vast en wikkelt zich van de as af: het punt van
de as dat het raakt is op dat ogenblik in rust, dus $v_G = r\omega$.

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$; $R^2/2r^2 = 9/(2 \times 0.16) = 28$: factor $29$ ---
bijna alle energie zit in de draaiing.

\textbf{6.} Gewicht $Mg$ omlaag, spankracht $T$ omhoog: $Ma_G = Mg - T$.

\textbf{7.} Alleen de spankracht heeft een moment om de as (\omterm{def:b1:angular-momentum:moment}{arm} $r$):
$J\dot\omega = Tr$.

\textbf{8.} $\dot\omega = a_G/r$: $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$; dan
$Ma_G = Mg - (R^2/2r^2)Ma_G$, $a_G = g/29 = \qty{0.34}{m/s^2}$.

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$, $97\%$ van
het gewicht: het touwtje houdt hem bijna vast.

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$; $v_G = a_Gt = \qty{0.82}{m/s}$.
Een steen: \qty{0.45}{s} en \qty{4.4}{m/s}.

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$.

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ tegen $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: $28$ keer meer energie in de draaiing dan in de val.

\textbf{13.} $Mgh = \qty{0.49}{J}$; $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$. De spankracht grijpt aan in het punt van
het touwtje dat in rust is (het touwtje beweegt niet): vermogen nul.

\textbf{14.} De stoppende kracht werkt langs het touwtje, door de as heen:
geen moment eromheen; het \omterm{def:b1:angular-momentum:L}{impulsmoment} $J\omega$ blijft onaangeroerd. Alleen een
tangentiële kracht (wrijving op de as) zou de draaiing kunnen afremmen.

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$, achttien keer het gewicht --- vandaar de ruk die de vinger
voelt.

\textbf{16.} $J\dot\omega = -\Gamma$: $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$.

\textbf{17.} Alle draaienergie $\tfrac12 J\omega^2 = \qty{0.47}{J}$ gaat over
in $Mgh'$: $h' = \qty{0.96}{m}$, bijna de volle hoogte; in de praktijk minder
door de wrijving op de as tijdens het slapen en de verliezen bij de
ruk.

\textbf{18.} $a_G$ wijst omlaag (de jojo vertraagt op de weg omhoog) terwijl
$v_G$ omhoog wijst: $T = M(g - a_G) < Mg$ --- kleiner dan het gewicht.

\textbf{19.} $r = R$: factor $1 + 1/2 = 1.5$, $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- een snelle val, weinig draaiing: een dunne as is wat een jojo tot jojo maakt.

\textbf{20.} \omterm{ex:b1:systems-of-points:rolling}{Rollen zonder slippen}: het materiële punt in $C$ heeft snelheid
nul; het snelheidsveld van een \omterm{def:b1:systems-of-points:solid}{star lichaam} met één punt in rust is
een rotatie om dat punt (as door $C$, dezelfde $\omega$).

\textbf{21.} $F$ grijpt aan onderaan de as, op hoogte $R - r$ boven
de tafel, horizontaal naar de hand toe: moment $F(R - r)$ om $C$ in
de zin die de klos naar de hand toe laat rollen; het gewicht en de
reactie gaan door $C$ (geen moment).

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$, $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: naar de hand toe.

\textbf{23.} De werklijn van $F$ gaat door $C$ wanneer de draad
vanaf de onderkant van de as de tafel in $C$ ontmoet: $\cos\beta = r/R$,
$\beta = 82^\circ$. Steiler: het moment om $C$ keert om, de klos rolt
weg (en bij voldoende grote $\beta$ komt hij omhoog).

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$; wrijving uit $Ma_G = F - f$ (de wrijving werkt $F$ tegen):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$; zij moet onder $\mu_sMg
\approx \qty{0.2}{N}$ blijven (voor $\mu_s = 0.4$), dus $F \lesssim \qty{0.5}{N}$.

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{Massamiddelpunt}: de translatie van $G$ en de spankracht;
\omterm{def:b1:angular-momentum:L}{impulsmoment} om een as: de draaiing en de rolrichting;
energie: de snelheid onderaan en de hoogte van de klim.
\end{solution}
