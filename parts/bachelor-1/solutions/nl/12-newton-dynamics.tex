\section*{Hoofdstuk \ref{ch:b1:newton-dynamics} --- De bewegingswetten van Newton}

\begin{solution}{exo:b1:newton-dynamics:1}
Aanwijzing van de weegschaal $N = m(g + a)$ met $a$ de opwaartse versnelling: omhoog,
$70 \times 11.81 = \qty{827}{N}$ (``\qty{84}{kg}''); omlaag, $70 \times 7.81 =
\qty{547}{N}$ (``\qty{56}{kg}''); constante snelheid, \qty{687}{N}
(\qty{70}{kg}); vrije val, $0$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:2}
Dezelfde versnelling $a$ voor beide (onrekbaar koord): $m_1a = T$,
$m_2a = m_2g - T$, dus $a = m_2g/(m_1 + m_2) = \qty{3.3}{m/s^2}$,
$T = m_1a = \qty{6.5}{N}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:3}
$a = g(\sin\alpha - \mu_d\cos\alpha) = 9.81(0.50 - 0.17) = \qty{3.2}{m/s^2}$;
$v = \sqrt{2 \times 3.2 \times 5.0} = \qty{5.7}{m/s}$. Nergens een \omterm{def:b1:newton-dynamics:momentum}{massa} te zien.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:4}
$k\Delta\ell = mg$: $\Delta\ell = 0.50 \times 9.81/200 = \qty{2.5}{cm}$.
Met $x$ de uitwijking uit het evenwicht is $m\ddot x = -kx$ (het gewicht
en de statische uitrekking vallen tegen elkaar weg): $\omega_0 = \sqrt{k/m} = \qty{20}{rad/s}$,
$T = \qty{0.31}{s}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:5}
$\mu_s = \tan 22^\circ = 0.40$. Bocht: $v \leq \sqrt{\mu_sgR}$: $\sqrt{0.8
\times 9.81 \times 50} = \qty{20}{m/s}$ (\qty{71}{km/h}); nat,
\qty{14}{m/s} (\qty{50}{km/h}).
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:6}
Parachutist: $v_\infty = \sqrt{2 \times 785/(1.2 \times 0.70)} = \qty{43}{m/s}$;
$\mathrm{Re} = \rho vL/\eta \approx 1.2 \times 43 \times 0.5/1.8\times10^{-5}
\approx 10^6$: kwadratisch. Druppeltje: $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-12}{kg}$,
$v_\infty = mg/6\pi\eta r = 4.1\times10^{-11}/3.4\times10^{-9} = \qty{1.2}{cm/s}$;
$\mathrm{Re} = 1.2 \times 0.012 \times 2\times10^{-5}/1.8\times10^{-5} = 0.02$:
lineair.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:7}
Onderin: $v^2 = 2g\ell(1 - \cos 60^\circ) = g\ell$; $T = mg + mv^2/\ell =
2mg$. Loslaatpunt: $v = 0$, $T = mg\cos\theta_0 = 0.5mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:8}
$v = \sqrt{gR\tan\beta} = \sqrt{9.81 \times 200 \times 0.268} = \qty{23}{m/s}$
(\qty{82}{km/h}). Langzamer: de auto neigt de helling af te glijden, de wrijving
moet omhoog langs de helling wijzen; sneller: de wrijving wijst omlaag om
\omterm{prop:b1:newton-dynamics:centripetal}{centripetale kracht} toe te voegen.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:9}
$T\cos\alpha = mg$, $T\sin\alpha = m\omega^2\ell\sin\alpha$: $\omega =
\sqrt{g/\ell\cos\alpha} = \sqrt{9.81/0.866} = \qty{3.4}{rad/s}$, periode
\qty{1.9}{s}; $T = mg/\cos\alpha = 1.15mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:10}
$a = 27.8/8.0 = \qty{3.5}{m/s^2}$, $F = \qty{3.5}{kN}$; uit $F \leq \mu_smg$
volgt $\mu_s \geq a/g = 0.35$. Op een natte weg begrenst $\mu_s \approx 0.4$ de
versnelling rond $0.4g$, wat de motor ook doet: grip, geen vermogen.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:11}
$v = v_\infty(1 - \eu^{-t/\tau})$, $v_\infty = g\tau = \qty{4.9}{m/s}$;
$z = \int v = v_\infty[t - \tau(1 - \eu^{-t/\tau})]$. Op \qty{2.0}{s}:
$4.9 \times (2.0 - 0.5 \times 0.98) = \qty{7.4}{m}$, tegen $\tfrac12 gt^2
= \qty{20}{m}$ in vacuüm.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:12}
Radiale projectie (naar het middelpunt): $N - mg\cos\theta = mv^2/R$, dus
$N = m(v^2/R + g\cos\theta)$. Bovenin ($\theta = \pi$): $N = m(v^2/R - g)
\geq 0$ precies als $v \geq \sqrt{gR}$. Met de snelheidswet is $v_{\text{boven}}^2 =
v_0^2 - 4gR \geq gR$: $v_0 \geq \sqrt{5gR}$; onderin is dan $N =
m(5g + g) = 6mg$.
\end{solution}

\begin{solution}{pb:b1:newton-dynamics:1}
\textbf{1.} $m\dot v = mg$: $v = gt$, $z = \tfrac12 gt^2$; op \qty{10}{s}:
\qty{98}{m/s}, \qty{490}{m}.

\textbf{2.} $mg = \qty{785}{N}$.

\textbf{3.} Op de grond duwt de vloer omhoog met $N = mg$ --- die duw
is wat men als gewicht voelt. In vrije val is er geen contactkracht: de
zwaartekracht werkt op elk deel gelijk en niets drukt.

\textbf{4.} Ja, tot op een fractie van een procent over een minuut; verwaarloosd worden
de draaiing van de aarde (middelpuntvliedende en coriolistermen,
\cref{ch:b1:non-inertial-frames}) en de wind.

\textbf{5.} $F_d = \tfrac12\rho C_xSv^2 = 0.48\,v^2$ (SI).

\textbf{6.} $m\,\dd v/\dd t = mg - 0.48v^2$.

\textbf{7.} $v_\infty = \sqrt{mg/0.48} = \sqrt{1635} = \qty{40}{m/s} =
\qty{146}{km/h}$.

\textbf{8.} Deel door $mg$: $\dd u/\dd t\,(v_\infty/g) = 1 - u^2$;
$\tau = 40.4/9.81 = \qty{4.1}{s}$.

\textbf{9.} $\int_0^u\dd u'/(1 - u'^2) = \operatorname{artanh}u = t/\tau$,
dus $u = \tanh(t/\tau)$.

\textbf{10.} $t = \tau\operatorname{artanh}(0.9) = 1.47\tau = \qty{6.1}{s}$;
voor $0.99$: $2.65\tau = \qty{11}{s}$.

\textbf{11.} $z = \int v_\infty\tanh(t/\tau)\dd t = v_\infty\tau\ln\cosh(t/\tau)$;
op \qty{10}{s}: $166 \times \ln\cosh(2.43) = 166 \times 1.74 = \qty{290}{m}$,
tegen \qty{490}{m}.

\textbf{12.} $\mathrm{Re} = 1.2 \times 40 \times 0.5/1.8\times10^{-5} \approx
\num{1.3e6}$: ver boven $1000$, dus kwadratisch.

\textbf{13.} $v_\infty' = \sqrt{785/(0.6 \times 25)} = \sqrt{52} = \qty{7.2}{m/s}$.

\textbf{14.} Weerstand $= 15 \times 1600 = \qty{24}{kN}$; $a = (24000 - 785)/80
= \qty{290}{m/s^2} \approx 30g$: het harnas zou breken, of de springer.

\textbf{15.} Gemiddeld $a \approx (40 - 7)/3 = \qty{11}{m/s^2} \approx 1.1g$;
\omterm{def:b1:newton-dynamics:momentum}{kracht} van het harnas $\approx m(g + a) \approx \qty{1.7}{kN}$.

\textbf{16.} $\tau' = 7.2/9.81 = \qty{0.73}{s}$: binnen enkele seconden na het
volledig openen is de daling stationair.

\textbf{17.} $\vect a = \vect 0$: \omterm{def:b1:newton-dynamics:momentum}{kracht} van het harnas $= mg = \qty{785}{N}$.

\textbf{18.} Uit het evenwicht $mg = \tfrac12\rho C_xSv^2$ volgt $v \propto (C_xS)^{-1/2}$:
halve oppervlakte, $\sqrt2$ keer de snelheid (\qty{10}{m/s}).

\textbf{19.} $v \propto \sqrt m$: $7.2 \times \sqrt{100/80} = \qty{8.1}{m/s}$.

\textbf{20.} $h = v^2/2g = 52/19.6 = \qty{2.6}{m}$: een sprong van de eerste
verdieping.

\textbf{21.} $a = v^2/2d$: \qty{52}{m/s^2} ($5.3g$) over \qty{0.5}{m},
\omterm{def:b1:newton-dynamics:momentum}{kracht} $m(g + a) \approx \qty{5}{kN}$; \qty{17}{m/s^2} ($1.8g$) over
\qty{1.5}{m}, \qty{2.2}{kN}. Rollen dus.

\textbf{22.} Een grotere weerstand (en enige lift) overtreft even het
gewicht: de springer vertraagt en komt onder $v_\infty'$ neer.

\textbf{23.} $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-6}{kg}$, $S = \pi r^2 =
\qty{3.1e-6}{m^2}$: $v_\infty = \sqrt{2mg/\rho C_xS} = \sqrt{8.2\times10^{-5}/
1.9\times10^{-6}} = \qty{6.6}{m/s}$; $\mathrm{Re} = 1.2 \times 6.6 \times
2\times10^{-3}/1.8\times10^{-5} \approx 900$: kwadratisch (ruwweg).

\textbf{24.} $v_\infty = \qty{1.2}{cm/s}$ (\cref{exo:b1:newton-dynamics:6});
$\mathrm{Re} \approx 0.02 \ll 1$: Stokes geldt, consistent.

\textbf{25.} $m\dot{\vect v} = m\vect g + \vect F_{\text{wr}}$; de weerstand is lineair
in $v$ voor kleine, trage dingen en kwadratisch voor grote, snelle. De
getallen: \qty{40}{m/s} voor een vallend lichaam, \qty{7}{m/s} onder een koepel
(of een regendruppel), \qty{1}{cm/s} voor mist --- dezelfde wet over vier
ordes van grootte.
\end{solution}
