\section*{Hoofdstuk \ref{ch:b1:potential-capacitors} --- Elektrische potentiaal; geleiders en condensatoren}

\begin{solution}{exo:b1:potential-capacitors:1}
$V = 8.99 \times 10^9 \times 10^{-6}/1 = \qty{9.0}{kV}$; $W = qV = \qty{9.0}{mJ}$;
losgelaten geeft $\tfrac12mv^2 = \qty{9}{mJ}$: $v = \sqrt{18} = \qty{4.2}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:2}
$E = 100/10^{-3} = \qty{1e5}{V/m}$; $E_k = eU = \qty{100}{eV} = \qty{1.6e-17}{J}$,
dus $v = \sqrt{2E_k/m_e} = \qty{5.9e6}{m/s}$; eenparig versneld is
$t = 2d/v = \qty{3.4e-10}{s}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:3}
$E_A = 10/0.002 = \qty{5}{kV/m}$, $E_B = 10/0.008 = \qty{1.25}{kV/m}$. Het
veld wijst naar afnemende potentiaal; een elektron ($q < 0$)
versnelt de andere kant op, naar stijgende $V$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:4}
$C = 4\pi\varepsilon_0R = 1.11 \times 10^{-10} \times 0.15 = \qty{17}{pF}$; $Q = CV =
\qty{3.3}{\micro C}$; $E = V/R = 2 \times 10^5/0.15 = \qty{1.3}{MV/m}$. Doorslag
bij $V = E_{\max}R = 3 \times 10^6 \times 0.15 = \qty{450}{kV}$ --- grote machines
gebruiken grote bollen.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:5}
Elk element van de ring ligt op afstand $\sqrt{R^2 + z^2}$: $V =
Q/4\pi\varepsilon_0\sqrt{R^2 + z^2}$; $E_z = -\dd V/\dd z = Qz/4\pi\varepsilon_0(R^2 +
z^2)^{3/2}$. Schijf: $\dd q = 2\pi\sigma a\,\dd a$ op $\sqrt{a^2 + z^2}$: $V =
\dfrac{\sigma}{2\varepsilon_0}\displaystyle\int_0^R\dfrac{a\,\dd a}{\sqrt{a^2 + z^2}} =
\dfrac{\sigma}{2\varepsilon_0}\bigl(\sqrt{z^2 + R^2} - z\bigr)$, en $-\dd V/\dd z =
\dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:6}
$V(r) = V(R) + \displaystyle\int_r^R\dfrac{Qr'\,\dd r'}{4\pi\varepsilon_0R^3} =
\dfrac{Q}{4\pi\varepsilon_0R} + \dfrac{Q(R^2 - r^2)}{8\pi\varepsilon_0R^3} = \dfrac{Q}{4\pi\varepsilon_0}
\dfrac{3R^2 - r^2}{2R^3}$. Goud: $V(0) = 1.5 \times 8.99 \times 10^9 \times 1.26 \times
10^{-17}/7 \times 10^{-15} = \qty{24}{MV}$: een proton heeft \qty{24}{MeV} nodig, een alfadeeltje
($2e$) \qty{48}{MeV} om het middelpunt te bereiken en \qty{32}{MeV} om het
oppervlak te raken. De alfadeeltjes van \qty{5}{MeV} van Rutherford keren om bij die $r$ waar $2eV =
\qty{5}{MeV}$: $V = \qty{2.5}{MV}$, $r = \qty{45}{fm}$ --- ver buiten de
kern, en daarom werkte zijn analyse met puntladingen.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:7}
As: $2p/4\pi\varepsilon_0r^3 = 2 \times 8.99 \times 10^9 \times 6.2 \times 10^{-30}/10^{-27} =
\qty{1.1e8}{V/m}$; loodrecht de helft: \qty{5.6e7}{V/m}. \omterm{def:b1:angular-momentum:moment}{Koppel} $pE =
\qty{6.2e-24}{N.m}$; energieverschil $2pE = \qty{1.2e-23}{J}$ tegen $k_BT =
\qty{4.1e-21}{J}$: $330$ keer kleiner --- slechts $0.1\%$ netto uitlijning.
Natriumion op \qty{0.3}{nm}: $E = e/4\pi\varepsilon_0r^2 = \qty{1.6e10}{V/m}$, $-pE
= \qty{-1.0e-19}{J} = \qty{-0.6}{eV}$, $24\,k_BT$: de watermoleculen klemmen
zich aan het ion vast --- hydratatie.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:8}
$Q = 4\pi\varepsilon_0R^2E = 10^{-4} \times 3 \times 10^6/8.99 \times 10^9 = \qty{33}{nC}$,
$V = ER = \qty{30}{kV}$. Verbonden bollen: $E_1 = V/R_1 = \qty{0.2}{MV/m}$,
$E_2 = V/R_2 = \qty{20}{MV/m} > \qty{3}{MV/m}$: de kleine bol ioniseert de
lucht en laat de lading weglekken --- het \omterm{ex:b1:potential-capacitors:point}{punteffect}.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:9}
$C = \varepsilon_r\varepsilon_0S/d = 2.2 \times 8.85 \times 10^{-12}/10^{-5} = \qty{2.0}{\micro F}$;
oprollen verandert de vorm, niet het oppervlak dat tegenover elkaar staat: dezelfde $C$ (met een tweede
film, zodat elke folie aan beide zijden tegenover de andere komt, verdubbelt zij).
$W = \tfrac12 \times 2 \times 10^{-6} \times 400^2 = \qty{0.16}{J}$. Defibrillator:
$Q = 150 \times 10^{-6} \times 2000 = \qty{0.30}{C}$, $W = \tfrac12CU^2 = \qty{300}{J}$,
$P = 300/0.005 = \qty{60}{kW}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:10}
(a) $E = Q/4\pi\varepsilon_0r^2$, $U = \dfrac{Q}{4\pi\varepsilon_0}\Bigl(\dfrac1{R_1} -
\dfrac1{R_2}\Bigr)$, $C = 4\pi\varepsilon_0R_1R_2/(R_2 - R_1)$; met $R_2 - R_1 = d \ll
R_1$ is $R_1R_2 \approx R^2$ en $C \approx 4\pi R^2\varepsilon_0/d = \varepsilon_0S/d$; met $R_2 \to
\infty$ gaat $C \to 4\pi\varepsilon_0R_1$. (b) $C/L = 2\pi\varepsilon_r\varepsilon_0/\ln(b/a) = 2\pi \times 2.3
\times 8.85 \times 10^{-12}/\ln5 = \qty{79}{pF/m}$, de juiste orde. (c) $E(a) =
U/[a\ln(b/a)] = 1000/(5 \times 10^{-4} \times 1.61) = \qty{1.2}{MV/m}$: onder de
\qty{20}{MV/m} van polyetheen, veilig; in lucht zou het krap zijn.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:11}
(a) $W = Q^2/2C = Q^2/8\pi\varepsilon_0R$. (b) $R = e^2/8\pi\varepsilon_0m_ec^2 = 2.31 \times
10^{-28}/(2 \times 8.19 \times 10^{-14}) = \qty{1.4e-15}{m}$ (de gebruikelijke ``klassieke
straal'' $e^2/4\pi\varepsilon_0m_ec^2 = \qty{2.8}{fm}$ laat de $\tfrac12$ weg). (c) $W =
Q^2/2C = Q^2d/2\varepsilon_0S$ verdubbelt: $\Delta W = Q^2d/2\varepsilon_0S = F\,d$ met $F =
Q^2/2\varepsilon_0S = \sigma^2S/2\varepsilon_0$, zoals rechtstreeks gevonden. (d) Bij vaste $U$ halveert $W =
\varepsilon_0SU^2/2d$: $\Delta W = -W/2$; wie trekt verricht nog steeds $W/2$
(integreer $F = \varepsilon_0SU^2/2x^2$ van $d$ tot $2d$); beide helften, samen $W$,
vloeien terug naar de batterij terwijl de lading $CU/2$ er bij
\omterm{def:b1:dc-circuits:voltage}{spanning} $U$ naar terugkeert.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:12}
(a) De bol met straal $r$ draagt $q = Qr^3/R^3$ bij oppervlaktepotentiaal
$q/4\pi\varepsilon_0r$; de schil $\dd q = 3Qr^2\,\dd r/R^3$ toevoegen kost $\dd W =
q\,\dd q/4\pi\varepsilon_0r = 3Q^2r^4\,\dd r/4\pi\varepsilon_0R^6$; integreren: $W =
3Q^2/20\pi\varepsilon_0R$. (b) $Q = 92e = \qty{1.47e-17}{C}$: $W = 3 \times 2.17 \times
10^{-34}/(20\pi \times 8.85 \times 10^{-12} \times 7.4 \times 10^{-15}) = \qty{1.6e-10}{J}
= \qty{990}{MeV}$. (c) Elk brokstuk: lading $Q/2$, straal $R/2^{1/3}$,
energie $\tfrac14 2^{1/3}W$; samen: $2^{1/3}W/2 = 0.63W$: vrijgekomen
$0.37 \times 990 = \qty{370}{MeV}$. (d) De ``oppervlaktespanning'' van de kern:
twee brokstukken hebben meer oppervlak dan één kern, wat ongeveer
\qty{170}{MeV} kost; het saldo, rond \qty{200}{MeV}, verschijnt vooral als de
\omterm{thm:b1:work-and-energy:ke}{kinetische energie} van de twee brokstukken die onder hun onderlinge
afstoting uit elkaar vliegen.
\end{solution}

\begin{solution}{pb:b1:potential-capacitors:1}
\textbf{1.} Gewicht $\tfrac43\pi r^3\rho g$ omlaag, weerstand $6\pi\eta rv$ omhoog,
\omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} $\tfrac43\pi r^3\rho_{\text{lucht}}g$ omhoog --- $1.2/900 = 0.13\%$ van het
gewicht, verwaarloosbaar. \omterm{prop:b1:newton-dynamics:terminal}{Eindsnelheid}: $\tfrac43\pi r^3\rho g = 6\pi\eta rv_1$, $v_1 =
2\rho gr^2/9\eta$.

\textbf{2.} $v_1 = \qty{1.0e-4}{m/s}$: $r = \sqrt{9\eta v_1/2\rho g} = \sqrt{9 \times
1.8 \times 10^{-5} \times 10^{-4}/(2 \times 900 \times 9.8)} = \qty{9.6e-7}{m}$, een
micrometer; $m = \tfrac43\pi r^3\rho = \qty{3.3e-15}{kg}$.

\textbf{3.} $\tau = m/6\pi\eta r = 3.3 \times 10^{-15}/3.3 \times 10^{-10} = \qty{1e-5}{s}$;
afstand $\sim v_1\tau = \qty{1}{nm}$. Nooit.

\textbf{4.} $\mathrm{Re} = 1.2 \times 10^{-4} \times 9.6 \times 10^{-7}/1.8 \times 10^{-5} =
6 \times 10^{-6} \ll 1$: Stokes geldt.

\textbf{5.} $eE = 1.6 \times 10^{-19} \times 3 \times 10^5 = \qty{4.8e-14}{N}$ tegen
$mg = \qty{3.2e-14}{N}$: alleen voor druppels van een micrometer vergelijkbaar. Een druppel van \qty{10}{\micro m}
weegt duizend keer meer; één \omterm{def:b1:electrostatics-gauss:charge}{elementaire lading} zou zijn
snelheid met $0.1\%$ veranderen, onzichtbaar.

\textbf{6.} Brownse beweging: het bombardement door luchtmoleculen schudt
de druppel willekeurig heen en weer; de tijden spreiden, wat de nauwkeurigheid begrenst en
een ondergrens aan de bruikbare druppelgrootte stelt.

\textbf{7.} $E = U/d = 5000/0.016 = \qty{3.1e5}{V/m}$; de platen zijn veel breder dan
$d$, zodat de superpositie van twee vlakken een homogeen veld geeft; de
equipotentialen zijn horizontale vlakken, $V = Uz/d$.

\textbf{8.} $C = \varepsilon_0\pi(0.1)^2/0.016 = \qty{17}{pF}$; $Q = CU = \qty{87}{nC}$;
$W = \tfrac12CU^2 = \qty{0.22}{mJ}$.

\textbf{9.} Bovenste plaat positief: $\vect E$ wijst omlaag; de kracht moet
omhoog wijzen, dus $q < 0$. $|q| = mgd/U_0 = 3.3 \times 10^{-15} \times 9.8 \times 0.016/
1080 = \qty{4.8e-19}{C} = 3e$: $q = -3e$.

\textbf{10.} Stijgend met constante snelheid: $|q|E = mg + 6\pi\eta rv_2$ en $mg =
6\pi\eta rv_1$, dus $|q|E = 6\pi\eta r(v_1 + v_2)$ en $|q| = 6\pi\eta r(v_1 +
v_2)d/U$. $v_2 = (|q|E - mg)/6\pi\eta r = (1.5 \times 10^{-13} - 3.2 \times 10^{-14})/
3.3 \times 10^{-10} = \qty{3.6e-4}{m/s}$: \qty{1}{mm} in \qty{2.8}{s}.

\textbf{11.} Uitbalanceren is een nuloordeel over een driftende, trillende stip
waarvan de lading intussen kan veranderen; tijden over een vaste afstand kan men
naar believen herhalen, $r$ ligt eens en voor altijd vast door $v_1$, en één druppel
levert dan een hele reeks ladingen.

\textbf{12.} $|q|U = 3 \times 1.6 \times 10^{-19} \times 5000 = \qty{2.4e-15}{J}$
(\qty{15}{keV}) over de spleet; weerstand over \qty{1}{cm}: $(|q|E - mg) \times 0.01 =
\qty{1.2e-15}{J}$, in de lucht omgezet in warmte.

\textbf{13.} Deel door $1.60$: $3.00$, $5.01$, $4.01$, $1.99$, $6.02$ ---
de gehele getallen $3, 5, 4, 2, 6$.

\textbf{14.} $q/n$: $1.600, 1.602, 1.605, 1.595, 1.605$; gemiddelde $1.601$,
standaardafwijking $0.004$, onzekerheid van het gemiddelde $0.004/\sqrt5 =
0.002$: $e = (1.601 \pm 0.002) \times 10^{-19}$~C.

\textbf{15.} De röntgenstraling ioniseert de lucht; de druppel pikt de ionen één
voor één op, elk met een geheel aantal $e$.

\textbf{16.} $r \propto \eta^{1/2}$ uit $v_1$, $m \propto r^3 \propto \eta^{3/2}$, en
$|q| = mgd/U_0 \propto \eta^{3/2}$: $1\%$ op $\eta$ geeft $1.5\%$ op $e$. De $\eta$ van
Millikan lag $0.4\%$ te laag, en zijn $e$ dus $0.6\%$.

\textbf{17.} Quarks komen nooit alleen: zij zitten opgesloten in hadronen,
en elk vrij voorwerp draagt een gehele lading.

\textbf{18.} $N_A = F/e = 96485/1.602 \times 10^{-19} = \qty{6.02e23}{mol^{-1}}$.

\textbf{19.} $m_e = 1.602 \times 10^{-19}/1.76 \times 10^{11} = \qty{9.1e-31}{kg}$: $1840$
keer lichter dan het waterstofatoom.

\textbf{20.} Horizontale vlakken tussen de platen, die bij de randen
uitbollen; verticale \omterm{def:b1:electrostatics-gauss:field}{veldlijnen} die aan de randen uitwaaieren over een breedte van de
orde $d = \qty{1.6}{cm}$ --- klein tegen de diameter van \qty{20}{cm}, en
de druppel wordt in het midden bekeken.

\textbf{21.} $E_p = qUz/d + mgz$, lineair in $z$: geen minimum; een stilhangende
druppel vraagt dat de helling verdwijnt, $q = -mgd/U$, en het evenwicht is
dan indifferent --- elke resterende drift houdt aan, en juist daarom is het
uitbalanceren zo moeilijk te beoordelen.

\textbf{22.} $2.4 \times 10^{-15}$ tegen $2.2 \times 10^{-4}$~J: $10^{-11}$. De druppel
verstoort het veld niet.

\textbf{23.} Veld omhoog, kracht op de negatieve druppel omlaag: hij valt met $(mg
+ |q|E)/6\pi\eta r = (3.2 \times 10^{-14} + 1.5 \times 10^{-13})/3.3 \times 10^{-10} =
\qty{5.6e-4}{m/s}$.

\textbf{24.} $F = 10^4 \times 1.6 \times 10^{-19} \times 3.1 \times 10^5 = \qty{5e-10}{N}$
tegen $\qty{9.8e-9}{N}$: $5\%$ van het gewicht --- niet op te tillen, en met
$10^4$ ladingen is een verandering van één $e$ onzichtbaar. Olienevel geeft druppels
die klein zijn, niet verdampen en bolvormig blijven.

\textbf{25.} Gemeten: $v_1$, $v_2$ en $U$ (met $d$, $\eta$, $\rho$ bekend);
afgeleid $r$, $m$, en daarna $q$, dat altijd een geheel veelvoud van
$e = (1.601 \pm 0.002) \times 10^{-19}$~C blijkt te zijn ($0.1\%$); daaruit $N_A = F/e$ en
$m_e = e/(e/m_e)$.
\end{solution}
