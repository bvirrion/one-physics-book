\section*{Hoofdstuk \ref{ch:b1:electrostatics-gauss} --- Elektrostatica: het veld en de wet van Gauss}

\begin{solution}{exo:b1:electrostatics-gauss:1}
$F_e = 8.99 \times 10^9 \times (1.6 \times 10^{-19})^2/(2 \times 10^{-15})^2 = \qty{58}{N}$;
$F_g = 6.67 \times 10^{-11} \times (1.67 \times 10^{-27})^2/(4 \times 10^{-30}) =
\qty{4.6e-35}{N}$; verhouding $1.2 \times 10^{36}$. Kernen houden samen omdat de sterke
wisselwerking, op femtometerschaal, zelfs deze afstoting overtreft.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:2}
$E = 8.99 \times 10^9 \times 10^{-6}/1 = \qty{9.0}{kV/m}$. Een neerwaarts veld is
dat van een negatieve lading: $Q = -4\pi\varepsilon_0R^2E = -1.11 \times 10^{-10} \times
(6.37 \times 10^6)^2 \times 100 = \qty{-4.5e5}{C}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:3}
Op de as, $|x| > a$: $E_x = \dfrac{q}{4\pi\varepsilon_0}\Bigl[\dfrac{1}{(x - a)^2} +
\dfrac{1}{(x + a)^2}\Bigr]$, naar buiten. Op de $y$-as vallen de $x$-componenten
weg: $E_y = \dfrac{2qy}{4\pi\varepsilon_0(y^2 + a^2)^{3/2}}$; $\dd E_y/\dd y \propto
(y^2 + a^2) - 3y^2 = 0$ in $y = a/\sqrt2$. In de oorsprong verdwijnt het
veld; lijnen die elke lading naar de andere toe verlaten buigen weg langs
$\pm y$ --- een zadelpunt.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:4}
Tussen de ladingen wijzen beide velden naar $-q$: geen nulpunt. Links van
$+2q$ geldt altijd $2q/x^2 > q/(x + a)^2$. Voorbij $-q$, op afstand $d$ ervan:
$2q/(a + d)^2 = q/d^2$, $a + d = \sqrt2\,d$, $d = a/(\sqrt2 - 1) = 2.4a$.
Er vertrekken twee lijnen bij $+2q$ voor elke lijn die op $-q$ eindigt; de andere helft gaat naar
het oneindige, waar het paar op een lading $+q$ lijkt.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:5}
Ring met straal $a$, dikte $\dd a$: $\dd q = 2\pi\sigma a\,\dd a$, $\dd E_z =
\dfrac{\sigma z}{2\varepsilon_0}\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}}$; integreer:
$\dfrac{\sigma z}{2\varepsilon_0}\Bigl[-\dfrac{1}{\sqrt{a^2 + z^2}}\Bigr]_0^R = \dfrac{\sigma}{2\varepsilon_0}
\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$. $z \ll R$: $\sigma/2\varepsilon_0$, het
oneindige vlak. $z \gg R$: $z/\sqrt{z^2 + R^2} \approx 1 - R^2/2z^2$, dus $E_z
\approx \sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, de puntlading.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:6}
$Q = 79e = \qty{1.26e-17}{C}$; $E(R) = 8.99 \times 10^9 \times 1.26 \times 10^{-17}/
(7 \times 10^{-15})^2 = \qty{2.3e21}{V/m}$; in $R/2$ de helft: \qty{1.2e21}{V/m}; op
\qty{1}{pm}: $1.14 \times 10^{-7}/10^{-24} = \qty{1.1e17}{V/m}$ --- $10^{15}$ keer
het doorslagveld van lucht, op een afstand waar de elektronen leven.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:7}
Cilinder met straal $r$ en lengte $h$: $2\pi rh\,E = \lambda h/\varepsilon_0$, $E =
\lambda/2\pi\varepsilon_0r$. Corona bij $\lambda = 2\pi\varepsilon_0 rE = 2\pi \times 8.85 \times
10^{-12} \times 0.01 \times 3 \times 10^6 = \qty{1.7}{\micro C/m}$; op \qty{5}{m} is
$E = 3 \times 10^6 \times 0.01/5 = \qty{6}{kV/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:8}
$E = \sigma/\varepsilon_0 = 10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$. Een plaat
oefent netto geen kracht op zichzelf uit (de bijdragen van haar eigen veld vallen
paarsgewijs weg); het veld van de andere plaat is op haar plaats $\sigma/2\varepsilon_0$, dus
$F = Q\sigma/2\varepsilon_0 = 10^{-6} \times 10^{-6}/(2 \times 8.85 \times 10^{-12}) =
\qty{5.6e-2}{N}$, aantrekkend; druk $\sigma^2/2\varepsilon_0 = \qty{0.056}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:9}
$r < a$: de lading van de binnengeleider zit op zijn oppervlak, geen lading
erbinnen, $E = 0$. $a < r < b$: $E = \lambda/2\pi\varepsilon_0r$. $r > b$: omsloten
lading $\lambda - \lambda = 0$, $E = 0$ --- de kabel straalt geen statisch
veld uit. In $r = a$: $2 \times 8.99 \times 10^9 \times 10^{-8}/(5 \times 10^{-4}) =
\qty{3.6e5}{V/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:10}
$\rho = 3Q/4\pi(R_2^3 - R_1^3)$. $r < R_1$: $E = 0$. $R_1 < r < R_2$: $4\pi r^2E =
\rho\,\tfrac43\pi(r^3 - R_1^3)/\varepsilon_0$, dus $E = \dfrac{Q}{4\pi\varepsilon_0r^2}\,
\dfrac{r^3 - R_1^3}{R_2^3 - R_1^3}$. $r > R_2$: $Q/4\pi\varepsilon_0r^2$. In $R_1$ geven beide
kanten $0$; in $R_2$ geven beide $Q/4\pi\varepsilon_0R_2^2$. In de schil neemt $E
\propto r - R_1^3/r^2$ toe, dus ligt het maximum in $R_2$, waarna
het als $1/r^2$ daalt. In de holte is het veld nul: een lading
daar voelt geen kracht --- waar zij ook zit.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:11}
(a) Kern $\tfrac43\pi(3.48 \times 10^6)^3 \times 11000 = \qty{1.94e24}{kg}$; mantel
$\tfrac43\pi[(6.37 \times 10^6)^3 - (3.48 \times 10^6)^3] \times 4500 = \qty{4.08e24}{kg}$;
totaal \qty{6.0e24}{kg}, tot op $1\%$. (b) Oppervlak $GM/R^2 = \qty{9.9}{m/s^2}$;
grens $6.67 \times 10^{-11} \times 1.94 \times 10^{24}/(3.48 \times 10^6)^2 =
\qty{10.7}{m/s^2}$. (c) $g = GM(r)/r^2$ met $\dd M = 4\pi r^2\rho_m\,\dd r$:
$\dd g/\dd r = 4\pi G\rho_m - 2GM(r)/r^3 = 4\pi G[\rho_m - \tfrac23\bar\rho(r)]$,
waarin $\bar\rho(r) = M(r)/\tfrac43\pi r^3$ de gemiddelde dichtheid onder $r$ is.
$g$ stijgt met de diepte ($\dd g/\dd r < 0$) dan en slechts dan als $\rho_m < \tfrac23\bar\rho$. Net
onder het oppervlak is $\bar\rho = 5500$, $\tfrac23\bar\rho = 3700 < 4500$: $g$
zakt eerst een beetje; op de kerngrens is $\bar\rho = 11000$, $7300 >
4500$: daar stijgt het, tot $10.7$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:12}
(a) Binnenin is $E = er/4\pi\varepsilon_0R^3$ naar buiten gericht; op het elektron werkt $\vect F =
-\dfrac{e^2}{4\pi\varepsilon_0R^3}\,\vect r$: een veer. (b) $\omega = \sqrt{e^2/4\pi\varepsilon_0mR^3}
= \sqrt{2.31 \times 10^{-28}/(9.11 \times 10^{-31} \times 10^{-30})} = \qty{1.6e16}{rad/s}$,
$f = \qty{2.5e15}{Hz}$, $\lambda = c/f = \qty{120}{nm}$. (c) Verbluffend dicht
bij de lijn van \qty{122}{nm}: de grootte en de sterkte van het atoom kloppen.
Maar het model geeft één frequentie en geen reeks, de kern is
geen bal van \qty{0.1}{nm} maar een punt van $10^{-15}$~m --- en een klassiek
cirkelend elektron zou zijn energie in nanoseconden wegstralen. De
oplossing is \cref{ch:b1:quantum-introduction}.
\end{solution}

\begin{solution}{pb:b1:electrostatics-gauss:1}
\textbf{1.} $\vect F = -e\vect E$: naar boven. Een neerwaarts veld aan het oppervlak
is het veld van een negatieve lading: de aarde is negatief.

\textbf{2.} Bol met straal $R_E$, $E_n = -100$: $Q = 4\pi\varepsilon_0R_E^2E_n =
1.11 \times 10^{-10} \times 4.06 \times 10^{13} \times (-100) = \qty{-4.5e5}{C}$.

\textbf{3.} $\sigma = \varepsilon_0E_n = \qty{-8.9e-10}{C/m^2}$; $8.9 \times 10^{-10}/1.6 \times
10^{-19} = 5.5 \times 10^9$ elektronen per vierkante meter, $5.5 \times 10^5$ per
vierkante centimeter.

\textbf{4.} $F = 100e \times 100 = \qty{1.6e-15}{N}$; gewicht $10^{-9} \times 9.8 =
\qty{9.8e-9}{N}$: zes miljoen keer groter.

\textbf{5.} Buitennormalen: boven $E_n = -5$, onder $E_n = +100$, zijkanten
nul: $\Phi = \qty{95}{V.m}$, dus $Q = 95\varepsilon_0 = \qty{+8.4e-10}{C}$ per vierkante
meter --- positieve ruimtelading, die de $-8.9 \times 10^{-10}$ van de grond
bijna opheft.

\textbf{6.} $\rho = 8.4 \times 10^{-10}/10^4 = \qty{8.4e-14}{C/m^3}$: $5 \times 10^5$
\omterm{def:b1:electrostatics-gauss:charge}{elementaire ladingen} per kubieke meter; tegen $10^9$ ionen van elk teken
een onbalans van $5 \times 10^{-4}$.

\textbf{7.} $J = \gamma E_0 = \qty{2e-12}{A/m^2}$; $I = J \times 5.1 \times 10^{14} =
\qty{1.0}{kA}$; $Q/I = 4.5 \times 10^5/1000 = \qty{450}{s}$. De aarde zou in
acht minuten neutraal zijn: iets laadt haar voortdurend bij.

\textbf{8.} De as ligt in elk vlak dat haar bevat, allemaal symmetrievlakken:
$\vect E$ is axiaal. Elke $\dd q$ geeft $\dd q/4\pi\varepsilon_0(a^2 + z^2)$
langs zijn eigen richting, waarvan het axiale deel de fractie
$z/\sqrt{a^2 + z^2}$ is: $E_z = qz/4\pi\varepsilon_0(a^2 + z^2)^{3/2}$.

\textbf{9.} $\dd q = 2\pi\sigma a\,\dd a$: $E_z = \dfrac{\sigma z}{2\varepsilon_0}\displaystyle\int_0^R
\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}} = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2
+ R^2}}\Bigr)$.

\textbf{10.} $z \ll R$: $\sigma/2\varepsilon_0$, het oneindige vlak (de waarde van Gauss);
$z \gg R$: $\sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, de puntlading.

\textbf{11.} $\sigma = -40/(\pi \times 9 \times 10^6) = \qty{-1.4e-6}{C/m^2}$; $\sigma/2\varepsilon_0 =
\qty{8.0e4}{V/m}$; op $z = \qty{3}{km}$ is $1 - 3/\sqrt{18} = 0.29$: $E =
\qty{2.3e4}{V/m}$, \emph{omhoog} gericht naar de negatieve lading.

\textbf{12.} Bovenste schijf, $z = \qty{9}{km}$: $8.0 \times 10^4 \times (1 - 9/\sqrt{90}) =
\qty{4.1e3}{V/m}$ naar beneden. Netto $\qty{1.9e4}{V/m}$ omhoog, ongeveer $4 \times 10^4$
met de geïnduceerde ladingen van de grond: de juiste orde van de \qty{10}{kV/m}
die wordt gemeten (de echte lading is meer verspreid en deels afgeschermd);
omgekeerd en honderd keer het mooiweerveld.

\textbf{13.} Hele wolk: netto lading nul, \omterm{def:b1:electrostatics-gauss:flux}{flux} nul. Alleen de onderste schijf:
$-Q/\varepsilon_0 = -40/8.85 \times 10^{-12} = \qty{-4.5e12}{V.m}$.

\textbf{14.} Elke schijf ligt op \qty{3}{km} en geeft $2.3 \times 10^4$; de
bovenste ($+$) duwt omlaag, de onderste ($-$) trekt omlaag: \qty{4.7e4}{V/m}
naar beneden --- dertig keer onder de doorslagwaarde.

\textbf{15.} $Q/4\pi\varepsilon_0R^2 = 8.99 \times 10^9 \times 40/(2.5 \times 10^5) =
\qty{1.4e6}{V/m}$: alleen een lading die in een halve kilometer wordt gepropt haalt
de doorslagwaarde, en de velden die men in wolken meet zijn tien keer te zwak. Plaatselijke
versterking bij druppels en ijs, of op hol geslagen elektronen die door kosmische
straling worden aangestoken, zijn de kandidaten --- hoe een flits begint staat nog ter discussie.

\textbf{16.} $I = 5/(5 \times 10^{-5}) = \qty{1e5}{A}$.

\textbf{17.} $W = QEd = 5 \times 10^5 \times 3000 = \qty{1.5e9}{J}$ (\qty{420}{kW h});
$P = 1.5 \times 10^9/5 \times 10^{-5} = \qty{3e13}{W}$ --- anderhalf keer
het vermogen van de hele mensheid, vijftig microseconden lang.

\textbf{18.} Volume $\pi(0.02)^2 \times 3000 = \qty{3.8}{m^3}$, massa \qty{4.5}{kg};
$\Delta T = 1.5 \times 10^9/(4.5 \times 720) = \qty{4.6e5}{K}$ als er niets ontsnapte.
Het kanaal explodeert naar buiten (de schok is de donder), straalt licht
en radiogolven uit, ioniseert en splitst de lucht: het uiteindelijke kanaal
tot $3 \times 10^4$~K verwarmen is maar een klein deel van de rekening.

\textbf{19.} \qty{2.9}{s} per kilometer. Het geluid vertrekt tegelijk uit elk punt van een
kanaal van \qty{3}{km}, vanaf afstanden die kilometers verschillen: het
komt over enkele seconden verspreid aan --- het gerommel.

\textbf{20.} $50 \times 5 = \qty{250}{A}$ naar de grond gevoerd; met de
rustiger onweersstromen (puntontlading onder wolken, geladen regen) haalt het
totaal de kiloampère van deel~I: onweersbuien zijn de generatoren van een
mondiale kring waarvan de mooiweerlucht de retourweg is.

\textbf{21.} $\Phi_{\text{uit}}(\vect g) = -4\pi GM_{\text{binnen}}$: voor een \omterm{def:b1:point-kinematics:frame}{puntmassa}
is $g = -GM/r^2$ op een bol met oppervlakte $4\pi r^2$, \omterm{def:b1:electrostatics-gauss:flux}{flux} $-4\pi GM$, onafhankelijk
van $r$; de rest volgt als bij Coulomb.

\textbf{22.} $g(r) = g_0r/R_E$; in $R_E/2$: \qty{4.9}{m/s^2}.

\textbf{23.} $M_{\text{kern}} = \qty{1.94e24}{kg}$, $g = GM_{\text{kern}}/R_c^2 =
\qty{10.7}{m/s^2}$ --- groter dan aan het oppervlak: de mantel erboven is
lichter dan het gemiddelde van wat eronder ligt.

\textbf{24.} $m\ddot x = -mg_0x/R_E$: $\omega = \sqrt{g_0/R_E} = \qty{1.24e-3}{rad/s}$,
$T = 2\pi/\omega = \qty{5060}{s} = \qty{84}{min}$, $v_{\max} = \omega R_E =
\qty{7.9}{km/s}$ --- de baansnelheid van een scherend cirkelende satelliet, waarvan de
periode dezelfde is.

\textbf{25.} Zij verving de som over elke lading van de aarde (I),
over de ionen van de lucht (I), en over elke schil gesteente (IV) --- en
in~II legde zij de grenzen van de som over schijven vast. Onthoud: \qty{-5e5}{C} en
\qty{1}{kA} voor de mooiweeraarde; \qty{40}{C} en \qty{e4}{V/m} onder een
bui; \qty{e5}{A} en \qty{e9}{J} per ontlading; \qty{10.7}{m/s^2} op de
kerngrens.
\end{solution}
