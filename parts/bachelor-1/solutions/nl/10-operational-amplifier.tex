\section*{Hoofdstuk \ref{ch:b1:operational-amplifier} --- De operationele versterker}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$; $V_s = \qty{5.0}{V}$; verzadiging bij $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$; ingangsimpedantie $R_1 = \qty{10}{k\ohm}$; $i = 1.0/10^4 =
\qty{0.10}{mA}$; $V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
Zonder: deler $1/(1 + 10) = 0.091$, \qty{91}{mV}. Met de volger:
\qty{1.0}{V} --- de volger trekt niets uit de sensor en levert
de \qty{1}{mA} die de belasting nodig heeft.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$; in $V_-$: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
dus $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$; met $V_- = V_+$:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. Getallen: $10 \times 0.05 = \qty{0.50}{V}$;
de gemeenschappelijke \qty{1}{V} draagt niets bij --- common-modeonderdrukking.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: helling $-1000 \times 1.0 = \qty{-1000}{V/s}$;
verzadiging na $15/1000 = \qty{15}{ms}$. Blokgolf: driehoek met
\omterm{def:b1:wave-propagation:signal}{amplitude} $ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(plus wat de beginlading als constante achterliet).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: gelijkspanningsversterking $-100$; \omterm{def:b1:filters-transfer-functions:bode}{afsnijfrequentie}
$1/(2\pi R'C) = \qty{1.6}{Hz}$; bij \qty{1}{kHz} is $R'C\omega = 628$ en $G =
100/628 = 0.16$ --- de waarde $1/RC\omega$ van de integrator. Zij integreert voor
$f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: $R_2 = \qty{10}{k\ohm}$; $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
Met eindige $\mu$: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, dus
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. Bandbreedte $f_T/G =
\qty{10}{kHz}$; volger: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
Elke ruisuitschieter over \qty{2.0}{V} klapt de uitgang om: een reeks
snelle omschakelingen (``ratelen'') rond de passage. Oplossing: een hysterese
breder dan de ruis, dus een \omterm{prop:b1:operational-amplifier:schmitt}{schmitttrigger}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: drempels $\pm\qty{1.5}{V}$. De
uitgang is $+\qty{15}{V}$ terwijl $V_e$ stijgt tot \qty{1.5}{V}, klapt naar
$-\qty{15}{V}$, blijft daar tot $V_e$ onder \qty{-1.5}{V} zakt, en klapt
terug: een blokgolf met de frequentie van de driehoek, met de flanken op de
drempelpassages.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
Met $V_s = +V_{\mathrm{sat}}$ laadt $V_C$ zich van $-\beta V_{\mathrm{sat}}$
naar $+V_{\mathrm{sat}}$: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, en bereikt $+\beta V_{\mathrm{sat}}$ wanneer $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$: halve periode $RC\ln\frac{1 + \beta}{1 - \beta}$,
en dus $T$. Voor $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$. $V_s$: blokgolf
$\pm\qty{15}{V}$; $V_C$: exponentiële bogen tussen $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} Twee delers: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone}); de bedrading
legt het teken vast.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ altijd; $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ op volle schaal. De
\qty{2.5}{V} die beide delen is de common-modespanning.

\textbf{4.} Elk middelpunt: $1 \parallel 1 = \qty{500}{\ohm}$; tussen de
twee middelpunten \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$: een verlies van $9\%$.

\textbf{6.} Oneindige ingangsimpedantie (geen stroom uit de brug, zodat
de middelpunten hun nullastpotentialen behouden) en nul
uitgangsimpedantie (de volgende trap kan trekken wat zij nodig heeft).

\textbf{7.} Precies $V_A$ en $V_B$; stroom nul.

\textbf{8.} Zoals in \cref{prop:b1:operational-amplifier:sumdiff}: $V_+ =
V_BR_2/(R_1 + R_2)$, knooppuntregel in $V_-$, $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$: bv.\ \qty{1}{k\ohm} en \qty{9}{k\ohm}.
Totale versterking $1000$; volle schaal $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} Aangepast: de gemeenschappelijke \qty{2.5}{V} valt precies weg (alleen
$V_B - V_A$ blijft over). Een afwijking van $1\%$ laat ongeveer $1\%$ ervan door: een
fout van de orde \qty{25}{mV} aan de ingang, tien keer het \omterm{def:b1:wave-propagation:signal}{signaal} op volle
schaal --- de vier weerstanden moeten beter dan $0.01\%$ overeenstemmen
(of de common-modespanning moet eerst worden weggenomen).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, veertig procent van de volle
schaal: onbruikbaar zonder offsetregeling, een versterker met lage offset
(chopper), of een nulijking met de ligger onbelast.

\textbf{13.} Trap 1: $f_T/100 = \qty{10}{kHz}$; trap 2: \qty{100}{kHz};
ver boven \qty{5}{Hz}.

\textbf{14.} Vereist $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
tweeduizend keer onder de \omterm{rem:b1:operational-amplifier:real}{slew rate}: nee.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$: $G = 1/\sqrt{26}$, \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$; vertraging $\varphi/\omega =
\qty{15}{ms}$ --- niets tegenover de passage van een vrachtwagen.

\textbf{18.} Geen belasting van de volgende trap (uitgangsimpedantie nul)
en versterking beschikbaar; bovendien geen verzwakking van het gewenste \omterm{def:b1:wave-propagation:signal}{signaal} onder
de \omterm{def:b1:filters-transfer-functions:bode}{afsnijfrequentie}.

\textbf{19.} \qty{-28}{dB} (versterkingen vermenigvuldigen, \omterm{def:b1:filters-transfer-functions:bode}{decibels} tellen
op), exact omdat de eerste trap een uitgangsimpedantie nul heeft: de
tweede trap belast haar niet.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$; uitgang $+V_{\mathrm{sat}}$ wanneer het \omterm{def:b1:wave-propagation:signal}{signaal} boven
\qty{2.0}{V} komt, $-V_{\mathrm{sat}}$ eronder (of omgekeerd, door de bedrading).

\textbf{21.} Hij ratelt: de lamp flikkert terwijl de ruis de referentie
passeert.

\textbf{22.} Met $V_{\mathrm{ref}}$ via $R_2$ en de uitgang via
$R_1$ staat de niet-inverterende ingang in de standaardvorm op
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$; met de referentie dienovereenkomstig
verschoven zijn de omschakelniveaus $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
Uit $\beta V_{\mathrm{sat}} = \qty{0.10}{V}$: $\beta = 1/150$, bv.\ $R_1 =
\qty{1.0}{k\ohm}$, $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (aanslaan) en \qty{1.9}{V} (loslaten): rek
\num{1.68e-3} en \num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. Bij
$-V_{\mathrm{sat}}$ staat de led \qty{15}{V} in sperrichting, boven haar
specificatie: een gewone diode antiparallel (of in serie) beschermt haar.

\textbf{25.} Brug (\qty{2.5}{mV} volle schaal) $\to$ twee volgers (geen
belasting) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{verschilversterker} $\times100$ $\to$ niet-inverterend
$\times10$ $\to$ actieve laagdoorlaat \qty{10}{Hz} $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{schmitttrigger} op
$\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ led. De werkelijke grenzen: de
ingangsoffsetspanning en het overeenstemmen van de weerstanden van de
\omterm{prop:b1:operational-amplifier:sumdiff}{verschilversterker}, beide vergelijkbaar met het \omterm{def:b1:wave-propagation:signal}{signaal} van één millivolt.
\end{solution}
