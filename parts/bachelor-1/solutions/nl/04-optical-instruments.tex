\section*{Hoofdstuk \ref{ch:b1:optical-instruments} --- Optische instrumenten en het oog}

\begin{solution}{exo:b1:optical-instruments:1}
Oneindig: $f' = \qty{17}{mm}$, $V = 1/0.017 = \qty{58.8}{\delta}$.
\qty{25}{cm}: $V = 58.8 + 1/0.25 = \qty{62.8}{\delta}$, $f' = \qty{15.9}{mm}$.
Breedte \qty{4.0}{\delta}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:2}
Bijziende: oneindig moet in het \omterm{def:b1:optical-instruments:punctum}{verpunt} worden afgebeeld, $f' = \qty{-40}{cm}$,
$V = \qty{-2.5}{\delta}$. Verziende: een voorwerp op \qty{25}{cm} moet
(virtueel) op \qty{1.0}{m} worden afgebeeld: $V = 1/\overline{OA'} - 1/\overline{OA}
= -1 + 4 = \qty{+3.0}{\delta}$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:3}
$G = 25/4.0 = 6.3$; postzegel in het brandvlak, \qty{4.0}{cm} van de
\omterm{def:b1:mirrors-thin-lenses:lens}{lens}. Detail van \qty{2}{mm}: $\theta' = 2/40 = \qty{0.05}{rad}$, tegenover
$2/250 = \qty{0.008}{rad}$ op \qty{25}{cm}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:4}
$G = 900/25 = 36$ en $900/9 = 100$; buislengtes $f_1' + f_2' =
\qty{925}{mm}$ en \qty{909}{mm}; maan: $36 \times 0.52^\circ = 19^\circ$
en $52^\circ$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:5}
$\gamma_1 = -160/4 = -40$; $G_2 = 250/20 = 12.5$; $G = 500$.
$\overline{F_1A} = -f_1'^2/\Delta = -16/160 = \qty{-0.10}{mm}$: voorwerp
\qty{4.1}{mm} van het \omterm{def:b1:optical-instruments:microscope}{objectief}. Het voorwerp moet op die afstand van het
\omterm{def:b1:optical-instruments:microscope}{objectief} blijven, dus verschuift de hele buis over \qty{10}{\micro m} ---
de fijnstelknop.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:6}
Blote \omterm{def:b1:optical-instruments:eye}{oog}: $1.5/3\times10^{-4} = \qty{5}{km}$. Kijker: $\theta_{\min}
= 1.22 \times 5.5\times10^{-7}/0.100 = \qty{6.7e-6}{rad}$; koplampen
gescheiden tot $1.5/6.7\times10^{-6} \approx \qty{220}{km}$ (in
beginsel --- lucht en aardkromming komen eerder tussenbeide). $G_r = 3\times10^{-4}/
6.7\times10^{-6} = 45$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Belichting $\propto N^2$: $(8/2)^2 = 16$ keer
langer, $16/500 \approx 1/30$~\unit{s}. Het levert \omterm{def:b1:optical-instruments:camera}{scherptediepte} op (en
minder aberraties) ten koste van bewegingsonscherpte of een statief.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$; scherp van $H/2 =
\qty{1.9}{m}$ tot oneindig. Vooraf op $H$ ingesteld hoeft de \omterm{def:b1:optical-instruments:camera}{camera} niet
scherp te stellen: de foto kan worden genomen op het moment zelf.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$: \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$, al het licht
komt het \omterm{def:b1:optical-instruments:eye}{oog} binnen. $7\times50$: \qty{7.1}{mm}, passend bij de nachtpupil,
en $(50/30)^2 = 2.8$ keer meer licht opgevangen: de schemerkijker.
$20\times30$: \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} \qty{1.5}{mm}, flets beeld, smal veld en
handtrilling twintig keer vergroot.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
Contactlens: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$. Bril:
het \omterm{def:b1:optical-instruments:punctum}{verpunt} ligt $25 - 1.5 = \qty{23.5}{cm}$ van de \omterm{def:b1:mirrors-thin-lenses:lens}{lens}, dus $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- sterker. De \omterm{def:b1:mirrors-thin-lenses:lens}{lens} moet het
beeld in het \omterm{def:b1:optical-instruments:punctum}{verpunt} van het \omterm{def:b1:optical-instruments:eye}{oog} vormen, en de afstand daarvan tot de \omterm{def:b1:mirrors-thin-lenses:lens}{lens}
hangt af van waar de \omterm{def:b1:mirrors-thin-lenses:lens}{lens} zit.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afocaal: $F_1' = F_2$, afstand $f_1' + f_2' = \qty{20}{cm}$ (korter
dan de \qty{40}{cm} van een keplerkijker met dezelfde $G$). $G = -30/(-10) =
+3$: rechtop. \omterm{prop:b1:optical-instruments:telescope}{Uittredepupil} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$: virtueel, binnen in de buis, waar het
\omterm{def:b1:optical-instruments:eye}{oog} niet kan zijn; het \omterm{def:b1:optical-instruments:eye}{oog} achter het \omterm{def:b1:optical-instruments:microscope}{oculair} vangt alleen de bundels op die
het toevallig bereiken --- een smal veld, de klassieke klacht over de
theaterkijker.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$; $\Delta m = 2.5\log_{10}820 = 7.3$: grensmagnitude
$13.3$. De flux gaat als $\propto 1/r^2$, dus is de lamp
$\sqrt{820} = 29$ keer verder te zien.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$: afstand $f_1' + f_2' = \qty{1225}{mm}$
(\omterm{def:b1:optical-instruments:microscope}{oculair} van \qty{25}{mm}) of \qty{1206}{mm} (\qty{6}{mm}).

\textbf{2.} $L_1$: $\overline{O_1A_1} = f_1'$ (beeld in $F_1'$). $L_2$:
$\overline{O_2A_1} = -f_2'$, dus $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$: beeld op oneindig.

\textbf{3.} De bundel komt samen in het gemeenschappelijke brandvlak op hoogte $h =
f_1'\alpha$ (straal door $O_1$); van daaruit bepaalt de straal door $O_2$
de uittredende richting, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ en $-200$; maan: $25^\circ$ en $104^\circ$.

\textbf{5.} Boven en onder betekenen niets aan de hemel; een spotting scope
voegt een oprichtend prismapaar (of lenzenrelais) toe.

\textbf{6.} In het brandvlak van $L_1$, diameter $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} Voorwerp $O_1$ op $\overline{O_2O_1} = -(f_1' + f_2')$:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$;
vergroting $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
diameter $D/\abs G$.

\textbf{8.} \qty{25}{mm}: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, diameter $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}:
\qty{6.0}{mm} erachter, diameter \qty{0.75}{mm}.

\textbf{9.} Elke straal die het \omterm{def:b1:optical-instruments:microscope}{objectief} is binnengekomen gaat door de
\omterm{prop:b1:optical-instruments:telescope}{uittredepupil}: een \omterm{def:b1:optical-instruments:eye}{oog} dat daar staat ontvangt het hele veld. Verder naar
achteren vangt het alleen de middelste bundels op: het veld krimpt tot een sleutelgat.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$;
blote \omterm{def:b1:optical-instruments:eye}{oog} $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$; het \omterm{def:b1:optical-instruments:microscope}{oculair} van \qty{25}{mm}
($48$) ligt er het dichtst bij, iets eronder; dat van \qty{6}{mm} ($200$) zit er
drie keer boven.

\textbf{14.} Bij $G = 200$ beslaat de buigingsvlek $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$: zichtbaar wazig. Het licht van
de maan wordt over een $(200/48)^2 = 17$ keer groter netvliesoppervlak
verdeeld: fletser.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Daarboven begrenst de luchtonrust het \omterm{rem:b1:optical-instruments:abbe}{oplossend vermogen} vanaf
de grond (adaptieve optiek daargelaten); een grotere $D$ vangt nog altijd meer
licht op en bereikt zwakkere objecten.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$: magnitude $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
$460$ keer dat van het blote \omterm{def:b1:optical-instruments:eye}{oog}. Het netvliesbeeld is $\abs G^2$ keer
groter in oppervlak: verhouding van oppervlaktehelderheid $460/G^2 = 0.20$ ($G = 48$),
$0.012$ ($G = 200$) --- gelijk aan $(D/\abs G)^2/d_p^2$, het kwadraat van de
verhouding \omterm{prop:b1:optical-instruments:telescope}{uittredepupil} tot oogpupil.

\textbf{19.} Voor een uitgebreid voorwerp groeit het opgevangen licht als $D^2$
maar het beeldoppervlak als $G^2 \geq (D/d_p)^2$: de oppervlaktehelderheid kan
hoogstens gelijk zijn aan de waarde met het blote \omterm{def:b1:optical-instruments:eye}{oog} (\omterm{prop:b1:optical-instruments:telescope}{uittredepupil} $=$ oogpupil).
Een ster wordt niet opgelost: al haar licht valt in één buigingsvlek, zodat haar
helderheid als $(D/d_p)^2$ groeit.

\textbf{20.} Een punt onder hoek $\beta$ wordt afgebeeld op hoogte $f_1'\beta$ in
het brandvlak; de stop laat door wat voldoet aan $\abs{f_1'\beta} \leq \phi/2$: volledig
veld $\phi/f_1'$ (kleine hoeken).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$: de hele maan
past erin. $5/1200 = 0.24^\circ$: niet.

\textbf{22.} Schijnbaar veld $\phi/f_2'$: $20/25 = \qty{0.8}{rad} = 46^\circ$;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$: $3420/15 = \qty{230}{s}$, bijna vier
minuten; $0.24^\circ = 864''$: \qty{58}{s}.

\textbf{24.} \omterm{def:b1:optical-instruments:microscope}{Oculair} van \qty{25}{mm} ($G = 48$, dicht bij $G_r$, helder, hele
maan) voor de schijf; \qty{6}{mm} ($G = 200$) om kraters van dichterbij te bekijken,
fletser en waziger. Het getal van het instrument: kraters tot
\qty{1.7}{km}.

\textbf{25.} Zelfde $f_1'$ en $D$: zelfde $G$, velden, uittredepupillen,
\omterm{rem:b1:optical-instruments:abbe}{oplossend vermogen} en licht (op een eventuele centrale afscherming na). Winst: geen
kleurafwijking. Praktisch verschil: het \omterm{def:b1:mirrors-thin-lenses:mirror}{brandpunt} ligt vóór de spiegel,
zodat een kleine hulpspiegel de bundel naar buiten naar het \omterm{def:b1:optical-instruments:microscope}{oculair} moet
vouwen.
\end{solution}
