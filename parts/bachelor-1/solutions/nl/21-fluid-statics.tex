\section*{Hoofdstuk \ref{ch:b1:fluid-statics} --- Hydrostatica}

\begin{solution}{exo:b1:fluid-statics:1}
$P = P_0 + \rho gh$: \qty{2.0e5}{Pa} (\qty{2.0}{bar}), \qty{1.1e6}{Pa}
(\qty{11}{bar}), \qty{1.0e7}{Pa} (\qty{101}{bar}). Op \qty{300}{m}: $P =
\qty{3.1e6}{Pa}$, kracht \qty{3.1}{MN} op \qty{1}{m^2} (netto, tegen
\qty{1}{atm} binnen: \qty{3.0}{MN}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:2}
$h = P/\rho g$: kwik \qty{0.760}{m}; water \qty{10.3}{m}. Een pomp kan hoogstens
een vacuüm boven het water maken: de atmosfeer duwt de kolom dan
\qty{10.3}{m} omhoog, niet meer.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:3}
$F_2 = F_1S_2/S_1 = \qty{10}{kN}$ (één ton); $d_2 = d_1S_1/S_2 = \qty{2.0}{mm}$
(hetzelfde verdrongen volume, dezelfde arbeid).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:4}
Dezelfde druk op het niveau van het grensvlak olie--water in beide \omterm{def:b1:angular-momentum:moment}{armen}: $\rho_og
h_o = \rho_wgh_w$, $h_w = 0.85 \times 12 = \qty{10.2}{cm}$ water boven
dat niveau in de andere \omterm{def:b1:angular-momentum:moment}{arm}; het olieoppervlak staat $12 - 10.2 = \qty{1.8}{cm}$
hoger.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:5}
$H = RT/Mg = 8.314 \times 273/(0.029 \times 9.81) = \qty{8.0}{km}$; $P/P_0 =
\eu^{-z/H}$: $0.54$ op \qty{5}{km}, $0.33$ op \qty{8848}{m}; de helft bij $H\ln2 =
\qty{5.5}{km}$ --- de getallen uit \cref{ch:b1:kinetic-theory}.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:6}
$F = \tfrac12\rho gh^2L = \qty{1.96e8}{N}$; \omterm{prop:b1:fluid-statics:wall}{drukmiddelpunt} op \qty{13.3}{m}
diepte, dat wil zeggen \qty{6.7}{m} boven de voet; moment om de voet $F \times
6.7 = \qty{1.3e9}{N.m}$. De dam naar het water toe krommen zet de stuwkracht
om in druk in de boog, die naar de dalwanden wordt afgevoerd.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:7}
$917/1025 = 89\%$; boomstam $60\%$. Vlot: \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} bij volledige onderdompeling $1000 \times
2.0 \times 9.81 = \qty{19.6}{kN}$, eigen gewicht \qty{11.8}{kN}: \qty{7.8}{kN}
over, tien mensen\dots{} de tiende met natte voeten: negen om droog te blijven
(\qty{800}{kg}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:8}
Ondergedompeld volume $m/\rho$: water \qty{20}{cm^3}; pekel $20/1.1 = \qty{18.2}{cm^3}$,
$\qty{1.8}{cm^3}$ minder, dus $1.8/0.50 = \qty{3.6}{cm}$ meer steel: \qty{7.6}{cm}
steekt boven water uit. Gevoeligheid $\approx m/(\rho^2S) \approx \qty{40}{cm}$ per eenheid
relatieve dichtheid (\qty{3.6}{cm} per $0.1$).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:9}
Verdrongen volume $2800/1.025 = \qty{2732}{m^3}$: \qty{268}{m^3} ($9\%$)
steekt boven water uit. Zwevend: massa $1025 \times 3000 = \qty{3075}{t}$: neem \qty{275}{t} in.
Zoet water: de \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} zakt tot \qty{3000}{t}: \qty{75}{t} te zwaar --- hij zinkt
tenzij \qty{75}{t} wordt weggepompt.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:10}
Verdrongen lucht: $10^{-2} \times 1.20 = \qty{12.0}{g}$; helium binnenin $12.0 \times
4/29 = \qty{1.7}{g}$; rubber \qty{3.0}{g}: nettodraagkracht \qty{7.3}{g} --- zeven meter
licht touw.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:11}
Moment $\int_0^h\rho gx\cdot xL\,\dd x = \rho gLh^3/3$, kracht $\rho gLh^2/2$:
\omterm{def:b1:angular-momentum:moment}{arm} $2h/3$. Van $h_1$ tot $h_2$: moment $\rho gL(h_2^3 - h_1^3)/3$, kracht
$\rho gL(h_2^2 - h_1^2)/2$: diepte $\frac23(h_2^3 - h_1^3)/(h_2^2 - h_1^2)$.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:12}
Een extra onderdompeling $x$ voegt de \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} $\rho gSx$ toe: $m\ddot x = -\rho gSx$,
$\omega_0^2 = \rho gS/m$, $T = 2\pi\sqrt{m/\rho gS} = 2\pi\sqrt{0.020/(1000 \times
9.81 \times 5\times10^{-5})} = \qty{1.3}{s}$.
\end{solution}

\begin{solution}{pb:b1:fluid-statics:1}
\textbf{1.} $P = 1.013\times10^5 + 1025 \times 9.81 \times 300 = \qty{3.12e6}{Pa}
= \qty{31}{bar}$.

\textbf{2.} $S = \pi(0.40)^2 = \qty{0.503}{m^2}$: nettokracht $(P - P_0)S =
\qty{1.52e6}{N}$ --- 155 ton, een volgeladen vrachtwagen op een luik.

\textbf{3.} De kiel \qty{10}{m} dieper: $+\rho g \times 10 = \qty{1.0e5}{Pa}$ meer,
$3\%$: de romp wordt vrijwel gelijkmatig belast.

\textbf{4.} $\Delta\rho/\rho = \chi_T\Delta P = 4.6\times10^{-10} \times 3.0\times10^6
= 0.14\%$: verwaarloosbaar voor de druk (enkele kilopascal op drie
miljoen).

\textbf{5.} Dertig bar buiten, één binnen: de romp is een drukvat
dat naar binnen wordt belast; een cilinder (of bol) zet de druk om in
druk langs zijn wand, waar staal goed tegen kan, terwijl een vlakke
plaat zou doorbuigen.

\textbf{6.} $\Delta P\,R/e = 3.0\times10^6 \times 5.0/0.030 = \qty{5.0e8}{Pa}
= \qty{500}{MPa}$: dicht bij de vloeigrens --- \qty{300}{m} ligt bij de
grens van deze romp, en diepere boten hebben dikker of sterker staal
nodig (of titaan).

\textbf{7.} Verdrongen volume $m_0/\rho = 2800/1.025 = \qty{2732}{m^3}$: \qty{268}{m^3}
steekt boven water uit ($9\%$).

\textbf{8.} Zwevend: $m = \rho V = \qty{3075}{t}$: neem \qty{275}{t} in.

\textbf{9.} De \omterm{thm:b1:fluid-statics:archimedes}{opwaartse kracht} daalt met $0.1\%$ van $\rho Vg$: $0.001 \times 3075 \times 9.81
\times10^3 = \qty{30}{kN}$ omlaag (drie ton). Dieper $\Rightarrow$ meer
samendrukking $\Rightarrow$ zwaarder: instabiel --- een onderzeeër moet
voortdurend trimmen (en een stalen romp drukt minder in dan water, wat helpt;
een te soepele romp zou zonder terugkeer zinken).

\textbf{10.} \omterm{thm:b1:fluid-statics:archimedes}{Opwaartse kracht} $1000 \times 3000 \times 9.81 = \qty{29.4}{MN}$ tegen
gewicht $3075 \times 9.81 \times 10^3 = \qty{30.2}{MN}$: \qty{0.74}{MN} omlaag
(\qty{75}{t}); pomp \qty{75}{t} weg.

\textbf{11.} Meteen \qty{20}{t} te licht: neem \qty{20}{t} water
(\qty{19.5}{m^3}) in de compensatietanks.

\textbf{12.} Water kan alleen worden verdreven door iets met een hogere druk
dan de zee buiten; perslucht in flessen duwt het naar buiten --- alleen
zolang de flesdruk de omgevingsdruk op die diepte
overtreft.

\textbf{13.} $PV$ constant: \qty{275}{m^3} bij \qty{31}{bar} vergt $275 \times 31
= \qty{8500}{m^3}$ lucht van \qty{1}{atm} --- opgeslagen bij \qty{200}{bar} in
\qty{43}{m^3} aan flessen.

\textbf{14.} \qty{2.0}{bar} bij \qty{10}{m}; \qty{4.0}{bar} bij \qty{30}{m}.

\textbf{15.} $PV$ constant: $6.0/4.0 = \qty{1.5}{L}$ --- de ribbenkast wordt
ingedrukt; vrijduikers voelen het.

\textbf{16.} Lucht die bij \qty{4}{bar} is ingeademd zet viervoudig uit: \qty{24}{L} in longen
die \qty{6}{L} bevatten: scheuring. Houd bij het stijgen nooit je adem in; adem
voortdurend uit.

\textbf{17.} $\Delta P = \rho g \times 1.0 = \qty{1.0e4}{Pa}$; kracht $\qty{1.0e4}{Pa}
\times 0.05 = \qty{500}{N}$ op de borstkas: de ademhalingsspieren kunnen
geen vijftig kilogram tillen --- een snorkel werkt alleen in de eerste decimeters.

\textbf{18.} $12 \times 200 = \qty{2400}{L}$ bij \qty{1}{bar}, dus \qty{600}{L}
bij \qty{4}{bar}: \qty{30}{min}.

\textbf{19.} $V = 10^8/1025 = \qty{97600}{m^3}$.

\textbf{20.} In zoet water moet hij $10^5\ \unit{m^3}$ verdringen: \qty{2400}{m^3}
meer, dus $2400/8000 = \qty{0.30}{m}$ dieper.

\textbf{21.} $10^4/1.025 = \qty{9756}{m^3}$ meer: \qty{1.2}{m} dieper.

\textbf{22.} Leeg ligt de $G$ van het schip hoog en is er weinig romp ondergedompeld:
het metacentrum kan onder $G$ zakken en het schip kan in een deining omslaan.
Zeewater laag in de romp verlaagt $G$ en vergroot de diepgang.

\textbf{23.} $\rho gh = \qty{2.0e5}{Pa}$, \qty{0.2}{MN/m^2}: vijftien keer minder
dan de \qty{3}{MN/m^2} van de onderzeeër.

\textbf{24.} Nee: de druk op de diepte van de onderzeeër is $P_0 + \rho gh$,
alleen door de diepte bepaald. Het gewicht van de tanker wordt gedragen door het
water dat hij verdringt, wat het zeeniveau overal met een onmeetbaar kleine
hoeveelheid verhoogt; er komt geen extra last bij de onderzeeër aan.

\textbf{25.} $\dd P/\dd z = -\rho g$ --- dus $P = P_0 + \rho gh$ voor elke
diepte, kracht en elk longvolume hier --- en de \omterm{thm:b1:fluid-statics:archimedes}{wet van Archimedes}, die
diezelfde betrekking is, geïntegreerd over een gesloten oppervlak.
\end{solution}
