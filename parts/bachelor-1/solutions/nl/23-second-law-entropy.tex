\section*{Hoofdstuk \ref{ch:b1:second-law-entropy} --- De tweede hoofdwet: entropie}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$: smelten $334000/273 = \qty{1.22}{kJ/K}$; koken $2260000/
373 = \qty{6.06}{kJ/K}$. Verdampen maakt de moleculen volledig van elkaar
los --- veel meer microscopische wanorde, en veel meer warmte per
\omterm{def:b1:kinetic-theory:temperature}{kelvin}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- een
toestandsfunctie: onafhankelijk van de methode (de \emph{geproduceerde} entropie niet).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$. Omkeerbaar isotherm: $Q = nRT\ln2$
uit het bad op $T$, $S_{\mathrm{exch}} = nR\ln2$, geproduceerd $0$. Joule:
$Q = 0$, $S_{\mathrm{exch}} = 0$, geproduceerd \qty{5.8}{J/K}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
$S = nC_{V,m}\ln T + nR\ln V$ constant: $T^{C_{V,m}}V^R$ constant, dus
$TV^{R/C_{V,m}} = TV^{\gamma - 1}$ constant. Plotselinge compressie: $Q = 0$, dus
$S_{\mathrm{exch}} = 0$ en $\Delta S = S_{\mathrm{created}}$; met $T$: $300 \to
643$ K, $V$: $24.9 \to 10.7$ L: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ geproduceerd.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
Blok: $450\ln(293/373) = \qty{-109}{J/K}$; meer: $+36000/293 = \qty{+123}{J/K}$;
heelal: $+\qty{14}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$; $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$, alles geproduceerd. $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$, dus is het argument van de logaritme $\geq 1$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$: $1000(1/280 - 1/300) = \qty{0.24}{J/K}$; van de oven naar de
kamer: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$, tien keer meer: hoe groter
het temperatuurverschil dat een warmtestroom oversteekt, hoe onomkeerbaarder --- de
val van oven naar kamer is waar de hoogwaardige energie wordt verkwist.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
Kringproces: $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$ met $S_{\mathrm{created}} \geq 0$,
dus $Q \leq 0$ en $W = -Q \geq 0$: de machine kan geen arbeid leveren. De
zee is één enkel warmtereservoir: haar warmte kan niet in arbeid worden omgezet zonder
een kouder lichaam om de entropie in te lozen.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
Elk gas zet uit van $V$ tot $2V$ bij constante $T$: $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$, geproduceerd (geen warmte uitgewisseld, de doos is geïsoleerd). Twee
helften stikstof: de wand wegnemen verandert macroscopisch niets ---
$\Delta S = 0$ (de paradox van Gibbs: gelijke moleculen worden niet ``gemengd'').
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
$Q = \qty{3.6e5}{J}$ naar de kamer op \qty{293}{K}: $S = \qty{1230}{J/K}$
geproduceerd --- in de gloeidraad (elektrische arbeid omgezet in warmte) en in de
stroom van die warmte van de hete gloeidraad naar de koele kamer; de elektrische
arbeid zelf droeg geen entropie.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
$\Delta S_{\text{lichaam}} = C\ln(T_0/T_1)$ in alle gevallen. (a) uitgewisseld
$C(T_0 - T_1)/T_0$: geproduceerd $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$. (b) twee stappen
met telkens verhouding $r = \sqrt{T_1/T_0}$: geproduceerd $2C[r - 1 - \ln r]$, kleiner
(voor $T_1/T_0 = 2$: $0.193C$ tegen $0.307C$). (c) oneindig veel stappen:
elk produceert $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$,
wat tot nul sommeert als $\epsilon \to 0$: omkeerbaar afkoelen vergt een reservoir
op elke tussenliggende temperatuur --- warmte mag nooit een eindig verschil oversteken.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$; $2^{-100} = \num{8e-31}$; $2^{-10^{22}} = 10^{-3\times10^{21}}$.
$k_B\ln2^N = Nk_B\ln2 = nR\ln2$. ``Onmogelijk'' betekent strikt genomen kans
nul; hier is zij $10^{-3\times10^{21}}$ --- niet nul, maar een getal zo klein dat
er nooit enig verschil waarneembaar is: onomkeerbaarheid is statistiek op
de schaal van $10^{23}$.
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$
voor de koffie: de kamer wint \qty{73}{kJ}, $\Delta S_{\text{kamer}} = 73150/293
= \qty{+250}{J/K}$.

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$.

\textbf{3.} $S_{\mathrm{created}} = -224 - (-250) = \qty{+26}{J/K} > 0$.

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$; $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$, $S_{\mathrm{exch}} = \qty{-117.7}{J/K}$: tot nu toe
\qty{18}{J/K} geproduceerd --- het grootste deel van de onomkeerbaarheid gebeurt terwijl de
koffie het heetst is.

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$ met $x = T_1/T_0$; $f'(x) = 1 - 1/x$ verdwijnt in $x = 1$, waar
$f = 0$, en $f'' = 1/x^2 > 0$: minimum $0$.

\textbf{6.} Nee: begin- en \omterm{def:b1:transient-regimes:firstorder}{eindtoestand} zijn dezelfde, de warmte is
dezelfde, de kamer is hetzelfde reservoir --- de geproduceerde entropie ligt vast door
de toestanden, niet door de snelheid.

\textbf{7.} $T_f = \qty{350}{K}$; geproduceerd $4180\ln(350^2/120000) = \qty{86}{J/K}$.

\textbf{8.} Omkeerbaar: de totale entropie blijft behouden: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$, dus $T_f'^2 = T_1T_2$.

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$; het water verliest $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$, geleverd als
arbeid.

\textbf{10.} Warm: $4180\ln(346.4/400) = \qty{-601}{J/K}$; koud: $4180\ln(346.4/
300) = \qty{+601}{J/K}$: som nul, zoals omkeerbaarheid eist.

\textbf{11.} $T_0S_{\mathrm{created}} = 300 \times 86 = \qty{26}{kJ}$, dicht bij de
\qty{30}{kJ} die terug te winnen was: de geproduceerde entropie, maal de
referentietemperatuur, is de arbeid die de onomkeerbare weg heeft weggegooid.

\textbf{12.} Het zou een motor vergen die werkt tussen twee lauwe
hoeveelheden water waarvan de temperaturen voortdurend veranderen --- enkele tientallen
kilojoule voor een grote machine: het loont niet, maar de tweede hoofdwet zegt
dat de arbeid er was.

\textbf{13.} Uit de binnenruimte ontvangen $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$;
aan de kamer afgestaan $Q_0/T_0 = (40 + W)/293$; balans over een kringproces:
$0 = 0.144 - (40 + W)/293 + S_c$.

\textbf{14.} $S_c \geq 0$: $(40 + W)/293 \geq 0.144$, $W \geq 42.2 - 40 =
\qty{2.2}{W}$; COP $\leq 40/2.2 = 18$ ($= T_c/(T_0 - T_c)$).

\textbf{15.} $\mathrm{COP}_{\max} = T_c/(T_0 - T_c)$: voor een vriezer $255/38 = 6.7$
--- hoe kouder de kast, hoe duurder elke verwijderde joule.

\textbf{16.} Met $W = 15$: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ geproduceerd
per seconde.

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$; de machine loost
\qty{200}{W} plus haar eigen arbeid in de kamer terwijl de binnenruimte koud
blijft: de nettowinst van de kamer is $W$ --- een open koelkast is een kachel.

\textbf{18.} $W = 0$ zou $S_c = Q_c(1/T_0 - 1/T_c) < 0$ geven: verboden ---
de \omterm{prop:b1:second-law-entropy:clausiuskelvin}{uitspraak van Clausius}.

\textbf{19.} Ervoor: $1$ (alles links); erna: $2^N$; $\Delta S = k_B\ln2^N =
Nk_B\ln2$.

\textbf{20.} $Nk_B = nR$: $\Delta S = nR\ln2$, het resultaat van de \omterm{rem:b1:first-law:irreversible}{expansie van Joule}.

\textbf{21.} Eén bit per molecuul: $N$ bits; en $\Delta S/(k_B\ln2) = N$ ---
entropie telt de informatie die men nodig zou hebben om de moleculen vast te
pinnen.

\textbf{22.} $2^{-50} = \num{9e-16}$; for a mole $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$.

\textbf{23.} Geen warmte ($Q = 0$): uitgewisseld $0$, geproduceerd $nR\ln2$ --- het
zuiverste geval: geen arbeid, geen warmte, alleen het uitspreiden van het gas.

\textbf{24.} $W = nRT\ln2$ ontvangen, $Q = -nRT\ln2$ aan de kamer afgestaan,
$S_{\mathrm{exch}} = -nR\ln2$ (de entropie van het gas keert omkeerbaar terug naar
haar beginwaarde): de vrije expansie ongedaan maken kost de arbeid $nRT\ln2$
en loost $nR\ln2$ entropie in de kamer --- het heelal houdt de
eens en voor altijd geproduceerde $nR\ln2$.

\textbf{25.} Entropie blijft in niets werkelijks behouden --- alleen in de
geïdealiseerde omkeerbare limiet; de wet verbiedt elk proces dat haar zou
vernietigen (warmte bergopwaarts, arbeid uit één reservoir, gratis ontmengen);
en elk beetje dat wordt geproduceerd is verloren arbeid, $T_0S_{\mathrm{created}}$ ervan.
\end{solution}
