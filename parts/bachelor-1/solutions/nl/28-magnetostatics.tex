\section*{Hoofdstuk \ref{ch:b1:magnetostatics} --- Magnetostatica: veld, krachten en dipolen}

\begin{solution}{exo:b1:magnetostatics:1}
$B = \mu_0I/2\pi r = 2 \times 10^{-7} \times 10/0.01 = \qty{2e-4}{T}$. Voor \qty{1}{T}:
$I = 2\pi rB/\mu_0 = \qty{5e4}{A}$. De waarde van de aarde bij $r = 2 \times 10^{-7} \times
100/5 \times 10^{-5} = \qty{0.4}{m}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:2}
Middelpunt: $\mu_0NI/2R = 4\pi \times 10^{-7} \times 200/0.1 = \qty{2.5}{mT}$. Bij $z = R$:
$\times(R^2/2R^2)^{3/2} = 2^{-3/2}$: \qty{0.89}{mT}. Bij $z = 10R$: $\times(1/101)^{3/2}
= 10^{-3}$: \qty{2.5}{\micro T}, al de dipoolwet in $1/z^3$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:3}
$B = \mu_0nI = 4\pi \times 10^{-7} \times 1000 \times 5 = \qty{6.3}{mT}$; aan het uiteinde
de helft: \qty{3.1}{mT}. Draad van \qty{0.5}{mm}: $2000$ windingen per meter, dus
\qty{12.6}{mT} bij \qty{5}{A}.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:4}
$F = ILB = 10 \times 1 \times 0.1 = \qty{1.0}{N}$; bij \qty{30}{\degree}, $\sin30^\circ$:
\qty{0.5}{N}. Motor: $100 \times \qty{1}{N} \times \qty{0.05}{m} = \qty{5}{N.m}$ ---
een kleine industriemotor.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:5}
Met $z = d\tan\alpha$: $B = \dfrac{\mu_0I}{4\pi}\displaystyle\int\dfrac{d\,\dd z}{(d^2 + z^2)^{3/2}}
= \dfrac{\mu_0I}{4\pi d}\displaystyle\int_{\alpha_1}^{\alpha_2}\cos\alpha\,\dd\alpha = \dfrac{\mu_0I}{4\pi d}
(\sin\alpha_2 - \sin\alpha_1)$; oneindige draad, $\alpha_{1,2} = \mp\pi/2$: $\mu_0I/2\pi d$.
Vierkant: elke zijde op $d = a/2$, gezien onder $\pm45^\circ$, geeft $\mu_0I\sqrt2/2\pi a$;
vier zijden: $B = 2\sqrt2\,\mu_0I/\pi a = 0.90\,\mu_0I/a$ (een cirkel met dezelfde
``straal'' $a/2$ zou $\mu_0I/a$ geven).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:6}
$B(z) = f(z - R/2) + f(z + R/2)$ met $f(u) = \mu_0NIR^2/2(R^2 + u^2)^{3/2}$.
Midden: $2f(R/2) = \mu_0NIR^2/(5R^2/4)^{3/2} = (4/5)^{3/2}\mu_0NI/R$. $f'$ is
oneven, dus $B'(0) = f'(-R/2) + f'(R/2) = 0$; $f''(u) \propto (4u^2 - R^2)(R^2 +
u^2)^{-7/2}$ verdwijnt in $u = R/2$, dus $B''(0) = 2f''(R/2) = 0$: het veld is
tot in de derde orde homogeen. Getallen: $0.716 \times 4\pi \times 10^{-7} \times 100/0.15 =
\qty{0.60}{mT}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:7}
Oppervlak: $\mu_0I/2\pi a = 2 \times 10^{-7} \times 100/0.002 = \qty{10}{mT}$; op
\qty{1}{mm}, binnenin, $\times r/a$: \qty{5}{mT}; op \qty{1}{cm}: \qty{2}{mT}. $B$ stijgt
lineair tot aan het oppervlak en daalt daarna als $1/r$. Coaxiaal: buiten de mantel
is de omsloten stroom nul, $B = 0$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:8}
$2 \times 10^{-7}$~N/m. Stroomrails: $2 \times 10^{-7} \times 10^8/0.1 = \qty{200}{N/m}$;
bij \qty{100}{kA}: $\qty{2e4}{N/m}$ --- twee ton per meter. De isolerende
steunen van schakelinstallaties worden op de kortsluitstroom gedimensioneerd, niet op de
bedrijfsstroom.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:9}
$\Gamma = mB_h = \qty{2e-6}{N.m}$; $W = 2mB_h = \qty{4e-6}{J}$; $J\ddot\theta =
-mB_h\theta$: $T = 2\pi\sqrt{J/mB_h} = 2\pi\sqrt{10^{-7}/2 \times 10^{-6}} =
\qty{1.4}{s}$. Klok de trillingen: $B_h = 4\pi^2J/mT^2$ zodra $m$ en $J$
bekend zijn (Gauss mat $m$ afzonderlijk uit de uitslag die het aan
een tweede naald geeft).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:10}
(a) $\mu_0IR^2/2z^3 = \mu_0(I\pi R^2)/2\pi z^3 = \mu_0m/2\pi z^3 = (\mu_0/4\pi)\,2m/z^3$:
de elektrische $2p/4\pi\varepsilon_0z^3$ met $1/\varepsilon_0 \to \mu_0$. (b) $m = 2\pi R_E^3B/
\mu_0 = 2\pi \times 2.58 \times 10^{20} \times 6 \times 10^{-5}/1.26 \times 10^{-6} = \qty{7.7e22}{A.m^2}$.
(c) $I = m/\pi R^2 = 7.7 \times 10^{22}/(\pi \times 9 \times 10^{12}) = \qty{2.7e9}{A}$. (d)
Evenaar ($\theta = 90^\circ$): $B_\theta = \mu_0m/4\pi R_E^3 = \qty{3e-5}{T}$, de helft van de
poolwaarde. Bij $\theta = 45^\circ$: $\tan(\text{inclinatie}) = B_r/B_\theta = 2\cot\theta = 2$,
inclinatie $= \qty{63}{\degree}$ --- op middelbare breedten duikt het veld steil de grond in.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:11}
$\mu_0nI = 4\pi \times 10^{-7} \times 2000 \times 3 = \qty{7.5}{mT}$. Midden: $\tan\alpha_1 =
R/(L/2) = 0.2$, $\cos\alpha_1 = -\cos\alpha_2 = 0.981$: $B = 0.981\,\mu_0nI =
\qty{7.4}{mT}$ ($98\%$). Uiteinde: $\cos\alpha_1 = 0$, $\tan\alpha_2 = R/L$, $\cos\alpha_2 =
-0.995$: $B = 0.497\,\mu_0nI = \qty{3.7}{mT}$ ($50\%$). Buiten, op afstand $x$ voorbij het
uiteinde, met beide uiteinden aan dezelfde kant: $B = \tfrac12\mu_0nI(\cos\alpha_1 - \cos\alpha_2)
\approx \tfrac12\mu_0nI\,\tfrac{R^2}{2}[1/x^2 - 1/(x + L)^2]$; $1\%$ van de waarde in het
midden bij $x \approx \qty{10}{cm}$, een halve lengte naar buiten. $B(z)$: een vlakke top over
het grootste deel van de lengte, gehalveerd aan de uiteinden en tot enkele procenten een
lengte verderop.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:12}
(a) Het eigen veld van de wikkelingen springt over hen heen van $0$ naar $B$; het
veld dat op hen werkt is het gemiddelde, $B/2$ (het vlak kan niet op
zichzelf duwen). Kracht op een wikkelelement met lengte $\dd l$ en breedte $\dd z$
dat $K\,\dd z$ voert: $K\,\dd z\,\dd l\,B/2$, radiaal naar buiten (controleer met $I\,\dd\vect l
\wedge\vect B$): druk $p = KB/2 = B^2/2\mu_0$ omdat $K = nI = B/\mu_0$.
(b) \qty{1}{T}: \qty{4e5}{Pa} = \qty{4}{bar}; \qty{10}{T}: \qty{400}{bar}; \qty{45}{T}:
\qty{8e8}{Pa} = \qty{8000}{bar}, dicht bij de \qty{1}{GPa} van staal. (c) De
wikkelingen moeten een druk weerstaan die als $B^2$ groeit: boven ongeveer \qty{40}{T}
is geen materiaal en geen koelschema meer opgewassen --- de grens van stationaire magneten
is mechanisch (en, bij supergeleiders, hun kritische veld).
\end{solution}

\begin{solution}{pb:b1:magnetostatics:1}
\textbf{1.} Elk vlak dat de as bevat is een \omterm{prop:b1:electrostatics-gauss:symmetry}{symmetrievlak} van
de stromen: $\vect B$ staat er loodrecht op, orthoradiaal; door invariantie
langs en om de as geldt $\vect B = B(r)\,\vect e_\theta$.

\textbf{2.} Cirkel met straal $r$: $2\pi rB = \mu_0I_{\text{oms}}$. Kern:
$I_{\text{oms}} = Ir^2/a^2$, $B = \mu_0Ir/2\pi a^2$; ertussen: $\mu_0I/2\pi r$;
erbuiten ($r > c$): $I - I = 0$, $B = 0$.

\textbf{3.} $B(a) = 2 \times 10^{-7}/5 \times 10^{-4} = \qty{4.0e-4}{T}$; $B(b) = 2 \times
10^{-7}/2.5 \times 10^{-3} = \qty{8.0e-5}{T}$.

\textbf{4.} Binnen de buis is geen stroom omsloten: daar is $B = 0$ in plaats
van de lineaire stijging; verder verandert er niets --- het veld buiten
een cilindrische stroom hangt alleen van de totale stroom af.

\textbf{5.} $I_{\text{oms}} = I - I(r^2 - b^2)/(c^2 - b^2) = I(c^2 - r^2)/(c^2 -
b^2)$: $B = \dfrac{\mu_0I}{2\pi r}\,\dfrac{c^2 - r^2}{c^2 - b^2}$, gelijk aan $\mu_0I/2\pi b$ in
$b$ en aan $0$ in $c$: continu. Schets: lineair tot \qty{4e-4}{T} op
\qty{0.5}{mm}, dan als $1/r$ omlaag tot \qty{8e-5}{T} op \qty{2.5}{mm}, dalend tot nul op
\qty{3}{mm}, en nul daarbuiten.

\textbf{6.} De stroom in de mantel is tegengesteld aan die in de kern: afstoting,
een naar buiten gerichte druk. Oppervlaktestroom $K = I/2\pi b = \qty{64}{A/m}$; gemiddeld veld
$\tfrac12(8 \times 10^{-5} + 0) = \qty{4e-5}{T}$: $p = KB = \qty{2.5e-3}{Pa}$. Bij
\qty{10}{kA}: $\times10^8$, \qty{2.5e5}{Pa} = \qty{2.5}{bar} --- kabels voor
pulsvermogen zijn erop gebouwd.

\textbf{7.} Geen veld erbuiten: de kabel straalt niets uit en pikt ook geen
magnetische storing van zijn buren op. Een paar draden: de twee
velden heffen elkaar bijna op, wat $\approx \mu_0Id/2\pi D^2 = 2 \times 10^{-7} \times 2 \times
10^{-3}/10^{-2} = \qty{4e-8}{T}$ op \qty{10}{cm} overlaat --- klein, maar de coaxkabel geeft
nul.

\textbf{8.} $D = \sqrt{h^2 + d^2/4} = \qty{10.0}{m}$: elk geeft $\mu_0I/2\pi D =
\qty{1.0e-5}{T}$, loodrecht op de lijn vanaf de geleider. Omdat de
stromen tegengesteld zijn heffen de componenten evenwijdig aan de verbindingslijn van
de geleiders elkaar op en tellen de verticale op: $B = 2 \times 10^{-5} \times
(d/2)/D = \qty{1.0e-6}{T}$, verticaal. Vijftig keer onder het aardveld,
honderd keer onder de richtlijn.

\textbf{9.} Voor $D \gg d$ is $B \approx \dfrac{\mu_0I}{2\pi}\Bigl(\dfrac1{D_1} - \dfrac1{D_2}\Bigr)
\sim \dfrac{\mu_0I}{2\pi}\dfrac{d}{D^2}$: de twee velden in $1/D$ heffen elkaar in eerste
orde op en alleen hun verschil, van relatieve grootte $d/D$, blijft over --- een
``tweedraadsdipool''.

\textbf{10.} Eigen geleider op \qty{1}{m}: \qty{1.0e-4}{T} horizontaal; de andere
op \qty{1.41}{m}: \qty{7.1e-5}{T} onder \qty{45}{\degree}, tegengesteld: resultante
$(5 \times 10^{-5}, 5 \times 10^{-5})$, \qty{7e-5}{T} --- \qty{70}{\micro T}, onder de
richtlijn voor het publiek en ver onder die voor beroepsmatige blootstelling.

\textbf{11.} $F/\ell = \mu_0I^2/2\pi d = 2 \times 10^{-7} \times 2.5 \times 10^5 =
\qty{0.05}{N/m}$, afstotend; bij \qty{20}{kA}: \qty{80}{N/m}.

\textbf{12.} \qty{15}{N} normaal; \qty{24}{kN} bij kortsluiting, acht keer
het gewicht van de overspanning van \qty{2.9}{kN}: de geleiders worden uiteengeslingerd, zwaaien
en kunnen elkaar bij het terugvallen raken --- vandaar afstandhouders en beveiligingen die
binnen een paar perioden openen.

\textbf{13.} $I\cos\omega t + I\cos(\omega t - 2\pi/3) + I\cos(\omega t + 2\pi/3) =
I\cos\omega t\,(1 + 2\cos2\pi/3) = 0$: de drie \omterm{def:b1:sinusoidal-impedance:complex}{fasoren} tellen tot nul op. De
velden in $1/D$ heffen elkaar op zoals bij de coaxkabel; het verre veld daalt als $1/D^2$ of
sneller.

\textbf{14.} Het veld van de lijn is daar verticaal; een kompas reageert alleen op
de horizontale component: geen uitslag (de inclinatie verandert met
$10^{-6}/(2 \times 10^{-5})$, drie graden, wat het kompas niet kan tonen).

\textbf{15.} Zijden $a$ (evenwijdig aan de as): het radiale $\vect B$ staat er
loodrecht op, kracht $NIaB$ op elk, loodrecht op het
vlak van de spoel en tegengesteld op de twee zijden: een \omterm{def:b1:angular-momentum:moment}{koppel}. Zijden $b$: $\dd\vect l$
is radiaal en evenwijdig aan $\vect B$: geen kracht.

\textbf{16.} $\Gamma = 2 \times NIaB \times b/2 = NIabB = NISB = 0.024\,I$ (N\,m, $I$
in A). Het radiale veld ligt altijd in het vlak van de spoel en loodrecht
op de zijden $a$: de krachten staan altijd loodrecht op dat vlak en hun
\omterm{def:b1:angular-momentum:moment}{arm} $b/2$ verandert nooit.

\textbf{17.} $C\theta = NISB$: $\theta = 2400\,I$ rad, dus \qty{2.4}{rad/mA}; volle
uitslag \qty{1.2}{rad} bij $I = \qty{0.50}{mA}$.

\textbf{18.} Lengte $200 \times 2(0.02 + 0.03) = \qty{20}{m}$, doorsnede $\pi(5 \times
10^{-5})^2 = \qty{7.9e-9}{m^2}$: $R = 1.7 \times 10^{-8} \times 20/7.9 \times 10^{-9} =
\qty{43}{\ohm}$; $U = 43 \times 5 \times 10^{-4} = \qty{22}{mV}$; $P = UI = \qty{11}{\micro W}$.

\textbf{19.} $J\ddot\theta = -C\theta + NISB\,I$: $\omega_0 = \sqrt{C/J} = \sqrt{10^{-5}/6.7
\times 10^{-8}} = \qty{12}{rad/s}$, $T = \qty{0.51}{s}$.

\textbf{20.} $J\ddot\theta + \alpha\dot\theta + C\theta = NISB\,I$: kritisch bij
$\alpha = 2\sqrt{JC} = 2\sqrt{6.7 \times 10^{-13}} = \qty{1.6e-6}{N.m.s}$; de naald
bereikt haar aanwijzing dan het snelst zonder door te schieten.

\textbf{21.} Voltmeter: $R_s = 10/5 \times 10^{-4} - 43 \approx \qty{20}{k\ohm}$.
Ampèremeter: de parallelweerstand voert \qty{0.9995}{A} bij \qty{22}{mV}: $R_{\text{sh}} =
\qty{22}{m\ohm}$.

\textbf{22.} $m = NIS = 200 \times 5 \times 10^{-4} \times 6 \times 10^{-4} = \qty{6e-5}{A.m^2}$;
$-mB = \qty{-1.2e-5}{J}$; veer $\tfrac12C\theta^2 = \tfrac12 \times 10^{-5} \times 1.44 =
\qty{7.2e-6}{J}$: dezelfde orde --- de veer slaat op wat het magnetische
\omterm{def:b1:angular-momentum:moment}{koppel} heeft verricht terwijl de spoel draaide.

\textbf{23.} In een homogeen veld is het \omterm{def:b1:angular-momentum:moment}{koppel} $mB\cos\theta$ (maximaal wanneer
het vlak $\vect B$ bevat, nul wanneer de normaal erlangs ligt): $C\theta =
NISB\,I\cos\theta$ --- alleen lineair voor kleine $\theta$, samengedrukt aan de
bovenkant van de schaal.

\textbf{24.} \qty{50}{Hz} tegen $\omega_0/2\pi = \qty{2}{Hz}$: de spoel kan het niet
volgen; zij blijft op het gemiddelde van de stroom staan, nul, met een trilling met
\omterm{def:b1:wave-propagation:signal}{amplitude} $\approx \theta_{\text{stat}}(\omega_0/\omega)^2 = 1.2 \times (12/314)^2 \approx
\qty{2e-3}{rad}$, onzichtbaar. Een draaispoelmeter leest het gemiddelde af --- voor
wisselstroom heeft zij een gelijkrichter nodig.

\textbf{25.} \qty{4e-4}{T} in de kern van een coaxkabel met \qty{1}{A} en niets erbuiten
(de \omterm{thm:b1:magnetostatics:ampere}{stelling van Ampère}); \qty{1}{\micro T} onder een lijn met \qty{500}{A} en \qty{0.05}{N/m}
tussen haar geleiders (Biot--Savart plus Laplace); \qty{2.4}{rad/mA} voor
de meter (laplacekoppel tegen een veer).
\end{solution}
