\section*{Hoofdstuk \ref{ch:b1:oscillators-resonance} --- Mechanische oscillatoren: demping en resonantie}

\begin{solution}{exo:b1:oscillators-resonance:1}
$\omega_0 = \sqrt{80/0.20} = \qty{20}{rad/s}$, $f_0 = \qty{3.2}{Hz}$, $T_0 =
\qty{0.31}{s}$; $v_{\max} = A\omega_0 = \qty{1.0}{m/s}$; $E = \tfrac12 kA^2 =
\qty{0.10}{J}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:2}
$T = 2\pi\sqrt{\ell/g}$: \qty{2.0}{s} op aarde, \qty{4.9}{s} op de maan.
De periode van massa en veer, $2\pi\sqrt{m/k}$, bevat geen $g$:
onveranderd.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:3}
$\tau = 2Q/\omega_0 = 100/(4\pi) = \qty{8.0}{s}$; de energie $\propto \eu^{-2t/\tau}$
halveert op $t = \tau\ln2/2 = \qty{2.8}{s}$; $5\%$ na $3\tau = \qty{24}{s}$,
ongeveer $48 \approx Q$ trillingen.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:4}
$X_0 = F_0/k = \qty{1.0}{cm}$; bij resonantie $\approx QX_0 = \qty{10}{cm}$;
bij $3\omega_0$: $X_0/\sqrt{64 + 0.09} = \qty{0.13}{cm}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:5}
$v = Lf_0 = 8.0 \times 1.3 = \qty{10}{m/s}$ (\qty{37}{km/h}). Bij
\qty{60}{km/h}: $f = 16.7/8.0 = \qty{2.1}{Hz}$, $x_r = 1.6$: $X/X_g =
\sqrt{(1 + 5.2)/(2.4 + 5.2)} = 0.90$ --- nauwelijks geïsoleerd; isolatie
vraagt $x_r \gg 1$ (bij \qty{120}{km/h} is $x_r = 3.2$: $0.45$).
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:6}
De arbeid van een duw is alleen positief wanneer die met de beweging mee werkt;
elke periode geduwd tellen alle duwen op; bij elk ander ritme
wisselen zij af. Stationair: $2\pi E/Q = \qty{25}{J}$, $E = \qty{80}{J}$;
uit $mg\ell(1 - \cos\theta_{\max}) = 80$ volgt $1 - \cos\theta_{\max} = 0.109$,
$\theta_{\max} = 27^\circ$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:7}
$\delta = \ln(1/0.9) = 0.105$, $Q = \pi/\delta = 30$; $\Delta f = f_0/Q =
\qty{0.067}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:8}
Uitwijking: in fase ($\varphi = 0$) ruim eronder, $-\pi/2$ bij
resonantie, $-\pi$ (tegenfase) ruim erboven. De snelheid bij resonantie: in
fase met de kracht. Het vermogen is $\vect F\cdot\vect v$: het is de fase van de snelheid,
niet die van de uitwijking, die bepaalt hoeveel arbeid de aandrijving
verricht.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:9}
$V = \omega X = \omega(F_0/m)/\sqrt{(\omega_0^2 - \omega^2)^2 + (\omega\omega_0/Q)^2}$;
deel teller en noemer door $\omega\omega_0/Q$:
\[
V = \frac{F_0Q/m\omega_0}{\sqrt{1 + Q^2(\omega_0/\omega - \omega/\omega_0)^2}},
\]
en $F_0Q/m\omega_0 = F_0/\alpha$.
De wortel is $\geq 1$ en precies gelijk aan $1$ bij $\omega = \omega_0$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:10}
$P = F_0^2/2\alpha = \qty{10}{W}$; $V_{\max} = F_0/\alpha = \qty{20}{m/s}$,
$E = \tfrac12 mV_{\max}^2 = \qty{40}{J}$; $E/P = \qty{4.0}{s}$ en $Q/\omega_0 =
m/\alpha = \qty{4.0}{s}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:11}
$3\tau = \qty{10}{s}$: $\tau = \qty{3.3}{s}$, $Q = \omega_0\tau/2 = 2\pi \times
440 \times 3.3/2 \approx 4600$; $\Delta f = f_0/Q = \qty{0.1}{Hz}$. Zij
reageert op, en zendt uit, in wezen één frequentie: een standaard die tot op
enkele delen in $10^4$ klopt.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:12}
$m\ddot x = mg - \rho_wgS(h + x) = -\rho_wgSx$ (want $mg = \rho_wgSh$):
$\omega_0^2 = \rho_wgS/m = g/h$; $T = 2\pi\sqrt{0.10/9.81} = \qty{0.63}{s}$.
Zowel de massa als de dichtheid van de vloeistof komt alleen via de
evenwichtsdiepte $h$ binnen.
\end{solution}

\begin{solution}{pb:b1:oscillators-resonance:1}
\textbf{1.} Krachten: het gewicht (in evenwicht met de statische uitrekking), de veer
$-ku$, de demper $-\alpha\dot u$; $m\ddot x = -ku - \alpha\dot u$ met $\ddot x =
\ddot x_g + \ddot u$.

\textbf{2.} $\ddot u + (\omega_0/Q)\dot u + \omega_0^2u = -\ddot x_g$, $\omega_0 =
\sqrt{k/m}$, $Q = \sqrt{km}/\alpha$.

\textbf{3.} $k = m\omega_0^2 = \pi^2 = \qty{9.9}{N/m}$; $\alpha = m\omega_0/Q =
3.14/0.70 = \qty{4.5}{N.s/m}$.

\textbf{4.} $u = -a_g/\omega_0^2$: de massa zakt over een vaste afstand --- het
instrument is bij frequentie nul een versnellingsopnemer (en een hellingmeter).

\textbf{5.} De relatieve uitwijking $u(t)$.

\textbf{6.} $-\omega^2\underline U + j\omega(\omega_0/Q)\underline U + \omega_0^2\underline U
= \omega^2X_g$; deel door $\omega_0^2$.

\textbf{7.} $U/X_g = x_r^2/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$.

\textbf{8.} $U \to X_g$: de massa beweegt niet in het inertiaalstelsel
(haar traagheid houdt haar stil terwijl de veer te slap is om haar mee te trekken);
het frame beweegt om haar heen, en de pen registreert de uitwijking van de grond
--- een seismometer.

\textbf{9.} $U \approx x_r^2X_g = \omega^2X_g/\omega_0^2 = a_g/\omega_0^2$: een
versnellingsopnemer.

\textbf{10.} $U = QX_g = 0.7X_g$: geen piek; de twee regimes sluiten vloeiend
op elkaar aan en het instrument galmt niet na.

\textbf{11.} Voor $Q = 0.7$: een lijn met helling $2$ die naar $1$ stijgt rond
$x_r = 1$ en daarna vlak op $1$ blijft; voor $Q = 5$: hetzelfde met een piek van hoogte
$5$ bij $x_r = 1$.

\textbf{12.} $\underline U \to -X_g$: $u = -x_g$, de massa stil terwijl het
frame beweegt.

\textbf{13.} $\Delta\ell = g/\omega_0^2 = 9.81/(2\pi \times 0.02)^2 = \qty{620}{m}$:
absurd.

\textbf{14.} $\ell = g/\omega_0^2 = \qty{620}{m}$.

\textbf{15.} Tuinhek: de slinger zwaait om een bijna verticale
as, zodat maar een klein deel van $g$ haar terugdrijft en een korte arm
een lange periode geeft. Krachtterugkoppeling: een spoel houdt de massa vast, en de
stroom die daarvoor nodig is, is evenredig met de grondversnelling --- de
stijfheid is elektronisch en instelbaar.

\textbf{16.} $\Delta\ell = 9.81/\pi^2 = \qty{0.99}{m}$: haalbaar.

\textbf{17.} $x_r = 0.2$: $U/X_g = 0.04/\sqrt{0.92 + 0.08} = 0.040$:
\qty{40}{\micro m} --- het regime van de versnellingsopnemer, dat versterking vraagt (optische
hefboom of elektronica).

\textbf{18.} $x_r = 10$: $100/\sqrt{9801 + 204} \approx 1.00$: \qty{1.0}{mm},
een getrouwe registratie van de uitwijking.

\textbf{19.} $\omega_0 = \qty{6.3e3}{rad/s}$: $U = g/\omega_0^2 = \qty{2.5e-7}{m}$
--- uitgelezen als de verandering van een condensatorspleet.

\textbf{20.} $U/(a_g/\omega_0^2) = 1/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$; met
$Q = 0.7$ daalt zij monotoon en blijft zij boven $0.9$ tot
$x_r \approx 0.7$: ongeveer \qty{700}{Hz}.

\textbf{21.} $\tau = 2Q/\omega_0 = \qty{0.22}{ms}$: hij volgt een val (tientallen
milliseconden) zonder achterstand.

\textbf{22.} $\tau = 1.4/3.14 = \qty{0.45}{s}$: na een stap van de grond komt de
registratie in een halve seconde tot rust, zonder nagalm.

\textbf{23.} Een hoge $Q$ zou het instrument na elke schok op $f_0$ laten
nagalmen (het zou zichzelf registreren in plaats van de grond) en rond
$f_0$ pieken; $Q \approx 0.7$ is vlak en snel.

\textbf{24.} $U = QX_g = 0.7X_g$ tegen het ideale $X_g$ (of $a_g/\omega_0^2
= X_g$): $30\%$ te laag.

\textbf{25.} Vergelijk de signaalfrequentie met $f_0$: ruim erboven is de
massa een vast referentiepunt en is $u$ de uitwijking; ruim eronder is $u$
de versnelling gedeeld door $\omega_0^2$.
\end{solution}
