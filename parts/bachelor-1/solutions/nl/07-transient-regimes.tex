\section*{Hoofdstuk \ref{ch:b1:transient-regimes} --- Overgangsverschijnselen: eerste en tweede orde}

\begin{solution}{exo:b1:transient-regimes:1}
$\tau = RC = 10^4 \times 47\times10^{-9} = \qty{0.47}{ms}$; $u_C(\tau)/E
= 1 - \eu^{-1} = 0.63$; $99\%$: $\eu^{-t/\tau} = 0.01$, $t = \tau\ln 100
= 4.6\tau = \qty{2.2}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:2}
$\tau = L/R = \qty{4.0}{ms}$; $i_\infty = E/R = \qty{2.0}{A}$; $i(\tau) =
0.63 \times 2.0 = \qty{1.26}{A}$; op $0^+$ is $i = 0$ (continuïteit), dus
$u_L = E - Ri = \qty{10}{V}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:3}
$\omega_0 = 1/\sqrt{LC} = \qty{3.16e4}{rad/s}$, $f_0 = \qty{5.03}{kHz}$;
$Q = \sqrt{L/C}/R = 316/100 = 3.2 > \tfrac12$: pseudoperiodiek;
$\tau = 2Q/\omega_0 = \qty{0.20}{ms}$; $\omega = \omega_0\sqrt{1 - 1/4Q^2}
= 0.987\omega_0$, $T = \qty{0.20}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:4}
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$, $Q = \tfrac12$. Bij \qty{2}{k\ohm} is
$Q = 0.16$: aperiodisch, een traag kruipen dat door de kleine wortel
$\abs{r_+} \approx \omega_0 Q = \qty{5e3}{s^{-1}}$ wordt beheerst (tijdschaal
\qty{0.2}{ms}, maar geen trilling).
\end{solution}

\begin{solution}{exo:b1:transient-regimes:5}
$u = U_0\eu^{-t/\tau}$ halveert wanneer $\eu^{-t/\tau} = \tfrac12$: $t_{1/2}
= \tau\ln 2$. $\tau = 3.5/0.693 = \qty{5.05}{ms}$, $R = \tau/C =
\qty{5.1}{k\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:6}
$i = (E/R)\eu^{-t/\tau}$: $\int_0^\infty Ri^2\,\dd t = (E^2/R)(\tau/2) =
\tfrac12 CE^2$, met geen $R$ in zicht. De bron levert $CE^2$: rendement
$50\%$ voor elke weerstand. Beter: laden via een spoel (de
stroom is dan niet evenredig met het spanningsverschil) of in meerdere
spanningsstappen --- elke stap van $E/n$ verliest maar $\tfrac12 C(E/n)^2$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:7}
$\delta = \ln(2.0/1.5) = 0.29$ (en $\ln(1.5/1.12) = 0.29$: consistent);
$Q \approx \pi/\delta = 11$. $T = \qty{0.40}{ms}$, dus $\omega_0 \approx
2\pi/T = \qty{1.57e4}{rad/s}$ (de correctie $1/8Q^2$ is $0.1\%$).
$L = 1/(\omega_0^2 C) = \qty{4.1}{mH}$; $R = \omega_0 L/Q = \qty{5.8}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:8}
$u_C \approx U_0\eu^{-t/\tau}\cos\omega_0 t$; $i = -C\,\dd u_C/\dd t
\approx CU_0\omega_0\sin\omega_0 t$ (voor $t \ll \tau$), piek $U_0\sqrt{C/L}
= 100 \times 0.1 = \qty{10}{A}$, bereikt een kwart periode na het sluiten,
wanneer $u_C = 0$: alle energie ($\tfrac12 CU_0^2 = \qty{50}{mJ}$) is dan
magnetisch, $\tfrac12 Li^2 = \qty{50}{mJ}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:9}
$\omega_0 = \sqrt{k/m} = \qty{7.1}{rad/s}$, $f_0 = \qty{1.1}{Hz}$;
$\alpha_c = 2\sqrt{km} = 2\sqrt{8\times10^6} = \qty{5.7e3}{N.s/m}$.
$\alpha = 2000$: $Q = \sqrt{km}/\alpha = 2828/2000 = 1.4$, pseudoperiodiek;
$T = 2\pi/(\omega_0\sqrt{1 - 1/4Q^2}) = 0.89/0.935 = \qty{0.95}{s}$;
$\tau = 2Q/\omega_0 = \qty{0.40}{s}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:10}
$i = I_0\eu^{-t/\tau}$, $\tau = L/R = \qty{5.0}{ms}$; piekspanning
$RI_0 = \qty{200}{V}$ op $t = 0^+$; energie $\tfrac12 LI_0^2 = \qty{1.0}{J}$
gedissipeerd in $R$. Zonder die weerstand heeft de stroom geen weg:
$\dd i/\dd t$ is enorm, de spoelspanning loopt op tot kilovolts, en er slaat
een vonk over de schakelaar.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:11}
Gezien vanuit $C$: $E_{\mathrm{Th}} = ER_2/(R_1 + R_2) = \qty{8.0}{V}$,
$R_{\mathrm{Th}} = R_1 \parallel R_2 = \qty{6.67}{k\ohm}$; $\tau =
R_{\mathrm{Th}}C = \qty{6.7}{ms}$; $u_C = 8.0(1 - \eu^{-t/\tau})$ V.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:12}
Dubbele wortel $-\omega_0$: $x = x_\infty + (A + Bt)\eu^{-\omega_0 t}$;
$x(0) = 0$ geeft $A = -x_\infty$, $x'(0) = 0$ geeft $B = \omega_0 A$:
$x = x_\infty[1 - (1 + \omega_0 t)\eu^{-\omega_0 t}]$. $95\%$:
$(1 + s)\eu^{-s} = 0.05$, $s \approx 4.7$: $t_{95} = 4.7/\omega_0$. Voor
$Q = 5$ heeft de omhullende $\eu^{-t/\tau}$ met $\tau = 10/\omega_0$
$3\tau = 30/\omega_0$ nodig; voor $Q = 0.1$ is de trage wortel $\abs{r_+} \approx
\omega_0 Q$ en is $3/\abs{r_+} = 30/\omega_0$. Beide zijn zes keer trager:
kritische demping is de snelste terugkeer zonder doorschot, en dat is wat een
meter, een deurdranger of een vering wil.
\end{solution}

\begin{solution}{pb:b1:transient-regimes:1}
\textbf{1.} $E = L\,\dd i/\dd t + Ri + u_C$, $i = C\,\dd u_C/\dd t$:
$LC\,u_C'' + RC\,u_C' + u_C = E$.

\textbf{2.} $u_C'' + (\omega_0/Q)u_C' + \omega_0^2u_C = \omega_0^2E$ met
$\omega_0 = 1/\sqrt{LC}$, $Q = \sqrt{L/C}/R$.

\textbf{3.} $\omega_0 = \qty{3.16e4}{rad/s}$, $f_0 = \qty{5.03}{kHz}$,
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$.

\textbf{4.} $r^2 + (\omega_0/Q)r + \omega_0^2 = 0$, $\Delta = \omega_0^2(1/Q^2
- 4)$: $Q < \tfrac12$ aperiodisch, $Q = \tfrac12$ kritisch, $Q > \tfrac12$
pseudoperiodiek.

\textbf{5.} $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$,
$u_C = E + \eu^{-t/\tau}(A\cos\omega t + B\sin\omega t)$.

\textbf{6.} $Q = 3.16$, $\tau = \qty{0.20}{ms}$, $T = 2\pi/\omega =
\qty{0.20}{ms}$, ongeveer drie trillingen.

\textbf{7.} $u_C(0^+) = U_0$ ($u_C$ continu: eindige stroom);
$u_C'(0^+) = i(0^+)/C = 0$ ($i_L$ continu en ervoor nul).

\textbf{8.} $m\ddot x = -k(x + x_{\text{eq}}) - \alpha\dot x + mg$ met
$kx_{\text{eq}} = mg$: $m\ddot x + \alpha\dot x + kx = 0$.

\textbf{9.} $\omega_0 = \sqrt{k/m}$, $Q = \sqrt{km}/\alpha$; $x \leftrightarrow
q$, $\dot x \leftrightarrow i$, $m \leftrightarrow L$, $\alpha \leftrightarrow
R$, $k \leftrightarrow 1/C$.

\textbf{10.} $\omega_0 = \sqrt{22000/350} = \qty{7.93}{rad/s}$, $f_0 =
\qty{1.26}{Hz}$; $\alpha_c = 2\sqrt{km} = \qty{5.55e3}{N.s/m}$.

\textbf{11.} $Q = 2775/4000 = 0.69$, $\xi = 0.72$: pseudoperiodiek, net
onder kritisch.

\textbf{12.} $\tau = 2Q/\omega_0 = \qty{0.18}{s}$; $\omega = \omega_0\sqrt{1
- \xi^2} = \qty{5.5}{rad/s}$, $T = \qty{1.1}{s}$.

\textbf{13.} Doorschot $\exp(-\pi \times 0.72/0.69) = \exp(-3.3) = 3.8\%$:
één kleine uitwijking voorbij het evenwicht, en klaar --- comfortabel.

\textbf{14.} $R = \sqrt{L/C}/Q = 316/0.69 = \qty{455}{\ohm}$.

\textbf{15.} $Q = 2775/1200 = 2.3$, $\xi = 0.22$; $\omega = 0.976\omega_0$,
$T = \qty{0.81}{s}$; $\tau = 2Q/\omega_0 = \qty{0.58}{s}$.

\textbf{16.} $\eu^{-T/\tau} = \eu^{-1.39} = 0.25$: elke wip is een
kwart van de vorige; $0.25^3 = 1.6\%$: drie zichtbare wippen.

\textbf{17.} $x(0) = -x_0$, $\dot x(0) = 0$; energie $\tfrac12 kx_0^2 =
0.5 \times 22000 \times 0.0025 = \qty{28}{J}$, als warmte gedissipeerd in de
olie van de demper.

\textbf{18.} $v \approx x_0\omega_0 = 0.05 \times 7.93 = \qty{0.40}{m/s}$;
dempkracht $\alpha v \approx 1200 \times 0.40 = \qty{480}{N}$.

\textbf{19.} Bij $Q = \tfrac12$ verloopt de terugkeer als $(1 + \omega_0 t)\eu^{-\omega_0
t}$, die asymptotisch kruipt: de laatste millimeters duren lang. Een
iets te weinig gedempt systeem ($\xi \approx 0.7$) schiet enkele
procenten door maar komt sneller binnen een tolerantieband en blijft daar.

\textbf{20.} $\exp(-\pi \times 0.22/0.976) = \exp(-0.70) = 50\%$, tegen
$3.8\%$: de versleten auto wipt halverwege terug voorbij het evenwicht.

\textbf{21.} $\delta = \ln 4 = 1.39$ (twee keer, consistent); $\xi =
\delta/\sqrt{4\pi^2 + \delta^2} = 1.39/6.43 = 0.215$, $Q = 1/2\xi = 2.3$:
de versleten waarde.

\textbf{22.} $-\ln 0.038 = 3.27 = \pi\xi/\sqrt{1 - \xi^2}$, dus $\xi =
3.27/\sqrt{\pi^2 + 3.27^2} = 0.72$, $Q = 0.69$.

\textbf{23.} Energie $\propto$ \omterm{def:b1:wave-propagation:signal}{amplitude}$^2$: overblijvende fractie per
periode $0.25^2 = 6\%$, dus $94\%$ gedissipeerd per pseudoperiode.

\textbf{24.} $R = 316/2.3 = \qty{137}{\ohm}$. In de variabele $\omega_0 t$
bevat de canonieke vergelijking alleen $Q$: twee systemen met dezelfde $Q$
tekenen dezelfde kromme.

\textbf{25.} De \omterm{def:b1:transient-regimes:secondorder}{kwaliteitsfactor} $Q$ (of $\xi$) alleen bepaalt de vorm; een
ontwerper streeft naar $Q \approx 0.7$ ($\xi \approx 0.7$); meer dan één
zichtbare wip betekent $Q$ ruim boven $1$: de demper is versleten.
\end{solution}
