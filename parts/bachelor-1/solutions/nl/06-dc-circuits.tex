\section*{Hoofdstuk \ref{ch:b1:dc-circuits} --- Gelijkstroomkringen: de wetten van Kirchhoff en enkele stellingen}

\begin{solution}{exo:b1:dc-circuits:1}
$q = 1.5 \times 3600 = \qty{5400}{C}$, $N = 5400/1.6\times10^{-19} =
\num{3.4e22}$ elektronen. $c/f$: \qty{6000}{km} bij \qty{50}{Hz} (huis:
ja); \qty{300}{m} bij \qty{1}{MHz} (schakeling van \qty{10}{cm}: ja);
\qty{30}{cm} bij \qty{1}{GHz}, vergelijkbaar met \qty{10}{cm}: nee.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:2}
$I = 2000/230 = \qty{8.7}{A}$; $R = U^2/P = \qty{26.5}{\ohm}$; $E =
\qty{6.0}{kWh} = \qty{2.2e7}{J}$. Snoer: één geleider $R = 1.7\times10^{-8}
\times 5.0/1.5\times10^{-6} = \qty{57}{m\ohm}$, twee geleiders
\qty{0.11}{\ohm}: $P = 0.11 \times 8.7^2 = \qty{8.6}{W}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:3}
Serie \qty{60}{\ohm}; parallel $1/(0.1 + 0.05 + 0.033) = \qty{5.5}{\ohm}$;
$10 \parallel 20 = \qty{6.7}{\ohm}$, plus $30$: \qty{36.7}{\ohm}.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:4}
$u_2 = 12 \times 2.2/6.9 = \qty{3.83}{V}$. Belast: $2.2 \parallel 2.2 =
\qty{1.1}{k\ohm}$, $u_2 = 12 \times 1.1/5.8 = \qty{2.28}{V}$: de belasting
trekt de deler $40\%$ omlaag --- een deler ``deelt'' alleen als hij
iets voedt met een veel hogere \omterm{def:b1:dc-circuits:resistor}{weerstand}.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:5}
$r = (12.6 - 11.4)/120 = \qty{10}{m\ohm}$; $I_N = 12.6/0.010 =
\qty{1260}{A}$. Naar de startmotor $11.4 \times 120 = \qty{1.37}{kW}$; verloren
$0.010 \times 120^2 = \qty{144}{W}$; rendement $1368/1512 = 90\%$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:6}
$E_{\mathrm{Th}} = \qty{3.83}{V}$ (deler in nullast); $R_{\mathrm{Th}}
= 4.7 \parallel 2.2 = \qty{1.50}{k\ohm}$ (bron kortgesloten). Belast:
$u = 3.83 \times 2.2/(1.50 + 2.2) = \qty{2.28}{V}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:7}
Superpositie: $E_1$ alleen ziet $r_2 \parallel R = \qty{1.0}{\ohm}$,
$u = 10 \times 1/(1 + 1) = \qty{5.0}{V}$; $E_2$ alleen ziet $r_1 \parallel R
= \qty{0.667}{\ohm}$, $u = 5 \times 0.667/2.667 = \qty{1.25}{V}$; samen
\qty{6.25}{V}. Millman: $(10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = \qty{6.25}{V}$.
Accu 1: $(10 - 6.25)/1 = \qty{3.75}{A}$ eruit; accu 2: $(5 - 6.25)/2
= \qty{-0.63}{A}$, dus \qty{0.63}{A} \emph{erin} --- zij wordt
geladen door de sterkere; belasting $6.25/2 = \qty{3.1}{A}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:8}
$i = 9/(R + 3)$, $P = Ri^2$, $\eta = R/(R + 3)$: $\qty{1}{\ohm}$:
\qty{5.1}{W}, $25\%$; $\qty{3}{\ohm}$: \qty{6.75}{W}, $50\%$;
$\qty{12}{\ohm}$: \qty{4.3}{W}, $80\%$. Een apparaat op batterijen wil
rendement (levensduur), dus $R \gg r$; de aangepaste belasting verspilt de helft
van de energie als warmte in de batterij.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:9}
$R = (5.0 - 2.0)/0.015 = \qty{200}{\ohm}$; $P_R = 3.0 \times 0.015 =
\qty{45}{mW}$, $P_{\mathrm{LED}} = 2.0 \times 0.015 = \qty{30}{mW}$. De
\omterm{met:b1:dc-circuits:loadline}{belastingslijn} van $(0, \qty{25}{mA})$ naar $(\qty{5}{V}, 0)$ snijdt de verticale
karakteristiek bij \qty{2.0}{V}. Bij \qty{3.3}{V} geeft dezelfde \omterm{def:b1:dc-circuits:resistor}{weerstand}
maar \qty{6.5}{mA}; voor \qty{15}{mA} is $R = 1.3/0.015 =
\qty{87}{\ohm}$ nodig.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:10}
$u = E\left[\dfrac12 - \dfrac{1 + \epsilon}{2 + \epsilon}\right] =
-E\dfrac{\epsilon}{2(2 + \epsilon)} \approx -\dfrac{E\epsilon}{4} =
\qty{-1.25}{mV}$. In een gewone deler ligt diezelfde verandering boven op een
offset van \qty{2.5}{V} (\qty{2.50125}{V}); de brug levert de
\qty{1.25}{mV} rond nul, en die kan duizendvoudig worden versterkt
zonder dat er iets in verzadiging raakt.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:11}
Door de symmetrie delen de drie buren van de intredehoek één
potentiaal en de drie buren van de uittredehoek een andere. De stroom
$I$ splitst zich in $I/3$ in de drie intredende ribben, elk daarna in twee
keer $I/6$ langs de zes middelste ribben, die weer samenkomen tot $I/3$ in de
drie uittredende ribben. \omterm{def:b1:dc-circuits:voltage}{Spanning}: $RI/3 + RI/6 + RI/3 = 5RI/6$, dus
$R_{\mathrm{eq}} = 5R/6$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:12}
Thévenin: $E = I_N r = \qty{8.0}{V}$ met \qty{4.0}{\ohm} in serie.
Maas: $8.0 - 6.0 = (4.0 + 2.0)i$, $i = \qty{0.33}{A}$ naar de accu toe.
De accu ontvangt $6.0 \times 0.33 = \qty{2.0}{W}$ (laden); de
\omterm{def:b1:dc-circuits:resistor}{weerstand} van \qty{2}{\ohm} dissipeert \qty{0.22}{W}; de klemspanning van
het Norton-paar is $8.0 - 4.0 \times 0.33 = \qty{6.67}{V}$, zodat zijn
\qty{4}{\ohm} \qty{1.67}{A} voert en \qty{11.1}{W} dissipeert, en de
\omterm{def:b1:dc-circuits:sources}{stroombron} $2.0 \times 6.67 = \qty{13.3}{W}$ levert --- wat
klopt.
\end{solution}

\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} Ideale emk $E$ in serie met $r$: $u = E - ri$.

\textbf{2.} $u(0) = \qty{12.6}{V}$; $I_N = E/r = \qty{1260}{A}$. Een sleutel
over de klemmen zou die voeren: $\qty{16}{kW}$ in een paar gram
staal --- hij smelt en spat uiteen.

\textbf{3.} \omterm{def:b1:dc-circuits:sources}{Stroombron} $I_N = \qty{1260}{A}$ parallel aan
\qty{10}{m\ohm}.

\textbf{4.} $i = E/(R + r)$; $u = ER/(R + r)$; $P_R = E^2R/(R + r)^2$;
$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$: maximum bij $R = r$,
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$; rendement
$R/(R + r) = 50\%$.

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$; $12.6 - 0.010 \times
120 = \qty{11.40}{V}$. $u$ tegen $i$ uitzetten geeft een rechte met
snijpunt $E$ en richtingscoëfficiënt $-r$ (\omterm{prop:b1:units-dimensions:regression}{kleinste kwadraten}, \cref{ch:b1:units-dimensions}).

\textbf{7.} Twee lampen: \qty{110}{W} bij \qty{12}{V}, \qty{9.2}{A};
$60/9.2 = \qty{6.5}{h}$. Energie $60 \times 3600 \times 12 = \qty{2.6}{MJ}$.

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$; parallel \qty{1.31}{\ohm}.

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$; $u = 12.6 - 1.4 = \qty{11.2}{V}$;
$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$; verloren $0.010 \times 140^2 =
\qty{196}{W}$.

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$; $i = 12.6/0.0854
= \qty{148}{A}$; $u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$; $i_h =
\qty{8.5}{A}$ voor de twee lampen.

\textbf{12.} Elke lamp: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$ tegen
\qty{55}{W}: $14\%$ minder.

\textbf{13.} Maasregel: de klemspanning is $u = E - ri$; de
\qty{140}{A} van de startmotor levert een verlies van \qty{1.4}{V} binnen in de accu,
en de lampen, parallel op dezelfde klemmen, ontvangen die verlaagde
\omterm{def:b1:dc-circuits:voltage}{spanning}.

\textbf{14.} Maar gedeeltelijk: dikkere kabels verkleinen het verlies in de kabels,
niet in $r$; lampen die vanaf de accuklemmen worden gevoed zien nog altijd $E - ri$.

\textbf{15.} Drie takken tussen dezelfde twee knooppunten: $(E, r)$,
$(E_a, r_a)$ en $R_h = \qty{1.31}{\ohm}$.

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} Dynamo $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$; accu
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (die dus \emph{erin} loopt); lampen
$12.79/1.31 = \qty{9.8}{A}$; $28.3 = 18.6 + 9.8$.

\textbf{18.} Laden. Zij ontvangt $12.79 \times 18.6 = \qty{238}{W}$
(\qty{234}{W} opgeslagen, \qty{3.5}{W} warmte); de dynamo levert
$12.79 \times 28.3 = \qty{362}{W}$ aan haar klemmen.

\textbf{19.} Alleen de accu (dynamo $\to$ \qty{50}{m\ohm}): belasting
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$, $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$. Alleen de dynamo (accu $\to$ \qty{10}{m\ohm}): belasting
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$, $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$. Samen \qty{12.79}{V}.

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$: \qty{3.60}{V} bij
\qty{20}{\degreeCelsius}, \qty{1.40}{V} bij \qty{90}{\degreeCelsius}.

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$;
$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$: $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$.

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$, nul wanneer $R_s =
\qty{775}{\ohm}$, dus bij de temperatuur midden in het bereik waar de
thermistor die waarde bereikt.

\textbf{23.} $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$ per geleider, \qty{3.2}{m\ohm} in totaal; verlies $0.0032
\times 140 = \qty{0.45}{V}$; vermogen \qty{63}{W}.

\textbf{24.} \qty{4}{mm^2}: \qty{6.4}{m\ohm} elk, \qty{12.8}{m\ohm}
totaal: verlies \qty{1.8}{V}, \qty{250}{W} verspild --- de startmotor zou
een zesde van zijn \omterm{def:b1:dc-circuits:voltage}{spanning} kwijtraken. Dikke kabel houdt $R_{\text{kabel}} \ll R_s$.

\textbf{25.} Ongeveer \qty{11.1}{V} aan de klemmen tijdens het starten; de
startmotor ontvangt $1568/(1568 + 196) \approx 89\%$ van het vermogen van de
accu; het doven is de maasregel, $u = E - ri$.
\end{solution}
