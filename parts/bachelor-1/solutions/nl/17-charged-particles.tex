\section*{Hoofdstuk \ref{ch:b1:charged-particles} --- Geladen deeltjes in E- en B-velden}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$: elektron
\qty{1.9e7}{m/s}; proton \qty{4.4e5}{m/s} ($\sqrt{1836}$ keer trager).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. Bij $30^\circ$: een schroeflijn met straal $R\sin 30^\circ
= \qty{10}{m}$ en spoed $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, die
langs de veldlijn opschuift.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Snellere ionen: de magnetische kracht
wint, zij buigen naar de magnetische kant af; de tegengestelde lading bij dezelfde
snelheid: beide krachten keren om en houden elkaar nog steeds in evenwicht --- het
\omterm{def:b1:filters-transfer-functions:H}{filter} is blind voor het teken en voor de massa.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; afbuiging
$0.30 \times 0.10 = \qty{3.0}{cm}$; gevoeligheid \qty{0.30}{mm/V}. Snellere
elektronen brengen minder tijd tussen de platen door: $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; afstand $2\Delta R =
\qty{1.2}{cm}$ tussen de aankomstpunten (de stralen schelen \qty{5.8}{mm},
de diameters twee keer zoveel).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; \qty{100}{keV} per omwenteling: $270$ omwentelingen; tijd $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
omlooptijd $2\pi m/eB = \qty{18}{ns}$, spoed $v_\parallel T = \qty{1.5}{cm}$:
ongeveer $64$ omwentelingen per meter.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: minder dan een microvolt. Met een $n$ dat tien miljoen keer
kleiner is levert een halfgeleider millivolts --- een bruikbare sensor.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; inproduct met $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ wat $\vect B(\vect r, t)$ ook is, mits er geen $\vect E$ is. De
richting van de snelheid, de \omterm{def:b1:point-kinematics:frenet}{kromtestraal} en de spoed kunnen alle
veranderen; de snelheid en de \omterm{thm:b1:work-and-energy:ke}{kinetische energie} (\qty{10}{keV}) niet.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. Met $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- een
zuivere cyclotronrotatie van $\vect w$ met $\omega_c$. Vertrekkend uit rust is
$\vect w(0) = -(E/B)\vect e_x$: $\vect w$ draait met constante norm $E/B$, dus is de
snelheid de drift $E/B$ plus een draaiende vector met dezelfde norm ---
een cycloïde, met \omterm{def:b1:work-and-energy:turning}{keerpunten} waar $\vect v = \vect 0$. Elektron: drift $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, booghoogte $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (ongeveer $15^\circ$):
afbuiging $\approx 0.30 \times 0.27 = \qty{8}{cm}$ voor een bescheiden
millitesla. Grote hoeken met platen zouden kilovolts en lange platen
vergen; een spoel buigt de bundel over tientallen graden zonder enige
\omterm{def:b1:dc-circuits:voltage}{spanning}.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: geen vermogen, constante
snelheid.

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, zonder
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ per passage, \qty{100}{keV} per omwenteling.

\textbf{5.} De spleet moet telkens wanneer het proton aankomt de versnellende
polariteit hebben, steeds een halve periode later; bij een verkeerde frequentie
loopt de fase weg en gaan de zetjes uiteindelijk afremmen.

\textbf{6.} Na één omwenteling is $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; na $n$ omwentelingen $E_k =
n \times \qty{100}{keV}$ en $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ omwentelingen; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} Nee: $E_k$ hangt alleen van $B$ en $R_{\max}$ af; $U$ bepaalt het
aantal omwentelingen en de tijd.

\textbf{11.} Weg \qty{560}{m}: nodig is $\lambda \gtrsim \qty{1}{km}$, dat wil zeggen
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
een hoog vacuüm.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; de omlooptijd groeit met $2.9\%$.
Over $270$ omwentelingen loopt de verschuiving op tot enkele perioden --- de laatste
omwentelingen zouden uit fase zijn; in de praktijk worden $B$ en de injectiefase
bijgesteld, en \qty{27}{MeV} ligt dicht bij de grens van de eenvoudige machine.

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; $R =
\qty{1.1}{m}$ bij \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ groeit met $24\%$ van het midden naar de rand: een
vaste frequentie raakt hopeloos uit de pas.

\textbf{15.} Synchrocyclotron: verlaag de frequentie naarmate het pakket energie
wint --- slechts één pakket tegelijk, lage gemiddelde stroom. Isochroon:
geef $B(R) \propto \gamma$ zo'n vorm dat de omlooptijd constant blijft --- een continue
bundel, de moderne keuze.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, iets meer dan
de niet-relativistische \qty{1.1}{m}.

\textbf{17.} Transversale impuls $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; hoek $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: hopeloos om mee te sturen.

\textbf{18.} Magnetisch: $R = p/eB = \qty{2.3}{m}$, hoek $L/R = \qty{0.43}{rad}
\approx 25^\circ$: duizend keer meer --- dus magneten.

\textbf{19.} Energie per proton $\qty{3.68e-11}{J}$; dosis $2.0 \times 0.50 =
\qty{1.0}{J}$: $2.7\times10^{10}$ protonen.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \qty{6.2e9}{protonen/s}$: ongeveer
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
een nachtlampje, precies afgeleverd waar men het hebben wil.

\textbf{22.} $\qty{58}{MeV}$ in \qty{1}{cm}: ongeveer \qty{6}{keV/\micro m} ---
dichte ionisatie langs het laatste stuk spoor.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionisaties per proton ---
de moleculaire schade die de cel doodt.

\textbf{24.} Röntgenstraling zet energie af over haar hele weg, het meeste vlak bij
de ingang; protonen zetten het grootste deel af aan het eind van hun dracht,
sparen het weefsel ervoor en laten er niets achter.

\textbf{25.} $R = mv/eB$ en $T = 2\pi m/eB$ (de cirkel), $E_k = (eBR)^2/2m$
(de energie), $E_k/2eU$ (de omwentelingen), dosis $=$ (protonen $\times$ energie)/massa
en stroom $= e \times$ tempo (de tijd).
\end{solution}
