\section*{Chapter \ref{ch:b1:newton-dynamics} --- Newton's Laws of Dynamics}

\begin{solution}{exo:b1:newton-dynamics:1}
Scale reading $N = m(g + a)$ with $a$ the upward acceleration: up,
$70 \times 11.81 = \qty{827}{N}$ (``\qty{84}{kg}''); down, $70 \times 7.81 =
\qty{547}{N}$ (``\qty{56}{kg}''); constant velocity, \qty{687}{N}
(\qty{70}{kg}); free fall, $0$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:2}
Same acceleration $a$ for both (inextensible string): $m_1a = T$,
$m_2a = m_2g - T$, so $a = m_2g/(m_1 + m_2) = \qty{3.3}{m/s^2}$,
$T = m_1a = \qty{6.5}{N}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:3}
$a = g(\sin\alpha - \mu_d\cos\alpha) = 9.81(0.50 - 0.17) = \qty{3.2}{m/s^2}$;
$v = \sqrt{2 \times 3.2 \times 5.0} = \qty{5.7}{m/s}$. No \omterm{def:b1:newton-dynamics:momentum}{mass} in sight.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:4}
$k\Delta\ell = mg$: $\Delta\ell = 0.50 \times 9.81/200 = \qty{2.5}{cm}$.
With $x$ the displacement from equilibrium, $m\ddot x = -kx$ (the weight
and the static stretch cancel): $\omega_0 = \sqrt{k/m} = \qty{20}{rad/s}$,
$T = \qty{0.31}{s}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:5}
$\mu_s = \tan 22^\circ = 0.40$. Bend: $v \leq \sqrt{\mu_sgR}$: $\sqrt{0.8
\times 9.81 \times 50} = \qty{20}{m/s}$ (\qty{71}{km/h}); wet,
\qty{14}{m/s} (\qty{50}{km/h}).
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:6}
Skydiver: $v_\infty = \sqrt{2 \times 785/(1.2 \times 0.70)} = \qty{43}{m/s}$;
$\mathrm{Re} = \rho vL/\eta \approx 1.2 \times 43 \times 0.5/1.8\times10^{-5}
\approx 10^6$: quadratic. Droplet: $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-12}{kg}$,
$v_\infty = mg/6\pi\eta r = 4.1\times10^{-11}/3.4\times10^{-9} = \qty{1.2}{cm/s}$;
$\mathrm{Re} = 1.2 \times 0.012 \times 2\times10^{-5}/1.8\times10^{-5} = 0.02$:
linear.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:7}
Bottom: $v^2 = 2g\ell(1 - \cos 60^\circ) = g\ell$; $T = mg + mv^2/\ell =
2mg$. Release point: $v = 0$, $T = mg\cos\theta_0 = 0.5mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:8}
$v = \sqrt{gR\tan\beta} = \sqrt{9.81 \times 200 \times 0.268} = \qty{23}{m/s}$
(\qty{82}{km/h}). Slower: the car tends to slide down the bank, friction
must point up-slope; faster: friction points down-slope to add
\omterm{prop:b1:newton-dynamics:centripetal}{centripetal force}.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:9}
$T\cos\alpha = mg$, $T\sin\alpha = m\omega^2\ell\sin\alpha$: $\omega =
\sqrt{g/\ell\cos\alpha} = \sqrt{9.81/0.866} = \qty{3.4}{rad/s}$, period
\qty{1.9}{s}; $T = mg/\cos\alpha = 1.15mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:10}
$a = 27.8/8.0 = \qty{3.5}{m/s^2}$, $F = \qty{3.5}{kN}$; $F \leq \mu_smg$
gives $\mu_s \geq a/g = 0.35$. On a wet road $\mu_s \approx 0.4$ caps the
acceleration near $0.4g$ whatever the engine: grip, not power.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:11}
$v = v_\infty(1 - \eu^{-t/\tau})$, $v_\infty = g\tau = \qty{4.9}{m/s}$;
$z = \int v = v_\infty[t - \tau(1 - \eu^{-t/\tau})]$. At \qty{2.0}{s}:
$4.9 \times (2.0 - 0.5 \times 0.98) = \qty{7.4}{m}$, against $\tfrac12 gt^2
= \qty{20}{m}$ in vacuum.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:12}
Radial projection (toward the center): $N - mg\cos\theta = mv^2/R$, so
$N = m(v^2/R + g\cos\theta)$. Top ($\theta = \pi$): $N = m(v^2/R - g)
\geq 0$ iff $v \geq \sqrt{gR}$. With the speed law, $v_{\text{top}}^2 =
v_0^2 - 4gR \geq gR$: $v_0 \geq \sqrt{5gR}$; then at the bottom $N =
m(5g + g) = 6mg$.
\end{solution}

\begin{solution}{pb:b1:newton-dynamics:1}
\textbf{1.} $m\dot v = mg$: $v = gt$, $z = \tfrac12 gt^2$; at \qty{10}{s}:
\qty{98}{m/s}, \qty{490}{m}.

\textbf{2.} $mg = \qty{785}{N}$.

\textbf{3.} On the ground the floor pushes up with $N = mg$ --- that push
is what one feels as weight. In free fall there is no contact \omterm{def:b1:newton-dynamics:momentum}{force}:
gravity acts on every part alike and nothing presses.

\textbf{4.} Yes to a fraction of a percent over a minute; neglected:
the Earth's rotation (centrifugal and Coriolis terms,
\cref{ch:b1:non-inertial-frames}) and the wind.

\textbf{5.} $F_d = \tfrac12\rho C_xSv^2 = 0.48\,v^2$ (SI).

\textbf{6.} $m\,\dd v/\dd t = mg - 0.48v^2$.

\textbf{7.} $v_\infty = \sqrt{mg/0.48} = \sqrt{1635} = \qty{40}{m/s} =
\qty{146}{km/h}$.

\textbf{8.} Divide by $mg$: $\dd u/\dd t\,(v_\infty/g) = 1 - u^2$;
$\tau = 40.4/9.81 = \qty{4.1}{s}$.

\textbf{9.} $\int_0^u\dd u'/(1 - u'^2) = \operatorname{artanh}u = t/\tau$,
so $u = \tanh(t/\tau)$.

\textbf{10.} $t = \tau\operatorname{artanh}(0.9) = 1.47\tau = \qty{6.1}{s}$;
$0.99$: $2.65\tau = \qty{11}{s}$.

\textbf{11.} $z = \int v_\infty\tanh(t/\tau)\dd t = v_\infty\tau\ln\cosh(t/\tau)$;
at \qty{10}{s}: $166 \times \ln\cosh(2.43) = 166 \times 1.74 = \qty{290}{m}$,
against \qty{490}{m}.

\textbf{12.} $\mathrm{Re} = 1.2 \times 40 \times 0.5/1.8\times10^{-5} \approx
\num{1.3e6}$: far above $1000$, quadratic.

\textbf{13.} $v_\infty' = \sqrt{785/(0.6 \times 25)} = \sqrt{52} = \qty{7.2}{m/s}$.

\textbf{14.} Drag $= 15 \times 1600 = \qty{24}{kN}$; $a = (24000 - 785)/80
= \qty{290}{m/s^2} \approx 30g$: the harness would break, or the jumper.

\textbf{15.} Mean $a \approx (40 - 7)/3 = \qty{11}{m/s^2} \approx 1.1g$;
harness \omterm{def:b1:newton-dynamics:momentum}{force} $\approx m(g + a) \approx \qty{1.7}{kN}$.

\textbf{16.} $\tau' = 7.2/9.81 = \qty{0.73}{s}$: within a few seconds of
full opening the descent is steady.

\textbf{17.} $\vect a = \vect 0$: harness \omterm{def:b1:newton-dynamics:momentum}{force} $= mg = \qty{785}{N}$.

\textbf{18.} Balance $mg = \tfrac12\rho C_xSv^2$ gives $v \propto (C_xS)^{-1/2}$:
half the area, $\sqrt2$ times the speed (\qty{10}{m/s}).

\textbf{19.} $v \propto \sqrt m$: $7.2 \times \sqrt{100/80} = \qty{8.1}{m/s}$.

\textbf{20.} $h = v^2/2g = 52/19.6 = \qty{2.6}{m}$: a jump from a first
floor.

\textbf{21.} $a = v^2/2d$: \qty{52}{m/s^2} ($5.3g$) over \qty{0.5}{m},
\omterm{def:b1:newton-dynamics:momentum}{force} $m(g + a) \approx \qty{5}{kN}$; \qty{17}{m/s^2} ($1.8g$) over
\qty{1.5}{m}, \qty{2.2}{kN}. Roll.

\textbf{22.} A larger drag (and some lift) momentarily exceeds the
weight: the jumper decelerates and touches down below $v_\infty'$.

\textbf{23.} $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-6}{kg}$, $S = \pi r^2 =
\qty{3.1e-6}{m^2}$: $v_\infty = \sqrt{2mg/\rho C_xS} = \sqrt{8.2\times10^{-5}/
1.9\times10^{-6}} = \qty{6.6}{m/s}$; $\mathrm{Re} = 1.2 \times 6.6 \times
2\times10^{-3}/1.8\times10^{-5} \approx 900$: quadratic (roughly).

\textbf{24.} $v_\infty = \qty{1.2}{cm/s}$ (\cref{exo:b1:newton-dynamics:6});
$\mathrm{Re} \approx 0.02 \ll 1$: Stokes applies, consistently.

\textbf{25.} $m\dot{\vect v} = m\vect g + \vect F_{\text{drag}}$; drag linear
in $v$ for small, slow things, quadratic for large, fast ones. The
numbers: \qty{40}{m/s} for a falling body, \qty{7}{m/s} under a canopy
(or a raindrop), \qty{1}{cm/s} for fog --- the same law spanning four
orders of magnitude.
\end{solution}
