\section*{Chapter \ref{ch:b1:systems-of-points} --- Systems of Points; Introduction to Rigid Bodies}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$. Earth--Moon:
$384000/82 = \qty{4700}{km}$ from the Earth's center, inside the Earth
(\qty{6370}{km}): the Earth wobbles about a point \qty{1700}{km} below
its surface.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: $v = \qty{2.0}{m/s}$. Bullet $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$, rifle \qty{8}{J}: from the powder's
chemical energy; the light body takes almost all of it ($E_k = p^2/2m$
with equal $p$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$; $E_k$: \qty{200}{kJ} before, $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$ after: $60\%$ lost.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
End: $\int_0^L x^2(M/L)\dd x = ML^2/3$; center: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$; Huygens with $d = L/2$: $ML^2/12 + ML^2/4 = ML^2/3$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: cylinder $1.5$, hoop $2$ times $\tfrac12 Mv^2$.
Same speed, more energy for the hoop: it rolls farther on a flat floor
(same friction) and climbs higher ($h = v^2(1 + k)/2g$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
HCl: $\mu = 35m_H/36 = 0.972m_H$; atom: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$. Frequencies $\propto 1/\sqrt\mu$: $+1.4\%$ for HCl, $+0.027\%$
for hydrogen (the isotope shift between H and D spectra rests on it).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$, $d = L/2$: $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$; equivalent length $2L/3 = \qty{0.67}{m}$. A blow at the
center of percussion sets the rod rotating about the pivot with no
impulsive reaction there: hit the ball at the bat's ``sweet spot'' and
the hands feel no sting.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$; $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$, $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$;
$\Gamma = J\omega/t = \qty{314}{N.m}$; turns: $\tfrac12\omega t/2\pi = 250$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: sphere ($k = 2/5$) $0.714$, cylinder ($1/2$)
$0.667$, hoop ($1$) $0.500$ times $g\sin\alpha$: sphere first, hoop last;
no mass or radius anywhere.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
Momentum: $0.010 \times 400 = 2.01\,v$: $v = \qty{2.0}{m/s}$; then energy:
$h = v^2/2g = \qty{0.20}{m}$. $E_k$: \qty{800}{J} before, $\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$ after: $99.5\%$ lost to deformation and
heat. The impact is inelastic --- energy is not conserved through it,
momentum is (the string's tension is vertical and the impact brief).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ and $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ give
$v_1 + v_1' = v_2'$ (divide the energy equation by the momentum one),
whence the formulas. Energy fraction to the target $4m_1m_2/(m_1 + m_2)^2$:
proton $100\%$; carbon $4 \times 12/169 = 28\%$; lead $4 \times 207/208^2 =
1.9\%$. Light nuclei take the neutron's energy in a few collisions
(water, graphite); lead would need hundreds.
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} On the axis, at the center of the disk, by symmetry.

\textbf{2.} Koenig: $E_k = \tfrac12 Mv_G^2 + E_k^*$, and in the \omterm{def:b1:systems-of-points:barycentric}{barycentric
frame} the yo-yo rotates about its axis: $E_k^* = \tfrac12 J\omega^2$.

\textbf{3.} Ring of radius $\rho$, mass $2M\rho\,\dd\rho/R^2$: $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$.

\textbf{4.} The string is fixed and unwinds from the axle: the point of
the axle in contact with it is momentarily at rest, so $v_G = r\omega$.

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$; $R^2/2r^2 = 9/(2 \times 0.16) = 28$: factor $29$ ---
almost all the energy is in the spin.

\textbf{6.} Weight $Mg$ downward, tension $T$ upward: $Ma_G = Mg - T$.

\textbf{7.} Only the tension has a moment about the axis (\omterm{def:b1:angular-momentum:moment}{lever arm} $r$):
$J\dot\omega = Tr$.

\textbf{8.} $\dot\omega = a_G/r$: $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$; then
$Ma_G = Mg - (R^2/2r^2)Ma_G$, $a_G = g/29 = \qty{0.34}{m/s^2}$.

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$, $97\%$ of
the weight: the string nearly holds it.

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$; $v_G = a_Gt = \qty{0.82}{m/s}$.
A stone: \qty{0.45}{s} and \qty{4.4}{m/s}.

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$.

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ against $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: $28$ times more energy in the spin than in the fall.

\textbf{13.} $Mgh = \qty{0.49}{J}$; $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$. The tension acts at the point of the
string that is at rest (the string does not move): zero power.

\textbf{14.} The stopping force acts along the string, through the axis:
no moment about it; the \omterm{def:b1:angular-momentum:L}{angular momentum} $J\omega$ is untouched. Only a
tangential force (friction on the axle) could slow the spin.

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$, eighteen times the weight --- hence the jerk felt by the
finger.

\textbf{16.} $J\dot\omega = -\Gamma$: $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$.

\textbf{17.} All the spin energy $\tfrac12 J\omega^2 = \qty{0.47}{J}$ converts
to $Mgh'$: $h' = \qty{0.96}{m}$, nearly the full height; less in practice
because of the axle friction during the sleep and the losses in the
jerk.

\textbf{18.} $a_G$ is downward (the yo-yo decelerates on the way up) while
$v_G$ is upward: $T = M(g - a_G) < Mg$ --- smaller than the weight.

\textbf{19.} $r = R$: factor $1 + 1/2 = 1.5$, $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- a fast drop, little spin: a thin axle is what makes a yo-yo.

\textbf{20.} \omterm{ex:b1:systems-of-points:rolling}{Rolling without slipping}: the material point at $C$ has zero
velocity; the velocity field of a \omterm{def:b1:systems-of-points:solid}{rigid body} with one point at rest is
a rotation about that point (axis through $C$, same $\omega$).

\textbf{21.} $F$ acts at the bottom of the axle, a height $R - r$ above
the table, horizontally toward the hand: moment $F(R - r)$ about $C$ in
the sense that rolls the spool toward the hand; the weight and the
reaction pass through $C$ (no moment).

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$, $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: toward the hand.

\textbf{23.} The line of action of $F$ passes through $C$ when the string
from the bottom of the axle meets the table at $C$: $\cos\beta = r/R$,
$\beta = 82^\circ$. Steeper: the moment about $C$ reverses, the spool rolls
away (and for $\beta$ large enough lifts).

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$; friction from $Ma_G = F - f$ (friction opposes $F$):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$; it must stay below $\mu_sMg
\approx \qty{0.2}{N}$ (for $\mu_s = 0.4$), i.e.\ $F \lesssim \qty{0.5}{N}$.

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{Center of mass}: the translation of $G$ and the tension;
\omterm{def:b1:angular-momentum:L}{angular momentum} about an axis: the spin and the rolling direction;
energy: the speed at the bottom and the height of the climb.
\end{solution}
