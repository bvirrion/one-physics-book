\section*{Chapter \ref{ch:b1:oscillators-resonance} --- Mechanical Oscillators: Damping and Resonance}

\begin{solution}{exo:b1:oscillators-resonance:1}
$\omega_0 = \sqrt{80/0.20} = \qty{20}{rad/s}$, $f_0 = \qty{3.2}{Hz}$, $T_0 =
\qty{0.31}{s}$; $v_{\max} = A\omega_0 = \qty{1.0}{m/s}$; $E = \tfrac12 kA^2 =
\qty{0.10}{J}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:2}
$T = 2\pi\sqrt{\ell/g}$: \qty{2.0}{s} on Earth, \qty{4.9}{s} on the Moon.
The mass--spring period, $2\pi\sqrt{m/k}$, does not involve $g$:
unchanged.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:3}
$\tau = 2Q/\omega_0 = 100/(4\pi) = \qty{8.0}{s}$; energy $\propto \eu^{-2t/\tau}$
halves at $t = \tau\ln2/2 = \qty{2.8}{s}$; $5\%$ after $3\tau = \qty{24}{s}$,
about $48 \approx Q$ oscillations.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:4}
$X_0 = F_0/k = \qty{1.0}{cm}$; at resonance $\approx QX_0 = \qty{10}{cm}$;
at $3\omega_0$: $X_0/\sqrt{64 + 0.09} = \qty{0.13}{cm}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:5}
$v = Lf_0 = 8.0 \times 1.3 = \qty{10}{m/s}$ (\qty{37}{km/h}). At
\qty{60}{km/h}: $f = 16.7/8.0 = \qty{2.1}{Hz}$, $x_r = 1.6$: $X/X_g =
\sqrt{(1 + 5.2)/(2.4 + 5.2)} = 0.90$ --- barely isolated; isolation
needs $x_r \gg 1$ (at \qty{120}{km/h}, $x_r = 3.2$: $0.45$).
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:6}
The work of a push is positive only when it acts along the motion;
pushed every period, all pushes add; at any other rhythm they
alternate. \omterm{def:b1:transient-regimes:firstorder}{Steady state}: $2\pi E/Q = \qty{25}{J}$, $E = \qty{80}{J}$;
$mg\ell(1 - \cos\theta_{\max}) = 80$: $1 - \cos\theta_{\max} = 0.109$,
$\theta_{\max} = 27^\circ$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:7}
$\delta = \ln(1/0.9) = 0.105$, $Q = \pi/\delta = 30$; $\Delta f = f_0/Q =
\qty{0.067}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:8}
Displacement: in phase ($\varphi = 0$) well below, $-\pi/2$ at
resonance, $-\pi$ (opposite) well above. Velocity at resonance: in
phase with the force. Power is $\vect F\cdot\vect v$: it is the velocity's
phase, not the displacement's, that decides how much work the drive
does.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:9}
$V = \omega X = \omega(F_0/m)/\sqrt{(\omega_0^2 - \omega^2)^2 + (\omega\omega_0/Q)^2}$;
divide numerator and denominator by $\omega\omega_0/Q$: $V = (F_0Q/m\omega_0)/
\sqrt{1 + Q^2(\omega_0/\omega - \omega/\omega_0)^2}$, and $F_0Q/m\omega_0 = F_0/\alpha$.
The square root is $\geq 1$, equal to $1$ exactly at $\omega = \omega_0$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:10}
$P = F_0^2/2\alpha = \qty{10}{W}$; $V_{\max} = F_0/\alpha = \qty{20}{m/s}$,
$E = \tfrac12 mV_{\max}^2 = \qty{40}{J}$; $E/P = \qty{4.0}{s}$ and $Q/\omega_0 =
m/\alpha = \qty{4.0}{s}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:11}
$3\tau = \qty{10}{s}$: $\tau = \qty{3.3}{s}$, $Q = \omega_0\tau/2 = 2\pi \times
440 \times 3.3/2 \approx 4600$; $\Delta f = f_0/Q = \qty{0.1}{Hz}$. It
responds to, and emits, essentially one frequency: a standard good to
a few parts in $10^4$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:12}
$m\ddot x = mg - \rho_wgS(h + x) = -\rho_wgSx$ (since $mg = \rho_wgSh$):
$\omega_0^2 = \rho_wgS/m = g/h$; $T = 2\pi\sqrt{0.10/9.81} = \qty{0.63}{s}$.
Both the mass and the liquid's density enter only through the
equilibrium depth $h$.
\end{solution}

\begin{solution}{pb:b1:oscillators-resonance:1}
\textbf{1.} Forces: weight (balanced by the static stretch), spring
$-ku$, damper $-\alpha\dot u$; $m\ddot x = -ku - \alpha\dot u$ with $\ddot x =
\ddot x_g + \ddot u$.

\textbf{2.} $\ddot u + (\omega_0/Q)\dot u + \omega_0^2u = -\ddot x_g$, $\omega_0 =
\sqrt{k/m}$, $Q = \sqrt{km}/\alpha$.

\textbf{3.} $k = m\omega_0^2 = \pi^2 = \qty{9.9}{N/m}$; $\alpha = m\omega_0/Q =
3.14/0.70 = \qty{4.5}{N.s/m}$.

\textbf{4.} $u = -a_g/\omega_0^2$: the mass sags by a fixed amount --- the
instrument is an accelerometer at zero frequency (and a tilt meter).

\textbf{5.} The relative displacement $u(t)$.

\textbf{6.} $-\omega^2\underline U + j\omega(\omega_0/Q)\underline U + \omega_0^2\underline U
= \omega^2X_g$; divide by $\omega_0^2$.

\textbf{7.} $U/X_g = x_r^2/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$.

\textbf{8.} $U \to X_g$: the mass does not move in the \omterm{thm:b1:newton-dynamics:laws}{inertial frame}
(its inertia keeps it still while the spring is too soft to drag it);
the frame moves around it, and the pen records the ground displacement
--- a seismometer.

\textbf{9.} $U \approx x_r^2X_g = \omega^2X_g/\omega_0^2 = a_g/\omega_0^2$: an
accelerometer.

\textbf{10.} $U = QX_g = 0.7X_g$: no peak; the two regimes join smoothly
and the instrument rings not.

\textbf{11.} For $Q = 0.7$: a line of slope $2$ rising to $1$ near
$x_r = 1$, then flat at $1$; for $Q = 5$: the same with a peak of height
$5$ at $x_r = 1$.

\textbf{12.} $\underline U \to -X_g$: $u = -x_g$, the mass still while the
frame moves.

\textbf{13.} $\Delta\ell = g/\omega_0^2 = 9.81/(2\pi \times 0.02)^2 = \qty{620}{m}$:
absurd.

\textbf{14.} $\ell = g/\omega_0^2 = \qty{620}{m}$.

\textbf{15.} Garden gate: the pendulum swings about a nearly vertical
axis, so only a tiny fraction of $g$ restores it and a short arm gives
a long period. Force feedback: a coil holds the mass fixed, and the
current needed is proportional to the ground acceleration --- the
stiffness is electronic and adjustable.

\textbf{16.} $\Delta\ell = 9.81/\pi^2 = \qty{0.99}{m}$: feasible.

\textbf{17.} $x_r = 0.2$: $U/X_g = 0.04/\sqrt{0.92 + 0.08} = 0.040$:
\qty{40}{\micro m} --- accelerometer regime, needs amplification (optical
lever or electronics).

\textbf{18.} $x_r = 10$: $100/\sqrt{9801 + 204} \approx 1.00$: \qty{1.0}{mm},
faithful displacement record.

\textbf{19.} $\omega_0 = \qty{6.3e3}{rad/s}$: $U = g/\omega_0^2 = \qty{2.5e-7}{m}$
--- read as the change of a capacitor gap.

\textbf{20.} $U/(a_g/\omega_0^2) = 1/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$; with
$Q = 0.7$ it decreases monotonically and stays above $0.9$ up to
$x_r \approx 0.7$: about \qty{700}{Hz}.

\textbf{21.} $\tau = 2Q/\omega_0 = \qty{0.22}{ms}$: it tracks a fall (tens
of milliseconds) without lag.

\textbf{22.} $\tau = 1.4/3.14 = \qty{0.45}{s}$: after a ground step the
record settles in half a second with no ringing.

\textbf{23.} A high $Q$ would make the instrument ring at $f_0$ after
every jolt (recording itself rather than the ground) and peak near
$f_0$; $Q \approx 0.7$ is flat and quick.

\textbf{24.} $U = QX_g = 0.7X_g$ against the ideal $X_g$ (or $a_g/\omega_0^2
= X_g$): $30\%$ low.

\textbf{25.} Compare the \omterm{def:b1:wave-propagation:signal}{signal} frequency with $f_0$: well above, the
mass is a fixed reference and $u$ is the displacement; well below, $u$
is the acceleration over $\omega_0^2$.
\end{solution}
