\section*{الفصل \ref{ch:b1:work-and-energy} --- الشغل والطاقة والطاقة الكامنة}

\begin{solution}{exo:b1:work-and-energy:1}
قوة الرفع $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$؛ والوزن
$-\qty{589}{J}$؛ والاستطاعة $589/5.0 = \qty{118}{W}$. والاحتكاك: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$؛ و $d = E_k/F =
\qty{55}{m}$؛ ومضاعفة $v$ تضاعف $E_k$ و $d$ أربع مرات: أي \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$؛ و $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$: ومنه $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
والشدّ: $T - mg = mv^2/\ell = mg$، ومنه $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
\omterm{def:b1:work-and-energy:work}{شغل} الاحتكاك $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$، ومنه
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$؛ أي إن $35\%$ ضاعت.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
يجب أن يكون $E_m = \tfrac12 mv^2 - GMm/R \geq 0$ حتى يبلغ الجسم اللانهاية
(حيث $E_p = 0$ و $v \to 0$): ومنه $v_e = \sqrt{2GM/R}$، أي
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
وبالكثافة نفسها، $M \propto R^3$: ومنه $v_e \propto R$، أي \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
في النقطة الدنيا: $mg(\ell_0 + x) = \tfrac12 kx^2$، أي $25x^2 - 687x - 13734 = 0$،
ومنه $x = \qty{41}{m}$؛ والهبوط الكلي \qty{61}{m}؛ و $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (أي محصلة \qty{2}{g} نحو الأعلى). وتكون السرعة أكبر ما تكون حيث تنعدم
القوة المحصلة، $kx = mg$: أي $x = \qty{14}{m}$، على بعد \qty{34}{m} تحت نقطة القفز.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
الثقالة: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$؛
والسحب: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$؛ والمجموع \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$: ومنه $x = 0, \pm a$. و $E_p'' = E_0(12x^2/a^4 -
4/a^2)$: فهو $-4E_0/a^2$ عند $0$ (نهاية عظمى غير مستقرة، $E_p = 0$)، وهو $+8E_0/a^2$
عند $\pm a$ (نهايتان صغريان مستقرتان، $E_p = -E_0$): أي بئران عمقهما $E_0$؛
و $\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$: أي إهليلج نصفا محورَيه $x_{\max}
= \sqrt{2E/k}$ و $\dot x_{\max} = \sqrt{2E/m}$. \omterm{prop:b1:point-kinematics:harmonic}{والحركة توافقية}
دورها $2\pi\sqrt{m/k}$ أيًّا كانت السعة (تساوي الأدوار)، ولذلك
يستغرق كل إهليلج الزمن نفسه.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
بالطاقة: $v^2 = 2gR(1 - \cos\theta)$. وعلى المحور الناظمي: $mg\cos\theta - N = mv^2/R$،
ومنه $N = mg(3\cos\theta - 2)$، وهو معدوم عند $\cos\theta = 2/3$ (أي $\theta =
48^\circ$): فالارتفاع $2R/3$ والسرعة $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$: ومنه $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
وقد صارت الطاقة حرارةً في تماسّ الإطار مع الطريق (وأثرَ انزلاق). أما العجلات
المتدحرجة فتستعمل الاحتكاك السكوني، $\mu_s > \mu_d$: أي قوة كبح أكبر ومسافة
أقصر، مع الاحتفاظ بالتوجيه.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} $r \to 0$: $+\infty$ (بسبب الحدّ $r^{-12}$)؛ و $r \to \infty$:
$0^-$. أي جدار حادّ وبئر وذيل طويل.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ إذا وفقط إذا كان
$(\sigma/r)^6 = 1$: أي $r = \sigma$ و $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$: ومنه $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$، وهي أصغرية عند $u = 1$ (أي $r = \sigma$) بالقيمة $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$:
موجبة (دافعة) من أجل $r < \sigma$، وسالبة (جاذبة) بعده.

\textbf{5.} $(1/1.2)^6 = 0.335$ و $(1/1.2)^{12} = 0.112$: ومنه $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. و $(1/1.2)^{13} = 0.0935$ و
$(1/1.2)^7 = 0.279$: ومنه $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$،
وهي قريبة من القيمة المقيسة \qty{431}{kJ/mol}.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$؛ وعند
$\sigma$: $72\epsilon/\sigma^2 > 0$: أي إن التوازن مستقرّ.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$: ومنه
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$؛ و $f = \qty{2.2e14}{Hz}$؛ و $\lambda = c/f = \qty{1.4}{\micro m}$؛
و $1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} أعلى بعامل $2.4$: فجدارا النموذج شديدا الانحدار
(إذ إن اللبّ $r^{-12}$ بديل خشن عن رابطة تساهمية)، لكن
\omterm{def:b1:units-dimensions:oom}{رتبة القدر} --- وهي اهتزاز في تحت الأحمر عند $10^{14}$ هرتز --- صحيحة،
والطريقة (أي تقوّس البئر) هي بعينها الطريقة المستعملة
مع كمونات أفضل.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$: ومنه $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$: أي إن الرابطة
تكاد لا ترتعش.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$: منحنيات مغلقة حول $(\sigma, 0)$ ---
أي اهتزاز \omterm{def:b1:work-and-energy:turning}{وحركة مرتبطة}؛ و $E_m > 0$: منحنيات مفتوحة --- أي انفصال الذرّتين؛
و $E_m = 0$: المنحني الفاصل، حيث تبلغ الذرة اللانهاية بالكاد.

\textbf{15.} $u^2 - 2u = -\tfrac12$ مع $u = (\sigma/r)^6$: ومنه $u = 1 \pm 1/\sqrt2
= 1.71$ أو $0.293$؛ ومن $r = \sigma u^{-1/6}$ نجد $0.915\sigma$ و $1.23\sigma$.

\textbf{16.} المنتصف $1.07\sigma > \sigma$: فالجدار الخارجي ألين من
الداخلي، ولذلك تجد الذرة حيّزًا أوسع في الخارج؛ والمسافة المتوسطة
زمنيًّا تكبر مع الطاقة --- فالروابط المسخّنة تطول، والأجسام الصلبة تتمدد.

\textbf{17.} أطول: لأن القوة المعيدة على الجانب الخارجي اللين
أضعف مما يفترضه التقريب التوافقي، فتستغرق العودة
زمنًا أطول.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$؛ وإذا قُدّمت طاقةً
حركية عند $\sigma$: $v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}
= \qty{2.0e4}{m/s}$.

\textbf{19.} بانحفاظ الطاقة: ما كان فيه $E_m > 0$ يبقى $> 0$، فالمنحني
مفتوح، وتتباعد الذرّتان بعد لقاء واحد. ولا بدّ من جسم ثالث
(جزيء آخر أو جدار أو فوتون) يحمل الفائض بعيدًا.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$، وهو موجب: فالجذب يسحب
الذرة إلى الداخل، أي إن القوة محرّكة.

\textbf{21.} $W_{\text{خارجي}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$ و $(1/0.9)^7 = 2.09$: ومنه $F = +22\epsilon/\sigma
= \qty{+120}{nN}$، أي عشرة أضعاف الجذب عند $1.2\sigma$: فضغط
رابطة بنسبة $10\%$ يكلّف أكثر بكثير من مطّها بنسبة $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. والغازات الحقيقية تتفكك دون ذلك بكثير لأن الطاقات
موزَّعة: فجزء من الجزيئات يحمل أضعاف $k_BT$
(\cref{ch:b1:kinetic-theory}).

\textbf{24.} تبقى: التوازن عند النهاية الصغرى، و $k = E_p''$ عند
النهاية الصغرى، وطاقة التفكك $=$ العمق، \omterm{def:b1:work-and-energy:turning}{ونقاط الرجوع} من $E_p = E_m$،
وعدم التناظر $\Rightarrow$ التمدد. ويتغير: قيمة $k$ وقيمة
التواتر وقيم \omterm{def:b1:work-and-energy:turning}{نقاط الرجوع}.

\textbf{25.} التوازن: $E_p' = 0$؛ والاهتزاز: $\omega_0^2 = E_p''/m$؛
والإفلات: مقارنة $E_m$ بحافة البئر.
\end{solution}
