\begin{solution}{exo:b1:dc-circuits:6}

\begin{solution}{exo:b1:dc-circuits:7}
بالتراكب: يرى $E_1$ وحده $r_2 \parallel R = \qty{1.0}{\ohm}$، ومنه
$u = 10 \times 1/(1 + 1) = \qty{5.0}{V}$؛ ويرى $E_2$ وحده $r_1 \parallel R
= \qty{0.667}{\ohm}$، ومنه $u = 5 \times 0.667/2.667 = \qty{1.25}{V}$؛ والمجموع
\qty{6.25}{V}. وبميلمان: $(10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = \qty{6.25}{V}$.
والبطارية 1: $(10 - 6.25)/1 = \qty{3.75}{A}$ خارجة؛ والبطارية 2: $(5 - 6.25)/2
= \qty{-0.63}{A}$، أي \qty{0.63}{A} \emph{داخلة} إليها --- فهي تُشحن
من الأقوى؛ والحمولة $6.25/2 = \qty{3.1}{A}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:8}
$i = 9/(R + 3)$ و $P = Ri^2$ و $\eta = R/(R + 3)$: فمن أجل $\qty{1}{\ohm}$:
\qty{5.1}{W} و$25\%$؛ ومن أجل $\qty{3}{\ohm}$: \qty{6.75}{W} و$50\%$؛
ومن أجل $\qty{12}{\ohm}$: \qty{4.3}{W} و$80\%$. والجهاز الذي يعمل ببطارية يريد
المردود (أي عمر البطارية)، ومن ثمّ $R \gg r$؛ أما الحمولة المكيَّفة فتهدر نصف
الطاقة حرارةً في البطارية.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:9}

\end{solution}

\begin{solution}{exo:b1:dc-circuits:10}
$u = E\left[\dfrac12 - \dfrac{1 + \epsilon}{2 + \epsilon}\right] =
-E\dfrac{\epsilon}{2(2 + \epsilon)} \approx -\dfrac{E\epsilon}{4} =
\qty{-1.25}{mV}$. وفي مجزّئ بسيط يقع التغيّر نفسه فوق إزاحة
\qty{2.5}{V} (أي \qty{2.50125}{V})؛ أما الجسر فيعطي
\qty{1.25}{mV} حول الصفر، ويمكن تضخيمها ألف مرة
دون إشباع أيّ شيء.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:11}
بالتناظر يشترك جيران زاوية الدخول الثلاثة في كمون
واحد، وجيران زاوية الخروج الثلاثة في كمون آخر. فينقسم التيار
$I$ إلى $I/3$ في أحرف الدخول الثلاثة، ثم ينقسم كل منها إلى
$I/6$ على امتداد الأحرف الوسطى الستة، ثم تتجمّع في $I/3$ في
أحرف الخروج الثلاثة. \omterm{def:b1:dc-circuits:voltage}{والتوتر}: $RI/3 + RI/6 + RI/3 = 5RI/6$، ومنه
$R_{\mathrm{eq}} = 5R/6$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:12}
بتيفنان: $E = I_N r = \qty{8.0}{V}$ مع \qty{4.0}{\ohm} على التسلسل.
وبقانون الحلقات: $8.0 - 6.0 = (4.0 + 2.0)i$، ومنه $i = \qty{0.33}{A}$ نحو البطارية.
فتتلقى البطارية $6.0 \times 0.33 = \qty{2.0}{W}$ (أي تُشحن)؛ وتبدد
الناقلة \qty{2}{\ohm} مقدار \qty{0.22}{W}؛ \omterm{def:b1:dc-circuits:voltage}{وتوتر} طرفَي
زوج نورتن هو $8.0 - 4.0 \times 0.33 = \qty{6.67}{V}$، ومنه تحمل
\qty{4}{\ohm} تيارًا قدره \qty{1.67}{A} وتبدد \qty{11.1}{W}، ويقدّم
\omterm{def:b1:dc-circuits:sources}{منبع التيار} $2.0 \times 6.67 = \qty{13.3}{W}$ --- والحصيلة
متوازنة.
\end{solution}

\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} قوة محركة مثالية $E$ على التسلسل مع $r$: أي $u = E - ri$.

\textbf{2.} $u(0) = \qty{12.6}{V}$؛ و$I_N = E/r = \qty{1260}{A}$. ومفتاح ربط
يصل بين الطرفين سيحمله: أي $\qty{16}{kW}$ في بضعة غرامات من
الفولاذ --- فينصهر ويتطاير.

\textbf{3.} \omterm{def:b1:dc-circuits:sources}{منبع تيار} $I_N = \qty{1260}{A}$ على التفرع مع
\qty{10}{m\ohm}.

\textbf{4.} $i = E/(R + r)$؛ و$u = ER/(R + r)$؛ و$P_R = E^2R/(R + r)^2$؛
و$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$: فالعظمى عند $R = r$، و
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$؛ والمردود
$R/(R + r) = 50\%$.

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$؛ و$12.6 - 0.010 \times
120 = \qty{11.40}{V}$. ورسم $u$ بدلالة $i$ يعطي مستقيمًا
مقطعه $E$ وميله $-r$ (\omterm{prop:b1:units-dimensions:regression}{بالمربعات الصغرى}، \cref{ch:b1:units-dimensions}).

\textbf{7.} المصباحان: \qty{110}{W} عند \qty{12}{V}، أي \qty{9.2}{A}؛
ومنه $60/9.2 = \qty{6.5}{h}$. والطاقة $60 \times 3600 \times 12 = \qty{2.6}{MJ}$.

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$؛ وعلى التفرع \qty{1.31}{\ohm}.

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$؛ و$u = 12.6 - 1.4 = \qty{11.2}{V}$؛
و$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$؛ والضائع $0.010 \times 140^2 =
\qty{196}{W}$.

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$؛ و$i = 12.6/0.0854
= \qty{148}{A}$؛ و$u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$؛ و$i_h =
\qty{8.5}{A}$ للمصباحين معًا.

\textbf{12.} كل مصباح: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$ في مقابل
\qty{55}{W}: أي خفوت بمقدار $14\%$.

\textbf{13.} بقانون الحلقات: \omterm{def:b1:dc-circuits:voltage}{توتر} الطرفين هو $u = E - ri$؛ فتيار
المقلع البالغ \qty{140}{A} يُحدث هبوطًا قدره \qty{1.4}{V} داخل البطارية،
والمصباحان، وهما على التفرع على الطرفين نفسيهما، يتلقيان ذلك \omterm{def:b1:dc-circuits:voltage}{التوتر}
المخفَّض.

\textbf{14.} جزئيًّا فقط: فالكبلات الأغلظ تخفض الهبوط في الكبلات،
لا في $r$؛ والمصباحان المغذَّيان من طرفَي البطارية يريان $E - ri$ دائمًا.

\textbf{15.} ثلاثة فروع بين العقدتين نفسيهما: $(E, r)$ و
$(E_a, r_a)$ و $R_h = \qty{1.31}{\ohm}$.

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} المولّد $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$؛ والبطارية
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (أي داخلًا \emph{إليها})؛ والمصباحان
$12.79/1.31 = \qty{9.8}{A}$؛ و$28.3 = 18.6 + 9.8$.

\textbf{18.} تُشحن. وتتلقى $12.79 \times 18.6 = \qty{238}{W}$
(منها \qty{234}{W} مخزَّنة و\qty{3.5}{W} حرارة)؛ ويقدّم المولّد
$12.79 \times 28.3 = \qty{362}{W}$ عند طرفيه.

\textbf{19.} البطارية وحدها (والمولّد $\to$ \qty{50}{m\ohm}): الحمولة
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$، ومنه $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$. والمولّد وحده (والبطارية $\to$ \qty{10}{m\ohm}): الحمولة
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$، ومنه $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$. والمجموع \qty{12.79}{V}.

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$: أي \qty{3.60}{V} عند
\qty{20}{\degreeCelsius}، و\qty{1.40}{V} عند \qty{90}{\degreeCelsius}.

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$؛
و$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$: ومنه $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$.

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$، وهو معدوم حين $R_s =
\qty{775}{\ohm}$، أي عند درجة الحرارة الوسطى التي تبلغ عندها
المقاومة الحرارية تلك القيمة.

\textbf{23.} $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$ لكل ناقل، أي \qty{3.2}{m\ohm} إجمالًا؛ والهبوط $0.0032
\times 140 = \qty{0.45}{V}$؛ والاستطاعة \qty{63}{W}.

\textbf{24.} ومع \qty{4}{mm^2}: \qty{6.4}{m\ohm} لكل ناقل، أي \qty{12.8}{m\ohm}
إجمالًا: فالهبوط \qty{1.8}{V} و\qty{250}{W} مهدورة --- وكان المقلع سيفقد
سُدس توتره. فالكبل الغليظ يبقي $R_{\text{الكبل}} \ll R_s$.

\textbf{25.} نحو \qty{11.1}{V} عند الطرفين أثناء الإقلاع؛ ويتلقى
المقلع $1568/(1568 + 196) \approx 89\%$ من استطاعة البطارية؛
والخفوت هو قانون الحلقات، $u = E - ri$.
\end{solution}
