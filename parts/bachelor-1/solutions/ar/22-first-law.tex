\begin{solution}{exo:b1:first-law:1}
$Q = C_{P,m}\Delta T = 29.1 \times 100 = \qty{2.91}{kJ}$؛ و $W = -nR\Delta T =
\qty{-0.83}{kJ}$؛ و $\Delta U = Q + W = \qty{2.08}{kJ} = C_{V,m}\Delta T$.
\end{solution}

\begin{solution}{exo:b1:first-law:2}

\end{solution}

\end{solution}

\begin{solution}{exo:b1:first-law:4}
الأرغون: $C_{V,m} = \tfrac32 R = 12.5$ و $C_{P,m} = \tfrac52 R = 20.8$ جول لكل \omterm{def:b1:kinetic-theory:scales}{مول} \omterm{def:b1:kinetic-theory:temperature}{وكلفن}،
و $\gamma = 1.67$؛ والآزوت: $20.8$ ثم $29.1$ و $\gamma = 1.40$. أما الماء فتمدده
بالتسخين ضئيل جدًّا، فيكون الشغل $P\,\dd V$ عند ثبات الضغط
مهمَلًا أمام \omterm{def:b1:first-law:heat}{الحرارة}: $c_P - c_V \approx 0$.
\end{solution}

\begin{solution}{exo:b1:first-law:5}
$T_f = (200 \times 80 + 300 \times 20)/500 = \qty{44}{\degreeCelsius}$. ومع
الكأس: $(200 \times 4.18 \times 80 + (300 \times 4.18 + 80) \times 20)/(500 \times
4.18 + 80) = \qty{43}{\degreeCelsius}$.
\end{solution}

\begin{solution}{exo:b1:first-law:6}
الجليد: تسخينه إلى $0$: $0.050 \times 2100 \times 10 = \qty{1.05}{kJ}$؛ وانصهاره:
\qty{16.7}{kJ}؛ فالمجموع \qty{17.8}{kJ}. ويستطيع الماء أن يعطي \qty{25.1}{kJ} نزولًا إلى
$0$: فينصهر كله؛ ثم $0.250 \times 4180\,T_f = 25.1 - 17.8$ كيلوجول، ومنه $T_f =
\qty{7}{\degreeCelsius}$. ومع \qty{150}{g}: يلزم \qty{53.3}{kJ} وهو $>$ \qty{25.1}{kJ}:
فلا ينصهر إلا $(25.1 - 3.15)/334 = \qty{66}{g}$ من الجليد، ويكون $T_f = \qty{0}{\degreeCelsius}$،
ويبقى \qty{84}{g} من الجليد.
\end{solution}

\begin{solution}{exo:b1:first-law:7}
$W = -P_1(V_2 - V_1) + 0 - P_2(V_1 - V_2) + 0 = (P_2 - P_1)(V_2 - V_1) > 0$:
فالدور يُقطع بعكس عقارب الساعة (تمدد عند ضغط منخفض،
وانضغاط عند ضغط مرتفع)، ومن ثمّ \emph{يتلقى} الغاز شغلًا صافيًا يساوي
المساحة المحصورة؛ ولو قُطع مع عقارب الساعة لقدّمه --- فكان محركًا.
\end{solution}

\begin{solution}{exo:b1:first-law:8}
$T_2 = 300 \times 7^{0.286} = \qty{523}{K}$. والهواء المضغوط الساخن يسخّن
جسم المنفاخ بالتوصيل في كل دفعة. وفي الإطار يبرد الهواء إلى
\omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} المحيط عند ثبات الحجم: فيكون $P \propto T$، ويهبط الضغط بنسبة
درجتَي \omterm{def:b1:first-law:heat}{الحرارة} --- فأعِد نفخه قليلًا.
\end{solution}

\begin{solution}{exo:b1:first-law:9}
$W = 0$ (إذ لا ضغط خارجي يدفع الغاز في مواجهته)، و $Q = 0$: ومنه $\Delta U = 0$،
ومن ثمّ $\Delta T = 0$ \omterm{def:b1:kinetic-theory:model}{لغاز مثالي}. \omterm{rem:b1:first-law:irreversible}{والتحول لاعكوس}: فالغاز لا يعود
أبدًا من تلقائه. أما الغاز الحقيقي، وجزيئاته يجذب بعضها بعضًا، فيبرد قليلًا
(إذ يصير جزء من \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} طاقةً كامنة وهي تتباعد).
\end{solution}

\begin{solution}{exo:b1:first-law:10}
$Q = L_v = \qty{2.26}{MJ}$؛ و $W = -P(V_g - V_l) = -1.013\times10^5 \times 1.67 =
\qty{-0.17}{MJ}$؛ و $\Delta U = \qty{2.09}{MJ}$: فيذهب $93\%$ من \omterm{def:b1:first-law:heat}{الحرارة} إلى
تفريق الجزيئات (أي إلى \omterm{def:b1:kinetic-theory:U}{الطاقة الداخلية})، و $7\%$ إلى دفع
الجوّ إلى الوراء.
\end{solution}

\begin{solution}{exo:b1:first-law:11}
$W = -P_2(V_2 - V_1)$ و $\Delta U = nC_{V,m}(T_2 - T_1)$ و $V_2 = nRT_2/P_2$ و $V_1
= nRT_1/P_1$: ومنه $\tfrac52(T_2 - 300) = -T_2 + 300 \times 5$ و $3.5T_2 = 2250$
و $T_2 = \qty{643}{K}$ و $V_2 = \qty{10.7}{L}$. أما العكوس: $T_2 = 300 \times
5^{0.286} = \qty{475}{K}$ و $V_2 = \qty{7.9}{L}$. فالانضغاط المفاجئ يؤدي على
الغاز شغلًا أكبر (إذ تؤثر \qty{5}{bar} كاملةً منذ البداية) وينتهي
أسخن وأقل انضغاطًا.
\end{solution}

\begin{solution}{exo:b1:first-law:12}
$\rho = PM/RT = \qty{1.20}{kg/m^3}$: ومنه $\sqrt{\gamma P/\rho} = \sqrt{1.4 \times
1.013\times10^5/1.20} = \qty{343}{m/s}$؛ و $\sqrt{P/\rho} = \qty{290}{m/s}$، أي أدنى بنسبة $15\%$.
ومع $P/\rho = RT/M$ يكون $c = \sqrt{\gamma RT/M}$؛ ومقابل $u = \sqrt{3RT/M}$
= \qty{502}{m/s}: يكون $c/u = \sqrt{\gamma/3} = 0.68$.
\end{solution}

\begin{solution}{pb:b1:first-law:1}
\textbf{1.} $n = P_1V_1/RT_1 = 10^5 \times 2.12\times10^{-4}/2494 = \qty{8.5e-3}{mol}$.

\textbf{2.} $PV^\gamma$ ثابت: ومنه $V_2 = V_1(P_1/P_2)^{1/\gamma} = 212 \times
7^{-0.714} = \qty{53}{cm^3}$.

\textbf{3.} $T_2 = T_1(P_2/P_1)^{(\gamma - 1)/\gamma} = 300 \times 7^{0.286} =
\qty{523}{K}$.

\textbf{4.} $W = nC_{V,m}(T_2 - T_1) = 8.5\times10^{-3} \times 20.8 \times 223 =
\qty{39}{J}$.

\textbf{5.} الهواء الواجب إضافته: $\Delta n = \Delta P\,V/RT = 6\times10^5 \times 2\times10^{-3}/
2494 = \qty{0.48}{mol}$: أي $0.48/8.5\times10^{-3} = 57$ دفعة، ونحو
\qty{2.2}{kJ} --- أي دقيقة من الجهد الصادق.

\textbf{6.} عند ثبات الحجم يكون $P \propto T$: فيهبط الضغط وهو يبرد
(بنسبة تبلغ $300/523$ لهواء الدفعة الأخيرة)؛ فيضيف الدرّاج
بضع دفعات بعد وقفة.

\textbf{7.} $W_{\text{المتساوي}} = nRT\ln(P_2/P_1) = 8.5\times10^{-3} \times 2494 \times
1.95 = \qty{41}{J} > \qty{39}{J}$: فالمسار المتساوي \omterm{def:b1:first-law:heat}{الحرارة} يضغط الغاز
أكثر (إلى \qty{30}{cm^3} بدل \qty{53}{cm^3}) ليبلغ \qty{7}{bar}،
لأنه يبقى باردًا --- فحجم مكنوس أكبر، وشغل أكبر.

\textbf{8.} $n = 10^5 \times 5\times10^{-4}/2494 = \qty{0.020}{mol}$؛ و $T_2 =
300 \times 20^{0.4} = \qty{994}{K}$.

\textbf{9.} $P_2 = 20^{1.4} = \qty{66}{bar}$.

\textbf{10.} تفوق \qty{994}{K} بكثير درجة الاشتعال الذاتي للوقود:
فيشتعل الوقود المحقون عند التماسّ. أما محرك البنزين فلا يضغط
مزيج الهواء والوقود إلا إلى نحو \qty{750}{K} كي لا يشتعل
قبل الشرارة؛ ونسبة انضغاط مفرطة تعطي ``الطرق''.

\textbf{11.} $W = nC_{V,m}(T_2 - T_1) = 0.020 \times 20.8 \times 694 = \qty{289}{J}$.

\textbf{12.} $25 \times 289 = \qty{7.2}{kW}$ لكل أسطوانة، تُخزَّن في الغاز
الساخن وتُردّ في التمدد.

\textbf{13.} اللاعكوسية (إذ لا يتبع الغاز المسار شبه السكوني،
ويتبدد شغل زائد) ترفع $T_2$؛ \omterm{def:b1:first-law:heat}{والحرارة} المفقودة إلى الجدران
تخفضها. وفي محرك حقيقي يغلب الأثر الثاني عادةً: فتكون $T_2$ أدنى قليلًا
من القيمة المثالية.

\textbf{14.} $Q = 20\times10^{-6} \times 43\times10^6 = \qty{860}{J}$.

\textbf{15.} متساوي الضغط: $Q = nC_{P,m}(T_3 - T_2)$: ومنه $T_3 - T_2 = 860/(0.020 \times
29.1) = \qty{1480}{K}$ و $T_3 = \qty{2470}{K}$؛ و $V_3 = V_2T_3/T_2 = 25 \times
2.48 = \qty{62}{cm^3}$.

\textbf{16.} $W = -P_2(V_3 - V_2) = -66\times10^5 \times 37\times10^{-6} =
\qty{-244}{J}$: وهو شغل يقدّمه الغاز.

\textbf{17.} $T_4 = T_3(V_3/V_1)^{\gamma - 1} = 2470 \times (0.124)^{0.4} =
\qty{1070}{K}$؛ و $W = nC_{V,m}(T_4 - T_3) = 0.020 \times 20.8 \times (-1400) =
\qty{-582}{J}$.

\textbf{18.} الشغل الصافي المقدَّم $= 244 + 582 - 289 = \qty{537}{J}$؛ والنسبة
$537/860 = 62\%$ (وهو مردود دور ديزل المثالي عند هذه الضبطيات؛
وتبلغ المحركات الحقيقية نحو $40\%$).

\textbf{19.} $1 - 300/2470 = 88\%$: فدور ديزل المثالي يبقى دون
حدّ كارنو بكثير، لأنه يتلقى حرارته على امتداد مجال من
درجات \omterm{def:b1:first-law:heat}{الحرارة} لا عند أعلاها.

\textbf{20.} على امتداد دور يكون $\Delta U = 0$: ومنه $Q_{\text{الخارجة}} = -(860 - 537) =
\qty{-323}{J}$، أي أن \qty{323}{J} تخرج مع العادم؛ وللتحقق:
$nC_{V,m}(T_4 - T_1) = 0.020 \times 20.8 \times 770 = \qty{320}{J}$.

\textbf{21.} الاهتزاز سريع (ثانية واحدة) بالقياس إلى توصيل
\omterm{def:b1:first-law:heat}{الحرارة} عبر \qty{10}{L} من الهواء: فهو كظوم. وبالنشر الخطي
للمقدار $PV^\gamma$: $\dd P/P = -\gamma\,\dd V/V = -\gamma Sx/V$.

\textbf{22.} القوة على الكرة $S\,\dd P = -\gamma PS^2x/V$: ومنه $m\ddot x =
-(\gamma PS^2/V)x$، وهي توافقية، مع $\omega_0^2 = \gamma PS^2/mV$.

\textbf{23.} $\omega_0^2 = 1.4 \times 10^5 \times 4\times10^{-8}/(0.0165 \times 0.010)
= \qty{33.9}{s^{-2}}$: ومنه $T = 2\pi/5.83 = \qty{1.08}{s}$. وانطلاقًا من $T = \qty{1.10}{s}$:
$\gamma = 4\pi^2mV/(T^2PS^2) = 4\pi^2 \times 0.0165 \times 0.010/(1.21 \times 10^5
\times 4\times10^{-8}) = 1.35$.

\textbf{24.} الاحتكاك يخمد الحركة، ويزيح وضع التوازن إن كان
جافًّا؛ والتسرّب بجانب الكرة يدع الهواء يفلت أثناء الانضغاط،
فيخفض القوة المُعيدة: وكلاهما يطيل الدور ويحيد بالمقدار $\gamma$
نحو الأدنى --- كما في القيمة $1.35$ أعلاه.

\textbf{25.} المنفاخ والديزل: $Q = 0$، ومنه $\Delta U = W$ --- فصار الشغل
طاقةً داخلية ودرجةَ \omterm{def:b1:first-law:heat}{حرارة}، وفق \omterm{prop:b1:first-law:transformations}{قانون لابلاس} بالأس
$\gamma$. أما الكرة: فالانضغاط السريع كظوم، و $\gamma$ هي التي تعيّن
صلابة نابض الهواء.
\end{solution}
