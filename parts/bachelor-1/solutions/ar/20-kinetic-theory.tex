\section*{الفصل \ref{ch:b1:kinetic-theory} --- نظرية الحركة والغاز المثالي}

\begin{solution}{exo:b1:kinetic-theory:1}
$N = PV/k_BT = 1.013\times10^5 \times 10^{-3}/(1.38\times10^{-23} \times 273) =
\num{2.7e22}$؛ و $n^{*-1/3} = (3.7\times10^{-26})^{1/3} = \qty{3.3}{nm}$، أي عشرة
أضعاف حجم الجزيء.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:2}
$u = \sqrt{3RT/M}$: فللغاز H$_2$ \qty{1930}{m/s}، وللغاز N$_2$ \qty{517}{m/s}، وللغاز O$_2$
\qty{484}{m/s}، وللغاز CO$_2$ \qty{412}{m/s}؛ وللغاز N$_2$ عند \qty{77}{K}: \qty{262}{m/s}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:3}
$P = \qty{3.5}{bar}$ مطلقًا: ومنه $n = PV/RT = 3.5\times10^5 \times 0.030/(8.314
\times 293) = \qty{4.3}{mol}$، أي $\qty{125}{g}$. وعند \qty{323}{K}، مع $P \propto T$:
يكون \qty{3.86}{bar} (أي \qty{2.85}{bar} مقياسيًّا).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:4}
$n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$؛ و $\lambda = 1/(\sqrt2\pi \times 1.37\times
10^{-19} \times 2.4\times10^{25}) = \qty{6.8e-8}{m}$؛ والتواتر $u/\lambda \approx
500/7\times10^{-8} = \qty{7e9}{s^{-1}}$. ومع $\lambda \propto 1/P$: يبلغ \qty{1}{m} عند
$P \approx 10^5 \times 7\times10^{-8} = \qty{7e-3}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:5}
$\rho = PM/RT$: أي \qty{1.20}{kg/m^3} عند \qty{293}{K}، و \qty{0.946}{kg/m^3} عند
\qty{373}{K}. والحمولة $(1.20 - 0.95) \times 2800 \times 9.81 = \qty{7.0}{kN}$:
أي نحو \qty{720}{kg}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:6}
$P_{\mathrm{O}_2} = 0.21P$: أي \qty{0.21}{atm}؛ و \qty{0.11}{atm} عند \qty{5000}{m}؛
و \qty{0.07}{atm} على إفرست --- أي ثلث ما عند سطح البحر. ويدخل الأكسجين إلى
الدم بنسبة ضغطه الجزئي؛ وعند الثلث لا تستطيع الرئتان
إشباع الهيموغلوبين.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:7}
\omterm{def:b1:kinetic-theory:model}{بالغاز المثالي}: $P = RT/V = 2494/10^{-3} = \qty{24.9}{bar}$. وبفان دير فالس:
$RT/(V - b) - a/V^2 = 2494/9.57\times10^{-4} - 0.364/10^{-6} = 26.1 - 3.6 =
\qty{22.4}{bar}$: فالتجاذب ($-3.6$ \omterm{def:b1:kinetic-theory:pressure}{بار}) يغلب الحجم
المستبعَد ($+1.2$ \omterm{def:b1:kinetic-theory:pressure}{بار})؛ و CO$_2$ الحقيقي أدنى بنسبة $10\%$ من القيمة المثالية.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:8}
$n = PV/RT = 10^5 \times 50/(8.314 \times 300) = \qty{2000}{mol}$؛ و $U = \tfrac52
nRT = \qty{1.25e7}{J}$؛ و $\Delta U = \tfrac52 nR \times 5 = \qty{208}{kJ}$؛
أي \qty{104}{s} عند \qty{2}{kW}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:9}
$\Delta L = \alpha L\Delta T = 1.2\times10^{-5} \times 30 \times 50 = \qty{18}{mm}$.
وللحلقة: $\Delta d/d = \alpha\Delta T = 0.05/49.95 = 10^{-3}$: ومنه $\Delta T = 10^{-3}/
1.9\times10^{-5} = \qty{53}{K}$ (فالثقب يتمدد كما يتمدد المعدن حوله).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:10}
$\tfrac32 k_BT = \qty{6.2e-21}{J} = \qty{0.039}{eV}$. وللأرغون $\tfrac32 R =
\qty{12.5}{J/(mol.K)}$؛ وللآزوت $\tfrac52 R = \qty{20.8}{J/(mol.K)}$: فالجزيء
الثنائي الذرة يخزن طاقة في الدوران أيضًا (بدرجتَي حرية
إضافيتين)، أيًّا كانت كتلته.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:11}
في نموذج الاتجاهات الستة: كما في المتن، $P = \tfrac13 n^*mu^2$. وبتماثل المناحي:
$\langle v_x^2\rangle = \langle v_y^2\rangle = \langle v_z^2\rangle$ ومجموعها
هو $\langle v^2\rangle$، فيكون كلٌّ منها $\tfrac13\langle v^2\rangle$. وحجّة التدفق
مع توزيع تعطي $P = n^*m\langle v_x^2\rangle$ (فالجزيئات
ذات $v_x > 0$ تضرب الجدار، \omterm{def:b1:newton-dynamics:momentum}{وكمية الحركة} $2mv_x$ لكلٍّ منها، والتدفق $n^*v_x/2$ لكل
صنف سرعة، بعد التوسيط)، أي $\tfrac13 n^*m\langle v^2\rangle$ --- وهو نفسه.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:12}
للغاز H$_2$: \qty{1.9}{km/s}، وللغاز O$_2$: \qty{0.48}{km/s}؛ وللأرض \qty{11.2}{km/s}، وللقمر
\qty{2.4}{km/s}. فمن أجل O$_2$ على الأرض، $v_e/u = 23$: وذيل
التوزيع وراء ذلك مهمل تمامًا؛ ومن أجل H$_2$، $v_e/u \approx 6$:
فتفلت نسبة ضئيلة لكن مطّردة في كل سنة، وعلى مدى مليارات
السنين ذهب الهدروجين. وعلى القمر، $v_e/u$ يساوي $5$ من أجل O$_2$ و
$1.2$ من أجل H$_2$: فتسرّب كل شيء.
\end{solution}

\begin{solution}{pb:b1:kinetic-theory:1}
\textbf{1.} $V = \qty{200}{m^3}$: ومنه $n = PV/RT = 1.013\times10^5 \times 200/(8.314
\times 293) = \qty{8320}{mol}$؛ و $N = nN_A = \num{5.0e27}$.

\textbf{2.} $m = nM = \qty{241}{kg}$: أي ثلاثة أشخاص.

\textbf{3.} $n^* = N/V = \qty{2.5e25}{m^{-3}}$؛ والمسافة $n^{*-1/3} = \qty{3.4}{nm}$،
أي تسعة أقطار جزيئية.

\textbf{4.} للغاز N$_2$: $\sqrt{3 \times 8.314 \times 293/0.028} = \qty{511}{m/s}$؛
وللغاز O$_2$: \qty{478}{m/s}. والمقدار $\tfrac32 k_BT$ نفسه لكل جزيء: فيكون $\tfrac12 mu^2$ ثابتًا،
ومنه $u \propto 1/\sqrt m$.

\textbf{5.} $\tfrac32 k_BT = \qty{6.1e-21}{J}$؛ والمجموع $N \times 6.1\times10^{-21}
= \qty{3.0e7}{J}$؛ وسيارة كتلتها \qty{1000}{kg} عند \qty{100}{km/h}: \qty{3.9e5}{J} ---
فحركة جزيئات القاعة تعادل ثمانين سيارة.

\textbf{6.} $\lambda = 1/(\sqrt2\pi \times 1.37\times10^{-19} \times 2.5\times10^{25})
= \qty{66}{nm}$؛ والزمن $\lambda/u = 66\times10^{-9}/500 = \qty{1.3e-10}{s}$.

\textbf{7.} الارتطامات في الثانية على $S$: $\tfrac16 n^*Su = \tfrac16 \times 2.5\times
10^{25} \times 10^{-4} \times 500 = \qty{2.1e23}{s^{-1}}$.

\textbf{8.} يسلّم كلٌّ منها $2mu = 2 \times 4.8\times10^{-26} \times 500 = \qty{4.8e-23}{kg.m/s}$
(مع $m = M/N_A = \qty{4.8e-26}{kg}$): فالقوة $2.1\times10^{23} \times 4.8\times10^{-23}
= \qty{10}{N}$ على \qty{1}{cm^2}: أي \qty{1.0e5}{Pa}. فالنموذج يعيد إنتاج
\omterm{def:b1:kinetic-theory:pressure}{الضغط} الجوي.

\textbf{9.} $F = PS = 1.013\times10^5 \times 20 = \qty{2.0e6}{N}$ --- أي مئتا
طن؛ ويؤثر \omterm{def:b1:kinetic-theory:pressure}{الضغط} نفسه على الجانب الآخر.

\textbf{10.} $P' = P \times 303/293 = \qty{1.048e5}{Pa}$؛ والمحصلة $\Delta P\,S = 3.5
\times10^3 \times 20 = \qty{70}{kN}$ إلى الخارج: أي سبعة أطنان على الجدار.

\textbf{11.} يخرج الهواء حتى يتساوى \omterm{def:b1:kinetic-theory:pressure}{الضغط}: مع $n \propto 1/T$ عند
$P$ و $V$ ثابتين: $8320 \times (1 - 293/303) = \qty{275}{mol}$، أي \qty{8}{kg}
تغادر.

\textbf{12.} $\rho = PM/RT$: أي \qty{1.205}{kg/m^3} عند \qty{293}{K}؛ و \qty{0.946}{kg/m^3}
عند \qty{373}{K}.

\textbf{13.} الهواء الساخن $0.946 \times 2800 \times 9.81 = \qty{26.0}{kN}$؛ والهواء
البارد المزاح $1.205 \times 2800 \times 9.81 = \qty{33.1}{kN}$؛ والحمولة \qty{7.1}{kN}
(أي \qty{725}{kg}).

\textbf{14.} $725 - 300 = \qty{425}{kg}$: أي خمسة ركاب.

\textbf{15.} الحمولة $\propto \rho_c - \rho_h = \rho_c(1 - 293/T)$: ومضاعفة
$1 - 293/373 = 0.215$ تقتضي $T = 293/0.57 = \qty{514}{K}$ (أي \qty{240}{\degreeCelsius})،
وهو أعلى بكثير مما يتحمله النايلون (نحو \qty{120}{\degreeCelsius}).

\textbf{16.} للهليوم $\rho = 1.205 \times 4/29 = \qty{0.166}{kg/m^3}$: فالحمولة
$(1.205 - 0.166) \times 2800 \times 9.81 = \qty{28.5}{kN}$ (أي \qty{2.9}{t})، أي أكثر
بأربع مرات --- لكن الهليوم غال ويضيع عند كل هبوط، بينما
الهواء الساخن مجاني ويمكن ضبط حمولته بالموقد.

\textbf{17.} $0.21 \times 1.013\times10^5 = \qty{2.1e4}{Pa}$.

\textbf{18.} $n^*_{\mathrm{O}_2} = P_{\mathrm{O}_2}/k_BT$: ففي القاعة $2.1\times10^4/
(1.38\times10^{-23} \times 293) = \qty{5.3e24}{m^{-3}}$؛ وعلى إفرست $0.21 \times 0.33
\times 1.013\times10^5/(1.38\times10^{-23} \times 250) = \qty{2.0e24}{m^{-3}}$:
أي $38\%$ من كثافة القاعة.

\textbf{19.} $H = RT/Mg = 8.314 \times 260/(0.029 \times 9.81) = \qty{7.6}{km}$؛
و $P_0/2$ عند $H\ln 2 = \qty{5.3}{km}$.

\textbf{20.} $c_s = \sqrt{1.4 \times 8.314 \times 293/0.029} = \qty{343}{m/s}$،
مقابل $u = \qty{503}{m/s}$ للمزيج: أي $c_s/u = \sqrt{\gamma/3} = 0.68$.
فالصوت اضطراب تحمله الجزيئات نفسها، ولذلك ينتقل
بسرعة من رتبة سرعتها --- أقلّ منها قليلًا، إذ
لا تهمّ إلا المركّبة على امتداد الانتشار.

\textbf{21.} $c_s \propto \sqrt T$: ومنه $343\sqrt{303/293} = \qty{349}{m/s}$.
وللهليوم: $\sqrt{(5/3) \times 8.314 \times 293/0.004} = \qty{1010}{m/s}$: فرنينات
الجهاز الصوتي تتناسب مع $c_s$، ومن ثمّ يرتفع كل رنين
ثلاث مرات.

\textbf{22.} أبرد: $343\sqrt{250/293} = \qty{317}{m/s}$، أي أبطأ بنسبة $8\%$؛
ولا يدخل \omterm{def:b1:kinetic-theory:pressure}{الضغط} في ذلك.

\textbf{23.} العدد $N$ نفسه (بالمقادير $P$ و $V$ و $T$ نفسها)؛ والكتلة $\times 40/29 = \qty{333}{kg}$؛
والسرعات $\times\sqrt{29/40}$: أي $u = \qty{428}{m/s}$؛ و $U = \tfrac32 nRT$ بدل
$\tfrac52$: أي $60\%$.

\textbf{24.} $U = \tfrac52 \times 8320 \times 8.314 \times 293 = \qty{5.1e7}{J}$؛
و $\Delta U = \tfrac52 nR \times 10 = \qty{1.7e6}{J}$.

\textbf{25.} متوسط واحد، $\langle v^2\rangle$، يحدّد \omterm{def:b1:kinetic-theory:pressure}{الضغط}
($\tfrac13 n^*m\langle v^2\rangle$)، ويعرّف \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة}
($\tfrac12 m\langle v^2\rangle = \tfrac32 k_BT$)، ويعطي معادلة الحالة
\omterm{def:b1:kinetic-theory:U}{والطاقة الداخلية}، ويضبط رتبة سرعة الصوت.
\end{solution}
