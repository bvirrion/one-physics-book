\section*{الفصل \ref{ch:b1:units-dimensions} --- الوحدات والأبعاد والقياس}

\begin{solution}{exo:b1:units-dimensions:1}
$\unit{J} = \unit{kg.m^2.s^{-2}}$; $\unit{W} = \unit{kg.m^2.s^{-3}}$;
$\unit{Pa} = \unit{kg.m^{-1}.s^{-2}}$; $\unit{C} = \unit{A.s}$.
$\unit{Pa.m^3} = \unit{kg.m^{-1}.s^{-2}.m^3} = \unit{kg.m^2.s^{-2}} = \unit{J}$
(شغل قوى الضغط، $P\,\dd V$).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:2}
الضغط $M L^{-1} T^{-2}$؛ والاستطاعة $M L^2 T^{-3}$؛ والشحنة $I\,T$؛
والتوتر هو الاستطاعة على التيار $= M L^2 T^{-3} I^{-1}$.
$[G] = [F r^2/m^2] = M L T^{-2} L^2 M^{-2} = L^3 M^{-1} T^{-2}$;
$[h] = [E/\nu] = M L^2 T^{-2} \cdot T = M L^2 T^{-1}$;
$[k_B] = [E/T] = M L^2 T^{-2} \Theta^{-1}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:3}
$\sqrt{2gh}$: $\sqrt{L T^{-2} \cdot L} = L T^{-1}$، متجانسة.
$\tfrac12 mv$: $M L T^{-1} \neq M L^2 T^{-2}$، ليست طاقة.
$\rho g h$: $M L^{-3} \cdot L T^{-2} \cdot L = M L^{-1} T^{-2}$، أي ضغط.
$v_0 t + \tfrac12 g t^2$: كلا الحدّين $L$، متجانسة.
$2\pi\sqrt{g/\ell}$: $\sqrt{T^{-2}} = T^{-1}$ --- أي تواتر لا دور.
$I_0\eu^{-t/RC}$: $RC$ \omterm{def:b1:units-dimensions:unit}{بوحدة} $\unit{\ohm.F} = \unit{s}$، والوسيط عديم البعد، فهي متجانسة.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:4}
المسطرة: $\delta = \qty{1}{mm}$، ومنه $u = 1/\sqrt{12} = \qty{0.29}{mm}$.
الميزان: $u = 0.01/\sqrt{12} = \qty{0.003}{g}$.
الفولطمتر: $a = 0.005 \times 4.80 = \qty{0.024}{V}$، ومنه
$u = 0.024/\sqrt3 = \qty{0.014}{V}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:5}
$v = k F^\alpha \mu^\beta$: $L T^{-1} = (M L T^{-2})^\alpha (M L^{-1})^\beta$،
ومنه $\alpha + \beta = 0$ و $\alpha - \beta = 1$ و $-2\alpha = -1$:
$\alpha = \tfrac12$ و $\beta = -\tfrac12$ و $v = k\sqrt{F/\mu}$ (والواقع أن $k = 1$).
الصوت: \omterm{def:b1:units-dimensions:dimension}{بُعد} $P/\rho$ هو $M L^{-1} T^{-2} / (M L^{-3}) = L^2 T^{-2}$،
ومنه $v = k\sqrt{P/\rho}$ (حيث $k = \sqrt\gamma \approx 1.2$ للهواء).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:6}
المجموع $= \qty{3620}{ms}$، والمتوسط $\bar t = \qty{452.5}{ms}$. والانحرافات
$-0.5, -5.5, 8.5, 2.5, -3.5, 5.5, -8.5, 1.5$؛ ومجموع مربعاتها $226$؛
ومنه $s = \sqrt{226/7} = \qty{5.7}{ms}$ و $u(\bar t) = 5.7/\sqrt8 = \qty{2.0}{ms}$.
النتيجة: $t = \qty{452.5\pm2.0}{ms}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:7}
لدينا $R = 4.80/0.0124 = \qty{387}{\ohm}$. والارتيابان النسبيان
$0.03/4.80 = 0.63\%$ و $0.2/12.4 = 1.6\%$؛ فالمركّب هو
$\sqrt{0.63^2 + 1.6^2} = 1.7\%$، و
$u(R) = \qty{7}{\ohm}$: أي $R = \qty{387\pm7}{\ohm}$. والتيار هو
المهيمن --- فيُحسَّن الأمبيرمتر (أو مجاله) أولًا.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:8}
$z = 7/\sqrt{16 + 9} = 1.4 \leq 2$: متوافقتان.
والثالثة مع الأولى: $z = 9/\sqrt{4 + 16} = 2.0$: على الحدّ، متوافقتان
بالكاد. والثالثة مع الثانية: $z = 16/\sqrt{4 + 9} = 4.4$: غير متوافقتين
--- فإحداهما على الأقل تحمل خطأً منهجيًّا أو ارتيابًا مقدَّرًا بأقلّ من
قيمته.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:9}
$V = \pi d^3/6 = \pi \times 25.40^3/6 = \qty{8580}{mm^3}$.
و$u(V)/V = 3\,u(d)/d = 3 \times 0.02/25.40 = 0.24\%$،
ومنه $u(V) = \qty{20}{mm^3}$: أي $V = \qty{8.58\pm0.02}{cm^3}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:10}
$F = k\eta^a R^b v^c$: من $M$ نجد $a = 1$؛ ومن $T$ نجد $-a - c = -2$ أي $c = 1$؛
ومن $L$ نجد $-a + b + c = 1$ أي $b = 1$: فيكون $F = k\eta R v$ (ستوكس، $k = 6\pi$).
وبأخذ $\rho$ بدلًا منها: من $M$ نجد $a = 1$؛ ومن $T$ نجد $-c = -2$ أي $c = 2$؛ ومن $L$
نجد $-3 + b + 2 = 1$ أي $b = 2$: فيكون $F = k\rho R^2 v^2$.
قطرة المطر: $\eta R v = 1.8\times10^{-5} \times 10^{-3} \times 5 \approx
\qty{1e-7}{N}$؛ و$\rho R^2 v^2 = 1.2 \times 10^{-6} \times 25 =
\qty{3e-5}{N}$، أي أكبر بثلاثمئة مرة: فينطبق النظام العُطالي (نظام
السرعات الكبيرة) --- والنسبة $\rho v R/\eta \approx 330$ هي عدد
رينولدز.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:11}
$\bar I = \qty{5.0}{mA}$ و $\bar U = \qty{1.000}{V}$؛ والانحرافات في $I$:
$-3, -1, 1, 3$؛ وفي $U$: $-0.59, -0.22, 0.22, 0.59$.
ومنه $\sum (I_i - \bar I)(U_i - \bar U) = 3.98$ و $\sum (I_i - \bar I)^2 = 20$:
فالميل $a = \qty{0.199}{V/mA} = \qty{199}{\ohm}$ والمقطع $b = 1.000 - 0.199 \times 5 =
\qty{0.005}{V}$. ومع $u(U) = \qty{0.02}{V}$ لكل نقطة يقع مقطعٌ مقداره
\qty{0.005}{V} داخل حدود الارتياب تمامًا: أي متوافق مع الصفر، والناقلة
أومية، و$R \approx \qty{199}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:12}
$\ell = G^a h^b c^c$: من $M$ نجد $-a + b = 0$؛ ومن $T$ نجد $-2a - b - c = 0$؛ ومن $L$
نجد $3a + 2b + c = 1$. ومنه $b = a$ و $c = -3a$ و $2a = 1$:
$\ell_P = \sqrt{Gh/c^3} = \sqrt{4.42\times10^{-44}/2.7\times10^{25}}
= \qty{4.1e-35}{m}$؛ و$t_P = \ell_P/c = \qty{1.4e-43}{s}$؛
و$m_P = \sqrt{hc/G} = \sqrt{1.99\times10^{-25}/6.67\times10^{-11}}
= \qty{5.5e-8}{kg}$. وطول بلانك أدنى من البروتون بعشرين رتبة من رتب
القدر؛ أما كتلة بلانك، وهي \qty{55}{\micro g}، فذرّة غبار --- هائلة
لجسيم، ضئيلة لإنسان (أخفّ $10^9$ مرة).
\end{solution}

\begin{solution}{pb:b1:units-dimensions:1}
\textbf{1.} $\sqrt{L/(L T^{-2})} = \sqrt{T^2} = T$.

\textbf{2.} $\tau = k\ell^\alpha m^\beta g^\gamma$ يعطي
$T = L^{\alpha+\gamma} M^\beta T^{-2\gamma}$، ومنه $\beta = 0$: فلا
تجميعة من $\ell$ و $g$ تستطيع إلغاء كتلة.

\textbf{3.} $\theta_0$ عديم البعد، فأيّ عامل $f(\theta_0)$ غير مرئي
للأبعاد. فأبقِ السعة صغيرة (دون $10^\circ$ تقريبًا) حيث
$f \approx 1$ بدقة أفضل من $0.2\%$.

\textbf{4.} $g = 4\pi^2\ell/\tau^2$.

\textbf{5.} $\ell = 99.0 + 1.00 = \qty{100.0}{cm} = \qty{1.000}{m}$.

\textbf{6.} $u = \qty{1}{mm}/\sqrt{12} = \qty{0.29}{mm}$.

\textbf{7.} المحاذاة: $5/\sqrt3 = \qty{2.9}{mm}$؛ والمركّب
$\sqrt{0.29^2 + 2.9^2} = \qty{2.9}{mm}$ --- فقراءة الشريط
مهملة. ومنه $u(\ell)/\ell = 0.29\%$.

\textbf{8.} خطأ زمن ردّ الفعل (نحو \qty{0.1}{s}) هو نفسه سواء وُقّت
دور واحد أو عشرة أدوار، فأثره على دور واحد أصغر عشر مرات. والتكرار
يكشف التبعثر (ومنه $s$) ويقسم ارتياب المتوسط على $\sqrt8$.

\textbf{9.} مجموع الأزمنة $\qty{160.80}{s}$، ومنه يكون
$\overline{t_{10}} = \qty{20.100}{s}$. والانحرافات هي، بالثانية،
$0.02, -0.05, 0.08, -0.01$ ثم $0.05, -0.08, 0.01, -0.02$؛ ومجموع
مربعاتها $\num{188e-4}$؛ ومنه $s = \sqrt{188\times10^{-4}/7} = \qty{0.052}{s}$.

\textbf{10.} $u(\overline{t_{10}}) = 0.052/\sqrt8 = \qty{0.018}{s}$؛ ومنه
$\tau = \qty{2.0100}{s}$ و $u(\tau) = \qty{0.0018}{s}$.

\textbf{11.} $u(\tau)/\tau = 0.09\%$، أي أصغر ثلاث مرات من
$u(\ell)/\ell = 0.29\%$.

\textbf{12.} $g = 4\pi^2 \times 1.000/2.0100^2 = 39.48/4.040 =
\qty{9.772}{m/s^2}$.

\textbf{13.} $g \propto \ell\,\tau^{-2}$:
$\dfrac{u(g)}{g} = \sqrt{\left(\dfrac{u(\ell)}{\ell}\right)^2
+ \left(2\dfrac{u(\tau)}{\tau}\right)^2} = \sqrt{0.29^2 + 0.18^2}\,\%
= 0.34\%$.

\textbf{14.} $u(g) = 0.0034 \times 9.772 = \qty{0.033}{m/s^2}$:
$g = \qty{9.77\pm0.03}{m/s^2}$.

\textbf{15.} $z = \abs{9.772 - 9.81}/0.033 = 1.2 \leq 2$: متوافقتان.

\textbf{16.} الطول ($0.29\%$ في مقابل $0.18\%$ لحدّ الدور).
وتحديد الطرفين إلى $\pm\qty{1}{mm}$ يخفض $u(\ell)/\ell$ إلى
$0.06\%$ و$u(g)/g$ إلى $\sqrt{0.06^2 + 0.18^2} = 0.19\%$؛ ومضاعفة
الطول تنصّف $u(\ell)/\ell$ وتطيل $\tau$، فتساعد الحدّين معًا. أما
توقيت عشرين دورًا بدل عشرة فينصّف حدّ الدور
وحده: $u(g)/g = \sqrt{0.29^2 + 0.09^2} = 0.30\%$ --- وهو مكسب ضئيل ما
دام الطول هو عنق الزجاجة.

\textbf{17.} لا: فالانزياح المشترك بين كل القراءات
\emph{خطأ منهجي} (أي تحيّز)؛ والميزانية تغطي التبعثر العشوائي
والتسامحات المعلنة، لا انزياحًا ثابتًا. ويُكشف بتغيير الطريقة أو
الجهاز، أو --- كما في الجزء V --- بمقطع غير معدوم لمستقيم
$\tau^2$--$\ell$.

\textbf{18.} العلاقة $\tau^2 = (4\pi^2/g)\,\ell$ مستقيم يمرّ بالمبدأ:
فالخطية والمقطع المعدوم كلاهما قابل للاختبار بالعين وبالمربعات
الصغرى؛ أما $\tau \propto \sqrt\ell$ فلا.

\textbf{19.} $\bar\ell = \qty{0.800}{m}$; $\overline{\tau^2} = 16.16/5 =
\qty{3.232}{s^2}$.

\textbf{20.} الانحرافات في $\ell$: $-0.4, -0.2, 0, 0.2, 0.4$؛ وفي
$\tau^2$: $-1.612, -0.822, 0.018, 0.808, 1.608$. والمجموع المتقاطع $= 1.614$،
و$\sum(\ell_i - \bar\ell)^2 = 0.400$: ومنه $a = \qty{4.035}{s^2/m}$ و
$b = 3.232 - 4.035 \times 0.800 = \qty{0.004}{s^2}$.

\textbf{21.} $g = 4\pi^2/a = 39.48/4.035 = \qty{9.78}{m/s^2}$.

\textbf{22.} $b \approx 0$ كما يتنبأ النموذج. أما $b$ غير المعدوم
بوضوح فيفضح انزياحًا منهجيًّا $\delta$ في كل طول
($\tau^2 = a(\ell_{\text{مقيس}} - \delta)$ يعطي $b = -a\delta$)، مثل
مركز كرة سيّئ التحديد --- أو أثر السعة.

\textbf{23.} $u(g)/g = u(a)/a = 0.05/4.035 = 1.2\%$ و $u(g) =
\qty{0.12}{m/s^2}$: أي $g = \qty{9.78\pm0.12}{m/s^2}$.

\textbf{24.} $z = \abs{9.772 - 9.784}/\sqrt{0.033^2 + 0.12^2} = 0.1$:
أي متوافقتان تمامًا. والقياس المفرد المتقن أدقّ؛ أما المستقيم فيصادق
على النموذج ويحمي من انزياح في الطول.

\textbf{25.} $\tau = 2\pi\sqrt{10/9.8} = \qty{6.3}{s}$؛ وتأرجحة واحدة
مؤقَّتة بزمن ردّ فعل نحو \qty{0.2}{s} تعطي $u(\tau)/\tau \approx
3\%$، ومنه $u(g)/g \approx 6\%$، بينما يهبط $u(\ell)/\ell$ إلى
$0.03\%$: أي أسوأ عشرين مرة من الجزء IV. فمكسب الطول يضيّعه توقيت دور
واحد. والاستراتيجية المثلى: نواس طويل \emph{مع} أدوار كثيرة، مع
التكرار --- عشر تأرجحات لنواس \qty{10}{m}، ثماني
مرات، تعطي $u(\tau)/\tau \approx 0.03\%$ و $u(g)/g \approx 0.07\%$.
\end{solution}
