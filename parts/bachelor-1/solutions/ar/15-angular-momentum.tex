\section*{الفصل \ref{ch:b1:angular-momentum} --- العزم الحركي}

\begin{solution}{exo:b1:angular-momentum:1}
$Fd = 50 \times 0.30 = \qty{15}{N.m}$؛ و $Fd\sin 60^\circ = \qty{13}{N.m}$؛
وعلى امتداد الذراع يمرّ الحامل بالصامولة: \omterm{def:b1:angular-momentum:moment}{فالعزم} $0$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:2}
$L = mrv = 5.97\times10^{24} \times 1.50\times10^{11} \times 2.98\times10^4 =
\qty{2.7e40}{kg.m^2/s}$؛ و $\dd A/\dd t = L/2m = rv/2 = \qty{2.2e15}{m^2/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:3}
$r = \ell\sin 30^\circ = \qty{0.50}{m}$ و $\omega = \qty{3.4}{rad/s}$:
ومنه $L_z/m = r^2\omega = \qty{0.84}{m^2/s}$. وحامل الشدّ
يمرّ بنقطة التعليق (فعزمه معدوم حول أيّ محور يمرّ بها)؛ والوزن
مواز للمحور الشاقولي (فعزمه معدوم حوله): ومنه ينحفظ $L_z$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:4}
$J_1 = 4.0 + 2 \times 3.0 \times 0.64 = \qty{7.8}{kg.m^2}$ و $J_2 = 4.0 +
2 \times 3.0 \times 0.04 = \qty{4.2}{kg.m^2}$؛ و $\omega_2 = \omega_1J_1/J_2 =
1.85\omega_1$: أي \qty{3.7}{} دورة في الثانية. و $E_k = L^2/2J$: فالنسبة
$J_1/J_2 = 1.85$؛ والعضلات، وهي تسحب الذراعين إلى الداخل ضدّ
النزوع الطاردَ، هي التي تبذل الشغل.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:5}
$L_\Delta = m\ell^2\dot\theta$؛ والشدّ: عزمه معدوم (فهو يقطع المحور)؛
والوزن: $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{وذراع الرافعة} $\ell\sin\theta$، وهو مُعيد).
ومنه $m\ell^2\ddot\theta = -mg\ell\sin\theta$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:6}
$r_pv_p = r_av_a$: ومنه $v_p/v_a = r_a/r_p = 1.034$؛ ومع $(v_p + v_a)/2 \approx
\qty{29.8}{km/s}$: يكون $v_p = \qty{30.3}{km/s}$ و $v_a = \qty{29.3}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:7}
$L = mrv$ منحفظ: ومنه $v_2 = v_1r_1/r_2 = \qty{4.0}{m/s}$ و $\omega_2 = v_2/r_2
= \qty{16}{rad/s}$ (بعد \qty{4}{rad/s}). و $E_1 = \qty{0.40}{J}$ و $E_2 =
\qty{1.6}{J}$: فالسحب يبذل \qty{1.2}{J}. والشدّ قطري، لكن
انتقال القرص صار له مركّبة قطرية (إذ يلتفّ إلى الداخل): فالشغل
$\vect T\cdot\dd\vect\ell$ لم يعد معدومًا.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:8}
العزوم حول المرتكز (بأبعاد مقيسة عنه): الطفل $30 \times 1.5 =
45$ (يسارًا)؛ ووزن اللوح عند مركزه، على بعد $0.5$ يمينًا: $20 \times
0.5 = 10$ (يمينًا)؛ والطفل الثاني على بعد $x$ يمينًا: $45x$. والتوازن:
$45 = 10 + 45x$، ومنه $x = \qty{0.78}{m}$. ويحمل المرتكز الوزن الكلي،
$95 \times 9.81 = \qty{932}{N}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:9}
للكتلتين: $m_1a = m_1g - T_1$ و $m_2a = T_2 - m_2g$؛ وللبكرة: $J\dot\omega =
(T_1 - T_2)R$ مع $a = R\dot\omega$. وبالجمع: $a = (m_1 - m_2)g/(m_1 + m_2 +
J/R^2) = 9.81/(3.0 + 1.0) = \qty{2.5}{m/s^2}$ (مقابل \qty{3.3}{m/s^2})؛
و $T_1 = m_1(g - a) = \qty{14.7}{N}$ و $T_2 = m_2(g + a) = \qty{12.3}{N}$ ---
وهما غير متساويين، لأن البكرة تحتاج إلى \omterm{def:b1:angular-momentum:moment}{عزم} محصّل.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:10}
$r_1v_1 = r_2v_{\perp 2}$: ومنه $v_{\perp 2} = 40 \times 0.5/10 = \qty{2.0}{km/s}$.
ولا يعطي قانون المساحات إلا المركّبة العمودية؛ أما القطرية
فتحتاج إلى انحفاظ الطاقة (\cref{ch:b1:central-forces}).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:11}
$J\omega$ ثابت مع $J \propto R^2$: ومنه $T' = T(R'/R)^2 = 25 \times 86400
\times (10^4/7\times10^8)^2 = \qty{4.4e-4}{s}$. والسرعة الاستوائية $2\pi R'/T'
= \qty{1.4e8}{m/s}$، أي نصف سرعة الضوء: فلا يمكن للانهيار أن يحتفظ
\omterm{def:b1:angular-momentum:L}{بالعزم الحركي} كله --- إذ يُطرح كثير منه (وتدخل النسبية).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:12}
$L_z = mr^2\omega$، و $r$ يتغير: ومنه $\dd L_z/\dd t = 2mr\dot r\omega \neq 0$.
ويجب أن يكون هذا \omterm{def:b1:angular-momentum:moment}{عزم} ردّ الفعل العرضي $N$ للقضيب: أي $rN =
2mr\dot r\omega$، ومنه $N = 2m\dot r\omega$ --- فالقضيب يدفع الخرزة جانبًا
وهي تنزلق نحو الخارج (وهو حدّ كوريوليس في \cref{ch:b1:non-inertial-frames}).
\end{solution}

\begin{solution}{pb:b1:angular-momentum:1}
\textbf{1.} $m\omega^2r = GMm/r^2$: ومنه $\omega = \sqrt{GM/r^3} = \sqrt{3.98\times
10^{14}/5.66\times10^{25}} = \qty{2.65e-6}{rad/s}$، والدور $2\pi/\omega =
\qty{2.37e6}{s} = \qty{27.4}{d}$.

\textbf{2.} $L_{\text{مداري}} = 7.35\times10^{22} \times 1.475\times10^{17}
\times 2.65\times10^{-6} = \qty{2.9e34}{kg.m^2/s}$.

\textbf{3.} $v = \sqrt{GM/r}$ و $L = mrv = m\sqrt{GMr}$؛ ومنه $\dd L/\dd r =
\tfrac12 m\sqrt{GM/r} = L/2r$.

\textbf{4.} $J = 0.33 \times 5.97\times10^{24} \times 4.06\times10^{13} =
\qty{8.0e37}{kg.m^2}$؛ و $\Omega = \qty{7.29e-5}{rad/s}$؛ و $L_{\text{ذاتي}} =
\qty{5.8e33}{kg.m^2/s}$، أي خُمس $L_{\text{مداري}}$.

\textbf{5.} $J_m \approx 0.4 \times 7.35\times10^{22} \times 3.0\times10^{12} =
\qty{8.9e34}{kg.m^2}$، مضروبًا في $\omega$: أي \qty{2.4e29}{kg.m^2/s}، وهو $10^{-5}$ من
\omterm{def:b1:angular-momentum:moment}{عزم} المدار: فهو مهمل.

\textbf{6.} يؤثر جذب الشمس في مركز كتلة الجملة
(فلا \omterm{def:b1:angular-momentum:moment}{عزم} له حوله من الرتبة الأولى)؛ وأثره المدّي في الثنائية
صغير: فينحفظ \omterm{def:b1:angular-momentum:moment}{العزم} الكلي $L$ بالدقة اللازمة هنا.

\textbf{7.} الانتفاخ الأقرب، وهو أمام مستقيم الأرض--القمر، يجذبه
القمر بقوة مركّبتها المماسية تعاكس دوران
الأرض.

\textbf{8.} \omterm{def:b1:angular-momentum:moment}{عزم} سالب على الدوران: فتتباطأ الأرض ويطول
اليوم.

\textbf{9.} يجذب الانتفاخ القمر إلى الأمام: \omterm{def:b1:angular-momentum:moment}{فعزم} موجب على
المدار، ويكبر $L_{\text{مداري}}$.

\textbf{10.} $L_{\text{مداري}} \propto \sqrt r$ يكبر، ومنه يكبر $r$؛ و $v \propto
1/\sqrt r$ يتناقص.

\textbf{11.} يبذل الشدّ الأمامي شغلًا موجبًا فيرفع طاقة
القمر؛ والمدار الأعلى مدار أبطأ: فتذهب الطاقة إلى الارتفاع
(أي إلى الكامنة)، بأكثر مما تفقده \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}.

\textbf{12.} القفل المدّي للأرض: أي يوم يساوي شهرًا، فلا انتفاخ
سابق ولا \omterm{def:b1:angular-momentum:moment}{عزم}.

\textbf{13.} $\Delta\Omega/\Omega = -\qty{2.3e-3}{s}/\qty{86164}{s} =
\num{-2.7e-8}$ في كل قرن.

\textbf{14.} $\Delta L_{\text{ذاتي}} = L_{\text{ذاتي}}\Delta\Omega/\Omega =
\qty{-1.55e26}{kg.m^2/s}$ في كل قرن.

\textbf{15.} $\Delta L_{\text{مداري}} = +\qty{1.55e26}{kg.m^2/s}$ في كل قرن.

\textbf{16.} $\Delta r = 2r\,\Delta L/L = 2 \times 3.84\times10^8 \times 1.55
\times10^{26}/2.9\times10^{34} = \qty{4.1}{m}$ في كل قرن، أي \qty{4.1}{cm/yr}:
وهي قريبة من القيمة المقيسة \qty{3.8}{cm/yr} (فطول اليوم له أيضًا
جزء غير مدّي).

\textbf{17.} $\Delta T/T = \tfrac32\Delta r/r = 1.6\times10^{-8}$: ومنه $2.37\times
10^6 \times 1.6\times10^{-8} = \qty{0.04}{s}$ في كل قرن.

\textbf{18.} $\Delta E_{\text{دوراني}} = L_{\text{ذاتي}}\Delta\Omega = 5.8\times10^{33}
\times 7.29\times10^{-5} \times (-2.7\times10^{-8}) = \qty{-1.1e22}{J}$ في كل
قرن.

\textbf{19.} $\Delta E_{\text{مداري}} = GMm\Delta r/2r^2 = 2.93\times10^{37}
\times 4.1/(2 \times 1.475\times10^{17}) = \qty{+4.1e20}{J}$ في كل قرن.

\textbf{20.} المحصلة $-1.1\times10^{22} + 4\times10^{20} \approx \qty{-1.06e22}{J}$
في كل قرن: أي $3.4\times10^{12}\ \unit{W}$، تتبدد حرارةً بالاحتكاك
المدّي في المحيطات وفي الأرض الصلبة.

\textbf{21.} أي خُمس استطاعة البشرية، وجزء من أربعة عشر من التدفق
الحراري الأرضي --- أي مدفأة صغيرة لكنها دائمة.

\textbf{22.} $L_{\text{كلي}} = 3.5\times10^{34}$. وعند $r = 5.5\times10^8$:
$\omega = \sqrt{GM/r^3} = \qty{1.55e-6}{rad/s}$ و $J\omega = 1.2\times10^{32}$ و
$m\sqrt{GMr} = 7.35\times10^{22} \times 4.68\times10^{11} = 3.44\times10^{34}$؛
والمجموع $3.45\times10^{34}$: فهو متوافق. والدور $2\pi/\omega = \qty{4.1e6}{s}
= \qty{47}{d}$.

\textbf{23.} تظلّ مدود الشمس تبطّئ الأرض دون معدّل القمر
المداري، فينعكس عندئذ \omterm{def:b1:angular-momentum:moment}{عزم} الأرض--القمر؛ ثم إن
الشمس ستكون قد تطورت قبل ذلك بزمن طويل (فسلّم الزمن عشرات
مليارات السنين).

\textbf{24.} أثارت الأرض مدودًا على القمر فكبحت دورانه
حتى بقي وجه واحد نحونا؛ ولأنه أخفّ وأقرب إلى شريك أثقل،
كان قفله أسرع بكثير.

\textbf{25.} ينحفظ: \omterm{def:b1:angular-momentum:L}{العزم الحركي} الكلي. ويُتبادل: \omterm{def:b1:angular-momentum:L}{عزم حركي}
من دوران الأرض إلى مدار القمر، نحو \qty{1.6e26}{SI}
في كل قرن. ويتبدد: \omterm{thm:b1:work-and-energy:em}{طاقة ميكانيكية}، بضعة تيراواط، حرارةً
مدّية.
\end{solution}
