\section*{الفصل \ref{ch:b1:charged-particles} --- الجسيمات المشحونة في مجالين كهربائي ومغناطيسي}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$ و $\qty{13.6}{eV} = \qty{2.18e-18}{J}$ و
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. و $v = \sqrt{2eU/m}$: فللإلكترون
\qty{1.9e7}{m/s}؛ وللبروتون \qty{4.4e5}{m/s} (أي أبطأ بالعامل $\sqrt{1836}$).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$؛ و $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$؛ و $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$؛ و $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. وعند $30^\circ$: حلزون نصف قطره $R\sin 30^\circ
= \qty{10}{m}$ وخطوته $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$، ينساق
على امتداد خط المجال.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. أما الأيونات الأسرع فتغلب فيها القوة
المغناطيسية، فتنحني نحو الجهة المغناطيسية؛ وأما الشحنة المعاكسة بالسرعة
نفسها فتنقلب القوتان معًا وتبقيان متوازنتين --- فالمنتقي أعمى عن
الإشارة وعن الكتلة.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$؛ والانحراف
$0.30 \times 0.10 = \qty{3.0}{cm}$؛ والحساسية \qty{0.30}{mm/V}. والإلكترونات
الأسرع تقضي زمنًا أقلّ بين اللوحين: أي $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$؛ فمع $m_{12} = \qty{1.99e-26}{kg}$ يكون البسط
$\sqrt{2 \times 1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}$،
مقسومًا على $B = 0.5$، أي $R_{12} = \qty{14.1}{cm}$؛
و $R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$؛ والفصل $2\Delta R =
\qty{1.2}{cm}$ بين نقطتَي الحطّ (فنصفا القطر يختلفان بمقدار \qty{5.8}{mm}،
والقطران بضعف ذلك).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$؛ و $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$؛ ومع \qty{100}{keV} في كل دورة: $270$ دورة؛ والزمن $270/f_c =
\qty{12}{\micro s}$؛ و $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$ و $v_\parallel = \qty{8.7e5}{m/s}$: ومنه $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$؛
والدور $2\pi m/eB = \qty{18}{ns}$، والخطوة $v_\parallel T = \qty{1.5}{cm}$:
أي نحو $64$ دورة في كل متر.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: أي دون الميكروفولط. ومع $n$ أصغر بعشرة ملايين مرة، يعطي
نصف الناقل ميلي فولطات --- أي مجسًّا صالحًا.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$؛ واضرب سلّميًّا في $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ أيًّا كان $\vect B(\vect r, t)$، ما دام لا يوجد $\vect E$. أما
اتجاه السرعة \omterm{def:b1:point-kinematics:frenet}{ونصف قطر التقوّس} والخطوة فيمكن أن تتغير كلها؛
وأما السرعة \omterm{thm:b1:work-and-energy:ke}{والطاقة الحركية} (\qty{10}{keV}) فلا.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$ و $m\dot v_y = qE - qv_xB$ و $v_z = 0$. ومع $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$ و $m\dot w_y = -qw_xB$ --- أي
دوران سيكلوتروني صرف للمتجهة $\vect w$ عند $\omega_c$. وانطلاقًا من السكون،
$\vect w(0) = -(E/B)\vect e_x$: فتدور $\vect w$ بمعيار ثابت $E/B$، ومن ثمّ تكون
السرعة هي الانسياق $E/B$ مضافًا إليه متجهة دائرة لها المعيار نفسه ---
أي منحن دويري، ذو نتوءات حيث $\vect v = \vect 0$. وللإلكترون: الانسياق $E/B =
\qty{1.0e5}{m/s}$ و $\omega_c = \qty{1.8e10}{rad/s}$، وارتفاع القوس $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$؛ و $\alpha \approx L/R = 0.04/0.15 = 0.27$ (أي نحو $15^\circ$):
فالانحراف $\approx 0.30 \times 0.27 = \qty{8}{cm}$ من أجل
ميلي تسلا متواضع. والزوايا الكبيرة باللوحين كانت ستحتاج إلى كيلوفولطات وألواح
طويلة؛ أما الوشيعة فتثني الحزمة عشرات الدرجات بلا أيّ توتر
البتة.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: فلا استطاعة، وسرعة
ثابتة.

\textbf{2.} $mv^2/R = evB$: ومنه $R = mv/eB$؛ و $T = 2\pi R/v = 2\pi m/eB$، وهو خال من
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ عند كل عبور، أي \qty{100}{keV} في كل دورة.

\textbf{5.} يجب أن تكون الفجوة على القطبية المسرِّعة في كل مرة يصل
فيها البروتون، على مدى نصف دور؛ وإذا حاد التواتر انزلق الطور
وصارت الركلات مبطّئة في النهاية.

\textbf{6.} بعد دورة واحدة $E_k = \qty{100}{keV}$ و $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$ و $R = mv/eB = \qty{3.0}{cm}$؛ وبعد $n$ دورة يكون $E_k =
n \times \qty{100}{keV}$ و $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$؛ و $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ دورة؛ و $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} لا: إذ يتوقف $E_k$ على $B$ و $R_{\max}$ وحدهما؛ أما $U$ فيحدد
عدد الدورات والزمن.

\textbf{11.} المسار \qty{560}{m}: فيلزم $\lambda \gtrsim \qty{1}{km}$، أي
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
أي خلاء عال.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$؛ فيكبر الدور بنسبة $2.9\%$.
وعلى مدى $270$ دورة يتراكم الانزلاق إلى عدة أدوار --- فالدورات
الأخيرة كانت ستخرج عن الطور؛ وعمليًّا يُضبط $B$ وطور الحقن،
و \qty{27}{MeV} قريبة من حدّ الآلة البسيطة.

\textbf{13.} $E_k = (eBR)^2/2m$: ومنه $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$؛ و $R =
\qty{1.1}{m}$ عند \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ يكبر بنسبة $24\%$ من المركز إلى الحافة: فتواتر
ثابت يخرج عن التساوق خروجًا لا رجعة فيه.

\textbf{15.} السيكلوترون المتزامن: يخفّض التواتر مع اكتساب الرزمة
طاقةً --- أي رزمة واحدة في كل مرة، وتيار متوسط ضعيف. والمتساوي الزمن:
يُشكّل $B(R) \propto \gamma$ فيبقى الدور ثابتًا --- أي حزمة
مستمرة، وهو الخيار الحديث.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$؛ و $R = p/eB = \qty{1.2}{m}$، أي أكثر قليلًا من
القيمة غير النسبية \qty{1.1}{m}.

\textbf{17.} الدفع العرضي $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$؛ والزاوية $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: أي ميؤوس منها للتوجيه.

\textbf{18.} مغناطيسيًّا: $R = p/eB = \qty{2.3}{m}$، والزاوية $L/R = \qty{0.43}{rad}
\approx 25^\circ$: أي أكثر بألف مرة --- فالمغانط إذن.

\textbf{19.} الطاقة لكل بروتون $\qty{3.68e-11}{J}$؛ والجرعة $2.0 \times 0.50 =
\qty{1.0}{J}$: أي $2.7\times10^{10}$ بروتون.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \num{6.2e9}$ بروتونًا في الثانية: أي نحو
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
أي مصباح ليلي، يُوصَل تمامًا حيث يُراد.

\textbf{22.} $\qty{58}{MeV}$ في \qty{1}{cm}: أي نحو \qty{6}{keV/\micro m} ---
أي تأيّن كثيف على امتداد الأثر الأخير.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ تأيّن لكل بروتون ---
وهو الضرر الجزيئي الذي يقتل الخلية.

\textbf{24.} تودع الأشعة السينية طاقتها على امتداد مسارها كله، ومعظمها قرب
المدخل؛ أما البروتونات فتودع معظم طاقتها في نهاية مداها،
فتوفّر النسيج الذي أمامها ولا تترك شيئًا وراءها.

\textbf{25.} $R = mv/eB$ و $T = 2\pi m/eB$ (للدائرة)، و $E_k = (eBR)^2/2m$
(للطاقة)، و $E_k/2eU$ (للدورات)، والجرعة $=$ (البروتونات $\times$ الطاقة)/الكتلة
والتيار $= e \times$ المعدل (للزمن).
\end{solution}
