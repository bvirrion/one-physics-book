\section*{الفصل \ref{ch:b1:transient-regimes} --- الأنظمة العابرة: الرتبة الأولى والرتبة الثانية}

\begin{solution}{exo:b1:transient-regimes:1}
$\tau = RC = 10^4 \times 47\times10^{-9} = \qty{0.47}{ms}$؛ و$u_C(\tau)/E
= 1 - \eu^{-1} = 0.63$؛ ومن أجل $99\%$: $\eu^{-t/\tau} = 0.01$ و $t = \tau\ln 100
= 4.6\tau = \qty{2.2}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:2}
$\tau = L/R = \qty{4.0}{ms}$؛ و$i_\infty = E/R = \qty{2.0}{A}$؛ و$i(\tau) =
0.63 \times 2.0 = \qty{1.26}{A}$؛ وعند $0^+$ يكون $i = 0$ (بالاتصال)، ومنه
$u_L = E - Ri = \qty{10}{V}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:3}
$\omega_0 = 1/\sqrt{LC} = \qty{3.16e4}{rad/s}$ و $f_0 = \qty{5.03}{kHz}$؛
و$Q = \sqrt{L/C}/R = 316/100 = 3.2 > \tfrac12$: \omterm{thm:b1:transient-regimes:regimes}{فالنظام شبه دوري}؛
و$\tau = 2Q/\omega_0 = \qty{0.20}{ms}$؛ و$\omega = \omega_0\sqrt{1 - 1/4Q^2}
= 0.987\omega_0$ و $T = \qty{0.20}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:4}
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$ و $Q = \tfrac12$. وعند \qty{2}{k\ohm} يكون
$Q = 0.16$: \omterm{thm:b1:transient-regimes:regimes}{فالنظام لا دوري}، أي زحف بطيء يحكمه الجذر الصغير
$\abs{r_+} \approx \omega_0 Q = \qty{5e3}{s^{-1}}$ (بمقياس زمني
\qty{0.2}{ms} لكن دون تذبذب).
\end{solution}

\begin{solution}{exo:b1:transient-regimes:5}
التوتر $u = U_0\eu^{-t/\tau}$ يتنصّف حين $\eu^{-t/\tau} = \tfrac12$: أي $t_{1/2}
= \tau\ln 2$. ومنه $\tau = 3.5/0.693 = \qty{5.05}{ms}$ و $R = \tau/C =
\qty{5.1}{k\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:6}
$i = (E/R)\eu^{-t/\tau}$: ومنه $\int_0^\infty Ri^2\,\dd t = (E^2/R)(\tau/2) =
\tfrac12 CE^2$، ولا أثر للمقاومة $R$. ويقدّم المنبع $CE^2$: فالمردود
$50\%$ مهما تكن الناقلة. وأفضل من ذلك: الشحن عبر وشيعة (فلا يكون
التيار حينئذٍ متناسبًا مع فجوة التوتر) أو على عدة
قفزات توتر --- إذ لا تضيّع كل قفزة قدرها $E/n$ إلا $\tfrac12 C(E/n)^2$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:7}
$\delta = \ln(2.0/1.5) = 0.29$ (و$\ln(1.5/1.12) = 0.29$: فمتوافقان)؛
ومنه $Q \approx \pi/\delta = 11$. و$T = \qty{0.40}{ms}$، ومنه $\omega_0 \approx
2\pi/T = \qty{1.57e4}{rad/s}$ (والتصحيح $1/8Q^2$ يبلغ $0.1\%$).
و$L = 1/(\omega_0^2 C) = \qty{4.1}{mH}$؛ و$R = \omega_0 L/Q = \qty{5.8}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:8}
$u_C \approx U_0\eu^{-t/\tau}\cos\omega_0 t$؛ و$i = -C\,\dd u_C/\dd t
\approx CU_0\omega_0\sin\omega_0 t$ (من أجل $t \ll \tau$)، وذروته $U_0\sqrt{C/L}
= 100 \times 0.1 = \qty{10}{A}$، تُبلغ بعد ربع دور من الإغلاق،
حين $u_C = 0$: فتكون الطاقة كلها ($\tfrac12 CU_0^2 = \qty{50}{mJ}$)
مغناطيسية، $\tfrac12 Li^2 = \qty{50}{mJ}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:9}
$\omega_0 = \sqrt{k/m} = \qty{7.1}{rad/s}$ و $f_0 = \qty{1.1}{Hz}$؛
و$\alpha_c = 2\sqrt{km} = 2\sqrt{8\times10^6} = \qty{5.7e3}{N.s/m}$.
ومن أجل $\alpha = 2000$: $Q = \sqrt{km}/\alpha = 2828/2000 = 1.4$، \omterm{thm:b1:transient-regimes:regimes}{فالنظام شبه دوري}؛
و$T = 2\pi/(\omega_0\sqrt{1 - 1/4Q^2}) = 0.89/0.935 = \qty{0.95}{s}$؛
و$\tau = 2Q/\omega_0 = \qty{0.40}{s}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:10}
$i = I_0\eu^{-t/\tau}$ مع $\tau = L/R = \qty{5.0}{ms}$؛ والتوتر الأعظمي
$RI_0 = \qty{200}{V}$ عند $t = 0^+$؛ والطاقة $\tfrac12 LI_0^2 = \qty{1.0}{J}$
تتبدد في $R$. وبلا الناقلة لا يجد التيار مسارًا:
فيصير $\dd i/\dd t$ هائلًا، ويبلغ توتر الوشيعة آلاف الفولطات، فتقفز شرارة
على القاطع.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:11}
منظورًا من $C$: $E_{\mathrm{Th}} = ER_2/(R_1 + R_2) = \qty{8.0}{V}$ و
$R_{\mathrm{Th}} = R_1 \parallel R_2 = \qty{6.67}{k\ohm}$؛ ومنه $\tau =
R_{\mathrm{Th}}C = \qty{6.7}{ms}$ و $u_C = 8.0(1 - \eu^{-t/\tau})$ فولط.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:12}
الجذر المضاعف $-\omega_0$: $x = x_\infty + (A + Bt)\eu^{-\omega_0 t}$؛
و$x(0) = 0$ يعطي $A = -x_\infty$، و$x'(0) = 0$ يعطي $B = \omega_0 A$:
ومنه $x = x_\infty[1 - (1 + \omega_0 t)\eu^{-\omega_0 t}]$. ومن أجل $95\%$:
$(1 + s)\eu^{-s} = 0.05$ و $s \approx 4.7$: أي $t_{95} = 4.7/\omega_0$. ومن أجل
$Q = 5$ يحتاج الغلاف $\eu^{-t/\tau}$ حيث $\tau = 10/\omega_0$ إلى
$3\tau = 30/\omega_0$؛ ومن أجل $Q = 0.1$ يكون الجذر البطيء $\abs{r_+} \approx
\omega_0 Q$ و $3/\abs{r_+} = 30/\omega_0$. وكلاهما أبطأ ست مرات:
فالتخميد الحرج هو أسرع عودة بلا تجاوز، وهو ما يريده
مقياسٌ أو مغلاق باب أو نظام تعليق.
\end{solution}

\begin{solution}{pb:b1:transient-regimes:1}
\textbf{1.} $E = L\,\dd i/\dd t + Ri + u_C$ مع $i = C\,\dd u_C/\dd t$:
ومنه $LC\,u_C'' + RC\,u_C' + u_C = E$.

\textbf{2.} $u_C'' + (\omega_0/Q)u_C' + \omega_0^2u_C = \omega_0^2E$ مع
$\omega_0 = 1/\sqrt{LC}$ و $Q = \sqrt{L/C}/R$.

\textbf{3.} $\omega_0 = \qty{3.16e4}{rad/s}$, $f_0 = \qty{5.03}{kHz}$,
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$.

\textbf{4.} $r^2 + (\omega_0/Q)r + \omega_0^2 = 0$ و $\Delta = \omega_0^2(1/Q^2
- 4)$: فمن أجل $Q < \tfrac12$ لا دوري، ومن أجل $Q = \tfrac12$ حرج، ومن أجل $Q > \tfrac12$
شبه دوري.

\textbf{5.} $\tau = 2Q/\omega_0$ و $\omega = \omega_0\sqrt{1 - 1/4Q^2}$ و
$u_C = E + \eu^{-t/\tau}(A\cos\omega t + B\sin\omega t)$.

\textbf{6.} $Q = 3.16$ و $\tau = \qty{0.20}{ms}$ و $T = 2\pi/\omega =
\qty{0.20}{ms}$، ونحو ثلاث تذبذبات.

\textbf{7.} $u_C(0^+) = U_0$ (لأن $u_C$ متصل: فالتيار منتهٍ)؛
و$u_C'(0^+) = i(0^+)/C = 0$ (لأن $i_L$ متصل وكان معدومًا قبلًا).

\textbf{8.} $m\ddot x = -k(x + x_{\text{توازن}}) - \alpha\dot x + mg$ مع
$kx_{\text{توازن}} = mg$: ومنه $m\ddot x + \alpha\dot x + kx = 0$.

\textbf{9.} $\omega_0 = \sqrt{k/m}$ و $Q = \sqrt{km}/\alpha$؛ و$x \leftrightarrow
q$ و $\dot x \leftrightarrow i$ و $m \leftrightarrow L$ و $\alpha \leftrightarrow
R$ و $k \leftrightarrow 1/C$.

\textbf{10.} $\omega_0 = \sqrt{22000/350} = \qty{7.93}{rad/s}$ و $f_0 =
\qty{1.26}{Hz}$؛ و$\alpha_c = 2\sqrt{km} = \qty{5.55e3}{N.s/m}$.

\textbf{11.} $Q = 2775/4000 = 0.69$ و $\xi = 0.72$: \omterm{thm:b1:transient-regimes:regimes}{فالنظام شبه دوري}، دون
الحرج بقليل.

\textbf{12.} $\tau = 2Q/\omega_0 = \qty{0.18}{s}$؛ و$\omega = \omega_0\sqrt{1
- \xi^2} = \qty{5.5}{rad/s}$ و $T = \qty{1.1}{s}$.

\textbf{13.} التجاوز $\exp(-\pi \times 0.72/0.69) = \exp(-3.3) = 3.8\%$:
أي ارتفاع صغير واحد بعد التوازن، ثم انتهى --- فهي مريحة.

\textbf{14.} $R = \sqrt{L/C}/Q = 316/0.69 = \qty{455}{\ohm}$.

\textbf{15.} $Q = 2775/1200 = 2.3$ و $\xi = 0.22$؛ و$\omega = 0.976\omega_0$ و
$T = \qty{0.81}{s}$؛ و$\tau = 2Q/\omega_0 = \qty{0.58}{s}$.

\textbf{16.} $\eu^{-T/\tau} = \eu^{-1.39} = 0.25$: فكل ارتجاجة ربع
سابقتها؛ و$0.25^3 = 1.6\%$: أي ثلاث ارتجاجات مرئية.

\textbf{17.} $x(0) = -x_0$ و $\dot x(0) = 0$؛ والطاقة $\tfrac12 kx_0^2 =
0.5 \times 22000 \times 0.0025 = \qty{28}{J}$، تتبدد حرارةً في
زيت المخمّد.

\textbf{18.} $v \approx x_0\omega_0 = 0.05 \times 7.93 = \qty{0.40}{m/s}$؛
وقوة المخمّد $\alpha v \approx 1200 \times 0.40 = \qty{480}{N}$.

\textbf{19.} عند $Q = \tfrac12$ تكون العودة $(1 + \omega_0 t)\eu^{-\omega_0
t}$، وهي تزحف زحفًا مقاربًا: فالمليمترات الأخيرة تطول. أما
جملة أقلّ تخميدًا قليلًا ($\xi \approx 0.7$) فتتجاوز ببضعة
بالمئة لكنها تدخل نطاق التسامح وتبقى فيه أسرع.

\textbf{20.} $\exp(-\pi \times 0.22/0.976) = \exp(-0.70) = 50\%$، في مقابل
$3.8\%$: فالسيارة البالية ترتدّ نصف الطريق فوق التوازن.

\textbf{21.} $\delta = \ln 4 = 1.39$ (مرتين، فمتوافق)؛ و$\xi =
\delta/\sqrt{4\pi^2 + \delta^2} = 1.39/6.43 = 0.215$ و $Q = 1/2\xi = 2.3$:
وهي القيمة البالية.

\textbf{22.} $-\ln 0.038 = 3.27 = \pi\xi/\sqrt{1 - \xi^2}$، ومنه $\xi =
3.27/\sqrt{\pi^2 + 3.27^2} = 0.72$ و $Q = 0.69$.

\textbf{23.} الطاقة $\propto$ السعة$^2$: فالكسر الباقي في كل
دور $0.25^2 = 6\%$، أي إن $94\%$ تتبدد في كل شبه دور.

\textbf{24.} $R = 316/2.3 = \qty{137}{\ohm}$. وفي المتغير $\omega_0 t$
لا تحتوي المعادلة النظامية إلا على $Q$: فجملتان لهما $Q$ نفسه
ترسمان المنحني نفسه.

\textbf{25.} \omterm{def:b1:transient-regimes:secondorder}{عامل الجودة} $Q$ (أو $\xi$) وحده يثبّت الشكل؛ و
المصمّم يستهدف $Q \approx 0.7$ (أي $\xi \approx 0.7$)؛ وأكثر من ارتجاجة
مرئية واحدة يعني $Q$ أعلى بكثير من $1$: أي إن المخمّد بالٍ.
\end{solution}
