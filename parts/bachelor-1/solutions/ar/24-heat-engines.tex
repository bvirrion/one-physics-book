\begin{solution}{exo:b1:heat-engines:8}
$Q_h = RT_h\ln2 = 8.314 \times 600 \times 0.693 = \qty{3.46}{kJ}$ و $Q_c = -8.314

\begin{solution}{exo:b1:heat-engines:9}
إنتروبيا الكتلتين: $\Delta S = C\ln(T_f/T_1) + C\ln(T_f/T_2) =
C\ln\bigl(T_f^2/T_1T_2\bigr) = S_{\mathrm{created}} \geq 0$ (فإنتروبيا المحرك نفسه
تعود إلى قيمتها في كل دور)، ومنه $T_f \geq \sqrt{T_1T_2}$، ويكون
$W = C(T_1 + T_2 - 2T_f)$ (بالمبدأ الأول) أكبر ما يكون عند أصغر $T_f$:
أي في الحالة العكوسة، $T_f = \sqrt{400 \times 300} = \qty{346.4}{K}$،
و $W_{\max} = 4000 \times (700 - 692.8) = \qty{29}{kJ}$. وفي التماسّ المباشر: $T_f =
\qty{350}{K}$، بلا شغل، و $S_{\mathrm{created}} = 4000\ln(350^2/120000) =
\qty{82}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:heat-engines:10}
(أ) $P = \dot Q_h - |\dot Q_c| = K(T_h - x) - K(y - T_c)$؛ ولا تتبادل النواة العكوسة
أيّ إنتروبيا صافية: $K(T_h - x)/x = K(y - T_c)/y$. (ب) بالضرب التبادلي:
$y(T_h - x) = x(y - T_c)$، ومنه $y(T_h - 2x) = -xT_c$ و $y = xT_c/(2x - T_h)$.
ثم $P = K(T_h - x)(1 - y/x) = K(T_h - x)(2x - T_h - T_c)/(2x - T_h)$؛ ومع
$u = 2x - T_h$ يكون $T_h - x = (T_h - u)/2$ و $P = K(T_h - u)(u - T_c)/(2u) =
\tfrac{K}{2}[(T_h + T_c) - u - T_hT_c/u]$. (ج) $\dd P/\dd u = \tfrac K2(-1 +
T_hT_c/u^2) = 0$ عند $u = \sqrt{T_hT_c}$، وهو أعظمي؛ وعندئذ $x = (T_h +
\sqrt{T_hT_c})/2$ و $y = xT_c/u = (\sqrt{T_hT_c} + T_c)/2$، ويكون $\eta^\ast = 1 - y/x
= 1 - \sqrt{T_c}(\sqrt{T_h} + \sqrt{T_c})/[\sqrt{T_h}(\sqrt{T_h} + \sqrt{T_c})] = 1 -
\sqrt{T_c/T_h}$؛ و $P_{\max} = \tfrac K2(\sqrt{T_h} - \sqrt{T_c})^2$. (د)
$\eta^\ast = 1 - \sqrt{0.375} = 0.39$، مقابل $\eta_C = 0.625$: \omterm{def:b1:heat-engines:efficiency}{فالمردود} الحقيقي $0.40$
في \cref{exo:b1:heat-engines:1} محرك بُني للاستطاعة لا
\omterm{def:b1:heat-engines:efficiency}{للمردود}.
\end{solution}

\begin{solution}{exo:b1:heat-engines:11}
في دور: $Q_h + Q_c + Q_0 = 0$ و $Q_h/T_h + Q_c/T_c + Q_0/T_0 \leq 0$.
ونحذف $Q_0 = -(Q_h + Q_c)$: فيكون $Q_h(1/T_h - 1/T_0) + Q_c(1/T_c - 1/T_0) \leq 0$،
أي $Q_c(T_0 - T_c)/(T_cT_0) \leq Q_h(T_h - T_0)/(T_hT_0)$، ومنه
$Q_c/Q_h \leq (1 - T_0/T_h)\,T_c/(T_0 - T_c)$. والعامل الأول هو
\omterm{def:b1:heat-engines:efficiency}{مردود} محرك كارنو بين الموقد والغرفة، والثاني
معامل أداء \omterm{def:b1:heat-engines:machine}{ثلاجة} كارنو بين الحجرة والغرفة: فالحدّ هو ما
يُنال بربط الاثنين ربطًا مثاليًّا. والأعداد: $0.25 \times 11 = 2.75$
(أما ثلاجات الامتصاص الحقيقية فنحو $0.6$).
\end{solution}

\begin{solution}{pb:b1:heat-engines:1}
\textbf{1.} $P = 250 \times 25 = \qty{6.25}{kW}$.

\textbf{2.} متوسط الفقد $250 \times 13 = \qty{3.25}{kW}$ على امتداد $200 \times 86400 =
\qty{1.73e7}{s}$: ومنه $E = \qty{5.6e10}{J} = \qty{15600}{kW h}$.

\textbf{3.} $15600 \times 0.25 = 3900$~\texteuro.

\textbf{4.} الغاز $15600/0.90 = \qty{17300}{kW h}$: أي $1730$~\texteuro.

\textbf{5.} بالمقاومات: $15600/0.40 = \qty{39000}{kW h}$ من الوقود؛ وبالمِرجل
\qty{17300}{kW h}: أي أقل بعامل قدره $2.25$. فالكهرباء شغل، وهو الصورة التي
تستطيع كل شيء؛ وتحويلها إلى حرارة عند \qty{20}{\degreeCelsius} في مقاومة
يرمي $60\%$ من الوقود الضائع أصلًا في المحطة، ويرمي معه
إمكان ضخّ الحرارة.

\textbf{6.} على امتداد دور: $W + Q_h + Q_c = 0$ و $Q_h/T_{\mathrm{in}} +
Q_c/T_{\mathrm{out}} \leq 0$ مع $Q_h < 0$ و $Q_c > 0$: ومنه $Q_c \leq
|Q_h|T_{\mathrm{out}}/T_{\mathrm{in}}$، فيكون $W = |Q_h| - Q_c \geq |Q_h|(1 -
T_{\mathrm{out}}/T_{\mathrm{in}})$ و $e_p = |Q_h|/W \leq T_{\mathrm{in}}/(T_{\mathrm{in}} -
T_{\mathrm{out}})$.

\textbf{7.} $293/25 = 11.7$؛ ومنه $6.25/11.7 = \qty{0.53}{kW}$.

\textbf{8.} $293/13 = 22.5$؛ ومنه $15600/22.5 = \qty{690}{kW h}$ (أي نحو
$170$~\texteuro).

\textbf{9.} الشغل لكل جول مرفوع هو $(T_{\mathrm{in}} - T_{\mathrm{out}})/
T_{\mathrm{in}}$، وهو ``ارتفاع'' الرفع؛ فكلما ازداد الخارج بردًا
ازداد الارتفاع الذي يجب رفع كل جول إليه --- وازداد عدد الجولات اللازمة، إذ
يتناسب الفقد أيضًا مع $T_{\mathrm{in}} - T_{\mathrm{out}}$: فتنمو
\omterm{thm:b1:dc-circuits:kirchhoff}{الاستطاعة الكهربائية} كمربع فرق درجتَي الحرارة، وتسوء
عند اللحظة التي تكون فيها التدفئة أشدّ ما تكون أهمية.

\textbf{10.} من الهواء الخارجي، الذي يبرد: $Q_c = 6.25 - 0.53 =
\qty{5.7}{kW}$. والمبدأ الثاني يمنع جريان الحرارة \emph{تلقائيًّا}
من البارد إلى الساخن؛ وهنا يدفع الشغل ثمن الرفع. وللتحقق: الإنتروبيا المأخوذة من
الهواء $5.72/268 = \qty{21.3}{W/K}$، والمعطاة للبيت $6.25/293 =
\qty{21.3}{W/K}$ --- فلا شيء أُفني ولا شيء أُنشئ: تلك هي المضخة المثالية.

\textbf{11.} الضاغط (يتلقى $W$)، والمكثِّف (يعطي $|Q_h|$)،
وصمام التمدد (لا هذا ولا ذاك)، والمبخِّر (يتلقى $Q_c$). ولا تجري الحرارة إلا
نازلةً في ميل حراري: فعلى \omterm{def:b1:heat-engines:vapourcycle}{المائع المبرِّد} أن يكون أبرد من الهواء كي
يأخذ منه الحرارة، وأسخن من ماء البيت كي يعطيه الحرارة.

\textbf{12.} $T_{\mathrm{ev}} = 268 - 8 = \qty{260}{K}$؛ و $e_{p,C} = 313/(313 - 260)
= 5.9$.

\textbf{13.} $e_p = 0.55 \times 5.9 = 3.25$؛ و $W = 6.25/3.25 = \qty{1.92}{kW}$؛
و $Q_c = 6.25 - 1.92 = \qty{4.33}{kW}$ من الهواء.

\textbf{14.} $\dot m = 4330/200000 = \qty{22}{g/s}$، أي \qty{1.3}{kg} في الدقيقة.

\textbf{15.} $T_{\mathrm{ev}} = \qty{272}{K}$ و $313/41 = 7.6$ و $e_p = 4.2$؛
ومنه $W = 3.25/4.2 = \qty{0.77}{kW}$.

\textbf{16.} $e_p = 0.55 \times 313/(321 - T_{\mathrm{out}}) = 172/(321 -
T_{\mathrm{out}})$ (مع $T_{\mathrm{out}}$ بالكلفن). وفي الموسم: $15600/4.2 =
\qty{3700}{kW h}$، أي $930$~\texteuro\ --- مقابل $3900$ بالمقاومات و
$1730$ بالغاز.

\textbf{17.} $3700/0.40 = \qty{9300}{kW h}$ من الوقود: أي أقل بنسبة $46\%$ من
المِرجل وأقل أربع مرات من المقاومات، وذلك عبر محطة توليد
مردودها $40\%$.

\textbf{18.} الطلب $250 \times 35 = \qty{8.75}{kW}$؛ و $T_{\mathrm{ev}} = \qty{250}{K}$
و $313/63 = 5.0$ و $e_p = 2.7$ و $W = \qty{3.2}{kW}$ لو استطاعت. أما السعة فهي
\qty{3.8}{kW}: فالاحتياطي $8.75 - 3.8 \approx \qty{5}{kW}$ من المقاومات. ومضخة
مقيسة على أبرد ساعة تكون مفرطة الحجم --- فتشتغل بتقطّع
وبلا \omterm{def:b1:heat-engines:efficiency}{مردود} --- طوال الشتاء؛ فيُقاس الجهاز على \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} التصميم
وتُترك مقاومة رخيصة تغطي النادر الأقصى.

\textbf{19.} $338/(338 - 260) = 4.33$ و $e_p = 2.4$ و $W = 6.25/2.4 =
\qty{2.6}{kW}$: أي كهرباء أكثر بنسبة $36\%$ مما مع الأرضية عند
\qty{35}{\degreeCelsius}. فأجهزة البثّ المنخفضة الحرارة نصف
التركيب.

\textbf{20.} الإنتروبيا على امتداد دور: $Q_h/T_{\mathrm{in}} + Q_c/T_{\mathrm{out}} +
S_{\mathrm{created}} = 0$ يعطي $Q_c = T_{\mathrm{out}}|Q_h|/T_{\mathrm{in}} -
T_{\mathrm{out}}S_{\mathrm{created}}$؛ فعوّض في $W = |Q_h| - Q_c$.

\textbf{21.} المكثِّف: $6250\,(1/293 - 1/313) = \qty{1.36}{W/K}$؛ والمبخِّر:
$4330\,(1/260 - 1/268) = \qty{0.50}{W/K}$.

\textbf{22.} $268 \times 1.86 = \qty{0.50}{kW}$: أي ربع
\qty{1.92}{kW}.

\textbf{23.} $(1.92 - 0.53)/268 = \qty{5.2}{W/K}$؛ ويصنع المبادلان منها
$36\%$؛ ويأتي الباقي من الضاغط (الاحتكاك، والانضغاط غير
المتساوي الإنتروبيا، وضياعات المحرك)، ومن الخنق في الصمام (وهو تمدد
حرّ)، ومن هبوط الضغط، ومن المراوح.

\textbf{24.} $T_{\mathrm{cd}} = \qty{303}{K}$ و $T_{\mathrm{ev}} = \qty{264}{K}$:
فالمكثِّف $6250\,(1/293 - 1/303) = \qty{0.70}{W/K}$، والمبخِّر
$4330\,(1/264 - 1/268) = \qty{0.25}{W/K}$: أي \qty{0.95}{W/K} بدل
$1.86$، ومنه $268 \times 0.91 \approx \qty{0.25}{kW}$ موفَّرة (أي $13\%$؛ أما
الآلة كلها، وهي عند $0.55$ من قيمة كارنو أفضل، فتكسب
أكثر). وتنصيف فرق يعني مضاعفة سطح التبادل: أي مبادلات أكبر
وأغلى واستطاعة مراوح أكبر.

\textbf{25.} $15600/3700 = 4.2$ كيلوواط ساعي من الحرارة لكل كيلوواط ساعي
من الكهرباء؛ و $930$~\texteuro\ مقابل $1730$ للغاز؛ والكربون: $3700
\times 0.060 = \qty{220}{kg}$ مقابل $17300 \times 0.20 = \qty{3500}{kg}$ ---
أي أقل خمس عشرة مرة.
\end{solution}
