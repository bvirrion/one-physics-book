\section*{الفصل \ref{ch:b1:sinusoidal-impedance} --- النظام الجيبي الدائم والممانعة}

\begin{solution}{exo:b1:sinusoidal-impedance:1}
$\underline U = 5\eu^{j\pi/3} = 2.5 + 4.33j$ (بالفولط). ومن أجل $3\cos\omega t +
4\sin\omega t$: $3 + 4\eu^{-j\pi/2} = 3 - 4j$، ومقياسه $5$ وعمدته
$-\arctan(4/3) = \qty{-0.927}{rad}$: أي $5\cos(\omega t - 0.927)$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:2}
$R$: \qty{100}{\ohm} عند التواترين. و$L\omega$: \qty{31.4}{\ohm} (عند \qty{50}{Hz})
و\qty{628}{\ohm} (عند \qty{1}{kHz}). و$1/C\omega$: \qty{318}{\ohm} و\qty{15.9}{\ohm}.
و$RL$ على التسلسل عند \qty{50}{Hz}: $\abs{\underline Z} = \sqrt{100^2 + 31.4^2} =
\qty{105}{\ohm}$، والطور $\arctan(0.314) = 17^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:3}
$U = 230\sqrt2 = \qty{325}{V}$. و$I_{\mathrm{rms}} = 2000/230 = \qty{8.7}{A}$،
والأعظمي \qty{12.3}{A}. و$\langle Ri^2\rangle = R\langle i^2\rangle =
RI_{\mathrm{rms}}^2$ بتعريف \omterm{def:b1:sinusoidal-impedance:rms}{القيمة الفعّالة}.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:4}
$f_0 = 1/(2\pi\sqrt{LC}) = \qty{5.03}{kHz}$؛ و$Q = \sqrt{L/C}/R = 3.16$؛
و$\Delta f = f_0/Q = \qty{1.6}{kHz}$؛ و$I_{\max} = E/R = \qty{10}{mA}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:5}
النسبة $1/\sqrt2$ عند $\omega = 1/RC$: أي $f_c = 1/(2\pi RC) = \qty{1.0}{kHz}$. وعند
\qty{1.0}{kHz}: النسبة $0.71$ والطور $-45^\circ$. وعند \qty{10}{kHz}:
$RC\omega = 10$ والنسبة $1/\sqrt{101} = 0.10$ والطور $-84^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:6}
$\underline Z = 100 + j(250 - 100) = 100 + 150j$: ومنه $Z = \qty{180}{\ohm}$ و
$\arg = 56^\circ$. و$I = 1.0/180 = \qty{5.6}{mA}$، متأخرًا عن المنبع بمقدار
$56^\circ$. و$U_R = \qty{0.56}{V}$ و $U_L = \qty{1.39}{V}$ و $U_C = \qty{0.56}{V}$؛
والمتجهات: $U_R$ على امتداد $I$، و$U_L$ إلى أعلى، و$U_C$ إلى أسفل، والمحصلة \qty{1.0}{V}.
و$L\omega > 1/C\omega$: أي فوق الرنين (تحريضي).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:7}
لدينا $f_0 = 1/(2\pi\sqrt{LC})$: أي من \qty{1.59}{MHz} عند \qty{50}{pF}
إلى \qty{0.50}{MHz} عند \qty{500}{pF}. وعند \qty{1.00}{MHz} يكون
$C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$.
و$\Delta f = \qty{10}{kHz}$. والمحطة عند \qty{1.009}{MHz}: $x = 1.009$ و
$Q(x - 1/x) = 1.8$ والنسبة $1/\sqrt{1 + 3.2} = 0.49$. وتوتر المكثف
$QE = \qty{1.0}{mV}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:8}
$Q(x - 1/x) = \pm1 \Leftrightarrow x^2 \mp x/Q - 1 = 0$؛ والجذران الموجبان
$x_{1,2} = \mp 1/2Q + \sqrt{1 + 1/4Q^2}$. والفرق $1/Q$، ومنه
$\Delta\omega = \omega_0/Q$؛ والجداء $(1 + 1/4Q^2) - 1/4Q^2 = 1$، ومنه
$\omega_1\omega_2 = \omega_0^2$ (فالنطاق متناظر على سُلّم لوغاريتمي).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:9}
$I = P/(U\cos\varphi) = 2000/(230 \times 0.70) = \qty{12.4}{A}$. \omterm{thm:b1:sinusoidal-impedance:power}{والاستطاعة
التفاعلية} $P\tan\varphi = 2000 \times 1.02 = \qty{2.0}{kvar}$. و$C = 2040/(314
\times 230^2) = \qty{123}{\micro F}$. والتيار الجديد $2000/230 = \qty{8.7}{A}$.
والضياعات $0.50 \times 12.4^2 = \qty{77}{W}$ قبل ذلك، و$0.50 \times 8.7^2 =
\qty{38}{W}$ بعده.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:10}
$\underline Y = 1/R + j(C\omega - 1/L\omega)$؛ و$\underline U = \underline I/
\underline Y$ أعظمي حين ينعدم الجزء التخيلي من \omterm{def:b1:sinusoidal-impedance:impedance}{المسايرة}، أي عند $\omega_0$،
حيث $U = RI$. وعند نصف الاستطاعة: $\abs{C\omega - 1/L\omega} = 1/R$، أي
$RC\omega_0(x - 1/x) = \pm1$: ومنه $\Delta\omega = \omega_0/(RC\omega_0) = 1/RC$
و $Q = RC\omega_0 = R\sqrt{C/L}$. والأعداد: $\omega_0 = \qty{1.0e6}{rad/s}$
(أي $f_0 = \qty{159}{kHz}$)، و $Q = 10^4\sqrt{10^{-6}} = 10$، و $\Delta f =
\qty{16}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:11}
$U_C = I/C\omega = E/[C\omega\sqrt{R^2 + (L\omega - 1/C\omega)^2}]$؛
وبالضرب داخل الجذر في $C\omega$: $\sqrt{(RC\omega)^2 + (LC\omega^2 - 1)^2} =
\sqrt{x^2/Q^2 + (1 - x^2)^2}$. ومع $s = x^2$ يكون $(1 - s)^2 + s/Q^2$
أصغريًّا عند $s = 1 - 1/2Q^2$ (إن كان موجبًا)، وأصغره $1/Q^2 - 1/4Q^4$:
ومنه $U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$. ومن أجل $Q = 3.16$: $x_r = 0.975$ و
$U_{C,\max} = 3.2E$. أما من أجل $Q < 1/\sqrt2$ فالأصغر عند $s = 0$: فتتناقص $U_C$
تناقصًا رتيبًا من $E$، بلا ذروة.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:12}
عند الرنين يكون $i = I\cos\omega_0 t$ و $u_C = (I/C\omega_0)\sin\omega_0 t$،
ومنه، باستعمال $1/C\omega_0^2 = L$،
\[
\tfrac12 Li^2 + \tfrac12 Cu_C^2 = \tfrac12 LI^2\cos^2\omega_0 t
+ \tfrac12 LI^2\sin^2\omega_0 t = \tfrac12 LI^2 .
\]
والمتبدد في كل دور: $\tfrac12 RI^2T$. والنسبة $\times 2\pi$: $2\pi L/(RT)
= L\omega_0/R = Q$.
\end{solution}

\begin{solution}{pb:b1:sinusoidal-impedance:1}
\textbf{1.} $\underline Z = R + j(L\omega - 1/C\omega)$.

\textbf{2.} $\underline I = E/\underline Z$؛ و$I = E/\sqrt{R^2 + (L\omega -
1/C\omega)^2}$؛ و$\varphi_i = -\arctan[(L\omega - 1/C\omega)/R]$.

\textbf{3.} تنعدم \omterm{def:b1:sinusoidal-impedance:impedance}{المفاعلة} عند $\omega_0 = 1/\sqrt{LC}$: ومنه $I_{\max}
= E/R$ و $\varphi_i = 0$.

\textbf{4.} $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$؛ و$L\omega_0 = 2\times10^{-4}
\times 6.28\times10^6 = \qty{1257}{\ohm}$.

\textbf{5.} $Q = 1257/12.6 = 100$.

\textbf{6.} $L\omega - 1/C\omega = L\omega_0(x - 1/x)$، ثم نقسم على $R$.

\textbf{7.} من أجل $x < 1$: \omterm{def:b1:sinusoidal-impedance:impedance}{المفاعلة} سالبة و $\varphi_i > 0$، فيتقدّم التيار
(سلوك سعوي)؛ ومن أجل $x > 1$: يتأخر (سلوك تحريضي).

\textbf{8.} $Q(x - 1/x) = \pm1$.

\textbf{9.} $x^2 \mp x/Q - 1 = 0$ و $x_{1,2} = \mp1/2Q + \sqrt{1 + 1/4Q^2}$ و
$x_2 - x_1 = 1/Q$: ومنه $\Delta\omega = \omega_0/Q$.

\textbf{10.} $\Delta f = \qty{10}{kHz}$.

\textbf{11.} عند \qty{1.010}{MHz}: $Q(x - 1/x) = 100 \times 0.0199 = 2.0$ و
النسبة $1/\sqrt{5.0} = 0.45$ والاستطاعة $0.20$: أي \qty{-7}{dB}. وعند \qty{1.020}{MHz}:
$3.96$ والنسبة $0.25$ والاستطاعة $0.06$: أي \qty{-12}{dB}.

\textbf{12.} على الحدّ: فالجارة لا تنخفض إلا \qty{7}{dB}. وعامل جودة
$Q$ أعلى (أي $R$ أدنى)، أو عدة مراحل مضبوطة على التتالي، يشحذ
الانتقاء.

\textbf{13.} من \qty{0.50}{MHz} إلى \qty{1.59}{MHz}: أي النطاق الإذاعي.

\textbf{14.} $\underline U_C = \underline I/jC\omega$؛ وعند $\omega_0$ يكون
$U_C = I_{\max}/C\omega_0 = E/(RC\omega_0) = QE$.

\textbf{15.} $I_{\max} = 10^{-5}/12.6 = \qty{0.79}{\micro A}$؛ و$U_C = QE =
\qty{1.0}{mV}$: أي ربح توتر مئة ضعف بلا مضخّم.

\textbf{16.} $\underline U_L = jL\omega_0\underline I = jQE$ و $\underline U_C
= -jQE$: متساويتان متعاكستان فتتلاشيان؛ فيرى المنبع $R$ وحدها.

\textbf{17.} المخزونة $\tfrac12 LI_{\max}^2 = \qty{6.2e-17}{J}$؛ والمتبددة
في كل دور $\tfrac12 RI_{\max}^2T = \qty{3.9e-18}{J}$؛ و$2\pi \times 6.2/0.39
\approx 100 = Q$.

\textbf{18.} $\tau = 2Q/\omega_0 = \qty{32}{\micro s}$. ولا تستطيع الدارة
متابعة تغيّرات أسرع من $\sim\tau$: فعرض نطاق $\Delta f$ لا يمرّر
إلا تضمينات أبطأ من $\Delta f$؛ ومع $Q = 1000$ يكون $\Delta f =
\qty{1}{kHz}$ فتضيع الأصوات الحادّة في البرنامج.

\textbf{19.} ينتصف $Q$ إلى $50$، ويتضاعف $\Delta f$ إلى \qty{20}{kHz}، وينتصف $U_C$
إلى \qty{0.5}{mV}.

\textbf{20.} $R_{\text{كلي}} = \qty{62.6}{\ohm}$: ومنه $Q = 1257/62.6 = 20$ و
$\Delta f = \qty{50}{kHz}$ و $U_C = 20 \times \qty{10}{\micro V} = \qty{0.2}{mV}$:
فتنهار الحساسية والانتقائية معًا.

\textbf{21.} $P = E^2/(2R_{\text{كلي}}) = 10^{-10}/125 = \qty{8e-13}{W}$؛
والكسر في $R_a$: $50/62.6 = 80\%$.

\textbf{22.} يطلب التكييف مقاومة حمولة تساوي $R_a$، أي
مقاومة كلية كبيرة؛ وتطلب الانتقائية أصغر مقاومة كلية
ممكنة (لأن $Q = L\omega_0/R_{\text{كلي}}$). فلا يمكن الجمع بينهما بوصل
مباشر.

\textbf{23.} المأخذ أو مكثف الاقتران الصغير يقدّم إلى الهوائي
جزءًا فقط من الدارة الرنّانة: فتُرى مقاومة الهوائي
من الدارة مخفَّضة كثيرًا، فيبقى $Q$ عاليًا، بينما يظلّ
الهوائي يقدّم استطاعة قرب الأمثل.

\textbf{24.} عند تواترَي نصف الاستطاعة تساوي \omterm{def:b1:sinusoidal-impedance:impedance}{المفاعلة} $\pm R$،
فيكون $\varphi = \pm45^\circ$ و $\cos\varphi = 1/\sqrt2$؛ ومع $I =
I_{\max}/\sqrt2$ أيضًا تكون $P = \tfrac12 EI\cos\varphi$ نصف قيمتها
عند الرنين.

\textbf{25.} يضرب $Q$ الإشارة في $Q$ ويضيّق النطاق إلى
$f_0/Q$، بثمن استجابة أبطأ ($\tau = 2Q/\omega_0$) و
اقتران هشّ بالهوائي؛ ومن أجل قنوات عرضها \qty{10}{kHz} عند
\qty{1}{MHz} يكون $Q \approx 100$ هو العدد المطلوب --- ويجب أن يُقرن الهوائي
اقترانًا خفيفًا للحفاظ عليه.
\end{solution}
