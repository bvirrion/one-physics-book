\section*{الفصل \ref{ch:b1:mirrors-thin-lenses} --- المرايا والعدسات الرقيقة}

\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$؛ و$1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. فعند \qty{60}{cm}: $\overline{SA'} = \qty{-30}{cm}$،
حقيقية، $\gamma = -0.5$. وعند \qty{30}{cm}: $\overline{SA'} = \qty{-60}{cm}$،
حقيقية، $\gamma = -2$. وعند \qty{10}{cm}: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$، ومنه $\overline{SA'} = \qty{+20}{cm}$ خلف المرآة، تخيلية،
$\gamma = +2$ (منتصبة مكبَّرة).
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:2}
$\overline{OA} = \qty{-20}{cm}$: $1/\overline{OA'} = 1/5 - 1/20 = 3/20$،
ومنه $\overline{OA'} = \qty{+6.7}{cm}$، حقيقية مقلوبة، $\gamma = -1/3$.
و$\overline{OA} = \qty{-3.0}{cm}$: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$،
ومنه $\overline{OA'} = \qty{-7.5}{cm}$، تخيلية منتصبة، $\gamma = +2.5$.
\end{solution}

و$V = 3.0 - 1.0 = \qty{+2.0}{\delta}$، ومنه $f' = \qty{50}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:4}
$\overline{FA} = \qty{-15}{cm}$: $\overline{F'A'} = -f'^2/\overline{FA}
= +100/15 = \qty{6.7}{cm}$ خلف $F'$، أي $\overline{OA'} = \qty{16.7}{cm}$؛
و$\gamma = f'/\overline{FA} = -0.67$. وبعلاقة ديكارت: $\overline{OA} =
\qty{-25}{cm}$ و $1/\overline{OA'} = 1/10 - 1/25 = 3/50$، ومنه
$\overline{OA'} = \qty{16.7}{cm}$ و $\gamma = 16.7/(-25) = -0.67$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}
$\overline{SF} = \qty{+1.0}{m}$ و $\overline{SA} = \qty{-20}{m}$:
$1/\overline{SA'} = 1 + 1/20$، ومنه $\overline{SA'} = \qty{0.95}{m}$ خلف
المرآة (تخيلية منتصبة)؛ و$\gamma = -0.95/(-20) = +0.048$، فارتفاع \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}
\qty{7.1}{cm}. والدماغ يحكم على المسافة من الحجم الظاهري: فسيارة
تظهر أصغر عشرين مرة على \qty{0.95}{m} تُقرأ سيارةً بعيدة
--- أبعد من \qty{20}{m}.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$، ومنه
$\overline{OA'} = \qty{51.3}{mm}$: أي تبتعد \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} \qty{1.3}{mm} عن
المستقبِل. و$\gamma = 51.3/(-2000) = -0.026$. ومن أجل $\abs\gamma = 24/1800
= 1/75$: $\overline{OA} = f'(1/\gamma - 1) = 50 \times (-76) =
\qty{-3.8}{m}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$؛ و$\overline{OA'} = \qty{5.0}{m}$، ومنه
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$؛
و$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$، ومنه $f' = \qty{98}{mm}$. \omterm{def:b1:mirrors-thin-lenses:image}{والصورة}
مقلوبة ($\gamma < 0$)، أعلى--أسفل ويمينًا--يسارًا: فتُدخل الشريحة
مُدارة بمقدار $180^\circ$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
الجسم في المستوي البؤري $\Rightarrow$ تغادر الأشعة الصادرة عن كل نقطة
العدسةَ متوازية؛ وترسلها \omterm{prop:b1:rays-reflection-refraction:mirror}{المرآة المستوية} راجعةً متوازية (في اتجاه
متناظر)؛ وتجمّع \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} حزمة متوازية في مستويها البؤري، عند
النقطة المناظرة لنقطة المنبع بالنسبة إلى المحور: أي \omterm{def:b1:mirrors-thin-lenses:image}{صورة} في مستوي
الجسم، مقلوبة، $\gamma = -1$. وإمالة المرآة بالزاوية $\alpha$ تدير
الحزمة الراجعة بمقدار $2\alpha$ وتزيح \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} جانبيًّا بمقدار
$2\alpha f'$، وهي تبقى واضحة في المستوي نفسه.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:9}
$f' = (100^2 - 40^2)/(4 \times 100) = \qty{21}{cm}$. وأقصر مسافة
$4f' = \qty{84}{cm}$، مع $\gamma = -1$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
تعطي $L_1$ \omterm{def:b1:mirrors-thin-lenses:image}{صورة} اللانهاية في $F_1'$، على \qty{20}{cm} خلف $L_1$، أي \qty{5}{cm}
خلف $L_2$: فهي جسم تخيلي، $\overline{O_2A_1} = \qty{+5}{cm}$.
و$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: \omterm{def:b1:mirrors-thin-lenses:image}{فالصورة} النهائية على \qty{10}{cm}
خلف $L_2$. وارتفاع \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} الوسيطة $h_1 = f_1'\alpha = 20\alpha$ (بالسنتيمتر)؛
و$\gamma_2 = 10/5 = 2$، ومنه $h = 40\alpha$: أي \omterm{def:b1:mirrors-thin-lenses:lens}{بعد بؤري} مكافئ
\qty{40}{cm}، في أسطوانة طولها $15 + 10 = \qty{25}{cm}$ --- وهي عدسة مقرِّبة.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ يعطي $\overline{SA'} =
3\overline{SA}$؛ وكلتاهما حقيقية أمام المرآة: $\overline{SA} = -x$ و
$\overline{SA'} = -3x$، والمسافة شاشة--مصباح $2x = \qty{80}{cm}$، ومنه
$x = \qty{40}{cm}$. فالمرآة على \qty{40}{cm} من المصباح و\qty{120}{cm}
من الشاشة؛ و$1/\overline{SF} = -1/40 - 1/120 = -1/30$، ومنه
$R = 2 \times 30 = \qty{60}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$؛ وقطر \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} $f'\theta =
\qty{4.6}{mm}$. والاستطاعة المجموعة $10^3 \times \pi(0.05)^2 = \qty{7.9}{W}$؛
ومساحة \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} $\pi(2.3\times10^{-3})^2 = \qty{1.7e-5}{m^2}$؛ فالإشعاع
$\qty{4.7e5}{W/m^2}$، أي $470$ ضعف ضوء الشمس المباشر: فيتفحّم الورق
ثم يشتعل --- وهي عدسة الإحراق. والربح عمومًا
$(D/f'\theta)^2$.
\end{solution}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$ و
$\gamma = \overline{OA'}/\overline{OA}$؛ مع محور موجَّه في اتجاه الضوء
ومسافات جبرية من $O$.

\textbf{2.} $p = -\overline{OA}$ و $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$، أي $p^2 - Dp + Df_1' = 0$؛ وللجذور وجود حقيقي إذا وفقط إذا كان
$D^2 - 4Df_1' \geq 0$، أي $D \geq 4f_1'$.

\textbf{3.} مجموع الجذرين $D$: أي $p_1 + p_2 = D$ ومنه $p_2 = D - p_1$ ---
فتتبادل مسافتا الجسم \omterm{def:b1:mirrors-thin-lenses:image}{والصورة}، والوضعان متناظران بالنسبة إلى
المنتصف. و$d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$، ومنه $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} لا يدخل في الحساب إلا انتقال الأسطوانة $d$ بين الوضعين
الواضحين والمسافة جسم--شاشة $D$: ولا يقتضي أيّ منهما
معرفة موضع $O$ داخل الأسطوانة.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$ و
$\partial f_1'/\partial d = -d/2D = -0.25$؛ ومنه
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ و $u(f_1') = \qty{0.14}{cm}$، أي
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ أو \qty{60}{cm}: فيكون $\gamma =
-60/20 = -3$ (والشريحة قريبة من \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة}، \omterm{def:b1:mirrors-thin-lenses:image}{والصورة} مكبَّرة: وهو جهاز العرض) و
$\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: \omterm{def:b1:mirrors-thin-lenses:image}{فالصورة الحقيقية} مُدارة بمقدار $180^\circ$
حول المحور، ومن ثمّ تُدخل الشريحة مُدارة بمقدار $180^\circ$ (مقلوبة
ومعكوسة يمينًا ويسارًا).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$، ومنه $\overline{OA} = \qty{-15.23}{cm}$: أي تتحرك \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} \qty{0.7}{mm}
نحو الشريحة. و$\gamma = 1000/(-15.23) = -65.7$: فعرض \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}
\qty{2.36}{m}.

\textbf{13.} باشتقاق علاقة ديكارت: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$، ومنه $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: فتنحرف \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}
عن الوضوح كلما سخن الفيلم --- ومن هنا الشرائح المركّبة على زجاج أو
ضبط ذاتي يتعقّب الفيلم.

\textbf{15.} بالمثلثات المتشابهة من الفتحة إلى نقطة \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}:
الضبابية $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$؛ وهي تُرى من \qty{5}{m}
تحت زاوية $\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$: فلا تُلاحَظ.

\textbf{16.} مساحة الشريحة $8.64\times10^{-4}\ \unit{m^2}$: فيكون $E_{\text{الشريحة}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$؛ ومساحة الشاشة أكبر $\gamma^2$
مرة، أي $\qty{2.16}{m^2}$: فيكون $E_{\text{الشاشة}} = \qty{0.93}{W/m^2}$؛
والاستطاعة نفسها، والمساحتان بنسبة $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$، ومنه $\overline{O_2A} = \qty{+29.3}{cm}$: أي إن
\omterm{def:b1:mirrors-thin-lenses:image}{الصورة} الوسيطة تقع على \qty{29.3}{cm} \emph{خلف} $L_2$ --- فهي جسم
تخيلي بالنسبة إلى $L_2$ (لأن الأشعة الآتية من $L_1$ تتقارب نحوها).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$؛
و$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$، ومنه $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$؛ و$\gamma_2 = 1200/29.3 =
41.0$؛ ومنه $\gamma = -60.9$.

\textbf{20.} العرض $60.9 \times 36 = \qty{2.19}{m}$. وبالعدسة $L_1$ وحدها:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$، فالعرض
\qty{2.86}{m}.

\textbf{21.} حزمة بالزاوية $\alpha$: تجمّعها $L_1$ على ارتفاع $h_1 = f_1'\alpha$
في مستويها البؤري، وهو على $\overline{O_2A_1} = f_1' - e$ من $L_2$؛
وتصوّر $L_2$ هذه النقطة عند $\overline{O_2A'}$ حيث $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ وتكبّرها بالمقدار $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$، ومنه $h = \gamma_2 f_1'\alpha$ و $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$، ومقلوبه هو الصيغة
المذكورة. ومنه $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$ و
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ و $15/19.6 = 0.77$. وهي قريبة فقط
لأن الشريحة على مسافة منتهية ولأن الرميين مقيسان
من عدستين مختلفتين؛ ومن أجل جسم في اللانهاية تكون النسبة
تامة.

\textbf{23.} جسم تخيلي عند $p > 0$ خلف \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة مفرّقة}:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$، وهو موجب (أي \omterm{def:b1:mirrors-thin-lenses:image}{صورة
حقيقية}) إذا وفقط إذا كان $p < \abs{f_2'}$.

\textbf{24.} الاستطاعة نفسها على \omterm{def:b1:mirrors-thin-lenses:image}{صورة} مساحتها أصغر بعامل $(2.19/2.86)^2 = 0.59$:
فيتضاعف الإشعاع $\times 1.7$.

\textbf{25.} يضرب المحوّل \omterm{def:b1:mirrors-thin-lenses:lens}{البعد البؤري} بالعامل
$f_{\mathrm{eq}}/f_1' = 1.3$؛ وعند رميٍ معطى تصير \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} أصغر $1.3$
مرة في كل اتجاه وأسطع $1.7$ مرة --- أي عدسة رميٍ
بعيد.
\end{solution}
