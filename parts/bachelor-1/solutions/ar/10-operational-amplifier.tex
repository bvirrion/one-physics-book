\section*{الفصل \ref{ch:b1:operational-amplifier} --- المضخّم العملياتي}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$؛ و$V_s = \qty{5.0}{V}$؛ والإشباع عند $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$؛ \omterm{def:b1:sinusoidal-impedance:impedance}{وممانعة} الدخل $R_1 = \qty{10}{k\ohm}$؛ و$i = 1.0/10^4 =
\qty{0.10}{mA}$؛ و$V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
دون تابع: المجزّئ $1/(1 + 10) = 0.091$، أي \qty{91}{mV}. ومع التابع:
\qty{1.0}{V} --- فالتابع لا يسحب شيئًا من المحسّ ويقدّم
\qty{1}{mA} التي تحتاجها الحمولة.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$؛ وعند $V_-$: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$،
ومنه $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$؛ ومع $V_- = V_+$:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. والأعداد: $10 \times 0.05 = \qty{0.50}{V}$؛
والمقدار \qty{1}{V} المشترك لا يسهم بشيء --- وهو رفض النمط المشترك.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: فالميل $-1000 \times 1.0 = \qty{-1000}{V/s}$؛
والإشباع بعد $15/1000 = \qty{15}{ms}$. وللموجة المربعة: مثلث سعته
$ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(زائد ما تركته الشحنة الابتدائية من ثابت).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: فالكسب في التيار المستمر $-100$؛ \omterm{def:b1:filters-transfer-functions:bode}{وتواتر القطع}
$1/(2\pi R'C) = \qty{1.6}{Hz}$؛ وعند \qty{1}{kHz} يكون $R'C\omega = 628$ و $G =
100/628 = 0.16$ --- أي قيمة المكامِل $1/RC\omega$. وهو يكامل من أجل
$f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: ومنه $R_2 = \qty{10}{k\ohm}$؛ و$C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
مع $\mu$ منتهٍ: $V_s = \mu(V_e - \beta V_s)$ حيث $\beta = 1/100$، ومنه
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. \omterm{thm:b1:sinusoidal-impedance:resonance}{وعرض النطاق} $f_T/G =
\qty{10}{kHz}$؛ وللتابع: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
كل جولة ضجيج تعبر \qty{2.0}{V} تقلب الخرج: أي رشقة من
التبديل السريع (``الرفرفة'') حول العبور. والعلاج: تخلّف
أعرض من الضجيج، أي \omterm{prop:b1:operational-amplifier:schmitt}{قادح شميت}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: فالعتبتان $\pm\qty{1.5}{V}$. ويكون
الخرج $+\qty{15}{V}$ بينما يرتفع $V_e$ حتى \qty{1.5}{V}، ثم ينقلب إلى
$-\qty{15}{V}$، ويبقى هناك حتى يهبط $V_e$ دون \qty{-1.5}{V}، ثم ينقلب
راجعًا: أي موجة مربعة بتواتر المثلث، وحوافها عند
عبور العتبتين.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
مع $V_s = +V_{\mathrm{sat}}$ يُشحن $V_C$ من $-\beta V_{\mathrm{sat}}$
نحو $+V_{\mathrm{sat}}$: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$، ويبلغ $+\beta V_{\mathrm{sat}}$ حين يكون $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$: فنصف الدور $RC\ln\frac{1 + \beta}{1 - \beta}$،
ومنه $T$. ومن أجل $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$؛ و$RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$ و $R = \qty{4.6}{k\ohm}$. و$V_s$: موجة مربعة
$\pm\qty{15}{V}$؛ و$V_C$: أقواس أسّية بين $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} مجزّئان: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone})؛ والتوصيل
يثبّت الإشارة.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ دائمًا؛ و$V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ عند المدى الكامل. و
المقدار \qty{2.5}{V} المشترك بينهما هو توتر النمط المشترك.

\textbf{4.} كل نقطة وسطى: $1 \parallel 1 = \qty{500}{\ohm}$؛ وبين
النقطتين الوسطيتين، \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$: أي ضياع قدره $9\%$.

\textbf{6.} \omterm{def:b1:sinusoidal-impedance:impedance}{ممانعة} دخل لانهائية (فلا تيار من الجسر، وتبقى
النقطتان الوسطيتان على كموني دارتهما المفتوحة) \omterm{def:b1:sinusoidal-impedance:impedance}{وممانعة} خرج
معدومة (فتستطيع المرحلة التالية أن تسحب ما تحتاجه).

\textbf{7.} $V_A$ و $V_B$ بالضبط؛ وتيار معدوم.

\textbf{8.} كما في \cref{prop:b1:operational-amplifier:sumdiff}: $V_+ =
V_BR_2/(R_1 + R_2)$، ثم قانون العقد عند $V_-$، ثم $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$: مثلًا \qty{1}{k\ohm} و\qty{9}{k\ohm}.
والكسب الكلي $1000$؛ والمدى الكامل $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} مع التطابق: يتلاشى المقدار المشترك \qty{2.5}{V} تمامًا (فلا يبقى إلا
$V_B - V_A$). أما عدم تطابق قدره $1\%$ فيمرّر نحو $1\%$ منه: أي
خطأ منسوب إلى الدخل من رتبة \qty{25}{mV}، وهو عشرة أضعاف إشارة المدى الكامل
--- فيجب أن تتطابق المقاومات الأربع إلى أفضل من $0.01\%$
(أو يُزال توتر النمط المشترك أولًا).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$، أي أربعون بالمئة من المدى
الكامل: فهو غير صالح للاستعمال دون تعديل الإزاحة، أو مضخّم منخفض
الإزاحة، أو معايرة صفرية بلا حمل على العارضة.

\textbf{13.} المرحلة 1: $f_T/100 = \qty{10}{kHz}$؛ والمرحلة 2: \qty{100}{kHz}؛
أي أعلى بكثير من \qty{5}{Hz}.

\textbf{14.} المطلوب $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$،
أي أدنى بألفَي مرة من \omterm{rem:b1:operational-amplifier:real}{معدل الانزلاق}: فلا.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$: ومنه $G = 1/\sqrt{26}$، أي \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$؛ والتأخير $\varphi/\omega =
\qty{15}{ms}$ --- وهو لا شيء أمام مرور شاحنة.

\textbf{18.} لا تحميل للمرحلة التالية (بفضل \omterm{def:b1:sinusoidal-impedance:impedance}{ممانعة} الخرج المعدومة)
ووجود كسب متاح؛ وكذلك لا توهين للإشارة المطلوبة دون
\omterm{def:b1:filters-transfer-functions:bode}{تواتر القطع}.

\textbf{19.} \qty{-28}{dB} (فالكسوب تتضارب والديسيبلات تُجمع)، وهو تامّ لأن
\omterm{def:b1:sinusoidal-impedance:impedance}{ممانعة} خرج المرحلة الأولى معدومة: فالمرحلة الثانية لا
تحمّلها.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$؛ والخرج $+V_{\mathrm{sat}}$ حين تتجاوز الإشارة
\qty{2.0}{V}، و $-V_{\mathrm{sat}}$ دونها (أو العكس، حسب التوصيل).

\textbf{21.} يرفرف: فيومض المصباح كلما عبر الضجيج
المرجع.

\textbf{22.} مع $V_{\mathrm{ref}}$ عبر $R_2$ والخرج عبر
$R_1$\dots{} في الصورة المعيارية يقع المدخل غير القالب عند
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$؛ وبأخذ المرجع منزاحًا
تبعًا لذلك، تكون مستويات التبديل $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
ومن $\beta V_{\mathrm{sat}} = \qty{0.10}{V}$ نجد $\beta = 1/150$، مثلًا $R_1 =
\qty{1.0}{k\ohm}$ و $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (للانطلاق) و\qty{1.9}{V} (للعودة): أي تشوّهان
\num{1.68e-3} و\num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. وعند
$-V_{\mathrm{sat}}$ يكون الثنائي منحازًا عكسيًّا بمقدار \qty{15}{V}، أي فوق
تقنينه: فيحميه ثنائي عادي على التوازي المعاكس (أو على التسلسل).

\textbf{25.} الجسر (\qty{2.5}{mV} عند المدى الكامل) $\to$ تابعان (بلا
تحميل) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{مضخّم فارقي} $\times100$ $\to$ \omterm{prop:b1:operational-amplifier:basic}{مضخّم غير قالب}
$\times10$ $\to$ تمرير منخفض فعّال عند \qty{10}{Hz} $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{قادح شميت} عند
$\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ ثنائي LED. والحدود الحقيقية: \omterm{rem:b1:operational-amplifier:real}{توتر
الإزاحة عند الدخل} وتطابق مقاومات \omterm{prop:b1:operational-amplifier:sumdiff}{المضخّم
الفارقي}، وكلاهما يقارن إشارة المليفولط.
\end{solution}
