\section*{الفصل \ref{ch:b1:fluid-statics} --- سكون الموائع}

\begin{solution}{exo:b1:fluid-statics:1}
$P = P_0 + \rho gh$: أي \qty{2.0e5}{Pa} وهو \qty{2.0}{bar}؛ ثم \qty{1.1e6}{Pa}
وهو \qty{11}{bar}؛ ثم \qty{1.0e7}{Pa} وهو \qty{101}{bar}. وعند \qty{300}{m}: $P =
\qty{3.1e6}{Pa}$، والقوة \qty{3.1}{MN} على \qty{1}{m^2} (صافيةً، مقابل
\qty{1}{atm} في الداخل: \qty{3.0}{MN}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:2}
$h = P/\rho g$: فالزئبق \qty{0.760}{m}، والماء \qty{10.3}{m}. وأقصى ما
تستطيعه المضخة أن تُحدث خلاءً فوق الماء: فيرفع الجوّ عندئذ
العمود \qty{10.3}{m}، لا أكثر.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:3}
$F_2 = F_1S_2/S_1 = \qty{10}{kN}$ أي طن واحد؛ و $d_2 = d_1S_1/S_2 = \qty{2.0}{mm}$
(الحجم المُزاح نفسه، والشغل نفسه).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:4}
الضغط نفسه عند مستوى السطح الفاصل بين الزيت والماء في
الذراعين: $\rho_og
h_o = \rho_wgh_w$، ومنه $h_w = 0.85 \times 12 = \qty{10.2}{cm}$ من الماء فوق
ذلك المستوى في الذراع الأخرى؛ فيعلو سطح الزيت بمقدار $12 - 10.2 = \qty{1.8}{cm}$.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:5}
$H = RT/Mg = 8.314 \times 273/(0.029 \times 9.81) = \qty{8.0}{km}$؛ و $P/P_0 =
\eu^{-z/H}$: أي $0.54$ عند \qty{5}{km}، و $0.33$ عند \qty{8848}{m}؛ وينتصف عند $H\ln2 =
\qty{5.5}{km}$ --- وهي الأعداد المستعملة في \cref{ch:b1:kinetic-theory}.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:6}
$F = \tfrac12\rho gh^2L = \qty{1.96e8}{N}$؛ \omterm{prop:b1:fluid-statics:wall}{ومركز الضغط} على عمق \qty{13.3}{m}،
أي \qty{6.7}{m} فوق القاعدة؛ وعزمه بالنسبة إلى القاعدة $F \times
6.7 = \qty{1.3e9}{N.m}$. وتقويس السدّ نحو الماء يحوّل الدفع
إلى انضغاط في القوس، يُنقل إلى جانبَي الوادي.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:7}
$917/1025 = 89\%$؛ والجذع $60\%$. أما الطوف: فالدافعة عند الغمر التام $1000 \times
2.0 \times 9.81 = \qty{19.6}{kN}$، وثقله هو \qty{11.8}{kN}: فيبقى \qty{7.8}{kN}
احتياطًا، أي عشرة أشخاص\dots{} والعاشر مبلول القدمين: فتسعة
يبقون جافّين (\qty{800}{kg}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:8}
الحجم المغمور $m/\rho$: أي \qty{20}{cm^3} في الماء؛ وفي المحلول الملحي $20/1.1 = \qty{18.2}{cm^3}$،
أي $\qty{1.8}{cm^3}$ أقل، ومنه $1.8/0.50 = \qty{3.6}{cm}$ من الساق زيادةً: فيبرز منها \qty{7.6}{cm}.
والحساسية $\approx m/(\rho^2S) \approx \qty{40}{cm}$ لواحدة الكثافة
النسبية (أي \qty{3.6}{cm} لكل $0.1$).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:9}
الحجم المُزاح $2800/1.025 = \qty{2732}{m^3}$: فيبرز \qty{268}{m^3} أي ($9\%$).
وللطفو المحايد: الكتلة $1025 \times 3000 = \qty{3075}{t}$، فتأخذ \qty{275}{t}.
وفي الماء العذب تهبط الدافعة إلى \qty{3000}{t}: فتزيد الكتلة \qty{75}{t} --- فتغوص
ما لم تُضخّ \qty{75}{t} إلى الخارج.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:10}
الهواء المُزاح: $10^{-2} \times 1.20 = \qty{12.0}{g}$؛ والهيليوم في الداخل $12.0 \times
4/29 = \qty{1.7}{g}$؛ والمطاط \qty{3.0}{g}: فالحمولة الصافية \qty{7.3}{g} --- أي سبعة أمتار
من خيط خفيف.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:11}
العزم $\int_0^h\rho gx\cdot xL\,\dd x = \rho gLh^3/3$، والقوة $\rho gLh^2/2$:
فالذراع $2h/3$. ومن $h_1$ إلى $h_2$: العزم $\rho gL(h_2^3 - h_1^3)/3$، والقوة
$\rho gL(h_2^2 - h_1^2)/2$: فالعمق $\frac23(h_2^3 - h_1^3)/(h_2^2 - h_1^2)$.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:12}
الغمر الزائد $x$ يضيف الدافعة $\rho gSx$ نحو الأعلى: ومنه $m\ddot x = -\rho gSx$،
و $\omega_0^2 = \rho gS/m$، و $T = 2\pi\sqrt{m/\rho gS} = 2\pi\sqrt{0.020/(1000 \times
9.81 \times 5\times10^{-5})} = \qty{1.3}{s}$.
\end{solution}

\begin{solution}{pb:b1:fluid-statics:1}
\textbf{1.} $P = 1.013\times10^5 + 1025 \times 9.81 \times 300 = \qty{3.12e6}{Pa}
= \qty{31}{bar}$.

\textbf{2.} $S = \pi(0.40)^2 = \qty{0.503}{m^2}$: فالقوة الصافية $(P - P_0)S =
\qty{1.52e6}{N}$ --- أي 155 طنًّا، شاحنة محمَّلة على فتحة.

\textbf{3.} العارضة السفلى أعمق بمقدار \qty{10}{m}: فيزيد الضغط $+\rho g \times 10 = \qty{1.0e5}{Pa}$،
أي $3\%$: فالبدن محمَّل تحميلًا يكاد يكون منتظمًا.

\textbf{4.} $\Delta\rho/\rho = \chi_T\Delta P = 4.6\times10^{-10} \times 3.0\times10^6
= 0.14\%$: وهو مهمَل بالنسبة إلى الضغط (بضعة كيلوباسكالات من
ثلاثة ملايين).

\textbf{5.} ثلاثون بارًا في الخارج وواحد في الداخل: فالبدن وعاء
ضغط محمَّل نحو الداخل؛ والأسطوانة (أو الكرة) تحوّل الضغط
إلى انضغاط على طول جدارها، وهو ما يقاومه الفولاذ جيدًا،
في حين تنثني صفيحة مسطّحة.

\textbf{6.} $\Delta P\,R/e = 3.0\times10^6 \times 5.0/0.030 = \qty{5.0e8}{Pa}
= \qty{500}{MPa}$: أي قريب من إجهاد الخضوع --- فعمق \qty{300}{m} قريب من
حدّ هذا البدن، وتحتاج المراكب الأعمق إلى فولاذ أسمك أو أمتن
(أو إلى التيتانيوم).

\textbf{7.} الحجم المُزاح $m_0/\rho = 2800/1.025 = \qty{2732}{m^3}$: فيبرز \qty{268}{m^3}
أي ($9\%$).

\textbf{8.} للطفو المحايد: $m = \rho V = \qty{3075}{t}$: فتأخذ \qty{275}{t}.

\textbf{9.} تهبط الدافعة بمقدار $0.1\%$ من $\rho Vg$: أي $0.001 \times 3075 \times 9.81
\times10^3 = \qty{30}{kN}$ نحو الأسفل (ثلاثة أطنان). فعمق أكبر $\Rightarrow$ انضغاط
أكبر $\Rightarrow$ ثقل أكبر: وهو توازن غير مستقر --- فعلى الغواصة أن
تعدّل موازنتها باستمرار (والبدن الفولاذي ينضغط أقل من الماء، وهو ما يساعد؛
أما بدن شديد الليونة فيغوص بلا عودة).

\textbf{10.} الدافعة $1000 \times 3000 \times 9.81 = \qty{29.4}{MN}$ مقابل
الثقل $3075 \times 9.81 \times 10^3 = \qty{30.2}{MN}$: أي \qty{0.74}{MN} نحو الأسفل
وهي (\qty{75}{t})؛ فتُضخّ \qty{75}{t} إلى الخارج.

\textbf{11.} تصير أخفّ في الحال بمقدار \qty{20}{t}: فتأخذ \qty{20}{t} من الماء
أي (\qty{19.5}{m^3}) في خزانات التعويض.

\textbf{12.} لا يُطرَد الماء إلا بشيء ضغطه أعلى من ضغط البحر
في الخارج؛ والهواء المضغوط في القوارير يدفعه إلى الخارج ---
ما دام ضغط القارورة يفوق الضغط المحيط عند ذلك
العمق.

\textbf{13.} $PV$ ثابت: فإن \qty{275}{m^3} عند \qty{31}{bar} تحتاج إلى $275 \times 31
= \qty{8500}{m^3}$ من الهواء عند \qty{1}{atm} --- مخزونة عند \qty{200}{bar} في
\qty{43}{m^3} من القوارير.

\textbf{14.} \qty{2.0}{bar} عند \qty{10}{m}؛ و\qty{4.0}{bar} عند \qty{30}{m}.

\textbf{15.} $PV$ ثابت: $6.0/4.0 = \qty{1.5}{L}$ --- فالقفص الصدري
ينضغط إلى الداخل؛ ويحسّ به الغوّاصون الحرّون.

\textbf{16.} الهواء المأخوذ عند \qty{4}{bar} يتمدد أربعة أضعاف: أي \qty{24}{L} في رئتين
تسعان \qty{6}{L}: فيقع التمزّق. فلا تحبس نفَسك أبدًا عند الصعود؛ بل ازفر
باستمرار.

\textbf{17.} $\Delta P = \rho g \times 1.0 = \qty{1.0e4}{Pa}$؛ والقوة $\qty{1.0e4}{Pa}
\times 0.05 = \qty{500}{N}$ على الصدر: فعضلات التنفس لا تستطيع
رفع خمسين كيلوغرامًا --- ولا يعمل أنبوب التنفس إلا في الديسيمترات الأولى.

\textbf{18.} $12 \times 200 = \qty{2400}{L}$ عند \qty{1}{bar}، أي \qty{600}{L}
عند \qty{4}{bar}: فتكفي \qty{30}{min}.

\textbf{19.} $V = 10^8/1025 = \qty{97600}{m^3}$.

\textbf{20.} في الماء العذب عليها أن تزيح $10^5\ \unit{m^3}$: أي \qty{2400}{m^3}
أكثر، ومنه $2400/8000 = \qty{0.30}{m}$ أعمق.

\textbf{21.} $10^4/1.025 = \qty{9756}{m^3}$ أكثر: أي \qty{1.2}{m} أعمق.

\textbf{22.} حين تفرغ السفينة يعلو مركز كتلتها $G$ ويقلّ الجزء المغمور
من بدنها: فقد يهبط \omterm{rem:b1:fluid-statics:stability}{مركز الاستقرار} دون $G$ فتنقلب في موج. وماء البحر
في أسفل البدن يخفض $G$ ويزيد الغاطس.

\textbf{23.} $\rho gh = \qty{2.0e5}{Pa}$، أي \qty{0.2}{MN/m^2}: وهو أقل خمس عشرة مرة
من \qty{3}{MN/m^2} عند الغواصة.

\textbf{24.} لا: فالضغط عند عمق الغواصة هو $P_0 + \rho gh$،
يعيّنه العمق وحده. أما ثقل الناقلة فيحمله الماء الذي
تزيحه، وهو يرفع مستوى البحر في كل مكان بمقدار لا يُقاس؛
ولا يبلغ الغواصةَ أيّ حمل زائد.

\textbf{25.} العلاقة $\dd P/\dd z = -\rho g$ --- ومنها $P = P_0 + \rho gh$ لكل
عمق وقوة وحجم رئة في هذه المسألة --- ثم \omterm{thm:b1:fluid-statics:archimedes}{مبرهنة أرخميدس}، وهي
تلك العلاقة مكاملةً على سطح مغلق.
\end{solution}
