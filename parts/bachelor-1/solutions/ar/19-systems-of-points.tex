\section*{الفصل \ref{ch:b1:systems-of-points} --- جمل النقاط؛ مدخل إلى الأجسام الصلبة}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$. وللأرض--القمر:
$384000/82 = \qty{4700}{km}$ من مركز الأرض، أي داخل الأرض
(\qty{6370}{km}): فتترنّح الأرض حول نقطة تقع \qty{1700}{km} تحت
سطحها.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: ومنه $v = \qty{2.0}{m/s}$. وللرصاصة $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$، وللبندقية \qty{8}{J}: من الطاقة الكيميائية
للبارود؛ ويأخذ الجسم الخفيف كل ذلك تقريبًا (إذ $E_k = p^2/2m$
مع $p$ متساوية).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$؛ و $E_k$: أي \qty{200}{kJ} قبل، و $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$ بعد: أي $60\%$ ضاعت.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
عند الطرف: $\int_0^L x^2(M/L)\dd x = ML^2/3$؛ وعند المركز: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$؛ وبهويغنس مع $d = L/2$: $ML^2/12 + ML^2/4 = ML^2/3$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: فالأسطوانة $1.5$ والطوق $2$ مرة $\tfrac12 Mv^2$.
وبالسرعة نفسها، للطوق طاقة أكبر: فيتدحرج أبعد على أرضية مستوية
(بالاحتكاك نفسه) ويصعد أعلى (إذ $h = v^2(1 + k)/2g$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
لكلوريد الهدروجين: $\mu = 35m_H/36 = 0.972m_H$؛ وللذرة: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$. والتواترات $\propto 1/\sqrt\mu$: أي $+1.4\%$ لكلوريد الهدروجين، و $+0.027\%$
للهدروجين (وعليه يقوم الانزياح النظيري بين طيفَي الهدروجين والديوتريوم).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$ و $d = L/2$: ومنه $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$؛ والطول المكافئ $2L/3 = \qty{0.67}{m}$. والضربة عند
مركز القرع تجعل القضيب يدور حول المفصل بلا
ردّ فعل دفعي هناك: فاضرب الكرة عند ``البقعة الحلوة'' من المضرب
ولن تشعر اليدان بأيّ لسعة.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$؛ و $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$ و $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$؛
و $\Gamma = J\omega/t = \qty{314}{N.m}$؛ وعدد الدورات: $\tfrac12\omega t/2\pi = 250$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: فالكرة ($k = 2/5$) $0.714$، والأسطوانة ($1/2$)
$0.667$، والطوق ($1$) $0.500$ مرة $g\sin\alpha$: فالكرة أولًا والطوق أخيرًا؛
ولا كتلة ولا نصف قطر في أيّ موضع.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
\omterm{def:b1:newton-dynamics:momentum}{بكمية الحركة}: $0.010 \times 400 = 2.01\,v$: ومنه $v = \qty{2.0}{m/s}$؛ ثم بالطاقة:
$h = v^2/2g = \qty{0.20}{m}$. و $E_k$: أي \qty{800}{J} قبل، و $\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$ بعد: أي $99.5\%$ ضاعت في التشوّه
والحرارة. والارتطام غير مرن --- فالطاقة لا تنحفظ خلاله،
أما \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة} فتنحفظ (إذ شدّ الخيط شاقولي والارتطام قصير).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ و $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ تعطيان
$v_1 + v_1' = v_2'$ (بقسمة معادلة الطاقة على معادلة \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة})،
ومن هنا الصيغتان. ونسبة الطاقة المنقولة إلى الهدف $4m_1m_2/(m_1 + m_2)^2$:
فللبروتون $100\%$؛ وللكربون $4 \times 12/169 = 28\%$؛ وللرصاص $4 \times 207/208^2 =
1.9\%$. فالنوى الخفيفة تأخذ طاقة النيوترون في بضعة تصادمات
(كالماء والغرافيت)؛ أما الرصاص فيحتاج إلى مئات.
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} على المحور، في مركز القرص، بالتناظر.

\textbf{2.} بكونيغ: $E_k = \tfrac12 Mv_G^2 + E_k^*$، وفي \omterm{def:b1:systems-of-points:barycentric}{المرجع
المرجَّحي} يدور اليويو حول محوره: $E_k^* = \tfrac12 J\omega^2$.

\textbf{3.} حلقة نصف قطرها $\rho$ وكتلتها $2M\rho\,\dd\rho/R^2$: ومنه $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$.

\textbf{4.} الخيط ثابت وينحلّ عن المحور: فنقطة
المحور الملامسة له ساكنة لحظيًّا، ومنه $v_G = r\omega$.

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$؛ و $R^2/2r^2 = 9/(2 \times 0.16) = 28$: أي العامل $29$ ---
فالطاقة كلها تقريبًا في الدوران.

\textbf{6.} الوزن $Mg$ نحو الأسفل، والشدّ $T$ نحو الأعلى: ومنه $Ma_G = Mg - T$.

\textbf{7.} للشدّ وحده عزم حول المحور (\omterm{def:b1:angular-momentum:moment}{وذراع الرافعة} $r$):
$J\dot\omega = Tr$.

\textbf{8.} $\dot\omega = a_G/r$: ومنه $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$؛ ثم
$Ma_G = Mg - (R^2/2r^2)Ma_G$، ومنه $a_G = g/29 = \qty{0.34}{m/s^2}$.

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$، أي $97\%$ من
الوزن: فالخيط يكاد يمسكه.

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$؛ و $v_G = a_Gt = \qty{0.82}{m/s}$.
أما الحجر: ففي \qty{0.45}{s} وبسرعة \qty{4.4}{m/s}.

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$.

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ مقابل $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: أي طاقة في الدوران أكبر $28$ مرة منها في السقوط.

\textbf{13.} $Mgh = \qty{0.49}{J}$؛ و $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$. والشدّ يؤثر عند نقطة
الخيط الساكنة (فالخيط لا يتحرك): أي استطاعة معدومة.

\textbf{14.} قوة الإيقاف تؤثر على امتداد الخيط، مارّةً بالمحور:
فلا عزم لها حوله؛ ويبقى \omterm{def:b1:angular-momentum:L}{العزم الحركي} $J\omega$ دون مسّ. ولا يستطيع
إبطاء الدوران إلا قوة مماسية (كاحتكاك على المحور).

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$، أي ثمانية عشر ضعف الوزن --- ومن هنا النتّة التي يشعر بها
الإصبع.

\textbf{16.} $J\dot\omega = -\Gamma$: ومنه $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$.

\textbf{17.} تتحول طاقة الدوران كلها $\tfrac12 J\omega^2 = \qty{0.47}{J}$
إلى $Mgh'$: ومنه $h' = \qty{0.96}{m}$، أي الارتفاع كله تقريبًا؛ وهو أقلّ عمليًّا
بسبب احتكاك المحور أثناء النوم والضياعات في
النتّة.

\textbf{18.} $a_G$ نحو الأسفل (فاليويو يتباطأ في الصعود) بينما
$v_G$ نحو الأعلى: ومنه $T = M(g - a_G) < Mg$ --- أي أصغر من الوزن.

\textbf{19.} $r = R$: فالعامل $1 + 1/2 = 1.5$ و $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- أي سقوط سريع ودوران قليل: فالمحور الرفيع هو ما يصنع اليويو.

\textbf{20.} في \omterm{ex:b1:systems-of-points:rolling}{التدحرج دون انزلاق}: تكون سرعة \omterm{def:b1:point-kinematics:frame}{النقطة المادية} عند $C$
معدومة؛ وحقل سرعات \omterm{def:b1:systems-of-points:solid}{جسم صلب} فيه نقطة ساكنة هو
دوران حول تلك النقطة (بمحور يمرّ بالنقطة $C$، وبالمقدار $\omega$ نفسه).

\textbf{21.} تؤثر $F$ عند أسفل المحور، على ارتفاع $R - r$ فوق
الطاولة، أفقيًّا نحو اليد: فعزمها $F(R - r)$ حول $C$ في
الجهة التي تدحرج البكرة نحو اليد؛ أما الوزن وردّ
الفعل فيمرّان بالنقطة $C$ (فلا عزم لهما).

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$، ومنه $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: أي نحو اليد.

\textbf{23.} يمرّ حامل $F$ بالنقطة $C$ حين يلتقي الخيط
الخارج من أسفل المحور بالطاولة عند $C$: أي $\cos\beta = r/R$، ومنه
$\beta = 82^\circ$. وإذا زاد الميل انعكس العزم حول $C$ فتتدحرج البكرة
بعيدًا (ومن أجل $\beta$ كبيرة بما يكفي ترتفع).

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$؛ والاحتكاك من $Ma_G = F - f$ (إذ يعاكس الاحتكاك $F$):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$؛ ويجب أن يبقى دون $\mu_sMg
\approx \qty{0.2}{N}$ (من أجل $\mu_s = 0.4$)، أي $F \lesssim \qty{0.5}{N}$.

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{مركز الكتلة}: انسحاب $G$ والشدّ؛
\omterm{def:b1:angular-momentum:L}{والعزم الحركي} حول محور: الدوران وجهة التدحرج؛
والطاقة: السرعة في الأسفل وارتفاع الصعود.
\end{solution}
