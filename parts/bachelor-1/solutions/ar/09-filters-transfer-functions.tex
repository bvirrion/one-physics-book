\section*{الفصل \ref{ch:b1:filters-transfer-functions} --- المرشّحات ودوال النقل}

\begin{solution}{exo:b1:filters-transfer-functions:1}
$0.5 \to \qty{-6.0}{dB}$؛ و$10 \to \qty{20}{dB}$؛ و$1/\sqrt2 \to \qty{-3.0}{dB}$؛
و$100 \to \qty{40}{dB}$. و$\qty{-40}{dB} \to 0.01$؛ و$\qty{+6}{dB} \to 2.0$.
وعلى التتالي: $-6 - 20 = \qty{-26}{dB}$، أي نسبة $0.05$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:2}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 10^4 \times 1.6\times10^{-8}) =
\qty{1.0}{kHz}$. والكسوب $-10\log(1 + (f/f_c)^2)$: فعند \qty{100}{Hz}:
\qty{-0.04}{dB}؛ وعند \qty{1}{kHz}: \qty{-3}{dB}؛ وعند \qty{10}{kHz}: \qty{-20}{dB}؛
وعند \qty{100}{kHz}: \qty{-40}{dB}.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:3}
(أ) تمرير منخفض ($C$ يقصر الخرج عند $f$ العالي)؛ (ب) تمرير عالٍ ($C$
ينفتح عند $f$ المنخفض)؛ (ج) تمرير منخفض من الرتبة الثانية ($L$ ينفتح عند $f$ العالي و
$C$ يقصر)؛ (د) تمرير نطاقي ($C$ يحجب $f$ المنخفض و $L$ يحجب $f$ العالي).
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:4}
$4E/n\pi$: أي \qty{6.37}{V} عند \qty{1}{kHz}، و\qty{2.12}{V} عند \qty{3}{kHz}، و
\qty{1.27}{V} عند \qty{5}{kHz}. والموجة المربعة متناظرة تناظرًا مضادًّا
بإزاحة نصف دور ($s(t + T/2) = -s(t)$)، وهو ما لا تحققه التوافقيات
الزوجية.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:5}
عند \qty{100}{Hz} وتوافقياتها المفيدة يكون $\omega \ll \omega_c$:
ومنه $\underline H \approx j\omega/\omega_c$، أي مشتقّ مقسوم على
$\omega_c$. وميل المثلث $\pm 4Ef = \pm\qty{400}{V/s}$، ومنه
يكون الخرج موجة مربعة سعتها $400/(2\pi \times 10^4) =
\qty{6.4}{mV}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:6}
$\omega \gg \omega_c$: ومنه $u_s \approx (1/\tau)\int u_e\,\dd t$ مع $\tau =
1/(2\pi \times 10) = \qty{16}{ms}$. وتكامل $\pm E$ مثلث
يرتفع بمقدار $ET/2$ في كل نصف دور: فالمدى $ET/2\tau$ والسعة
$ET/4\tau = 1.0 \times 10^{-3}/(4 \times 0.016) = \qty{16}{mV}$ حول
الصفر.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:7}
$f_0 = \qty{5.03}{kHz}$. و$Q = \sqrt{L/C}/R = 1/\sqrt2$: ومنه $R = \sqrt2 \times
316 = \qty{447}{\ohm}$. وعند $f_0$: \qty{-3}{dB}؛ وعند $10f_0$: $1/\sqrt{1 +
10^4} \to \qty{-40}{dB}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:8}
$G = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ مع $x = f/3$: فعند \qty{1}{kHz}: $x = 1/3$ و
$Q(x - 1/x) = -53$ و $G = 0.019$؛ وعند \qty{3}{kHz}: $G = 1$؛ وعند \qty{5}{kHz}:
$x = 5/3$ و $21$ و $G = 0.047$. والخرج: \qty{2.1}{V} عند \qty{3}{kHz}، مع
\qty{0.12}{V} عند \qty{1}{kHz} و\qty{0.06}{V} عند \qty{5}{kHz} --- أي
جيب شبه خالص عند \qty{3}{kHz}.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:9}
العقد: مع $a = jRC\omega$، يرى المكثف الأول $R$ قبله و
$R + 1/jC\omega$ بعده: ومنه $\underline H = 1/(1 + 3a + a^2)$، أي
$1/[1 - (\omega/\omega_c)^2 + 3j\omega/\omega_c]$. وعند $\omega_c$: $1/3j$ و
$G = 1/3$، أي \qty{-9.5}{dB}؛ بينما جداء الكسبين المنفصلين
$1/2$، أي \qty{-6}{dB}. و$f_c = \qty{1.0}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:10}
دون $\omega_1$: \qty{0}{dB}؛ وبين $\omega_1$ و $\omega_2$: ارتفاع
\qty{20}{dB/decade}؛ وفوق $\omega_2$: مستوٍ عند
$20\log(\omega_2/\omega_1) = \qty{40}{dB}$. والطور هو
$\arctan(\omega/\omega_1) - \arctan(\omega/\omega_2)$: فعند
$\omega = 10^3$ يكون $\arctan 10 - \arctan 0.1 = 84^\circ - 6^\circ$،
أي $78^\circ$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:11}
فوق \omterm{def:b1:filters-transfer-functions:bode}{تواتر القطع} بكثير يكون $G \approx f_c/f$؛ ونسبة التموّج بعد الترشيح
هي
\[
\frac{(4E/3\pi)(f_c/100)}{2E/\pi} = \frac{2}{3}\,\frac{f_c}{100} < 0.01,
\]
ومنه $f_c < \qty{1.5}{Hz}$ و $\tau = 1/2\pi f_c > \qty{0.1}{s}$. والحمولة نفسها هي
$R$: فمكثف على طرفيها لا يكلّف شيئًا في هبوط التوتر ولا في الاستطاعة،
بينما تهدر ناقلة على التسلسل الاثنين معًا.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:12}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 1.6\times10^5 \times 10^{-7}) =
\qty{10}{Hz}$ و $\tau = RC = \qty{16}{ms}$. وخلال قمة مسطّحة يُشحن
المكثف عبر $R$: فيتناقص الخرج بالعامل $\eu^{-t/\tau}$
من كل حافة. والتهدّل على $T/2$ هو $1 - \eu^{-T/2\tau}$: فعند \qty{50}{Hz}
(حيث $T/2 = \qty{10}{ms}$): $1 - \eu^{-0.63} = 47\%$؛ وعند \qty{1}{kHz}
(حيث $T/2 = \qty{0.5}{ms}$): $3\%$.
\end{solution}

\begin{solution}{pb:b1:filters-transfer-functions:1}
\textbf{1.} $\underline H = \underline U_{\text{المجهار}}/\underline U_e$؛
و$G_{\mathrm{dB}} = 20\log\abs{\underline H}$.

\textbf{2.} $\underline H_L = R/(R + jL\omega) = 1/(1 + j\omega/\omega_c)$ مع
$\omega_c = R/L$.

\textbf{3.} $L = R/\omega_c = 8.0/(2\pi \times 2000) = \qty{0.64}{mH}$.

\textbf{4.} $\underline H_H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$ مع
$\omega_c = 1/RC$؛ ومنه $C = 1/(R\omega_c) = \qty{9.9}{\micro F} \approx
\qty{10}{\micro F}$.

\textbf{5.} عند $f_c$: \qty{-3}{dB} لكلٍّ منهما. وعند \qty{200}{Hz} (حيث $x = 0.1$): المجهار الجهيري
\qty{-0.04}{dB} والمجهار الحادّ \qty{-20}{dB}. وعند \qty{20}{kHz} (حيث $x = 10$): المجهار الجهيري
\qty{-20}{dB} والمجهار الحادّ \qty{-0.04}{dB}.

\textbf{6.} $G_L^2 + G_H^2 = (1 + x^2)/(1 + x^2) = 1$: فمع حمولتين متساويتين
تتجمّع الاستطاعتان لتساويا ما كان سيأخذه مجهار واحد \qty{8}{\ohm}
--- أي مقتسَمة لا ضائعة.

\textbf{7.} $-45^\circ$ (المجهار الجهيري) و $+45^\circ$ (المجهار الحادّ): أي $90^\circ$
بينهما؛ فعند $f_c$ لا يتحرك المخروطان معًا.

\textbf{8.} \qty{\pm 6}{dB} لكل أوكتاف (أي \qty{20}{dB} لكل عقد).

\textbf{9.} مجزّئ بين $jL\omega$ و $R \parallel 1/jC\omega =
R/(1 + jRC\omega)$: ومنه $\underline H = R/[R + jL\omega(1 + jRC\omega)] =
1/(1 - LC\omega^2 + jL\omega/R)$.

\textbf{10.} $\omega_0 = 1/\sqrt{LC}$؛ و$L\omega/R = x/Q$ يعطي $Q =
R/L\omega_0 = R\sqrt{C/L}$.

\textbf{11.} $\abs{\underline H}^{-2} = (1 - x^2)^2 + x^2/Q^2 = 1 + x^4 +
x^2(1/Q^2 - 2) = 1 + x^4$ من أجل $Q^2 = \tfrac12$؛ وعند $x = 1$ يكون $G = 1/\sqrt2$.

\textbf{12.} $L = R/(Q\omega_0) = 8\sqrt2/(1.257\times10^4) = \qty{0.90}{mH}$؛
و$C = 1/(L\omega_0^2) = \qty{7.0}{\micro F}$.

\textbf{13.} من أجل $x = 2$: $1/\sqrt{17}$، أي \qty{-12.3}{dB}؛ ومن أجل $x = 4$: $1/\sqrt{257}$،
أي \qty{-24.1}{dB}: أي \qty{12}{dB} لكل أوكتاف.

\textbf{14.} تبادل $L$ و $C$ يبادل $x \leftrightarrow 1/x$ (ويبقى
$Q$): ومنه $\underline H_H = -x^2/(1 - x^2 + jx/Q)$.

\textbf{15.} عند $x = 1$: $\underline H_L = Q/j = -jQ$ (أي $-90^\circ$)، و
$\underline H_H = jQ$ (أي $+90^\circ$): فالمجهاران في تعاكس
طور عند تواتر الفصل وكانا سيتلاشيان صوتيًّا؛ ولذلك يُوصَّل المجهار الحادّ
بقطبية معكوسة.

\textbf{16.} التوافقيات فوق \qty{2}{kHz}: أي $n \geq 10$ (من \qty{2.2}{kHz}
فصاعدًا) تذهب في معظمها إلى المجهار الحادّ.

\textbf{17.} $9 \times 220 = \qty{1980}{Hz} \approx f_c$: فيُقسم بالتساوي،
أي \qty{-3}{dB} لكلٍّ منهما.

\textbf{18.} المجهار الجهيري ($G_L$): من أجل $x = 0.5$: $0.89$؛ ومن أجل $x = 1.5$: $0.55$؛ ومن أجل $x = 2.5$:
$0.37$. والمجهار الحادّ ($G_H$): $0.45$ و $0.83$ و $0.93$.

\textbf{19.} المجهار الحادّ: فمكثفه المتسلسل يحجب التيار المستمر. أما المجهار الجهيري
فيأخذ الإزاحة (\qty{1}{V} على \qty{8}{\ohm}: أي \qty{125}{mA} و
\qty{0.125}{W} حرارةً ومخروطًا منزاحًا).

\textbf{20.} في الرتبة الأولى، من أجل $x = 0.25$: $G_H = 0.25/\sqrt{1.0625} = 0.24$
(أي \qty{-12}{dB})، وكسر الاستطاعة $6\%$. وفي الرتبة الثانية: $0.0625/\sqrt{1 +
0.0039} = 0.06$ (أي \qty{-24}{dB})، و$0.4\%$. فالمجهار الحادّ المقنَّن ببضعة واطات
يتلقى $6\%$ من نغمة جهير عالية في تصميم الرتبة الأولى --- وقد
يحترق.

\textbf{21.} $\underline Z = 8 + jL_{vc}\omega$: فعند \qty{2}{kHz}: $8 + 6.3j$ و
$\abs{\underline Z} = \qty{10.2}{\ohm}$؛ وعند \qty{8}{kHz}: $8 + 25.1j$ و
\qty{26.3}{\ohm}.

\textbf{22.} $\underline H = \underline Z/(\underline Z + jL\omega)$:
فعند \qty{2}{kHz}: $10.2/\abs{8 + 14.3j} = 10.2/16.4 = 0.62$ (أي \qty{-4.2}{dB})؛
وعند \qty{8}{kHz}: $26.3/\abs{8 + 57.3j} = 0.45$ (أي \qty{-6.9}{dB}) بدلًا من
\qty{-12.3}{dB}: \omterm{def:b1:sinusoidal-impedance:impedance}{فممانعة} الوشيعة المتزايدة تقاوم الوشيعة المتسلسلة، و
يتسطّح الانحدار إلى بضعة ديسيبلات لكل أوكتاف.

\textbf{23.} المسايرات: $1/(R + jL_{vc}\omega) + jC_z\omega/(1 + jRC_z\omega)$؛
ومع $C_z = L_{vc}/R^2$ يكون الحدّ الثاني $jL_{vc}\omega/[R(R +
jL_{vc}\omega)]$، والمجموع $(R + jL_{vc}\omega)/[R(R + jL_{vc}\omega)]
= 1/R$. و$C_z = 0.50\times10^{-3}/64 = \qty{7.8}{\micro F}$.

\textbf{24.} $U = \sqrt{PR} = \sqrt{400} = \qty{20}{V}$ فعّالة. وعند $f_c$ يأخذ كل
مجهار النصف: أي \qty{25}{W}. وعند \qty{500}{Hz} يأخذ المجهار الحادّ $6\%$:
أي \qty{3}{W}.

\textbf{25.} الرتبة الأولى: عنصران، متتامّان في الاستطاعة، وميلان لطيفان
\qty{6}{dB/oct}، و$90^\circ$ بين المجهارين، وتسرّب الجهير إلى
المجهار الحادّ. والرتبة الثانية: أربعة عناصر، و\qty{12}{dB/oct}، و$180^\circ$ عند
تواتر الفصل (فيُقلب المجهار الحادّ)، وحماية أفضل بكثير للمجهار الحادّ. وفي
الحالتين معًا يجب ترويض تحريضية المجهار الجهيري بشبكة زوبل كي
يصحّ التصميم.
\end{solution}
