\begin{solution}{exo:b1:rays-reflection-refraction:2}

فلا شعاع منكسر --- بل \omterm{thm:b1:rays-reflection-refraction:tir}{انعكاس كلي} ($i_c = 48.8^\circ$).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
العينان على \qty{1.60}{m}. والشعاع الآتي من القدمين يصل إلى العين بعد
انعكاس عند نصف ارتفاع العين، أي \qty{0.80}{m}؛ والشعاع الآتي من قمة
الرأس ينعكس في منتصف ما بين القمة والعينين، عند \qty{1.65}{m}.
فالمرآة من \qty{0.80}{m} إلى \qty{1.65}{m}: ارتفاعها \qty{0.85}{m}، أي نصف
ارتفاع الجسم --- ولا يتوقف على المسافة (فقاعدة المنتصف
تصحّ عند أيّ مسافة).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
للزوايا الصغيرة يكون الموضع الظاهري $h' = h/n$ عند النظر من
الوسط الأقلّ كثافة، و $h' = nh$ من الوسط الأكثف. القطعة النقدية: $1.2/1.33 =
\qty{0.90}{m}$. والصياد: $1.33 \times 1.8 = \qty{2.4}{m}$. والتبرير:
من أجل نقطة سطح على بعد أفقي $x$ يكون $\tan i_2 \approx i_2 =
x/h$ و $\tan i_1 \approx i_1 = x/h'$ مع $i_1 = n i_2$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$، ومنه
$\frac{A + D_{\min}}{2} = 49.5^\circ$ و $D_{\min} = 38.9^\circ$.
والخروج يقتضي $r' \leq i_c$؛ كما أن $\sin r = \sin i/n \leq 1/n$ يعطي
$r \leq i_c$؛ ومنه $A = r + r' \leq 2i_c$. وهنا $i_c = \arcsin(1/1.52) =
41.1^\circ$ و $2i_c = 82.3^\circ > 60^\circ$: فالشرط محقق.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$ و
$\theta_a = 14.1^\circ$. و$\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$. وعلى \qty{1}{km}:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}

$5.30^\circ$؛ وعرض الطيف $0.20^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
الدخول: $\sin r = \sin 45^\circ/1.5 = 0.471$ و $r = 28.1^\circ$. والوجهان
متوازيان، فالورود على الوجه الثاني هو $r$ ويكون
$\sin i' = n\sin r = \sin i$: أي $i' = i$، فالشعاع الخارج مواز
للشعاع الوارد. وعلى امتداد المسار الداخلي الذي طوله $e/\cos r$ يكون
الانزياح العمودي $(e/\cos r)\sin(i - r)$:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
انكساران ($2(i - r)$) وانعكاسان ($2(\pi - 2r)$):
$D = 2\pi + 2i - 6r$. و$\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ يعطي
$3\cos i = n\cos r$ ثم $9\cos^2 i = n^2 - 1 + \cos^2 i$ ثم $\cos^2 i =
(n^2 - 1)/8$. ومن أجل $n = 1.333$: $\cos^2 i = 0.0971$ و $i = 71.8^\circ$ و
$\sin r = 0.950/1.333$ و $r = 45.5^\circ$ و $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$: فالقوس عند $230.9^\circ - 180^\circ =
51^\circ$ من النقطة المقابلة للشمس، خارج الابتدائي ($42^\circ$).
والأحمر ($n = 1.331$) عند $50.4^\circ$، والبنفسجي ($1.343$) عند $53.6^\circ$:
فالألوان معكوسة، والبنفسجي خارجًا.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$ و $i_c = 89.56^\circ$: فالأشعة
الواقعة داخل $0.44^\circ = \qty{7.7}{mrad}$ من سطح الطريق تنعكس
انعكاسًا كليًّا (وعمومًا الزاوية الماسّة $\approx \sqrt{2\Delta n/n}$). ومن
\qty{1.2}{m} علوًّا يمسّ خط النظر بهذه الزاوية بعد
$1.2/0.0077 \approx \qty{160}{m}$: فالطريق هناك تعكس السماء ---
وهي ``البركة''.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$؛
و$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$، وهو معدوم تمامًا حين
$\sin i_1/v_1 = \sin i_2/v_2$، أي $n_1\sin i_1 = n_2\sin i_2$ مع
$n = c/v$.
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$، ولا يتغير في الزجاج؛
و$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$.

\textbf{4.} \omterm{def:b1:rays-reflection-refraction:fiber}{القلب} العاري لا يوجّه الضوء إلا ما دام سطحه نظيفًا تمامًا
وغير ممسوس: فالغبار أو الماء أو قبضة يد تستبدل موضعيًّا بالهواء قرينةً
أعلى فيهرب الضوء، والخدوش تبعثره. أما \omterm{def:b1:rays-reflection-refraction:fiber}{الغلاف} فسطح فاصل محكم مضبوط
بقفزة قرينية مختارة عن قصد.

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$؛ ويكون موجَّهًا إذا وفقط إذا كان $\cos\theta' \geq
n_2/n_1$، أي $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$، أي
$\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$:
ومنه $\theta_a = 12.1^\circ$.

\textbf{8.} $\mathrm{NA} = 0.210$. والقفزة الأكبر توسّع المخروط، فيُلتقط
قدر أكبر من ضوء منبع متباعد (مثل LED).

\textbf{9.} الشعاع مماسٌّ للدائرة ذات نصف القطر $R_b$ (الحافة
الداخلية) ويلاقي الدائرة ذات نصف القطر $R_b + 2a$ (الحافة الخارجية)؛ وفي
المثلث القائم المركز--نقطة التماس--نقطة الاصطدام تحقق الزاوية عند
نقطة الاصطدام بين الشعاع ونصف القطر (الناظم) العلاقة
$\sin i = R_b/(R_b + 2a)$. ويكون موجَّهًا إذا وفقط إذا كان $\sin i \geq n_2/n_1$:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$. والانحناءات الأضيق تسرّب الضوء.

\textbf{10.} $t_0 = Ln_1/c$؛ ومسار الشعاع الأشدّ ميلًا $L/\cos\theta'_{\max} =
Ln_1/n_2$، ومنه $t_1 = Ln_1^2/(n_2 c)$.

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$؛ وعلى \qty{2}{km}: \qty{101}{ns}.

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} الأشعة البعيدة عن المحور تقطع مسارات أطول لكن في قرينة
أدنى، أي أسرع؛ ومع مقطع مختار جيدًا يتعاوض الأثران
فتصل الأشعة كلها معًا.

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$: \omterm{def:b1:rays-reflection-refraction:fiber}{فالقلب} ليس
كبيرًا في مقابل طول الموجة، فيهيمن الحيود ولا معنى للأشعة.
وتبيّن المعالجة الموجية نمطًا موجَّهًا واحدًا: أي لا اتساع
بين الأنماط البتة؛ ولا يبقى إلا التبدد اللوني والوهن.

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$:
ومنه $P/P_0 = 10^{-2}$، أي واحد بالمئة.

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$، ومنه $L_{\max} =
30/0.20 = \qty{150}{km}$.

\textbf{18.} تكلّف \qty{100}{km} مقدار \qty{20}{dB}: فالهامش \qty{10}{dB}، أي
عامل $10$ في الاستطاعة.

\textbf{19.} المنبع LED: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$ و $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$. والليزر:
$\Delta t_c = \qty{170}{ps}$ و $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$.

\textbf{20.} $30/2.0 = \qty{15}{km}$، أي أقلّ عشر مرات. فالسيليس أشدّ ما
يكون شفافية قرب \qty{1550}{nm} (عند أصغر وهنٍ له)، وهناك توجد
المضخّمات البصرية.

\textbf{21.} $30/30 = \qty{1}{km}$ بين المضخّمات، في مقابل
\qty{150}{km}: فالليف يفوز بعامل $150$.

\textbf{22.} بين الأنماط: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s})؛
واللوني (منبع ليزري): $17 \times 0.1 \times 2 = \qty{3.4}{ps}$؛
والوهن \qty{0.4}{dB}، وهو مهمل. فالاتساع بين الأنماط هو الحادّ:
\qty{5}{Mbit/s}.

\textbf{23.} لا اتساع بين الأنماط؛ واللوني \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}؛ والوهن \qty{20}{dB}، وهو داخل الميزانية. فالتبدد
اللوني هو الحادّ، عند نحو \qty{3}{Gbit/s}.

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} القفزة القرينية: \qty{10}{Mbit.s^{-1}.km}؛ وقناة أحادية النمط:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- أي عامل نحو
$3\times10^4$. وحقيقة واحدة: مسار موجَّه وحيد، ومنه انعدام الاتساع
الهندسي (بين الأنماط) في أزمنة الوصول.
\end{solution}
