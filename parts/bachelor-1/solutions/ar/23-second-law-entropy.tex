\section*{الفصل \ref{ch:b1:second-law-entropy} --- المبدأ الثاني: الإنتروبيا}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$: فالانصهار $334000/273 = \qty{1.22}{kJ/K}$؛ والغليان $2260000/
373 = \qty{6.06}{kJ/K}$. فالتبخر يحرّر الجزيئات بعضها من بعض
تحريرًا تامًّا --- أي فوضى مجهرية أكبر بكثير، وحرارة أكبر بكثير لكل
\omterm{def:b1:kinetic-theory:temperature}{كلفن}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- وهي دالة
حالة: فلا تتوقف على الطريقة (أما الإنتروبيا \emph{المُنشأة} فتتوقف عليها).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$. وفي التمدد المتساوي الحرارة العكوس: $Q = nRT\ln2$
من الحوض عند $T$، و $S_{\mathrm{exch}} = nR\ln2$، والمُنشأة $0$. وفي \omterm{rem:b1:first-law:irreversible}{تمدد جول}:
$Q = 0$ و $S_{\mathrm{exch}} = 0$، والمُنشأة \qty{5.8}{J/K}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
ثبات $S = nC_{V,m}\ln T + nR\ln V$ يعني ثبات $T^{C_{V,m}}V^R$، أي
ثبات $TV^{R/C_{V,m}} = TV^{\gamma - 1}$. وفي الانضغاط المفاجئ: $Q = 0$، ومنه
$S_{\mathrm{exch}} = 0$ و $\Delta S = S_{\mathrm{created}}$؛ ومع $T$: $300 \to
643$ \omterm{def:b1:kinetic-theory:temperature}{كلفن}، و $V$: $24.9 \to 10.7$ لتر: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ مُنشأة.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
الكتلة: $450\ln(293/373) = \qty{-109}{J/K}$؛ والبحيرة: $+36000/293 = \qty{+123}{J/K}$؛
والكون: $+\qty{14}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$؛ و $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$، وكلها مُنشأة. ولدينا $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$، ومنه أن وسيط اللوغاريتم $\geq 1$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$: أي $1000(1/280 - 1/300) = \qty{0.24}{J/K}$؛ ومن الفرن إلى
الغرفة: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$، أي عشرة أضعاف: فكلما اتسع
فرق \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} الذي يعبره جريان حراري ازداد لاعكوسية --- والهبوط
من الفرن إلى الغرفة هو موضع تبديد الطاقة الرفيعة.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
في دور: $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$ مع $S_{\mathrm{created}} \geq 0$،
ومنه $Q \leq 0$ و $W = -Q \geq 0$: فلا تستطيع الآلة أن تقدّم شغلًا. والبحر
منظِّم حراري واحد: فلا تتحول حرارته إلى شغل من دون
جسم أبرد تُلقى فيه الإنتروبيا.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
يتمدد كل غاز من $V$ إلى $2V$ عند ثبات $T$: ومنه $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$، وهي مُنشأة (إذ لا حرارة تُتبادَل، والصندوق معزول). أما نصفان
من الآزوت: فرفع الجدار لا يغيّر شيئًا عيانيًّا ---
ويكون $\Delta S = 0$ (وهي مفارقة غيبس: فالجزيئات المتطابقة لا ``تمتزج'').
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
$Q = \qty{3.6e5}{J}$ تدخل الغرفة عند \qty{293}{K}: فتُنشأ $S = \qty{1230}{J/K}$
--- في الشعيرة (إذ يتحول الشغل الكهربائي إلى حرارة) وفي
جريان تلك الحرارة من الشعيرة الساخنة إلى الغرفة الباردة؛ أما الشغل
الكهربائي نفسه فلم يحمل إنتروبيا.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
$\Delta S_{\text{الجسم}} = C\ln(T_0/T_1)$ في كل حالة. (أ) المتبادَلة
$C(T_0 - T_1)/T_0$: فالمُنشأة $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$. (ب) مرحلتان
نسبة كلٍّ منهما $r = \sqrt{T_1/T_0}$: فالمُنشأة $2C[r - 1 - \ln r]$، وهي أصغر
(فمن أجل $T_1/T_0 = 2$: $0.193C$ مقابل $0.307C$). (ج) عدد لا منتهٍ من المراحل:
تُنشئ كلٌّ منها $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$،
ومجموعها إلى الصفر حين $\epsilon \to 0$: فالتبريد العكوس يقتضي منظِّمًا حراريًّا
عند كل \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} وسيطة --- فلا يجوز للحرارة أن تعبر فرقًا منتهيًا أبدًا.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$؛ و $2^{-100} = \num{8e-31}$؛ و $2^{-10^{22}} = 10^{-3\times10^{21}}$.
ولدينا $k_B\ln2^N = Nk_B\ln2 = nR\ln2$. و``المستحيل'' يعني بدقة أن الاحتمال
معدوم؛ وهو هنا $10^{-3\times10^{21}}$ --- ليس صفرًا، لكنه من الصغر بحيث
لا يمكن رصد أيّ فرق أبدًا: فاللاعكوسية إحصاء عند
مقياس $10^{23}$.
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$
للقهوة: فتكسب الغرفة \qty{73}{kJ}، ويكون $\Delta S_{\text{الغرفة}} = 73150/293
= \qty{+250}{J/K}$.

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$.

\textbf{3.} $S_{\mathrm{created}} = -224 - (-250) = \qty{+26}{J/K} > 0$.

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$؛ و $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$ و $S_{\mathrm{exch}} = \qty{-117.7}{J/K}$: فالمُنشأة
\qty{18}{J/K} حتى الآن --- ومعظم اللاعكوسية يقع بينما تكون
القهوة أشدّ سخونة.

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$ مع $x = T_1/T_0$؛ و $f'(x) = 1 - 1/x$ ينعدم عند $x = 1$ حيث
$f = 0$، و $f'' = 1/x^2 > 0$: فالأصغر هو $0$.

\textbf{6.} لا: فالحالتان الابتدائية والنهائية هما هما، والحرارة هي
هي، والغرفة هي المنظِّم الحراري نفسه --- فالإنتروبيا المُنشأة تعيّنها
الحالتان، لا السرعة.

\textbf{7.} $T_f = \qty{350}{K}$؛ والمُنشأة $4180\ln(350^2/120000) = \qty{86}{J/K}$.

\textbf{8.} في \omterm{def:b1:first-law:transformation}{التحول العكوس} تنحفظ الإنتروبيا الكلية: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$، ومنه $T_f'^2 = T_1T_2$.

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$؛ وتفقد الكتلتان المائيتان $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$، تُقدَّم
شغلًا.

\textbf{10.} الساخنة: $4180\ln(346.4/400) = \qty{-601}{J/K}$؛ والباردة: $4180\ln(346.4/
300) = \qty{+601}{J/K}$: فالمجموع معدوم، كما تقتضي العكوسية.

\textbf{11.} $T_0S_{\mathrm{created}} = 300 \times 86 = \qty{26}{kJ}$، وهي قريبة من
\qty{30}{kJ} القابلة للاستعادة: فالإنتروبيا المُنشأة، مضروبةً \omterm{def:b1:kinetic-theory:temperature}{بدرجة الحرارة}
المرجعية، هي الشغل الذي رماه المسار اللاعكوس.

\textbf{12.} كان ذلك ليقتضي محركًا يعمل بين كتلتين مائيتين
فاترتين تتغير درجتا حرارتهما باستمرار --- أي بضع عشرات من الكيلوجولات
لآلة كبيرة: فلا يستحق العناء، لكن المبدأ الثاني يقول إن الشغل كان
هناك.

\textbf{13.} المتلقاة من الداخل $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$؛
والمعطاة للغرفة $Q_0/T_0 = (40 + W)/293$؛ والموازنة على امتداد دور:
$0 = 0.144 - (40 + W)/293 + S_c$.

\textbf{14.} $S_c \geq 0$: ومنه $(40 + W)/293 \geq 0.144$ و $W \geq 42.2 - 40 =
\qty{2.2}{W}$؛ ومعامل الأداء $\leq 40/2.2 = 18$ (وهو $= T_c/(T_0 - T_c)$).

\textbf{15.} $\mathrm{COP}_{\max} = T_c/(T_0 - T_c)$: فمن أجل مجمّد $255/38 = 6.7$
--- فكلما ازداد الصندوق بردًا غلا كل جول يُنزَع منه.

\textbf{16.} مع $W = 15$: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ تُنشأ
في كل ثانية.

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$؛ وتلقي الآلة
\qty{200}{W} مع شغلها هي في الغرفة بينما يبقى الداخل
باردًا: فمكسب الغرفة الصافي هو $W$ --- والثلاجة المفتوحة مدفأة.

\textbf{18.} لو كان $W = 0$ لأعطى $S_c = Q_c(1/T_0 - 1/T_c) < 0$: وهو ممنوع ---
وذلك هو \omterm{prop:b1:second-law-entropy:clausiuskelvin}{نصّ كلاوزيوس}.

\textbf{19.} قبل ذلك: $1$ (كلها في اليسار)؛ وبعده: $2^N$؛ ومنه $\Delta S = k_B\ln2^N =
Nk_B\ln2$.

\textbf{20.} لدينا $Nk_B = nR$: ومنه $\Delta S = nR\ln2$، وهي نتيجة \omterm{rem:b1:first-law:irreversible}{تمدد جول}.

\textbf{21.} بتّ واحد لكل جزيء: أي $N$ بتًّا؛ ولدينا $\Delta S/(k_B\ln2) = N$ ---
فالإنتروبيا تعدّ المعلومات التي يحتاج إليها المرء ليحدّد مواضع
الجزيئات.

\textbf{22.} $2^{-50} = \num{9e-16}$؛ وللمول $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$.

\textbf{23.} لا حرارة ($Q = 0$): فالمتبادَلة $0$، والمُنشأة $nR\ln2$ --- وهي
أصفى الحالات: لا شغل ولا حرارة، بل انتشار الغاز لا غير.

\textbf{24.} يتلقى الغاز $W = nRT\ln2$، ويعطي الغرفة $Q = -nRT\ln2$،
وتكون $S_{\mathrm{exch}} = -nR\ln2$ (فتعود إنتروبيا الغاز إلى قيمتها
الابتدائية، على نحو عكوس): فنقض التمدد الحرّ يكلّف الشغل $nRT\ln2$
ويلقي $nR\ln2$ من الإنتروبيا في الغرفة --- ويحتفظ الكون بالمقدار $nR\ln2$
التي أُنشئت مرة واحدة إلى الأبد.

\textbf{25.} لا تنحفظ الإنتروبيا في شيء حقيقي --- بل في النهاية
العكوسة المثالية وحدها؛ والمبدأ يمنع كل سيرورة تُفنيها (حرارة
تصعد، وشغل من منظِّم حراري واحد، وفصل مجاني لمزيج)؛
وكل بتّ يُنشأ هو شغل ضائع، مقداره $T_0S_{\mathrm{created}}$.
\end{solution}
