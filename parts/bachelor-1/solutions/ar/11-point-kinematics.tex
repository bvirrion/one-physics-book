\section*{الفصل \ref{ch:b1:point-kinematics} --- تحريك النقطة المادية}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: أي \qty{12}{m/s} عند \qty{3.0}{s}؛ و$a = \qty{4.0}{m/s^2}$؛
و$x(5) = \qty{50}{m}$؛ و$\qty{100}{km/h} = \qty{27.8}{m/s}$ عند $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$ و $\dot\theta = \qty{3.0}{rad/s}$: فيكون $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (بوحدة \unit{m/s})؛ و $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (بوحدة \unit{m/s^2}). وهي \omterm{cor:b1:point-kinematics:circular}{حركة دائرية} منتظمة: قيمة \omterm{def:b1:point-kinematics:velocity}{سرعة} ثابتة و
\omterm{cor:b1:point-kinematics:circular}{تسارع مركزي}.
\end{solution}

و$a = v^2/R = \qty{0.034}{m/s^2}$، أي $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$ و $v_0\cos\alpha = \qty{9.19}{m/s}$.
والقمة عند $t = 7.71/9.81 = \qty{0.79}{s}$، وارتفاعها $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. والمدى $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
وعند الوصول: $(9.19, -7.71)$، أي \qty{12}{m/s} بزاوية $40^\circ$ تحت
الأفق.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$ و $a_N = 10^2/25 = \qty{4.0}{m/s^2}$ و $a =
\qty{4.1}{m/s^2}$. وبعد \qty{5}{s}: $v = \qty{15}{m/s}$ و $a_N = 225/25 =
\qty{9.0}{m/s^2}$ و $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$ و $\dot r = b\omega$ و $\ddot r = 0$ و $\dot\theta = \omega$:
فيكون $\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$ و $v = b\omega
\sqrt{1 + \omega^2t^2}$؛ و $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. والحدّ $2\dot r\dot\theta$ نصفان متساويان: فاتجاه
\omterm{def:b1:point-kinematics:velocity}{السرعة} القطرية يدور بمعدل $\dot\theta$ (فيعطي
$\dot r\dot\theta\,\vect e_\theta$)، والانتقال إلى الخارج يزيد
\omterm{def:b1:point-kinematics:velocity}{السرعة} العمودية على القطر $r\dot\theta$ (فيعطي $\dot r\dot\theta$ آخر).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$؛ و$a = R\omega^2 =
\qty{2.0}{m/s^2}$ نحو المحور؛ والخطوة $v_z \times 2\pi/\omega =
\qty{3.1}{m}$؛ و$\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$؛ و$a_{\max} = A\omega^2 = \qty{10}{m/s^2}$؛
وتتقدّم $v$ على $x$ بربع دور، ويكون $a = -\omega^2x$ معاكسًا للموضع $x$.
وتكون $\abs v$ أكبر ما تكون عند $x = 0$، و $\abs a$ عند الطرفين $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$؛ و$\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. ومنه
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$، وباشتقاق
مرة أخرى بقاعدة الجداء نحصل على \omterm{def:b1:point-kinematics:velocity}{تسارع}
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
عند القمة: $v = v_0\cos\alpha$ و $a_N = g$: ومنه $\rho = v_0^2\cos^2\alpha/g$.
وعند القذف: $v = v_0$ و $a_N = g\cos\alpha$: ومنه $\rho = v_0^2/(g\cos\alpha)$.
فنصف قطر القمة أصغر بالعامل $\cos^3\alpha$: إذ ينحني القطع المكافئ أشدّ ما ينحني
حيث يكون أبطأ وحيث يكون $\vect g$ كله ناظميًّا.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$ و $\dot y = v\sin(vt/R)$؛ و$\ddot x = (v^2/R)
\sin(vt/R)$ و $\ddot y = (v^2/R)\cos(vt/R)$. وعند الأرض (حيث $vt/R = 2\pi n$):
$\dot x = \dot y = 0$ و $\vect a = (0, v^2/R)$، أي إلى أعلى. وعند القمة
(حيث $vt/R = \pi$): $\dot x = 2v$ و $\dot y = 0$: فقيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$ و $\dot r = u$ و $\dot\theta = \omega$: فيكون $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. وعند $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (بوحدة \unit{m/s^2}). والأول هو الحدّ المركزي
للدوران؛ أما الثاني، $2\dot r\dot\theta$، فلا يوجد إلا لأن
الاتجاه القطري نفسه يدور --- فعلى الأرض لا يعطي المشي مستقيمًا
\omterm{def:b1:point-kinematics:velocity}{بسرعة} ثابتة أيّ \omterm{def:b1:point-kinematics:velocity}{تسارع} البتة.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$؛ و$\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (والحركة في اتجاه $\theta$ المتزايدة)، و
$\vect N = -\vect e_r$ (نحو $C$)، و $\rho = R$.

\textbf{3.} عند القمة: $v^2 = 625 - 392 = 233$ و $v = \qty{15.3}{m/s}$. وعند الجانبين:
$625 - 196 = 429$ و $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ و $v = R\dot\theta$:
ومنه $\dot v = -g\sin\theta$. وهو سالب من أجل $0 < \theta < \pi$ (أي إبطاء في
الصعود)، وموجب في النزول.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: فعند الأسفل
\qty{62.5}{m/s^2}؛ وعند الجانبين $62.5 - 19.6 = \qty{42.9}{m/s^2}$؛ وعند القمة $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} عند الأسفل: \qty{62.5}{m/s^2} (بلا جزء مماسي)؛ وعند الجانبين:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$؛ وعند القمة: \qty{23.3}{m/s^2}،
مشيرًا شاقوليًّا إلى الأسفل نحو $C$: فالزاوية $0$ مع الشاقول.

\textbf{7.} عمودية عند الأسفل والقمة (حيث $a_T = 0$)؛ وليست موازية
أبدًا (لأن $a_N > 0$ في كل مكان).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
فعند الأسفل \qty{2.50}{rad/s}، وعند القمة \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: وهي معادلة
نواس بسيط طوله $R$ --- فالعربة على مضمارها ثقل نواس
يعبر القمة.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} متوسط $25$ و$20.7$ و$15.3$ و$20.7$: نحو \qty{20}{m/s}؛
والطول $2\pi R = \qty{63}{m}$: أي نحو \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: فنعم. وأصغر قيمة: $v_0^2 \geq
gR + 4gR = 5gR = 490$، أي $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} عند الأسفل: $\vect a = 62.5$ إلى أعلى و $\vect g$ إلى أسفل:
ومنه $\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} عند القمة: كلاهما إلى أسفل، $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$،
يضغط الراكب في مقعده (أي نحو المركز، وهو إلى أعلى بالنسبة إلى
الراكب المقلوب).

\textbf{15.} عند الجانبين: $\vect a = 42.9$ أفقيًّا نحو $C$ زائد $9.8$
إلى أسفل (مماسيًّا)؛ والمقدار $\vect a - \vect g$ يساوي $42.9$ أفقيًّا: أي $4.4g$
جانبيًّا نحو المركز.

\textbf{16.} تفشل: $7.4g$ عند الأسفل (أما القمة والجانبان فلا بأس بهما).

\textbf{17.} وزن ظاهري معدوم عند القمة: $v_{\text{القمة}}^2 = gR$؛
ومنه $v_0^2 = gR + 4gR = 5gR$ و $a_N = 5g$ عند الأسفل، فتكون الحمولة
$5g + g = 6g$، مستقلة عن $R$.

\textbf{18.} $a_N = v^2/\rho$: فقيمة $v$ يثبّتها الارتفاع (بالطاقة)، و $\rho$
تثبّتها هندسة المضمار؛ فقيمة مختلفة لنصف القطر $\rho$ عند الارتفاع نفسه تغيّر
$a_N$ وحده.

\textbf{19.} $a_N + g = 3g$: ومنه $a_N = 2g$ و $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{القمة}} = \qty{15.3}{m/s}$؛ ومن $a_N - g = 1.5g$ نجد $a_N =
\qty{24.5}{m/s^2}$ و $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: فهو ينمو خطيًّا
من الصفر --- أي بمعدل تغيّر ثابت \omterm{def:b1:point-kinematics:velocity}{للتسارع} --- بدلًا من
القفز من $0$ إلى $v^2/R$ عند مدخل دائرة، وهو ما يشعر به العنق ضربةً.

\textbf{22.} على شكل دمعة: عريضة لطيفة التقوّس عند الأسفل حيث تكون
العربة سريعة، وضيقة عند القمة حيث تكون بطيئة. فتُبقي الحمولة قرب
$3g$ في كل مكان وتأتي بها تدريجيًّا.

\textbf{23.} خط النظر: البعد الأفقي \qty{50}{m} والارتفاع
\qty{20}{m}: ومنه $r = \qty{53.9}{m}$ وزاوية الارتفاع $\beta = 21.8^\circ$. و
المركّبة المستعرضة \omterm{def:b1:point-kinematics:velocity}{للسرعة} الأفقية هي $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$؛ ومنه $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} منظورًا من نقطة خارج المركز لا يكون المعدل الزاوي لخط
النظر ثابتًا (فهو سريع حين تقترب العربة، وبطيء حين تبتعد):
فتنزلق آلة تصوير ذات معدل ثابت عن العربة.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{الإحداثيات القطبية} عند $C$ \omterm{def:b1:point-kinematics:velocity}{للسرعة} \omterm{def:b1:point-kinematics:velocity}{والتسارع} على
الدائرة؛ وفصل فريني بين ``التسريع'' و``الدوران'' و
من أجل الكلوثويد؛ \omterm{def:b1:point-kinematics:cylindrical}{والإحداثيات القطبية} عند \omterm{def:b1:optical-instruments:camera}{آلة التصوير} لمعدل
التعقّب؛ والطاقة (في الفصول التالية) لقانون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة}.
\end{solution}
