\section*{الفصل \ref{ch:b1:magnetostatics} --- المغناطيسية الساكنة: المجال والقوى وثنائيات القطب}

\begin{solution}{exo:b1:magnetostatics:1}
$B = \mu_0I/2\pi r = 2 \times 10^{-7} \times 10/0.01 = \qty{2e-4}{T}$. ومن أجل \qty{1}{T}:
$I = 2\pi rB/\mu_0 = \qty{5e4}{A}$. ويُبلَغ مجال الأرض عند $r = 2 \times 10^{-7} \times
100/5 \times 10^{-5} = \qty{0.4}{m}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:2}
عند المركز: $\mu_0NI/2R = 4\pi \times 10^{-7} \times 200/0.1 = \qty{2.5}{mT}$. وعند $z = R$:
$\times(R^2/2R^2)^{3/2} = 2^{-3/2}$، أي \qty{0.89}{mT}. وعند $z = 10R$: $\times(1/101)^{3/2}
= 10^{-3}$، أي \qty{2.5}{\micro T}، وهو قانون ثنائي القطب $1/z^3$ من الآن.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:3}

\end{solution}

\begin{solution}{exo:b1:magnetostatics:4}
$F = ILB = 10 \times 1 \times 0.1 = \qty{1.0}{N}$؛ وعند \qty{30}{\degree}، بالعامل $\sin30^\circ$:
\qty{0.5}{N}. وفي المحرك: $100 \times \qty{1}{N} \times \qty{0.05}{m} = \qty{5}{N.m}$ ---

\end{solution}

\begin{solution}{exo:b1:magnetostatics:5}
مع $z = d\tan\alpha$: $B = \dfrac{\mu_0I}{4\pi}\displaystyle\int\dfrac{d\,\dd z}{(d^2 + z^2)^{3/2}}
= \dfrac{\mu_0I}{4\pi d}\displaystyle\int_{\alpha_1}^{\alpha_2}\cos\alpha\,\dd\alpha = \dfrac{\mu_0I}{4\pi d}
(\sin\alpha_2 - \sin\alpha_1)$؛ وفي السلك اللانهائي $\alpha_{1,2} = \mp\pi/2$: ومنه $\mu_0I/2\pi d$.
وفي المربع: كل ضلع عند $d = a/2$ يُرى تحت $\pm45^\circ$ فيعطي $\mu_0I\sqrt2/2\pi a$؛
وبأربعة أضلاع: $B = 2\sqrt2\,\mu_0I/\pi a = 0.90\,\mu_0I/a$ (ودائرة ``نصف قطرها''
$a/2$ كانت لتعطي $\mu_0I/a$).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:6}
$B(z) = f(z - R/2) + f(z + R/2)$ مع $f(u) = \mu_0NIR^2/2(R^2 + u^2)^{3/2}$.
وعند نقطة المنتصف: $2f(R/2) = \mu_0NIR^2/(5R^2/4)^{3/2} = (4/5)^{3/2}\mu_0NI/R$. والدالة $f'$
فردية، ومنه $B'(0) = f'(-R/2) + f'(R/2) = 0$؛ و $f''(u) \propto (4u^2 - R^2)(R^2 +
u^2)^{-7/2}$ ينعدم عند $u = R/2$، ومنه $B''(0) = 2f''(R/2) = 0$: فالمجال
منتظم إلى المرتبة الثالثة. والأعداد: $0.716 \times 4\pi \times 10^{-7} \times 100/0.15 =
\qty{0.60}{mT}$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:7}
عند السطح: $\mu_0I/2\pi a = 2 \times 10^{-7} \times 100/0.002 = \qty{10}{mT}$؛ وعند
\qty{1}{mm} في الداخل، بالعامل $\times r/a$: \qty{5}{mT}؛ وعند \qty{1}{cm}: \qty{2}{mT}. ويتزايد $B$
خطيًّا حتى السطح، ثم يتناقص بنسبة $1/r$. وفي الكبل المحوري: خارج الغلاف
يكون التيار المحصور معدومًا، ومنه $B = 0$.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:8}
$2 \times 10^{-7}$~نيوتن في المتر. وفي القضبان الناقلة: $2 \times 10^{-7} \times 10^8/0.1 = \qty{200}{N/m}$؛
وعند \qty{100}{kA}: $\qty{2e4}{N/m}$ --- أي طنان في المتر. ولذا تُقاس دعائم
معدات القطع العازلة على تيار العطل لا على تيار
التشغيل.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:9}
$\Gamma = mB_h = \qty{2e-6}{N.m}$؛ و $W = 2mB_h = \qty{4e-6}{J}$؛ و $J\ddot\theta =
-mB_h\theta$: ومنه $T = 2\pi\sqrt{J/mB_h} = 2\pi\sqrt{10^{-7}/2 \times 10^{-6}} =
\qty{1.4}{s}$. ويُقاس بتوقيت الاهتزازات: $B_h = 4\pi^2J/mT^2$ متى عُرفت $m$ و $J$
(وقد قاس غاوس $m$ على حدة من الانحراف الذي تُحدثه في
إبرة ثانية).
\end{solution}

\begin{solution}{exo:b1:magnetostatics:10}
(أ) $\mu_0IR^2/2z^3 = \mu_0(I\pi R^2)/2\pi z^3 = \mu_0m/2\pi z^3 = (\mu_0/4\pi)\,2m/z^3$:
أي المقدار الكهربائي $2p/4\pi\varepsilon_0z^3$ مع $1/\varepsilon_0 \to \mu_0$. (ب) و $m = 2\pi R_E^3B/
\mu_0 = 2\pi \times 2.58 \times 10^{20} \times 6 \times 10^{-5}/1.26 \times 10^{-6} = \qty{7.7e22}{A.m^2}$.
(ج) و $I = m/\pi R^2 = 7.7 \times 10^{22}/(\pi \times 9 \times 10^{12}) = \qty{2.7e9}{A}$. (د)
وعند خط الاستواء ($\theta = 90^\circ$): $B_\theta = \mu_0m/4\pi R_E^3 = \qty{3e-5}{T}$، أي نصف
القيمة القطبية. وعند $\theta = 45^\circ$: $\tan(\text{الميل}) = B_r/B_\theta = 2\cot\theta = 2$،
فيكون الميل $= \qty{63}{\degree}$ --- فالمجال يغوص في الأرض بحدّة عند
خطوط العرض الوسطى.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:11}
$\mu_0nI = 4\pi \times 10^{-7} \times 2000 \times 3 = \qty{7.5}{mT}$. وعند المركز: $\tan\alpha_1 =
R/(L/2) = 0.2$ و $\cos\alpha_1 = -\cos\alpha_2 = 0.981$: ومنه $B = 0.981\,\mu_0nI =
\qty{7.4}{mT}$ (أي $98\%$). وعند الطرف: $\cos\alpha_1 = 0$ و $\tan\alpha_2 = R/L$ و $\cos\alpha_2 =
-0.995$: ومنه $B = 0.497\,\mu_0nI = \qty{3.7}{mT}$ (أي $50\%$). وفي الخارج عند $x$ وراء
الطرف، والطرفان في الجهة نفسها: $B = \tfrac12\mu_0nI(\cos\alpha_1 - \cos\alpha_2)
\approx \tfrac12\mu_0nI\,\tfrac{R^2}{2}[1/x^2 - 1/(x + L)^2]$؛ ويبلغ $1\%$ من القيمة
المركزية عند $x \approx \qty{10}{cm}$، أي على نصف طول في الخارج. وأما $B(z)$: فقمة مسطّحة على
معظم الطول، تهبط إلى النصف عند الطرفين وإلى بضعة بالمئة على بعد
طول واحد.
\end{solution}

\begin{solution}{exo:b1:magnetostatics:12}
(أ) يقفز مجال الملفات الخاص من $0$ إلى $B$ عبرها؛ و
المجال المؤثر فيها هو المتوسط، $B/2$ (فالصفيحة لا تدفع
نفسها). والقوة على عنصر من الملف طوله $\dd l$ وعرضه $\dd z$
يحمل $K\,\dd z$: هي $K\,\dd z\,\dd l\,B/2$، قطرية نحو الخارج (تحقّق بالمقدار $I\,\dd\vect l
\wedge\vect B$): فالضغط $p = KB/2 = B^2/2\mu_0$ لأن $K = nI = B/\mu_0$.
(ب) وعند \qty{1}{T}: \qty{4e5}{Pa} أي \qty{4}{bar}؛ وعند \qty{10}{T}: \qty{400}{bar}؛ وعند \qty{45}{T}:
\qty{8e8}{Pa} أي \qty{8000}{bar}، وهو قريب من \qty{1}{GPa} للفولاذ. (ج) وعلى
الملفات أن تحمل ضغطًا ينمو بنسبة $B^2$: ففوق \qty{40}{T} أو نحوها
لا تكفي مادة ولا خطة تبريد --- فحدّ المغانط المستقرة
حدّ ميكانيكي (وهو، في فائقات الناقلية، مجالها الحرج).
\end{solution}

\begin{solution}{pb:b1:magnetostatics:1}
\textbf{1.} كل مستوٍ يحوي المحور \omterm{prop:b1:electrostatics-gauss:symmetry}{مستوي تناظر}
للتيارات: فيكون $\vect B$ عموديًّا عليه، أي معامدًا؛ وبعدم التغيّر
على امتداد المحور وحوله: $\vect B = B(r)\,\vect e_\theta$.

\textbf{2.} دائرة نصف قطرها $r$: $2\pi rB = \mu_0I_{\text{المحصور}}$. وفي القلب:
$I_{\text{المحصور}} = Ir^2/a^2$ و $B = \mu_0Ir/2\pi a^2$؛ وبين الناقلين: $\mu_0I/2\pi r$؛
وفي الخارج ($r > c$): $I - I = 0$ و $B = 0$.

\textbf{3.} $B(a) = 2 \times 10^{-7}/5 \times 10^{-4} = \qty{4.0e-4}{T}$؛ و $B(b) = 2 \times
10^{-7}/2.5 \times 10^{-3} = \qty{8.0e-5}{T}$.

\textbf{4.} داخل الأنبوب لا تيار محصور: فيكون $B = 0$ هناك بدل
التزايد الخطي؛ ولا يتغير شيء في كل موضع آخر --- فالمجال خارج
تيار أسطواني لا يتوقف إلا على التيار الكلي.

\textbf{5.} $I_{\text{المحصور}} = I - I(r^2 - b^2)/(c^2 - b^2) = I(c^2 - r^2)/(c^2 -
b^2)$: ومنه $B = \dfrac{\mu_0I}{2\pi r}\,\dfrac{c^2 - r^2}{c^2 - b^2}$، ويساوي $\mu_0I/2\pi b$ عند
$b$ و $0$ عند $c$: فهو متصل. والرسم: تزايد خطي إلى \qty{4e-4}{T} عند
\qty{0.5}{mm}، ثم هبوط بالمقدار $1/r$ إلى \qty{8e-5}{T} عند \qty{2.5}{mm}، ثم هبوط إلى الصفر عند
\qty{3}{mm}، ثم صفر فيما وراء ذلك.

\textbf{6.} تيار الغلاف معاكس لتيار القلب: فالتنافر،
أي ضغط نحو الخارج. والتيار السطحي $K = I/2\pi b = \qty{64}{A/m}$؛ ومتوسط المجال
$\tfrac12(8 \times 10^{-5} + 0) = \qty{4e-5}{T}$: ومنه $p = KB = \qty{2.5e-3}{Pa}$. وعند
\qty{10}{kA}: بالعامل $\times10^8$، أي \qty{2.5e5}{Pa} = \qty{2.5}{bar} --- وكبلات
الاستطاعة النبضية مبنية لتحتمله.

\textbf{7.} لا مجال في الخارج: فالكبل لا يشعّ ولا يلتقط
تشويشًا مغناطيسيًّا من جيرانه. أما زوج الأسلاك: فيكاد المجالان
يتلاشيان، ولا يبقى إلا $\approx \mu_0Id/2\pi D^2 = 2 \times 10^{-7} \times 2 \times
10^{-3}/10^{-2} = \qty{4e-8}{T}$ عند \qty{10}{cm} --- وهو صغير، لكن المحوري يعطي
صفرًا.

\textbf{8.} $D = \sqrt{h^2 + d^2/4} = \qty{10.0}{m}$: فيعطي كلٌّ منهما $\mu_0I/2\pi D =
\qty{1.0e-5}{T}$، عموديًّا على المستقيم الواصل من الناقل. ولأن
التيارين متعاكسان تتلاشى المركّبات الموازية للمستقيم الواصل
بين الناقلين وتتجمع الشاقولية: ومنه $B = 2 \times 10^{-5} \times
(d/2)/D = \qty{1.0e-6}{T}$، شاقوليًّا. أي دون مجال الأرض خمسين مرة،
ودون الحدّ الإرشادي مئة مرة.

\textbf{9.} من أجل $D \gg d$ يكون $B \approx \dfrac{\mu_0I}{2\pi}\Bigl(\dfrac1{D_1} - \dfrac1{D_2}\Bigr)
\sim \dfrac{\mu_0I}{2\pi}\dfrac{d}{D^2}$: فمجالا $1/D$ يتلاشيان إلى المرتبة
الأولى ولا ينجو إلا فرقهما، وقياسه النسبي $d/D$ --- وهو
``ثنائي قطب من سلكين''.

\textbf{10.} الناقل نفسه على \qty{1}{m}: \qty{1.0e-4}{T} أفقيًّا؛ والآخر
على \qty{1.41}{m}: \qty{7.1e-5}{T} بزاوية \qty{45}{\degree}، معاكسًا: فالمحصلة
$(5 \times 10^{-5}, 5 \times 10^{-5})$، أي \qty{7e-5}{T} --- أي \qty{70}{\micro T}، وهو دون
الحدّ الإرشادي العام ودون الحدّ المهني بكثير.

\textbf{11.} $F/\ell = \mu_0I^2/2\pi d = 2 \times 10^{-7} \times 2.5 \times 10^5 =
\qty{0.05}{N/m}$، وهي تنافرية؛ وعند \qty{20}{kA}: \qty{80}{N/m}.

\textbf{12.} \qty{15}{N} في الحال العادية؛ و\qty{24}{kN} في دارة القصر، أي ثمانية أضعاف
ثقل الامتداد البالغ \qty{2.9}{kN}: فيُقذف الناقلان متباعدين، ويتأرجحان
وقد يصطدمان حين يعودان --- ومن هنا الفواصل والقواطع التي
تفتح في غضون بضعة أدوار.

\textbf{13.} $I\cos\omega t + I\cos(\omega t - 2\pi/3) + I\cos(\omega t + 2\pi/3) =
I\cos\omega t\,(1 + 2\cos2\pi/3) = 0$: فمجموع الأشعة الطورية الثلاثة معدوم. و
مجالات $1/D$ تتلاشى كما في المحوري؛ ويتناقص المجال البعيد بنسبة $1/D^2$ أو
أسرع.

\textbf{14.} مجال الخط هناك شاقولي؛ والبوصلة لا تستجيب إلا
للمركّبة الأفقية: فلا انحراف (ويتغير الميل بمقدار
$10^{-6}/(2 \times 10^{-5})$، أي ثلاث درجات، وهو ما لا تستطيع البوصلة إظهاره).

\textbf{15.} الضلعان $a$ (الموازيان للمحور): المجال $\vect B$ القطري
عمودي عليهما، فالقوة $NIaB$ على كلٍّ منهما، عمودية على
مستوي الوشيعة، ومتعاكسة على الضلعين: فهي مزدوجة. أما الضلعان $b$: فيكون $\dd\vect l$
فيهما قطريًّا، موازيًا للمتجهة $\vect B$: فلا قوة.

\textbf{16.} $\Gamma = 2 \times NIaB \times b/2 = NIabB = NISB = 0.024\,I$ (بالنيوتن متر، مع $I$
\omterm{def:b1:dc-circuits:current}{بالأمبير}). والمجال القطري يقع دائمًا في مستوي الوشيعة وعموديًّا
على الضلعين $a$: فتكون القوتان دائمًا ناظميتين على المستوي ولا يتغير
ذراعهما $b/2$ أبدًا.

\textbf{17.} $C\theta = NISB$: ومنه $\theta = 2400\,I$ راديان، أي \qty{2.4}{rad/mA}؛ ويقع
الانحراف التامّ \qty{1.2}{rad} عند $I = \qty{0.50}{mA}$.

\textbf{18.} الطول $200 \times 2(0.02 + 0.03) = \qty{20}{m}$، والمقطع $\pi(5 \times
10^{-5})^2 = \qty{7.9e-9}{m^2}$: ومنه $R = 1.7 \times 10^{-8} \times 20/7.9 \times 10^{-9} =
\qty{43}{\ohm}$؛ و $U = 43 \times 5 \times 10^{-4} = \qty{22}{mV}$؛ و $P = UI = \qty{11}{\micro W}$.

\textbf{19.} $J\ddot\theta = -C\theta + NISB\,I$: ومنه $\omega_0 = \sqrt{C/J} = \sqrt{10^{-5}/6.7
\times 10^{-8}} = \qty{12}{rad/s}$ و $T = \qty{0.51}{s}$.

\textbf{20.} $J\ddot\theta + \alpha\dot\theta + C\theta = NISB\,I$: ويكون حرجًا عند
$\alpha = 2\sqrt{JC} = 2\sqrt{6.7 \times 10^{-13}} = \qty{1.6e-6}{N.m.s}$؛ وعندئذ تبلغ الإبرة
قراءتها بأسرع ما يمكن بلا تجاوز.

\textbf{21.} في الفولطمتر: $R_s = 10/5 \times 10^{-4} - 43 \approx \qty{20}{k\ohm}$.
وفي الأمبيرمتر: تحمل مقاومة التحويل \qty{0.9995}{A} تحت \qty{22}{mV}: ومنه $R_{\text{التحويل}} =
\qty{22}{m\ohm}$.

\textbf{22.} $m = NIS = 200 \times 5 \times 10^{-4} \times 6 \times 10^{-4} = \qty{6e-5}{A.m^2}$؛
و $-mB = \qty{-1.2e-5}{J}$؛ وفي النابض $\tfrac12C\theta^2 = \tfrac12 \times 10^{-5} \times 1.44 =
\qty{7.2e-6}{J}$: أي الرتبة نفسها --- فالنابض يخزّن ما أدّاه \omterm{def:b1:magnetostatics:moment}{العزم
المغناطيسي} من شغل بينما كانت الوشيعة تدور.

\textbf{23.} في مجال منتظم يكون العزم المزدوج $mB\cos\theta$ (أعظميًّا حين
يحوي المستوي $\vect B$، ومعدومًا حين يكون الناظم على امتداده): ومنه $C\theta =
NISB\,I\cos\theta$ --- وهو خطي من أجل $\theta$ الصغيرة وحدها، ومنضغط في
أعلى التدريج.

\textbf{24.} \qty{50}{Hz} مقابل $\omega_0/2\pi = \qty{2}{Hz}$: فلا تستطيع الوشيعة
اللحاق؛ فتستقر عند متوسط التيار، وهو صفر، مع ارتعاش
\omterm{def:b1:potential-capacitors:capacitance}{سعته} $\approx \theta_{\text{الساكن}}(\omega_0/\omega)^2 = 1.2 \times (12/314)^2 \approx
\qty{2e-3}{rad}$، وهو غير مرئي. ومقياس الوشيعة المتحركة يقرأ المتوسط --- فمن أجل
التيار المتناوب يحتاج إلى مقوِّم.

\textbf{25.} \qty{4e-4}{T} عند قلب كبل محوري يحمل \qty{1}{A} ولا شيء في الخارج
(بمبرهنة أمبير)؛ و\qty{1}{\micro T} تحت خط يحمل \qty{500}{A} و\qty{0.05}{N/m}
بين ناقليه (ببيو-سافار مع لابلاس)؛ و\qty{2.4}{rad/mA} في
المقياس (بمزدوجة لابلاس في مواجهة نابض).
\end{solution}
