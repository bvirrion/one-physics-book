\section*{Bab \ref{ch:b1:newton-dynamics} --- Hukum Dinamika Newton}

\begin{solution}{exo:b1:newton-dynamics:1}
Bacaan timbangannya $N = m(g + a)$ dengan $a$ percepatan ke atasnya: ke
atas, $70 \times 11.81 = \qty{827}{N}$ (``\qty{84}{kg}''); ke bawah, $70 \times 7.81 =
\qty{547}{N}$ (``\qty{56}{kg}''); kecepatan tetap, \qty{687}{N}
(\qty{70}{kg}); jatuh bebas, $0$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:2}
Percepatan $a$ yang sama bagi keduanya (tali tak dapat memanjang): $m_1a = T$,
$m_2a = m_2g - T$, sehingga $a = m_2g/(m_1 + m_2) = \qty{3.3}{m/s^2}$,
$T = m_1a = \qty{6.5}{N}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:3}
$a = g(\sin\alpha - \mu_d\cos\alpha) = 9.81(0.50 - 0.17) = \qty{3.2}{m/s^2}$;
$v = \sqrt{2 \times 3.2 \times 5.0} = \qty{5.7}{m/s}$. Tak ada \omterm{def:b1:newton-dynamics:momentum}{massa} yang terlihat.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:4}
$k\Delta\ell = mg$: $\Delta\ell = 0.50 \times 9.81/200 = \qty{2.5}{cm}$.
Dengan $x$ sebagai simpangan dari kesetimbangannya, $m\ddot x = -kx$
(beratnya dan regangan statisnya saling meniadakan): $\omega_0 = \sqrt{k/m} = \qty{20}{rad/s}$,
$T = \qty{0.31}{s}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:5}
$\mu_s = \tan 22^\circ = 0.40$. Tikungan: $v \leq \sqrt{\mu_sgR}$: $\sqrt{0.8
\times 9.81 \times 50} = \qty{20}{m/s}$ (\qty{71}{km/h}); ketika basah,
\qty{14}{m/s} (\qty{50}{km/h}).
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:6}
Penerjun payung: $v_\infty = \sqrt{2 \times 785/(1.2 \times 0.70)} = \qty{43}{m/s}$;
$\mathrm{Re} = \rho vL/\eta \approx 1.2 \times 43 \times 0.5/1.8\times10^{-5}
\approx 10^6$: kuadratik. Tetes kabut: $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-12}{kg}$,
$v_\infty = mg/6\pi\eta r = 4.1\times10^{-11}/3.4\times10^{-9} = \qty{1.2}{cm/s}$;
$\mathrm{Re} = 1.2 \times 0.012 \times 2\times10^{-5}/1.8\times10^{-5} = 0.02$:
linear.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:7}
Dasar: $v^2 = 2g\ell(1 - \cos 60^\circ) = g\ell$; $T = mg + mv^2/\ell =
2mg$. Titik pelepasannya: $v = 0$, $T = mg\cos\theta_0 = 0.5mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:8}
$v = \sqrt{gR\tan\beta} = \sqrt{9.81 \times 200 \times 0.268} = \qty{23}{m/s}$
(\qty{82}{km/h}). Lebih lambat: mobilnya cenderung tergelincir menuruni
kemiringannya, gesekannya harus menunjuk mendaki; lebih cepat:
gesekannya menunjuk menurun untuk menambah \omterm{prop:b1:newton-dynamics:centripetal}{gaya sentripetalnya}.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:9}
$T\cos\alpha = mg$, $T\sin\alpha = m\omega^2\ell\sin\alpha$: $\omega =
\sqrt{g/\ell\cos\alpha} = \sqrt{9.81/0.866} = \qty{3.4}{rad/s}$, periodenya
\qty{1.9}{s}; $T = mg/\cos\alpha = 1.15mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:10}
$a = 27.8/8.0 = \qty{3.5}{m/s^2}$, $F = \qty{3.5}{kN}$; $F \leq \mu_smg$
memberi $\mu_s \geq a/g = 0.35$. Di jalan basah $\mu_s \approx 0.4$
membatasi percepatannya di dekat $0.4g$ berapa pun mesinnya: cengkeraman, bukan tenaga.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:11}
$v = v_\infty(1 - \eu^{-t/\tau})$, $v_\infty = g\tau = \qty{4.9}{m/s}$;
$z = \int v = v_\infty[t - \tau(1 - \eu^{-t/\tau})]$. Pada \qty{2.0}{s}:
$4.9 \times (2.0 - 0.5 \times 0.98) = \qty{7.4}{m}$, terhadap $\tfrac12 gt^2
= \qty{20}{m}$ di hampa.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:12}
Proyeksi radial (ke arah pusatnya): $N - mg\cos\theta = mv^2/R$, sehingga
$N = m(v^2/R + g\cos\theta)$. Puncak ($\theta = \pi$): $N = m(v^2/R - g)
\geq 0$ jika dan hanya jika $v \geq \sqrt{gR}$. Dengan hukum kelajuannya, $v_{\text{puncak}}^2 =
v_0^2 - 4gR \geq gR$: $v_0 \geq \sqrt{5gR}$; lalu di dasarnya $N =
m(5g + g) = 6mg$.
\end{solution}

\begin{solution}{pb:b1:newton-dynamics:1}
\textbf{1.} $m\dot v = mg$: $v = gt$, $z = \tfrac12 gt^2$; pada \qty{10}{s}:
\qty{98}{m/s}, \qty{490}{m}.

\textbf{2.} $mg = \qty{785}{N}$.

\textbf{3.} Di atas tanah, lantainya mendorong ke atas dengan $N = mg$
--- dorongan itulah yang dirasakan orang sebagai berat. Dalam jatuh
bebas tak ada \omterm{def:b1:newton-dynamics:momentum}{gaya} sentuh: gravitasi bekerja pada setiap bagiannya sama
rata dan tak ada yang menekan.

\textbf{4.} Ya, sampai sepersekian persen selama satu menit; yang
diabaikan: perputaran Bumi (suku sentrifugal dan Coriolis,
\cref{ch:b1:non-inertial-frames}) dan anginnya.

\textbf{5.} $F_d = \tfrac12\rho C_xSv^2 = 0.48\,v^2$ (SI).

\textbf{6.} $m\,\dd v/\dd t = mg - 0.48v^2$.

\textbf{7.} $v_\infty = \sqrt{mg/0.48} = \sqrt{1635} = \qty{40}{m/s} =
\qty{146}{km/h}$.

\textbf{8.} Bagi dengan $mg$: $\dd u/\dd t\,(v_\infty/g) = 1 - u^2$;
$\tau = 40.4/9.81 = \qty{4.1}{s}$.

\textbf{9.} $\int_0^u\dd u'/(1 - u'^2) = \operatorname{artanh}u = t/\tau$,
sehingga $u = \tanh(t/\tau)$.

\textbf{10.} $t = \tau\operatorname{artanh}(0.9) = 1.47\tau = \qty{6.1}{s}$;
$0.99$: $2.65\tau = \qty{11}{s}$.

\textbf{11.} $z = \int v_\infty\tanh(t/\tau)\dd t = v_\infty\tau\ln\cosh(t/\tau)$;
pada \qty{10}{s}: $166 \times \ln\cosh(2.43) = 166 \times 1.74 = \qty{290}{m}$,
terhadap \qty{490}{m}.

\textbf{12.} $\mathrm{Re} = 1.2 \times 40 \times 0.5/1.8\times10^{-5} \approx
\num{1.3e6}$: jauh di atas $1000$, jadi kuadratik.

\textbf{13.} $v_\infty' = \sqrt{785/(0.6 \times 25)} = \sqrt{52} = \qty{7.2}{m/s}$.

\textbf{14.} Hambatnya $= 15 \times 1600 = \qty{24}{kN}$; $a = (24000 - 785)/80
= \qty{290}{m/s^2} \approx 30g$: tali pengikatnya akan putus, atau penerjunnya.

\textbf{15.} Rata-rata $a \approx (40 - 7)/3 = \qty{11}{m/s^2} \approx 1.1g$;
\omterm{def:b1:newton-dynamics:momentum}{gaya} tali pengikatnya $\approx m(g + a) \approx \qty{1.7}{kN}$.

\textbf{16.} $\tau' = 7.2/9.81 = \qty{0.73}{s}$: dalam beberapa sekon
setelah terbuka penuh, penurunannya sudah tunak.

\textbf{17.} $\vect a = \vect 0$: \omterm{def:b1:newton-dynamics:momentum}{gaya} tali pengikatnya $= mg = \qty{785}{N}$.

\textbf{18.} Keseimbangan $mg = \tfrac12\rho C_xSv^2$ memberi $v \propto (C_xS)^{-1/2}$:
luasnya separuh, lajunya $\sqrt2$ kali (\qty{10}{m/s}).

\textbf{19.} $v \propto \sqrt m$: $7.2 \times \sqrt{100/80} = \qty{8.1}{m/s}$.

\textbf{20.} $h = v^2/2g = 52/19.6 = \qty{2.6}{m}$: sebuah lompatan dari
lantai satu.

\textbf{21.} $a = v^2/2d$: \qty{52}{m/s^2} ($5.3g$) sepanjang \qty{0.5}{m},
\omterm{def:b1:newton-dynamics:momentum}{gayanya} $m(g + a) \approx \qty{5}{kN}$; \qty{17}{m/s^2} ($1.8g$) sepanjang
\qty{1.5}{m}, \qty{2.2}{kN}. Bergulinglah.

\textbf{22.} \omterm{prop:b1:newton-dynamics:drag}{Gaya hambat} yang lebih besar (dan sedikit \omterm{def:b1:newton-dynamics:momentum}{gaya} angkat)
sesaat melampaui beratnya: penerjunnya melambat lalu mendarat di bawah $v_\infty'$.

\textbf{23.} $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-6}{kg}$, $S = \pi r^2 =
\qty{3.1e-6}{m^2}$: $v_\infty = \sqrt{2mg/\rho C_xS} = \sqrt{8.2\times10^{-5}/
1.9\times10^{-6}} = \qty{6.6}{m/s}$; $\mathrm{Re} = 1.2 \times 6.6 \times
2\times10^{-3}/1.8\times10^{-5} \approx 900$: kuadratik (kira-kira).

\textbf{24.} $v_\infty = \qty{1.2}{cm/s}$ (\cref{exo:b1:newton-dynamics:6});
$\mathrm{Re} \approx 0.02 \ll 1$: Stokes berlaku, secara taat asas.

\textbf{25.} $m\dot{\vect v} = m\vect g + \vect F_{\text{hambat}}$; \omterm{prop:b1:newton-dynamics:drag}{gaya
hambatnya} linear terhadap $v$ bagi benda kecil dan lambat, dan kuadratik
bagi benda besar dan cepat. Angkanya: \qty{40}{m/s} bagi benda yang
jatuh, \qty{7}{m/s} di bawah payung (atau bagi tetes hujan),
\qty{1}{cm/s} bagi kabut --- hukum yang sama membentang empat \omterm{def:b1:units-dimensions:oom}{orde besar}.
\end{solution}
