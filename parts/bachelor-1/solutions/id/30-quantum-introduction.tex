\begin{solution}{exo:b1:quantum-introduction:7}
$p = \sqrt{2 \times 9.11 \times 10^{-31} \times 5 \times 10^4 \times 1.6 \times 10^{-19}} =
\qty{1.2e-22}{kg.m/s}$, $\lambda = \qty{5.5}{pm}$ (kerelatifan mengoreksinya menjadi

\begin{solution}{exo:b1:quantum-introduction:8}
Atomnya: $\Delta p \geq \hbar/2\Delta x = \qty{5.3e-25}{kg.m/s}$, $E \sim \Delta p^2/2m_e =
\qty{1.5e-19}{J} \approx \qty{1}{eV}$: skalanya tepat. Intinya: $\Delta p \geq
\qty{1.1e-20}{kg.m/s}$; $\Delta p^2/2m_e$ akan bernilai \qty{400}{MeV}, jauh di atas
$m_ec^2$, sehingga pakailah $E \approx \Delta p\,c = \qty{3e-12}{J} = \qty{20}{MeV}$: tak ada gaya
di dalam intinya yang dapat menahan elektron setenaga itu --- inti tak memuat
elektron (elektron beta tercipta ketika dipancarkan). Butirannya: $\Delta v \geq
\hbar/2m\Delta x = 1.05 \times 10^{-34}/(2 \times 10^{-9} \times 10^{-6}) = \qty{5e-20}{m/s}$:
tak berarti.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:9}
(a) $m_ev^2/r = e^2/4\pi\varepsilon_0r^2$: $v = \sqrt{e^2/4\pi\varepsilon_0m_er}$, $E =
\tfrac12m_ev^2 - e^2/4\pi\varepsilon_0r = -e^2/8\pi\varepsilon_0r$. (b) $m_evr = n\hbar$ dan
$m_ev^2r = e^2/4\pi\varepsilon_0$: bagilah kuadrat yang pertama dengan yang kedua:
$m_er = n^2\hbar^2 \cdot 4\pi\varepsilon_0/e^2$, $r_n = n^2a_0$, dan $E_n = -e^2/8\pi\varepsilon_0n^2a_0
= E_1/n^2$ dengan $a_0 = \qty{52.9}{pm}$, $E_1 = \qty{-13.6}{eV}$. (c) $v_1 =
\hbar/m_ea_0 = \qty{2.19e6}{m/s}$, $v_1/c = 1/137$. (d) $2 \to 1$: $\tfrac34 \times
13.6 = \qty{10.2}{eV}$, \qty{122}{nm}; $3 \to 2$: $(\tfrac14 - \tfrac19) \times 13.6 =
\qty{1.89}{eV}$, \qty{656}{nm}. (e) $\lambda_n = h/m_ev_n = nh/m_ev_1 = 2\pi na_0$
(karena $m_ev_1a_0 = \hbar$), dan $2\pi r_n = 2\pi n^2a_0 = n\lambda_n$.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:12}
(a) $E(\Delta x) = \hbar^2/8m\Delta x^2 + \tfrac12m\omega^2\Delta x^2$; turunannya nol pada
$\Delta x^2 = \hbar/2m\omega$, yang di sana kedua sukunya sama dengan $\hbar\omega/4$: $E_{\min} =
\hbar\omega/2$. (b) $\tfrac12 \times 1.055 \times 10^{-34} \times 8.3 \times 10^{14} = \qty{4.4e-20}{J}
= \qty{0.27}{eV}$; $T \sim \hbar\omega/k_B = \qty{6300}{K}$ --- pada suhu ruang
getarannya beku dalam \omterm{prop:b1:quantum-introduction:box}{keadaan dasarnya}. (c) $\Delta p = \hbar/2\Delta x =
\qty{1.05e-24}{kg.m/s}$, $E = \Delta p^2/2m_{\mathrm{He}} = 1.1 \times 10^{-48}/1.33 \times
10^{-26} = \qty{8e-23}{J} = \qty{0.5}{meV}$: separuh energi ikatnya --- atomnya tak dapat
duduk cukup diam untuk mengkristal; helium tetap cair sampai nol
mutlak kecuali bila diperas (\qty{25}{bar}). (d) $E_1 = \pi^2\hbar^2/2mL^2$ terhadap
$\hbar^2/2mL^2$: sepuluh kali lebih besar, sebagaimana seharusnya --- batasnya adalah batas
bawah, dan keadaan kotaknya memiliki $\Delta x \approx 0.18L$, bukan $L/2$.
\end{solution}

\begin{solution}{pb:b1:quantum-introduction:1}
\textbf{1.} Cahaya membebaskan elektron dari katodenya dengan berbagai
energi; ditahan pada $-V_s$, anodenya menolaknya dan hanya yang berenergi
$E_k > eV_s$ yang tiba. Arusnya lenyap ketika $eV_s = E_{k,\max}$.

\textbf{2.} $eV_s = h\nu - W$: kemiringannya $h/e$, pintasannya $-W/e$; $V_s = 0$ pada
ambang $\nu_0 = W/h$.

\textbf{3.} Titik ujungnya: $(2.36 - 0.30)/(5.0 \times 10^{14}) = \qty{4.12e-15}{V.s}$,
$h = 1.602 \times 10^{-19} \times 4.12 \times 10^{-15} = \qty{6.60e-34}{J.s}$ (pencocokan pada
kelimanya memberi $6.62$): berselisih $0.5\%$ dari $6.626 \times 10^{-34}$.

\textbf{4.} $W = h\nu - eV_s = 2.47 - 0.30 = \qty{2.2}{eV}$; $\nu_0 = W/h =
\qty{5.3e14}{Hz}$, $\lambda_0 = 1240/2.2 = \qty{560}{nm}$. Sebuah penunjuk merah
(\qty{650}{nm}, \qty{1.9}{eV}) tak melontarkan apa pun.

\textbf{5.} $V_s$ tetap, arusnya berlipat dua: fotonnya dua kali lipat, masing-masing
berenergi sama. Sebuah gelombang akan memberi tiap elektron energi lebih besar dengan
intensitas lebih besar, dan sebuah ambang dalam intensitas alih-alih dalam frekuensi.

\textbf{6.} $10^{-6} \times 10^{-20} = \qty{1e-26}{W}$ pada atomnya; \qty{2.2}{eV} $=
\qty{3.5e-19}{J}$ memerlukan $3.5 \times 10^7$~s --- yaitu satu tahun. Pancarannya teramati
dalam hitungan nanosekon: energinya tiba dalam kuanta yang utuh.

\textbf{7.} $m_ev^2/r = e^2/4\pi\varepsilon_0r^2$: $v = \sqrt{e^2/4\pi\varepsilon_0m_er}$; $E =
\tfrac12m_ev^2 - e^2/4\pi\varepsilon_0r = -e^2/8\pi\varepsilon_0r$.

\textbf{8.} $(m_evr)^2 = n^2\hbar^2$ dan $m_ev^2r = e^2/4\pi\varepsilon_0$: $r_n = n^2 \cdot
4\pi\varepsilon_0\hbar^2/m_ee^2 = n^2a_0$, $a_0 = (1.055 \times 10^{-34})^2/(8.99 \times 10^9
\times 9.11 \times 10^{-31} \times 2.57 \times 10^{-38}) = \qty{52.9}{pm}$.

\textbf{9.} $E_n = -e^2/8\pi\varepsilon_0r_n = -m_ee^4/2(4\pi\varepsilon_0)^2\hbar^2n^2$; $E_1 =
-2.31 \times 10^{-28}/(2 \times 5.29 \times 10^{-11}) = \qty{-2.18e-18}{J} = \qty{-13.6}{eV}$.

\textbf{10.} $v_1 = \hbar/m_ea_0 = \qty{2.19e6}{m/s}$; $v_1/c = 7.3 \times 10^{-3} =
1/137$.

\textbf{11.} $2 \to 1$: \qty{10.2}{eV}, \qty{122}{nm} (ultraungu); $3 \to 2$:
\qty{1.89}{eV}, \qty{656}{nm} (merah); $4 \to 2$: \qty{2.55}{eV}, \qty{486}{nm}
(biru-hijau): garis Balmerlah yang tampak.

\textbf{12.} $hc/\lambda = |E_1|(1/n^2 - 1/p^2)$: $R_H = |E_1|/hc = 13.6/1240 =
\qty{1.097e-2}{nm^{-1}} = \qty{1.097e7}{m^{-1}}$. Batas Balmernya ($p \to \infty$):
$\lambda = 4/R_H = \qty{365}{nm}$.

\textbf{13.} \qty{13.6}{eV}; $\lambda = 1240/13.6 = \qty{91}{nm}$.

\textbf{14.} $v_n = v_1/n$, $\lambda_n = h/m_ev_n = nh/m_ev_1 = 2\pi na_0$ (karena $m_ev_1a_0
= \hbar$); $2\pi r_n = 2\pi n^2a_0 = n\lambda_n$: orbitnya mengemban $n$ gelombang
yang utuh --- aturan Bohr adalah aturan \omterm{thm:b1:wave-propagation:standing}{gelombang berdiri}.

\textbf{15.} Yang keliru: elektronnya tak memiliki orbit (ia akan memancar;
\omterm{prop:b1:quantum-introduction:box}{keadaan dasarnya} bermomentum sudut nol) dan modelnya gagal bagi atom mana pun
yang berelektron dua. Yang tepat: tingkat hidrogennya, jadi setiap \omterm{prop:b1:quantum-introduction:hydrogen}{garis
spektrumnya}. Orbitnya digantikan sebuah \omterm{def:b1:quantum-introduction:wavefunction}{fungsi gelombang}, yaitu awan peluang
berukuran $\sim n^2a_0$.

\textbf{16.} \omterm{thm:b1:wave-propagation:standing}{Gelombang berdirinya}: $L = n\lambda/2$, $p_n = h/\lambda_n = nh/2L$, $E_n =
p_n^2/2m = n^2h^2/8mL^2$. Rumus bolanya sama dengan $L \to R$:
\omterm{prop:b1:quantum-introduction:box}{keadaan dasarnya} memuat setengah panjang gelombang dalam jari-jarinya.

\textbf{17.} $E_e = 6.02 \times 10^{-38}/(0.13 \times 4 \times 10^{-18}) = \qty{0.72}{eV}$;
$E_h = 0.13/0.45 \times 0.72 = \qty{0.21}{eV}$.

\textbf{18.} $R = \qty{2}{nm}$: $1.74 + 0.93 = \qty{2.67}{eV}$, \qty{464}{nm}, biru;
\qty{3}{nm}: \qty{2.15}{eV}, \qty{577}{nm}, kuning; \qty{4}{nm}: \qty{1.97}{eV},
\qty{629}{nm}, merah.

\textbf{19.} Energi pengurungannya tumbuh sebanding $1/R^2$: kotak yang lebih kecil menaikkan
tingkatnya, fotonnya lebih bertenaga, cahayanya lebih biru.

\textbf{20.} $\Delta p \sim \hbar/R = \qty{5.3e-26}{kg.m/s}$, $E \sim \Delta p^2/2m_e^\ast =
\qty{1.2e-20}{J} = \qty{0.07}{eV}$: ordenya tepat (koefisien eksaknya
$\pi^2/2$ dan $\Delta p$ yang kasar itu menjelaskan faktor sepuluhnya).

\textbf{21.} $1240/577 = \qty{2.15}{eV} = \qty{3.4e-19}{J}$: $10^{-9}/3.4 \times 10^{-19}
= 2.9 \times 10^9$ foton tiap sekon.

\textbf{22.} $10^{-3}/(1.96 \times 1.6 \times 10^{-19}) = 3.2 \times 10^{15}$ tiap sekon.

\textbf{23.} $100 \times 2.48 \times 1.6 \times 10^{-19} = \qty{4e-17}{J}$ dalam \qty{0.1}{s}:
\qty{4e-16}{W}.

\textbf{24.} $h\nu = \qty{6.6e-26}{J}$: $1.5 \times 10^{13}$ foton tiap sekon ---
dan masing-masing $10^5$ kali lebih kecil daripada energi termal $k_BT$
antenanya: tak ada harapan, dan tak ada keperluan; gambaran gelombangnya eksak di sana.

\textbf{25.} Ukuran atom, $a_0 = \qty{53}{pm}$; skala energinya,
\qty{13.6}{eV} (jadi foton beberapa eV, kimia, spektrum tampak); dan
kelajuan elektronnya, $\alpha c \approx \qty{2e6}{m/s}$ --- ketiganya
dari $h$, $e$ dan $m_e$ (serta $\varepsilon_0$), tanpa satu pun disetel.
\end{solution}
