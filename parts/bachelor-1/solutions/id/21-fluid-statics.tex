\section*{Bab \ref{ch:b1:fluid-statics} --- Statika Fluida}

\begin{solution}{exo:b1:fluid-statics:1}
$P = P_0 + \rho gh$: \qty{2.0e5}{Pa} (\qty{2.0}{bar}), \qty{1.1e6}{Pa}
(\qty{11}{bar}), \qty{1.0e7}{Pa} (\qty{101}{bar}). Pada \qty{300}{m}: $P =
\qty{3.1e6}{Pa}$, gayanya \qty{3.1}{MN} pada \qty{1}{m^2} (total, berbanding
\qty{1}{atm} di dalamnya: \qty{3.0}{MN}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:2}
$h = P/\rho g$: raksa \qty{0.760}{m}; air \qty{10.3}{m}. Sebuah pompa paling
banter dapat menciptakan hampa di atas airnya: atmosfernya lalu mendorong
kolomnya naik sejauh \qty{10.3}{m}, tidak lebih.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:3}
$F_2 = F_1S_2/S_1 = \qty{10}{kN}$ (satu ton); $d_2 = d_1S_1/S_2 = \qty{2.0}{mm}$
(volume yang dipindahkannya sama, usahanya sama).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:4}
Tekanannya sama pada aras antarmuka minyak--air di kedua kakinya: $\rho_og
h_o = \rho_wgh_w$, $h_w = 0.85 \times 12 = \qty{10.2}{cm}$ air di atas
aras itu di kaki yang lain; permukaan minyaknya berdiri $12 - 10.2 = \qty{1.8}{cm}$
lebih tinggi.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:5}
$H = RT/Mg = 8.314 \times 273/(0.029 \times 9.81) = \qty{8.0}{km}$; $P/P_0 =
\eu^{-z/H}$: $0.54$ pada \qty{5}{km}, $0.33$ pada \qty{8848}{m}; separuhnya pada $H\ln2 =
\qty{5.5}{km}$ --- angka-angka yang dipakai di \cref{ch:b1:kinetic-theory}.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:6}
$F = \tfrac12\rho gh^2L = \qty{1.96e8}{N}$; \omterm{prop:b1:fluid-statics:wall}{pusat tekanannya} pada kedalaman
\qty{13.3}{m}, yakni \qty{6.7}{m} di atas dasarnya; momennya terhadap dasarnya $F \times
6.7 = \qty{1.3e9}{N.m}$. Melengkungkan bendungannya ke arah airnya mengubah dorongannya
menjadi tekanan pada lengkungnya, yang diteruskan ke dinding lembahnya.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:7}
$917/1025 = 89\%$; kayunya $60\%$. Rakitnya: \omterm{thm:b1:fluid-statics:archimedes}{gaya apung} pada pencelupan penuh $1000 \times
2.0 \times 9.81 = \qty{19.6}{kN}$, beratnya sendiri \qty{11.8}{kN}: tersisa \qty{7.8}{kN},
sepuluh orang\dots{} yang kesepuluh berkaki basah: sembilan agar tetap kering
(\qty{800}{kg}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:8}
Volume tercelupnya $m/\rho$: air \qty{20}{cm^3}; air garam $20/1.1 = \qty{18.2}{cm^3}$,
$\qty{1.8}{cm^3}$ lebih kecil, yakni $1.8/0.50 = \qty{3.6}{cm}$ batang tambahan: \qty{7.6}{cm}
muncul. Kepekaannya $\approx m/(\rho^2S) \approx \qty{40}{cm}$ tiap satuan massa jenis
nisbi (\qty{3.6}{cm} tiap $0.1$).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:9}
Volume yang didesaknya $2800/1.025 = \qty{2732}{m^3}$: \qty{268}{m^3} ($9\%$)
muncul. Netral: massanya $1025 \times 3000 = \qty{3075}{t}$: masukkan \qty{275}{t}.
Air tawar: \omterm{thm:b1:fluid-statics:archimedes}{gaya apungnya} turun ke \qty{3000}{t}: \qty{75}{t} terlalu berat --- ia tenggelam
kecuali \qty{75}{t} dipompa keluar.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:10}
Udara yang didesaknya: $10^{-2} \times 1.20 = \qty{12.0}{g}$; helium di dalamnya $12.0 \times
4/29 = \qty{1.7}{g}$; karetnya \qty{3.0}{g}: daya angkat totalnya \qty{7.3}{g} --- tujuh meter
tali yang ringan.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:11}
Momennya $\int_0^h\rho gx\cdot xL\,\dd x = \rho gLh^3/3$, gayanya $\rho gLh^2/2$:
lengannya $2h/3$. Dari $h_1$ sampai $h_2$: momennya $\rho gL(h_2^3 - h_1^3)/3$, gayanya
$\rho gL(h_2^2 - h_1^2)/2$: kedalamannya $\frac23(h_2^3 - h_1^3)/(h_2^2 - h_1^2)$.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:12}
Pencelupan tambahan $x$ menambahkan \omterm{thm:b1:fluid-statics:archimedes}{gaya apung} $\rho gSx$ ke atas: $m\ddot x = -\rho gSx$,
$\omega_0^2 = \rho gS/m$, $T = 2\pi\sqrt{m/\rho gS} = 2\pi\sqrt{0.020/(1000 \times
9.81 \times 5\times10^{-5})} = \qty{1.3}{s}$.
\end{solution}

\begin{solution}{pb:b1:fluid-statics:1}
\textbf{1.} $P = 1.013\times10^5 + 1025 \times 9.81 \times 300 = \qty{3.12e6}{Pa}
= \qty{31}{bar}$.

\textbf{2.} $S = \pi(0.40)^2 = \qty{0.503}{m^2}$: gaya totalnya $(P - P_0)S =
\qty{1.52e6}{N}$ --- 155 ton, seperti truk bermuatan di atas sebuah palka.

\textbf{3.} Lunasnya \qty{10}{m} lebih dalam: $+\rho g \times 10 = \qty{1.0e5}{Pa}$ lebih besar,
$3\%$: lambungnya terbebani hampir seragam.

\textbf{4.} $\Delta\rho/\rho = \chi_T\Delta P = 4.6\times10^{-10} \times 3.0\times10^6
= 0.14\%$: dapat diabaikan bagi tekanannya (beberapa kilopascal dari tiga
juta).

\textbf{5.} Tiga puluh bar di luar, satu di dalam: lambungnya adalah bejana
tekan yang terbebani ke dalam; tabung (atau bola) mengubah tekanannya menjadi
tekanan menyusuri dindingnya, yang ditahan baja dengan baik, sedangkan pelat
datar akan melengkung.

\textbf{6.} $\Delta P\,R/e = 3.0\times10^6 \times 5.0/0.030 = \qty{5.0e8}{Pa}
= \qty{500}{MPa}$: dekat dengan tegangan luluhnya --- \qty{300}{m} sudah dekat
batas bagi lambung ini, dan kapal yang menyelam lebih dalam menuntut baja yang lebih tebal atau lebih kuat
(atau titanium).

\textbf{7.} Volume yang didesaknya $m_0/\rho = 2800/1.025 = \qty{2732}{m^3}$: \qty{268}{m^3}
muncul ($9\%$).

\textbf{8.} Netral: $m = \rho V = \qty{3075}{t}$: masukkan \qty{275}{t}.

\textbf{9.} \omterm{thm:b1:fluid-statics:archimedes}{Gaya apungnya} turun sebesar $0.1\%$ dari $\rho Vg$: $0.001 \times 3075 \times 9.81
\times10^3 = \qty{30}{kN}$ ke bawah (tiga ton). Makin dalam $\Rightarrow$ makin
termampatkan $\Rightarrow$ makin berat: tak mantap --- sebuah kapal selam harus terus
menyetel pemberatnya (dan lambung baja termampatkan lebih sedikit daripada air, dan itu membantu;
lambung yang terlalu lentur akan tenggelam tanpa kembali).

\textbf{10.} \omterm{thm:b1:fluid-statics:archimedes}{Gaya apungnya} $1000 \times 3000 \times 9.81 = \qty{29.4}{MN}$ berbanding
beratnya $3075 \times 9.81 \times 10^3 = \qty{30.2}{MN}$: \qty{0.74}{MN} ke bawah
(\qty{75}{t}); pompalah keluar \qty{75}{t}.

\textbf{11.} Seketika \qty{20}{t} lebih ringan: masukkan \qty{20}{t} air
(\qty{19.5}{m^3}) ke dalam tangki pengimbangnya.

\textbf{12.} Air hanya dapat disemburkan keluar oleh sesuatu yang bertekanan lebih tinggi
daripada laut di luarnya; udara mampat dalam tabung mendorongnya keluar --- hanya
selama tekanan tabungnya melampaui tekanan sekitarnya pada kedalaman
itu.

\textbf{13.} $PV$ tetap: \qty{275}{m^3} pada \qty{31}{bar} menuntut $275 \times 31
= \qty{8500}{m^3}$ udara pada \qty{1}{atm} --- disimpan pada \qty{200}{bar} dalam
tabung \qty{43}{m^3}.

\textbf{14.} \qty{2.0}{bar} pada \qty{10}{m}; \qty{4.0}{bar} pada \qty{30}{m}.

\textbf{15.} $PV$ tetap: $6.0/4.0 = \qty{1.5}{L}$ --- rongga dadanya
terhimpit ke dalam; penyelam bebas merasakannya.

\textbf{16.} Udara yang dihirup pada \qty{4}{bar} memuai empat kali lipat: \qty{24}{L} di paru-paru
yang hanya memuat \qty{6}{L}: pecah. Jangan pernah menahan napas ketika naik; embuskan
terus-menerus.

\textbf{17.} $\Delta P = \rho g \times 1.0 = \qty{1.0e4}{Pa}$; gayanya $\qty{1.0e4}{Pa}
\times 0.05 = \qty{500}{N}$ pada dadanya: otot pernapasannya tak dapat
mengangkat lima puluh kilogram --- snorkel hanya berhasil pada beberapa desimeter pertama.

\textbf{18.} $12 \times 200 = \qty{2400}{L}$ pada \qty{1}{bar}, yakni \qty{600}{L}
pada \qty{4}{bar}: \qty{30}{min}.

\textbf{19.} $V = 10^8/1025 = \qty{97600}{m^3}$.

\textbf{20.} Di air tawar ia harus mendesak $10^5\ \unit{m^3}$: \qty{2400}{m^3}
lebih banyak, yakni $2400/8000 = \qty{0.30}{m}$ lebih dalam.

\textbf{21.} $10^4/1.025 = \qty{9756}{m^3}$ lebih banyak: \qty{1.2}{m} lebih dalam.

\textbf{22.} Dalam keadaan kosong, $G$ kapalnya duduk tinggi dan sedikit lambungnya yang tercelup:
metasentrumnya dapat jatuh di bawah $G$ dan kapalnya terbalik dalam ombak. Air
laut yang rendah di dalam lambungnya menurunkan $G$ dan memperdalam saratnya.

\textbf{23.} $\rho gh = \qty{2.0e5}{Pa}$, \qty{0.2}{MN/m^2}: lima belas kali lebih kecil
daripada \qty{3}{MN/m^2} kapal selamnya.

\textbf{24.} Tidak: tekanan pada kedalaman kapal selamnya adalah $P_0 + \rho gh$,
yang ditetapkan oleh kedalamannya saja. Berat kapal tankernya dipikul air yang
didesaknya, yang menaikkan aras lautnya di mana-mana sebesar jumlah yang tak
terukur; tak ada beban tambahan yang sampai ke kapal selamnya.

\textbf{25.} $\dd P/\dd z = -\rho g$ --- dari sinilah $P = P_0 + \rho gh$ bagi setiap
kedalaman, gaya dan volume paru-paru di sini --- dan teorema Archimedes, yaitu
hubungan itu yang diintegralkan atas sebuah permukaan tertutup.
\end{solution}
