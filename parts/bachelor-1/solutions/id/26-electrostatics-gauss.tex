\section*{Bab \ref{ch:b1:electrostatics-gauss} --- Elektrostatika: Medan dan Hukum Gauss}

\begin{solution}{exo:b1:electrostatics-gauss:1}
$F_e = 8.99 \times 10^9 \times (1.6 \times 10^{-19})^2/(2 \times 10^{-15})^2 = \qty{58}{N}$;
$F_g = 6.67 \times 10^{-11} \times (1.67 \times 10^{-27})^2/(4 \times 10^{-30}) =
\qty{4.6e-35}{N}$; nisbahnya $1.2 \times 10^{36}$. Inti atom bertahan karena

\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:2}
$E = 8.99 \times 10^9 \times 10^{-6}/1 = \qty{9.0}{kV/m}$. Medan yang mengarah ke bawah adalah
medan sebuah muatan negatif: $Q = -4\pi\varepsilon_0R^2E = -1.11 \times 10^{-10} \times
(6.37 \times 10^6)^2 \times 100 = \qty{-4.5e5}{C}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:3}
Pada sumbunya, $|x| > a$: $E_x = \dfrac{q}{4\pi\varepsilon_0}\Bigl[\dfrac{1}{(x - a)^2} +
\dfrac{1}{(x + a)^2}\Bigr]$, mengarah keluar. Pada sumbu $y$ komponen $x$-nya
saling meniadakan: $E_y = \dfrac{2qy}{4\pi\varepsilon_0(y^2 + a^2)^{3/2}}$; $\dd E_y/\dd y \propto
(y^2 + a^2) - 3y^2 = 0$ pada $y = a/\sqrt2$. Di titik asalnya medannya
lenyap; garis yang meninggalkan tiap muatan menuju yang lain membelok menjauh menyusuri
$\pm y$ --- sebuah pelana.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:4}
Di antara kedua muatannya kedua medannya menunjuk ke arah $-q$: tak ada nol. Di kiri
$+2q$, selalu berlaku $2q/x^2 > q/(x + a)^2$. Di luar $-q$, pada jarak $d$ darinya:
$2q/(a + d)^2 = q/d^2$, $a + d = \sqrt2\,d$, $d = a/(\sqrt2 - 1) = 2.4a$.
Dua garis meninggalkan $+2q$ bagi tiap satu garis yang berakhir pada $-q$; separuh lainnya pergi ke
tak hingga, tempat pasangannya tampak seperti muatan $+q$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:5}
Cincin berjari-jari $a$, berlebar $\dd a$: $\dd q = 2\pi\sigma a\,\dd a$.
Maka $\dd E_z =
\dfrac{\sigma z}{2\varepsilon_0}\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}}$; integralkan:
$\dfrac{\sigma z}{2\varepsilon_0}\Bigl[-\dfrac{1}{\sqrt{a^2 + z^2}}\Bigr]_0^R = \dfrac{\sigma}{2\varepsilon_0}
\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$. Untuk $z \ll R$: $\sigma/2\varepsilon_0$, yaitu
bidang tak hingganya. Untuk $z \gg R$: $z/\sqrt{z^2 + R^2} \approx 1 - R^2/2z^2$, sehingga $E_z
\approx \sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, yaitu muatan titiknya.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:6}
$Q = 79e = \qty{1.26e-17}{C}$; $E(R) = 8.99 \times 10^9 \times 1.26 \times 10^{-17}/
(7 \times 10^{-15})^2 = \qty{2.3e21}{V/m}$; pada $R/2$, separuhnya: \qty{1.2e21}{V/m}; pada
\qty{1}{pm}: $1.14 \times 10^{-7}/10^{-24} = \qty{1.1e17}{V/m}$ --- $10^{15}$ kali
medan dadal udara, pada jarak tempat elektronnya hidup.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:7}
Tabung berjari-jari $r$, sepanjang $h$: $2\pi rh\,E = \lambda h/\varepsilon_0$, $E =
\lambda/2\pi\varepsilon_0r$. Korona terjadi ketika $\lambda = 2\pi\varepsilon_0 rE = 2\pi \times 8.85 \times
10^{-12} \times 0.01 \times 3 \times 10^6 = \qty{1.7}{\micro C/m}$; pada \qty{5}{m},
$E = 3 \times 10^6 \times 0.01/5 = \qty{6}{kV/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:8}
$E = \sigma/\varepsilon_0 = 10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$. Sebuah keping
tak mengerahkan gaya total pada dirinya sendiri (sumbangan medannya sendiri saling meniadakan
berpasangan); medan keping yang lain di letaknya adalah $\sigma/2\varepsilon_0$, sehingga
$F = Q\sigma/2\varepsilon_0 = 10^{-6} \times 10^{-6}/(2 \times 8.85 \times 10^{-12}) =
\qty{5.6e-2}{N}$, tarik-menarik; tekanannya $\sigma^2/2\varepsilon_0 = \qty{0.056}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:9}
Untuk $r < a$: muatan penghantar dalamnya duduk di permukaannya, tak ada muatan
di dalamnya, $E = 0$. Untuk $a < r < b$: $E = \lambda/2\pi\varepsilon_0r$. Untuk $r > b$: muatan
yang dilingkupinya $\lambda - \lambda = 0$, $E = 0$ --- kabelnya tak memancarkan medan
statik. Pada $r = a$: $2 \times 8.99 \times 10^9 \times 10^{-8}/(5 \times 10^{-4}) =
\qty{3.6e5}{V/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:10}
$\rho = 3Q/4\pi(R_2^3 - R_1^3)$. Untuk $r < R_1$: $E = 0$. Untuk $R_1 < r < R_2$: $4\pi r^2E =
\rho\,\tfrac43\pi(r^3 - R_1^3)/\varepsilon_0$, sehingga $E = \dfrac{Q}{4\pi\varepsilon_0r^2}\,
\dfrac{r^3 - R_1^3}{R_2^3 - R_1^3}$. Untuk $r > R_2$: $Q/4\pi\varepsilon_0r^2$. Di $R_1$ kedua
sisinya memberikan $0$; di $R_2$ keduanya memberikan $Q/4\pi\varepsilon_0R_2^2$. Di dalam kulitnya $E
\propto r - R_1^3/r^2$ menaik, sehingga maksimumnya di $R_2$, dan sesudah
itu ia jatuh sebagai $1/r^2$. Di dalam rongganya medannya nol: sebuah muatan
di sana tak merasakan gaya --- di mana pun ia duduk.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:11}
(a) Intinya $\tfrac43\pi(3.48 \times 10^6)^3 \times 11000 = \qty{1.94e24}{kg}$; mantelnya
$\tfrac43\pi[(6.37 \times 10^6)^3 - (3.48 \times 10^6)^3] \times 4500 = \qty{4.08e24}{kg}$;
totalnya \qty{6.0e24}{kg}, meleset $1\%$. (b) Permukaannya $GM/R^2 = \qty{9.9}{m/s^2}$;
batasnya $6.67 \times 10^{-11} \times 1.94 \times 10^{24}/(3.48 \times 10^6)^2 =
\qty{10.7}{m/s^2}$. (c) $g = GM(r)/r^2$ dengan $\dd M = 4\pi r^2\rho_m\,\dd r$:
$\dd g/\dd r = 4\pi G\rho_m - 2GM(r)/r^3 = 4\pi G[\rho_m - \tfrac23\bar\rho(r)]$,
dengan $\bar\rho(r) = M(r)/\tfrac43\pi r^3$ adalah massa jenis rata-rata di bawah $r$.
Nilai $g$ naik terhadap kedalaman ($\dd g/\dd r < 0$) jika dan hanya jika $\rho_m < \tfrac23\bar\rho$. Tepat
di bawah permukaannya $\bar\rho = 5500$, $\tfrac23\bar\rho = 3700 < 4500$: $g$
mula-mula turun sedikit; di batas intinya $\bar\rho = 11000$, $7300 >
4500$: di sana ia naik, sampai $10.7$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:12}
(a) Di dalamnya, $E = er/4\pi\varepsilon_0R^3$ mengarah keluar; pada elektronnya $\vect F =
-\dfrac{e^2}{4\pi\varepsilon_0R^3}\,\vect r$: sebuah pegas. (b) $\omega = \sqrt{e^2/4\pi\varepsilon_0mR^3}
= \sqrt{2.31 \times 10^{-28}/(9.11 \times 10^{-31} \times 10^{-30})} = \qty{1.6e16}{rad/s}$,
$f = \qty{2.5e15}{Hz}$, $\lambda = c/f = \qty{120}{nm}$. (c) Mengejutkan betapa dekatnya
dengan garis \qty{122}{nm}: ukuran dan kekuatan atomnya tepat.
Tetapi modelnya memberikan satu frekuensi, bukan sederet, dan intinya
bukan bola \qty{0.1}{nm} melainkan titik $10^{-15}$~m --- dan elektron
klasik yang mengorbit akan memancarkan energinya habis dalam beberapa nanosekon. Pembetulannya
ada di \cref{ch:b1:quantum-introduction}.
\end{solution}

\begin{solution}{pb:b1:electrostatics-gauss:1}
\textbf{1.} $\vect F = -e\vect E$: ke atas. Medan yang mengarah ke bawah di permukaannya
adalah medan sebuah muatan negatif: Bumi bermuatan negatif.

\textbf{2.} Bola berjari-jari $R_E$, dengan $E_n = -100$: $Q = 4\pi\varepsilon_0R_E^2E_n =
1.11 \times 10^{-10} \times 4.06 \times 10^{13} \times (-100) = \qty{-4.5e5}{C}$.

\textbf{3.} $\sigma = \varepsilon_0E_n = \qty{-8.9e-10}{C/m^2}$; $8.9 \times 10^{-10}/1.6 \times
10^{-19} = 5.5 \times 10^9$ elektron tiap meter persegi, $5.5 \times 10^5$ tiap
sentimeter persegi.

\textbf{4.} $F = 100e \times 100 = \qty{1.6e-15}{N}$; beratnya $10^{-9} \times 9.8 =
\qty{9.8e-9}{N}$: enam juta kali lebih besar.

\textbf{5.} Normal keluarnya: atas $E_n = -5$, bawah $E_n = +100$, sisinya
nol: $\Phi = \qty{95}{V.m}$, sehingga $Q = 95\varepsilon_0 = \qty{+8.4e-10}{C}$ tiap meter
persegi --- muatan ruang yang positif, yang hampir meniadakan
$-8.9 \times 10^{-10}$ milik tanahnya.

\textbf{6.} $\rho = 8.4 \times 10^{-10}/10^4 = \qty{8.4e-14}{C/m^3}$: $5 \times 10^5$
\omterm{def:b1:electrostatics-gauss:charge}{muatan elementer} tiap meter kubik; berbanding $10^9$ ion tiap tanda,
ketakseimbangannya $5 \times 10^{-4}$.

\textbf{7.} $J = \gamma E_0 = \qty{2e-12}{A/m^2}$; $I = J \times 5.1 \times 10^{14} =
\qty{1.0}{kA}$; $Q/I = 4.5 \times 10^5/1000 = \qty{450}{s}$. Bumi akan menjadi
netral dalam delapan menit: ada yang mengisinya kembali terus-menerus.

\textbf{8.} Sumbunya terletak di dalam setiap bidang yang memuatnya, dan semuanya \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang
simetri}: $\vect E$ bersifat aksial. Tiap $\dd q$ memberikan $\dd q/4\pi\varepsilon_0(a^2 + z^2)$
menyusuri arahnya sendiri, yang bagian aksialnya adalah pecahan
$z/\sqrt{a^2 + z^2}$: $E_z = qz/4\pi\varepsilon_0(a^2 + z^2)^{3/2}$.

\textbf{9.} $\dd q = 2\pi\sigma a\,\dd a$: $E_z = \dfrac{\sigma z}{2\varepsilon_0}\displaystyle\int_0^R
\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}} = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2
+ R^2}}\Bigr)$.

\textbf{10.} Untuk $z \ll R$: $\sigma/2\varepsilon_0$, yaitu bidang tak hingganya (nilai Gauss);
untuk $z \gg R$: $\sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, yaitu muatan titiknya.

\textbf{11.} $\sigma = -40/(\pi \times 9 \times 10^6) = \qty{-1.4e-6}{C/m^2}$; $\sigma/2\varepsilon_0 =
\qty{8.0e4}{V/m}$; pada $z = \qty{3}{km}$, $1 - 3/\sqrt{18} = 0.29$: $E =
\qty{2.3e4}{V/m}$, yang mengarah \emph{ke atas} menuju muatan negatifnya.

\textbf{12.} Cakram atasnya, $z = \qty{9}{km}$: $8.0 \times 10^4 \times (1 - 9/\sqrt{90}) =
\qty{4.1e3}{V/m}$ ke bawah. Totalnya $\qty{1.9e4}{V/m}$ ke atas, kira-kira $4 \times 10^4$
dengan muatan imbas tanahnya: seorde dengan \qty{10}{kV/m}
yang terukur (muatan yang sebenarnya lebih tersebar dan sebagian terperisai);
terbalik dan seratus kali medan cuaca cerahnya.

\textbf{13.} Seluruh awannya: muatan totalnya nol, fluksnya nol. Cakram bawahnya saja:
$-Q/\varepsilon_0 = -40/8.85 \times 10^{-12} = \qty{-4.5e12}{V.m}$.

\textbf{14.} Tiap cakramnya berjarak \qty{3}{km} dan memberikan $2.3 \times 10^4$; yang
atas ($+$) mendorong ke bawah, yang bawah ($-$) menarik ke bawah: \qty{4.7e4}{V/m}
ke bawah --- tiga puluh kali di bawah dadalnya.

\textbf{15.} $Q/4\pi\varepsilon_0R^2 = 8.99 \times 10^9 \times 40/(2.5 \times 10^5) =
\qty{1.4e6}{V/m}$: hanya muatan yang terjejal dalam setengah kilometer yang mencapai
dadalnya, dan medan di dalam awan yang terukur sepuluh kali terlalu lemah. Penguatan
setempat pada tetesan dan es, atau elektron lari yang disemai sinar
kosmik, adalah calonnya --- bagaimana sebuah kilat bermula masih diperdebatkan.

\textbf{16.} $I = 5/(5 \times 10^{-5}) = \qty{1e5}{A}$.

\textbf{17.} $W = QEd = 5 \times 10^5 \times 3000 = \qty{1.5e9}{J}$ (\qty{420}{kW h});
$P = 1.5 \times 10^9/5 \times 10^{-5} = \qty{3e13}{W}$ --- satu setengah kali
daya seluruh umat manusia, selama lima puluh mikrosekon.

\textbf{18.} Volumenya $\pi(0.02)^2 \times 3000 = \qty{3.8}{m^3}$, massanya \qty{4.5}{kg};
$\Delta T = 1.5 \times 10^9/(4.5 \times 720) = \qty{4.6e5}{K}$ seandainya tak ada yang lolos.
Salurannya meledak ke luar (kejutannya adalah guruhnya), memancarkan cahaya
dan gelombang radio, mengionkan dan mengurai udaranya: memanaskan saluran
akhirnya sampai $3 \times 10^4$~K hanyalah bagian kecil tagihannya.

\textbf{19.} \qty{2.9}{s} tiap kilometer. Bunyinya meninggalkan tiap titik
saluran \qty{3}{km} pada saat yang sama, dari jarak yang berbeda bermil-mil: ia
tiba terbentang selama beberapa sekon --- itulah gemuruhnya.

\textbf{20.} $50 \times 5 = \qty{250}{A}$ yang dibawa ke tanah; bersama
arus badai yang lebih tenang (pelucutan titik di bawah awan, hujan bermuatan)
totalnya mencapai kiloampere Bagian~I: badai adalah pembangkit sebuah
rangkaian global yang jalan baliknya adalah udara cuaca cerah.

\textbf{21.} $\Phi_{\text{keluar}}(\vect g) = -4\pi GM_{\text{dalam}}$: bagi sebuah massa titik
$g = -GM/r^2$ pada bola seluas $4\pi r^2$, fluksnya $-4\pi GM$, tak bergantung
pada $r$; sisanya menyusul seperti pada Coulomb.

\textbf{22.} $g(r) = g_0r/R_E$; pada $R_E/2$: \qty{4.9}{m/s^2}.

\textbf{23.} $M_{\text{inti}} = \qty{1.94e24}{kg}$, $g = GM_{\text{inti}}/R_c^2 =
\qty{10.7}{m/s^2}$ --- lebih besar daripada di permukaannya: mantel di atasnya lebih
ringan daripada rata-rata yang ada di bawahnya.

\textbf{24.} $m\ddot x = -mg_0x/R_E$: $\omega = \sqrt{g_0/R_E} = \qty{1.24e-3}{rad/s}$,
$T = 2\pi/\omega = \qty{5060}{s} = \qty{84}{min}$, $v_{\max} = \omega R_E =
\qty{7.9}{km/s}$ --- yaitu \omterm{thm:b1:central-forces:circular}{laju orbit} sebuah satelit yang menyerempet, yang
periodenya sama.

\textbf{25.} Ia menggantikan penjumlahan atas tiap muatan Bumi (I),
atas ion udaranya (I), dan atas tiap kulit batuannya (IV) --- dan
di~II ia menetapkan batas penjumlahan cakramnya. Ingatlah: \qty{-5e5}{C} dan
\qty{1}{kA} bagi Bumi cuaca cerah; \qty{40}{C} dan \qty{e4}{V/m} di bawah
badai; \qty{e5}{A} dan \qty{e9}{J} tiap sambaran; \qty{10.7}{m/s^2} di batas
intinya.
\end{solution}
