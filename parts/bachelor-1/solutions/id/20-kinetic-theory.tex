\section*{Bab \ref{ch:b1:kinetic-theory} --- Teori Kinetik dan Gas Ideal}

\begin{solution}{exo:b1:kinetic-theory:1}
$N = PV/k_BT = 1.013\times10^5 \times 10^{-3}/(1.38\times10^{-23} \times 273) =
\num{2.7e22}$; $n^{*-1/3} = (3.7\times10^{-26})^{1/3} = \qty{3.3}{nm}$, sepuluh
kali ukuran molekulnya.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:2}
$u = \sqrt{3RT/M}$: H$_2$ \qty{1930}{m/s}, N$_2$ \qty{517}{m/s}, O$_2$
\qty{484}{m/s}, CO$_2$ \qty{412}{m/s}; N$_2$ pada \qty{77}{K}: \qty{262}{m/s}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:3}
$P = \qty{3.5}{bar}$ mutlak: $n = PV/RT = 3.5\times10^5 \times 0.030/(8.314
\times 293) = \qty{4.3}{mol}$, $\qty{125}{g}$. Pada \qty{323}{K}, $P \propto T$:
\qty{3.86}{\omterm{def:b1:kinetic-theory:pressure}{bar}} (tolok \qty{2.85}{\omterm{def:b1:kinetic-theory:pressure}{bar}}).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:4}
$n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$; $\lambda = 1/(\sqrt2\pi \times 1.37\times
10^{-19} \times 2.4\times10^{25}) = \qty{6.8e-8}{m}$; frekuensinya $u/\lambda \approx
500/7\times10^{-8} = \qty{7e9}{s^{-1}}$. $\lambda \propto 1/P$: \qty{1}{m} pada
$P \approx 10^5 \times 7\times10^{-8} = \qty{7e-3}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:5}
$\rho = PM/RT$: \qty{1.20}{kg/m^3} pada \qty{293}{K}, \qty{0.946}{kg/m^3} pada
\qty{373}{K}. Daya angkatnya $(1.20 - 0.95) \times 2800 \times 9.81 = \qty{7.0}{kN}$:
kira-kira \qty{720}{kg}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:6}
$P_{\mathrm{O}_2} = 0.21P$: \qty{0.21}{atm}; \qty{0.11}{atm} pada \qty{5000}{m};
\qty{0.07}{atm} di Everest --- sepertiga nilainya di permukaan laut. Oksigen masuk ke
dalam darah sebanding dengan \omterm{def:b1:kinetic-theory:pressure}{tekanan} parsialnya; pada sepertiganya, paru-paru
tak dapat menjenuhkan hemoglobinnya.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:7}
\omterm{def:b1:kinetic-theory:model}{Gas ideal}: $P = RT/V = 2494/10^{-3} = \qty{24.9}{bar}$. Van der Waals:
$RT/(V - b) - a/V^2 = 2494/9.57\times10^{-4} - 0.364/10^{-6} = 26.1 - 3.6 =
\qty{22.4}{bar}$: tarikannya ($-3.6$ \omterm{def:b1:kinetic-theory:pressure}{bar}) mengalahkan volume
tersingkirnya ($+1.2$ \omterm{def:b1:kinetic-theory:pressure}{bar}); CO$_2$ yang sebenarnya $10\%$ di bawah nilai idealnya.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:8}
$n = PV/RT = 10^5 \times 50/(8.314 \times 300) = \qty{2000}{mol}$; $U = \tfrac52
nRT = \qty{1.25e7}{J}$; $\Delta U = \tfrac52 nR \times 5 = \qty{208}{kJ}$;
\qty{104}{s} dengan \qty{2}{kW}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:9}
$\Delta L = \alpha L\Delta T = 1.2\times10^{-5} \times 30 \times 50 = \qty{18}{mm}$.
Cincin: $\Delta d/d = \alpha\Delta T = 0.05/49.95 = 10^{-3}$: $\Delta T = 10^{-3}/
1.9\times10^{-5} = \qty{53}{K}$ (lubangnya memuai seperti logam di sekelilingnya).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:10}
$\tfrac32 k_BT = \qty{6.2e-21}{J} = \qty{0.039}{eV}$. Argon $\tfrac32 R =
\qty{12.5}{J/(mol.K)}$; nitrogen $\tfrac52 R = \qty{20.8}{J/(mol.K)}$:
molekul diatomiknya juga menyimpan \omterm{def:b1:kinetic-theory:U}{energi dalam} rotasi (dua derajat
kebebasan tambahan), berapa pun massanya.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:11}
Model enam arah: seperti dalam teksnya, $P = \tfrac13 n^*mu^2$. Keisotropikan:
$\langle v_x^2\rangle = \langle v_y^2\rangle = \langle v_z^2\rangle$ dan jumlah
ketiganya $\langle v^2\rangle$, sehingga masing-masing $\tfrac13\langle v^2\rangle$. Alasan
fluksnya dengan sebuah distribusi memberikan $P = n^*m\langle v_x^2\rangle$ (molekul
dengan $v_x > 0$ yang menghantam dindingnya, masing-masing bermomentum $2mv_x$, berfluks $n^*v_x/2$ tiap
kelas kelajuan, lalu dirata-ratakan), yakni $\tfrac13 n^*m\langle v^2\rangle$ --- sama saja.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:12}
H$_2$: \qty{1.9}{km/s}, O$_2$: \qty{0.48}{km/s}; Bumi \qty{11.2}{km/s}, Bulan
\qty{2.4}{km/s}. Bagi O$_2$ di Bumi, $v_e/u = 23$: ekor
distribusi di luar itu sama sekali dapat diabaikan; bagi H$_2$, $v_e/u \approx 6$:
pecahan yang amat kecil tetapi tetap lolos tiap tahun, dan sepanjang miliaran
tahun hidrogennya habis. Di Bulan, $v_e/u$ bernilai $5$ bagi O$_2$ dan
$1.2$ bagi H$_2$: semuanya bocor pergi.
\end{solution}

\begin{solution}{pb:b1:kinetic-theory:1}
\textbf{1.} $V = \qty{200}{m^3}$: $n = PV/RT = 1.013\times10^5 \times 200/(8.314
\times 293) = \qty{8320}{mol}$; $N = nN_A = \num{5.0e27}$.

\textbf{2.} $m = nM = \qty{241}{kg}$: tiga orang.

\textbf{3.} $n^* = N/V = \qty{2.5e25}{m^{-3}}$; jaraknya $n^{*-1/3} = \qty{3.4}{nm}$,
sembilan garis tengah molekul.

\textbf{4.} N$_2$: $\sqrt{3 \times 8.314 \times 293/0.028} = \qty{511}{m/s}$;
O$_2$: \qty{478}{m/s}. Dengan $\tfrac32 k_BT$ yang sama tiap molekul: $\tfrac12 mu^2$ tetap,
sehingga $u \propto 1/\sqrt m$.

\textbf{5.} $\tfrac32 k_BT = \qty{6.1e-21}{J}$; totalnya $N \times 6.1\times10^{-21}
= \qty{3.0e7}{J}$; sebuah mobil \qty{1000}{kg} pada \qty{100}{km/h}: \qty{3.9e5}{J} ---
gerak molekul di ruangan itu setara delapan puluh mobil.

\textbf{6.} $\lambda = 1/(\sqrt2\pi \times 1.37\times10^{-19} \times 2.5\times10^{25})
= \qty{66}{nm}$; waktunya $\lambda/u = 66\times10^{-9}/500 = \qty{1.3e-10}{s}$.

\textbf{7.} Hantaman tiap sekon pada $S$: $\tfrac16 n^*Su = \tfrac16 \times 2.5\times
10^{25} \times 10^{-4} \times 500 = \qty{2.1e23}{s^{-1}}$.

\textbf{8.} Masing-masing memberi $2mu = 2 \times 4.8\times10^{-26} \times 500 = \qty{4.8e-23}{kg.m/s}$
($m = M/N_A = \qty{4.8e-26}{kg}$): gayanya $2.1\times10^{23} \times 4.8\times10^{-23}
= \qty{10}{N}$ pada \qty{1}{cm^2}: \qty{1.0e5}{Pa}. Modelnya menghasilkan kembali
\omterm{def:b1:kinetic-theory:pressure}{tekanan} atmosfer.

\textbf{9.} $F = PS = 1.013\times10^5 \times 20 = \qty{2.0e6}{N}$ --- dua ratus
ton; \omterm{def:b1:kinetic-theory:pressure}{tekanan} yang sama bekerja di sisi lainnya.

\textbf{10.} $P' = P \times 303/293 = \qty{1.048e5}{Pa}$; selisihnya $\Delta P\,S = 3.5
\times10^3 \times 20 = \qty{70}{kN}$ ke luar: tujuh ton pada dindingnya.

\textbf{11.} Udaranya mengalir keluar sampai \omterm{def:b1:kinetic-theory:pressure}{tekanannya} menyamai: $n \propto 1/T$ pada
$P$, $V$ tetap: $8320 \times (1 - 293/303) = \qty{275}{mol}$, \qty{8}{kg}
pergi.

\textbf{12.} $\rho = PM/RT$: \qty{1.205}{kg/m^3} pada \qty{293}{K}; \qty{0.946}{kg/m^3}
pada \qty{373}{K}.

\textbf{13.} Udara panasnya $0.946 \times 2800 \times 9.81 = \qty{26.0}{kN}$; udara
dingin yang didesaknya $1.205 \times 2800 \times 9.81 = \qty{33.1}{kN}$; daya angkatnya \qty{7.1}{kN}
(\qty{725}{kg}).

\textbf{14.} $725 - 300 = \qty{425}{kg}$: lima penumpang.

\textbf{15.} Daya angkatnya $\propto \rho_c - \rho_h = \rho_c(1 - 293/T)$: untuk melipatduakan
$1 - 293/373 = 0.215$ dituntut $T = 293/0.57 = \qty{514}{K}$ (\qty{240}{\degreeCelsius}),
jauh di atas yang ditahan nilon (kira-kira \qty{120}{\degreeCelsius}).

\textbf{16.} Helium $\rho = 1.205 \times 4/29 = \qty{0.166}{kg/m^3}$: daya angkatnya
$(1.205 - 0.166) \times 2800 \times 9.81 = \qty{28.5}{kN}$ (\qty{2.9}{t}), empat
kali lebih besar --- tetapi helium mahal dan hilang pada tiap pendaratan, sedangkan
udara panas gratis dan daya angkatnya dapat diatur dengan pembakarnya.

\textbf{17.} $0.21 \times 1.013\times10^5 = \qty{2.1e4}{Pa}$.

\textbf{18.} $n^*_{\mathrm{O}_2} = P_{\mathrm{O}_2}/k_BT$: ruangan $2.1\times10^4/
(1.38\times10^{-23} \times 293) = \qty{5.3e24}{m^{-3}}$; Everest $0.21 \times 0.33
\times 1.013\times10^5/(1.38\times10^{-23} \times 250) = \qty{2.0e24}{m^{-3}}$:
$38\%$ dari nilai di ruangan itu.

\textbf{19.} $H = RT/Mg = 8.314 \times 260/(0.029 \times 9.81) = \qty{7.6}{km}$;
$P_0/2$ pada $H\ln 2 = \qty{5.3}{km}$.

\textbf{20.} $c_s = \sqrt{1.4 \times 8.314 \times 293/0.029} = \qty{343}{m/s}$,
berbanding $u = \qty{503}{m/s}$ bagi campurannya: $c_s/u = \sqrt{\gamma/3} = 0.68$.
Bunyi adalah usikan yang dibawa oleh molekulnya sendiri, sehingga ia
merambat dengan laju yang seorde dengan laju molekulnya --- sedikit lebih kecil, karena
hanya komponen menyusuri arah rambatnya yang berperan.

\textbf{21.} $c_s \propto \sqrt T$: $343\sqrt{303/293} = \qty{349}{m/s}$.
Helium: $\sqrt{(5/3) \times 8.314 \times 293/0.004} = \qty{1010}{m/s}$: resonansi
saluran suaranya berskala dengan $c_s$, sehingga setiap forman menjadi
tiga kali lebih tinggi.

\textbf{22.} Lebih dingin: $343\sqrt{250/293} = \qty{317}{m/s}$, $8\%$ lebih lambat;
\omterm{def:b1:kinetic-theory:pressure}{tekanannya} tak masuk hitungan.

\textbf{23.} $N$ sama (dengan $P$, $V$, $T$ yang sama); massanya $\times 40/29 = \qty{333}{kg}$;
kelajuannya $\times\sqrt{29/40}$: $u = \qty{428}{m/s}$; $U = \tfrac32 nRT$ dan bukan
$\tfrac52$: $60\%$.

\textbf{24.} $U = \tfrac52 \times 8320 \times 8.314 \times 293 = \qty{5.1e7}{J}$;
$\Delta U = \tfrac52 nR \times 10 = \qty{1.7e6}{J}$.

\textbf{25.} Satu rerata, $\langle v^2\rangle$, menetapkan \omterm{def:b1:kinetic-theory:pressure}{tekanannya}
($\tfrac13 n^*m\langle v^2\rangle$), mendefinisikan \omterm{def:b1:kinetic-theory:temperature}{suhunya}
($\tfrac12 m\langle v^2\rangle = \tfrac32 k_BT$), memberikan persamaan keadaannya
dan \omterm{def:b1:kinetic-theory:U}{energi dalamnya}, serta menetapkan skala laju bunyinya.
\end{solution}
