\section*{Bab \ref{ch:b1:rays-reflection-refraction} --- Optika Geometri: Sinar, Pemantulan, Pembiasan}

\begin{solution}{exo:b1:rays-reflection-refraction:1}
$v = c/n$: air \qty{2.26e8}{m/s}, kaca \qty{2.00e8}{m/s}, intan
\qty{1.24e8}{m/s}. Serat: $t = nL/c = 1.48 \times 10^3/3.00\times10^8 =
\qty{4.9}{\micro s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:2}
$\sin i_2 = \sin i_1/1.33$: $30^\circ \to 22.1^\circ$; $60^\circ \to
40.6^\circ$. Dari dalam air pada $50^\circ$: $1.33\sin 50^\circ = 1.02 > 1$,
tak ada sinar bias --- \omterm{thm:b1:rays-reflection-refraction:tir}{pemantulan sempurna} ($i_c = 48.8^\circ$).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}
$i_c = \arcsin(1/n)$: kaca $41.8^\circ$, intan $24.4^\circ$, air
$48.8^\circ$. Cakram langitnya membentang $2 \times 48.8^\circ = 97.5^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
Matanya pada \qty{1.60}{m}. Sinar dari kakinya mencapai mata setelah
memantul pada separuh tinggi matanya, \qty{0.80}{m}; sinar dari puncak
kepalanya memantul di tengah antara puncak dan mata, pada \qty{1.65}{m}.
Cermin dari \qty{0.80}{m} sampai \qty{1.65}{m}: tingginya \qty{0.85}{m},
separuh tinggi badannya --- bebas dari jaraknya (kaidah titik tengah
berlaku pada jarak berapa pun).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
Untuk sudut kecil: kedudukan semunya $h' = h/n$ bila dilihat dari medium
yang lebih renggang, dan $h' = nh$ dari medium yang lebih rapat. Uang
logam: $1.2/1.33 =
\qty{0.90}{m}$. Pemancing: $1.33 \times 1.8 = \qty{2.4}{m}$. Alasannya:
untuk titik permukaan pada jarak mendatar $x$, $\tan i_2 \approx i_2 =
x/h$ dan $\tan i_1 \approx i_1 = x/h'$ dengan $i_1 = n i_2$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$, sehingga
$\frac{A + D_{\min}}{2} = 49.5^\circ$, $D_{\min} = 38.9^\circ$.
Agar keluar diperlukan $r' \leq i_c$; selain itu $\sin r = \sin i/n \leq 1/n$ memberi
$r \leq i_c$; jadi $A = r + r' \leq 2i_c$. Di sini $i_c = \arcsin(1/1.52) =
41.1^\circ$, $2i_c = 82.3^\circ > 60^\circ$: memenuhi.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$,
$\theta_a = 14.1^\circ$. $\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$. Sepanjang \qty{1}{km}:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}
$D \approx (n - 1)A$: merah $0.510 \times 10^\circ = 5.10^\circ$, biru
$5.30^\circ$; lebar spektrumnya $0.20^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
Masuk: $\sin r = \sin 45^\circ/1.5 = 0.471$, $r = 28.1^\circ$. Kedua
mukanya sejajar, sehingga \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang} pada muka kedua adalah $r$ dan
$\sin i' = n\sin r = \sin i$: $i' = i$, sinar keluarnya sejajar dengan
sinar datangnya. Sepanjang lintasan di dalamnya yang panjangnya
$e/\cos r$, geseran tegak lurusnya adalah $(e/\cos r)\sin(i - r)$:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
Dua pembiasan ($2(i - r)$) dan dua pemantulan ($2(\pi - 2r)$):
$D = 2\pi + 2i - 6r$. $\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ memberi
$3\cos i = n\cos r$, $9\cos^2 i = n^2 - 1 + \cos^2 i$, $\cos^2 i =
(n^2 - 1)/8$. $n = 1.333$: $\cos^2 i = 0.0971$, $i = 71.8^\circ$,
$\sin r = 0.950/1.333$, $r = 45.5^\circ$, $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$: pelanginya berada pada $230.9^\circ - 180^\circ =
51^\circ$ dari titik antimatahari, di luar pelangi utama ($42^\circ$).
Merah ($n = 1.331$) pada $50.4^\circ$, ungu ($1.343$) pada $53.6^\circ$:
warnanya terbalik, ungu di luar.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$, $i_c = 89.56^\circ$: sinar
dalam $0.44^\circ = \qty{7.7}{mrad}$ dari permukaan jalan dipantulkan
sempurna (secara umum, sudut susurnya $\approx \sqrt{2\Delta n/n}$). Dari
ketinggian \qty{1.2}{m}, garis pandangnya menyusur dengan sudut ini di
luar jarak $1.2/0.0077 \approx \qty{160}{m}$: jalan di sana memantulkan
langit --- ``genangan air'' itu.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$;
$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$, yang nol tepat ketika
$\sin i_1/v_1 = \sin i_2/v_2$, yaitu $n_1\sin i_1 = n_2\sin i_2$ dengan
$n = c/v$.
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$, tak berubah di dalam kaca;
$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$.

\textbf{4.} \omterm{def:b1:rays-reflection-refraction:fiber}{Inti} telanjang hanya menuntun selama permukaannya benar-benar
bersih dan tak tersentuh: debu, air atau genggaman tangan setempat
menggantikan udara dengan indeks yang lebih tinggi dan cahayanya lolos,
sedangkan goresan menghamburkannya. \omterm{def:b1:rays-reflection-refraction:fiber}{Selubungnya} adalah antarmuka
tertutup dan terkendali dengan tangga indeks yang dipilih dengan sengaja.

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$; terpandu jika dan hanya jika $\cos\theta' \geq
n_2/n_1$, yaitu $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$, yaitu
$\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$:
$\theta_a = 12.1^\circ$.

\textbf{8.} $\mathrm{NA} = 0.210$. Tangga yang lebih besar melebarkan
kerucutnya, sehingga cahaya sumber yang menyebar (sebuah LED) lebih banyak tertangkap.

\textbf{9.} Sinarnya menyinggung lingkaran berjari-jari $R_b$ (tepi
dalam) lalu menemui lingkaran berjari-jari $R_b + 2a$ (tepi luar); dalam
segitiga siku-siku pusat--titik singgung--titik tumbukan, sudut di titik
tumbukan antara sinarnya dan jari-jari (garis normalnya) memenuhi
$\sin i = R_b/(R_b + 2a)$. Terpandu jika dan hanya jika $\sin i \geq n_2/n_1$:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$. Lengkungan yang lebih tajam membocorkannya.

\textbf{10.} $t_0 = Ln_1/c$; lintasan sinar paling curam $L/\cos\theta'_{\max} =
Ln_1/n_2$, sehingga $t_1 = Ln_1^2/(n_2 c)$.

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$; sepanjang \qty{2}{km}: \qty{101}{ns}.

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} Sinar yang jauh dari sumbu menempuh lintasan lebih panjang
tetapi di indeks yang lebih rendah, jadi lebih cepat; dengan profil yang
dipilih baik kedua pengaruh itu saling meniadakan dan semua sinar tiba bersamaan.

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$: \omterm{def:b1:rays-reflection-refraction:fiber}{intinya} tidak
besar dibandingkan panjang gelombangnya, sehingga difraksi mendominasi
dan sinar tak lagi bermakna. Pembahasan gelombang memperlihatkan satu
pola terpandu tunggal: sama sekali tak ada pemuluran antarmoda; hanya
dispersi kromatik dan redaman yang tersisa.

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$:
$P/P_0 = 10^{-2}$, satu persen.

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$, sehingga $L_{\max} =
30/0.20 = \qty{150}{km}$.

\textbf{18.} \qty{100}{km} memakan \qty{20}{dB}: marginnya \qty{10}{dB},
sebuah faktor $10$ pada dayanya.

\textbf{19.} LED: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$, $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$. Laser:
$\Delta t_c = \qty{170}{ps}$, $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$.

\textbf{20.} $30/2.0 = \qty{15}{km}$, sepuluh kali lebih pendek. Silika
paling bening di dekat \qty{1550}{nm} (minimum redamannya), dan penguat
optik tersedia di sana.

\textbf{21.} $30/30 = \qty{1}{km}$ antarpenguat, terhadap
\qty{150}{km}: seratnya menang dengan faktor $150$.

\textbf{22.} Antarmoda: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s});
kromatik (laser): $17 \times 0.1 \times 2 = \qty{3.4}{ps}$;
redaman \qty{0.4}{dB}, dapat diabaikan. Pemuluran antarmoda yang
membatasi: \qty{5}{Mbit/s}.

\textbf{23.} Tak ada pemuluran antarmoda; kromatiknya \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}; redamannya \qty{20}{dB}, masih di dalam anggaran.
Dispersi kromatik yang membatasi, pada sekitar \qty{3}{Gbit/s}.

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} Indeks tangga: \qty{10}{Mbit.s^{-1}.km}; kanal moda tunggal:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- sebuah faktor sekitar
$3\times10^4$. Satu fakta: lintasan terpandu yang tunggal, sehingga tak
ada pemuluran waktu tiba yang bersifat geometri (antarmoda).
\end{solution}
