\begin{solution}{exo:b1:heat-engines:5}
Adiabatnya: $V_C = V_B(T_h/T_c)^{1/(\gamma-1)} = 3.0 \times (5/3)^{2.5} =
\qty{10.8}{L}$, $V_D = 1.0 \times 3.59 = \qty{3.59}{L}$ (sehingga $V_C/V_D = 3 =
V_B/V_A$). $Q_h = RT_h\ln3 = 8.314 \times 500 \times 1.099 = \qty{4.57}{kJ}$,
$Q_c = -RT_c\ln3 = \qty{-2.74}{kJ}$, $W = -(4.57 - 2.74) = \qty{-1.83}{kJ}$,
$\eta = 1.83/4.57 = 0.40 = 1 - 300/500$.
\end{solution}

\begin{solution}{exo:b1:heat-engines:6}
Pengakuan pertama: energinya $200 = 1000 - 800$ seimbang; Clausius $1000/400 - 800/300
= 2.5 - 2.67 = \qty{-0.17}{J/K}$, sehingga $S_{\mathrm{cipta}} = \qty{0.17}{J/K} \geq 0$:
mungkin ($\eta = 0.20$). Yang kedua: $1000/400 - 700/300 = +\qty{0.17}{J/K} > 0$:
entropinya harus dimusnahkan --- mustahil. Maksimumnya: $\eta_C = 1 -
300/400 = 0.25$, yakni \qty{250}{J}.
\end{solution}

\begin{solution}{exo:b1:heat-engines:7}
$r^{\gamma-1} = 18^{0.4} = 3.18$; $\rho^\gamma = 2^{1.4} = 2.64$: $\eta = 1 -
1.64/(1.4 \times 3.18 \times 1) = 0.63$. Otto pada $r = 18$: $1 - 1/3.18 = 0.69$;
pada $r = 9$: $0.585$. Diesel memampatkan udara murni: tak ada bahan bakar yang hadir untuk
menyala mendahului, dan bahan bakarnya, yang disuntikkan ke dalam udara pada \qty{900}{K}, terbakar begitu ia

\end{solution}

\begin{solution}{exo:b1:heat-engines:8}
$Q_h = RT_h\ln2 = 8.314 \times 600 \times 0.693 = \qty{3.46}{kJ}$, $Q_c = -8.314
\times 300 \times 0.693 = \qty{-1.73}{kJ}$: $W = \qty{-1.73}{kJ}$ tiap siklus. Dengan
regeneratornya $\eta = 1.73/3.46 = 0.50 = 1 - 300/600$. Tanpa regeneratornya sumber
panasnya harus memasok pula $C_{V,m}(T_h - T_c) = 2.5 \times 8.314 \times 300 =
\qty{6.24}{kJ}$: $\eta = 1.73/(3.46 + 6.24) = 0.18$.
\end{solution}

\begin{solution}{exo:b1:heat-engines:9}
Entropi kedua baloknya: $\Delta S = C\ln(T_f/T_1) + C\ln(T_f/T_2) =
C\ln\bigl(T_f^2/T_1T_2\bigr) = S_{\mathrm{cipta}} \geq 0$ (entropi mesinnya sendiri
kembali ke nilainya tiap siklus), sehingga $T_f \geq \sqrt{T_1T_2}$, dan
$W = C(T_1 + T_2 - 2T_f)$ (hukum pertama) terbesar bagi $T_f$ yang terkecil:
yaitu kasus reversibelnya, $T_f = \sqrt{400 \times 300} = \qty{346.4}{K}$,
$W_{\max} = 4000 \times (700 - 692.8) = \qty{29}{kJ}$. Sentuhan langsung: $T_f =
\qty{350}{K}$, tanpa usaha, $S_{\mathrm{cipta}} = 4000\ln(350^2/120000) =
\qty{82}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:heat-engines:10}
(a) $P = \dot Q_h - |\dot Q_c| = K(T_h - x) - K(y - T_c)$; teras reversibelnya
tak mempertukarkan entropi total: $K(T_h - x)/x = K(y - T_c)/y$. (b) Kalikan silang:
$y(T_h - x) = x(y - T_c)$, sehingga $y(T_h - 2x) = -xT_c$ dan $y = xT_c/(2x - T_h)$.
Lalu $P = K(T_h - x)(1 - y/x) = K(T_h - x)(2x - T_h - T_c)/(2x - T_h)$; dengan
$u = 2x - T_h$, $T_h - x = (T_h - u)/2$ dan $P = K(T_h - u)(u - T_c)/(2u) =
\tfrac{K}{2}[(T_h + T_c) - u - T_hT_c/u]$. (c) $\dd P/\dd u = \tfrac K2(-1 +
T_hT_c/u^2) = 0$ pada $u = \sqrt{T_hT_c}$, sebuah maksimum; lalu $x = (T_h +
\sqrt{T_hT_c})/2$, $y = xT_c/u = (\sqrt{T_hT_c} + T_c)/2$, dan $\eta^\ast = 1 - y/x
= 1 - \sqrt{T_c}(\sqrt{T_h} + \sqrt{T_c})/[\sqrt{T_h}(\sqrt{T_h} + \sqrt{T_c})] = 1 -
\sqrt{T_c/T_h}$; $P_{\max} = \tfrac K2(\sqrt{T_h} - \sqrt{T_c})^2$. (d)
$\eta^\ast = 1 - \sqrt{0.375} = 0.39$, berbanding $\eta_C = 0.625$: nilai nyata $0.40$
pada \cref{exo:b1:heat-engines:1} adalah mesin yang dibangun untuk daya, bukan untuk
\omterm{def:b1:heat-engines:efficiency}{efisiensi}.
\end{solution}

\begin{solution}{exo:b1:heat-engines:11}
Siklus: $Q_h + Q_c + Q_0 = 0$ dan $Q_h/T_h + Q_c/T_c + Q_0/T_0 \leq 0$.
Hilangkan $Q_0 = -(Q_h + Q_c)$: $Q_h(1/T_h - 1/T_0) + Q_c(1/T_c - 1/T_0) \leq 0$,
yakni $Q_c(T_0 - T_c)/(T_cT_0) \leq Q_h(T_h - T_0)/(T_hT_0)$, sehingga
$Q_c/Q_h \leq (1 - T_0/T_h)\,T_c/(T_0 - T_c)$. Faktor pertamanya adalah
\omterm{def:b1:heat-engines:efficiency}{efisiensi} sebuah mesin Carnot di antara pembakar dan ruangannya, yang kedua
COP sebuah \omterm{def:b1:heat-engines:machine}{lemari es} Carnot di antara ruang dingin dan ruangannya: batasnya adalah apa
yang diperoleh dengan menggandengkan keduanya secara ideal. Angkanya: $0.25 \times 11 = 2.75$
(\omterm{def:b1:heat-engines:machine}{lemari es} serapan yang nyata: kira-kira $0.6$).
\end{solution}

\end{solution}

\begin{solution}{pb:b1:heat-engines:1}
\textbf{1.} $P = 250 \times 25 = \qty{6.25}{kW}$.

\textbf{2.} Kerugian rata-ratanya $250 \times 13 = \qty{3.25}{kW}$ selama $200 \times 86400 =
\qty{1.73e7}{s}$: $E = \qty{5.6e10}{J} = \qty{15600}{kW h}$.

\textbf{3.} $15600 \times 0.25 = 3900$~\texteuro.

\textbf{4.} Gas $15600/0.90 = \qty{17300}{kW h}$: $1730$~\texteuro.

\textbf{5.} \omterm{def:b1:dc-circuits:resistor}{Resistornya}: $15600/0.40 = \qty{39000}{kW h}$ bahan bakar; ketelnya
\qty{17300}{kW h}: $2.25$ kali lebih sedikit. Listrik adalah usaha, yaitu bentuk yang dapat
melakukan apa saja; mengubahnya menjadi kalor \qty{20}{\degreeCelsius} di dalam sebuah \omterm{def:b1:dc-circuits:resistor}{resistor}
membuang $60\%$ bahan bakar yang sudah hilang di pembangkitnya dan
membuang pula kemungkinan memompa kalor.

\textbf{6.} Sepanjang satu siklus $W + Q_h + Q_c = 0$ dan $Q_h/T_{\mathrm{dalam}} +
Q_c/T_{\mathrm{luar}} \leq 0$ dengan $Q_h < 0$, $Q_c > 0$: $Q_c \leq
|Q_h|T_{\mathrm{luar}}/T_{\mathrm{dalam}}$, sehingga $W = |Q_h| - Q_c \geq |Q_h|(1 -
T_{\mathrm{luar}}/T_{\mathrm{dalam}})$ dan $e_p = |Q_h|/W \leq T_{\mathrm{dalam}}/(T_{\mathrm{dalam}} -
T_{\mathrm{luar}})$.

\textbf{7.} $293/25 = 11.7$; $6.25/11.7 = \qty{0.53}{kW}$.

\textbf{8.} $293/13 = 22.5$; $15600/22.5 = \qty{690}{kW h}$ (kira-kira
$170$~\texteuro).

\textbf{9.} Usaha tiap joule yang diangkat adalah $(T_{\mathrm{dalam}} - T_{\mathrm{luar}})/
T_{\mathrm{dalam}}$, yaitu ``ketinggian'' angkatannya; makin dingin di luar, makin
tinggi tiap joule harus dipompa --- dan makin banyak joule yang diperlukan, karena
kerugiannya juga sebanding dengan $T_{\mathrm{dalam}} - T_{\mathrm{luar}}$: maka
\omterm{thm:b1:dc-circuits:kirchhoff}{daya listriknya} tumbuh sebagai kuadrat selisih suhunya, yang terburuk
tepat ketika pemanasannya paling dibutuhkan.

\textbf{10.} Dari udara luarnya, yang menjadi dingin: $Q_c = 6.25 - 0.53 =
\qty{5.7}{kW}$. Hukum kedua melarang kalor mengalir \emph{secara spontan}
dari yang dingin ke yang panas; di sini usahalah yang membayar angkatannya. Periksa: entropi yang diambil dari
udaranya $5.72/268 = \qty{21.3}{W/K}$, entropi yang diberikan ke rumahnya $6.25/293 =
\qty{21.3}{W/K}$ --- tak ada yang dimusnahkan, tak ada yang tercipta: pompa yang ideal.

\textbf{11.} Kompresornya (menerima $W$), kondensornya (memberikan $|Q_h|$),
katup ekspansinya (tak keduanya), evaporatornya (menerima $Q_c$). Kalor hanya mengalir
menuruni lereng suhu: \omterm{def:b1:heat-engines:vapourcycle}{refrigerannya} harus lebih dingin daripada udaranya untuk
mengambil kalor darinya dan lebih panas daripada air rumahnya untuk memberi kalor kepadanya.

\textbf{12.} $T_{\mathrm{ev}} = 268 - 8 = \qty{260}{K}$; $e_{p,C} = 313/(313 - 260)
= 5.9$.

\textbf{13.} $e_p = 0.55 \times 5.9 = 3.25$; $W = 6.25/3.25 = \qty{1.92}{kW}$;
$Q_c = 6.25 - 1.92 = \qty{4.33}{kW}$ dari udaranya.

\textbf{14.} $\dot m = 4330/200000 = \qty{22}{g/s}$, \qty{1.3}{kg} tiap menit.

\textbf{15.} $T_{\mathrm{ev}} = \qty{272}{K}$, $313/41 = 7.6$, $e_p = 4.2$;
$W = 3.25/4.2 = \qty{0.77}{kW}$.

\textbf{16.} $e_p = 0.55 \times 313/(321 - T_{\mathrm{luar}}) = 172/(321 -
T_{\mathrm{luar}})$ ($T_{\mathrm{luar}}$ dalam \omterm{def:b1:kinetic-theory:temperature}{kelvin}). Semusim: $15600/4.2 =
\qty{3700}{kW h}$, $930$~\texteuro\ --- berbanding $3900$ dengan \omterm{def:b1:dc-circuits:resistor}{resistor} dan
$1730$ dengan gas.

\textbf{17.} $3700/0.40 = \qty{9300}{kW h}$ bahan bakar: $46\%$ lebih sedikit daripada
ketelnya dan empat kali lebih sedikit daripada \omterm{def:b1:dc-circuits:resistor}{resistornya}, bahkan melalui pembangkit
berefisiensi $40\%$.

\textbf{18.} Tuntutannya $250 \times 35 = \qty{8.75}{kW}$; $T_{\mathrm{ev}} = \qty{250}{K}$,
$313/63 = 5.0$, $e_p = 2.7$, $W = \qty{3.2}{kW}$ seandainya ia mampu. Kemampuannya
\qty{3.8}{kW}: cadangannya $8.75 - 3.8 \approx \qty{5}{kW}$ \omterm{def:b1:dc-circuits:resistor}{resistor}. Pompa yang
ditakar bagi jam terdingin akan berlebihan ukurannya --- dan berhidup-mati
secara tak efisien --- sepanjang musim dingin; kita menakar bagi suhu rancangannya lalu
membiarkan \omterm{def:b1:dc-circuits:resistor}{resistor} yang murah menutup keadaan ekstrem yang jarang itu.

\textbf{19.} $338/(338 - 260) = 4.33$, $e_p = 2.4$, $W = 6.25/2.4 =
\qty{2.6}{kW}$: $36\%$ lebih banyak listrik daripada dengan lantai pada
\qty{35}{\degreeCelsius}. Pemancar kalor bersuhu rendah adalah separuh dari
pemasangannya.

\textbf{20.} Entropi sepanjang satu siklus: $Q_h/T_{\mathrm{dalam}} + Q_c/T_{\mathrm{luar}} +
S_{\mathrm{cipta}} = 0$ memberikan $Q_c = T_{\mathrm{luar}}|Q_h|/T_{\mathrm{dalam}} -
T_{\mathrm{luar}}S_{\mathrm{cipta}}$; sisipkan ke dalam $W = |Q_h| - Q_c$.

\textbf{21.} Kondensornya: $6250\,(1/293 - 1/313) = \qty{1.36}{W/K}$; evaporatornya:
$4330\,(1/260 - 1/268) = \qty{0.50}{W/K}$.

\textbf{22.} $268 \times 1.86 = \qty{0.50}{kW}$: seperempat dari
\qty{1.92}{kW}.

\textbf{23.} $(1.92 - 0.53)/268 = \qty{5.2}{W/K}$; penukar kalornya menyumbang
$36\%$; sisanya datang dari kompresornya (gesekan, pemampatan yang tak
isentropik, kerugian motornya), pencekikan di katupnya (yaitu sebuah ekspansi
bebas), penurunan tekanannya, dan kipasnya.

\textbf{24.} $T_{\mathrm{cd}} = \qty{303}{K}$, $T_{\mathrm{ev}} = \qty{264}{K}$:
kondensornya $6250\,(1/293 - 1/303) = \qty{0.70}{W/K}$, evaporatornya
$4330\,(1/264 - 1/268) = \qty{0.25}{W/K}$: \qty{0.95}{W/K} dan bukan
$1.86$, yakni $268 \times 0.91 \approx \qty{0.25}{kW}$ yang dihemat ($13\%$;
seluruh mesinnya, yang bernilai $0.55$ dari nilai Carnot yang lebih baik, akan memperoleh
lebih). Menyeparuhkan sebuah selisih berarti melipatduakan luas penukarnya: penukar kalor yang lebih besar,
lebih mahal, dan daya kipas yang lebih besar.

\textbf{25.} $15600/3700 = 4.2$ kilowatt-jam kalor tiap kilowatt-jam
listrik; $930$~\texteuro\ berbanding $1730$ untuk gas; karbonnya: $3700
\times 0.060 = \qty{220}{kg}$ berbanding $17300 \times 0.20 = \qty{3500}{kg}$ ---
lima belas kali lebih sedikit.
\end{solution}
