\begin{solution}{exo:b1:dc-circuits:7}
Superposisi: $E_1$ sendirian melihat $r_2 \parallel R = \qty{1.0}{\ohm}$,
$u = 10 \times 1/(1 + 1) = \qty{5.0}{V}$; $E_2$ sendirian melihat $r_1 \parallel R
= \qty{0.667}{\ohm}$, $u = 5 \times 0.667/2.667 = \qty{1.25}{V}$; totalnya
\qty{6.25}{V}. Millman: $(10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = \qty{6.25}{V}$.
Aki 1: $(10 - 6.25)/1 = \qty{3.75}{A}$ keluar; aki 2: $(5 - 6.25)/2
= \qty{-0.63}{A}$, yaitu \qty{0.63}{A} \emph{masuk} ke dalamnya --- ia
sedang diisi oleh aki yang lebih kuat; bebannya $6.25/2 = \qty{3.1}{A}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:8}
$i = 9/(R + 3)$, $P = Ri^2$, $\eta = R/(R + 3)$: $\qty{1}{\ohm}$:
\qty{5.1}{W}, $25\%$; $\qty{3}{\ohm}$: \qty{6.75}{W}, $50\%$;
$\qty{12}{\ohm}$: \qty{4.3}{W}, $80\%$. Perangkat bertenaga baterai
menginginkan efisiensi (umur baterai), jadi $R \gg r$; beban yang
tersesuaikan membuang separuh energinya sebagai kalor di dalam baterai.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:9}
$R = (5.0 - 2.0)/0.015 = \qty{200}{\ohm}$; $P_R = 3.0 \times 0.015 =
\qty{45}{mW}$, $P_{\mathrm{LED}} = 2.0 \times 0.015 = \qty{30}{mW}$. \omterm{met:b1:dc-circuits:loadline}{Garis

karakteristik tegaknya pada \qty{2.0}{V}. Pada \qty{3.3}{V} \omterm{def:b1:dc-circuits:resistor}{resistor} yang
sama hanya memberi \qty{6.5}{mA}; untuk \qty{15}{mA} diperlukan $R = 1.3/0.015 =
\qty{87}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:10}
$u = E\left[\dfrac12 - \dfrac{1 + \epsilon}{2 + \epsilon}\right] =
-E\dfrac{\epsilon}{2(2 + \epsilon)} \approx -\dfrac{E\epsilon}{4} =
\qty{-1.25}{mV}$. Dalam pembagi biasa perubahan yang sama duduk di atas
simpangan \qty{2.5}{V} (\qty{2.50125}{V}); jembatannya menyerahkan
\qty{1.25}{mV} di sekitar nol, yang dapat diperkuat seribu kali lipat
tanpa menjenuhkan apa pun.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:11}
Oleh kesetangkupannya, ketiga tetangga sudut masuknya berbagi satu
potensial, dan ketiga tetangga sudut keluarnya berbagi potensial yang
lain. Arus $I$ terbelah menjadi $I/3$ pada ketiga rusuk masuknya,
masing-masing lalu menjadi dua $I/6$ menyusuri keenam rusuk tengahnya,
yang bergabung kembali menjadi $I/3$ pada ketiga rusuk keluarnya.
\omterm{def:b1:dc-circuits:voltage}{Tegangannya}: $RI/3 + RI/6 + RI/3 = 5RI/6$, sehingga $R_{\mathrm{eq}} = 5R/6$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:12}
Thévenin: $E = I_N r = \qty{8.0}{V}$ dengan \qty{4.0}{\ohm} terangkai seri.
Loop: $8.0 - 6.0 = (4.0 + 2.0)i$, $i = \qty{0.33}{A}$ menuju akinya.
Akinya menerima $6.0 \times 0.33 = \qty{2.0}{W}$ (sedang diisi); \omterm{def:b1:dc-circuits:resistor}{resistor}
\qty{2}{\ohm} melesapkan \qty{0.22}{W}; \omterm{def:b1:dc-circuits:voltage}{tegangan} pada ujung pasangan
Nortonnya adalah $8.0 - 4.0 \times 0.33 = \qty{6.67}{V}$, sehingga
\qty{4}{\ohm} miliknya membawa \qty{1.67}{A} dan melesapkan
\qty{11.1}{W}, dan sumber arusnya memberi $2.0 \times 6.67 = \qty{13.3}{W}$
--- yang semuanya cocok.
\end{solution}

\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} Ggl ideal $E$ terangkai seri dengan $r$: $u = E - ri$.

\textbf{2.} $u(0) = \qty{12.6}{V}$; $I_N = E/r = \qty{1260}{A}$. Kunci pas
yang melintasi kutubnya akan membawanya: $\qty{16}{kW}$ dalam beberapa
gram baja --- ia melebur dan menyembur.

\textbf{3.} \omterm{def:b1:dc-circuits:sources}{Sumber arus} $I_N = \qty{1260}{A}$ terangkai paralel dengan
\qty{10}{m\ohm}.

\textbf{4.} $i = E/(R + r)$; $u = ER/(R + r)$; $P_R = E^2R/(R + r)^2$;
$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$: maksimum pada $R = r$,
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$; efisiensinya
$R/(R + r) = 50\%$.

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$; $12.6 - 0.010 \times
120 = \qty{11.40}{V}$. Menggambarkan $u$ terhadap $i$ memberi garis lurus
dengan perpotongan $E$ dan kemiringan $-r$ (\omterm{prop:b1:units-dimensions:regression}{kuadrat terkecil}, \cref{ch:b1:units-dimensions}).

\textbf{7.} Dua lampu: \qty{110}{W} pada \qty{12}{V}, \qty{9.2}{A};
$60/9.2 = \qty{6.5}{h}$. Energinya $60 \times 3600 \times 12 = \qty{2.6}{MJ}$.

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$; terangkai paralel \qty{1.31}{\ohm}.

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$; $u = 12.6 - 1.4 = \qty{11.2}{V}$;
$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$; hilang $0.010 \times 140^2 =
\qty{196}{W}$.

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$; $i = 12.6/0.0854
= \qty{148}{A}$; $u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$; $i_h =
\qty{8.5}{A}$ untuk kedua lampunya.

\textbf{12.} Tiap lampu: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$ terhadap
\qty{55}{W}: turun $14\%$.

\textbf{13.} \omterm{thm:b1:dc-circuits:kirchhoff}{Hukum loop}: \omterm{def:b1:dc-circuits:voltage}{tegangan} pada ujungnya adalah $u = E - ri$;
\qty{140}{A} milik starternya menghasilkan jatuh \omterm{def:b1:dc-circuits:voltage}{tegangan} \qty{1.4}{V}
di dalam akinya, dan lampunya, yang terangkai paralel pada ujung yang
sama, menerima \omterm{def:b1:dc-circuits:voltage}{tegangan} yang sudah berkurang itu.

\textbf{14.} Hanya sebagian: kabel yang lebih tebal memotong jatuh
\omterm{def:b1:dc-circuits:voltage}{tegangan} di kabelnya, bukan di $r$; lampu yang diberi daya dari ujung
akinya tetap melihat $E - ri$.

\textbf{15.} Tiga cabang di antara dua simpul yang sama: $(E, r)$,
$(E_a, r_a)$ dan $R_h = \qty{1.31}{\ohm}$.

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} Alternator $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$; aki
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (mengalir \emph{masuk} ke
dalamnya); lampu $12.79/1.31 = \qty{9.8}{A}$; $28.3 = 18.6 + 9.8$.

\textbf{18.} Sedang diisi. Ia menerima $12.79 \times 18.6 = \qty{238}{W}$
(\qty{234}{W} tersimpan, \qty{3.5}{W} menjadi kalor); alternatornya
memasok $12.79 \times 28.3 = \qty{362}{W}$ pada ujungnya.

\textbf{19.} Aki sendirian (alternator $\to$ \qty{50}{m\ohm}): bebannya
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$, $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$. Alternator sendirian (aki $\to$ \qty{10}{m\ohm}): bebannya
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$, $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$. Jumlahnya \qty{12.79}{V}.

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$: \qty{3.60}{V} pada
\qty{20}{\degreeCelsius}, \qty{1.40}{V} pada \qty{90}{\degreeCelsius}.

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$;
$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$: $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$.

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$, nol ketika $R_s =
\qty{775}{\ohm}$, yaitu pada suhu pertengahan tempat termistornya
mencapai nilai itu.

\textbf{23.} $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$ tiap penghantar, \qty{3.2}{m\ohm} seluruhnya; jatuh \omterm{def:b1:dc-circuits:voltage}{tegangannya} $0.0032
\times 140 = \qty{0.45}{V}$; dayanya \qty{63}{W}.

\textbf{24.} \qty{4}{mm^2}: \qty{6.4}{m\ohm} tiap penghantar, \qty{12.8}{m\ohm}
seluruhnya: jatuh \omterm{def:b1:dc-circuits:voltage}{tegangannya} \qty{1.8}{V}, \qty{250}{W} terbuang --- starternya
akan kehilangan seperenam \omterm{def:b1:dc-circuits:voltage}{tegangannya}. Kabel tebal menjaga $R_{\text{kabel}} \ll R_s$.

\textbf{25.} Sekitar \qty{11.1}{V} pada ujungnya selama pengengkolan;
starternya menerima $1568/(1568 + 196) \approx 89\%$ daya akinya;
meredupnya lampu adalah \omterm{thm:b1:dc-circuits:kirchhoff}{hukum loop}, $u = E - ri$.
\end{solution}
