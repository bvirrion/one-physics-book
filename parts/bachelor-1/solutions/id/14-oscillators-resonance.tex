\section*{Bab \ref{ch:b1:oscillators-resonance} --- Osilator Mekanik: Peredaman dan Resonansi}

\begin{solution}{exo:b1:oscillators-resonance:1}
$\omega_0 = \sqrt{80/0.20} = \qty{20}{rad/s}$, $f_0 = \qty{3.2}{Hz}$, $T_0 =
\qty{0.31}{s}$; $v_{\max} = A\omega_0 = \qty{1.0}{m/s}$; $E = \tfrac12 kA^2 =

\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:2}
$T = 2\pi\sqrt{\ell/g}$: \qty{2.0}{s} di Bumi, \qty{4.9}{s} di Bulan.
Periode massa--pegasnya, $2\pi\sqrt{m/k}$, tidak melibatkan $g$: tak
berubah.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:3}
$\tau = 2Q/\omega_0 = 100/(4\pi) = \qty{8.0}{s}$; energinya $\propto \eu^{-2t/\tau}$
berkurang separuh pada $t = \tau\ln2/2 = \qty{2.8}{s}$; $5\%$ setelah $3\tau = \qty{24}{s}$,
yaitu sekitar $48 \approx Q$ osilasi.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:4}
$X_0 = F_0/k = \qty{1.0}{cm}$; pada resonansi $\approx QX_0 = \qty{10}{cm}$;
pada $3\omega_0$: $X_0/\sqrt{64 + 0.09} = \qty{0.13}{cm}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:5}
$v = Lf_0 = 8.0 \times 1.3 = \qty{10}{m/s}$ (\qty{37}{km/h}). Pada
\qty{60}{km/h}: $f = 16.7/8.0 = \qty{2.1}{Hz}$, $x_r = 1.6$: $X/X_g =
\sqrt{(1 + 5.2)/(2.4 + 5.2)} = 0.90$ --- nyaris tak terisolasi; isolasi
menuntut $x_r \gg 1$ (pada \qty{120}{km/h}, $x_r = 3.2$: $0.45$).
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:6}
Usaha sebuah dorongan positif hanya bila ia bekerja searah geraknya;
ketika didorong tiap periode, semua dorongannya berjumlah; pada irama
lain dorongannya berganti-ganti. \omterm{def:b1:transient-regimes:firstorder}{Keadaan tunaknya}: $2\pi E/Q = \qty{25}{J}$, $E = \qty{80}{J}$;
$mg\ell(1 - \cos\theta_{\max}) = 80$: $1 - \cos\theta_{\max} = 0.109$,
$\theta_{\max} = 27^\circ$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:7}
$\delta = \ln(1/0.9) = 0.105$, $Q = \pi/\delta = 30$; $\Delta f = f_0/Q =
\qty{0.067}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:8}
Simpangannya: sefase ($\varphi = 0$) jauh di bawahnya, $-\pi/2$ pada
resonansi, $-\pi$ (berlawanan) jauh di atasnya. Kecepatannya pada
resonansi: sefase dengan gayanya. Dayanya adalah $\vect F\cdot\vect v$:
fase kecepatannyalah, bukan fase simpangannya, yang memutuskan berapa
banyak usaha yang dilakukan penggeraknya.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:9}
$V = \omega X = \omega(F_0/m)/\sqrt{(\omega_0^2 - \omega^2)^2 + (\omega\omega_0/Q)^2}$;
bagi pembilang dan penyebutnya dengan $\omega\omega_0/Q$: $V = (F_0Q/m\omega_0)/
\sqrt{1 + Q^2(\omega_0/\omega - \omega/\omega_0)^2}$, dan $F_0Q/m\omega_0 = F_0/\alpha$.
Akar kuadratnya $\geq 1$, dan sama dengan $1$ tepat pada $\omega = \omega_0$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:10}
$P = F_0^2/2\alpha = \qty{10}{W}$; $V_{\max} = F_0/\alpha = \qty{20}{m/s}$,
$E = \tfrac12 mV_{\max}^2 = \qty{40}{J}$; $E/P = \qty{4.0}{s}$ dan $Q/\omega_0 =
m/\alpha = \qty{4.0}{s}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:11}
$3\tau = \qty{10}{s}$: $\tau = \qty{3.3}{s}$, $Q = \omega_0\tau/2 = 2\pi \times
440 \times 3.3/2 \approx 4600$; $\Delta f = f_0/Q = \qty{0.1}{Hz}$. Ia
menanggapi, dan memancarkan, pada dasarnya satu frekuensi saja: sebuah
acuan yang baik sampai beberapa bagian dalam $10^4$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:12}
$m\ddot x = mg - \rho_wgS(h + x) = -\rho_wgSx$ (karena $mg = \rho_wgSh$):
$\omega_0^2 = \rho_wgS/m = g/h$; $T = 2\pi\sqrt{0.10/9.81} = \qty{0.63}{s}$.
Baik massanya maupun massa jenis cairannya masuk hanya lewat kedalaman
kesetimbangannya $h$.
\end{solution}

\begin{solution}{pb:b1:oscillators-resonance:1}
\textbf{1.} Gayanya: berat (diimbangi regangan statisnya), pegas
$-ku$, peredam $-\alpha\dot u$; $m\ddot x = -ku - \alpha\dot u$ dengan $\ddot x =
\ddot x_g + \ddot u$.

\textbf{2.} $\ddot u + (\omega_0/Q)\dot u + \omega_0^2u = -\ddot x_g$, $\omega_0 =
\sqrt{k/m}$, $Q = \sqrt{km}/\alpha$.

\textbf{3.} $k = m\omega_0^2 = \pi^2 = \qty{9.9}{N/m}$; $\alpha = m\omega_0/Q =
3.14/0.70 = \qty{4.5}{N.s/m}$.

\textbf{4.} $u = -a_g/\omega_0^2$: massanya melendut dengan jumlah tetap ---
alatnya menjadi akselerometer pada frekuensi nol (dan pengukur kemiringan).

\textbf{5.} Simpangan relatifnya $u(t)$.

\textbf{6.} $-\omega^2\underline U + j\omega(\omega_0/Q)\underline U + \omega_0^2\underline U
= \omega^2X_g$; bagi dengan $\omega_0^2$.

\textbf{7.} $U/X_g = x_r^2/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$.

\textbf{8.} $U \to X_g$: massanya tak bergerak dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersialnya}
(kelembamannya menahannya diam sementara pegasnya terlalu lunak untuk
menyeretnya); rangkanya bergerak di sekelilingnya, dan penanya merekam
simpangan tanahnya --- sebuah seismometer.

\textbf{9.} $U \approx x_r^2X_g = \omega^2X_g/\omega_0^2 = a_g/\omega_0^2$: sebuah
akselerometer.

\textbf{10.} $U = QX_g = 0.7X_g$: tanpa puncak; kedua rezimnya bergabung
dengan mulus dan alatnya tidak berdering.

\textbf{11.} Untuk $Q = 0.7$: sebuah garis berkemiringan $2$ yang naik
ke $1$ di dekat $x_r = 1$, lalu mendatar pada $1$; untuk $Q = 5$: sama
dengan itu tetapi dengan puncak setinggi $5$ pada $x_r = 1$.

\textbf{12.} $\underline U \to -X_g$: $u = -x_g$, massanya diam sementara
rangkanya bergerak.

\textbf{13.} $\Delta\ell = g/\omega_0^2 = 9.81/(2\pi \times 0.02)^2 = \qty{620}{m}$:
mustahil.

\textbf{14.} $\ell = g/\omega_0^2 = \qty{620}{m}$.

\textbf{15.} Pintu pagar: bandulnya berayun terhadap sumbu yang nyaris
tegak, sehingga hanya sebagian kecil $g$ yang memulihkannya dan lengan
pendek pun memberi periode panjang. Umpan balik gaya: sebuah kumparan
menahan massanya tetap, dan arus yang diperlukan sebanding dengan
percepatan tanahnya --- kekakuannya bersifat elektronik dan dapat disetel.

\textbf{16.} $\Delta\ell = 9.81/\pi^2 = \qty{0.99}{m}$: dapat diwujudkan.

\textbf{17.} $x_r = 0.2$: $U/X_g = 0.04/\sqrt{0.92 + 0.08} = 0.040$:
\qty{40}{\micro m} --- rezim akselerometer, yang menuntut penguatan
(tuas optik atau elektronik).

\textbf{18.} $x_r = 10$: $100/\sqrt{9801 + 204} \approx 1.00$: \qty{1.0}{mm},
yaitu rekaman simpangan yang setia.

\textbf{19.} $\omega_0 = \qty{6.3e3}{rad/s}$: $U = g/\omega_0^2 = \qty{2.5e-7}{m}$
--- dibaca sebagai perubahan celah sebuah kapasitor.

\textbf{20.} $U/(a_g/\omega_0^2) = 1/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$; dengan
$Q = 0.7$ ia menurun secara monoton dan tetap di atas $0.9$ sampai
$x_r \approx 0.7$: kira-kira \qty{700}{Hz}.

\textbf{21.} $\tau = 2Q/\omega_0 = \qty{0.22}{ms}$: ia mengikuti sebuah
jatuh (puluhan milisekon) tanpa ketertinggalan.

\textbf{22.} $\tau = 1.4/3.14 = \qty{0.45}{s}$: setelah sebuah tangga
tanah, rekamannya mapan dalam setengah sekon tanpa dering.

\textbf{23.} $Q$ yang tinggi akan membuat alatnya berdering pada $f_0$
setelah tiap sentakan (merekam dirinya sendiri alih-alih tanahnya) dan
memuncak di dekat $f_0$; $Q \approx 0.7$ mendatar dan cepat.

\textbf{24.} $U = QX_g = 0.7X_g$ terhadap $X_g$ yang ideal (atau $a_g/\omega_0^2
= X_g$): meleset $30\%$ terlalu rendah.

\textbf{25.} Bandingkan frekuensi \omterm{def:b1:wave-propagation:signal}{sinyalnya} dengan $f_0$: jauh di
atasnya, massanya menjadi acuan tetap dan $u$ adalah simpangannya; jauh
di bawahnya, $u$ adalah percepatannya dibagi $\omega_0^2$.
\end{solution}
