\begin{solution}{exo:b1:filters-transfer-functions:1}
$0.5 \to \qty{-6.0}{dB}$; $10 \to \qty{20}{dB}$; $1/\sqrt2 \to \qty{-3.0}{dB}$;
$100 \to \qty{40}{dB}$. $\qty{-40}{dB} \to 0.01$; $\qty{+6}{dB} \to 2.0$.
Berderet: $-6 - 20 = \qty{-26}{dB}$, yaitu nisbah $0.05$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:2}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 10^4 \times 1.6\times10^{-8}) =
\qty{1.0}{kHz}$. \omterm{def:b1:filters-transfer-functions:H}{Penguatannya} $-10\log(1 + (f/f_c)^2)$: \qty{100}{Hz}:
\qty{-0.04}{dB}; \qty{1}{kHz}: \qty{-3}{dB}; \qty{10}{kHz}: \qty{-20}{dB};
\qty{100}{kHz}: \qty{-40}{dB}.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:3}
(a) lolos rendah ($C$ menghubungsingkatkan keluarannya pada $f$ tinggi);
(b) lolos tinggi ($C$ membuka pada $f$ rendah); (c) lolos rendah orde
kedua ($L$ membuka pada $f$ tinggi, $C$ menghubungsingkatkan); (d) lolos

\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:4}
$4E/n\pi$: \qty{6.37}{V} pada \qty{1}{kHz}, \qty{2.12}{V} pada \qty{3}{kHz},
\qty{1.27}{V} pada \qty{5}{kHz}. Gelombang perseginya antisimetrik
terhadap pergeseran setengah periode ($s(t + T/2) = -s(t)$), sifat yang
tak dimiliki harmonik genap.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:5}
Pada \qty{100}{Hz} dan harmonik bergunanya, $\omega \ll \omega_c$:
$\underline H \approx j\omega/\omega_c$, yaitu turunan yang dibagi
$\omega_c$. Kemiringan segitiganya $\pm 4Ef = \pm\qty{400}{V/s}$, sehingga
keluarannya adalah gelombang persegi beramplitudo $400/(2\pi \times 10^4) =
\qty{6.4}{mV}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:6}
$\omega \gg \omega_c$: $u_s \approx (1/\tau)\int u_e\,\dd t$, $\tau =
1/(2\pi \times 10) = \qty{16}{ms}$. Integral $\pm E$ adalah segitiga
yang naik sebesar $ET/2$ tiap setengah periode: ayunannya $ET/2\tau$, \omterm{def:b1:wave-propagation:signal}{amplitudonya}
$ET/4\tau = 1.0 \times 10^{-3}/(4 \times 0.016) = \qty{16}{mV}$ di sekitar
nol.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:7}
$f_0 = \qty{5.03}{kHz}$. $Q = \sqrt{L/C}/R = 1/\sqrt2$: $R = \sqrt2 \times
316 = \qty{447}{\ohm}$. Pada $f_0$: \qty{-3}{dB}; pada $10f_0$: $1/\sqrt{1 +
10^4} \to \qty{-40}{dB}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:8}
$G = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ dengan $x = f/3$: \qty{1}{kHz}: $x = 1/3$,
$Q(x - 1/x) = -53$, $G = 0.019$; \qty{3}{kHz}: $G = 1$; \qty{5}{kHz}:
$x = 5/3$, $21$, $G = 0.047$. Keluarannya: \qty{2.1}{V} pada \qty{3}{kHz},
dengan \qty{0.12}{V} pada \qty{1}{kHz} dan \qty{0.06}{V} pada \qty{5}{kHz} ---
sinusoid \qty{3}{kHz} yang nyaris murni.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:9}
Simpul: dengan $a = jRC\omega$, kapasitor pertamanya melihat $R$ di
hulunya dan $R + 1/jC\omega$ di hilirnya: $\underline H = 1/(1 + 3a + a^2)$, yaitu
$1/[1 - (\omega/\omega_c)^2 + 3j\omega/\omega_c]$. Pada $\omega_c$: $1/3j$,
$G = 1/3$, \qty{-9.5}{dB}; hasil kali kedua \omterm{def:b1:filters-transfer-functions:H}{penguatan} terpisahnya akan
$1/2$, \qty{-6}{dB}. $f_c = \qty{1.0}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:10}
Di bawah $\omega_1$: \qty{0}{dB}; antara $\omega_1$ dan $\omega_2$: naik
\qty{20}{dB/dekade}; di atas $\omega_2$: mendatar pada $20\log(\omega_2/\omega_1) =
\qty{40}{dB}$. Fasenya $\arctan(\omega/\omega_1) - \arctan(\omega/\omega_2)$:
pada $\omega = 10^3$, $\arctan 10 - \arctan 0.1 = 84^\circ - 6^\circ =
78^\circ$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:11}
Jauh di atas \omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya}, $G \approx f_c/f$; nisbah riaknya
setelah ditapis adalah
\[
\frac{(4E/3\pi)(f_c/100)}{2E/\pi} = \frac{2}{3}\,\frac{f_c}{100} < 0.01,
\]
sehingga $f_c < \qty{1.5}{Hz}$ dan $\tau = 1/2\pi f_c > \qty{0.1}{s}$. Bebannya sendiri menjadi
$R$: kapasitor yang dipasang melintasinya tak memakan biaya apa pun pada
jatuh \omterm{def:b1:dc-circuits:voltage}{tegangan} dan dayanya, sedangkan \omterm{def:b1:dc-circuits:resistor}{resistor} seri akan memboroskan keduanya.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:12}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 1.6\times10^5 \times 10^{-7}) =
\qty{10}{Hz}$, $\tau = RC = \qty{16}{ms}$. Selama sebuah puncak datar
kapasitornya mengisi lewat $R$: keluarannya meluruh sebagai $\eu^{-t/\tau}$
dari tiap tepinya. Kemerosotan sepanjang $T/2$: $1 - \eu^{-T/2\tau}$: \qty{50}{Hz}
($T/2 = \qty{10}{ms}$): $1 - \eu^{-0.63} = 47\%$; \qty{1}{kHz}
($T/2 = \qty{0.5}{ms}$): $3\%$.
\end{solution}

\begin{solution}{pb:b1:filters-transfer-functions:1}
\textbf{1.} $\underline H = \underline U_{\text{penggerak}}/\underline U_e$;
$G_{\mathrm{dB}} = 20\log\abs{\underline H}$.

\textbf{2.} $\underline H_L = R/(R + jL\omega) = 1/(1 + j\omega/\omega_c)$,
$\omega_c = R/L$.

\textbf{3.} $L = R/\omega_c = 8.0/(2\pi \times 2000) = \qty{0.64}{mH}$.

\textbf{4.} $\underline H_H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$,
$\omega_c = 1/RC$; $C = 1/(R\omega_c) = \qty{9.9}{\micro F} \approx
\qty{10}{\micro F}$.

\textbf{5.} Pada $f_c$: \qty{-3}{dB} masing-masing. \qty{200}{Hz} ($x = 0.1$): woofer
\qty{-0.04}{dB}, tweeter \qty{-20}{dB}. \qty{20}{kHz} ($x = 10$): woofer
\qty{-20}{dB}, tweeter \qty{-0.04}{dB}.

\textbf{6.} $G_L^2 + G_H^2 = (1 + x^2)/(1 + x^2) = 1$: dengan beban yang
sama kedua dayanya berjumlah sama dengan daya yang akan diambil satu
penggerak \qty{8}{\ohm} --- dibagi, bukan hilang.

\textbf{7.} $-45^\circ$ (woofer) dan $+45^\circ$ (tweeter): terpisah
$90^\circ$; pada $f_c$ kedua kerucutnya tidak bergerak bersama.

\textbf{8.} \qty{\pm 6}{dB} tiap oktaf (\qty{20}{dB} tiap dekade).

\textbf{9.} Pembagi antara $jL\omega$ dan $R \parallel 1/jC\omega =
R/(1 + jRC\omega)$: $\underline H = R/[R + jL\omega(1 + jRC\omega)] =
1/(1 - LC\omega^2 + jL\omega/R)$.

\textbf{10.} $\omega_0 = 1/\sqrt{LC}$; $L\omega/R = x/Q$ memberi $Q =
R/L\omega_0 = R\sqrt{C/L}$.

\textbf{11.} $\abs{\underline H}^{-2} = (1 - x^2)^2 + x^2/Q^2 = 1 + x^4 +
x^2(1/Q^2 - 2) = 1 + x^4$ untuk $Q^2 = \tfrac12$; pada $x = 1$, $G = 1/\sqrt2$.

\textbf{12.} $L = R/(Q\omega_0) = 8\sqrt2/(1.257\times10^4) = \qty{0.90}{mH}$;
$C = 1/(L\omega_0^2) = \qty{7.0}{\micro F}$.

\textbf{13.} $x = 2$: $1/\sqrt{17}$, \qty{-12.3}{dB}; $x = 4$: $1/\sqrt{257}$,
\qty{-24.1}{dB}: \qty{12}{dB} tiap oktaf.

\textbf{14.} Menukar $L$ dan $C$ menukar $x \leftrightarrow 1/x$ (dan
$Q$ tetap): $\underline H_H = -x^2/(1 - x^2 + jx/Q)$.

\textbf{15.} Pada $x = 1$: $\underline H_L = Q/j = -jQ$ ($-90^\circ$),
$\underline H_H = jQ$ ($+90^\circ$): kedua penggeraknya berlawanan fase
pada frekuensi silangnya dan akan saling meniadakan secara akustik;
tweeternya diawatkan dengan kutubnya dibalik.

\textbf{16.} Harmonik di atas \qty{2}{kHz}: $n \geq 10$ (\qty{2.2}{kHz}
ke atas) sebagian besar pergi ke tweeternya.

\textbf{17.} $9 \times 220 = \qty{1980}{Hz} \approx f_c$: terbagi rata,
\qty{-3}{dB} untuk masing-masing.

\textbf{18.} Woofer ($G_L$): $x = 0.5$: $0.89$; $x = 1.5$: $0.55$; $x = 2.5$:
$0.37$. Tweeter ($G_H$): $0.45$, $0.83$, $0.93$.

\textbf{19.} Tweeternya: kapasitor serinya menghalangi arus searah.
Woofernya menerima simpangan tetapnya (\qty{1}{V} pada \qty{8}{\ohm}:
\qty{125}{mA}, \qty{0.125}{W} kalor dan kerucut yang tergeser).

\textbf{20.} Orde pertama, $x = 0.25$: $G_H = 0.25/\sqrt{1.0625} = 0.24$
(\qty{-12}{dB}), bagian dayanya $6\%$. Orde kedua: $0.0625/\sqrt{1 +
0.0039} = 0.06$ (\qty{-24}{dB}), $0.4\%$. Tweeter berdaya beberapa watt
menerima $6\%$ dari sebuah nada rendah yang keras dengan rancangan orde
pertama --- ia bisa terbakar.

\textbf{21.} $\underline Z = 8 + jL_{vc}\omega$: \qty{2}{kHz}: $8 + 6.3j$,
$\abs{\underline Z} = \qty{10.2}{\ohm}$; \qty{8}{kHz}: $8 + 25.1j$,
\qty{26.3}{\ohm}.

\textbf{22.} $\underline H = \underline Z/(\underline Z + jL\omega)$:
\qty{2}{kHz}: $10.2/\abs{8 + 14.3j} = 10.2/16.4 = 0.62$ (\qty{-4.2}{dB});
\qty{8}{kHz}: $26.3/\abs{8 + 57.3j} = 0.45$ (\qty{-6.9}{dB}) alih-alih
\qty{-12.3}{dB}: \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} kumparannya yang menaik melawan kumparan
serinya, dan landaiannya mendatar menjadi beberapa dB tiap oktaf.

\textbf{23.} \omterm{def:b1:sinusoidal-impedance:impedance}{Admitansinya}: $1/(R + jL_{vc}\omega) + jC_z\omega/(1 + jRC_z\omega)$;
dengan $C_z = L_{vc}/R^2$ suku keduanya menjadi $jL_{vc}\omega/[R(R +
jL_{vc}\omega)]$, dan jumlahnya $(R + jL_{vc}\omega)/[R(R + jL_{vc}\omega)]
= 1/R$. $C_z = 0.50\times10^{-3}/64 = \qty{7.8}{\micro F}$.

\textbf{24.} $U = \sqrt{PR} = \sqrt{400} = \qty{20}{V}$ rms. Pada $f_c$ tiap
penggeraknya memperoleh separuh: \qty{25}{W}. Pada \qty{500}{Hz} tweeternya
memperoleh $6\%$: \qty{3}{W}.

\textbf{25.} Orde pertama: dua komponen, saling melengkapi dalam daya,
kemiringan \qty{6}{dB/okt} yang landai, $90^\circ$ antara penggeraknya,
dan nada rendah bocor ke tweeternya. Orde kedua: empat komponen,
\qty{12}{dB/okt}, $180^\circ$ pada frekuensi silangnya (balikkan
tweeternya), perlindungan tweeter yang jauh lebih baik. Bagaimanapun
juga, induktansi woofernya harus dijinakkan jaringan Zobel agar
rancangannya bertahan.
\end{solution}
