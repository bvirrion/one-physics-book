\section*{Bab \ref{ch:b1:work-and-energy} --- Usaha, Energi, dan Energi Potensial}

\begin{solution}{exo:b1:work-and-energy:1}
Gaya angkatnya $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$; beratnya
$-\qty{589}{J}$; dayanya $589/5.0 = \qty{118}{W}$. Gesekan: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$; $d = E_k/F =
\qty{55}{m}$; melipatduakan $v$ melipatempatkan $E_k$ dan $d$: \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$; $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$: $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
Tegangannya: $T - mg = mv^2/\ell = mg$, $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
Gesekannya berusaha $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$, sehingga
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$; $35\%$ hilang.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
$E_m = \tfrac12 mv^2 - GMm/R \geq 0$ agar bendanya mencapai tak hingga
(tempat $E_p = 0$ dan $v \to 0$): $v_e = \sqrt{2GM/R}$, yaitu
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
Massa jenis sama, $M \propto R^3$: $v_e \propto R$, jadi \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
Titik terendah: $mg(\ell_0 + x) = \tfrac12 kx^2$, $25x^2 - 687x - 13734 = 0$,
$x = \qty{41}{m}$; jatuh totalnya \qty{61}{m}; $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (bersihnya \qty{2}{g} ke atas). Lajunya terbesar di tempat gaya
totalnya lenyap, $kx = mg$: $x = \qty{14}{m}$, \qty{34}{m} di bawah titik lompatnya.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
Gravitasi: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$;
hambat: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$; totalnya \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$: $x = 0, \pm a$. $E_p'' = E_0(12x^2/a^4 -
4/a^2)$: $-4E_0/a^2$ di $0$ (maksimum yang goyah, $E_p = 0$), $+8E_0/a^2$
di $\pm a$ (minimum yang mantap, $E_p = -E_0$): sumur sedalam $E_0$;
$\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$: sebuah elips bersetengah sumbu $x_{\max}
= \sqrt{2E/k}$ dan $\dot x_{\max} = \sqrt{2E/m}$. Geraknya harmonik
dengan periode $2\pi\sqrt{m/k}$ berapa pun \omterm{def:b1:wave-propagation:signal}{amplitudonya} (keisokronan),
sehingga setiap elipsnya memakan waktu yang sama.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
Energi: $v^2 = 2gR(1 - \cos\theta)$. Radial: $mg\cos\theta - N = mv^2/R$,
sehingga $N = mg(3\cos\theta - 2)$, yang nol pada $\cos\theta = 2/3$ ($\theta =
48^\circ$): tingginya $2R/3$, lajunya $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$: $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
Menjadi kalor pada sentuhan ban--jalan (dan sebuah bekas gelincir). Roda
yang menggelinding memakai gesekan statis, $\mu_s > \mu_d$: gaya rem yang
lebih besar, jarak yang lebih pendek, dan kemudi yang tetap terjaga.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} $r \to 0$: $+\infty$ (suku $r^{-12}$); $r \to \infty$:
$0^-$. Dinding yang curam, sebuah sumur, dan ekor yang panjang.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ jika dan hanya jika
$(\sigma/r)^6 = 1$: $r = \sigma$, $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$: $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$, minimum pada $u = 1$ ($r = \sigma$) dengan nilai $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$:
positif (menolak) untuk $r < \sigma$, negatif (menarik) di luarnya.

\textbf{5.} $(1/1.2)^6 = 0.335$, $(1/1.2)^{12} = 0.112$: $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. $(1/1.2)^{13} = 0.0935$,
$(1/1.2)^7 = 0.279$: $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$,
dekat dengan \qty{431}{kJ/mol} yang terukur.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$; pada
$\sigma$: $72\epsilon/\sigma^2 > 0$: mantap.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$:
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$; $f = \qty{2.2e14}{Hz}$; $\lambda = c/f = \qty{1.4}{\micro m}$;
$1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} Faktor $2.4$ terlalu tinggi: dinding modelnya terlalu curam
(inti $r^{-12}$ itu pengganti yang kasar bagi sebuah ikatan kovalen),
tetapi \omterm{def:b1:units-dimensions:oom}{orde besarnya} --- getaran inframerah pada $10^{14}$ Hz --- sudah
tepat, dan metodenya (kelengkungan sumurnya) persis metode yang dipakai
dengan potensial yang lebih baik.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$: $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$: ikatannya
nyaris hanya menggigil.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$: kurva tertutup di sekitar $(\sigma, 0)$ ---
getaran, terikat; $E_m > 0$: kurva terbuka --- atomnya berpisah;
$E_m = 0$: separatriksnya, yaitu atom yang tepat dapat mencapai tak hingga.

\textbf{15.} $u^2 - 2u = -\tfrac12$ dengan $u = (\sigma/r)^6$: $u = 1 \pm 1/\sqrt2
= 1.71$ atau $0.293$; $r = \sigma u^{-1/6}$: $0.915\sigma$ dan $1.23\sigma$.

\textbf{16.} Titik tengahnya $1.07\sigma > \sigma$: dinding luarnya lebih
lunak daripada dinding dalamnya, sehingga atomnya menghabiskan lebih
banyak ruang di luar; jarak rata-rata waktunya tumbuh bersama energinya
--- ikatan yang dipanaskan memanjang, dan zat padat memuai.

\textbf{17.} Lebih panjang: gaya pemulih pada sisi luarnya yang lunak
lebih lemah daripada yang diandaikan hampiran harmoniknya, sehingga
kembalinya memakan lebih banyak waktu.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$; sebagai \omterm{thm:b1:work-and-energy:ke}{energi
kinetik} di $\sigma$: $v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}$
$= \qty{2.0e4}{m/s}$.

\textbf{19.} Kekekalan energi: $E_m > 0$ tetap $> 0$, kurvanya terbuka,
dan atomnya terbang berpisah setelah satu perjumpaan. Benda ketiga
(molekul lain, sebuah dinding, sebuah foton) harus membawa kelebihannya pergi.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$, positif: tarikannya menarik
atomnya masuk, jadi memacu.

\textbf{21.} $W_{\text{luar}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$, $(1/0.9)^7 = 2.09$: $F = +22\epsilon/\sigma
= \qty{+120}{nN}$, sepuluh kali tarikannya pada $1.2\sigma$: menekan
sebuah ikatan sebesar $10\%$ jauh lebih mahal daripada meregangkannya sebesar $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. Gas sungguhan tercerai-berai jauh di bawahnya karena
energinya terdistribusi: sebagian molekulnya membawa berkali-kali $k_BT$
(\cref{ch:b1:kinetic-theory}).

\textbf{24.} Yang bertahan: kesetimbangan di minimumnya, $k = E_p''$ di
minimumnya, energi disosiasi $=$ kedalamannya, \omterm{def:b1:work-and-energy:turning}{titik balik} dari
$E_p = E_m$, dan ketaksetangkupan $\Rightarrow$ pemuaian. Yang berubah:
nilai $k$, nilai frekuensinya dan nilai \omterm{def:b1:work-and-energy:turning}{titik baliknya}.

\textbf{25.} Kesetimbangan: $E_p' = 0$; getaran: $\omega_0^2 = E_p''/m$;
lepas: $E_m$ dibandingkan dengan bibir sumurnya.
\end{solution}
