\begin{solution}{exo:b1:optical-instruments:6}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Pajanan $\propto N^2$: $(8/2)^2 = 16$ kali
lebih lama, $16/500 \approx 1/30$~\unit{s}. Ia membeli \omterm{def:b1:optical-instruments:camera}{kedalaman medan}
(dan lebih sedikit aberasi) dengan harga kabur gerak atau sebuah tripod.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$; tajam dari $H/2 =
\qty{1.9}{m}$ sampai tak hingga. Disetel lebih dulu pada $H$, \omterm{def:b1:optical-instruments:camera}{kameranya}

\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupil keluarnya} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$,
seluruh cahayanya masuk ke \omterm{def:b1:optical-instruments:eye}{mata}. $7\times50$: \qty{7.1}{mm}, cocok dengan
pupil malamnya, dan $(50/30)^2 = 2.8$ kali lebih banyak cahaya yang
terkumpul: itulah pasangan untuk senja hari. $20\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupil
keluarnya} \qty{1.5}{mm}, bayangannya pudar, medannya sempit dan getar
tangan diperbesar dua puluh kali.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
\omterm{def:b1:mirrors-thin-lenses:lens}{Lensa} kontak: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$. Kacamata:
\omterm{def:b1:optical-instruments:punctum}{titik jauhnya} $25 - 1.5 = \qty{23.5}{cm}$ dari \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya}, sehingga $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- lebih kuat. \omterm{def:b1:mirrors-thin-lenses:lens}{Lensanya}
harus membentuk bayangan di \omterm{def:b1:optical-instruments:punctum}{titik jauh} \omterm{def:b1:optical-instruments:eye}{matanya}, yang jaraknya dari
\omterm{def:b1:mirrors-thin-lenses:lens}{lensa} bergantung pada di mana \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} duduk.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afokal: $F_1' = F_2$, pisahnya $f_1' + f_2' = \qty{20}{cm}$ (lebih pendek
daripada \qty{40}{cm} milik teropong Kepler dengan $G$ yang sama).
$G = -30/(-10) =
+3$: tegak. \omterm{prop:b1:optical-instruments:telescope}{Pupil keluarnya} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$: maya, di dalam tabungnya, tempat
\omterm{def:b1:optical-instruments:eye}{matanya} tak bisa berada; \omterm{def:b1:optical-instruments:eye}{mata} di belakang \omterm{def:b1:optical-instruments:microscope}{okulernya} hanya menangkap
berkas yang kebetulan sampai kepadanya --- medan yang sempit, keluhan
klasik tentang teropong opera.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$; $\Delta m = 2.5\log_{10}820 = 7.3$: magnitudo
pembatasnya $13.3$. Fluksnya $\propto 1/r^2$, sehingga lampunya terlihat
$\sqrt{820} = 29$ kali lebih jauh.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$: pisahnya $f_1' + f_2' = \qty{1225}{mm}$
(\omterm{def:b1:optical-instruments:microscope}{okuler} \qty{25}{mm}) atau \qty{1206}{mm} (\omterm{def:b1:optical-instruments:microscope}{okuler} \qty{6}{mm}).

\textbf{2.} $L_1$: $\overline{O_1A_1} = f_1'$ (bayangan di $F_1'$). $L_2$:
$\overline{O_2A_1} = -f_2'$, sehingga $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$: bayangannya di tak hingga.

\textbf{3.} Berkasnya memfokus di bidang fokus bersamanya pada ketinggian $h =
f_1'\alpha$ (sinar melalui $O_1$); dari sana sinar yang melalui $O_2$
menetapkan arah keluarnya, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ dan $-200$; Bulan: $25^\circ$ dan $104^\circ$.

\textbf{5.} Atas dan bawah tak bermakna di langit; teropong pengamat
menambahkan sepasang prisma penegak (atau relai \omterm{def:b1:mirrors-thin-lenses:lens}{lensa}).

\textbf{6.} Di bidang fokus $L_1$, bergaris tengah $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} Benda $O_1$ pada $\overline{O_2O_1} = -(f_1' + f_2')$:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$;
perbesarannya $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
garis tengahnya $D/\abs G$.

\textbf{8.} \qty{25}{mm}: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, garis tengahnya $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}:
\qty{6.0}{mm} di belakang, garis tengahnya \qty{0.75}{mm}.

\textbf{9.} Setiap sinar yang masuk ke \omterm{def:b1:optical-instruments:microscope}{objektifnya} melewati \omterm{prop:b1:optical-instruments:telescope}{pupil
keluarnya}: \omterm{def:b1:optical-instruments:eye}{mata} yang diletakkan di sana menerima seluruh medannya. Lebih
ke belakang ia hanya mencegat berkas pusatnya: medannya menyusut menjadi lubang kunci.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$;
\omterm{def:b1:optical-instruments:eye}{mata} telanjang $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$; \omterm{def:b1:optical-instruments:microscope}{okuler} \qty{25}{mm}
($48$) yang paling dekat, sedikit di bawahnya; \omterm{def:b1:optical-instruments:microscope}{okuler} \qty{6}{mm} ($200$)
tiga kali di atasnya.

\textbf{14.} Pada $G = 200$ kekaburan difraksinya membentang $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$: tampak berbulu. Cahaya
Bulan tersebar pada luas $(200/48)^2 = 17$ kali lebih besar di
\omterm{def:b1:optical-instruments:eye}{retinanya}: lebih pudar.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Di atas itu, keburaman atmosfer yang membatasi \omterm{rem:b1:optical-instruments:abbe}{daya
urai} dari permukaan bumi (kecuali dengan optika adaptif); $D$ yang lebih
besar tetap mengumpulkan lebih banyak cahaya, sehingga mencapai objek
yang lebih redup.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$: magnitudo $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
$460$ kali milik \omterm{def:b1:optical-instruments:eye}{mata} telanjang. Bayangan \omterm{def:b1:optical-instruments:eye}{retinanya} $\abs G^2$ kali
lebih luas: nisbah kecerahan permukaannya $460/G^2 = 0.20$ ($G = 48$),
$0.012$ ($G = 200$) --- sama dengan $(D/\abs G)^2/d_p^2$, yaitu kuadrat
nisbah \omterm{prop:b1:optical-instruments:telescope}{pupil keluar} terhadap pupil \omterm{def:b1:optical-instruments:eye}{mata}.

\textbf{19.} Untuk benda yang terurai, cahaya yang terkumpul tumbuh
sebagai $D^2$ tetapi luas bayangannya sebagai $G^2 \geq (D/d_p)^2$:
kecerahan permukaannya paling banter menyamai nilai \omterm{def:b1:optical-instruments:eye}{mata} telanjang
(\omterm{prop:b1:optical-instruments:telescope}{pupil keluar} $=$ pupil \omterm{def:b1:optical-instruments:eye}{mata}). Bintang tak terurai: seluruh cahayanya
mendarat di satu bintik difraksi, sehingga kecerahannya tumbuh sebagai $(D/d_p)^2$.

\textbf{20.} Titik bersudut $\beta$ dibayangkan pada ketinggian $f_1'\beta$
di bidang fokusnya; diafragmanya meloloskan $\abs{f_1'\beta} \leq \phi/2$:
medan penuhnya $\phi/f_1'$ (sudut kecil).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$: seluruh Bulan
muat. $5/1200 = 0.24^\circ$: tidak muat.

\textbf{22.} Medan semunya $\phi/f_2'$: $20/25 = \qty{0.8}{rad} = 46^\circ$;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$: $3420/15 = \qty{230}{s}$, hampir
empat menit; $0.24^\circ = 864''$: \qty{58}{s}.

\textbf{24.} \omterm{def:b1:optical-instruments:microscope}{Okuler} \qty{25}{mm} ($G = 48$, dekat $G_r$, terang, seluruh
Bulan) untuk cakramnya; \omterm{def:b1:optical-instruments:microscope}{okuler} \qty{6}{mm} ($G = 200$) untuk menengok
kawah lebih dekat, lebih pudar dan lebih kabur. Angka alat itu: kawah
sampai sekecil \qty{1.7}{km}.

\textbf{25.} $f_1'$ dan $D$ yang sama: $G$, medan, \omterm{prop:b1:optical-instruments:telescope}{pupil keluar}, \omterm{rem:b1:optical-instruments:abbe}{daya
urai} dan cahaya yang sama pula (dikurangi penghalang pusat, bila ada).
Yang diperoleh: tak ada aberasi warna. Perubahan praktisnya: fokusnya
terletak di depan cerminnya, sehingga cermin sekunder yang kecil harus
melipat berkasnya ke luar menuju \omterm{def:b1:optical-instruments:microscope}{okulernya}.
\end{solution}
