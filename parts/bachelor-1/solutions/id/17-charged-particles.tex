\section*{Bab \ref{ch:b1:charged-particles} --- Partikel Bermuatan dalam Medan E dan B}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$: elektron
\qty{1.9e7}{m/s}; proton \qty{4.4e5}{m/s} ($\sqrt{1836}$ kali lebih lambat).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. Pada $30^\circ$: sebuah heliks berjari-jari $R\sin 30^\circ
= \qty{10}{m}$ dan berlangkah $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, menghanyut
menyusuri garis medannya.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Ion yang lebih cepat: gaya magnetnya
menang, mereka melengkung ke sisi magnetnya; muatan berlawanan pada
kelajuan yang sama: kedua gayanya membalik dan tetap berimbang --- selektornya
buta terhadap tanda dan terhadap massanya.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; pembelokannya
$0.30 \times 0.10 = \qty{3.0}{cm}$; kepekaannya \qty{0.30}{mm/V}. Elektron yang
lebih cepat lebih singkat berada di antara kepingnya: $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; pemisahan $2\Delta R =
\qty{1.2}{cm}$ di antara kedua titik jatuhnya (jari-jarinya berselisih \qty{5.8}{mm},
garis tengahnya dua kali itu).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; \qty{100}{keV} tiap putaran: $270$ putaran; waktunya $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
periode $2\pi m/eB = \qty{18}{ns}$, langkah $v_\parallel T = \qty{1.5}{cm}$:
kira-kira $64$ putaran tiap meter.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: di bawah satu mikrovolt. Dengan $n$ sepuluh juta kali lebih
kecil, semikonduktor memberikan milivolt --- penginderaan yang dapat dipakai.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; kalikan secara skalar dengan $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ berapa pun $\vect B(\vect r, t)$, asalkan tak ada $\vect E$. Arah
kecepatannya, \omterm{def:b1:point-kinematics:frenet}{jari-jari kelengkungannya}, dan langkahnya semuanya dapat
berubah; kelajuannya dan \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} (\qty{10}{keV}) tidak.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. Dengan $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- rotasi
siklotron murni dari $\vect w$ pada $\omega_c$. Mulai dari keadaan diam,
$\vect w(0) = -(E/B)\vect e_x$: $\vect w$ berputar dengan norma tetap $E/B$, sehingga
kecepatannya adalah hanyutan $E/B$ ditambah vektor berputar bernorma sama ---
sebuah sikloida, dengan titik runcing di tempat $\vect v = \vect 0$. Elektron: hanyutan $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, tinggi lengkungnya $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (kira-kira $15^\circ$):
pembelokannya $\approx 0.30 \times 0.27 = \qty{8}{cm}$ untuk satu militesla
yang sederhana. Sudut yang besar dengan keping akan menuntut kilovolt dan
keping yang panjang; sebuah kumparan membelokkan berkasnya sampai puluhan
derajat tanpa \omterm{def:b1:dc-circuits:voltage}{tegangan} sama sekali.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: tak ada daya, kelajuannya
tetap.

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, bebas dari
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ tiap penyeberangan, \qty{100}{keV} tiap putaran.

\textbf{5.} Celahnya harus berpolaritas mempercepat tiap kali protonnya
tiba, berselang setengah periode; bila frekuensinya meleset, fasenya
tergelincir dan tendangannya lama-lama justru memperlambat.

\textbf{6.} Setelah satu putaran $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; setelah $n$ putaran $E_k =
n \times \qty{100}{keV}$ dan $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ putaran; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} Tidak: $E_k$ hanya bergantung pada $B$ dan $R_{\max}$; $U$ menetapkan
jumlah putarannya dan waktunya.

\textbf{11.} \omterm{def:b1:point-kinematics:frame}{Lintasannya} \qty{560}{m}: dituntut $\lambda \gtrsim \qty{1}{km}$, yakni
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
hampa yang tinggi.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; periodenya tumbuh sebesar $2.9\%$.
Sepanjang $270$ putaran gelincirnya menumpuk sampai beberapa periode ---
putaran terakhirnya akan keluar fase; dalam praktiknya $B$ dan fase
suntikannya disetel, dan \qty{27}{MeV} sudah dekat batas mesin sederhananya.

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; $R =
\qty{1.1}{m}$ pada \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ tumbuh sebesar $24\%$ dari pusat ke tepi:
frekuensi yang tetap kehilangan langkahnya sama sekali.

\textbf{15.} Sinkrosiklotron: turunkan frekuensinya seiring bertambahnya
energi gumpalannya --- hanya satu gumpalan sekali jalan, arus rata-ratanya
kecil. Isokron: bentuklah $B(R) \propto \gamma$ sehingga periodenya tinggal
tetap --- berkas yang sinambung, pilihan modern.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, sedikit lebih besar daripada
\qty{1.1}{m} yang tak relativistik.

\textbf{17.} Impuls melintangnya $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; sudutnya $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: tak ada harapan untuk mengemudi.

\textbf{18.} Secara magnetik: $R = p/eB = \qty{2.3}{m}$, sudutnya $L/R = \qty{0.43}{rad}
\approx 25^\circ$: seribu kali lebih besar --- maka magnetlah pilihannya.

\textbf{19.} Energi tiap proton $\qty{3.68e-11}{J}$; dosisnya $2.0 \times 0.50 =
\qty{1.0}{J}$: $2.7\times10^{10}$ proton.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \qty{6.2e9}{proton/s}$: kira-kira
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
sebuah lampu malam, disampaikan tepat di tempat yang dikehendaki.

\textbf{22.} $\qty{58}{MeV}$ dalam \qty{1}{cm}: kira-kira \qty{6}{keV/\micro m} ---
ionisasi yang padat menyusuri \omterm{def:b1:point-kinematics:frame}{lintasan} terakhirnya.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionisasi tiap proton ---
kerusakan molekuler yang membunuh selnya.

\textbf{24.} Sinar-X menyimpan energinya di sepanjang \omterm{def:b1:point-kinematics:frame}{lintasannya}, terbanyak
di dekat pintu masuknya; proton menyimpan sebagian besarnya di ujung
jangkauannya, menyisihkan jaringan di depannya dan tak meninggalkan apa pun di belakang.

\textbf{25.} $R = mv/eB$ dan $T = 2\pi m/eB$ (lingkarannya), $E_k = (eBR)^2/2m$
(energinya), $E_k/2eU$ (putarannya), dosis $=$ (proton $\times$ energi)/massa
dan arus $= e \times$ laju (waktunya).
\end{solution}
