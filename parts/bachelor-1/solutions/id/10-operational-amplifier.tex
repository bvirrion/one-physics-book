\section*{Bab \ref{ch:b1:operational-amplifier} --- Penguat Operasional}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$; $V_s = \qty{5.0}{V}$; jenuh pada $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$; \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} masukannya $R_1 = \qty{10}{k\ohm}$; $i = 1.0/10^4 =
\qty{0.10}{mA}$; $V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
Tanpa pengikut: pembagi $1/(1 + 10) = 0.091$, \qty{91}{mV}. Dengan
pengikutnya: \qty{1.0}{V} --- pengikutnya tak menarik apa pun dari
sensornya dan memasok \qty{1}{mA} yang diperlukan bebannya.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$; di $V_-$: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
sehingga $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$; dengan $V_- = V_+$:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. Angkanya: $10 \times 0.05 = \qty{0.50}{V}$;
\qty{1}{V} yang bersama tak menyumbang apa pun --- penolakan modus bersama.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: kemiringannya $-1000 \times 1.0 = \qty{-1000}{V/s}$;
jenuh setelah $15/1000 = \qty{15}{ms}$. Gelombang persegi: segitiga
beramplitudo $ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(ditambah tetapan apa pun yang ditinggalkan muatan awalnya).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: penguatan searahnya $-100$;
\omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya} $1/(2\pi R'C) = \qty{1.6}{Hz}$; pada \qty{1}{kHz},
$R'C\omega = 628$, $G =
100/628 = 0.16$ --- yaitu nilai pengintegralnya $1/RC\omega$. Ia
mengintegralkan untuk $f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: $R_2 = \qty{10}{k\ohm}$; $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
Dengan $\mu$ berhingga: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, sehingga
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. \omterm{thm:b1:sinusoidal-impedance:resonance}{Lebar pitanya} $f_T/G =
\qty{10}{kHz}$; pengikutnya: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
Tiap lonjakan derau yang melintasi \qty{2.0}{V} membalikkan
keluarannya: sebuah letupan penyaklaran cepat di sekitar
persilangannya. Obatnya: histeresis yang lebih lebar daripada deraunya,
yaitu sebuah \omterm{prop:b1:operational-amplifier:schmitt}{pemicu Schmitt}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: ambangnya $\pm\qty{1.5}{V}$.
Keluarannya $+\qty{15}{V}$ selagi $V_e$ naik sampai \qty{1.5}{V},
berbalik ke $-\qty{15}{V}$, tinggal di sana sampai $V_e$ jatuh di bawah
\qty{-1.5}{V}, lalu berbalik lagi: sebuah gelombang persegi berfrekuensi
segitiganya, dengan tepinya pada persilangan ambangnya.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
Dengan $V_s = +V_{\mathrm{sat}}$, $V_C$ mengisi dari $-\beta V_{\mathrm{sat}}$
menuju $+V_{\mathrm{sat}}$: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, yang mencapai $+\beta V_{\mathrm{sat}}$ ketika $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$: setengah periodenya $RC\ln\frac{1 + \beta}{1 - \beta}$,
jadi $T$. $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$. $V_s$: gelombang persegi
$\pm\qty{15}{V}$; $V_C$: busur eksponensial antara $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} Dua pembagi: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone});
pengawatannya menetapkan tandanya.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ selalu; $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ pada skala penuh.
\qty{2.5}{V} yang dibagi keduanya adalah \omterm{def:b1:dc-circuits:voltage}{tegangan} modus bersamanya.

\textbf{4.} Tiap titik tengahnya: $1 \parallel 1 = \qty{500}{\ohm}$; antara
kedua titik tengahnya, \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$: rugi $9\%$.

\textbf{6.} \omterm{def:b1:sinusoidal-impedance:impedance}{Impedansi} masukan tak hingga (tak ada arus dari jembatannya,
sehingga titik tengahnya menjaga potensial rangkaian terbukanya) dan
\omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} keluaran nol (tingkat berikutnya dapat menarik apa yang diperlukannya).

\textbf{7.} $V_A$ dan $V_B$ persis; arusnya nol.

\textbf{8.} Seperti dalam \cref{prop:b1:operational-amplifier:sumdiff}: $V_+ =
V_BR_2/(R_1 + R_2)$, \omterm{thm:b1:dc-circuits:kirchhoff}{hukum simpul} di $V_-$, $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$: misalnya \qty{1}{k\ohm} dan \qty{9}{k\ohm}.
Penguatan totalnya $1000$; skala penuhnya $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} Bila tersesuaikan: \qty{2.5}{V} yang bersama meniadakan
dengan tepat (hanya $V_B - V_A$ yang bertahan). Ketaksesuaian $1\%$
meloloskan sekitar $1\%$ darinya: galat teracu masukan berorde
\qty{25}{mV}, sepuluh kali \omterm{def:b1:wave-propagation:signal}{sinyal} skala penuhnya --- keempat \omterm{def:b1:dc-circuits:resistor}{resistornya}
harus disesuaikan lebih baik daripada $0.01\%$ (atau \omterm{def:b1:dc-circuits:voltage}{tegangan} modus
bersamanya disingkirkan lebih dahulu).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, empat puluh persen skala
penuhnya: tak terpakai tanpa peneraan simpangan, penguat bersimpangan
rendah (jenis cincang), atau kalibrasi nol tanpa beban di gelagarnya.

\textbf{13.} Tingkat 1: $f_T/100 = \qty{10}{kHz}$; tingkat 2: \qty{100}{kHz};
jauh di atas \qty{5}{Hz}.

\textbf{14.} Yang diperlukan $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
dua ribu kali di bawah \omterm{rem:b1:operational-amplifier:real}{laju ayunnya}: tidak.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$: $G = 1/\sqrt{26}$, \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$; tundaannya $\varphi/\omega =
\qty{15}{ms}$ --- bukan apa-apa terhadap lewatnya sebuah truk.

\textbf{18.} Tak ada pembebanan pada tingkat berikutnya (\omterm{def:b1:sinusoidal-impedance:impedance}{impedansi}
keluaran nol) dan tersedia penguatan; selain itu tak ada pelemahan
\omterm{def:b1:wave-propagation:signal}{sinyal} yang dikehendaki di bawah \omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya}.

\textbf{19.} \qty{-28}{dB} (penguatannya berkalian, \omterm{def:b1:filters-transfer-functions:bode}{desibelnya}
berjumlahan), yang eksak karena \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} keluaran tingkat pertamanya
nol: tingkat kedua tidak membebaninya.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$; keluarannya $+V_{\mathrm{sat}}$ ketika \omterm{def:b1:wave-propagation:signal}{sinyalnya} melampaui
\qty{2.0}{V}, dan $-V_{\mathrm{sat}}$ di bawahnya (atau sebaliknya, menurut pengawatannya).

\textbf{21.} Ia menyaklar berulang-ulang: lampunya berkedip-kedip
selagi derau melintasi acuannya.

\textbf{22.} Dengan $V_{\mathrm{ref}}$ lewat $R_2$ dan keluarannya lewat
$R_1$\dots dalam bentuk bakunya, masukan tak membaliknya duduk pada
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$; dengan mengambil acuan yang
tergeser sesuai dengan itu, taraf penyaklarannya adalah $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
$\beta V_{\mathrm{sat}} = \qty{0.10}{V}$: $\beta = 1/150$, misalnya $R_1 =
\qty{1.0}{k\ohm}$, $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (terpicu) dan \qty{1.9}{V} (terlepas):
regangannya \num{1.68e-3} dan \num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. Pada
$-V_{\mathrm{sat}}$ LED-nya terprategang mundur sebesar \qty{15}{V}, di
luar batasnya: sebuah dioda biasa yang dipasang antiparalel (atau seri) melindunginya.

\textbf{25.} Jembatan (\qty{2.5}{mV} skala penuh) $\to$ dua pengikut
(tanpa pembebanan) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{penguat selisih} $\times100$ $\to$ tak membalik
$\times10$ $\to$ lolos rendah aktif \qty{10}{Hz} $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{pemicu Schmitt} pada
$\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ LED. Batas sejatinya: \omterm{rem:b1:operational-amplifier:real}{tegangan
simpangan masukan} dan penyesuaian \omterm{def:b1:dc-circuits:resistor}{resistor} \omterm{prop:b1:operational-amplifier:sumdiff}{penguat selisihnya}, yang
keduanya sebanding dengan \omterm{def:b1:wave-propagation:signal}{sinyal} bertaraf milivolt itu.
\end{solution}
