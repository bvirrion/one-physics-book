\section*{Bab \ref{ch:b1:potential-capacitors} --- Potensial Listrik; Penghantar dan Kapasitor}

\begin{solution}{exo:b1:potential-capacitors:1}
$V = 8.99 \times 10^9 \times 10^{-6}/1 = \qty{9.0}{kV}$; $W = qV = \qty{9.0}{mJ}$;
setelah dilepaskan, $\tfrac12mv^2 = \qty{9}{mJ}$: $v = \sqrt{18} = \qty{4.2}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:2}
$E = 100/10^{-3} = \qty{1e5}{V/m}$; $E_k = eU = \qty{100}{eV} = \qty{1.6e-17}{J}$,
sehingga $v = \sqrt{2E_k/m_e} = \qty{5.9e6}{m/s}$; dalam gerak berubah beraturan,
$t = 2d/v = \qty{3.4e-10}{s}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:3}
$E_A = 10/0.002 = \qty{5}{kV/m}$, $E_B = 10/0.008 = \qty{1.25}{kV/m}$. Medannya
menunjuk ke arah potensial yang mengecil; sebuah elektron ($q < 0$)
dipercepat ke arah sebaliknya, menuju $V$ yang membesar.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:4}
$C = 4\pi\varepsilon_0R = 1.11 \times 10^{-10} \times 0.15 = \qty{17}{pF}$; $Q = CV =
\qty{3.3}{\micro C}$; $E = V/R = 2 \times 10^5/0.15 = \qty{1.3}{MV/m}$. Dadalnya
pada $V = E_{\max}R = 3 \times 10^6 \times 0.15 = \qty{450}{kV}$ --- mesin besar
memakai bola besar.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:5}
Setiap unsur cincinnya berjarak $\sqrt{R^2 + z^2}$: $V =
Q/4\pi\varepsilon_0\sqrt{R^2 + z^2}$; $E_z = -\dd V/\dd z = Qz/4\pi\varepsilon_0(R^2 +
z^2)^{3/2}$. Cakramnya: $\dd q = 2\pi\sigma a\,\dd a$ pada $\sqrt{a^2 + z^2}$: $V =
\dfrac{\sigma}{2\varepsilon_0}\displaystyle\int_0^R\dfrac{a\,\dd a}{\sqrt{a^2 + z^2}} =
\dfrac{\sigma}{2\varepsilon_0}\bigl(\sqrt{z^2 + R^2} - z\bigr)$, dan $-\dd V/\dd z =
\dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:6}
$V(r) = V(R) + \displaystyle\int_r^R\dfrac{Qr'\,\dd r'}{4\pi\varepsilon_0R^3} =
\dfrac{Q}{4\pi\varepsilon_0R} + \dfrac{Q(R^2 - r^2)}{8\pi\varepsilon_0R^3} = \dfrac{Q}{4\pi\varepsilon_0}
\dfrac{3R^2 - r^2}{2R^3}$. Emas: $V(0) = 1.5 \times 8.99 \times 10^9 \times 1.26 \times
10^{-17}/7 \times 10^{-15} = \qty{24}{MV}$: sebuah proton perlu \qty{24}{MeV}, sebuah alfa
($2e$) \qty{48}{MeV} untuk mencapai pusatnya dan \qty{32}{MeV} untuk menyentuh
permukaannya. Alfa \qty{5}{MeV} milik Rutherford berbalik pada $r$ dengan $2eV =
\qty{5}{MeV}$: $V = \qty{2.5}{MV}$, $r = \qty{45}{fm}$ --- jauh di luar
intinya, dan itulah sebabnya analisis muatan titiknya berhasil.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:7}
Pada sumbunya: $2p/4\pi\varepsilon_0r^3 = 2 \times 8.99 \times 10^9 \times 6.2 \times 10^{-30}/10^{-27} =
\qty{1.1e8}{V/m}$; tegak lurus, separuhnya: \qty{5.6e7}{V/m}. \omterm{def:b1:angular-momentum:moment}{Momen gaya} $pE =
\qty{6.2e-24}{N.m}$; selang energinya $2pE = \qty{1.2e-23}{J}$ terhadap $k_BT =
\qty{4.1e-21}{J}$: $330$ kali lebih kecil --- hanya penyearahan total $0.1\%$.
Ion natrium pada \qty{0.3}{nm}: $E = e/4\pi\varepsilon_0r^2 = \qty{1.6e10}{V/m}$, $-pE
= \qty{-1.0e-19}{J} = \qty{-0.6}{eV}$, yaitu $24\,k_BT$: molekul airnya terkunci
pada ionnya --- itulah hidrasi.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:8}
$Q = 4\pi\varepsilon_0R^2E = 10^{-4} \times 3 \times 10^6/8.99 \times 10^9 = \qty{33}{nC}$,
$V = ER = \qty{30}{kV}$. Bola yang terhubung: $E_1 = V/R_1 = \qty{0.2}{MV/m}$,
$E_2 = V/R_2 = \qty{20}{MV/m} > \qty{3}{MV/m}$: bola kecilnya mengionkan
udaranya dan membocorkan muatannya --- \omterm{ex:b1:potential-capacitors:point}{efek ujung runcing}.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:9}
$C = \varepsilon_r\varepsilon_0S/d = 2.2 \times 8.85 \times 10^{-12}/10^{-5} = \qty{2.0}{\micro F}$;
penggulungan mengubah bentuknya, bukan luas yang saling berhadapan: $C$ sama (dengan
lapisan kedua sehingga tiap kertas timah menghadapi yang lain pada kedua sisinya, ia
berlipat dua). $W = \tfrac12 \times 2 \times 10^{-6} \times 400^2 = \qty{0.16}{J}$. Defibrilator:
$Q = 150 \times 10^{-6} \times 2000 = \qty{0.30}{C}$, $W = \tfrac12CU^2 = \qty{300}{J}$,
$P = 300/0.005 = \qty{60}{kW}$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:10}
(a) $E = Q/4\pi\varepsilon_0r^2$, $U = \dfrac{Q}{4\pi\varepsilon_0}\Bigl(\dfrac1{R_1} -
\dfrac1{R_2}\Bigr)$, $C = 4\pi\varepsilon_0R_1R_2/(R_2 - R_1)$; dengan $R_2 - R_1 = d \ll
R_1$, $R_1R_2 \approx R^2$ dan $C \approx 4\pi R^2\varepsilon_0/d = \varepsilon_0S/d$; dengan $R_2 \to
\infty$, $C \to 4\pi\varepsilon_0R_1$. (b) $C/L = 2\pi\varepsilon_r\varepsilon_0/\ln(b/a) = 2\pi \times 2.3
\times 8.85 \times 10^{-12}/\ln5 = \qty{79}{pF/m}$, ordenya tepat. (c) $E(a) =
U/[a\ln(b/a)] = 1000/(5 \times 10^{-4} \times 1.61) = \qty{1.2}{MV/m}$: di bawah
\qty{20}{MV/m} milik polietilena, aman; di udara nilainya akan mepet.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:11}
(a) $W = Q^2/2C = Q^2/8\pi\varepsilon_0R$. (b) $R = e^2/8\pi\varepsilon_0m_ec^2 = 2.31 \times
10^{-28}/(2 \times 8.19 \times 10^{-14}) = \qty{1.4e-15}{m}$ (``jari-jari klasik''
yang lazim, $e^2/4\pi\varepsilon_0m_ec^2 = \qty{2.8}{fm}$, membuang $\tfrac12$-nya). (c) $W =
Q^2/2C = Q^2d/2\varepsilon_0S$ berlipat dua: $\Delta W = Q^2d/2\varepsilon_0S = F\,d$ dengan $F =
Q^2/2\varepsilon_0S = \sigma^2S/2\varepsilon_0$, seperti yang diperoleh langsung. (d) Pada $U$ tetap, $W =
\varepsilon_0SU^2/2d$ berkurang separuh: $\Delta W = -W/2$; penariknya tetap melakukan usaha $W/2$
(integralkan $F = \varepsilon_0SU^2/2x^2$ dari $d$ sampai $2d$); kedua separuhnya, seluruhnya
$W$, mengalir balik ke baterainya seiring muatan $CU/2$ kembali padanya pada
\omterm{def:b1:dc-circuits:voltage}{tegangan} $U$.
\end{solution}

\begin{solution}{exo:b1:potential-capacitors:12}
(a) Bola berjari-jari $r$ mengemban $q = Qr^3/R^3$ pada potensial permukaan
$q/4\pi\varepsilon_0r$; menambahkan kulit $\dd q = 3Qr^2\,\dd r/R^3$ menelan $\dd W =
q\,\dd q/4\pi\varepsilon_0r = 3Q^2r^4\,\dd r/4\pi\varepsilon_0R^6$; integralkan: $W =
3Q^2/20\pi\varepsilon_0R$. (b) $Q = 92e = \qty{1.47e-17}{C}$: $W = 3 \times 2.17 \times
10^{-34}/(20\pi \times 8.85 \times 10^{-12} \times 7.4 \times 10^{-15}) = \qty{1.6e-10}{J}
= \qty{990}{MeV}$. (c) Tiap serpihannya: muatan $Q/2$, jari-jari $R/2^{1/3}$,
energi $\tfrac14 2^{1/3}W$; keduanya: $2^{1/3}W/2 = 0.63W$: yang dilepaskan
$0.37 \times 990 = \qty{370}{MeV}$. (d) ``Tegangan permukaan'' intinya:
dua serpihan berpermukaan lebih luas daripada satu inti, dan itu menelan kira-kira
\qty{170}{MeV}; sisanya, dekat \qty{200}{MeV}, muncul sebagian besar sebagai
\omterm{thm:b1:work-and-energy:ke}{energi kinetik} kedua serpihan yang terbang berpisah di bawah tolakan
timbal baliknya.
\end{solution}

\begin{solution}{pb:b1:potential-capacitors:1}
\textbf{1.} Berat $\tfrac43\pi r^3\rho g$ ke bawah, seretan $6\pi\eta rv$ ke atas,
\omterm{thm:b1:fluid-statics:archimedes}{gaya apung} $\tfrac43\pi r^3\rho_{\text{udara}}g$ ke atas --- $1.2/900 = 0.13\%$ dari
beratnya, terabaikan. Terminalnya: $\tfrac43\pi r^3\rho g = 6\pi\eta rv_1$, $v_1 =
2\rho gr^2/9\eta$.

\textbf{2.} $v_1 = \qty{1.0e-4}{m/s}$: $r = \sqrt{9\eta v_1/2\rho g} = \sqrt{9 \times
1.8 \times 10^{-5} \times 10^{-4}/(2 \times 900 \times 9.8)} = \qty{9.6e-7}{m}$, yaitu satu
mikrometer; $m = \tfrac43\pi r^3\rho = \qty{3.3e-15}{kg}$.

\textbf{3.} $\tau = m/6\pi\eta r = 3.3 \times 10^{-15}/3.3 \times 10^{-10} = \qty{1e-5}{s}$;
jaraknya $\sim v_1\tau = \qty{1}{nm}$. Tak pernah.

\textbf{4.} $\mathrm{Re} = 1.2 \times 10^{-4} \times 9.6 \times 10^{-7}/1.8 \times 10^{-5} =
6 \times 10^{-6} \ll 1$: \omterm{prop:b1:newton-dynamics:drag}{hukum Stokes} berlaku.

\textbf{5.} $eE = 1.6 \times 10^{-19} \times 3 \times 10^5 = \qty{4.8e-14}{N}$ terhadap
$mg = \qty{3.2e-14}{N}$: sebanding hanya bagi tetes semikron. Sebuah tetes
\qty{10}{\micro m} berbobot seribu kali lebih besar; satu \omterm{def:b1:electrostatics-gauss:charge}{muatan elementer} akan mengubah
kelajuannya sebesar $0.1\%$, tak terlihat.

\textbf{6.} Gerak Brown: gempuran molekul udara menyentak-nyentak
tetesnya secara acak; waktu ukurnya berserak, membatasi ketelitiannya dan
menetapkan batas bawah bagi ukuran tetes yang berguna.

\textbf{7.} $E = U/d = 5000/0.016 = \qty{3.1e5}{V/m}$; kepingnya jauh lebih lebar daripada
$d$, sehingga superposisi kedua bidangnya memberi medan yang seragam;
ekuipotensialnya bidang mendatar, $V = Uz/d$.

\textbf{8.} $C = \varepsilon_0\pi(0.1)^2/0.016 = \qty{17}{pF}$; $Q = CU = \qty{87}{nC}$;
$W = \tfrac12CU^2 = \qty{0.22}{mJ}$.

\textbf{9.} Keping atasnya positif: $\vect E$ menunjuk ke bawah; gayanya harus
menunjuk ke atas, sehingga $q < 0$. $|q| = mgd/U_0 = 3.3 \times 10^{-15} \times 9.8 \times 0.016/
1080 = \qty{4.8e-19}{C} = 3e$: $q = -3e$.

\textbf{10.} Naik dengan kelajuan tetap: $|q|E = mg + 6\pi\eta rv_2$ dan $mg =
6\pi\eta rv_1$, sehingga $|q|E = 6\pi\eta r(v_1 + v_2)$ dan $|q| = 6\pi\eta r(v_1 +
v_2)d/U$. $v_2 = (|q|E - mg)/6\pi\eta r = (1.5 \times 10^{-13} - 3.2 \times 10^{-14})/
3.3 \times 10^{-10} = \qty{3.6e-4}{m/s}$: \qty{1}{mm} dalam \qty{2.8}{s}.

\textbf{11.} Penyeimbangan adalah penilaian nol atas sebutir noktah yang menghanyut dan
bergetar, yang muatannya mungkin berubah sementara itu; pengukuran waktu atas jarak tetap dapat
diulang sesuka hati, $r$ ditetapkan sekali untuk selamanya oleh $v_1$, dan satu tetes lalu
memberi seluruh deret muatan.

\textbf{12.} $|q|U = 3 \times 1.6 \times 10^{-19} \times 5000 = \qty{2.4e-15}{J}$
(\qty{15}{keV}) melintasi celahnya; seretan sepanjang \qty{1}{cm}: $(|q|E - mg) \times 0.01 =
\qty{1.2e-15}{J}$, berubah menjadi kalor di udaranya.

\textbf{13.} Bagilah dengan $1.60$: $3.00$, $5.01$, $4.01$, $1.99$, $6.02$ ---
bilangan bulatnya $3, 5, 4, 2, 6$.

\textbf{14.} $q/n$: $1.600, 1.602, 1.605, 1.595, 1.605$; reratanya $1.601$,
simpangan bakunya $0.004$, ketakpastian reratanya $0.004/\sqrt5 =
0.002$: $e = (1.601 \pm 0.002) \times 10^{-19}$~C.

\textbf{15.} Sinar-X-nya mengionkan udaranya; tetesnya memungut ion satu per
satu, masing-masing mengemban bilangan bulat $e$.

\textbf{16.} $r \propto \eta^{1/2}$ dari $v_1$, $m \propto r^3 \propto \eta^{3/2}$, dan
$|q| = mgd/U_0 \propto \eta^{3/2}$: $1\%$ pada $\eta$ memberi $1.5\%$ pada $e$. Nilai $\eta$
milik Millikan $0.4\%$ terlalu rendah, karena itu $e$-nya $0.6\%$ terlalu rendah.

\textbf{17.} Kuark tak pernah datang sendirian: ia terkurung di dalam hadron,
dan setiap benda bebas mengemban muatan bulat.

\textbf{18.} $N_A = F/e = 96485/1.602 \times 10^{-19} = \qty{6.02e23}{mol^{-1}}$.

\textbf{19.} $m_e = 1.602 \times 10^{-19}/1.76 \times 10^{11} = \qty{9.1e-31}{kg}$: $1840$
kali lebih ringan daripada atom hidrogen.

\textbf{20.} Bidang mendatar di antara kedua kepingnya, menggelembung di sekeliling
tepinya; \omterm{def:b1:electrostatics-gauss:field}{garis medan} tegak yang mengembang keluar di tepinya sepanjang lebar
berorde $d = \qty{1.6}{cm}$ --- kecil terhadap diameter \qty{20}{cm}, dan
tetesnya diamati di tengah.

\textbf{21.} $E_p = qUz/d + mgz$, linear dalam $z$: tak ada minimum; sebuah tetes
yang diam menuntut kemiringannya lenyap, $q = -mgd/U$, dan kesetimbangannya
lalu bersifat netral --- setiap hanyutan sisa bertahan, dan itulah sebabnya
penyeimbangannya sukar dinilai.

\textbf{22.} $2.4 \times 10^{-15}$ terhadap $2.2 \times 10^{-4}$~J: $10^{-11}$. Tetesnya
tak mengganggu medannya.

\textbf{23.} Medannya ke atas, gaya pada tetes negatifnya ke bawah: ia jatuh dengan $(mg
+ |q|E)/6\pi\eta r = (3.2 \times 10^{-14} + 1.5 \times 10^{-13})/3.3 \times 10^{-10} =
\qty{5.6e-4}{m/s}$.

\textbf{24.} $F = 10^4 \times 1.6 \times 10^{-19} \times 3.1 \times 10^5 = \qty{5e-10}{N}$
terhadap $\qty{9.8e-9}{N}$: $5\%$ dari beratnya --- tak terangkat, dan dengan
$10^4$ muatan sebuah perubahan satu $e$ tak terlihat. Kabut minyak memberi tetes
yang kecil, tak menguap dan berbentuk bola.

\textbf{25.} Yang diukur: $v_1$, $v_2$ dan $U$ (dengan $d$, $\eta$, $\rho$ diketahui);
yang disimpulkan $r$, $m$, lalu $q$, yang ternyata selalu kelipatan bulat
$e = (1.601 \pm 0.002) \times 10^{-19}$~C ($0.1\%$); darinya $N_A = F/e$ dan
$m_e = e/(e/m_e)$.
\end{solution}
