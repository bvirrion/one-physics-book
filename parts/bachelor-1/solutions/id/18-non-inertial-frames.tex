\section*{Bab \ref{ch:b1:non-inertial-frames} --- Kerangka Tak Inersial; Dinamika di Bumi}

\begin{solution}{exo:b1:non-inertial-frames:1}
$\tan\alpha = a/g = 0.255$, $\alpha = 14^\circ$ (bebannya tertinggal ke belakang);
$T = m\sqrt{g^2 + a^2} = 1.03\,mg$. Mengerem: sudut dan tegangan yang sama,
bebannya condong ke depan.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:2}
$\Omega = 2\pi/4.0 = \qty{1.57}{rad/s}$; $\Omega^2r = \qty{7.4}{m/s^2} = 0.75g$,
terarah keluar dalam kerangka berputarnya; tempat duduknya (gesekan dan
sabuk penahan) memasok gaya nyata ke arah dalam.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:3}
$\Omega = 2\pi/86164 = \qty{7.29e-5}{rad/s}$; $\Omega^2R = \qty{0.034}{m/s^2}$,
$0.35\%$ dari $g$. Untuk melayang lepas dituntut $\Omega^2R = g$: $T = 2\pi\sqrt{R/g} =
\qty{84}{min}$ --- tujuh belas kali lebih cepat daripada sekarang.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:4}
$a_C = 2\Omega v\sin\lambda = 2 \times 7.29\times10^{-5} \times 27.8 \times 0.707
= \qty{2.9e-3}{m/s^2}$; $F = \qty{1.1}{kN}$. Menuju utara: terdorong ke timur,
ke rel kanannya (rel timurnya); menuju timur: terdorong ke selatan, sekali lagi
rel kanannya.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:5}
$z = \Omega^2r^2/2g$: \qty{2.0}{mm} pada \qty{2}{rad/s}, \qty{5.1}{cm} pada
\qty{10}{rad/s}. Paraboloid memfokuskan berkas sejajar ke satu titik:
raksa yang berputar adalah cermin parabolik yang sempurna, murah, dan memoles
dirinya sendiri (tetapi tak dapat dimiringkan).
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:6}
Percepatan sentrifugal $\Omega^2R\cos\lambda$ menunjuk menjauhi
sumbunya; komponennya yang tegak lurus pada vertikal setempat adalah $\Omega^2R\cos\lambda
\sin\lambda$, yang memiringkan $\vect g$ sebesar $\delta \approx \Omega^2R\sin\lambda\cos\lambda/g
= 0.034 \times 0.5/9.81 = \qty{1.7e-3}{rad} = 6'$. Khatulistiwa: $g$ berkurang sebesar
$\Omega^2R = \qty{0.034}{m/s^2}$ karena rotasinya (pemepatan Bumi menambahnya lagi).
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:7}
$\ddot x_E = 2\Omega gt\cos\lambda$: $x_E = \tfrac13\Omega g\cos\lambda\,t^3$; $t =
\sqrt{2h/g} = \qty{4.5}{s}$: $x_E = \tfrac13 \times 7.29\times10^{-5} \times 9.81
\times 0.707 \times 92 = \qty{1.6}{cm}$. Ke timur: mulut sumurnya
bergerak ke timur lebih cepat (jari-jarinya lebih besar) daripada dasarnya; batunya
menyimpan kelebihan itu --- atau, dalam kerangka berputarnya, \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis} pada
kecepatan ke bawah menunjuk ke timur.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:8}
$T_F = 23.93/\sin\lambda$: Paris \qty{31.8}{h}; kutub \qty{23.9}{h}; khatulistiwa
tak hingga (tanpa presesi). Paris: $360^\circ/31.8 = 11.3^\circ$ tiap jam.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:9}
$a_C = 2\Omega v\sin\lambda = \qty{2.1e-3}{m/s^2}$, ke kanan. Udara yang
memusat dari segala sisi dibelokkan ke kanan, sehingga ia berpilin masuk
berlawanan arah jarum jam (dilihat dari atas): siklon belahan utara. Wastafel
terkuras dalam beberapa sekon dan tornado terbentuk dalam beberapa menit: pembelokan
Coriolis selama waktu sesingkat itu dapat diabaikan terhadap pusaran awal apa pun.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:10}
Dalam kerangka mobilnya gravitasi semu $\vect g - \vect a$ condong ke belakang.
Udaranya, yang lebih berat, ``jatuh'' ke belakang; balonnya, yang lebih ringan daripada
udara yang didesaknya, terapung naik menyusuri vertikal semunya --- ia condong \emph{ke depan},
berlawanan dengan unting-unting.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:11}
$\Omega^2R = g$: $\Omega = \qty{0.31}{rad/s}$, $T = \qty{20}{s}$. $a_C = 2\Omega v
= \qty{0.94}{m/s^2}$, kira-kira $0.1g$: berjalan searah rotasinya ia merasa
lebih berat sebesar $10\%$ (gaya Coriolisnya menunjuk keluar), berlawanan arah
lebih ringan. Sebuah bola yang dijatuhkan: dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial} ia bergerak lurus
sementara lantainya melengkung naik; sepanjang waktu jatuh $\sqrt{2h/g} = \qty{0.64}{s}$
simpangannya berorde $\tfrac13\Omega\,g\,t^3 \cdot 2 \approx 2 \times 0.31 \times
9.81 \times 0.26/3 \approx \qty{0.5}{m}$ --- beberapa desimeter, berlawanan dengan
rotasinya.
\end{solution}

\begin{solution}{exo:b1:non-inertial-frames:12}
$GM/(d - r)^2 - GM/d^2 \approx 2GMr/d^3$ (orde pertama dalam $r/d$). Bulan:
$2 \times 6.67\times10^{-11} \times 7.35\times10^{22} \times 6.37\times10^6/
(3.84\times10^8)^3 = \qty{1.1e-6}{m/s^2}$; Matahari: $2 \times 1.33\times10^{20}
\times 6.37\times10^6/(1.50\times10^{11})^3 = \qty{5.0e-7}{m/s^2}$; nisbahnya
$0.46$: \omterm{rem:b1:non-inertial-frames:tides}{pasang surut} Matahari hampir separuh \omterm{rem:b1:non-inertial-frames:tides}{pasang surut} Bulan, walaupun
tarikannya $180$ kali lebih besar --- \omterm{rem:b1:non-inertial-frames:tides}{pasang surut} berjalan sebagai $M/d^3$.
\end{solution}

\begin{solution}{pb:b1:non-inertial-frames:1}
\textbf{1.} $T = 2\pi\sqrt{67/9.81} = \qty{16.4}{s}$; $x_0/\ell = 0.045$
($2.6^\circ$): kecil.

\textbf{2.} $v_{\max} = x_0\omega_0 = 3.0 \times 0.383 = \qty{1.15}{m/s}$; tingginya
$x_0^2/2\ell = \qty{6.7}{cm}$.

\textbf{3.} $T = mg + mv_{\max}^2/\ell = mg(1 + 0.002)$: $1.002\,mg$.

\textbf{4.} $\tfrac12\rho C_xSv_{\max}^2 = 0.5 \times 1.2 \times 0.05 \times 1.3 =
\qty{0.04}{N}$, berbanding $m\omega_0^2x_0 = \qty{12}{N}$: tiga ratus kali lebih
kecil, namun cukup untuk menghentikan bandulnya dalam beberapa jam.

\textbf{5.} Kawat yang panjang memberikan periode yang panjang, sehingga bebannya lambat
dan hambatan nisbinya kecil, dan membuat presesi tiap ayunannya terlihat; beban
yang berat menyimpan banyak energi untuk dimakan hambatannya --- berjam-jam berayun.

\textbf{6.} Berat (tegak) dan tegangan tali (dalam bidang kawatnya dan
vertikalnya, yakni dalam bidang ayunnya): tak ada gaya yang berkomponen
tegak lurus pada bidang itu.

\textbf{7.} Bidangnya tetap di antara bintang-bintang; esnya berputar di bawahnya
ke timur pada $\Omega$, sehingga pengamatnya melihat bidang itu berputar searah jarum jam,
kembali ke arah semula setelah satu hari sideris, \qty{23.93}{h}.

\textbf{8.} Tiap setengah ayunan melukiskan tali busur yang sedikit terputar: sebuah roset;
$86164/8.2 \approx 10500$ setengah ayunan, masing-masing $0.034^\circ$ lebih jauh berputar.

\textbf{9.} $\vect\Omega = \Omega(0, \cos\lambda, \sin\lambda)$.

\textbf{10.} Dengan $\vect v' = (\dot x, \dot y, 0)$:
$\vect\Omega\wedge\vect v' = \Omega(-\sin\lambda\,\dot y,\ \sin\lambda\,\dot x,\ -\cos\lambda\,\dot x)$,
sehingga $-2m\vect\Omega\wedge\vect v'$ berkomponen mendatar
$2m\Omega\sin\lambda\,(\dot y, -\dot x)$: hanya $\Omega\sin\lambda$ yang masuk.

\textbf{11.} $\ddot x = -\omega_0^2x + 2\omega_F\dot y$, $\ddot y = -\omega_0^2y -
2\omega_F\dot x$; $\omega_F = 7.29\times10^{-5} \times 0.753 = \qty{5.5e-5}{rad/s}
\ll \omega_0 = \qty{0.38}{rad/s}$.

\textbf{12.} $\ddot u = -\omega_0^2u + 2\omega_F(\dot y - j\dot x) = -\omega_0^2u -
2j\omega_F\dot u$.

\textbf{13.} $-\alpha^2 - 2\omega_F\alpha + \omega_0^2 = 0$: $\alpha = -\omega_F \pm
\sqrt{\omega_F^2 + \omega_0^2} \approx -\omega_F \pm \omega_0$.

\textbf{14.} Kurungnya adalah osilasi bidang (sebuah garis tetap melalui
titik asal untuk $A = B$ real); faktor $\eu^{-j\omega_Ft}$ memutar seluruh
polanya searah jarum jam pada $\omega_F$: bidang ayunnya berpresesi searah jarum jam pada
$\Omega\sin\lambda$.

\textbf{15.} $T_F = 2\pi/\omega_F = \qty{1.14e5}{s} = \qty{31.8}{h}$; $11.3^\circ$
tiap jam.

\textbf{16.} $\sin\lambda = 0$: komponen vertikal setempat $\vect\Omega$
lenyap; \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis} mendatarnya nol bagi gerak mendatar.

\textbf{17.} $2m\omega_Fv_{\max} = 2 \times 28 \times 5.5\times10^{-5} \times 1.15 =
\qty{3.5}{mN}$; beratnya \qty{275}{N} ($10^{-5}$ darinya), gaya pemulihnya
$m\omega_0^2x_0 = \qty{12}{N}$: tiga ribu kali lebih kecil.

\textbf{18.} $\omega_FT/2 = 5.5\times10^{-5} \times 8.2 = \qty{4.5e-4}{rad}$:
$x_0 \times 4.5\times10^{-4} = \qty{1.4}{mm}$ tiap setengah ayunan.

\textbf{19.} Pasak yang berjarak satu sentimeter pada lingkarannya jatuh tiap $\sim 7$
setengah ayunan, kira-kira semenit sekali.

\textbf{20.} Dorongan tangan memberikan kecepatan menyamping: bebannya lalu
melukiskan sebuah elips, yang presesi lambatnya sendiri (akibat \omterm{def:b1:wave-propagation:signal}{amplitudo}
yang berhingga) menutupi presesi Foucault.

\textbf{21.} Tendangan radial menambah energi menyusuri ayunannya tanpa menambah
kecepatan melintang, sehingga bidangnya tak tersentuh; komponen menyamping
akan menggiring bandulnya menjadi elips dan memalsukan atau menyembunyikan
presesinya.

\textbf{22.} Laju presesinya mengukur $\Omega\sin\lambda$, yaitu komponen tegak
rotasi Bumi; dengan $\Omega$ yang diketahui dari astronomi, laju itu
memberikan lintangnya (dan sebaliknya).

\textbf{23.} $\sin\lambda = 23.93/48 = 0.499$: $\lambda = 30^\circ$.

\textbf{24.} \omterm{def:b1:angular-momentum:L}{Momentum sudutnya} kekal, sehingga sumbunya tinggal tetap
di antara bintang-bintang sementara Bumi berputar di bawahnya: itulah kompas giro, yang
menemukan utara sejati tanpa magnet.

\textbf{25.} \omterm{thm:b1:non-inertial-frames:dynamics}{Gaya Coriolis} $-2m\vect\Omega\wedge\vect v'$; bagian mendatarnya
tumbuh sebagai $\sin\lambda$; satu putaran dalam $23.93\,\text{h}/\sin\lambda$,
\qty{31.8}{h} di Paris.
\end{solution}
