\section*{Bab \ref{ch:b1:point-kinematics} --- Kinematika Titik}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{12}{m/s} pada \qty{3.0}{s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ pada $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). \omterm{cor:b1:point-kinematics:circular}{Gerak melingkar} beraturan: kelajuannya
tetap, \omterm{def:b1:point-kinematics:velocity}{percepatannya} sentripetal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, yaitu $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Puncaknya pada $t = 7.71/9.81 = \qty{0.79}{s}$, tingginya $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Jangkauannya $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
Saat mendarat: $(9.19, -7.71)$, \qty{12}{m/s} pada $40^\circ$ di bawah
mendatar.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. Setelah \qty{5}{s}: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. Suku $2\dot r\dot\theta$ mempunyai dua paruh yang sama
besar: arah \omterm{def:b1:point-kinematics:velocity}{kecepatan} radialnya berputar dengan laju $\dot\theta$
(memberi $\dot r\dot\theta\,\vect e_\theta$), dan bergerak ke luar
memperbesar kelajuan ortoradialnya $r\dot\theta$ (satu $\dot r\dot\theta$ lagi).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ ke arah sumbunya; kisarnya $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ mendahului $x$ seperempat periode, dan $a = -\omega^2x$ berlawanan dengan $x$.
$\abs v$ terbesar pada $x = 0$, $\abs a$ pada ujungnya $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Lalu
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, dan dengan
menurunkannya sekali lagi memakai kaidah hasil kali diperoleh \omterm{def:b1:point-kinematics:velocity}{percepatan}
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Puncak: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$.
Titik tembak: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$.
Jari-jari di puncaknya lebih kecil dengan faktor $\cos^3\alpha$:
parabolanya paling membengkok di tempat ia paling lambat dan di tempat
seluruh $\vect g$ bersifat normal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. Di tanahnya ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ dan $\vect a = (0, v^2/R)$, ke atas. Di puncaknya
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: kelajuannya $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. Pada $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). Yang pertama adalah suku
sentripetal perputarannya; yang kedua, $2\dot r\dot\theta$, ada hanya
karena arah radialnya sendiri berputar --- di atas tanah, berjalan lurus
dengan laju tetap sama sekali tak memberi \omterm{def:b1:point-kinematics:velocity}{percepatan}.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (gerak dengan $\theta$ yang menaik),
$\vect N = -\vect e_r$ (ke arah $C$), $\rho = R$.

\textbf{3.} Puncak: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$. Samping:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ dan $v = R\dot\theta$:
$\dot v = -g\sin\theta$. Negatif untuk $0 < \theta < \pi$ (melambat pada
perjalanan naik), positif pada perjalanan turun.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: dasarnya
\qty{62.5}{m/s^2}; sampingnya $62.5 - 19.6 = \qty{42.9}{m/s^2}$; puncaknya $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} Dasar: \qty{62.5}{m/s^2} (tanpa bagian tangensial); samping:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; puncak: \qty{23.3}{m/s^2},
menunjuk lurus ke bawah ke arah $C$: sudutnya $0$ dengan garis tegaknya.

\textbf{7.} Tegak lurus di dasarnya dan di puncaknya ($a_T = 0$); tak
pernah sejajar ($a_N > 0$ di mana-mana).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
dasarnya \qty{2.50}{rad/s}, puncaknya \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: persamaan bandul
sederhana sepanjang $R$ --- keretanya di atas \omterm{def:b1:point-kinematics:frame}{lintasannya} adalah bandulan
yang melewati puncaknya.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Rata-rata $25$, $20.7$, $15.3$, $20.7$: sekitar \qty{20}{m/s};
panjangnya $2\pi R = \qty{63}{m}$: kira-kira \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: ya. Minimumnya: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} Dasar: $\vect a = 62.5$ ke atas, $\vect g$ ke bawah:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} Puncak: keduanya ke bawah, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
yang menekan penumpangnya ke kursinya (ke arah pusatnya, yaitu ke atas
bagi penumpang yang terbalik).

\textbf{15.} Samping: $\vect a = 42.9$ mendatar ke arah $C$ ditambah $9.8$
ke bawah (tangensial); $\vect a - \vect g$ adalah $42.9$ mendatar: $4.4g$
ke samping, ke arah pusatnya.

\textbf{16.} Tak lolos: $7.4g$ di dasarnya (puncak dan sampingnya baik-baik saja).

\textbf{17.} Berat semunya nol di puncaknya: $v_{\text{puncak}}^2 = gR$;
lalu $v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ di dasarnya dan bebannya
$5g + g = 6g$, bebas dari $R$.

\textbf{18.} $a_N = v^2/\rho$: $v$ ditetapkan oleh ketinggiannya
(energi), $\rho$ oleh geometri \omterm{def:b1:point-kinematics:frame}{lintasannya}; $\rho$ yang berbeda pada
ketinggian yang sama mengubah $a_N$ saja.

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{puncak}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: ia tumbuh linear
dari nol --- laju perubahan \omterm{def:b1:point-kinematics:velocity}{percepatan} yang tetap --- alih-alih melompat
dari $0$ ke $v^2/R$ di jalan masuk sebuah lingkaran, yang dirasakan
leher sebagai sebuah pukulan.

\textbf{22.} Sebuah tetes air \omterm{def:b1:optical-instruments:eye}{mata}: lebar dan melengkung landai di
dasarnya tempat keretanya cepat, sempit di puncaknya tempat ia lambat.
Ia menjaga bebannya di dekat $3g$ di sepanjangnya dan mendatangkannya
secara berangsur.

\textbf{23.} Garis pandangnya: jarak mendatar \qty{50}{m}, tinggi
\qty{20}{m}: $r = \qty{53.9}{m}$, sudut ketinggiannya $\beta = 21.8^\circ$.
Komponen melintang \omterm{def:b1:point-kinematics:velocity}{kecepatan} mendatarnya adalah $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Dilihat dari titik yang tak sepusat, laju sudut garis
pandangnya tidak tetap (cepat ketika keretanya dekat, lambat ketika
jauh): \omterm{def:b1:optical-instruments:camera}{kamera} berlaju tetap akan hanyut meninggalkan keretanya.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Koordinat polar} di $C$ untuk \omterm{def:b1:point-kinematics:velocity}{kecepatan} dan \omterm{def:b1:point-kinematics:velocity}{percepatan} pada
lingkarannya; pemisahan Frenet untuk ``mempercepat'' lawan ``berbelok''
dan untuk klotoidnya; \omterm{def:b1:point-kinematics:cylindrical}{koordinat polar} di \omterm{def:b1:optical-instruments:camera}{kameranya} untuk laju
pengikutannya; energi (bab berikutnya) untuk hukum kelajuannya.
\end{solution}
