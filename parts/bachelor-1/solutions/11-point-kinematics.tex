\section*{Chapter \ref{ch:b1:point-kinematics} --- Kinematics of a Point}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{12}{m/s} at \qty{3.0}{s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ at $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). Uniform \omterm{cor:b1:point-kinematics:circular}{circular motion}: constant speed,
\omterm{cor:b1:point-kinematics:circular}{centripetal acceleration}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Apex at $t = 7.71/9.81 = \qty{0.79}{s}$, height $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Range $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
Landing: $(9.19, -7.71)$, \qty{12}{m/s} at $40^\circ$ below the
horizontal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. After \qty{5}{s}: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. The $2\dot r\dot\theta$ term has two equal halves: the
radial \omterm{def:b1:point-kinematics:velocity}{velocity}'s direction turns at rate $\dot\theta$ (giving
$\dot r\dot\theta\,\vect e_\theta$), and moving outward increases the
orthoradial speed $r\dot\theta$ (another $\dot r\dot\theta$).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ toward the axis; pitch $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ leads $x$ by a quarter period, $a = -\omega^2x$ is opposite to $x$.
$\abs v$ is largest at $x = 0$, $\abs a$ at the extremes $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Then
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, and differentiating
once more with the product rule gives the \omterm{def:b1:point-kinematics:velocity}{acceleration} of
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Apex: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$.
Launch: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$.
The apex radius is smaller by $\cos^3\alpha$: the parabola bends most
where it is slowest and where all of $\vect g$ is normal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. At the ground ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ and $\vect a = (0, v^2/R)$, upward. At the top
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: speed $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. At $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). The first is the centripetal term
of the rotation; the second, $2\dot r\dot\theta$, exists only because the
radial direction itself rotates --- on the ground, walking straight at
constant speed gives no \omterm{def:b1:point-kinematics:velocity}{acceleration} at all.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (motion with $\theta$ increasing),
$\vect N = -\vect e_r$ (toward $C$), $\rho = R$.

\textbf{3.} Top: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$. Sides:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ and $v = R\dot\theta$:
$\dot v = -g\sin\theta$. Negative for $0 < \theta < \pi$ (slowing on the
way up), positive on the way down.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: bottom
\qty{62.5}{m/s^2}; sides $62.5 - 19.6 = \qty{42.9}{m/s^2}$; top $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} Bottom: \qty{62.5}{m/s^2} (no tangential part); sides:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; top: \qty{23.3}{m/s^2},
pointing straight down toward $C$: angle $0$ with the vertical.

\textbf{7.} Perpendicular at the bottom and top ($a_T = 0$); never
parallel ($a_N > 0$ everywhere).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
bottom \qty{2.50}{rad/s}, top \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: the equation of a
simple pendulum of length $R$ --- the car on its track is a pendulum
bob that goes over the top.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Mean of $25$, $20.7$, $15.3$, $20.7$: about \qty{20}{m/s};
length $2\pi R = \qty{63}{m}$: roughly \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: yes. Minimum: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} Bottom: $\vect a = 62.5$ upward, $\vect g$ downward:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} Top: both downward, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
pressing the rider into the seat (toward the center, i.e.\ upward for
the inverted rider).

\textbf{15.} Sides: $\vect a = 42.9$ horizontal toward $C$ plus $9.8$
downward (tangential); $\vect a - \vect g$ is $42.9$ horizontal: $4.4g$
sideways, toward the center.

\textbf{16.} Fails: $7.4g$ at the bottom (the top and sides are fine).

\textbf{17.} Zero apparent weight at the top: $v_{\text{top}}^2 = gR$;
then $v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ at the bottom and the load is
$5g + g = 6g$, independent of $R$.

\textbf{18.} $a_N = v^2/\rho$: $v$ is fixed by the height (energy), $\rho$
by the track's geometry; a different $\rho$ at the same height changes
$a_N$ alone.

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{top}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: it grows linearly
from zero --- a constant rate of change of \omterm{def:b1:point-kinematics:velocity}{acceleration} --- instead of
jumping from $0$ to $v^2/R$ at the entry of a circle, which the neck
feels as a blow.

\textbf{22.} A teardrop: wide and gently curved at the bottom where the
car is fast, tight at the top where it is slow. It keeps the load near
$3g$ throughout and brings it on gradually.

\textbf{23.} Line of sight: horizontal distance \qty{50}{m}, height
\qty{20}{m}: $r = \qty{53.9}{m}$, elevation $\beta = 21.8^\circ$. The
horizontal \omterm{def:b1:point-kinematics:velocity}{velocity}'s transverse component is $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Seen from an off-center point the angular rate of the line
of sight is not constant (fast when the car is near, slow when far):
a constant-rate \omterm{def:b1:optical-instruments:camera}{camera} drifts off the car.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Polar coordinates} at $C$ for \omterm{def:b1:point-kinematics:velocity}{velocity} and \omterm{def:b1:point-kinematics:velocity}{acceleration} on
the circle; the Frenet split for ``speeding up'' versus ``turning'' and
for the clothoid; \omterm{def:b1:point-kinematics:cylindrical}{polar coordinates} at the \omterm{def:b1:optical-instruments:camera}{camera} for the tracking
rate; energy (next chapters) for the speed law.
\end{solution}
