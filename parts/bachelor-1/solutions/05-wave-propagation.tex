\begin{solution}{exo:b1:wave-propagation:6}
$\lambda = \qty{0.40}{m}$. On the segment, at distance $x$ from speaker
1, $\delta = (2.0 - x) - x = 2.0 - 2x$; silence for $\delta = \pm\lambda/2,
\pm 3\lambda/2, \dots$: $x = 0.1, 0.3, \dots, 1.9$ m, every \qty{20}{cm};

midpoint and every \qty{20}{cm} from it.
\end{solution}

\begin{solution}{exo:b1:wave-propagation:11}
Mode $n$ has shape $\sin(n\pi x/L)$. A node forced at $x = L/3$ requires
$\sin(n\pi/3) = 0$, i.e.\ $n$ a multiple of $3$: frequencies $3f_1,
6f_1, \dots$ Plucking at $L/2$ gives the string a maximum displacement
there, which mode $n$ can contribute to only if $\sin(n\pi/2) \neq 0$:
even modes, which have a node at $L/2$, receive nothing.
\end{solution}

\begin{solution}{exo:b1:wave-propagation:12}
$\lambda = c/f = \qty{3.0}{m}$. \omterm{def:b1:wave-propagation:signal}{Phase} difference $2\pi(2d)/\lambda + \pi$;
destructive when it equals $(2m + 1)\pi$: $2d/\lambda = m$,
$d = m\lambda/2 = 0, 1.5, 3.0, \dots$ m; constructive when $2d/\lambda =
m + \tfrac12$: $d = 0.75, 2.25, \dots$ m. Tunnel walls and roof reflect
the wave: a standing pattern with dead spots every half-wavelength along
the car's path.
\end{solution}

\begin{solution}{pb:b1:wave-propagation:1}
\textbf{1.} $[F/\mu] = M L T^{-2}/(M L^{-1}) = L^2 T^{-2}$: a squared
speed.

\textbf{2.} A wave toward $-x$ depends on $t + x/c$, i.e.\ on
$\omega t + kx$; $k = \omega/c = 2\pi f/c$.

\textbf{3.} $c = \sqrt{0.90/1.0\times10^{-3}} = \qty{30}{m/s}$;
$\lambda = c/f = \qty{0.60}{m}$; $k = 2\pi/\lambda = \qty{10.5}{rad/m}$.

\textbf{4.} $\Delta\varphi = kd = 10.5 \times 0.15 = \qty{1.6}{rad} = \pi/2$.

\textbf{5.} $v_{\max} = A\omega = 2.0\times10^{-3} \times 2\pi \times 50 =
\qty{0.63}{m/s}$, fifty times less than $c$: the string's points move
slowly while the shape races along.

\textbf{6.} It travels toward $+x$: \omterm{def:b1:wave-propagation:signal}{phase} $\omega t - kx$. At the fixed
end the total displacement vanishes at all times: $s_i(0, t) + s_r(0, t)
= A\cos\omega t + (-A)\cos\omega t = 0$.

\textbf{7.} $\cos(\omega t + kx) - \cos(\omega t - kx) =
-2\sin(\omega t)\sin(kx)$, so $s = -2A\sin(kx)\sin(\omega t)$.

\textbf{8.} Nodes: $\sin kx = 0$, $x = n\lambda/2$; antinodes $x = (n +
\tfrac12)\lambda/2$; neighboring nodes $\lambda/2$ apart.

\textbf{9.} $\sin kL = 0$: $L = n\lambda/2$, i.e.\ $f_n = nc/2L$.

\textbf{10.} $n = 2L/\lambda = 2.4/0.60 = 4$: four spindles, an integer
--- resonance.

\textbf{11.} $3$ spindles: $\lambda = 2L/3 = \qty{0.80}{m}$, $c = \lambda f
= \qty{40}{m/s}$, $F = \mu c^2 = \qty{1.6}{N}$. $6$ spindles: $\lambda =
\qty{0.40}{m}$, $c = \qty{20}{m/s}$, $F = \qty{0.40}{N}$.

\textbf{12.} $c = 2Lf/n = 2 \times 1.20 \times 50/3 = \qty{40}{m/s}$,
$\mu = F/c^2 = 1.6/1600 = \qty{1.0e-3}{kg/m} = \qty{1.0}{g/m}$.

\textbf{13.} $\mu = mgn^2/(4L^2f^2)$: $u(\mu)/\mu = \sqrt{(u_m/m)^2 +
(2u_L/L)^2} = \sqrt{0.61^2 + 0.33^2}\,\% = 0.7\%$; the mass limits.

\textbf{14.} The fundamental. $c = 2Lf_1$: low E $2 \times 0.650 \times
82.4 = \qty{107}{m/s}$; high E \qty{428}{m/s}.

\textbf{15.} $F = \mu c^2$: low E $6.0\times10^{-3} \times 107^2 =
\qty{69}{N}$; high E $4.0\times10^{-4} \times 428^2 = \qty{73}{N}$ ---
comparable tensions. Lowering $f$ by a factor $4$ at fixed $L$ and $\mu$
would need $16$ times less tension: a floppy, buzzing string. Winding
raises $\mu$ instead.

\textbf{16.} $f_n = 2^{n/12}f_1$ and $f \propto 1/L$: $L_n = 2^{-n/12}L$,
distance from the nut $L - L_n = L(1 - 2^{-n/12})$: fret 1 \qty{36.5}{mm},
fret 5 \qty{163}{mm}, fret 7 \qty{216}{mm}, fret 12 \qty{325}{mm}.

\textbf{17.} Each spacing is $2^{-1/12} = 0.944$ times the previous one.

\textbf{18.} Factor $2^{3/12} = 1.19$ (three semitones); length
$650 \times 2^{-3/12} = \qty{547}{mm}$.

\textbf{19.} Touching at $L/2$: even modes only, pitch $2f_1 =
\qty{165}{Hz}$; at $L/3$: multiples of $3$, pitch $3f_1 = \qty{247}{Hz}$.

\textbf{20.} Near an end, $\sin(n\pi x/L) \approx n\pi x/L$ grows with
$n$: high modes are excited relatively more --- bright. Over the sound
hole (near the middle) even modes are weak and the fundamental dominates
--- mellow.

\textbf{21.} $3f_1 = \qty{330}{Hz}$ beats at \qty{2}{Hz} with $2f_2$:
$f_2 = 166$ or \qty{164}{Hz}, ratio $1.509$ or $1.491$ (the equal-tempered
fifth is $2^{7/12} = 1.498$).

\textbf{22.} The fourth harmonic, $4f$. Three beats: $4f = 437$ or
\qty{443}{Hz}, $f = 109.25$ or \qty{110.75}{Hz}.

\textbf{23.} Beats quicken when tightening, so the string was sharp:
\qty{110.75}{Hz}. It must drop by $0.75/110.75 = 0.68\%$, i.e.\ the
tension by $1.4\%$.

\textbf{24.} $f = (n/2L)\sqrt{F/\mu}$, so $\ln f = \tfrac12\ln F + \text{const}$
and $\dd f/f = \tfrac12\,\dd F/F$.

\textbf{25.} $f_1 = \dfrac{1}{2L}\sqrt{\dfrac{F}{\mu}}$: the length fixes
$\lambda_1 = 2L$, tension and mass fix $c$, frets scale $L$ by
$2^{-n/12}$, beats reveal $f - f_{\text{ref}}$. Behind all of it, one
relation: $\lambda f = c$, with the standing-wave condition $L = n\lambda/2$.
\end{solution}
