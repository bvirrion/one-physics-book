\section*{Chapter \ref{ch:b1:charged-particles} --- Charged Particles in E and B Fields}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$: electron
\qty{1.9e7}{m/s}; proton \qty{4.4e5}{m/s} ($\sqrt{1836}$ times slower).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. At $30^\circ$: a helix of radius $R\sin 30^\circ
= \qty{10}{m}$ and pitch $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, drifting
along the field line.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Faster ions: the magnetic force
wins, they curve toward the magnetic side; opposite charge at the same
speed: both forces reverse, still balanced --- the selector is blind to
the sign and to the mass.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; deflection
$0.30 \times 0.10 = \qty{3.0}{cm}$; sensitivity \qty{0.30}{mm/V}. Faster
electrons spend less time between the plates: $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; separation $2\Delta R =
\qty{1.2}{cm}$ between the landing points (radii differ by \qty{5.8}{mm},
diameters by twice that).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; \qty{100}{keV} per turn: $270$ turns; time $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
period $2\pi m/eB = \qty{18}{ns}$, pitch $v_\parallel T = \qty{1.5}{cm}$:
about $64$ turns per meter.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: sub-microvolt. With $n$ ten million times smaller, a
semiconductor gives millivolts --- a usable sensor.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; dot with $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ whatever $\vect B(\vect r, t)$, provided there is no $\vect E$. The
direction of the velocity, the \omterm{def:b1:point-kinematics:frenet}{radius of curvature}, the pitch can all
change; the speed, the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} (\qty{10}{keV}) cannot.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. With $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- a
pure cyclotron rotation of $\vect w$ at $\omega_c$. Starting from rest,
$\vect w(0) = -(E/B)\vect e_x$: $\vect w$ turns at constant norm $E/B$, so the
velocity is the drift $E/B$ plus a rotating vector of the same norm ---
a cycloid, with cusps where $\vect v = \vect 0$. Electron: drift $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, arch height $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (about $15^\circ$):
deflection $\approx 0.30 \times 0.27 = \qty{8}{cm}$ for a modest
millitesla. Large angles with plates would need kilovolts and long
plates; a coil bends the beam through tens of degrees with no \omterm{def:b1:dc-circuits:voltage}{voltage}
at all.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: no power, constant
speed.

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, free of
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ per crossing, \qty{100}{keV} per turn.

\textbf{5.} The gap must be at the accelerating polarity each time the
proton arrives, half a period apart; off frequency, the phase slips and
the kicks eventually decelerate.

\textbf{6.} After one turn $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; after $n$ turns $E_k =
n \times \qty{100}{keV}$ and $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ turns; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} No: $E_k$ depends on $B$ and $R_{\max}$ only; $U$ sets the
number of turns and the time.

\textbf{11.} Path \qty{560}{m}: need $\lambda \gtrsim \qty{1}{km}$, i.e.\
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
a high vacuum.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; the period grows by $2.9\%$.
Over $270$ turns the slip accumulates to several periods --- the last
turns would be out of phase; in practice $B$ and the injection phase are
trimmed, and \qty{27}{MeV} is near the limit of the simple machine.

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; $R =
\qty{1.1}{m}$ at \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ grows by $24\%$ from center to rim: a
fixed frequency falls hopelessly out of step.

\textbf{15.} Synchrocyclotron: lower the frequency as the bunch gains
energy --- only one bunch at a time, low average current. Isochronous:
shape $B(R) \propto \gamma$ so the period stays constant --- continuous
beam, the modern choice.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, a little more than
the non-relativistic \qty{1.1}{m}.

\textbf{17.} Transverse impulse $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; angle $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: hopeless for steering.

\textbf{18.} Magnetic: $R = p/eB = \qty{2.3}{m}$, angle $L/R = \qty{0.43}{rad}
\approx 25^\circ$: a thousand times more --- magnets it is.

\textbf{19.} Energy per proton $\qty{3.68e-11}{J}$; dose $2.0 \times 0.50 =
\qty{1.0}{J}$: $2.7\times10^{10}$ protons.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \qty{6.2e9}{protons/s}$: about
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
a night light, delivered exactly where wanted.

\textbf{22.} $\qty{58}{MeV}$ in \qty{1}{cm}: about \qty{6}{keV/\micro m} ---
dense ionization along the last track.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionizations per proton ---
the molecular damage that kills the cell.

\textbf{24.} X-rays deposit energy all along their path, most near the
entrance; protons deposit most of theirs at the end of their range,
sparing the tissue in front and leaving nothing behind.

\textbf{25.} $R = mv/eB$ and $T = 2\pi m/eB$ (the circle), $E_k = (eBR)^2/2m$
(the energy), $E_k/2eU$ (the turns), dose $=$ (protons $\times$ energy)/mass
and current $= e \times$ rate (the time).
\end{solution}
