\section*{Chapitre \ref{ch:b1:sinusoidal-impedance} --- Régime sinusoïdal forcé et impédance}

\begin{solution}{exo:b1:sinusoidal-impedance:1}
$\underline U = 5\eu^{j\pi/3} = 2.5 + 4.33j$ (V). Pour $3\cos\omega t +
4\sin\omega t$~: $3 + 4\eu^{-j\pi/2} = 3 - 4j$, de module $5$ et
d'argument $-\arctan(4/3) = \qty{-0.927}{rad}$~: $5\cos(\omega t - 0.927)$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:2}
$R$~: \qty{100}{\ohm} dans les deux cas. $L\omega$~: \qty{31.4}{\ohm} (\qty{50}{Hz}),
\qty{628}{\ohm} (\qty{1}{kHz}). $1/C\omega$~: \qty{318}{\ohm}, \qty{15.9}{\ohm}.
$RL$ en série à \qty{50}{Hz}~: $\abs{\underline Z} = \sqrt{100^2 + 31.4^2} =
\qty{105}{\ohm}$, phase $\arctan(0.314) = 17^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:3}
$U = 230\sqrt2 = \qty{325}{V}$. $I_{\mathrm{eff}} = 2000/230 = \qty{8.7}{A}$,
crête \qty{12.3}{A}. $\langle Ri^2\rangle = R\langle i^2\rangle =
RI_{\mathrm{eff}}^2$ par définition de la \omterm{def:b1:sinusoidal-impedance:rms}{valeur efficace}.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:4}
$f_0 = 1/(2\pi\sqrt{LC}) = \qty{5.03}{kHz}$~; $Q = \sqrt{L/C}/R = 3.16$~;
$\Delta f = f_0/Q = \qty{1.6}{kHz}$~; $I_{\max} = E/R = \qty{10}{mA}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:5}
$1/\sqrt2$ en $\omega = 1/RC$~: $f_c = 1/(2\pi RC) = \qty{1.0}{kHz}$. À
\qty{1.0}{kHz}~: rapport $0.71$, phase $-45^\circ$. À \qty{10}{kHz}~:
$RC\omega = 10$, rapport $1/\sqrt{101} = 0.10$, phase $-84^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:6}
$\underline Z = 100 + j(250 - 100) = 100 + 150j$~: $Z = \qty{180}{\ohm}$,
$\arg = 56^\circ$. $I = 1.0/180 = \qty{5.6}{mA}$, en retard de $56^\circ$
sur la source. $U_R = \qty{0.56}{V}$, $U_L = \qty{1.39}{V}$, $U_C = \qty{0.56}{V}$~;
diagramme~: $U_R$ le long de $I$, $U_L$ vers le haut, $U_C$ vers le bas,
résultante \qty{1.0}{V}. $L\omega > 1/C\omega$~: au-dessus de la
résonance (inductif).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:7}
$f_0 = 1/(2\pi\sqrt{LC})$~: de \qty{1.59}{MHz} (\qty{50}{pF}) à \qty{0.50}{MHz}
(\qty{500}{pF}). À \qty{1.00}{MHz}~: $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$.
$\Delta f = \qty{10}{kHz}$. Station à \qty{1.009}{MHz}~: $x = 1.009$,
$Q(x - 1/x) = 1.8$, rapport $1/\sqrt{1 + 3.2} = 0.49$. Tension du
condensateur $QE = \qty{1.0}{mV}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:8}
$Q(x - 1/x) = \pm1 \Leftrightarrow x^2 \mp x/Q - 1 = 0$~; racines positives
$x_{1,2} = \mp 1/2Q + \sqrt{1 + 1/4Q^2}$. Différence $1/Q$, donc
$\Delta\omega = \omega_0/Q$~; produit $(1 + 1/4Q^2) - 1/4Q^2 = 1$, donc
$\omega_1\omega_2 = \omega_0^2$ (la bande est symétrique en échelle logarithmique).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:9}
$I = P/(U\cos\varphi) = 2000/(230 \times 0.70) = \qty{12.4}{A}$. \omterm{thm:b1:sinusoidal-impedance:power}{Puissance
réactive} $P\tan\varphi = 2000 \times 1.02 = \qty{2.0}{kvar}$. $C = 2040/(314
\times 230^2) = \qty{123}{\micro F}$. Nouveau courant $2000/230 = \qty{8.7}{A}$.
Pertes $0.50 \times 12.4^2 = \qty{77}{W}$ avant, $0.50 \times 8.7^2 =
\qty{38}{W}$ après.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:10}
$\underline Y = 1/R + j(C\omega - 1/L\omega)$~; $\underline U = \underline I/
\underline Y$ est maximale lorsque la susceptance s'annule, en $\omega_0$,
où $U = RI$. Demi-puissance~: $\abs{C\omega - 1/L\omega} = 1/R$, c'est-à-dire
$RC\omega_0(x - 1/x) = \pm1$~: $\Delta\omega = \omega_0/(RC\omega_0) = 1/RC$
et $Q = RC\omega_0 = R\sqrt{C/L}$. Application~: $\omega_0 = \qty{1.0e6}{rad/s}$
($f_0 = \qty{159}{kHz}$), $Q = 10^4\sqrt{10^{-6}} = 10$, $\Delta f =
\qty{16}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:11}
$U_C = I/C\omega = E/[C\omega\sqrt{R^2 + (L\omega - 1/C\omega)^2}]$~;
en multipliant sous la racine par $C\omega$, il vient
$\sqrt{(RC\omega)^2 + (LC\omega^2 - 1)^2} =
\sqrt{x^2/Q^2 + (1 - x^2)^2}$. Avec $s = x^2$, $(1 - s)^2 + s/Q^2$ est
minimal en $s = 1 - 1/2Q^2$ (si cette valeur est positive), de minimum $1/Q^2 - 1/4Q^4$~:
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$. Pour $Q = 3.16$~: $x_r = 0.975$,
$U_{C,\max} = 3.2E$. Pour $Q < 1/\sqrt2$, le minimum est en $s = 0$~: $U_C$
décroît de $E$ de façon monotone, sans maximum.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:12}
À la résonance, $i = I\cos\omega_0 t$ et $u_C = (I/C\omega_0)\sin\omega_0 t$,
donc, en utilisant $1/C\omega_0^2 = L$,
\[
\tfrac12 Li^2 + \tfrac12 Cu_C^2 = \tfrac12 LI^2\cos^2\omega_0 t
+ \tfrac12 LI^2\sin^2\omega_0 t = \tfrac12 LI^2 .
\]
Dissipée par période~: $\tfrac12 RI^2T$. Rapport $\times 2\pi$~: $2\pi L/(RT)
= L\omega_0/R = Q$.
\end{solution}

\begin{solution}{pb:b1:sinusoidal-impedance:1}
\textbf{1.} $\underline Z = R + j(L\omega - 1/C\omega)$.

\textbf{2.} $\underline I = E/\underline Z$~; $I = E/\sqrt{R^2 + (L\omega -
1/C\omega)^2}$~; $\varphi_i = -\arctan[(L\omega - 1/C\omega)/R]$.

\textbf{3.} La \omterm{def:b1:sinusoidal-impedance:impedance}{réactance} s'annule en $\omega_0 = 1/\sqrt{LC}$~: $I_{\max}
= E/R$, $\varphi_i = 0$.

\textbf{4.} $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$~; $L\omega_0 = 2\times10^{-4}
\times 6.28\times10^6 = \qty{1257}{\ohm}$.

\textbf{5.} $Q = 1257/12.6 = 100$.

\textbf{6.} $L\omega - 1/C\omega = L\omega_0(x - 1/x)$, que l'on divise par $R$.

\textbf{7.} Pour $x < 1$~: \omterm{def:b1:sinusoidal-impedance:impedance}{réactance} négative, $\varphi_i > 0$, le
courant est en avance (capacitif)~; pour $x > 1$~: en retard (inductif).

\textbf{8.} $Q(x - 1/x) = \pm1$.

\textbf{9.} $x^2 \mp x/Q - 1 = 0$, $x_{1,2} = \mp1/2Q + \sqrt{1 + 1/4Q^2}$,
$x_2 - x_1 = 1/Q$~: $\Delta\omega = \omega_0/Q$.

\textbf{10.} $\Delta f = \qty{10}{kHz}$.

\textbf{11.} \qty{1.010}{MHz}~: $Q(x - 1/x) = 100 \times 0.0199 = 2.0$,
rapport $1/\sqrt{5.0} = 0.45$, puissance $0.20$~: \qty{-7}{dB}. \qty{1.020}{MHz}~:
$3.96$, rapport $0.25$, puissance $0.06$~: \qty{-12}{dB}.

\textbf{12.} Tout juste~: la voisine n'est atténuée que de \qty{7}{dB}.
Un $Q$ plus élevé (un $R$ plus faible), ou plusieurs étages accordés en
cascade, affine la sélection.

\textbf{13.} De \qty{0.50}{MHz} à \qty{1.59}{MHz}~: la bande AM.

\textbf{14.} $\underline U_C = \underline I/jC\omega$~; en $\omega_0$,
$U_C = I_{\max}/C\omega_0 = E/(RC\omega_0) = QE$.

\textbf{15.} $I_{\max} = 10^{-5}/12.6 = \qty{0.79}{\micro A}$~; $U_C = QE =
\qty{1.0}{mV}$~: un gain en tension de cent, sans aucun amplificateur.

\textbf{16.} $\underline U_L = jL\omega_0\underline I = jQE$, $\underline U_C
= -jQE$~: égales et opposées, elles se compensent~; la source ne voit que $R$.

\textbf{17.} Stockée $\tfrac12 LI_{\max}^2 = \qty{6.2e-17}{J}$~; dissipée
par période $\tfrac12 RI_{\max}^2T = \qty{3.9e-18}{J}$~; $2\pi \times 6.2/0.39
\approx 100 = Q$.

\textbf{18.} $\tau = 2Q/\omega_0 = \qty{32}{\micro s}$. Le circuit ne peut
pas suivre des variations plus rapides que $\sim\tau$~: une bande
passante $\Delta f$ ne laisse passer que des modulations plus lentes que
$\Delta f$~; avec $Q = 1000$, $\Delta f =
\qty{1}{kHz}$ et les aigus de l'émission seraient perdus.

\textbf{19.} $Q$ est divisé par deux, à $50$, $\Delta f$ double, à \qty{20}{kHz},
et $U_C$ est divisée par deux, à \qty{0.5}{mV}.

\textbf{20.} $R_{\text{tot}} = \qty{62.6}{\ohm}$~: $Q = 1257/62.6 = 20$,
$\Delta f = \qty{50}{kHz}$, $U_C = 20 \times \qty{10}{\micro V} = \qty{0.2}{mV}$~:
sensibilité et sélectivité s'effondrent ensemble.

\textbf{21.} $P = E^2/(2R_{\text{tot}}) = 10^{-10}/125 = \qty{8e-13}{W}$~;
fraction dans $R_a$~: $50/62.6 = 80\%$.

\textbf{22.} L'adaptation demande une résistance de charge égale à $R_a$,
c'est-à-dire une grande résistance totale~; la sélectivité demande la
plus petite résistance totale possible ($Q = L\omega_0/R_{\text{tot}}$).
On ne peut pas avoir les deux avec une liaison directe.

\textbf{23.} Une prise intermédiaire, ou un petit condensateur de
couplage, ne présente à l'antenne qu'une fraction du circuit résonnant~:
la résistance de l'antenne est «~vue~» très réduite par le circuit, ce
qui garde $Q$ élevé, tandis que l'antenne délivre encore une puissance
proche de son optimum.

\textbf{24.} Aux fréquences de demi-puissance, la \omterm{def:b1:sinusoidal-impedance:impedance}{réactance} vaut $\pm R$,
donc $\varphi = \pm45^\circ$ et $\cos\varphi = 1/\sqrt2$~; comme $I =
I_{\max}/\sqrt2$ également, $P = \tfrac12 EI\cos\varphi$ vaut la moitié
de sa valeur à la résonance.

\textbf{25.} $Q$ multiplie le \omterm{def:b1:wave-propagation:signal}{signal} par $Q$ et resserre la bande à
$f_0/Q$, au prix d'une réponse plus lente ($\tau = 2Q/\omega_0$) et d'un
couplage fragile à l'antenne~; pour des canaux de \qty{10}{kHz} à
\qty{1}{MHz}, $Q \approx 100$ est le bon nombre --- et il faut coupler
l'antenne faiblement pour le conserver.
\end{solution}
