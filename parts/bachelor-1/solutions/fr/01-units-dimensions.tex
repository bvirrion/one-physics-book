\section*{Chapitre \ref{ch:b1:units-dimensions} --- Unités, dimensions et mesure}

\begin{solution}{exo:b1:units-dimensions:1}
$\unit{J} = \unit{kg.m^2.s^{-2}}$; $\unit{W} = \unit{kg.m^2.s^{-3}}$;
$\unit{Pa} = \unit{kg.m^{-1}.s^{-2}}$; $\unit{C} = \unit{A.s}$.
$\unit{Pa.m^3} = \unit{kg.m^{-1}.s^{-2}.m^3} = \unit{kg.m^2.s^{-2}} = \unit{J}$
(le travail des forces de pression, $P\,\dd V$).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:2}
Pression $M L^{-1} T^{-2}$~; puissance $M L^2 T^{-3}$~; charge $I\,T$~;
tension $= $ puissance/intensité $= M L^2 T^{-3} I^{-1}$.
$[G] = [F r^2/m^2] = M L T^{-2} L^2 M^{-2} = L^3 M^{-1} T^{-2}$;
$[h] = [E/\nu] = M L^2 T^{-2} \cdot T = M L^2 T^{-1}$;
$[k_B] = [E/T] = M L^2 T^{-2} \Theta^{-1}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:3}
$\sqrt{2gh}$~: $\sqrt{L T^{-2} \cdot L} = L T^{-1}$, homogène.
$\tfrac12 mv$~: $M L T^{-1} \neq M L^2 T^{-2}$, ce n'est pas une énergie.
$\rho g h$~: $M L^{-3} \cdot L T^{-2} \cdot L = M L^{-1} T^{-2}$, une pression.
$v_0 t + \tfrac12 g t^2$~: les deux termes en $L$, homogène.
$2\pi\sqrt{g/\ell}$~: $\sqrt{T^{-2}} = T^{-1}$ --- une fréquence, pas une période.
$I_0\eu^{-t/RC}$~: $RC$ en $\unit{\ohm.F} = \unit{s}$, argument \omterm{def:b1:units-dimensions:dimension}{sans dimension}, homogène.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:4}
Règle~: $\delta = \qty{1}{mm}$, $u = 1/\sqrt{12} = \qty{0.29}{mm}$.
Balance~: $u = 0.01/\sqrt{12} = \qty{0.003}{g}$.
Voltmètre~: $a = 0.005 \times 4.80 = \qty{0.024}{V}$,
$u = 0.024/\sqrt3 = \qty{0.014}{V}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:5}
$v = k F^\alpha \mu^\beta$~: $L T^{-1} = (M L T^{-2})^\alpha (M L^{-1})^\beta$,
d'où $\alpha + \beta = 0$, $\alpha - \beta = 1$, $-2\alpha = -1$~:
$\alpha = \tfrac12$, $\beta = -\tfrac12$, $v = k\sqrt{F/\mu}$ (en fait $k = 1$).
Son~: $P/\rho$ a pour \omterm{def:b1:units-dimensions:dimension}{dimension} $M L^{-1} T^{-2} / (M L^{-3}) = L^2 T^{-2}$,
donc $v = k\sqrt{P/\rho}$ ($k = \sqrt\gamma \approx 1.2$ pour l'air).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:6}
Somme $= \qty{3620}{ms}$, $\bar t = \qty{452.5}{ms}$. Écarts
$-0.5, -5.5, 8.5, 2.5, -3.5, 5.5, -8.5, 1.5$~; la somme de leurs carrés vaut $226$~;
$s = \sqrt{226/7} = \qty{5.7}{ms}$~; $u(\bar t) = 5.7/\sqrt8 = \qty{2.0}{ms}$.
Résultat~: $t = \qty{452.5\pm2.0}{ms}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:7}
$R = 4.80/0.0124 = \qty{387}{\ohm}$. Incertitudes relatives $0.03/4.80 =
0.63\%$ et $0.2/12.4 = 1.6\%$~; composées, $\sqrt{0.63^2 + 1.6^2} = 1.7\%$,
$u(R) = \qty{7}{\ohm}$~: $R = \qty{387\pm7}{\ohm}$. L'intensité domine
--- améliorer d'abord l'ampèremètre (ou son calibre).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:8}
$z = 7/\sqrt{16 + 9} = 1.4 \leq 2$~: compatibles.
Le troisième contre le premier~: $z = 9/\sqrt{4 + 16} = 2.0$~: à la
limite, tout juste compatibles. Le troisième contre le second~:
$z = 16/\sqrt{4 + 9} = 4.4$~: incompatibles --- au moins l'un des deux
souffre d'une erreur systématique ou d'une incertitude sous-estimée.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:9}
$V = \pi d^3/6 = \pi \times 25.40^3/6 = \qty{8580}{mm^3}$.
$u(V)/V = 3\,u(d)/d = 3 \times 0.02/25.40 = 0.24\%$,
$u(V) = \qty{20}{mm^3}$~: $V = \qty{8.58\pm0.02}{cm^3}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:10}
$F = k\eta^a R^b v^c$~: en $M$, $a = 1$~; en $T$, $-a - c = -2$, $c = 1$~;
en $L$, $-a + b + c = 1$, $b = 1$~: $F = k\eta R v$ (Stokes, $k = 6\pi$).
Avec $\rho$ à la place~: en $M$, $a = 1$~; en $T$, $-c = -2$, $c = 2$~; en $L$,
$-3 + b + 2 = 1$, $b = 2$~: $F = k\rho R^2 v^2$.
Goutte de pluie~: $\eta R v = 1.8\times10^{-5} \times 10^{-3} \times 5 \approx
\qty{1e-7}{N}$~; $\rho R^2 v^2 = 1.2 \times 10^{-6} \times 25 =
\qty{3e-5}{N}$, trois cents fois plus grand~: c'est le régime inertiel
(grande vitesse) qui s'applique --- le rapport $\rho v R/\eta \approx 330$
est le nombre de Reynolds.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:11}
$\bar I = \qty{5.0}{mA}$, $\bar U = \qty{1.000}{V}$~; écarts en $I$~:
$-3, -1, 1, 3$~; en $U$~: $-0.59, -0.22, 0.22, 0.59$.
$\sum (I_i - \bar I)(U_i - \bar U) = 3.98$, $\sum (I_i - \bar I)^2 = 20$~:
$a = \qty{0.199}{V/mA} = \qty{199}{\ohm}$, $b = 1.000 - 0.199 \times 5 =
\qty{0.005}{V}$. Avec $u(U) = \qty{0.02}{V}$ par point, une ordonnée à
l'origine de \qty{0.005}{V} est largement dans l'incertitude~:
compatible avec zéro, le conducteur est ohmique, $R \approx \qty{199}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:12}
$\ell = G^a h^b c^c$~: en $M$, $-a + b = 0$~; en $T$, $-2a - b - c = 0$~; en $L$,
$3a + 2b + c = 1$. Donc $b = a$, $c = -3a$, $2a = 1$~:
$\ell_P = \sqrt{Gh/c^3} = \sqrt{4.42\times10^{-44}/2.7\times10^{25}}
= \qty{4.1e-35}{m}$~; $t_P = \ell_P/c = \qty{1.4e-43}{s}$~;
$m_P = \sqrt{hc/G} = \sqrt{1.99\times10^{-25}/6.67\times10^{-11}}
= \qty{5.5e-8}{kg}$. La longueur de Planck est vingt \omterm{def:b1:units-dimensions:oom}{ordres de grandeur}
sous un proton~; la masse de Planck, \qty{55}{\micro g}, est un grain de
poussière --- énorme pour une particule, minuscule pour un être humain
($10^9$ fois plus léger).
\end{solution}

\begin{solution}{pb:b1:units-dimensions:1}
\textbf{1.} $\sqrt{L/(L T^{-2})} = \sqrt{T^2} = T$.

\textbf{2.} $\tau = k\ell^\alpha m^\beta g^\gamma$ donne
$T = L^{\alpha+\gamma} M^\beta T^{-2\gamma}$, donc $\beta = 0$~: aucune
combinaison de $\ell$ et de $g$ ne peut compenser une masse.

\textbf{3.} $\theta_0$ est \omterm{def:b1:units-dimensions:dimension}{sans dimension}, donc tout facteur $f(\theta_0)$
est invisible aux \omterm{def:b1:units-dimensions:dimension}{dimensions}. Il faut garder l'amplitude petite (sous
$10^\circ$ environ), où $f \approx 1$ à mieux que $0.2\%$ près.

\textbf{4.} $g = 4\pi^2\ell/\tau^2$.

\textbf{5.} $\ell = 99.0 + 1.00 = \qty{100.0}{cm} = \qty{1.000}{m}$.

\textbf{6.} $u = \qty{1}{mm}/\sqrt{12} = \qty{0.29}{mm}$.

\textbf{7.} Alignement~: $5/\sqrt3 = \qty{2.9}{mm}$~; au total
$\sqrt{0.29^2 + 2.9^2} = \qty{2.9}{mm}$ --- la lecture du ruban est
négligeable. $u(\ell)/\ell = 0.29\%$.

\textbf{8.} L'erreur de temps de réaction (environ \qty{0.1}{s}) est la
même que l'on chronomètre une période ou dix, donc son effet sur une
période est dix fois plus petit. Répéter révèle la dispersion (d'où $s$)
et divise l'incertitude de la moyenne par $\sqrt8$.

\textbf{9.} Somme $= \qty{160.80}{s}$, $\overline{t_{10}} = \qty{20.100}{s}$.
Écarts $0.02, -0.05, 0.08, -0.01, 0.05, -0.08, 0.01, -0.02$~; la somme de
leurs carrés vaut $\num{188e-4}$~; $s = \sqrt{188\times10^{-4}/7} = \qty{0.052}{s}$.

\textbf{10.} $u(\overline{t_{10}}) = 0.052/\sqrt8 = \qty{0.018}{s}$;
$\tau = \qty{2.0100}{s}$, $u(\tau) = \qty{0.0018}{s}$.

\textbf{11.} $u(\tau)/\tau = 0.09\%$, trois fois plus petit que
$u(\ell)/\ell = 0.29\%$.

\textbf{12.} $g = 4\pi^2 \times 1.000/2.0100^2 = 39.48/4.040 =
\qty{9.772}{m/s^2}$.

\textbf{13.} $g \propto \ell\,\tau^{-2}$:
$\dfrac{u(g)}{g} = \sqrt{\left(\dfrac{u(\ell)}{\ell}\right)^2
+ \left(2\dfrac{u(\tau)}{\tau}\right)^2} = \sqrt{0.29^2 + 0.18^2}\,\%
= 0.34\%$.

\textbf{14.} $u(g) = 0.0034 \times 9.772 = \qty{0.033}{m/s^2}$:
$g = \qty{9.77\pm0.03}{m/s^2}$.

\textbf{15.} $z = \abs{9.772 - 9.81}/0.033 = 1.2 \leq 2$~: compatibles.

\textbf{16.} La longueur ($0.29\%$ contre $0.18\%$ pour le terme de
période). Repérer les deux extrémités à $\pm\qty{1}{mm}$ près ramène
$u(\ell)/\ell$ à $0.06\%$ et $u(g)/g$ à
$\sqrt{0.06^2 + 0.18^2} = 0.19\%$~; doubler la longueur divise par deux
$u(\ell)/\ell$ et allonge $\tau$, ce qui aide les deux termes.
Chronométrer vingt périodes au lieu de dix ne divise par deux que le
terme de période~: $u(g)/g = \sqrt{0.29^2 + 0.09^2} = 0.30\%$ --- un
maigre gain tant que la longueur est le goulet d'étranglement.

\textbf{17.} Non~: un décalage commun à toutes les lectures est une
\emph{erreur systématique} (un biais)~; le bilan couvre la dispersion
aléatoire et les tolérances annoncées, pas un décalage constant. On le
détecte en changeant de méthode ou d'instrument, ou --- partie~V --- par
une ordonnée à l'origine non nulle de la droite $\tau^2$--$\ell$.

\textbf{18.} $\tau^2 = (4\pi^2/g)\,\ell$ est une droite passant par
l'origine~: la linéarité et l'ordonnée à l'origine nulle se testent
toutes deux à l'œil et par les \omterm{prop:b1:units-dimensions:regression}{moindres carrés}~; $\tau \propto \sqrt\ell$, non.

\textbf{19.} $\bar\ell = \qty{0.800}{m}$~; $\overline{\tau^2} = 16.16/5 =
\qty{3.232}{s^2}$.

\textbf{20.} Écarts en $\ell$~: $-0.4, -0.2, 0, 0.2, 0.4$~; en
$\tau^2$~: $-1.612, -0.822, 0.018, 0.808, 1.608$. Somme des produits $= 1.614$,
$\sum(\ell_i - \bar\ell)^2 = 0.400$~: $a = \qty{4.035}{s^2/m}$,
$b = 3.232 - 4.035 \times 0.800 = \qty{0.004}{s^2}$.

\textbf{21.} $g = 4\pi^2/a = 39.48/4.035 = \qty{9.78}{m/s^2}$.

\textbf{22.} $b \approx 0$, comme le modèle le prévoit. Un $b$
nettement non nul trahirait un décalage systématique $\delta$ sur toutes
les longueurs ($\tau^2 = a(\ell_{\text{meas}} - \delta)$ donne $b = -a\delta$),
par exemple un centre de bille mal repéré --- ou un effet d'amplitude.

\textbf{23.} $u(g)/g = u(a)/a = 0.05/4.035 = 1.2\%$, $u(g) =
\qty{0.12}{m/s^2}$: $g = \qty{9.78\pm0.12}{m/s^2}$.

\textbf{24.} $z = \abs{9.772 - 9.784}/\sqrt{0.033^2 + 0.12^2} = 0.1$~:
parfaitement compatibles. La mesure unique soignée est la plus précise~;
la droite valide le modèle et protège d'un décalage sur les longueurs.

\textbf{25.} $\tau = 2\pi\sqrt{10/9.8} = \qty{6.3}{s}$~; une seule
oscillation chronométrée avec un temps de réaction d'environ
\qty{0.2}{s} donne $u(\tau)/\tau \approx
3\%$, donc $u(g)/g \approx 6\%$, alors que $u(\ell)/\ell$ tombe à
$0.03\%$~: vingt fois pire que la partie~IV. Le gain sur la longueur est
gâché par le chronométrage d'une période unique. Meilleure stratégie~:
pendule long \emph{et} périodes nombreuses, en répétant --- dix
oscillations du pendule de \qty{10}{m}, huit fois, donnent
$u(\tau)/\tau \approx 0.03\%$ et $u(g)/g \approx 0.07\%$.
\end{solution}
