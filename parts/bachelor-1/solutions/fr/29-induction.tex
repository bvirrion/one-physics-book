\section*{Chapitre \ref{ch:b1:induction} --- Induction électromagnétique et applications}

\begin{solution}{exo:b1:induction:1}
$\Phi = NBS\cos30^\circ = 100 \times 0.2 \times 10^{-3} \times 0.866 = \qty{1.7e-2}{Wb}$~;
$\langle e\rangle = \Delta\Phi/\Delta t = \qty{1.7}{V}$ en \qty{10}{ms}, \qty{17}{V} en
\qty{1}{ms}. Le courant circule de façon à recréer le flux qui disparaît~: son
champ est dirigé selon l'ancien $\vect B$.
\end{solution}

\end{solution}

\begin{solution}{exo:b1:induction:3}
$L = \mu_0n^2S\ell = 4\pi \times 10^{-7} \times 10^6 \times 4 \times 10^{-4} \times 0.2 =
\qty{0.10}{mH}$~; $W = \tfrac12LI^2 = \qty{1.3}{mJ}$~; $\tau = L/R = \qty{0.1}{ms}$.
$B = \mu_0nI = \qty{6.3}{mT}$, $B^2/2\mu_0 = \qty{15.7}{J/m^3}$ sur
$S\ell = \qty{8e-5}{m^3}$~: \qty{1.3}{mJ}.
\end{solution}

\begin{solution}{exo:b1:induction:4}
$N_2/N_1 = 12/230 = 0.052$~; $N_1 = 50/0.052 = 960$ spires~; $i_1 = 2 \times 0.052 =
\qty{0.10}{A}$. Un courant permanent donne un flux permanent~: pas de $\dd\varphi/\dd t$, pas de
tension au secondaire --- et le primaire, en court-circuit, brûlerait.
\end{solution}

\begin{solution}{exo:b1:induction:5}
(a) Le flux (selon l'approche) croît~: le courant crée un champ qui s'y
oppose --- son propre pôle nord fait face au nord de l'aimant --- et l'anneau
est repoussé. (b) Un courant dont le champ s'oppose à la croissance~; l'anneau,
dipôle anti-aligné sur $\vect B$, est poussé vers les champs plus
faibles. (c) En tombant, le flux croît~: même courant qu'en (a), l'anneau est
freiné. (d) Chaque tronçon de tube en avant de l'aimant voit un flux croître,
en arrière décroître~; les deux courants induits freinent l'aimant~; à la
vitesse pour laquelle la \omterm{thm:b1:magnetostatics:laplace}{force de Laplace} équilibre le poids, l'aimant tombe
régulièrement et son \omterm{def:b1:work-and-energy:conservative}{énergie potentielle} perdue devient de la chaleur Joule
dans le tube.
\end{solution}

\begin{solution}{exo:b1:induction:6}
$E_{\max} = NBS\omega = 200 \times 2 \times 10^{-3} \times 0.1 \times 314 = \qty{12.6}{V}$, soit
\qty{8.9}{V} efficaces~; $I_{\max} = \qty{1.26}{A}$, $\Gamma_{\max} = NBSI_{\max} = \qty{0.050}{N.m}$
(quand $\sin\omega t = 1$)~; $\langle P\rangle = E_{\max}^2/2R = \qty{7.9}{W}$ $=
\langle\Gamma\omega\rangle = \tfrac12 \times 0.05 \times 314$~; quand $e = 0$ le courant est
nul et le couple s'annule --- le couple pulse à \qty{100}{Hz}.
\end{solution}

\begin{solution}{exo:b1:induction:7}
$e = Bav$, $i = Bav/R$, $F = B^2a^2v/R = 10\,v$ (en N). $m\,\dd v/\dd t = -B^2a^2v/R$~:
$v = v_0\eu^{-t/\tau}$, $\tau = mR/B^2a^2 = \qty{0.10}{s}$~; distance $v_0\tau =
\qty{10}{cm}$ (si la région de champ est assez longue). L'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique},
\qty{0.5}{J}, est dissipée en $Ri^2$ dans la spire.
\end{solution}

\begin{solution}{exo:b1:induction:8}
Flux du champ du solénoïde, $\mu_0n_1i_1$, à travers les $N_2$ spires~: $M =
\mu_0n_1N_2S_2 = 4\pi \times 10^{-7} \times 2000 \times 50 \times 2 \times 10^{-4} = \qty{25}{\micro H}$.
Rampe~: $e = M\,\dd i_1/\dd t = \qty{2.5}{mV}$~; à \qty{10}{kHz}, \qty{1}{A}~: $e_{\max} =
M\omega I = 25 \times 10^{-6} \times 6.3 \times 10^4 = \qty{1.6}{V}$. $M$ est symétrique~:
le même nombre donne le flux que la petite bobine envoie dans le
solénoïde --- bien plus difficile à calculer directement.
\end{solution}

\begin{solution}{exo:b1:induction:9}
$W = B^2V/2\mu_0 = 2.25/(2.51 \times 10^{-6}) = \qty{0.90}{MJ}$~; $L = 2W/I^2 =
\qty{7.2}{H}$~; $h = W/mg = \qty{92}{m}$~; hélium~: $9 \times 10^5/2600 = \qty{350}{L}$.
Champ terrestre~: $(5 \times 10^{-5})^2/2\mu_0 = \qty{1e-3}{J/m^3}$, soit \qty{50}{mJ} dans
la pièce.
\end{solution}

\begin{solution}{exo:b1:induction:10}
(a) $E - B\ell v = Ri$ (la f.c.é.m. s'oppose à la source), $m\,\dd v/\dd t =
B\ell i$~: $m\,\dd v/\dd t = B\ell(E - B\ell v)/R$. (b) $v_\infty = E/B\ell$, $\tau =
mR/B^2\ell^2$. (c) On multiplie l'équation électrique par $i$~: $Ei = Ri^2 +
B\ell vi$, et $B\ell vi = Fv$ est la puissance mécanique~; \omterm{def:b1:heat-engines:efficiency}{rendement} $Fv/Ei =
B\ell v/E = v/v_\infty$ --- $100\%$ seulement à vide, où rien n'est
fourni. (d) $v_\infty = 2/(0.5 \times 0.2) = \qty{20}{m/s}$, $\tau = 0.05 \times
0.1/0.01 = \qty{0.5}{s}$. (e) $F = B\ell I = 1 \times 0.1 \times 10^6 = \qty{1e5}{N}$
(en prenant $\ell = \qty{10}{cm}$), $a = \qty{1e7}{m/s^2}$, $v = \sqrt{2a \times 5} =
\qty{1e4}{m/s}$ --- dix kilomètres par seconde, la promesse et le
problème des canons électromagnétiques (les rails s'érodent).
\end{solution}

\begin{solution}{exo:b1:induction:11}
(a) Au rayon $r$, le métal se déplace à $v = \omega r$~; $\vect v\wedge\vect B$ est
radial, de norme $\omega rB$~: $e = \int_0^R\omega Br\,\dd r = \tfrac12B\omega R^2$.
(b) $\omega = 314$~: $e = 0.5 \times 1 \times 314 \times 0.01 = \qty{1.6}{V}$. (c) Une
seule «~spire~», donc quelques volts au plus~; dans \qty{1}{m\ohm}~: \qty{1.6}{kA}, couple
$= P/\omega = ei/\omega = 2500/314 = \qty{8}{N.m}$. (d) Le circuit balai--rayon--
bord--balai balaie l'aire $\tfrac12R^2\omega\,\dd t$ pendant $\dd t$~: $\dd\Phi/\dd t =
\tfrac12BR^2\omega$ --- le flux à travers le circuit mobile et déformable.
\end{solution}

\begin{solution}{exo:b1:induction:12}
(a) $e_2 = -M\,\dd i_1/\dd t$, de valeur maximale $M\omega I_1 = 10^{-6} \times 1.57 \times 10^5 \times 30 =
\qty{4.7}{V}$. (b) $I_2 = 4.7/0.01 = \qty{470}{A}$ en crête~; $\langle P\rangle = R_pI_2^2/2
= \qty{1.1}{kW}$. (c) La f.é.m. est $\propto\omega$~: une fréquence élevée donne une
f.é.m. utile avec un $M$ modeste~; le verre est isolant, rien n'y circule
--- la chaleur naît dans le métal. (d) Le courant de la casserole induit dans
la bobine $M\omega I_2 = 10^{-6} \times 1.57 \times 10^5 \times 470 = \qty{74}{V}$ en crête,
en phase avec $i_1$ (le courant de la casserole, purement résistif, ne
retarde pas sur $e_2$, et $e_2$ retarde sur $i_1$ de \qty{90}{\degree}\ldots\ de sorte que la
f.é.m. réfléchie vaut $-M\,\dd i_2/\dd t$, en opposition de phase avec la
tension \emph{motrice} de $i_1$)~: le générateur fournit $\tfrac12 \times 74 \times 30 =
\qty{1.1}{kW}$ de plus --- exactement ce que dissipe la casserole.
\end{solution}

\begin{solution}{pb:b1:induction:1}
\textbf{1.} $\Phi = NBS\cos\omega t$, $e = NBS\omega\sin\omega t$~; $NBS = 300 \times
0.3 \times 3 \times 10^{-4} = \qty{0.027}{Wb}$~; $\omega = v/r = 5/0.01 = \qty{500}{rad/s}$~:
\omterm{def:b1:wave-propagation:signal}{amplitude} \qty{13.5}{V}, fréquence $\omega/2\pi = \qty{80}{Hz}$.

\textbf{2.} $I_{\text{eff}} = 13.5/\sqrt2/16 = \qty{0.60}{A}$~; lampe $RI^2 = \qty{4.3}{W}$
sous \qty{7.2}{V}~: une lampe \qty{6}{V}, \qty{3}{W} est surchargée et ne durera
pas.

\textbf{3.} Mécanique $=$ électrique~: $(R + r_c)I_{\text{eff}}^2 = 16 \times 0.36 =
\qty{5.7}{W}$~; $F = P/v = 5.7/5 = \qty{1.1}{N}$ --- la moitié du frottement de
roulement~; le cycliste le sent.

\textbf{4.} Le couple de Laplace du courant induit s'oppose à la rotation
(Lenz)~; l'énergie de la lampe vient des jambes du cycliste, \qty{5.7}{W} de leur part.

\textbf{5.} $\underline Z = R + r_c + jL\omega$~; $I = NBS\omega/\sqrt{(R + r_c)^2 + L^2\omega^2}$~;
pour $L\omega \gg R + r_c$, $I \to NBS/L = 0.027/0.02 = \qty{1.35}{A}$. À $70\%$~: $L\omega
= 0.7(R + r_c)/\sqrt{1 - 0.49} = 15.7$, $\omega = \qty{784}{rad/s}$, $v = \qty{7.8}{m/s}
= \qty{28}{km/h}$.

\textbf{6.} Sans $L$~: $e_{\max} = 3 \times 13.5 = \qty{40.5}{V}$, $I_{\text{eff}} =
\qty{1.8}{A}$, \qty{39}{W} dans la lampe --- elle grille. Avec $L$~: $I = 40.5/\sqrt{256 +
900} = \qty{1.2}{A}$ en crête, \qty{8.5}{W}~: chaud, mais dix fois moins.

\textbf{7.} À \qty{9}{km/h}~: $e_{\max} = 6.75$, $|Z| = \sqrt{256 + 25} = 16.8$, $I = 0.40$,
$P = \tfrac12 \times 0.16 \times 12 = \qty{1.0}{W}$~: faible. À \qty{36}{km/h}~: $e_{\max} = 27$,
$|Z| = 25.6$, $I = 1.05$, $P = \qty{6.6}{W}$. Quatre fois la vitesse, $2.6$ fois
le courant~: l'inductance régule.

\textbf{8.} Seul le \omterm{def:b1:non-inertial-frames:frames}{mouvement relatif} compte~: le flux à travers la bobine fixe
varie de la même façon, même f.é.m. Aucun contact glissant (balai) n'est alors
nécessaire pour extraire le courant d'une bobine tournante.

\textbf{9.} Sans mouvement, pas de variation de flux, pas de lumière --- d'où le
\omterm{def:b1:potential-capacitors:capacitance}{condensateur} ou la petite batterie que l'on charge en roulant (l'éclairage permanent).

\textbf{10.} Chaque élément de fil est perpendiculaire au $\vect B$ radial~:
$F = B\ell i$, axiale. En se déplaçant à $v$, chaque élément voit $\vect v\wedge\vect B$
selon le fil~: $e = -B\ell v$, opposée au courant qui produit le
mouvement (en \omterm{thm:b1:dc-circuits:kirchhoff}{convention récepteur}, une tension $B\ell v$).

\textbf{11.} $u = Ri + L\,\dd i/\dd t + B\ell v$~; $m\,\dd v/\dd t = -kx - hv + B\ell i$.

\textbf{12.} $\underline i = (\underline u - B\ell\underline v)/(R + jL\omega)$~; $\underline v\,(jm\omega + h +
k/j\omega) = B\ell\,\underline i$~; on substitue et l'on résout en $\underline v$.

\textbf{13.} $i \approx (u - B\ell v)/R$ donne $m\dot v + (h + B^2\ell^2/R)v + kx =
B\ell u/R$~: amortissement supplémentaire $25/8 = \qty{3.1}{N.s/m}$. $f_0 = \tfrac1{2\pi}\sqrt{k/m} =
\qty{50}{Hz}$~; $Q = \sqrt{km}/h_{\text{tot}}$~: $3.2$ en circuit ouvert, $0.77$ avec
l'amplificateur.

\textbf{14.} $F = \qty{5.0}{N}$, $a = F/m = \qty{500}{m/s^2}$, $x = a/\omega^2 =
\qty{13}{\micro m}$, $v = a/\omega = \qty{0.080}{m/s}$~; $B\ell v = \qty{0.40}{V}$ contre
$Ri = \qty{8}{V}$~: $5\%$.

\textbf{15.} À courant imposé, $v = F/h = \qty{5}{m/s}$ (une \omterm{def:b1:wave-propagation:signal}{amplitude} de
\qty{16}{mm} --- une belle excursion pour \qty{1}{A})~; $B\ell v =
\qty{25}{V}$, trois fois $Ri$~: l'amplificateur doit fournir \qty{33}{V} pour
imposer \qty{1}{A} à la résonance --- c'est le pic d'\omterm{def:b1:sinusoidal-impedance:impedance}{impédance}. Un amplificateur
en \omterm{def:b1:dc-circuits:sources}{source de tension} laisserait au contraire l'amortissement électrique tenir la membrane.

\textbf{16.} $\tfrac12RI^2 = \qty{4.0}{W}$~; $\tfrac12hv^2 = \tfrac12 \times 0.0064 =
\qty{3.2}{mW}$~: \omterm{def:b1:heat-engines:efficiency}{rendement} $0.08\%$ --- les haut-parleurs sont des radiateurs
qui chuchotent.

\textbf{17.} Au-dessus de la résonance, $a = F/m = B\ell i/m$ ne dépend pas de
la fréquence à courant donné, de sorte que la pression, $\propto a$, est plate~:
la bande contrôlée par la masse est la bande utile.

\textbf{18.} Champ radial~: la force est axiale et la même quelle que soit la
position de la bobine dans l'entrefer. Un tweeter doit rester contrôlé par sa
masse jusqu'à \qty{20}{kHz} avec une excursion minuscule~: bobine et membrane
légères~; et une petite membrane rayonne les aigus dans toutes les directions.

\textbf{19.} À basse fréquence, $v$ est petite et $|Z| \approx R = \qty{8}{\ohm}$~; à
\qty{50}{Hz}~: $u = Ri + B\ell v$ avec $v = B\ell i/h$~: $Z = R + B^2\ell^2/h = 8 + 25 =
\qty{33}{\ohm}$~; à \qty{10}{kHz}~: $|R + jL\omega| = |8 + 31j| = \qty{32}{\ohm}$. Un
pic à la résonance, un plancher plat à \qty{8}{\ohm}, une remontée en $L\omega$.

\textbf{20.} $e = B\ell v = 5 \times 10^{-3} = \qty{5}{mV}$.

\textbf{21.} $i = 5 \times 10^{-3}/608 = \qty{8.2}{\micro A}$~; $F = B\ell i = \qty{41}{\micro N}$,
opposée à $v$~: $F = [B^2\ell^2/(R + R_L)]\,v$, soit un amortissement de \qty{0.041}{N.s/m}~;
la puissance $Fv = \qty{4e-8}{W}$ est prélevée au son et finit dans les
deux résistances.

\textbf{22.} Carcasse~: $B^2\ell_f^2/R_f = 1 \times 0.01/10^{-3} = \qty{10}{N.s/m}$, dix
fois $h$~: elle éteindrait la membrane et gaspillerait la puissance en chaleur~;
une fente brise la spire fermée.

\textbf{23.} Bornes ouvertes~: seul $h$ amortit, $Q = 3.2$, elle résonne. Bornes
court-circuitées~: la f.c.é.m. fait passer un courant dans $R$, l'amortissement
$B^2\ell^2/R$ s'ajoute, $Q = 0.77$~: elle s'arrête aussitôt. (À essayer.)

\textbf{24.} Sur une période, les énergies stockées reprennent leurs valeurs~:
$\langle ui\rangle = \langle Ri^2\rangle + \langle hv^2\rangle$ --- entrée électrique $=$
chaleur Joule dans la bobine $+$ pertes mécaniques (son rayonné et
frottement de la suspension)~; le terme de couplage $B\ell iv$ figure dans les
deux équations avec des signes opposés et se compense.

\textbf{25.} Dynamo~: Faraday sur une bobine tournante --- \qty{13.5}{V} à
\qty{18}{km/h}, courant plafonné à \qty{1.35}{A} par sa propre inductance.
Haut-parleur~: \omterm{thm:b1:magnetostatics:laplace}{force de Laplace} et f.c.é.m. --- \qty{3.1}{N.s/m} d'amortissement
électrique, $0.08\%$ de \omterm{def:b1:heat-engines:efficiency}{rendement}. Microphone~: la même bobine à l'envers ---
\qty{5}{mV} par millimètre par seconde.
\end{solution}
