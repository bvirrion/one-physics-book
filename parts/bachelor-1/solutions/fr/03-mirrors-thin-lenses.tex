\section*{Chapitre \ref{ch:b1:mirrors-thin-lenses} --- Miroirs et lentilles minces}

\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$~; $1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. À \qty{60}{cm}~: $\overline{SA'} = \qty{-30}{cm}$,
réelle, $\gamma = -0.5$. À \qty{30}{cm}~: $\overline{SA'} = \qty{-60}{cm}$,
réelle, $\gamma = -2$. À \qty{10}{cm}~: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$, $\overline{SA'} = \qty{+20}{cm}$ derrière le miroir, virtuelle,
$\gamma = +2$ (droite, agrandie).
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:2}
$\overline{OA} = \qty{-20}{cm}$~: $1/\overline{OA'} = 1/5 - 1/20 = 3/20$,
$\overline{OA'} = \qty{+6.7}{cm}$, réelle et renversée, $\gamma = -1/3$.
$\overline{OA} = \qty{-3.0}{cm}$~: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$,
$\overline{OA'} = \qty{-7.5}{cm}$, virtuelle et droite, $\gamma = +2.5$.
\end{solution}

$V = 3.0 - 1.0 = \qty{+2.0}{\delta}$, $f' = \qty{50}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:4}
$\overline{FA} = \qty{-15}{cm}$~: $\overline{F'A'} = -f'^2/\overline{FA}
= +100/15 = \qty{6.7}{cm}$ au-delà de $F'$, soit $\overline{OA'} = \qty{16.7}{cm}$~;
$\gamma = f'/\overline{FA} = -0.67$. Par Descartes~: $\overline{OA} =
\qty{-25}{cm}$, $1/\overline{OA'} = 1/10 - 1/25 = 3/50$,
$\overline{OA'} = \qty{16.7}{cm}$, $\gamma = 16.7/(-25) = -0.67$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}
$\overline{SF} = \qty{+1.0}{m}$, $\overline{SA} = \qty{-20}{m}$~:
$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ derrière
le miroir (virtuelle, droite)~; $\gamma = -0.95/(-20) = +0.048$, \omterm{def:b1:mirrors-thin-lenses:image}{image}
haute de \qty{7.1}{cm}. Le cerveau juge la distance sur la taille
apparente~: une voiture vue vingt fois plus petite à \qty{0.95}{m} se lit
comme une voiture lointaine --- plus loin que \qty{20}{m}.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$,
$\overline{OA'} = \qty{51.3}{mm}$~: l'objectif s'éloigne de \qty{1.3}{mm}
du capteur. $\gamma = 51.3/(-2000) = -0.026$. Pour $\abs\gamma = 24/1800
= 1/75$~: $\overline{OA} = f'(1/\gamma - 1) = 50 \times (-76) =
\qty{-3.8}{m}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$~; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$~;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$.
L'\omterm{def:b1:mirrors-thin-lenses:image}{image} est renversée ($\gamma < 0$), haut--bas et gauche--droite~: la
diapositive s'introduit tournée de $180^\circ$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
Objet dans le plan focal $\Rightarrow$ les rayons issus de chaque point
quittent la \omterm{def:b1:mirrors-thin-lenses:lens}{lentille} parallèles~; le \omterm{prop:b1:rays-reflection-refraction:mirror}{miroir plan} les renvoie parallèles
(direction symétrique)~; la \omterm{def:b1:mirrors-thin-lenses:lens}{lentille} focalise un faisceau parallèle dans
son plan focal, au point symétrique du point source par rapport à
l'axe~: \omterm{def:b1:mirrors-thin-lenses:image}{image} dans le plan de l'objet, renversée, $\gamma = -1$. Incliner
le miroir de $\alpha$ fait tourner le faisceau de retour de $2\alpha$ et
décale l'\omterm{def:b1:mirrors-thin-lenses:image}{image} latéralement de $2\alpha f'$, toujours nette dans le même plan.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:9}
$f' = (100^2 - 40^2)/(4 \times 100) = \qty{21}{cm}$. Distance minimale
$4f' = \qty{84}{cm}$, avec $\gamma = -1$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ \omterm{def:b1:mirrors-thin-lenses:image}{image} l'infini en $F_1'$, \qty{20}{cm} après $L_1$, soit \qty{5}{cm}
après $L_2$~: c'est un objet virtuel, $\overline{O_2A_1} = \qty{+5}{cm}$.
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$~: \omterm{def:b1:mirrors-thin-lenses:image}{image} finale \qty{10}{cm}
après $L_2$. Hauteur de l'\omterm{def:b1:mirrors-thin-lenses:image}{image} intermédiaire $h_1 = f_1'\alpha = 20\alpha$ (cm)~;
$\gamma_2 = 10/5 = 2$, donc $h = 40\alpha$~: \omterm{def:b1:mirrors-thin-lenses:lens}{distance focale} équivalente
\qty{40}{cm}, dans un barillet long de $15 + 10 = \qty{25}{cm}$ --- un téléobjectif.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ donne $\overline{SA'} =
3\overline{SA}$~; les deux sont réels, en avant~: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, et la distance écran--lampe vaut $2x = \qty{80}{cm}$~:
$x = \qty{40}{cm}$. Le miroir est à \qty{40}{cm} de la lampe et \qty{120}{cm}
de l'écran~; $1/\overline{SF} = -1/40 - 1/120 = -1/30$~:
$R = 2 \times 30 = \qty{60}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$~; diamètre de l'\omterm{def:b1:mirrors-thin-lenses:image}{image} $f'\theta =
\qty{4.6}{mm}$. Puissance recueillie $10^3 \times \pi(0.05)^2 = \qty{7.9}{W}$~;
aire de l'\omterm{def:b1:mirrors-thin-lenses:image}{image} $\pi(2.3\times10^{-3})^2 = \qty{1.7e-5}{m^2}$~; éclairement
$\qty{4.7e5}{W/m^2}$, soit $470$ fois le soleil direct~: le papier
noircit puis s'enflamme --- c'est la loupe ardente. En général le gain
vaut $(D/f'\theta)^2$.
\end{solution}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$~; axe orienté dans le sens de la
lumière, distances algébriques comptées depuis $O$.

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$~:
$1/(D - p) + 1/p = 1/f_1'$, soit $p^2 - Dp + Df_1' = 0$~; racines réelles si et seulement si
$D^2 - 4Df_1' \geq 0$, c'est-à-dire $D \geq 4f_1'$.

\textbf{3.} La somme des racines vaut $D$~: $p_1 + p_2 = D$, donc $p_2 = D - p_1$ ---
les distances objet et \omterm{def:b1:mirrors-thin-lenses:image}{image} s'échangent, les positions sont symétriques
par rapport au milieu. $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$, d'où $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} Seuls interviennent le déplacement $d$ du barillet entre les
deux réglages nets et la distance objet--écran $D$~: ni l'un ni l'autre
n'exige de savoir où se trouve $O$ à l'intérieur du barillet.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$~; donc
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ et $u(f_1') = \qty{0.14}{cm}$~:
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ ou \qty{60}{cm}~: $\gamma =
-60/20 = -3$ (diapositive près de la \omterm{def:b1:mirrors-thin-lenses:lens}{lentille}, \omterm{def:b1:mirrors-thin-lenses:image}{image} agrandie~: le
projecteur) et $\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$~; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$~: l'\omterm{def:b1:mirrors-thin-lenses:image}{image réelle} est tournée de $180^\circ$
autour de l'axe, donc la diapositive s'introduit tournée de $180^\circ$
(à l'envers et inversée gauche--droite).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$~: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$~: la \omterm{def:b1:mirrors-thin-lenses:lens}{lentille} se rapproche de
\qty{0.7}{mm} de la diapositive. $\gamma = 1000/(-15.23) = -65.7$~: \omterm{def:b1:mirrors-thin-lenses:image}{image}
large de \qty{2.36}{m}.

\textbf{13.} Différentions Descartes~: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, donc $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$~: l'\omterm{def:b1:mirrors-thin-lenses:image}{image} sort
de la netteté à mesure que le film chauffe --- d'où les diapositives
montées sous verre, ou une mise au point qui suit le film.

\textbf{15.} Triangles semblables entre l'ouverture et le point \omterm{def:b1:mirrors-thin-lenses:image}{image}~:
flou $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$~; vu de \qty{5}{m}, cela
sous-tend $\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$~: imperceptible.

\textbf{16.} Aire de la diapositive $8.64\times10^{-4}\ \unit{m^2}$~: $E_{\text{diapo}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$~; aire de l'écran $\gamma^2$
fois plus grande, $\qty{2.16}{m^2}$~: $E_{\text{écran}} = \qty{0.93}{W/m^2}$~;
même puissance, aires dans le rapport $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$~: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$~: l'\omterm{def:b1:mirrors-thin-lenses:image}{image}
intermédiaire se trouve \qty{29.3}{cm} \emph{au-delà} de $L_2$ --- un
objet virtuel pour $L_2$ (les rayons issus de $L_1$ convergent vers elle).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$~;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$~; $\gamma_2 = 1200/29.3 =
41.0$~; $\gamma = -60.9$.

\textbf{20.} Largeur $60.9 \times 36 = \qty{2.19}{m}$. Avec $L_1$ seule~:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, largeur
\qty{2.86}{m}.

\textbf{21.} Faisceau incliné de $\alpha$~: $L_1$ le focalise à la hauteur $h_1 = f_1'\alpha$
dans son plan focal, situé à $\overline{O_2A_1} = f_1' - e$ de $L_2$~;
$L_2$ \omterm{def:b1:mirrors-thin-lenses:image}{image} ce point en $\overline{O_2A'}$ avec $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ et le grandit de $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$, donc $h = \gamma_2 f_1'\alpha$ et $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, dont l'inverse est la formule
annoncée. $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ et $15/19.6 = 0.77$. Seulement de façon
approchée, parce que la diapositive est à distance finie et que les deux
portées se mesurent depuis des \omterm{def:b1:mirrors-thin-lenses:lens}{lentilles} différentes~; pour un objet à
l'infini le rapport serait exact.

\textbf{23.} Objet virtuel à $p > 0$ au-delà d'une \omterm{def:b1:mirrors-thin-lenses:lens}{lentille divergente}~:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, positif (\omterm{def:b1:mirrors-thin-lenses:image}{image
réelle}) si et seulement si $p < \abs{f_2'}$.

\textbf{24.} Même puissance sur une \omterm{def:b1:mirrors-thin-lenses:image}{image} d'aire $(2.19/2.86)^2 = 0.59$
fois plus petite~: éclairement $\times 1.7$.

\textbf{25.} Le convertisseur multiplie la \omterm{def:b1:mirrors-thin-lenses:lens}{distance focale} par
$f_{\mathrm{eq}}/f_1' = 1.3$~; à portée donnée, l'\omterm{def:b1:mirrors-thin-lenses:image}{image} est $1.3$ fois
plus petite dans chaque direction et $1.7$ fois plus lumineuse --- un
objectif longue portée.
\end{solution}
