\begin{solution}{exo:b1:wave-propagation:6}
$\lambda = \qty{0.40}{m}$. Sur le segment, à la distance $x$ du
haut-parleur 1, $\delta = (2.0 - x) - x = 2.0 - 2x$~; silence pour $\delta = \pm\lambda/2,

\end{solution}

\begin{solution}{exo:b1:wave-propagation:10}

\begin{solution}{exo:b1:wave-propagation:11}
Le mode $n$ a la forme $\sin(n\pi x/L)$. Imposer un nœud en $x = L/3$
exige $\sin(n\pi/3) = 0$, c'est-à-dire $n$ multiple de $3$~: fréquences $3f_1,
6f_1, \dots$ Pincer en $L/2$ donne à la corde un déplacement maximal en
ce point, auquel le mode $n$ ne peut contribuer que si $\sin(n\pi/2) \neq 0$~:
les modes pairs, qui ont un nœud en $L/2$, ne reçoivent rien.
\end{solution}

\begin{solution}{exo:b1:wave-propagation:12}
$\lambda = c/f = \qty{3.0}{m}$. Déphasage $2\pi(2d)/\lambda + \pi$~;
destructif lorsqu'il vaut $(2m + 1)\pi$~: $2d/\lambda = m$,
$d = m\lambda/2 = 0, 1.5, 3.0, \dots$ m~; constructif lorsque $2d/\lambda =
m + \tfrac12$~: $d = 0.75, 2.25, \dots$ m. Les parois et le plafond d'un
tunnel réfléchissent l'onde~: il s'établit un régime stationnaire, avec
des zones mortes toutes les demi-longueurs d'onde sur le trajet de la voiture.
\end{solution}

\begin{solution}{pb:b1:wave-propagation:1}
\textbf{1.} $[F/\mu] = M L T^{-2}/(M L^{-1}) = L^2 T^{-2}$~: le carré
d'une vitesse.

\textbf{2.} Une onde qui va vers $-x$ dépend de $t + x/c$, c'est-à-dire de
$\omega t + kx$~; $k = \omega/c = 2\pi f/c$.

\textbf{3.} $c = \sqrt{0.90/1.0\times10^{-3}} = \qty{30}{m/s}$~;
$\lambda = c/f = \qty{0.60}{m}$~; $k = 2\pi/\lambda = \qty{10.5}{rad/m}$.

\textbf{4.} $\Delta\varphi = kd = 10.5 \times 0.15 = \qty{1.6}{rad} = \pi/2$.

\textbf{5.} $v_{\max} = A\omega = 2.0\times10^{-3} \times 2\pi \times 50 =
\qty{0.63}{m/s}$, cinquante fois moins que $c$~: les points de la corde
se déplacent lentement tandis que la forme file.

\textbf{6.} Elle va vers $+x$~: sa \omterm{def:b1:wave-propagation:signal}{phase} est $\omega t - kx$. À
l'extrémité fixe, le déplacement total est nul à tout instant~: $s_i(0, t) + s_r(0, t)
= A\cos\omega t + (-A)\cos\omega t = 0$.

\textbf{7.} $\cos(\omega t + kx) - \cos(\omega t - kx) =
-2\sin(\omega t)\sin(kx)$, donc $s = -2A\sin(kx)\sin(\omega t)$.

\textbf{8.} Nœuds~: $\sin kx = 0$, $x = n\lambda/2$~; ventres $x = (n +
\tfrac12)\lambda/2$~; nœuds voisins distants de $\lambda/2$.

\textbf{9.} $\sin kL = 0$~: $L = n\lambda/2$, c'est-à-dire $f_n = nc/2L$.

\textbf{10.} $n = 2L/\lambda = 2.4/0.60 = 4$~: quatre fuseaux, nombre
entier --- il y a résonance.

\textbf{11.} $3$ fuseaux~: $\lambda = 2L/3 = \qty{0.80}{m}$, $c = \lambda f
= \qty{40}{m/s}$, $F = \mu c^2 = \qty{1.6}{N}$. $6$ fuseaux~: $\lambda =
\qty{0.40}{m}$, $c = \qty{20}{m/s}$, $F = \qty{0.40}{N}$.

\textbf{12.} $c = 2Lf/n = 2 \times 1.20 \times 50/3 = \qty{40}{m/s}$,
$\mu = F/c^2 = 1.6/1600 = \qty{1.0e-3}{kg/m} = \qty{1.0}{g/m}$.

\textbf{13.} $\mu = mgn^2/(4L^2f^2)$~: $u(\mu)/\mu = \sqrt{(u_m/m)^2 +
(2u_L/L)^2} = \sqrt{0.61^2 + 0.33^2}\,\% = 0.7\%$~; c'est la masse qui limite.

\textbf{14.} Le fondamental. $c = 2Lf_1$~: mi grave $2 \times 0.650 \times
82.4 = \qty{107}{m/s}$~; mi aigu \qty{428}{m/s}.

\textbf{15.} $F = \mu c^2$~: mi grave $6.0\times10^{-3} \times 107^2 =
\qty{69}{N}$~; mi aigu $4.0\times10^{-4} \times 428^2 = \qty{73}{N}$ ---
tensions comparables. Abaisser $f$ d'un facteur $4$ à $L$ et $\mu$ fixés
demanderait $16$ fois moins de tension~: une corde molle et bourdonnante.
Le filage augmente $\mu$ à la place.

\textbf{16.} $f_n = 2^{n/12}f_1$ et $f \propto 1/L$~: $L_n = 2^{-n/12}L$,
distance au sillet $L - L_n = L(1 - 2^{-n/12})$~: case 1 \qty{36.5}{mm},
case 5 \qty{163}{mm}, case 7 \qty{216}{mm}, case 12 \qty{325}{mm}.

\textbf{17.} Chaque écart vaut $2^{-1/12} = 0.944$ fois le précédent.

\textbf{18.} Facteur $2^{3/12} = 1.19$ (trois demi-tons)~; longueur
$650 \times 2^{-3/12} = \qty{547}{mm}$.

\textbf{19.} En effleurant en $L/2$~: modes pairs seulement, hauteur $2f_1 =
\qty{165}{Hz}$~; en $L/3$~: multiples de $3$, hauteur $3f_1 = \qty{247}{Hz}$.

\textbf{20.} Près d'une extrémité, $\sin(n\pi x/L) \approx n\pi x/L$
croît avec $n$~: les modes élevés sont relativement plus excités ---
son clair. Au-dessus de la rosace (près du milieu), les modes pairs sont
faibles et le fondamental domine --- son rond.

\textbf{21.} $3f_1 = \qty{330}{Hz}$ bat à \qty{2}{Hz} avec $2f_2$~:
$f_2 = 166$ ou \qty{164}{Hz}, rapport $1.509$ ou $1.491$ (la quinte
tempérée vaut $2^{7/12} = 1.498$).

\textbf{22.} La quatrième harmonique, $4f$. Trois battements~: $4f = 437$ ou
\qty{443}{Hz}, donc $f = 109.25$ ou \qty{110.75}{Hz}.

\textbf{23.} Les battements s'accélèrent quand on tend, donc la corde
était trop haute~: \qty{110.75}{Hz}. Elle doit baisser de
$0.75/110.75 = 0.68\%$, c'est-à-dire la tension de $1.4\%$.

\textbf{24.} $f = (n/2L)\sqrt{F/\mu}$, donc $\ln f = \tfrac12\ln F + \text{const}$
et $\dd f/f = \tfrac12\,\dd F/F$.

\textbf{25.} $f_1 = \dfrac{1}{2L}\sqrt{\dfrac{F}{\mu}}$~: la longueur fixe
$\lambda_1 = 2L$, la tension et la masse fixent $c$, les cases
multiplient $L$ par $2^{-n/12}$, les battements révèlent $f - f_{\text{réf}}$.
Derrière tout cela, une seule relation~: $\lambda f = c$, avec la
condition d'\omterm{thm:b1:wave-propagation:standing}{onde stationnaire} $L = n\lambda/2$.
\end{solution}
