\section*{Chapitre \ref{ch:b1:angular-momentum} --- Moment cinétique}

\begin{solution}{exo:b1:angular-momentum:1}
$Fd = 50 \times 0.30 = \qty{15}{N.m}$~; $Fd\sin 60^\circ = \qty{13}{N.m}$~;
dans le prolongement du manche, la droite d'action passe par l'écrou~: $0$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:2}
$L = mrv = 5.97\times10^{24} \times 1.50\times10^{11} \times 2.98\times10^4 =
\qty{2.7e40}{kg.m^2/s}$~; $\dd A/\dd t = L/2m = rv/2 = \qty{2.2e15}{m^2/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:3}
$r = \ell\sin 30^\circ = \qty{0.50}{m}$, $\omega = \qty{3.4}{rad/s}$~:
$L_z/m = r^2\omega = \qty{0.84}{m^2/s}$. La droite d'action de la tension
passe par le point de suspension (\omterm{def:b1:angular-momentum:moment}{moment} nul par rapport à tout axe qui
y passe)~; le poids est parallèle à l'axe vertical (\omterm{def:b1:angular-momentum:moment}{moment} nul par
rapport à lui)~: $L_z$ se conserve.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:4}
$J_1 = 4.0 + 2 \times 3.0 \times 0.64 = \qty{7.8}{kg.m^2}$, $J_2 = 4.0 +
2 \times 3.0 \times 0.04 = \qty{4.2}{kg.m^2}$~; $\omega_2 = \omega_1J_1/J_2 =
1.85\omega_1$~: \qty{3.7}{} tours par seconde. $E_k = L^2/2J$~: rapport
$J_1/J_2 = 1.85$~; ce sont les muscles, qui ramènent les bras contre la
tendance centrifuge, qui fournissent ce travail.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:5}
$L_\Delta = m\ell^2\dot\theta$~; tension~: \omterm{def:b1:angular-momentum:moment}{moment} nul (elle rencontre
l'axe)~; poids~: $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{bras de levier} $\ell\sin\theta$,
effet de rappel). D'où $m\ell^2\ddot\theta = -mg\ell\sin\theta$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:6}
$r_pv_p = r_av_a$~: $v_p/v_a = r_a/r_p = 1.034$~; avec $(v_p + v_a)/2 \approx
\qty{29.8}{km/s}$~: $v_p = \qty{30.3}{km/s}$, $v_a = \qty{29.3}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:7}
$L = mrv$ se conserve~: $v_2 = v_1r_1/r_2 = \qty{4.0}{m/s}$, $\omega_2 = v_2/r_2
= \qty{16}{rad/s}$ (contre \qty{4}{rad/s}). $E_1 = \qty{0.40}{J}$, $E_2 =
\qty{1.6}{J}$~: la traction fournit \qty{1.2}{J}. La tension est radiale,
mais le déplacement du palet a désormais une composante radiale (il
spirale vers l'intérieur)~: le travail $\vect T\cdot\dd\vect\ell$ n'est plus nul.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:8}
\omterm{def:b1:angular-momentum:moment}{Moments} par rapport au pivot (longueurs depuis le pivot)~: enfant $30 \times 1.5 =
45$ (à gauche)~; poids de la planche en son milieu, à $0.5$ à droite~: $20 \times
0.5 = 10$ (à droite)~; second enfant à $x$ à droite~: $45x$. Équilibre~:
$45 = 10 + 45x$, $x = \qty{0.78}{m}$. Le pivot supporte le poids total,
$95 \times 9.81 = \qty{932}{N}$.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:9}
Masses~: $m_1a = m_1g - T_1$, $m_2a = T_2 - m_2g$~; poulie~: $J\dot\omega =
(T_1 - T_2)R$ avec $a = R\dot\omega$. En additionnant~: $a = (m_1 - m_2)g/(m_1 + m_2 +
J/R^2) = 9.81/(3.0 + 1.0) = \qty{2.5}{m/s^2}$ (contre \qty{3.3}{m/s^2})~;
$T_1 = m_1(g - a) = \qty{14.7}{N}$, $T_2 = m_2(g + a) = \qty{12.3}{N}$ ---
inégales, puisque la poulie réclame un \omterm{def:b1:angular-momentum:moment}{moment} résultant.
\end{solution}

\begin{solution}{exo:b1:angular-momentum:10}
$r_1v_1 = r_2v_{\perp 2}$~: $v_{\perp 2} = 40 \times 0.5/10 = \qty{2.0}{km/s}$.
La \omterm{cor:b1:angular-momentum:central}{loi des aires} ne donne que la composante perpendiculaire~; la
composante radiale demande la conservation de l'énergie (\cref{ch:b1:central-forces}).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:11}
$J\omega$ constant avec $J \propto R^2$~: $T' = T(R'/R)^2 = 25 \times 86400
\times (10^4/7\times10^8)^2 = \qty{4.4e-4}{s}$. Vitesse à l'équateur $2\pi R'/T'
= \qty{1.4e8}{m/s}$, la moitié de la vitesse de la lumière~: l'effondrement
ne peut pas garder tout le \omterm{def:b1:angular-momentum:L}{moment cinétique} --- il en évacue beaucoup (et
la relativité entre en jeu).
\end{solution}

\begin{solution}{exo:b1:angular-momentum:12}
$L_z = mr^2\omega$, et $r$ varie~: $\dd L_z/\dd t = 2mr\dot r\omega \neq 0$.
Ce doit être le \omterm{def:b1:angular-momentum:moment}{moment} de la réaction transversale $N$ de la tige~: $rN =
2mr\dot r\omega$, donc $N = 2m\dot r\omega$ --- la tige pousse la perle
latéralement pendant qu'elle s'éloigne (le terme de Coriolis du
\cref{ch:b1:non-inertial-frames}).
\end{solution}

\begin{solution}{pb:b1:angular-momentum:1}
\textbf{1.} $m\omega^2r = GMm/r^2$~: $\omega = \sqrt{GM/r^3} = \sqrt{3.98\times
10^{14}/5.66\times10^{25}} = \qty{2.65e-6}{rad/s}$, de période $2\pi/\omega =
\qty{2.37e6}{s} = \qty{27.4}{d}$.

\textbf{2.} $L_{\text{orb}} = 7.35\times10^{22} \times 1.475\times10^{17}
\times 2.65\times10^{-6} = \qty{2.9e34}{kg.m^2/s}$.

\textbf{3.} $v = \sqrt{GM/r}$, $L = mrv = m\sqrt{GMr}$~; $\dd L/\dd r =
\tfrac12 m\sqrt{GM/r} = L/2r$.

\textbf{4.} $J = 0.33 \times 5.97\times10^{24} \times 4.06\times10^{13} =
\qty{8.0e37}{kg.m^2}$~; $\Omega = \qty{7.29e-5}{rad/s}$~; $L_{\text{rot}} =
\qty{5.8e33}{kg.m^2/s}$, le cinquième de $L_{\text{orb}}$.

\textbf{5.} $J_m \approx 0.4 \times 7.35\times10^{22} \times 3.0\times10^{12} =
\qty{8.9e34}{kg.m^2}$, multiplié par $\omega$~: \qty{2.4e29}{kg.m^2/s}, soit $10^{-5}$
de celui de l'orbite~: négligeable.

\textbf{6.} L'attraction du Soleil s'exerce sur le centre de masse du
système (pas de \omterm{def:b1:angular-momentum:moment}{moment} par rapport à lui au premier ordre)~; son effet
de marée sur le \omterm{def:b1:angular-momentum:moment}{couple} est faible~: le $L$ total se conserve à la
précision voulue ici.

\textbf{7.} Le bourrelet le plus proche, en avance sur la ligne
Terre--Lune, est tiré par la Lune avec une force dont la composante
tangentielle s'oppose à la rotation terrestre.

\textbf{8.} \omterm{def:b1:angular-momentum:moment}{Moment} négatif sur la rotation~: la Terre ralentit, le jour
s'allonge.

\textbf{9.} Le bourrelet tire la Lune en avant~: \omterm{def:b1:angular-momentum:moment}{moment} positif sur
l'orbite, $L_{\text{orb}}$ croît.

\textbf{10.} $L_{\text{orb}} \propto \sqrt r$ croît, donc $r$ croît~; $v \propto
1/\sqrt r$ décroît.

\textbf{11.} La traction vers l'avant fournit un travail positif, qui
élève l'énergie de la Lune~; or une orbite plus haute est une orbite
plus lente~: l'énergie passe en altitude (potentielle), plus que ne se
perd d'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique}.

\textbf{12.} Vers le verrouillage par les marées de la Terre~: jour égal
au mois, plus de bourrelet en avance, plus de \omterm{def:b1:angular-momentum:moment}{couple}.

\textbf{13.} $\Delta\Omega/\Omega = -\qty{2.3e-3}{s}/\qty{86164}{s} =
\num{-2.7e-8}$ par siècle.

\textbf{14.} $\Delta L_{\text{rot}} = L_{\text{rot}}\Delta\Omega/\Omega =
\qty{-1.55e26}{kg.m^2/s}$ par siècle.

\textbf{15.} $\Delta L_{\text{orb}} = +\qty{1.55e26}{kg.m^2/s}$ par siècle.

\textbf{16.} $\Delta r = 2r\,\Delta L/L = 2 \times 3.84\times10^8 \times 1.55
\times10^{26}/2.9\times10^{34} = \qty{4.1}{m}$ par siècle, soit \qty{4.1}{cm/yr}~:
proche des \qty{3.8}{cm/yr} mesurés (l'allongement du jour comporte aussi
une part non due aux marées).

\textbf{17.} $\Delta T/T = \tfrac32\Delta r/r = 1.6\times10^{-8}$~: $2.37\times
10^6 \times 1.6\times10^{-8} = \qty{0.04}{s}$ par siècle.

\textbf{18.} $\Delta E_{\text{rot}} = L_{\text{rot}}\Delta\Omega = 5.8\times10^{33}
\times 7.29\times10^{-5} \times (-2.7\times10^{-8}) = \qty{-1.1e22}{J}$ par
siècle.

\textbf{19.} $\Delta E_{\text{orb}} = GMm\Delta r/2r^2 = 2.93\times10^{37}
\times 4.1/(2 \times 1.475\times10^{17}) = \qty{+4.1e20}{J}$ par siècle.

\textbf{20.} Bilan $-1.1\times10^{22} + 4\times10^{20} \approx \qty{-1.06e22}{J}$
par siècle~: $3.4\times10^{12}\ \unit{W}$, dissipés en chaleur par le
frottement des marées dans les océans et dans la Terre solide.

\textbf{21.} Un cinquième de la puissance de l'humanité, un quatorzième
du flux géothermique --- un petit chauffage, mais permanent.

\textbf{22.} $L_{\text{tot}} = 3.5\times10^{34}$. En $r = 5.5\times10^8$~:
$\omega = \sqrt{GM/r^3} = \qty{1.55e-6}{rad/s}$, $J\omega = 1.2\times10^{32}$,
$m\sqrt{GMr} = 7.35\times10^{22} \times 4.68\times10^{11} = 3.44\times10^{34}$~;
somme $3.45\times10^{34}$~: cohérent. Période $2\pi/\omega = \qty{4.1e6}{s}
= \qty{47}{d}$.

\textbf{23.} Les marées solaires continuent de ralentir la Terre
au-dessous du rythme orbital de la Lune, si bien que le \omterm{def:b1:angular-momentum:moment}{couple}
Terre--Lune s'inverserait~; et de toute façon le Soleil aura évolué bien
avant (l'échelle de temps est de plusieurs dizaines de milliards d'années).

\textbf{24.} La Terre a soulevé des marées sur la Lune, qui ont freiné
sa rotation propre jusqu'à ce qu'une face reste tournée vers nous~;
étant plus légère et voisine d'un partenaire plus lourd, la Lune s'est
verrouillée bien plus vite.

\textbf{25.} Se conserve~: le \omterm{def:b1:angular-momentum:L}{moment cinétique} total. S'échange~: du
\omterm{def:b1:angular-momentum:L}{moment cinétique}, de la rotation de la Terre vers l'orbite de la Lune,
environ \qty{1.6e26}{SI} par siècle. Se dissipe~: de l'\omterm{thm:b1:work-and-energy:em}{énergie
mécanique}, quelques térawatts, en chaleur de marée.
\end{solution}
