\section*{Chapitre \ref{ch:b1:optical-instruments} --- Instruments d'optique et œil}

\begin{solution}{exo:b1:optical-instruments:1}
À l'infini~: $f' = \qty{17}{mm}$, $V = 1/0.017 = \qty{58.8}{\delta}$.
À \qty{25}{cm}~: $V = 58.8 + 1/0.25 = \qty{62.8}{\delta}$, $f' = \qty{15.9}{mm}$.
Amplitude \qty{4.0}{\delta}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:2}
Myope~: l'infini doit être imagé sur le \omterm{def:b1:optical-instruments:punctum}{punctum remotum}, $f' = \qty{-40}{cm}$,
$V = \qty{-2.5}{\delta}$. Hypermétrope~: un objet à \qty{25}{cm} doit
être imagé (virtuellement) à \qty{1.0}{m}~: $V = 1/\overline{OA'} - 1/\overline{OA}
= -1 + 4 = \qty{+3.0}{\delta}$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:3}
$G = 25/4.0 = 6.3$~; timbre dans le plan focal, à \qty{4.0}{cm} de la
\omterm{def:b1:mirrors-thin-lenses:lens}{lentille}. Détail de \qty{2}{mm}~: $\theta' = 2/40 = \qty{0.05}{rad}$,
contre $2/250 = \qty{0.008}{rad}$ à \qty{25}{cm}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:4}
$G = 900/25 = 36$ et $900/9 = 100$~; longueurs de tube $f_1' + f_2' =
\qty{925}{mm}$ et \qty{909}{mm}~; Lune~: $36 \times 0.52^\circ = 19^\circ$
et $52^\circ$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:5}
$\gamma_1 = -160/4 = -40$~; $G_2 = 250/20 = 12.5$~; $G = 500$.
$\overline{F_1A} = -f_1'^2/\Delta = -16/160 = \qty{-0.10}{mm}$~: objet à
\qty{4.1}{mm} de l'\omterm{def:b1:optical-instruments:microscope}{objectif}. L'objet doit rester à cette distance de
l'\omterm{def:b1:optical-instruments:microscope}{objectif}, donc tout le tube se déplace de \qty{10}{\micro m} --- c'est
la vis micrométrique.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:6}
\omterm{def:b1:optical-instruments:eye}{Œil} nu~: $1.5/3\times10^{-4} = \qty{5}{km}$. Lunette~: $\theta_{\min}
= 1.22 \times 5.5\times10^{-7}/0.100 = \qty{6.7e-6}{rad}$~; les phares
restent séparés jusqu'à $1.5/6.7\times10^{-6} \approx \qty{220}{km}$ (en
principe --- l'air et la courbure interviennent bien avant). $G_r = 3\times10^{-4}/
6.7\times10^{-6} = 45$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Exposition $\propto N^2$~: $(8/2)^2 = 16$ fois
plus longue, $16/500 \approx 1/30$~\unit{s}. On y gagne de la \omterm{def:b1:optical-instruments:camera}{profondeur
de champ} (et moins d'aberrations), au prix d'un flou de bougé ou d'un trépied.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$~; net de $H/2 =
\qty{1.9}{m}$ à l'infini. Préréglé sur $H$, l'appareil n'a plus de mise
au point à faire~: la photo se prend à l'instant où la scène a lieu.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$~: \omterm{prop:b1:optical-instruments:telescope}{cercle oculaire} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$, toute
la lumière entre dans l'\omterm{def:b1:optical-instruments:eye}{œil}. $7\times50$~: \qty{7.1}{mm}, à la mesure de
la pupille nocturne, et $(50/30)^2 = 2.8$ fois plus de lumière
recueillie~: la paire du crépuscule. $20\times30$~: \omterm{prop:b1:optical-instruments:telescope}{cercle oculaire}
\qty{1.5}{mm}, image terne, champ étroit et tremblement de la main
grossi vingt fois.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
\omterm{def:b1:mirrors-thin-lenses:lens}{Lentille} de contact~: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$.
Lunettes~: le \omterm{def:b1:optical-instruments:punctum}{punctum remotum} est à $25 - 1.5 = \qty{23.5}{cm}$ du verre,
donc $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- plus forte. Le verre doit
former l'image sur le \omterm{def:b1:optical-instruments:punctum}{punctum remotum} de l'\omterm{def:b1:optical-instruments:eye}{œil}, dont la distance au
verre dépend de l'endroit où le verre est placé.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afocal~: $F_1' = F_2$, écart $f_1' + f_2' = \qty{20}{cm}$ (plus court que
les \qty{40}{cm} d'une lunette de Kepler de même $G$). $G = -30/(-10) =
+3$~: image droite. \omterm{prop:b1:optical-instruments:telescope}{Cercle oculaire} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$~: virtuel, à l'intérieur du tube, là
où l'\omterm{def:b1:optical-instruments:eye}{œil} ne peut se placer~; derrière l'\omterm{def:b1:optical-instruments:microscope}{oculaire}, l'\omterm{def:b1:optical-instruments:eye}{œil} ne capte que les
faisceaux qui l'atteignent par hasard --- d'où le champ étroit, reproche
classique fait aux jumelles de théâtre.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$~; $\Delta m = 2.5\log_{10}820 = 7.3$~: magnitude limite
$13.3$. Le flux étant $\propto 1/r^2$, la lampe est vue
$\sqrt{820} = 29$ fois plus loin.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$~: écart $f_1' + f_2' = \qty{1225}{mm}$
(\omterm{def:b1:optical-instruments:microscope}{oculaire} de \qty{25}{mm}) ou \qty{1206}{mm} (\qty{6}{mm}).

\textbf{2.} $L_1$~: $\overline{O_1A_1} = f_1'$ (image en $F_1'$). $L_2$~:
$\overline{O_2A_1} = -f_2'$, donc $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$~: image à l'infini.

\textbf{3.} Le faisceau converge dans le plan focal commun à la hauteur $h =
f_1'\alpha$ (rayon passant par $O_1$)~; de là, le rayon passant par $O_2$
fixe la direction d'émergence, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ et $-200$~; Lune~: $25^\circ$ et $104^\circ$.

\textbf{5.} Le haut et le bas n'ont pas de sens dans le ciel~; une
lunette terrestre ajoute une paire de prismes redresseurs (ou un relais).

\textbf{6.} Dans le plan focal de $L_1$, de diamètre $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} $O_1$ vu comme objet en $\overline{O_2O_1} = -(f_1' + f_2')$~:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$~;
grandissement $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
diamètre $D/\abs G$.

\textbf{8.} \qty{25}{mm}~: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, diamètre $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}~:
\qty{6.0}{mm} derrière, diamètre \qty{0.75}{mm}.

\textbf{9.} Tout rayon entré dans l'\omterm{def:b1:optical-instruments:microscope}{objectif} passe par le \omterm{prop:b1:optical-instruments:telescope}{cercle
oculaire}~: un \omterm{def:b1:optical-instruments:eye}{œil} placé là reçoit tout le champ. Plus en arrière, il
n'intercepte que les faisceaux centraux~: le champ se réduit à un trou
de serrure.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$~: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$~;
à l'\omterm{def:b1:optical-instruments:eye}{œil} nu, $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$~; l'\omterm{def:b1:optical-instruments:microscope}{oculaire} de \qty{25}{mm}
($48$) en est le plus proche, un peu en dessous~; celui de \qty{6}{mm}
($200$) est trois fois au-dessus.

\textbf{14.} À $G = 200$, la tache de diffraction est vue sous $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$~: visiblement floue. La
lumière de la Lune s'étale sur une aire $(200/48)^2 = 17$ fois plus
grande sur la \omterm{def:b1:optical-instruments:eye}{rétine}~: image plus terne.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$~: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Au-delà, c'est la turbulence qui limite la résolution
depuis le sol (sauf optique adaptative)~; un $D$ plus grand recueille
encore plus de lumière et atteint des objets plus faibles.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$~: magnitude $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
soit $460$ fois ce que reçoit l'\omterm{def:b1:optical-instruments:eye}{œil} nu. L'image rétinienne est $\abs G^2$
fois plus grande en aire~: rapport des luminances $460/G^2 = 0.20$
($G = 48$), $0.012$ ($G = 200$) --- soit $(D/\abs G)^2/d_p^2$, le carré
du rapport du \omterm{prop:b1:optical-instruments:telescope}{cercle oculaire} à la pupille.

\textbf{19.} Pour un objet étendu, la lumière recueillie croît comme
$D^2$ mais l'aire de l'image comme $G^2 \geq (D/d_p)^2$~: la luminance
ne peut au mieux qu'égaler celle de l'\omterm{def:b1:optical-instruments:eye}{œil} nu (\omterm{prop:b1:optical-instruments:telescope}{cercle oculaire} $=$
pupille). Une étoile n'est pas résolue~: toute sa lumière tombe dans une
seule tache de diffraction, donc son éclat croît comme $(D/d_p)^2$.

\textbf{20.} Un point vu sous l'angle $\beta$ s'image à la hauteur
$f_1'\beta$ dans le plan focal~; le diaphragme laisse passer
$\abs{f_1'\beta} \leq \phi/2$~: champ total $\phi/f_1'$ (petits angles).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$~: la Lune entière
tient. $5/1200 = 0.24^\circ$~: elle ne tient pas.

\textbf{22.} Champ apparent $\phi/f_2'$~: $20/25 = \qty{0.8}{rad} = 46^\circ$~;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$~: $3420/15 = \qty{230}{s}$, près de
quatre minutes~; $0.24^\circ = 864''$~: \qty{58}{s}.

\textbf{24.} L'\omterm{def:b1:optical-instruments:microscope}{oculaire} de \qty{25}{mm} ($G = 48$, proche de $G_r$,
lumineux, Lune entière) pour le disque~; celui de \qty{6}{mm} ($G = 200$)
pour regarder les cratères de plus près, plus terne et plus flou. Le
nombre de l'instrument~: des cratères jusqu'à \qty{1.7}{km}.

\textbf{25.} Mêmes $f_1'$ et $D$~: mêmes $G$, champs, \omterm{prop:b1:optical-instruments:telescope}{cercles oculaires},
résolution et lumière (à l'obstruction centrale près). Gain~: plus
d'aberration chromatique. Changement pratique~: le foyer se trouve
devant le miroir, si bien qu'un petit miroir secondaire doit renvoyer le
faisceau vers l'\omterm{def:b1:optical-instruments:microscope}{oculaire}.
\end{solution}
