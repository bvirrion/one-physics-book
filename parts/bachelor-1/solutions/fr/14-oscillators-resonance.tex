\section*{Chapitre \ref{ch:b1:oscillators-resonance} --- Oscillateurs mécaniques~: amortissement et résonance}

\begin{solution}{exo:b1:oscillators-resonance:1}
$\omega_0 = \sqrt{80/0.20} = \qty{20}{rad/s}$, $f_0 = \qty{3.2}{Hz}$, $T_0 =
\qty{0.31}{s}$~; $v_{\max} = A\omega_0 = \qty{1.0}{m/s}$~; $E = \tfrac12 kA^2 =
\qty{0.10}{J}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:2}
$T = 2\pi\sqrt{\ell/g}$~: \qty{2.0}{s} sur Terre, \qty{4.9}{s} sur la Lune.
La période du système masse--ressort, $2\pi\sqrt{m/k}$, ne fait pas
intervenir $g$~: elle est inchangée.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:3}
$\tau = 2Q/\omega_0 = 100/(4\pi) = \qty{8.0}{s}$~; l'énergie, $\propto \eu^{-2t/\tau}$,
est divisée par deux en $t = \tau\ln2/2 = \qty{2.8}{s}$~; $5\%$ après $3\tau = \qty{24}{s}$,
soit environ $48 \approx Q$ oscillations.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:4}
$X_0 = F_0/k = \qty{1.0}{cm}$~; à la résonance, $\approx QX_0 = \qty{10}{cm}$~;
en $3\omega_0$~: $X_0/\sqrt{64 + 0.09} = \qty{0.13}{cm}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:5}
$v = Lf_0 = 8.0 \times 1.3 = \qty{10}{m/s}$ (\qty{37}{km/h}). À
\qty{60}{km/h}~: $f = 16.7/8.0 = \qty{2.1}{Hz}$, $x_r = 1.6$~: $X/X_g =
\sqrt{(1 + 5.2)/(2.4 + 5.2)} = 0.90$ --- à peine isolée~; l'isolation
demande $x_r \gg 1$ (à \qty{120}{km/h}, $x_r = 3.2$~: $0.45$).
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:6}
Le travail d'une poussée n'est positif que lorsqu'elle agit dans le sens
du mouvement~; poussée à chaque période, toutes les poussées
s'ajoutent~; sur tout autre rythme, elles alternent. Régime établi~:
$2\pi E/Q = \qty{25}{J}$, $E = \qty{80}{J}$~;
$mg\ell(1 - \cos\theta_{\max}) = 80$~: $1 - \cos\theta_{\max} = 0.109$,
$\theta_{\max} = 27^\circ$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:7}
$\delta = \ln(1/0.9) = 0.105$, $Q = \pi/\delta = 30$~; $\Delta f = f_0/Q =
\qty{0.067}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:8}
Déplacement~: en phase ($\varphi = 0$) bien en dessous, $-\pi/2$ à la
résonance, $-\pi$ (en opposition) bien au-dessus. Vitesse à la
résonance~: en phase avec la force. La puissance vaut $\vect F\cdot\vect v$~:
c'est la phase de la vitesse, non celle du déplacement, qui décide du
travail fourni par l'excitation.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:9}
$V = \omega X = \omega(F_0/m)/\sqrt{(\omega_0^2 - \omega^2)^2 + (\omega\omega_0/Q)^2}$~;
en divisant numérateur et dénominateur par $\omega\omega_0/Q$~: $V = (F_0Q/m\omega_0)/
\sqrt{1 + Q^2(\omega_0/\omega - \omega/\omega_0)^2}$, et $F_0Q/m\omega_0 = F_0/\alpha$.
La racine est $\geq 1$, et vaut $1$ exactement en $\omega = \omega_0$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:10}
$P = F_0^2/2\alpha = \qty{10}{W}$~; $V_{\max} = F_0/\alpha = \qty{20}{m/s}$,
$E = \tfrac12 mV_{\max}^2 = \qty{40}{J}$~; $E/P = \qty{4.0}{s}$ et $Q/\omega_0 =
m/\alpha = \qty{4.0}{s}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:11}
$3\tau = \qty{10}{s}$~: $\tau = \qty{3.3}{s}$, $Q = \omega_0\tau/2 = 2\pi \times
440 \times 3.3/2 \approx 4600$~; $\Delta f = f_0/Q = \qty{0.1}{Hz}$. Il ne
répond qu'à une fréquence, et n'en émet qu'une~: un étalon bon à
quelques $10^4$\textsuperscript{es} près.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:12}
$m\ddot x = mg - \rho_wgS(h + x) = -\rho_wgSx$ (puisque $mg = \rho_wgSh$)~:
$\omega_0^2 = \rho_wgS/m = g/h$~; $T = 2\pi\sqrt{0.10/9.81} = \qty{0.63}{s}$.
La masse et la masse volumique du liquide n'interviennent qu'à travers
la profondeur d'équilibre $h$.
\end{solution}

\begin{solution}{pb:b1:oscillators-resonance:1}
\textbf{1.} Forces~: le poids (compensé par l'allongement statique), le
ressort $-ku$, l'amortisseur $-\alpha\dot u$~; $m\ddot x = -ku - \alpha\dot u$ avec $\ddot x =
\ddot x_g + \ddot u$.

\textbf{2.} $\ddot u + (\omega_0/Q)\dot u + \omega_0^2u = -\ddot x_g$, avec $\omega_0 =
\sqrt{k/m}$ et $Q = \sqrt{km}/\alpha$.

\textbf{3.} $k = m\omega_0^2 = \pi^2 = \qty{9.9}{N/m}$~; $\alpha = m\omega_0/Q =
3.14/0.70 = \qty{4.5}{N.s/m}$.

\textbf{4.} $u = -a_g/\omega_0^2$~: la masse s'affaisse d'une quantité fixe
--- l'instrument est un accéléromètre à fréquence nulle (et un inclinomètre).

\textbf{5.} Le déplacement relatif $u(t)$.

\textbf{6.} $-\omega^2\underline U + j\omega(\omega_0/Q)\underline U + \omega_0^2\underline U
= \omega^2X_g$~; on divise par $\omega_0^2$.

\textbf{7.} $U/X_g = x_r^2/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$.

\textbf{8.} $U \to X_g$~: la masse ne bouge pas dans le \omterm{thm:b1:newton-dynamics:laws}{référentiel
galiléen} (son inertie la maintient, le ressort étant trop souple pour
l'entraîner)~; le bâti se déplace autour d'elle, et la plume enregistre
le déplacement du sol --- c'est un sismomètre.

\textbf{9.} $U \approx x_r^2X_g = \omega^2X_g/\omega_0^2 = a_g/\omega_0^2$~: un
accéléromètre.

\textbf{10.} $U = QX_g = 0.7X_g$~: pas de pic~; les deux régimes se
raccordent en douceur et l'instrument ne sonne pas.

\textbf{11.} Pour $Q = 0.7$~: une droite de pente $2$ qui monte jusqu'à
$1$ au voisinage de $x_r = 1$, puis un plateau à $1$~; pour $Q = 5$~: la
même chose avec un pic de hauteur $5$ en $x_r = 1$.

\textbf{12.} $\underline U \to -X_g$~: $u = -x_g$, la masse immobile
pendant que le bâti se déplace.

\textbf{13.} $\Delta\ell = g/\omega_0^2 = 9.81/(2\pi \times 0.02)^2 = \qty{620}{m}$~:
absurde.

\textbf{14.} $\ell = g/\omega_0^2 = \qty{620}{m}$.

\textbf{15.} Porte de jardin~: le pendule oscille autour d'un axe presque
vertical, si bien qu'une infime fraction de $g$ seulement le rappelle et
qu'un bras court donne une longue période. Contre-réaction de force~: une
bobine maintient la masse fixe, et le courant nécessaire est
proportionnel à l'accélération du sol --- la raideur devient
électronique et réglable.

\textbf{16.} $\Delta\ell = 9.81/\pi^2 = \qty{0.99}{m}$~: réalisable.

\textbf{17.} $x_r = 0.2$~: $U/X_g = 0.04/\sqrt{0.92 + 0.08} = 0.040$~:
\qty{40}{\micro m} --- régime d'accéléromètre, qui demande une
amplification (levier optique ou électronique).

\textbf{18.} $x_r = 10$~: $100/\sqrt{9801 + 204} \approx 1.00$~: \qty{1.0}{mm},
un enregistrement fidèle du déplacement.

\textbf{19.} $\omega_0 = \qty{6.3e3}{rad/s}$~: $U = g/\omega_0^2 = \qty{2.5e-7}{m}$
--- lu comme la variation de l'entrefer d'un condensateur.

\textbf{20.} $U/(a_g/\omega_0^2) = 1/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$~; avec
$Q = 0.7$, ce rapport décroît de façon monotone et reste au-dessus de
$0.9$ jusqu'à $x_r \approx 0.7$~: environ \qty{700}{Hz}.

\textbf{21.} $\tau = 2Q/\omega_0 = \qty{0.22}{ms}$~: il suit une chute
(quelques dizaines de millisecondes) sans retard.

\textbf{22.} $\tau = 1.4/3.14 = \qty{0.45}{s}$~: après un échelon du sol,
l'enregistrement se stabilise en une demi-seconde, sans sonnerie.

\textbf{23.} Un $Q$ élevé ferait sonner l'instrument à $f_0$ après chaque
secousse (il s'enregistrerait lui-même plutôt que le sol) et
présenterait un pic près de $f_0$~; $Q \approx 0.7$ est plat et rapide.

\textbf{24.} $U = QX_g = 0.7X_g$ au lieu de l'idéal $X_g$ (ou $a_g/\omega_0^2
= X_g$)~: $30\%$ trop bas.

\textbf{25.} On compare la fréquence du \omterm{def:b1:wave-propagation:signal}{signal} à $f_0$~: bien au-dessus,
la masse est une référence fixe et $u$ est le déplacement~; bien en
dessous, $u$ est l'accélération divisée par $\omega_0^2$.
\end{solution}
