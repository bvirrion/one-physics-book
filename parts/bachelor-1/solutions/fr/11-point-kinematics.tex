\section*{Chapitre \ref{ch:b1:point-kinematics} --- Cinématique du point}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$~: \qty{12}{m/s} à \qty{3.0}{s}~; $a = \qty{4.0}{m/s^2}$~;
$x(5) = \qty{50}{m}$~; $\qty{100}{km/h} = \qty{27.8}{m/s}$ à $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$~: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s})~; $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). \omterm{cor:b1:point-kinematics:circular}{Mouvement circulaire} uniforme~: \omterm{def:b1:point-kinematics:velocity}{vitesse}
constante en norme, \omterm{cor:b1:point-kinematics:circular}{accélération centripète}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$~;
$a = v^2/R = \qty{0.034}{m/s^2}$, soit $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Sommet à $t = 7.71/9.81 = \qty{0.79}{s}$, hauteur $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Portée $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
À l'arrivée~: $(9.19, -7.71)$, soit \qty{12}{m/s} à $40^\circ$ sous
l'horizontale.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. Après \qty{5}{s}~: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$~:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$~; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. Le terme $2\dot r\dot\theta$ a deux moitiés égales~: la
direction de la \omterm{def:b1:point-kinematics:velocity}{vitesse} radiale tourne à la \omterm{def:b1:point-kinematics:velocity}{vitesse} $\dot\theta$ (d'où
$\dot r\dot\theta\,\vect e_\theta$), et s'éloigner augmente la \omterm{def:b1:point-kinematics:velocity}{vitesse}
orthoradiale $r\dot\theta$ (encore $\dot r\dot\theta$).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$~; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ vers l'axe~; pas de l'hélice $v_z \times 2\pi/\omega =
\qty{3.1}{m}$~; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$~; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$~;
$v$ est en avance d'un quart de période sur $x$, et $a = -\omega^2x$ est
opposée à $x$. $\abs v$ est maximale en $x = 0$, $\abs a$ aux extrémités $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$~; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Puis
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, et une seconde
dérivation par la règle du produit donne l'\omterm{def:b1:point-kinematics:velocity}{accélération} de la
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Sommet~: $v = v_0\cos\alpha$, $a_N = g$~: $\rho = v_0^2\cos^2\alpha/g$.
Lancement~: $v = v_0$, $a_N = g\cos\alpha$~: $\rho = v_0^2/(g\cos\alpha)$.
Le rayon au sommet est plus petit d'un facteur $\cos^3\alpha$~: la
parabole se courbe le plus là où elle est la plus lente et où tout
$\vect g$ est normal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$~; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. Au sol ($vt/R = 2\pi n$)~:
$\dot x = \dot y = 0$ et $\vect a = (0, v^2/R)$, vers le haut. Au sommet
($vt/R = \pi$)~: $\dot x = 2v$, $\dot y = 0$, donc une \omterm{def:b1:point-kinematics:velocity}{vitesse} $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$~: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. À $t = \qty{2.0}{s}$~: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). Le premier est le terme centripète
de la rotation~; le second, $2\dot r\dot\theta$, n'existe que parce que
la direction radiale tourne elle-même --- sur le sol, marcher droit à
\omterm{def:b1:point-kinematics:velocity}{vitesse} constante ne donne aucune \omterm{def:b1:point-kinematics:velocity}{accélération}.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$~; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (mouvement à $\theta$ croissant),
$\vect N = -\vect e_r$ (vers $C$), $\rho = R$.

\textbf{3.} Sommet~: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$. Côtés~:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ et $v = R\dot\theta$~:
$\dot v = -g\sin\theta$. Négative pour $0 < \theta < \pi$ (ralentissement
à la montée), positive à la descente.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$~: en bas
\qty{62.5}{m/s^2}~; côtés $62.5 - 19.6 = \qty{42.9}{m/s^2}$~; sommet $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} En bas~: \qty{62.5}{m/s^2} (pas de part tangentielle)~; côtés~:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$~; sommet~: \qty{23.3}{m/s^2},
dirigée droit vers le bas, vers $C$~: angle $0$ avec la verticale.

\textbf{7.} Perpendiculaires en bas et au sommet ($a_T = 0$)~; jamais
parallèles ($a_N > 0$ partout).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$~:
en bas \qty{2.50}{rad/s}, au sommet \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$~: l'équation du
pendule simple de longueur $R$ --- la voiture sur son rail est un
pendule qui passe par le sommet.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Moyenne de $25$, $20.7$, $15.3$, $20.7$~: environ \qty{20}{m/s}~;
longueur $2\pi R = \qty{63}{m}$~: à peu près \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$~: oui. Minimum~: $v_0^2 \geq
gR + 4gR = 5gR = 490$, soit $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} En bas~: $\vect a = 62.5$ vers le haut, $\vect g$ vers le bas~:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} Au sommet~: toutes deux vers le bas, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
qui presse le passager dans son siège (vers le centre, donc vers le haut
pour le passager la tête en bas).

\textbf{15.} Côtés~: $\vect a = 42.9$ horizontale vers $C$ plus $9.8$
vers le bas (tangentielle)~; $\vect a - \vect g$ vaut $42.9$ à
l'horizontale~: $4.4g$ latéralement, vers le centre.

\textbf{16.} Elle échoue~: $7.4g$ en bas (le sommet et les côtés vont bien).

\textbf{17.} Poids apparent nul au sommet~: $v_{\text{sommet}}^2 = gR$~;
alors $v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ en bas et la charge vaut
$5g + g = 6g$, indépendamment de $R$.

\textbf{18.} $a_N = v^2/\rho$~: $v$ est fixée par la hauteur (l'énergie),
$\rho$ par la géométrie du rail~; un $\rho$ différent à la même hauteur
change $a_N$ et rien d'autre.

\textbf{19.} $a_N + g = 3g$~: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{sommet}} = \qty{15.3}{m/s}$~; $a_N - g = 1.5g$~: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$~: elle croît
linéairement depuis zéro --- un taux de variation constant de
l'\omterm{def:b1:point-kinematics:velocity}{accélération} --- au lieu de sauter de $0$ à $v^2/R$ à l'entrée d'un
cercle, ce que la nuque ressent comme un choc.

\textbf{22.} Une goutte d'eau~: large et peu courbée en bas, où la
voiture est rapide, resserrée en haut, où elle est lente. Elle maintient
la charge au voisinage de $3g$ tout du long et l'amène progressivement.

\textbf{23.} Ligne de visée~: distance horizontale \qty{50}{m}, hauteur
\qty{20}{m}~: $r = \qty{53.9}{m}$, élévation $\beta = 21.8^\circ$. La
composante transversale de la \omterm{def:b1:point-kinematics:velocity}{vitesse} horizontale vaut $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$~; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Vu d'un point excentré, la \omterm{cor:b1:point-kinematics:circular}{vitesse angulaire} de la ligne de
visée n'est pas constante (rapide quand la voiture est proche, lente
quand elle est loin)~: une caméra à \omterm{def:b1:point-kinematics:velocity}{vitesse} constante décroche.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Coordonnées polaires} en $C$ pour la \omterm{def:b1:point-kinematics:velocity}{vitesse} et
l'\omterm{def:b1:point-kinematics:velocity}{accélération} sur le cercle~; la décomposition de Frenet pour séparer
«~accélérer~» de «~tourner~» et pour la clothoïde~; \omterm{def:b1:point-kinematics:cylindrical}{coordonnées polaires}
depuis la caméra pour la \omterm{def:b1:point-kinematics:velocity}{vitesse} de suivi~; l'énergie (chapitres
suivants) pour la loi des \omterm{def:b1:point-kinematics:velocity}{vitesses}.
\end{solution}
