\section*{Chapitre \ref{ch:b1:phase-changes} --- Changements d'état d'un corps pur}

\begin{solution}{exo:b1:phase-changes:1}
(a) Vapeur~: \qty{0.5}{kPa} est au-dessous de $P_s(\qty{20}{\degreeCelsius}) = \qty{2.3}{kPa}$.
(b) Solide. (c) Au-dessus de $T_c = \qty{374}{\degreeCelsius}$~: fluide
supercritique, sans distinction liquide--gaz. (d) Le \omterm{prop:b1:phase-changes:PTdiagram}{point triple}~: les trois.
La flaque s'évapore par sa surface tant que la pression de vapeur dans l'air
reste au-dessous de $P_s(\qty{20}{\degreeCelsius})$~; l'ébullition n'est que le
cas où des bulles peuvent se former au sein du liquide.
\end{solution}

\begin{solution}{exo:b1:phase-changes:2}
$0.1 \times 2257 + 0.1 \times 4.18\,(100 - T_f) = 1.0 \times 4.18\,(T_f - 20)$~:
$351.1 = 4.60\,T_f$, $T_f = \qty{76}{\degreeCelsius}$. Avec de l'eau chaude à
la place~: $100 - T_f = 10(T_f - 20)$, $T_f = \qty{27}{\degreeCelsius}$ --- la
\omterm{def:b1:phase-changes:phase}{chaleur latente} vaut \qty{540}{K} de chaleur sensible.
\end{solution}

\begin{solution}{exo:b1:phase-changes:3}
$v = \qty{2.0e-3}{m^3/kg}$~; $x = (2.0 - 1.09) \times 10^{-3}/0.392 = 2.3 \times 10^{-3}$~:
\qty{2.3}{g} de vapeur qui remplissent \qty{0.91}{L}~; les \qty{0.998}{kg} de
liquide occupent $0.998 \times 1.09 = \qty{1.09}{L}$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:4}
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, soit $\times 0.018 = \qty{109}{J/(mol.K)}$~;
fusion $334/273 \times 0.018 = \qty{22}{J/(mol.K)}$. Le $109$ de l'eau est bien
au-dessus des $88$ de Trouton~: les liaisons hydrogène rendent l'eau liquide
plus ordonnée qu'un liquide ordinaire, si bien que la vaporiser fait gagner davantage de désordre.
\end{solution}

\begin{solution}{exo:b1:phase-changes:5}
$1/T = 1/373.15 - \ln(P/P_0)/4890$. Everest~: $\ln(0.33/1.013) = -1.12$,
$1/T = 2.680 \times 10^{-3} + 2.29 \times 10^{-4}$, $T = \qty{344}{K} =
\qty{71}{\degreeCelsius}$. À \qty{0.80}{bar}~: $\ln = -0.236$, $T = \qty{366}{K} =
\qty{93}{\degreeCelsius}$. À \qty{71}{\degreeCelsius}, le blanc prend lentement
et le jaune presque pas~: l'œuf de \qty{3}{min} devient un œuf de
\qty{10}{min}, voire davantage.
\end{solution}

\begin{solution}{exo:b1:phase-changes:6}
$\ln(1.8/1.013) = 0.575$~: $1/T = 2.680 \times 10^{-3} - 1.18 \times 10^{-4}$,
$T = \qty{390}{K} = \qty{117}{\degreeCelsius}$~; à \qty{2.5}{bar}~: $\ln = 0.903$,
$T = \qty{401}{K} = \qty{128}{\degreeCelsius}$. Vapeur~: $\qty{2}{L}/0.98 =
\qty{2.0}{g}$ --- l'autocuiseur ne contient presque que de l'eau et un peu de
vapeur.
\end{solution}

\begin{solution}{exo:b1:phase-changes:7}
$\dd P/\dd T = 3.34 \times 10^5/[273 \times (-9.1 \times 10^{-5})] = \qty{-1.3e7}{Pa/K}
= \qty{-130}{bar/K}$~: la glace fond à \qty{-1}{\degreeCelsius} sous environ
\qty{130}{bar}. Sous \qty{3}{km} de glace, $P = 917 \times 9.8 \times 3000 =
\qty{270}{bar}$~: fusion à \qty{-2}{\degreeCelsius}~; le lit d'un glacier peut
être à sa température de fusion alors que la surface est bien plus froide, et
le film d'eau de fonte laisse la calotte glisser.
\end{solution}

\begin{solution}{exo:b1:phase-changes:8}
$W = -P(v_v - v_l) = -1.013 \times 10^5 \times 1.672 = \qty{-169}{kJ}$ (l'eau
repousse l'atmosphère)~; $Q = L_v = \qty{2257}{kJ}$~; $\Delta U = 2257 - 169
= \qty{2088}{kJ}$~; $\Delta H = \qty{2257}{kJ}$~; $\Delta S = 2257/373 = \qty{6.05}{kJ/K}$.
Avec une flamme à \qty{1500}{K}~: $S_{\text{éch}} = 2257/1500 = \qty{1.50}{kJ/K}$,
et \qty{4.5}{kJ/K} créés --- l'essentiel de la qualité de la flamme est gaspillé
à chauffer de l'eau.
\end{solution}

\begin{solution}{exo:b1:phase-changes:9}
$P_s(\qty{22}{\degreeCelsius}) \approx 2.34 + 0.4 \times 0.83 = \qty{2.67}{kPa}$,
$P_v = 0.70 \times 2.67 = \qty{1.87}{kPa}$~; \omterm{ex:b1:phase-changes:dew}{point de rosée} là où $P_s = 1.87$~: $15 +
5 \times 0.17/0.64 = \qty{16}{\degreeCelsius}$. Masse~: $\rho_v = 1870 \times 0.018/
(8.314 \times 295) = \qty{13.7}{g/m^3}$, soit \qty{0.69}{kg} dans la pièce. Vitre à
\qty{12}{\degreeCelsius}~: $P_s \approx \qty{1.42}{kPa} < 1.87$, elle s'embue. À
\qty{25}{\degreeCelsius}, l'humidité tombe à $1.87/3.17 = 59\%$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:10}
(a) $h$ est une fonction d'état~: passer solide $\to$ liquide $\to$ vapeur au
\omterm{prop:b1:phase-changes:PTdiagram}{point triple}, ou solide $\to$ vapeur directement, donne le même $\Delta h$~:
$L_s = L_f + L_v$. (b) Avec $v_s, v_l \ll v_v$, les deux pentes valent $L/(Tv_v)$ aux
mêmes $T$ et $v_v$~: leur rapport est $L_s/L_v > 1$. (c) $v_v = RT/(MP) =
8.314 \times 273.16/(0.018 \times 611) = \qty{206}{m^3/kg}$~; vaporisation
$2.50 \times 10^6/(273.16 \times 206) = \qty{44}{Pa/K}$, sublimation $2.83 \times 10^6/
(273.16 \times 206) = \qty{50}{Pa/K}$. (d) Au-dessous de \qty{0}{\degreeCelsius}
sous \qty{1}{bar}, aucun liquide n'existe~; si la pression de vapeur de l'air
est au-dessous de $P_s$ au-dessus de la glace (\qty{260}{Pa} à
\qty{-10}{\degreeCelsius}), la neige se sublime --- le vent froid et sec la mange.
\end{solution}

\begin{solution}{exo:b1:phase-changes:11}
(a) $P_s(\qty{105}{\degreeCelsius}) \approx 1.013 + 5 \times 0.036 = \qty{1.19}{bar}$~:
$\Delta P = \qty{0.18}{bar}$, $r^\ast = 2 \times 0.059/(1.8 \times 10^4) = \qty{6.6}{\micro m}$.
(b) $\Delta P = \qty{3.6}{kPa}$, $r^\ast = \qty{33}{\micro m}$. (c) $x = c\,\Delta T/L_v =
4.18 \times 5/2257 = 0.93\%$~: \qty{9.3}{g} par litre, soit $9.3 \times 10^{-3}
\times 1.67 = \qty{16}{L}$ de vapeur produits en une fraction de seconde dans
une tasse --- l'eau est projetée au-dehors, en pleine ébullition. (d) Les grains
régulateurs retiennent de l'air dans leurs pores~: des bulles de quelques
dizaines de micromètres préexistent, de sorte que l'ébullition démarre dès que
$P_s$ atteint $P_0$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:12}
(a) Après le détendeur, $h = \qty{256}{kJ/kg}$ à \qty{-10}{\degreeCelsius}~: $x =
(256 - 187)/(392 - 187) = 0.34$. (b) Évaporateur $q_c = 392 - 256 =
\qty{136}{kJ/kg}$~; compresseur $w = 430 - 392 = \qty{38}{kJ/kg}$~; condenseur
$q_h = 256 - 430 = \qty{-174}{kJ/kg}$~; $136 + 38 - 174 = 0$. (c) $e_p =
174/38 = 4.6$, $e_f = 136/38 = 3.6$~; Carnot $313/50 = 6.3$ et $263/50 =
5.3$~: environ $0.7$ de l'idéal. (d) $\dot m = 6000/174000 = \qty{34}{g/s}$,
soit \qty{2.1}{kg} par minute.
\end{solution}

\begin{solution}{pb:b1:phase-changes:1}
\textbf{1.} Sous \qty{1}{atm}, l'eau bout à \qty{100}{\degreeCelsius}~; tant
qu'il reste du liquide, toute chaleur fournie fabrique de la vapeur à cette
température (\omterm{def:b1:phase-changes:phase}{chaleur latente}) au lieu de réchauffer le liquide~: la température
est imposée par la pression.

\textbf{2.} $\dd P_s/\dd T = L_v/[T(v_v - v_l)]$ avec $v_l \ll v_v = RT/(MP_s)$
(vapeur supposée parfaite)~: $\dd P_s/P_s = (ML_v/R)\,\dd T/T^2$~; avec $L_v$
prise constante, on intègre~: $\ln(P_s/P_0) = (ML_v/R)(1/T_0 - 1/T)$.

\textbf{3.} $1/T = 1/373.15 - \ln(1.8/1.013)/4890 = 2.562 \times 10^{-3}$~: $T =
\qty{390}{K} = \qty{117}{\degreeCelsius}$.

\textbf{4.} $m = \Delta P\,A/g = 0.8 \times 10^5 \times 2.0 \times 10^{-5}/9.8 =
\qty{0.16}{kg}$.

\textbf{5.} $Q = 2.0 \times 4.18 \times 97 = \qty{811}{kJ}$~; $t = 811/2.0 =
\qty{405}{s} \approx \qty{7}{min}$.

\textbf{6.} $2000/(2.2 \times 10^6) = \qty{0.91}{g/s} = \qty{55}{g/min}$ --- deux
litres partis en \qty{37}{min}. Baisser la plaque au minimum qui garde la
soupape sifflante~: la température est fixée par la pression, non par la flamme.

\textbf{7.} $+\qty{17}{K}$~: $2^{1.7} = 3.2$ fois plus vite, \qty{30}{min}
$\to$ \qty{9}{min}. Sur l'Everest, $-\qty{30}{K}$~: $2^{-3}$, soit quatre heures.

\textbf{8.} \omterm{ex:b1:phase-changes:dew}{Humidité relative} $= P_v/P_s(T)$. $P_v = 0.60 \times 3.17 =
\qty{1.90}{kPa}$~; $\rho_v = P_vM/(RT) = 1900 \times 0.018/(8.314 \times 298) =
\qty{13.8}{g/m^3}$.

\textbf{9.} $P_s(T_d) = \qty{1.90}{kPa}$~: entre \qty{15}{\degreeCelsius}
(\qty{1.70}{kPa}) et \qty{20}{\degreeCelsius} (\qty{2.34}{kPa})~: $T_d = 15 + 5 \times 0.20/0.64
= \qty{16.6}{\degreeCelsius}$. L'herbe qui rayonne vers le ciel nocturne, ou une
bouteille sortie du \omterm{def:b1:heat-engines:machine}{réfrigérateur}, descend au-dessous de $T_d$~: la vapeur à son
contact est alors sursaturée et se condense.

\textbf{10.} $TP^{(1-\gamma)/\gamma} = \text{const}$~: $\dd T/T = \frac{\gamma - 1}{\gamma}
\dd P/P$~; $\dd P = -\rho g\,\dd z$ avec $\rho = PM/(RT)$~: $\dd T/T = -\frac{\gamma -
1}{\gamma}\frac{Mg}{RT}\dd z$, donc $\dd T/\dd z = -\frac{\gamma - 1}{\gamma}\frac{Mg}{R} =
-g/c_p$ puisque $c_p = \gamma R/[(\gamma - 1)M]$. Numériquement $9.8/1000 = \qty{9.8}{K/km}$.

\textbf{11.} L'écart $T - T_d$ se referme à $9.8 - 2 = \qty{7.8}{K/km}$~: $z = (25 - 16.6)/
7.8 = \qty{1.1}{km}$.

\textbf{12.} $1 \times 2.45 = \qty{2.45}{kJ}$ par kilogramme d'air~: $\Delta T =
\qty{2.45}{K}$. Cette libération compense une partie du refroidissement
adiabatique~: une masse d'air saturée ne se refroidit plus que de
\qtyrange{5}{6}{K/km}, reste plus chaude et plus légère que son entourage et
continue de monter --- la \omterm{def:b1:phase-changes:phase}{chaleur latente} est le carburant de l'orage.

\textbf{13.} $10^9 \times \qty{1}{g} = \qty{1000}{t}$~; $10^6 \times 2.45 \times 10^6 =
\qty{2.5e12}{J} = \qty{0.68}{GW h}$ --- quarante minutes d'une grosse centrale.

\textbf{14.} $P_v = 0.80 \times 3.17 = \qty{2.54}{kPa}$~; $P_s(\qty{18}{\degreeCelsius})
\approx 1.70 + 0.6 \times 0.64 = \qty{2.08}{kPa} < 2.54$~: il s'embue, jusqu'à ce
que l'humidité descende au-dessous de $2.08/3.17 = 66\%$.

\textbf{15.} $P = P_s(\qty{20}{\degreeCelsius}) = \qty{2.3}{kPa}$~; $v = \qty{2.0e-3}{m^3/kg}$~;
$v_v = RT/(MP_s) = 8.314 \times 293/(0.018 \times 2340) = \qty{58}{m^3/kg}$~; $x =
(2.0 - 1.0) \times 10^{-3}/58 = 1.7 \times 10^{-5}$~: \qty{9}{mg} de vapeur~;
liquide $0.50 \times 10^{-3} = \qty{0.50}{L}$, la moitié du récipient.

\textbf{16.} $v = \qty{2.0e-3}{m^3/kg} < v_c$~: l'isochore rencontre la branche
liquide de la cloche --- le liquide se dilate plus vite qu'il ne s'évapore, le
niveau \emph{monte}, et le récipient est plein de liquide lorsque $v_l(T) =
\qty{2.0e-3}{m^3/kg}$, un peu au-dessus de \qty{360}{\degreeCelsius} (environ
\qty{190}{bar}). Au-delà, un liquide quasi incompressible chauffé à volume
constant~: la pression grimpe de plusieurs dizaines de bars par \omterm{def:b1:kinetic-theory:temperature}{kelvin} --- le
récipient éclate. Ne jamais sceller un récipient plein de liquide.

\textbf{17.} $v = \qty{1.0e-2}{m^3/kg} > v_c$~: le niveau \emph{descend}~; la
dernière goutte disparaît lorsque $v_v(T) = 0.010$, entre \qty{340}{\degreeCelsius} ($0.0108$)
et \qty{350}{\degreeCelsius} ($0.0088$)~: vers \qty{344}{\degreeCelsius},
\qty{150}{bar}.

\textbf{18.} $v = \qty{3.2e-3}{m^3/kg} \approx v_c$~: le ménisque reste vers la
mi-hauteur, s'aplatit et pâlit à mesure que les deux masses volumiques
convergent, le fluide devient laiteux (opalescence critique), et à
\qty{374}{\degreeCelsius}, \qty{221}{bar}, le ménisque disparaît~: une seule \omterm{def:b1:phase-changes:phase}{phase}.

\textbf{19.} Trois verticales~: $v = 2.0 \times 10^{-3}$ quitte la cloche par la
branche liquide, à gauche de C~; $10^{-2}$ par la branche vapeur, à droite de
C~; $3.2 \times 10^{-3}$ par le sommet.

\textbf{20.} Air~: \qty{0.5}{L} à \qty{1.0}{bar}, \qty{293}{K}, même volume à
\qty{373}{K}~: $P_{\text{air}} = 1.0 \times 373/293 = \qty{1.27}{bar}$~; plus
$P_s = \qty{1.01}{bar}$~: \qty{2.3}{bar} à l'intérieur, \qty{1.3}{bar} de plus
qu'au-dehors~: $1.3 \times 10^5 \times 10^{-2} = \qty{1.3}{kN}$ sur le couvercle
--- le poids de \qty{130}{kg}.

\textbf{21.} En \omterm{def:b1:phase-changes:metastable}{surfusion}, \omterm{def:b1:phase-changes:metastable}{état métastable}~: aucun site de \omterm{def:b1:phase-changes:metastable}{germination} (eau
propre, sans poussière, immobile) pour le premier cristal. Un choc amorce la
glace, qui envahit la bouteille en libérant $L_f$~; la température monte
jusqu'à \qty{0}{\degreeCelsius} et s'y arrête, seule température où glace et
eau coexistent sous \qty{1}{atm}~: la congélation cesse dès que la chaleur
libérée y a porté le mélange.

\textbf{22.} $x = c\,\Delta T/L_f = 4.18 \times 8/334 = 0.10$.

\textbf{23.} $\Delta s = c\ln(273/265) - xL_f/273 = 124.3 - 122.3 =
\qty{+2.0}{J/(kg.K)}$~: le processus est adiabatique, donc tout est créé~; et
c'est positif, comme doit l'être un processus spontané.

\textbf{24.} $x = 4.18 \times 5/2257 = 0.93\%$~: \qty{9.3}{g} de vapeur par litre,
$9.3 \times 10^{-3} \times 1.67 = \qty{16}{L}$ de vapeur nés au sein du liquide en
une fraction de seconde~: la tasse déborde violemment.

\textbf{25.} $T_0S_{\text{créée}} = 273 \times 2.0 = \qty{550}{J}$ par litre,
contre $mgh = \qty{9.8}{J}$~: cinquante-six fois plus. Un \omterm{def:b1:phase-changes:metastable}{état métastable} stocke
du travail --- celui qu'un chemin réversible aurait pu extraire --- et un choc
le transforme intégralement en entropie.
\end{solution}
