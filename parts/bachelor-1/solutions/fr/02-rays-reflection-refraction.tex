\section*{Chapitre \ref{ch:b1:rays-reflection-refraction} --- Optique géométrique~: rayons, réflexion, réfraction}

\begin{solution}{exo:b1:rays-reflection-refraction:1}
$v = c/n$~: eau \qty{2.26e8}{m/s}, verre \qty{2.00e8}{m/s}, diamant
\qty{1.24e8}{m/s}. Fibre~: $t = nL/c = 1.48 \times 10^3/3.00\times10^8 =
\qty{4.9}{\micro s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:2}
$\sin i_2 = \sin i_1/1.33$~: $30^\circ \to 22.1^\circ$~; $60^\circ \to
40.6^\circ$. Depuis l'eau sous $50^\circ$~: $1.33\sin 50^\circ = 1.02 > 1$,
pas de rayon réfracté --- \omterm{thm:b1:rays-reflection-refraction:tir}{réflexion totale} ($i_c = 48.8^\circ$).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}
$i_c = \arcsin(1/n)$~: verre $41.8^\circ$, diamant $24.4^\circ$, eau
$48.8^\circ$. Le disque de ciel s'étend sur $2 \times 48.8^\circ = 97.5^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
Les yeux sont à \qty{1.60}{m}. Le rayon venu des pieds atteint l'œil
après réflexion à mi-hauteur des yeux, \qty{0.80}{m}~; celui venu du
sommet du crâne se réfléchit à mi-chemin entre le sommet et les yeux, à
\qty{1.65}{m}. Miroir de \qty{0.80}{m} à \qty{1.65}{m}~: hauteur
\qty{0.85}{m}, la moitié de la taille --- indépendante de la distance
(la règle du milieu vaut à toute distance).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
Aux petits angles, la position apparente est $h' = h/n$ vue depuis le
milieu le moins réfringent, $h' = nh$ depuis le plus réfringent. Pièce~:
$1.2/1.33 =
\qty{0.90}{m}$. Pêcheur~: $1.33 \times 1.8 = \qty{2.4}{m}$. Justification~:
pour un point de la surface à la distance horizontale $x$, $\tan i_2 \approx i_2 =
x/h$ et $\tan i_1 \approx i_1 = x/h'$ avec $i_1 = n i_2$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$, donc
$\frac{A + D_{\min}}{2} = 49.5^\circ$ et $D_{\min} = 38.9^\circ$.
L'émergence exige $r' \leq i_c$~; par ailleurs $\sin r = \sin i/n \leq 1/n$ donne
$r \leq i_c$~; donc $A = r + r' \leq 2i_c$. Ici $i_c = \arcsin(1/1.52) =
41.1^\circ$, $2i_c = 82.3^\circ > 60^\circ$~: c'est vérifié.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$,
$\theta_a = 14.1^\circ$. $\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$. Sur \qty{1}{km}~:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}
$D \approx (n - 1)A$~: rouge $0.510 \times 10^\circ = 5.10^\circ$, bleu
$5.30^\circ$~; largeur du spectre $0.20^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
Entrée~: $\sin r = \sin 45^\circ/1.5 = 0.471$, $r = 28.1^\circ$. Les deux
faces étant parallèles, l'incidence sur la seconde face vaut $r$ et
$\sin i' = n\sin r = \sin i$~: $i' = i$, le rayon émergent est parallèle
à l'incident. Le long du trajet intérieur, de longueur $e/\cos r$, le
décalage perpendiculaire vaut $(e/\cos r)\sin(i - r)$~:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
Deux réfractions ($2(i - r)$) et deux réflexions ($2(\pi - 2r)$)~:
$D = 2\pi + 2i - 6r$. $\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ donne
$3\cos i = n\cos r$, $9\cos^2 i = n^2 - 1 + \cos^2 i$, $\cos^2 i =
(n^2 - 1)/8$. Pour $n = 1.333$~: $\cos^2 i = 0.0971$, $i = 71.8^\circ$,
$\sin r = 0.950/1.333$, $r = 45.5^\circ$, $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$~: l'arc est à $230.9^\circ - 180^\circ =
51^\circ$ du point antisolaire, en dehors du primaire ($42^\circ$).
Le rouge ($n = 1.331$) à $50.4^\circ$, le violet ($1.343$) à $53.6^\circ$~:
les couleurs sont inversées, violet à l'extérieur.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$, $i_c = 89.56^\circ$~: les rayons
situés à moins de $0.44^\circ = \qty{7.7}{mrad}$ de la chaussée sont
totalement réfléchis (en général, angle rasant $\approx \sqrt{2\Delta n/n}$).
Depuis \qty{1.2}{m} de hauteur, la ligne de visée rase sous cet angle
au-delà de $1.2/0.0077 \approx \qty{160}{m}$~: la route y réfléchit le
ciel --- c'est la «~flaque~».
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$~;
$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$, nul exactement lorsque
$\sin i_1/v_1 = \sin i_2/v_2$, c'est-à-dire $n_1\sin i_1 = n_2\sin i_2$ avec
$n = c/v$.
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$~; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$, inchangée dans le verre~;
$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$.

\textbf{4.} Un \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} nu ne guide que tant que sa surface reste
parfaitement propre et intacte~: poussière, eau ou simple contact
remplacent localement l'air par un indice plus élevé et la lumière
s'échappe, tandis que les rayures la diffusent. La \omterm{def:b1:rays-reflection-refraction:fiber}{gaine} est une
interface scellée et maîtrisée, dont le saut d'indice est choisi.

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$~; guidé si et seulement si $\cos\theta' \geq
n_2/n_1$, c'est-à-dire $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$, soit
$\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$~:
$\theta_a = 12.1^\circ$.

\textbf{8.} $\mathrm{NA} = 0.210$. Un saut plus grand élargit le cône,
donc capte une plus grande part de la lumière d'une source divergente.

\textbf{9.} Le rayon est tangent au cercle de rayon $R_b$ (bord
intérieur) et rencontre le cercle de rayon $R_b + 2a$ (bord extérieur)~;
dans le triangle rectangle centre--point de tangence--point d'impact,
l'angle au point d'impact entre le rayon et le rayon du cercle (la
normale) vérifie $\sin i = R_b/(R_b + 2a)$. Guidé si et seulement si
$\sin i \geq n_2/n_1$~:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$. Des courbures plus serrées fuient.

\textbf{10.} $t_0 = Ln_1/c$~; le rayon le plus incliné parcourt $L/\cos\theta'_{\max} =
Ln_1/n_2$, donc $t_1 = Ln_1^2/(n_2 c)$.

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$~; sur \qty{2}{km}~: \qty{101}{ns}.

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} Les rayons éloignés de l'axe suivent des trajets plus
longs, mais dans un indice plus faible, donc plus vite~; avec un profil
bien choisi les deux effets se compensent et tous arrivent ensemble.

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$~: le \omterm{def:b1:rays-reflection-refraction:fiber}{cœur} n'est pas
grand devant la longueur d'onde, la diffraction domine et les rayons
n'ont plus de sens. Le traitement ondulatoire montre un unique mode
guidé~: plus aucun étalement intermodal~; ne restent que la dispersion
chromatique et l'atténuation.

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$~:
$P/P_0 = 10^{-2}$, un pour cent.

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$, donc $L_{\max} =
30/0.20 = \qty{150}{km}$.

\textbf{18.} \qty{100}{km} coûtent \qty{20}{dB}~: marge \qty{10}{dB},
soit un facteur $10$ en puissance.

\textbf{19.} Diode~: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$, $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$. Laser~:
$\Delta t_c = \qty{170}{ps}$, $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$.

\textbf{20.} $30/2.0 = \qty{15}{km}$, dix fois moins. La silice est la
plus transparente au voisinage de \qty{1550}{nm} (son minimum
d'atténuation), et c'est là qu'existent les amplificateurs optiques.

\textbf{21.} $30/30 = \qty{1}{km}$ entre amplificateurs, contre
\qty{150}{km}~: la fibre l'emporte d'un facteur $150$.

\textbf{22.} Intermodal~: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s})~;
chromatique (laser)~: $17 \times 0.1 \times 2 = \qty{3.4}{ps}$~;
atténuation \qty{0.4}{dB}, négligeable. C'est l'étalement intermodal qui
limite~: \qty{5}{Mbit/s}.

\textbf{23.} Pas d'étalement intermodal~; chromatique \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}~; atténuation \qty{20}{dB}, dans le bilan. C'est la
dispersion chromatique qui limite, vers \qty{3}{Gbit/s}.

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} Saut d'indice~: \qty{10}{Mbit.s^{-1}.km}~; canal monomode~:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- un facteur d'environ
$3\times10^4$. Un seul fait~: un unique trajet guidé, donc aucun
étalement géométrique (intermodal) des temps d'arrivée.
\end{solution}
