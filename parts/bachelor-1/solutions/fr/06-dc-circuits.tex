\section*{Chapitre \ref{ch:b1:dc-circuits} --- Circuits en courant continu~: lois de Kirchhoff et théorèmes}

\begin{solution}{exo:b1:dc-circuits:1}
$q = 1.5 \times 3600 = \qty{5400}{C}$, $N = 5400/1.6\times10^{-19} =
\num{3.4e22}$ électrons. $c/f$~: \qty{6000}{km} à \qty{50}{Hz} (maison~:
oui)~; \qty{300}{m} à \qty{1}{MHz} (circuit de \qty{10}{cm}~: oui)~;
\qty{30}{cm} à \qty{1}{GHz}, comparable à \qty{10}{cm}~: non.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:2}
$I = 2000/230 = \qty{8.7}{A}$~; $R = U^2/P = \qty{26.5}{\ohm}$~; $E =
\qty{6.0}{kWh} = \qty{2.2e7}{J}$. Cordon~: un conducteur $R = 1.7\times10^{-8}
\times 5.0/1.5\times10^{-6} = \qty{57}{m\ohm}$, deux conducteurs
\qty{0.11}{\ohm}~: $P = 0.11 \times 8.7^2 = \qty{8.6}{W}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:3}

$10 \parallel 20 = \qty{6.7}{\ohm}$, plus $30$~: \qty{36.7}{\ohm}.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:4}
$u_2 = 12 \times 2.2/6.9 = \qty{3.83}{V}$. En charge~: $2.2 \parallel 2.2 =
\qty{1.1}{k\ohm}$, $u_2 = 12 \times 1.1/5.8 = \qty{2.28}{V}$~: la charge

\end{solution}

\begin{solution}{exo:b1:dc-circuits:5}
$r = (12.6 - 11.4)/120 = \qty{10}{m\ohm}$~; $I_N = 12.6/0.010 =
\qty{1260}{A}$. Au démarreur $11.4 \times 120 = \qty{1.37}{kW}$~; perdus
$0.010 \times 120^2 = \qty{144}{W}$~; rendement $1368/1512 = 90\%$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:6}
$E_{\mathrm{Th}} = \qty{3.83}{V}$ (\omterm{def:b1:dc-circuits:voltage}{tension} à vide du diviseur)~; $R_{\mathrm{Th}}
= 4.7 \parallel 2.2 = \qty{1.50}{k\ohm}$ (source court-circuitée). En
charge~: $u = 3.83 \times 2.2/(1.50 + 2.2) = \qty{2.28}{V}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:7}
Superposition~: $E_1$ seule voit $r_2 \parallel R = \qty{1.0}{\ohm}$,
$u = 10 \times 1/(1 + 1) = \qty{5.0}{V}$~; $E_2$ seule voit $r_1 \parallel R
= \qty{0.667}{\ohm}$, $u = 5 \times 0.667/2.667 = \qty{1.25}{V}$~; au total
\qty{6.25}{V}. Millman~: $(10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = \qty{6.25}{V}$.
Batterie 1~: $(10 - 6.25)/1 = \qty{3.75}{A}$ débités~; batterie 2~: $(5 - 6.25)/2
= \qty{-0.63}{A}$, c'est-à-dire \qty{0.63}{A} \emph{vers} elle --- elle
est chargée par la plus forte~; charge $6.25/2 = \qty{3.1}{A}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:8}
$i = 9/(R + 3)$, $P = Ri^2$, $\eta = R/(R + 3)$~: $\qty{1}{\ohm}$~:
\qty{5.1}{W}, $25\%$~; $\qty{3}{\ohm}$~: \qty{6.75}{W}, $50\%$~;
$\qty{12}{\ohm}$~: \qty{4.3}{W}, $80\%$. Un appareil sur pile veut du
rendement (l'autonomie), donc $R \gg r$~; la charge adaptée gaspille la
moitié de l'énergie en chaleur dans la pile.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:9}
$R = (5.0 - 2.0)/0.015 = \qty{200}{\ohm}$~; $P_R = 3.0 \times 0.015 =
\qty{45}{mW}$, $P_{\mathrm{LED}} = 2.0 \times 0.015 = \qty{30}{mW}$. La
\omterm{met:b1:dc-circuits:loadline}{droite de charge}, de $(0, \qty{25}{mA})$ à $(\qty{5}{V}, 0)$, coupe la
caractéristique verticale en \qty{2.0}{V}. Sous \qty{3.3}{V}, la même
résistance ne donne plus que \qty{6.5}{mA}~; pour \qty{15}{mA} il faut $R = 1.3/0.015 =
\qty{87}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:10}
$u = E\left[\dfrac12 - \dfrac{1 + \epsilon}{2 + \epsilon}\right] =
-E\dfrac{\epsilon}{2(2 + \epsilon)} \approx -\dfrac{E\epsilon}{4} =
\qty{-1.25}{mV}$. Dans un simple diviseur, la même variation se
superpose à un décalage de \qty{2.5}{V} (\qty{2.50125}{V})~; le pont
délivre les \qty{1.25}{mV} autour de zéro, que l'on peut amplifier mille
fois sans rien saturer.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:11}
Par symétrie, les trois voisins du sommet d'entrée sont au même
potentiel, et les trois voisins du sommet de sortie à un autre. Le
courant $I$ se divise en $I/3$ dans les trois arêtes d'entrée, puis
chacun en deux $I/6$ le long des six arêtes médianes, qui se
recombinent en $I/3$ dans les trois arêtes de sortie. \omterm{def:b1:dc-circuits:voltage}{Tension}~:
$RI/3 + RI/6 + RI/3 = 5RI/6$, donc $R_{\mathrm{eq}} = 5R/6$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:12}
Thévenin~: $E = I_N r = \qty{8.0}{V}$ en série avec \qty{4.0}{\ohm}.
Maille~: $8.0 - 6.0 = (4.0 + 2.0)i$, $i = \qty{0.33}{A}$ vers la
batterie. La batterie reçoit $6.0 \times 0.33 = \qty{2.0}{W}$ (elle se
charge)~; la résistance de \qty{2}{\ohm} dissipe \qty{0.22}{W}~; la
\omterm{def:b1:dc-circuits:voltage}{tension} aux bornes de l'ensemble de Norton vaut
$8.0 - 4.0 \times 0.33 = \qty{6.67}{V}$, donc sa \qty{4}{\ohm} est
traversée par \qty{1.67}{A} et dissipe \qty{11.1}{W}, et la \omterm{def:b1:dc-circuits:sources}{source de
courant} fournit $2.0 \times 6.67 = \qty{13.3}{W}$ --- le bilan est
équilibré.
\end{solution}

\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} Fém idéale $E$ en série avec $r$~: $u = E - ri$.

\textbf{2.} $u(0) = \qty{12.6}{V}$~; $I_N = E/r = \qty{1260}{A}$. Une clé
posée en travers des bornes le porterait~: $\qty{16}{kW}$ dans quelques
grammes d'acier --- elle fond et projette du métal.

\textbf{3.} \omterm{def:b1:dc-circuits:sources}{Source de courant} $I_N = \qty{1260}{A}$ en dérivation avec
\qty{10}{m\ohm}.

\textbf{4.} $i = E/(R + r)$~; $u = ER/(R + r)$~; $P_R = E^2R/(R + r)^2$~;
$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$~: maximum en $R = r$,
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$~; rendement
$R/(R + r) = 50\%$.

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$~; $12.6 - 0.010 \times
120 = \qty{11.40}{V}$. Porter $u$ en fonction de $i$ donne une droite
d'ordonnée à l'origine $E$ et de pente $-r$ (\omterm{prop:b1:units-dimensions:regression}{moindres carrés}, \cref{ch:b1:units-dimensions}).

\textbf{7.} Deux lampes~: \qty{110}{W} sous \qty{12}{V}, \qty{9.2}{A}~;
$60/9.2 = \qty{6.5}{h}$. Énergie $60 \times 3600 \times 12 = \qty{2.6}{MJ}$.

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$~; en dérivation \qty{1.31}{\ohm}.

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$~; $u = 12.6 - 1.4 = \qty{11.2}{V}$~;
$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$~; perdus $0.010 \times 140^2 =
\qty{196}{W}$.

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$~; $i = 12.6/0.0854
= \qty{148}{A}$~; $u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$~; $i_h =
\qty{8.5}{A}$ pour les deux lampes.

\textbf{12.} Chaque lampe~: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$ contre
\qty{55}{W}~: en baisse de $14\%$.

\textbf{13.} \omterm{thm:b1:dc-circuits:kirchhoff}{Loi des mailles}~: la \omterm{def:b1:dc-circuits:voltage}{tension} aux bornes vaut $u = E - ri$~;
les \qty{140}{A} du démarreur produisent une chute de \qty{1.4}{V} dans
la batterie, et les lampes, en dérivation sur les mêmes bornes,
reçoivent cette \omterm{def:b1:dc-circuits:voltage}{tension} diminuée.

\textbf{14.} En partie seulement~: des câbles plus gros réduisent la
chute dans les câbles, pas dans $r$~; alimentées aux bornes de la
batterie, les lampes voient toujours $E - ri$.

\textbf{15.} Trois branches entre les deux mêmes nœuds~: $(E, r)$,
$(E_a, r_a)$ et $R_h = \qty{1.31}{\ohm}$.

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} Alternateur $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$~; batterie
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (qui vont \emph{vers} elle)~; lampes
$12.79/1.31 = \qty{9.8}{A}$~; $28.3 = 18.6 + 9.8$.

\textbf{18.} Elle se charge. Elle reçoit $12.79 \times 18.6 = \qty{238}{W}$
(\qty{234}{W} stockés, \qty{3.5}{W} en chaleur)~; l'alternateur fournit
$12.79 \times 28.3 = \qty{362}{W}$ à ses bornes.

\textbf{19.} Batterie seule (alternateur $\to$ \qty{50}{m\ohm})~: charge
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$, $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$. Alternateur seul (batterie $\to$ \qty{10}{m\ohm})~: charge
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$, $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$. Somme \qty{12.79}{V}.

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$~: \qty{3.60}{V} à
\qty{20}{\degreeCelsius}, \qty{1.40}{V} à \qty{90}{\degreeCelsius}.

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$~;
$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$~: $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$.

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$, nul lorsque $R_s =
\qty{775}{\ohm}$, c'est-à-dire à la température médiane où la
thermistance atteint cette valeur.

\textbf{23.} $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$ par conducteur, \qty{3.2}{m\ohm} au total~; chute $0.0032
\times 140 = \qty{0.45}{V}$~; puissance \qty{63}{W}.

\textbf{24.} En \qty{4}{mm^2}~: \qty{6.4}{m\ohm} chacun, \qty{12.8}{m\ohm}
au total~: chute \qty{1.8}{V}, \qty{250}{W} gaspillés --- le démarreur
perdrait un sixième de sa \omterm{def:b1:dc-circuits:voltage}{tension}. Un câble gros maintient $R_{\text{câble}} \ll R_s$.

\textbf{25.} Environ \qty{11.1}{V} aux bornes pendant le démarrage~; le
démarreur reçoit $1568/(1568 + 196) \approx 89\%$ de la puissance de la
batterie~; l'affaiblissement des phares, c'est la \omterm{thm:b1:dc-circuits:kirchhoff}{loi des mailles}, $u = E - ri$.
\end{solution}
