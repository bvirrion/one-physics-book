\section*{Chapitre \ref{ch:b1:work-and-energy} --- Travail, énergie et énergie potentielle}

\begin{solution}{exo:b1:work-and-energy:1}
Force de levage $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$~; poids
$-\qty{589}{J}$~; puissance $589/5.0 = \qty{118}{W}$. Frottement~: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$~; $d = E_k/F =
\qty{55}{m}$~; doubler $v$ quadruple $E_k$ et $d$~: \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$~; $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$~: $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
Tension~: $T - mg = mv^2/\ell = mg$, donc $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
Le frottement travaille $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$, donc
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$~; $35\%$ perdus.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
$E_m = \tfrac12 mv^2 - GMm/R \geq 0$ pour que le corps atteigne l'infini
(où $E_p = 0$ et $v \to 0$)~: $v_e = \sqrt{2GM/R}$, c'est-à-dire
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
Même masse volumique, $M \propto R^3$~: $v_e \propto R$, donc \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
Point le plus bas~: $mg(\ell_0 + x) = \tfrac12 kx^2$, $25x^2 - 687x - 13734 = 0$,
$x = \qty{41}{m}$~; chute totale \qty{61}{m}~; $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (soit \qty{2}{g} net vers le haut). La vitesse est maximale là
où la force résultante s'annule, $kx = mg$~: $x = \qty{14}{m}$, à
\qty{34}{m} sous le point de départ.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
Pesanteur~: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$~;
traînée~: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$~; au total \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$~: $x = 0, \pm a$. $E_p'' = E_0(12x^2/a^4 -
4/a^2)$~: $-4E_0/a^2$ en $0$ (maximum instable, $E_p = 0$), $+8E_0/a^2$
en $\pm a$ (minimums stables, $E_p = -E_0$)~: des puits de profondeur $E_0$~;
$\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$~: une ellipse de demi-axes $x_{\max}
= \sqrt{2E/k}$ et $\dot x_{\max} = \sqrt{2E/m}$. Le mouvement est
harmonique, de période $2\pi\sqrt{m/k}$ quelle que soit l'\omterm{def:b1:wave-propagation:signal}{amplitude}
(isochronisme), donc toutes les ellipses sont parcourues en le même temps.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
Énergie~: $v^2 = 2gR(1 - \cos\theta)$. Radialement~: $mg\cos\theta - N = mv^2/R$,
donc $N = mg(3\cos\theta - 2)$, nulle pour $\cos\theta = 2/3$ ($\theta =
48^\circ$)~: hauteur $2R/3$, vitesse $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$~: $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
Elle est passée en chaleur dans le contact pneu--route (et en trace de
gomme). Des roues qui roulent exploitent le frottement statique,
$\mu_s > \mu_d$~: une force de freinage plus grande, une distance plus
courte, et la direction conservée.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} Quand $r \to 0$~: $+\infty$ (le terme en $r^{-12}$)~; quand $r \to \infty$~:
$0^-$. Une paroi raide, un puits, une longue queue.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ si et seulement si
$(\sigma/r)^6 = 1$~: $r = \sigma$, $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$~: $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$, minimale en $u = 1$ ($r = \sigma$) et valant $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$~:
positive (répulsive) pour $r < \sigma$, négative (attractive) au-delà.

\textbf{5.} $(1/1.2)^6 = 0.335$, $(1/1.2)^{12} = 0.112$~: $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. $(1/1.2)^{13} = 0.0935$,
$(1/1.2)^7 = 0.279$~: $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$,
proche des \qty{431}{kJ/mol} mesurés.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$~; en
$\sigma$~: $72\epsilon/\sigma^2 > 0$~: stable.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$~:
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$~; $f = \qty{2.2e14}{Hz}$~; $\lambda = c/f = \qty{1.4}{\micro m}$~;
$1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} Un facteur $2.4$ de trop~: les parois du modèle sont trop
raides (le cœur en $r^{-12}$ est un substitut grossier d'une liaison
covalente), mais l'\omterm{def:b1:units-dimensions:oom}{ordre de grandeur} --- une vibration infrarouge à
$10^{14}$ Hz --- est juste, et la méthode (la courbure du puits) est
exactement celle que l'on emploie avec de meilleurs potentiels.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$~: $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$~: la liaison
frissonne à peine.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$~: courbes fermées autour de $(\sigma, 0)$ ---
vibration, état lié~; $E_m > 0$~: courbes ouvertes --- les atomes se
séparent~; $E_m = 0$~: la séparatrice, un atome tout juste capable
d'atteindre l'infini.

\textbf{15.} $u^2 - 2u = -\tfrac12$ avec $u = (\sigma/r)^6$~: $u = 1 \pm 1/\sqrt2
= 1.71$ ou $0.293$~; $r = \sigma u^{-1/6}$~: $0.915\sigma$ et $1.23\sigma$.

\textbf{16.} Milieu en $1.07\sigma > \sigma$~: la paroi extérieure est plus
molle que l'intérieure, si bien que l'atome dispose de plus de place au
dehors~; la distance moyennée dans le temps croît avec l'énergie --- les
liaisons chauffées s'allongent, les solides se dilatent.

\textbf{17.} Plus longue~: du côté extérieur, plus mou, la force de
rappel est plus faible que ne le suppose l'approximation harmonique, et
le retour prend donc plus de temps.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$~; en
\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} à la distance $\sigma$, cela donne
$v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}
= \qty{2.0e4}{m/s}$.

\textbf{19.} Conservation de l'énergie~: si $E_m > 0$, il reste $> 0$, la
courbe est ouverte, et les atomes se séparent après une seule rencontre.
Il faut un troisième corps (une autre molécule, une paroi, un photon)
pour emporter l'excédent.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$, positif~: l'attraction ramène
l'atome, le \omterm{def:b1:work-and-energy:work}{travail} est moteur.

\textbf{21.} $W_{\text{ext}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$, $(1/0.9)^7 = 2.09$~: $F = +22\epsilon/\sigma
= \qty{+120}{nN}$, dix fois l'attraction en $1.2\sigma$~: comprimer une
liaison de $10\%$ coûte bien plus que l'étirer de $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. Les gaz réels se dissocient bien en dessous, parce que
les énergies sont distribuées~: une fraction des molécules porte
plusieurs fois $k_BT$ (\cref{ch:b1:kinetic-theory}).

\textbf{24.} Survivent~: l'équilibre au minimum, $k = E_p''$ au minimum,
l'énergie de dissociation $=$ profondeur, les \omterm{def:b1:work-and-energy:turning}{points de rebroussement}
tirés de $E_p = E_m$, la dissymétrie $\Rightarrow$ dilatation. Changent~:
les valeurs de $k$, de la fréquence et des \omterm{def:b1:work-and-energy:turning}{points de rebroussement}.

\textbf{25.} Équilibre~: $E_p' = 0$~; vibration~: $\omega_0^2 = E_p''/m$~;
échappée~: $E_m$ comparée au bord du puits.
\end{solution}
