\section*{Chapitre \ref{ch:b1:operational-amplifier} --- L'amplificateur opérationnel}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$~; $V_s = \qty{5.0}{V}$~; saturation pour $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$~; \omterm{def:b1:sinusoidal-impedance:impedance}{impédance} d'entrée $R_1 = \qty{10}{k\ohm}$~; $i = 1.0/10^4 =
\qty{0.10}{mA}$~; $V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
Sans~: diviseur $1/(1 + 10) = 0.091$, soit \qty{91}{mV}. Avec le
suiveur~: \qty{1.0}{V} --- le suiveur ne tire rien du capteur et fournit
le \qty{1}{mA} dont la charge a besoin.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$~; en $V_-$~: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
donc $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$~; avec $V_- = V_+$~:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. Application~: $10 \times 0.05 = \qty{0.50}{V}$~;
le \qty{1}{V} commun n'apporte rien --- c'est le rejet de mode commun.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$~: pente $-1000 \times 1.0 = \qty{-1000}{V/s}$~;
saturation au bout de $15/1000 = \qty{15}{ms}$. \omterm{def:b1:wave-propagation:signal}{Signal} carré~: triangle
d'\omterm{def:b1:wave-propagation:signal}{amplitude} $ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(à la constante près qu'a laissée la charge initiale).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$~: gain continu $-100$~; coupure
$1/(2\pi R'C) = \qty{1.6}{Hz}$~; à \qty{1}{kHz}, $R'C\omega = 628$, $G =
100/628 = 0.16$ --- la valeur $1/RC\omega$ de l'intégrateur. Il intègre
pour $f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$~: $R_2 = \qty{10}{k\ohm}$~; $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
Avec $\mu$ fini~: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, donc
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. \omterm{def:b1:filters-transfer-functions:bode}{Bande passante} $f_T/G =
\qty{10}{kHz}$~; pour un suiveur~: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
Chaque excursion du bruit à travers \qty{2.0}{V} fait basculer la
sortie~: une rafale de commutations rapides autour du passage. Remède~:
une hystérésis plus large que le bruit, c'est-à-dire un \omterm{prop:b1:operational-amplifier:schmitt}{trigger de Schmitt}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$~: seuils $\pm\qty{1.5}{V}$. La sortie
vaut $+\qty{15}{V}$ tant que $V_e$ monte jusqu'à \qty{1.5}{V}, bascule à
$-\qty{15}{V}$, y reste jusqu'à ce que $V_e$ descende sous \qty{-1.5}{V},
puis rebascule~: un \omterm{def:b1:wave-propagation:signal}{signal} carré à la fréquence du triangle, dont les
fronts sont aux passages des seuils.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
Avec $V_s = +V_{\mathrm{sat}}$, $V_C$ se charge depuis $-\beta V_{\mathrm{sat}}$
vers $+V_{\mathrm{sat}}$~: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, et atteint $+\beta V_{\mathrm{sat}}$ lorsque $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$~: demi-période $RC\ln\frac{1 + \beta}{1 - \beta}$,
d'où $T$. Pour $\beta = \tfrac12$~: $T = 2RC\ln 3 = 2.2RC$~; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$. $V_s$~: \omterm{def:b1:wave-propagation:signal}{signal} carré
$\pm\qty{15}{V}$~; $V_C$~: arcs exponentiels entre $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} Deux diviseurs~: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone})~; le
câblage en fixe le signe.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ toujours~; $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ en pleine échelle. Les
\qty{2.5}{V} communs aux deux sont la tension de mode commun.

\textbf{4.} Chaque point milieu~: $1 \parallel 1 = \qty{500}{\ohm}$~;
entre les deux points milieux, \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$~: une perte de $9\%$.

\textbf{6.} \omterm{def:b1:sinusoidal-impedance:impedance}{Impédance} d'entrée infinie (aucun courant tiré du pont, donc
les points milieux gardent leurs potentiels à vide) et \omterm{def:b1:sinusoidal-impedance:impedance}{impédance} de
sortie nulle (l'étage suivant peut tirer ce dont il a besoin).

\textbf{7.} Exactement $V_A$ et $V_B$~; courant nul.

\textbf{8.} Comme à la \cref{prop:b1:operational-amplifier:sumdiff}~: $V_+ =
V_BR_2/(R_1 + R_2)$, \omterm{thm:b1:dc-circuits:kirchhoff}{loi des nœuds} en $V_-$, puis $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$~: par exemple \qty{1}{k\ohm} et \qty{9}{k\ohm}.
Gain total $1000$~; pleine échelle $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} Apparié~: les \qty{2.5}{V} communs s'annulent exactement
(seul $V_B - V_A$ survit). Un désappariement de $1\%$ en laisse passer
environ $1\%$~: une erreur ramenée à l'entrée de l'ordre de \qty{25}{mV},
dix fois le \omterm{def:b1:wave-propagation:signal}{signal} de pleine échelle --- les quatre résistances doivent
être appariées à mieux que $0.01\%$ (ou la tension de mode commun
retirée d'abord).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, soit quarante pour
cent de la pleine échelle~: inutilisable sans compensation du décalage,
sans amplificateur à faible décalage, ou sans un étalonnage du zéro
pont non chargé.

\textbf{13.} Étage 1~: $f_T/100 = \qty{10}{kHz}$~; étage 2~: \qty{100}{kHz}~;
très au-dessus de \qty{5}{Hz}.

\textbf{14.} Il faut $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
deux mille fois moins que la \omterm{rem:b1:operational-amplifier:real}{vitesse de balayage}~: non.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$~: $G = 1/\sqrt{26}$, soit \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$~; retard $\varphi/\omega =
\qty{15}{ms}$ --- rien au regard du passage d'un camion.

\textbf{18.} Aucune charge de l'étage suivant (\omterm{def:b1:sinusoidal-impedance:impedance}{impédance} de sortie
nulle) et un gain disponible~; de plus, aucune atténuation du \omterm{def:b1:wave-propagation:signal}{signal}
utile sous la coupure.

\textbf{19.} \qty{-28}{dB} (les gains se multiplient, les \omterm{def:b1:filters-transfer-functions:bode}{décibels}
s'ajoutent), exactement, parce que l'\omterm{def:b1:sinusoidal-impedance:impedance}{impédance} de sortie du premier
étage est nulle~: le second ne le charge pas.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$~; sortie $+V_{\mathrm{sat}}$ lorsque le \omterm{def:b1:wave-propagation:signal}{signal} dépasse
\qty{2.0}{V}, $-V_{\mathrm{sat}}$ en dessous (ou l'inverse, selon le câblage).

\textbf{21.} Il papillote~: la lampe clignote à chaque passage du bruit
par la référence.

\textbf{22.} Avec $V_{\mathrm{ref}}$ par $R_2$ et la sortie par
$R_1$\dots{} sous la forme standard, l'entrée non inverseuse est au
potentiel $V_{\mathrm{ref}}(1 - \beta) + \beta V_s$~; en décalant la
référence en conséquence, les niveaux de basculement valent $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
Comme $\beta V_{\mathrm{sat}} = \qty{0.10}{V}$~: $\beta = 1/150$, par exemple $R_1 =
\qty{1.0}{k\ohm}$ et $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (déclenchement) et \qty{1.9}{V} (retour)~:
déformations \num{1.68e-3} et \num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. À
$-V_{\mathrm{sat}}$, la diode est polarisée en inverse sous \qty{15}{V},
au-delà de ce qu'elle supporte~: une diode ordinaire en antiparallèle
(ou en série) la protège.

\textbf{25.} Pont (\qty{2.5}{mV} en pleine échelle) $\to$ deux suiveurs
(pas d'effet de charge) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{amplificateur différentiel} $\times100$
$\to$ non inverseur $\times10$ $\to$ passe-bas actif à \qty{10}{Hz}
$\to$ \omterm{prop:b1:operational-amplifier:schmitt}{trigger de Schmitt} à $\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ diode.
Les vraies limites~: la \omterm{rem:b1:operational-amplifier:real}{tension de décalage d'entrée} et l'appariement
des résistances de l'\omterm{prop:b1:operational-amplifier:sumdiff}{amplificateur différentiel}, toutes deux comparables
au \omterm{def:b1:wave-propagation:signal}{signal} de quelques millivolts.
\end{solution}
