\section*{Chapitre \ref{ch:b1:kinetic-theory} --- Théorie cinétique et gaz parfait}

\begin{solution}{exo:b1:kinetic-theory:1}
$N = PV/k_BT = 1.013\times10^5 \times 10^{-3}/(1.38\times10^{-23} \times 273) =
\num{2.7e22}$~; $n^{*-1/3} = (3.7\times10^{-26})^{1/3} = \qty{3.3}{nm}$, dix
fois la taille d'une molécule.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:2}
$u = \sqrt{3RT/M}$~: H$_2$ \qty{1930}{m/s}, N$_2$ \qty{517}{m/s}, O$_2$
\qty{484}{m/s}, CO$_2$ \qty{412}{m/s}~; N$_2$ à \qty{77}{K}~: \qty{262}{m/s}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:3}
$P = \qty{3.5}{bar}$ en absolu~: $n = PV/RT = 3.5\times10^5 \times 0.030/(8.314
\times 293) = \qty{4.3}{mol}$, soit $\qty{125}{g}$. À \qty{323}{K}, $P \propto T$~:
\qty{3.86}{\omterm{def:b1:kinetic-theory:pressure}{bar}} (\qty{2.85}{\omterm{def:b1:kinetic-theory:pressure}{bar}} au manomètre).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:4}
$n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$~; $\lambda = 1/(\sqrt2\pi \times 1.37\times
10^{-19} \times 2.4\times10^{25}) = \qty{6.8e-8}{m}$~; fréquence $u/\lambda \approx
500/7\times10^{-8} = \qty{7e9}{s^{-1}}$. $\lambda \propto 1/P$~: \qty{1}{m} pour
$P \approx 10^5 \times 7\times10^{-8} = \qty{7e-3}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:5}
$\rho = PM/RT$~: \qty{1.20}{kg/m^3} à \qty{293}{K}, \qty{0.946}{kg/m^3} à
\qty{373}{K}. Force ascensionnelle $(1.20 - 0.95) \times 2800 \times 9.81 = \qty{7.0}{kN}$~:
environ \qty{720}{kg}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:6}
$P_{\mathrm{O}_2} = 0.21P$~: \qty{0.21}{atm}~; \qty{0.11}{atm} à \qty{5000}{m}~;
\qty{0.07}{atm} sur l'Everest --- le tiers de la valeur au niveau de la mer.
L'oxygène passe dans le sang proportionnellement à sa \omterm{def:b1:kinetic-theory:pressure}{pression} partielle~;
réduite au tiers, elle ne permet plus aux poumons de saturer l'hémoglobine.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:7}
\omterm{def:b1:kinetic-theory:model}{Gaz parfait}~: $P = RT/V = 2494/10^{-3} = \qty{24.9}{bar}$. Van der Waals~:
$RT/(V - b) - a/V^2 = 2494/9.57\times10^{-4} - 0.364/10^{-6} = 26.1 - 3.6 =
\qty{22.4}{bar}$~: l'attraction ($-3.6$ \omterm{def:b1:kinetic-theory:pressure}{bar}) l'emporte sur le volume
exclu ($+1.2$ \omterm{def:b1:kinetic-theory:pressure}{bar})~; le CO$_2$ réel est $10\%$ au-dessous de la valeur idéale.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:8}
$n = PV/RT = 10^5 \times 50/(8.314 \times 300) = \qty{2000}{mol}$~; $U = \tfrac52
nRT = \qty{1.25e7}{J}$~; $\Delta U = \tfrac52 nR \times 5 = \qty{208}{kJ}$~;
\qty{104}{s} à \qty{2}{kW}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:9}
$\Delta L = \alpha L\Delta T = 1.2\times10^{-5} \times 30 \times 50 = \qty{18}{mm}$.
Anneau~: $\Delta d/d = \alpha\Delta T = 0.05/49.95 = 10^{-3}$~: $\Delta T = 10^{-3}/
1.9\times10^{-5} = \qty{53}{K}$ (le trou se dilate comme le métal qui l'entoure).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:10}
$\tfrac32 k_BT = \qty{6.2e-21}{J} = \qty{0.039}{eV}$. Argon~: $\tfrac32 R =
\qty{12.5}{J/(mol.K)}$~; azote~: $\tfrac52 R = \qty{20.8}{J/(mol.K)}$~: la
molécule diatomique stocke aussi de l'énergie en rotation (deux degrés de
liberté de plus), quelle que soit sa masse.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:11}
Modèle à six directions~: comme dans le cours, $P = \tfrac13 n^*mu^2$. Isotropie~:
$\langle v_x^2\rangle = \langle v_y^2\rangle = \langle v_z^2\rangle$ et leur
somme vaut $\langle v^2\rangle$, donc chacun vaut $\tfrac13\langle v^2\rangle$. L'argument
de flux avec une distribution donne $P = n^*m\langle v_x^2\rangle$ (les molécules
de $v_x > 0$ frappent la paroi, chacune cède $2mv_x$, avec un flux $n^*v_x/2$ par
classe de vitesse, que l'on moyenne), c'est-à-dire $\tfrac13 n^*m\langle v^2\rangle$ --- le même.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:12}
H$_2$~: \qty{1.9}{km/s}, O$_2$~: \qty{0.48}{km/s}~; Terre \qty{11.2}{km/s}, Lune
\qty{2.4}{km/s}. Pour O$_2$ sur Terre, $v_e/u = 23$~: la queue de la
distribution au-delà est parfaitement négligeable~; pour H$_2$, $v_e/u \approx 6$~:
une fraction infime mais constante s'échappe chaque année, et en quelques
milliards d'années l'hydrogène a disparu. Sur la Lune, $v_e/u$ vaut $5$ pour O$_2$ et
$1.2$ pour H$_2$~: tout s'est échappé.
\end{solution}

\begin{solution}{pb:b1:kinetic-theory:1}
\textbf{1.} $V = \qty{200}{m^3}$~: $n = PV/RT = 1.013\times10^5 \times 200/(8.314
\times 293) = \qty{8320}{mol}$~; $N = nN_A = \num{5.0e27}$.

\textbf{2.} $m = nM = \qty{241}{kg}$~: trois personnes.

\textbf{3.} $n^* = N/V = \qty{2.5e25}{m^{-3}}$~; distance $n^{*-1/3} = \qty{3.4}{nm}$,
soit neuf diamètres moléculaires.

\textbf{4.} N$_2$~: $\sqrt{3 \times 8.314 \times 293/0.028} = \qty{511}{m/s}$~;
O$_2$~: \qty{478}{m/s}. Même $\tfrac32 k_BT$ par molécule~: $\tfrac12 mu^2$ est fixé,
donc $u \propto 1/\sqrt m$.

\textbf{5.} $\tfrac32 k_BT = \qty{6.1e-21}{J}$~; au total $N \times 6.1\times10^{-21}
= \qty{3.0e7}{J}$~; une voiture de \qty{1000}{kg} à \qty{100}{km/h}~: \qty{3.9e5}{J} ---
l'agitation moléculaire de la salle vaut quatre-vingts voitures.

\textbf{6.} $\lambda = 1/(\sqrt2\pi \times 1.37\times10^{-19} \times 2.5\times10^{25})
= \qty{66}{nm}$~; durée $\lambda/u = 66\times10^{-9}/500 = \qty{1.3e-10}{s}$.

\textbf{7.} Chocs par seconde sur $S$~: $\tfrac16 n^*Su = \tfrac16 \times 2.5\times
10^{25} \times 10^{-4} \times 500 = \qty{2.1e23}{s^{-1}}$.

\textbf{8.} Chacun cède $2mu = 2 \times 4.8\times10^{-26} \times 500 = \qty{4.8e-23}{kg.m/s}$
($m = M/N_A = \qty{4.8e-26}{kg}$)~: force $2.1\times10^{23} \times 4.8\times10^{-23}
= \qty{10}{N}$ sur \qty{1}{cm^2}, soit \qty{1.0e5}{Pa}. Le modèle reproduit
la \omterm{def:b1:kinetic-theory:pressure}{pression} atmosphérique.

\textbf{9.} $F = PS = 1.013\times10^5 \times 20 = \qty{2.0e6}{N}$ --- deux cents
tonnes~; la même \omterm{def:b1:kinetic-theory:pressure}{pression} s'exerce de l'autre côté.

\textbf{10.} $P' = P \times 303/293 = \qty{1.048e5}{Pa}$~; résultante $\Delta P\,S = 3.5
\times10^3 \times 20 = \qty{70}{kN}$ vers l'extérieur~: sept tonnes sur le mur.

\textbf{11.} L'air sort jusqu'à ce que les \omterm{def:b1:kinetic-theory:pressure}{pressions} s'égalisent~: $n \propto 1/T$ à
$P$ et $V$ fixés~: $8320 \times (1 - 293/303) = \qty{275}{mol}$, soit \qty{8}{kg}
qui s'en vont.

\textbf{12.} $\rho = PM/RT$~: \qty{1.205}{kg/m^3} à \qty{293}{K}~; \qty{0.946}{kg/m^3}
à \qty{373}{K}.

\textbf{13.} Air chaud $0.946 \times 2800 \times 9.81 = \qty{26.0}{kN}$~; air froid
déplacé $1.205 \times 2800 \times 9.81 = \qty{33.1}{kN}$~; force ascensionnelle \qty{7.1}{kN}
(\qty{725}{kg}).

\textbf{14.} $725 - 300 = \qty{425}{kg}$~: cinq passagers.

\textbf{15.} Force ascensionnelle $\propto \rho_c - \rho_h = \rho_c(1 - 293/T)$~: doubler
$1 - 293/373 = 0.215$ exige $T = 293/0.57 = \qty{514}{K}$ (\qty{240}{\degreeCelsius}),
bien au-delà de ce que le nylon supporte (environ \qty{120}{\degreeCelsius}).

\textbf{16.} Hélium~: $\rho = 1.205 \times 4/29 = \qty{0.166}{kg/m^3}$, d'où
$(1.205 - 0.166) \times 2800 \times 9.81 = \qty{28.5}{kN}$ (\qty{2.9}{t}), quatre
fois plus --- mais l'hélium coûte cher et se perd à chaque atterrissage,
alors que l'air chaud est gratuit et que le brûleur en règle la force ascensionnelle.

\textbf{17.} $0.21 \times 1.013\times10^5 = \qty{2.1e4}{Pa}$.

\textbf{18.} $n^*_{\mathrm{O}_2} = P_{\mathrm{O}_2}/k_BT$~: dans la salle $2.1\times10^4/
(1.38\times10^{-23} \times 293) = \qty{5.3e24}{m^{-3}}$~; sur l'Everest $0.21 \times 0.33
\times 1.013\times10^5/(1.38\times10^{-23} \times 250) = \qty{2.0e24}{m^{-3}}$~:
$38\%$ de la valeur dans la salle.

\textbf{19.} $H = RT/Mg = 8.314 \times 260/(0.029 \times 9.81) = \qty{7.6}{km}$~;
$P_0/2$ à $H\ln 2 = \qty{5.3}{km}$.

\textbf{20.} $c_s = \sqrt{1.4 \times 8.314 \times 293/0.029} = \qty{343}{m/s}$,
contre $u = \qty{503}{m/s}$ pour le mélange~: $c_s/u = \sqrt{\gamma/3} = 0.68$.
Le son est une perturbation transportée par les molécules elles-mêmes~: il
se propage donc à une vitesse de leur \omterm{def:b1:units-dimensions:oom}{ordre de grandeur} --- un peu moindre,
puisque seule compte la composante selon la direction de propagation.

\textbf{21.} $c_s \propto \sqrt T$~: $343\sqrt{303/293} = \qty{349}{m/s}$.
Hélium~: $\sqrt{(5/3) \times 8.314 \times 293/0.004} = \qty{1010}{m/s}$~: les
résonances du conduit vocal suivent $c_s$, de sorte que chaque formant monte
d'un facteur trois.

\textbf{22.} Plus froid~: $343\sqrt{250/293} = \qty{317}{m/s}$, $8\%$ plus lent~;
la \omterm{def:b1:kinetic-theory:pressure}{pression} n'intervient pas.

\textbf{23.} Même $N$ (mêmes $P$, $V$, $T$)~; masse $\times 40/29 = \qty{333}{kg}$~;
vitesses $\times\sqrt{29/40}$~: $u = \qty{428}{m/s}$~; $U = \tfrac32 nRT$ au lieu de
$\tfrac52$~: $60\%$.

\textbf{24.} $U = \tfrac52 \times 8320 \times 8.314 \times 293 = \qty{5.1e7}{J}$~;
$\Delta U = \tfrac52 nR \times 10 = \qty{1.7e6}{J}$.

\textbf{25.} Une seule moyenne, $\langle v^2\rangle$, fixe la \omterm{def:b1:kinetic-theory:pressure}{pression}
($\tfrac13 n^*m\langle v^2\rangle$), définit la \omterm{def:b1:kinetic-theory:temperature}{température}
($\tfrac12 m\langle v^2\rangle = \tfrac32 k_BT$), donne l'équation d'état
et l'\omterm{def:b1:kinetic-theory:U}{énergie interne}, et fixe l'\omterm{def:b1:units-dimensions:oom}{ordre de grandeur} de la vitesse du son.
\end{solution}
