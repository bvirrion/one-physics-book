\section*{Chapitre \ref{ch:b1:second-law-entropy} --- Le second principe~: l'entropie}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$~: fusion $334000/273 = \qty{1.22}{kJ/K}$~; ébullition $2260000/
373 = \qty{6.06}{kJ/K}$. La vaporisation libère entièrement les molécules les
unes des autres --- bien plus de désordre microscopique, et bien plus de
chaleur par \omterm{def:b1:kinetic-theory:temperature}{kelvin}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- c'est une
fonction d'état~: indépendante de la méthode (l'entropie \emph{créée}, elle, ne l'est pas).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$. Isotherme réversible~: $Q = nRT\ln2$
venus du thermostat à $T$, $S_{\text{éch}} = nR\ln2$, création $0$. Joule~:
$Q = 0$, $S_{\text{éch}} = 0$, création \qty{5.8}{J/K}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
$S = nC_{V,m}\ln T + nR\ln V$ constante~: $T^{C_{V,m}}V^R$ constant, c'est-à-dire
$TV^{R/C_{V,m}} = TV^{\gamma - 1}$ constant. Compression brusque~: $Q = 0$, donc
$S_{\text{éch}} = 0$ et $\Delta S = S_{\text{créée}}$~; avec $T$~: $300 \to
643$ K, $V$~: $24.9 \to 10.7$ L~: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ créés.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
Bloc~: $450\ln(293/373) = \qty{-109}{J/K}$~; lac~: $+36000/293 = \qty{+123}{J/K}$~;
univers~: $+\qty{14}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$~; $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$, entièrement créés. $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$, donc l'argument du logarithme est $\geq 1$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$~: $1000(1/280 - 1/300) = \qty{0.24}{J/K}$~; du four à la
pièce~: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$, dix fois plus~: plus grand est
l'écart de température que franchit un flux de chaleur, plus il est
irréversible --- c'est dans la chute du four à la pièce que l'énergie noble est gaspillée.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
Sur un cycle, $\Delta S = 0 = Q/T_0 + S_{\text{créée}}$ avec $S_{\text{créée}} \geq 0$,
donc $Q \leq 0$ et $W = -Q \geq 0$~: la machine ne peut pas fournir de travail.
La mer est un thermostat unique~: sa chaleur ne peut devenir travail sans un
corps plus froid où déverser l'entropie.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
Chaque gaz se détend de $V$ à $2V$ à $T$ constante~: $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$, créés (aucune chaleur échangée, la boîte est isolée). Deux
moitiés d'azote~: retirer la cloison ne change rien de macroscopique ---
$\Delta S = 0$ (paradoxe de Gibbs~: des molécules identiques ne se «~mélangent~» pas).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
$Q = \qty{3.6e5}{J}$ versés dans la pièce à \qty{293}{K}~: $S = \qty{1230}{J/K}$
créés --- dans le filament (travail électrique dégradé en chaleur) et dans
l'écoulement de cette chaleur du filament chaud vers la pièce froide~; le
travail électrique lui-même ne transportait aucune entropie.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
$\Delta S_{\text{corps}} = C\ln(T_0/T_1)$ dans tous les cas. (a) échangée
$C(T_0 - T_1)/T_0$~: créée $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$. (b) deux étapes de
rapport $r = \sqrt{T_1/T_0}$ chacune~: créée $2C[r - 1 - \ln r]$, plus petite
(pour $T_1/T_0 = 2$~: $0.193C$ contre $0.307C$). (c) une infinité d'étapes~:
chacune crée $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$,
dont la somme tend vers zéro lorsque $\epsilon \to 0$~: un refroidissement réversible
exige un thermostat à chaque température intermédiaire --- la chaleur ne doit jamais franchir d'écart fini.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$~; $2^{-100} = \num{8e-31}$~; $2^{-10^{22}} = 10^{-3\times10^{21}}$.
$k_B\ln2^N = Nk_B\ln2 = nR\ln2$. «~Impossible~» signifie au sens strict une
probabilité nulle~; ici elle vaut $10^{-3\times10^{21}}$ --- non nulle, mais si
petite qu'aucune différence ne pourra jamais être observée~: l'irréversibilité
est une statistique à l'échelle de $10^{23}$.
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$
pour le café~: la pièce gagne \qty{73}{kJ}, $\Delta S_{\text{pièce}} = 73150/293
= \qty{+250}{J/K}$.

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$.

\textbf{3.} $S_{\text{créée}} = -224 - (-250) = \qty{+26}{J/K} > 0$.

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$~; $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$, $S_{\text{éch}} = \qty{-117.7}{J/K}$~: soit
\qty{18}{J/K} créés jusque-là --- l'essentiel de l'irréversibilité se produit
quand le café est le plus chaud.

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$ avec $x = T_1/T_0$~; $f'(x) = 1 - 1/x$ s'annule en $x = 1$, où
$f = 0$, et $f'' = 1/x^2 > 0$~: minimum $0$.

\textbf{6.} Non~: les états initial et final sont les mêmes, la chaleur est la
même, la pièce est le même thermostat --- l'entropie créée est fixée par les
états, non par la vitesse.

\textbf{7.} $T_f = \qty{350}{K}$~; créée $4180\ln(350^2/120000) = \qty{86}{J/K}$.

\textbf{8.} Cas réversible~: l'entropie totale se conserve~: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$, donc $T_f'^2 = T_1T_2$.

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$~; les deux eaux perdent $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$, fournis en
travail.

\textbf{10.} Chaude~: $4180\ln(346.4/400) = \qty{-601}{J/K}$~; froide~: $4180\ln(346.4/
300) = \qty{+601}{J/K}$~: somme nulle, comme l'exige la réversibilité.

\textbf{11.} $T_0S_{\text{créée}} = 300 \times 86 = \qty{26}{kJ}$, proche des
\qty{30}{kJ} récupérables~: l'entropie créée, multipliée par la température de
référence, est le travail que le chemin irréversible a jeté.

\textbf{12.} Il y faudrait un moteur fonctionnant entre deux eaux tièdes dont
les températures changent sans cesse --- quelques dizaines de kilojoules pour
une grosse machine~: cela n'en vaut pas la peine, mais le second principe dit
que le travail était là.

\textbf{13.} Reçue de l'intérieur $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$~;
cédée à la pièce $Q_0/T_0 = (40 + W)/293$~; bilan sur un cycle~:
$0 = 0.144 - (40 + W)/293 + S_c$.

\textbf{14.} $S_c \geq 0$~: $(40 + W)/293 \geq 0.144$, $W \geq 42.2 - 40 =
\qty{2.2}{W}$~; efficacité $\leq 40/2.2 = 18$ ($= T_c/(T_0 - T_c)$).

\textbf{15.} $\mathrm{e}_{\max} = T_c/(T_0 - T_c)$~: pour un congélateur $255/38 = 6.7$
--- plus le compartiment est froid, plus chaque joule extrait coûte cher.

\textbf{16.} Avec $W = 15$~: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ créés
chaque seconde.

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$~; la machine déverse
\qty{200}{W} plus son propre travail dans la pièce tandis que l'intérieur
reste froid~: le gain net de la pièce vaut $W$ --- un réfrigérateur ouvert est un radiateur.

\textbf{18.} $W = 0$ donnerait $S_c = Q_c(1/T_0 - 1/T_c) < 0$~: interdit ---
c'est l'\omterm{prop:b1:second-law-entropy:clausiuskelvin}{énoncé de Clausius}.

\textbf{19.} Avant~: $1$ (tout à gauche)~; après~: $2^N$~; $\Delta S = k_B\ln2^N =
Nk_B\ln2$.

\textbf{20.} $Nk_B = nR$~: $\Delta S = nR\ln2$, le résultat de la détente de Joule.

\textbf{21.} Un bit par molécule~: $N$ bits~; et $\Delta S/(k_B\ln2) = N$ ---
l'entropie compte l'information qu'il faudrait pour fixer la position des
molécules.

\textbf{22.} $2^{-50} = \num{9e-16}$~; pour une \omterm{def:b1:kinetic-theory:scales}{mole} $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$.

\textbf{23.} Aucune chaleur ($Q = 0$)~: échangée $0$, créée $nR\ln2$ --- le cas
le plus pur~: ni travail, ni chaleur, seulement l'étalement du gaz.

\textbf{24.} $W = nRT\ln2$ reçus, $Q = -nRT\ln2$ cédés à la pièce,
$S_{\text{éch}} = -nR\ln2$ (l'entropie du gaz revient à sa valeur
initiale, réversiblement)~: défaire la détente libre coûte le travail $nRT\ln2$
et déverse $nR\ln2$ d'entropie dans la pièce --- l'univers garde les
$nR\ln2$ créés une fois pour toutes.

\textbf{25.} L'entropie n'est conservée dans rien de réel --- seulement dans
la limite réversible idéalisée~; le principe interdit tout processus qui la
détruirait (chaleur remontant la pente, travail tiré d'un seul thermostat,
démixtion gratuite)~; et chaque bribe créée est du travail perdu, à hauteur de $T_0S_{\text{créée}}$.
\end{solution}
