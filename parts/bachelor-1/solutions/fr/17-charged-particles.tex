\section*{Chapitre \ref{ch:b1:charged-particles} --- Particules chargées dans les champs E et B}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$~: électron
\qty{1.9e7}{m/s}~; proton \qty{4.4e5}{m/s} ($\sqrt{1836}$ fois moins vite).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$~; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$~; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$~; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. À $30^\circ$~: une hélice de rayon $R\sin 30^\circ
= \qty{10}{m}$ et de pas $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, qui
dérive le long de la ligne de champ.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Ions plus rapides~: la force
magnétique l'emporte, ils s'incurvent du côté magnétique~; charge opposée
à la même vitesse~: les deux forces changent de sens, l'équilibre
demeure --- le sélecteur est aveugle au signe et à la masse.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$~; déviation
$0.30 \times 0.10 = \qty{3.0}{cm}$~; sensibilité \qty{0.30}{mm/V}. Des
électrons plus rapides passent moins de temps entre les plaques~:
$\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$~: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$~;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$~; écart $2\Delta R =
\qty{1.2}{cm}$ entre les points d'arrivée (les rayons diffèrent de
\qty{5.8}{mm}, les diamètres du double).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$~; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$~; \qty{100}{keV} par tour~: $270$ tours~; durée $270/f_c =
\qty{12}{\micro s}$~; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$~: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$~;
période $2\pi m/eB = \qty{18}{ns}$, pas $v_\parallel T = \qty{1.5}{cm}$~:
environ $64$ tours par mètre.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$~: moins d'un microvolt. Avec un $n$ dix millions de
fois plus petit, un semi-conducteur donne des millivolts --- un capteur utilisable.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$~; en multipliant scalairement par $\vect v$~: $\dd(\tfrac12 mv^2)/
\dd t = 0$ quel que soit $\vect B(\vect r, t)$, pourvu qu'il n'y ait pas de $\vect E$. La
direction de la vitesse, le \omterm{def:b1:point-kinematics:frenet}{rayon de courbure}, le pas peuvent tous
changer~; la valeur de la vitesse et l'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} (\qty{10}{keV}), non.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. Avec $\vect v =
(E/B)\vect e_x + \vect w$~: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- une
pure rotation cyclotron de $\vect w$ à $\omega_c$. Partant du repos,
$\vect w(0) = -(E/B)\vect e_x$~: $\vect w$ tourne à norme constante $E/B$, si bien que la
vitesse est la dérive $E/B$ plus un vecteur tournant de même norme ---
une cycloïde, à rebroussements là où $\vect v = \vect 0$. Électron~: dérive $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, hauteur des arches $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$~; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (environ $15^\circ$)~:
déviation $\approx 0.30 \times 0.27 = \qty{8}{cm}$ pour un modeste
millitesla. De grands angles avec des plaques demanderaient des
kilovolts et de longues plaques~; une bobine courbe le faisceau de
dizaines de degrés sans aucune tension.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$~: puissance nulle,
vitesse constante.

\textbf{2.} $mv^2/R = evB$~: $R = mv/eB$~; $T = 2\pi R/v = 2\pi m/eB$,
indépendante de $v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ par traversée, \qty{100}{keV} par tour.

\textbf{5.} L'intervalle doit être à la polarité accélératrice chaque
fois que le proton s'y présente, à une demi-période d'écart~; hors
fréquence, la phase glisse et les impulsions finissent par freiner.

\textbf{6.} Après un tour, $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$~; après $n$ tours, $E_k =
n \times \qty{100}{keV}$ et $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$~; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ tours~; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} Non~: $E_k$ ne dépend que de $B$ et de $R_{\max}$~; $U$ fixe
le nombre de tours et la durée.

\textbf{11.} Trajet \qty{560}{m}~: il faut $\lambda \gtrsim \qty{1}{km}$, c'est-à-dire
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
un vide poussé.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$~; la période croît de $2.9\%$.
Sur $270$ tours, le glissement s'accumule sur plusieurs périodes --- les
derniers tours seraient déphasés~; en pratique on ajuste $B$ et la phase
d'injection, et \qty{27}{MeV} est proche de la limite de la machine simple.

\textbf{13.} $E_k = (eBR)^2/2m$~: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$~; $R =
\qty{1.1}{m}$ à \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ croît de $24\%$ du centre au bord~: une
fréquence fixe se déphase sans espoir.

\textbf{15.} Synchrocyclotron~: on abaisse la fréquence à mesure que le
paquet gagne de l'énergie --- un seul paquet à la fois, donc un courant
moyen faible. Isochrone~: on profile $B(R) \propto \gamma$ pour que la
période reste constante --- faisceau continu, le choix moderne.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$~; $R = p/eB = \qty{1.2}{m}$, un peu plus que les
\qty{1.1}{m} non relativistes.

\textbf{17.} Impulsion transversale $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$~; angle $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$~: sans espoir pour piloter.

\textbf{18.} En magnétique~: $R = p/eB = \qty{2.3}{m}$, angle $L/R = \qty{0.43}{rad}
\approx 25^\circ$~: mille fois plus --- ce seront donc des aimants.

\textbf{19.} Énergie par proton $\qty{3.68e-11}{J}$~; dose $2.0 \times 0.50 =
\qty{1.0}{J}$~: $2.7\times10^{10}$ protons.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \qty{6.2e9}{protons/s}$~: environ
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$~:
une veilleuse, délivrée exactement là où il faut.

\textbf{22.} $\qty{58}{MeV}$ sur \qty{1}{cm}~: environ \qty{6}{keV/\micro m}
--- une ionisation dense le long du dernier segment.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionisations par proton ---
les dégâts moléculaires qui tuent la cellule.

\textbf{24.} Les rayons X déposent de l'énergie tout au long de leur
parcours, surtout près de l'entrée~; les protons déposent l'essentiel de
la leur en fin de parcours, épargnant les tissus en amont et ne laissant
rien en aval.

\textbf{25.} $R = mv/eB$ et $T = 2\pi m/eB$ (le cercle), $E_k = (eBR)^2/2m$
(l'énergie), $E_k/2eU$ (les tours), dose $=$ (protons $\times$ énergie)/masse
et courant $= e \times$ débit (la durée).
\end{solution}
