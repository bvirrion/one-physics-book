\end{solution}

\begin{solution}{exo:b1:central-forces:3}

\begin{solution}{exo:b1:central-forces:4}

\begin{solution}{exo:b1:central-forces:5}
$\Delta E = \tfrac12 mv^2 + GMm(1/R - 1/r) = 2.94\times10^{10} + 3.99\times
10^{17} \times (1.570 - 1.477)\times10^{-7} = 2.94\times10^{10} + 0.37\times
10^{10} = \qty{3.3e10}{J}$ --- soit \qty{33}{MJ/kg}, l'énergie de \qty{0.7}{kg}
d'essence par kilogramme mis en orbite (les fusées en demandent bien
plus, ayant à soulever leur propre carburant).
\end{solution}

\begin{solution}{exo:b1:central-forces:6}
$a = (0.586 + 35.1)/2 = \qty{17.8}{AU}$~; $e = (35.1 - 0.586)/35.7 = 0.967$~;
$T = 17.8^{3/2} = \qty{75}{yr}$. $v_p^2 = GM(2/r_p - 1/a)$ avec $r_p =
8.77\times10^{10}$ m et $a = 2.67\times10^{12}$ m~: $v_p = \qty{54}{km/s}$~;
$v_a = v_pr_p/r_a = \qty{0.91}{km/s}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:7}
$a = \qty{24490}{km}$. $v_1 = \qty{7.67}{km/s}$~; sur l'ellipse, en $r_1$~:
$\sqrt{GM(2/r_1 - 1/a)} = \qty{10.07}{km/s}$, d'où $\Delta v_1 = \qty{2.4}{km/s}$.
En $r_2$~: $\sqrt{GM(2/r_2 - 1/a)} = \qty{1.62}{km/s}$, contre
\qty{3.07}{km/s} en circulaire~: $\Delta v_2 = \qty{1.45}{km/s}$. Total
\qty{3.85}{km/s}~; durée $\pi\sqrt{a^3/GM} = \qty{1.9e4}{s} = \qty{5.3}{h}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:8}
$E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ en $r_c = L^2/GMm^2$.
$E_{p,\mathrm{eff}}'' = -2GMm/r^3 + 3L^2/mr^4$~; en $r_c$, $L^2 = GMm^2r_c$
donne $E'' = GMm/r_c^3$, donc $\omega_r^2 = E''/m = GM/r_c^3 = \omega_{\text{orb}}^2$.
Une petite perturbation radiale oscille exactement une fois par
révolution~: l'orbite se referme sur elle-même --- une ellipse peu
excentrique, non une rosace.
\end{solution}

\begin{solution}{exo:b1:central-forces:9}
Seule la pesanteur agit, sur la station comme sur l'équipage~; tous deux
tombent sur la même orbite, si bien qu'aucune force de contact n'est
nécessaire entre eux --- aucun plancher n'appuie, d'où l'«~impesanteur~».
$g = GM/r^2 = 3.99\times10^{14}/(6.77\times10^6)^2
= \qty{8.7}{m/s^2}$~: la pesanteur est presque entière~; le poids n'est
pas absent, il est entièrement employé à courber la \omterm{def:b1:point-kinematics:frame}{trajectoire}.
\end{solution}

\begin{solution}{exo:b1:central-forces:10}
$a = 4\pi^2r/T^2 = 39.5 \times 3.84\times10^8/(2.36\times10^6)^2 =
\qty{2.7e-3}{m/s^2}$~; $g(R/r)^2 = 9.81/60.3^2 = \qty{2.7e-3}{m/s^2}$~: la
loi de force qui fait tomber une pomme retient la Lune, diluée en $1/r^2$.
\end{solution}

\begin{solution}{exo:b1:central-forces:11}
$E_k = 2Ze^2/4\pi\varepsilon_0r_{\min}$~: $r_{\min} = 8.99\times10^9 \times 2 \times 79
\times (1.60\times10^{-19})^2/8.0\times10^{-13} = \qty{4.5e-14}{m} = \qty{45}{fm}$,
six fois le rayon nucléaire~: la particule $\alpha$ rebrousse chemin
avant de toucher. Pour atteindre \qty{7}{fm}~: $E_k \approx \qty{32}{MeV}$.
\end{solution}

\begin{solution}{exo:b1:central-forces:12}
$E_m = -GMm/2r$ décroît $\Rightarrow$ $r$ décroît $\Rightarrow$ $v =
\sqrt{GM/r}$ croît. L'\omterm{thm:b1:work-and-energy:ke}{énergie cinétique} $GMm/2r$ augmente de $\abs{\Delta E}$
tandis que l'\omterm{def:b1:work-and-energy:conservative}{énergie potentielle} $-GMm/r$ diminue de $2\abs{\Delta E}$~: la
moitié de l'\omterm{def:b1:work-and-energy:conservative}{énergie potentielle} libérée devient cinétique, l'autre
moitié est dissipée en chaleur par le frottement.
\end{solution}

\begin{solution}{pb:b1:central-forces:1}
\textbf{1.} $T = \qty{43082}{s}$~: $r = (GMT^2/4\pi^2)^{1/3} = (3.986\times10^{14}
\times 1.856\times10^9/39.48)^{1/3} = \qty{2.656e7}{m}$~; altitude
\qty{20190}{km}.

\textbf{2.} $v = \sqrt{GM/r} = \qty{3.87}{km/s}$.

\textbf{3.} $a = v^2/r = GM/r^2 = \qty{0.565}{m/s^2} = g(R/r)^2$, dix-sept
fois plus faible qu'à la surface --- mais non nulle~: le satellite tombe
continûment autour de la Terre.

\textbf{4.} $E_m/m = -GM/2r = \qty{-7.5}{MJ/kg}$~; $E_k/m = \qty{+7.5}{MJ/kg}$,
$E_p/m = \qty{-15.0}{MJ/kg}$.

\textbf{5.} La Terre tourne une fois par jour sidéral par rapport aux
étoiles (le \omterm{def:b1:point-kinematics:frame}{référentiel} de l'orbite)~; après deux orbites, le satellite
repasse au-dessus du même point du sol~: les traces au sol se répètent
chaque jour sidéral.

\textbf{6.} $\sin\beta = R/r = 0.240$~: $\beta = 13.9^\circ$, soit un
disque large de $28^\circ$.

\textbf{7.} La géostationnaire est à \qty{42200}{km}~: $1.59$ fois plus
haut~; l'orbite de navigation est en dessous.

\textbf{8.} $GM(1/R - 1/2r) = 3.986\times10^{14}(1.570\times10^{-7} - 1.88\times
10^{-8}) = \qty{5.5e7}{J/kg} = \qty{55}{MJ/kg}$.

\textbf{9.} $r_1 = \qty{6771}{km}$~: $v = \qty{7.67}{km/s}$.

\textbf{10.} $a = \qty{16665}{km}$~; en $r_1$~: $\sqrt{GM(2/r_1 - 1/a)} =
\qty{9.69}{km/s}$, $\Delta v_1 = \qty{2.0}{km/s}$~; en $r_2$~: \qty{2.47}{km/s}
contre \qty{3.87}{km/s} en circulaire, $\Delta v_2 = \qty{1.4}{km/s}$~; total
\qty{3.4}{km/s}.

\textbf{11.} $\pi\sqrt{a^3/GM} = \qty{1.07e4}{s} \approx \qty{3.0}{h}$.

\textbf{12.} Libération depuis l'orbite basse~: $(\sqrt2 - 1)\times 7.67 = \qty{3.2}{km/s}$~;
les \qty{3.4}{km/s} de la mission la dépassent --- atteindre cette orbite
coûte à peu près autant que quitter la Terre tout entière.

\textbf{13.} $m_0/m_1 = \eu^{3.4/3.0} = 3.1$~: $m_1 = \qty{640}{kg}$, soit
\qty{1360}{kg} d'ergols --- les deux tiers de la masse.

\textbf{14.} $\cos\theta = 0.240$, $\theta = 76^\circ$~; fraction $(1 - 0.240)/2
= 0.38$.

\textbf{15.} $24 \times 0.38 = 9$ en moyenne~; quatre est largement
dépassé presque toujours.

\textbf{16.} Au zénith~: $2.02\times10^7/c = \qty{67}{ms}$~; à l'horizon~:
distance $\sqrt{r^2 - R^2} = \qty{2.58e7}{m}$, soit \qty{86}{ms}.

\textbf{17.} $3/c = \qty{10}{ns}$.

\textbf{18.} Le décalage de l'horloge du récepteur est une quatrième
inconnue, en plus de ses trois coordonnées~: quatre distances, quatre équations.

\textbf{19.} $10^{-8}\ \unit{s}/\qty{86400}{s} \approx \num{1e-13}$.

\textbf{20.} $v^2/2c^2 = (3874)^2/(2 \times 8.99\times10^{16}) = \num{8.3e-11}$~:
$8.3\times10^{-11} \times 86400 = \qty{7.2}{\micro s}$ de retard par jour.

\textbf{21.} $GM(1/R - 1/r)/c^2 = 3.986\times10^{14} \times 1.19\times10^{-7}/
8.99\times10^{16} = \num{5.3e-10}$~: \qty{45.7}{\micro s} d'avance par jour.

\textbf{22.} Bilan $+\qty{38.5}{\micro s}$ par jour~; $c \times 38.5\times10^{-6}
= \qty{11.5}{km}$ d'erreur par jour.

\textbf{23.} $(10.23 - 10.22999999543)/10.23 = \num{4.47e-10}$, ce qui
correspond au bilan $4.5\times10^{-10}$ (\qty{38.5}{\micro s} sur \qty{86400}{s}).

\textbf{24.} Le bourrelet équatorial de la Terre, l'attraction de la
Lune et du Soleil, la pression de radiation solaire (et de petites
poussées des propulseurs)~: le segment sol mesure les orbites en continu
et téléverse des éphémérides fraîches, que les satellites diffusent.

\textbf{25.} L'énergie et le \omterm{def:b1:angular-momentum:L}{moment cinétique} fixent une \omterm{prop:b1:central-forces:effective}{orbite
circulaire} à partir de sa période~; Kepler~III la place~; la géométrie
fixe la couverture et les temps de propagation~; et les horloges --- le
cœur du système --- doivent être corrigées de la relativité de
\qty{38}{\micro s} par jour, faute de quoi le point dérive de dix
kilomètres avant la nuit.
\end{solution}
