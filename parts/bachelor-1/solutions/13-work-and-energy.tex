\section*{Chapter \ref{ch:b1:work-and-energy} --- Work, Energy, and Potential Energy}

\begin{solution}{exo:b1:work-and-energy:1}
Lifting force $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$; weight
$-\qty{589}{J}$; power $589/5.0 = \qty{118}{W}$. Friction: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$; $d = E_k/F =
\qty{55}{m}$; doubling $v$ quadruples $E_k$ and $d$: \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$; $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$: $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
Tension: $T - mg = mv^2/\ell = mg$, $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
Friction \omterm{def:b1:work-and-energy:work}{works} $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$, so
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$; $35\%$ lost.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
$E_m = \tfrac12 mv^2 - GMm/R \geq 0$ for the body to reach infinity
(where $E_p = 0$ and $v \to 0$): $v_e = \sqrt{2GM/R}$, i.e.
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
Same density, $M \propto R^3$: $v_e \propto R$, so \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
Lowest point: $mg(\ell_0 + x) = \tfrac12 kx^2$, $25x^2 - 687x - 13734 = 0$,
$x = \qty{41}{m}$; total drop \qty{61}{m}; $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (net \qty{2}{g} upward). Speed is greatest where the net force
vanishes, $kx = mg$: $x = \qty{14}{m}$, \qty{34}{m} below the jump.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
Gravity: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$;
drag: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$; total \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$: $x = 0, \pm a$. $E_p'' = E_0(12x^2/a^4 -
4/a^2)$: $-4E_0/a^2$ at $0$ (unstable maximum, $E_p = 0$), $+8E_0/a^2$
at $\pm a$ (stable minima, $E_p = -E_0$): wells of depth $E_0$;
$\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$: an ellipse of semi-axes $x_{\max}
= \sqrt{2E/k}$ and $\dot x_{\max} = \sqrt{2E/m}$. The motion is harmonic
with period $2\pi\sqrt{m/k}$ whatever the \omterm{def:b1:wave-propagation:signal}{amplitude} (isochronism), so
every ellipse takes the same time.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
Energy: $v^2 = 2gR(1 - \cos\theta)$. Radial: $mg\cos\theta - N = mv^2/R$,
so $N = mg(3\cos\theta - 2)$, zero at $\cos\theta = 2/3$ ($\theta =
48^\circ$): height $2R/3$, speed $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$: $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
Heat in the tire--road contact (and a skid mark). Rolling wheels use
static friction, $\mu_s > \mu_d$: a larger braking force, a shorter
distance, and steering kept.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} $r \to 0$: $+\infty$ (the $r^{-12}$ term); $r \to \infty$:
$0^-$. A steep wall, a well, a long tail.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ iff
$(\sigma/r)^6 = 1$: $r = \sigma$, $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$: $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$, minimal at $u = 1$ ($r = \sigma$) with value $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$:
positive (repulsive) for $r < \sigma$, negative (attractive) beyond.

\textbf{5.} $(1/1.2)^6 = 0.335$, $(1/1.2)^{12} = 0.112$: $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. $(1/1.2)^{13} = 0.0935$,
$(1/1.2)^7 = 0.279$: $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$,
close to the measured \qty{431}{kJ/mol}.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$; at
$\sigma$: $72\epsilon/\sigma^2 > 0$: stable.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$:
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$; $f = \qty{2.2e14}{Hz}$; $\lambda = c/f = \qty{1.4}{\micro m}$;
$1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} A factor $2.4$ too high: the model's walls are too steep
(the $r^{-12}$ core is a crude stand-in for a covalent bond), but the
\omterm{def:b1:units-dimensions:oom}{order of magnitude} --- an infrared vibration at $10^{14}$ Hz --- is
right, and the method (curvature of the well) is exactly the one used
with better potentials.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$: $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$: the bond
barely shivers.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$: closed curves around $(\sigma, 0)$ ---
vibration, bound; $E_m > 0$: open curves --- the atoms separate;
$E_m = 0$: the separatrix, an atom just able to reach infinity.

\textbf{15.} $u^2 - 2u = -\tfrac12$ with $u = (\sigma/r)^6$: $u = 1 \pm 1/\sqrt2
= 1.71$ or $0.293$; $r = \sigma u^{-1/6}$: $0.915\sigma$ and $1.23\sigma$.

\textbf{16.} Midpoint $1.07\sigma > \sigma$: the outer wall is softer than
the inner one, so the atom spends more room outside; the time-averaged
distance grows with energy --- heated bonds lengthen, solids expand.

\textbf{17.} Longer: the restoring force on the soft outer side is
weaker than the harmonic approximation assumes, so the return takes
more time.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$; as \omterm{thm:b1:work-and-energy:ke}{kinetic
energy} at $\sigma$: $v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}
= \qty{2.0e4}{m/s}$.

\textbf{19.} Energy conservation: $E_m > 0$ stays $> 0$, the curve is
open, the atoms fly apart after one encounter. A third body (another
molecule, a wall, a photon) must carry the excess away.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$, positive: the attraction pulls
the atom in, motive.

\textbf{21.} $W_{\text{ext}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$, $(1/0.9)^7 = 2.09$: $F = +22\epsilon/\sigma
= \qty{+120}{nN}$, ten times the attraction at $1.2\sigma$: compressing a
bond by $10\%$ costs far more than stretching it by $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. Real gases dissociate well below because energies are
distributed: a fraction of molecules carries many times $k_BT$
(\cref{ch:b1:kinetic-theory}).

\textbf{24.} Survive: equilibrium at the minimum, $k = E_p''$ at the
minimum, dissociation energy $=$ depth, \omterm{def:b1:work-and-energy:turning}{turning points} from $E_p = E_m$,
asymmetry $\Rightarrow$ expansion. Change: the values of $k$, of the
frequency and of the \omterm{def:b1:work-and-energy:turning}{turning points}.

\textbf{25.} Equilibrium: $E_p' = 0$; vibration: $\omega_0^2 = E_p''/m$;
escape: $E_m$ compared with the well's rim.
\end{solution}
