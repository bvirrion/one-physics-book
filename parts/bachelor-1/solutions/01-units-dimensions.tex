\section*{Chapter \ref{ch:b1:units-dimensions} --- Units, Dimensions, and Measurement}

\begin{solution}{exo:b1:units-dimensions:1}
$\unit{J} = \unit{kg.m^2.s^{-2}}$; $\unit{W} = \unit{kg.m^2.s^{-3}}$;
$\unit{Pa} = \unit{kg.m^{-1}.s^{-2}}$; $\unit{C} = \unit{A.s}$.
$\unit{Pa.m^3} = \unit{kg.m^{-1}.s^{-2}.m^3} = \unit{kg.m^2.s^{-2}} = \unit{J}$
(the work of pressure forces, $P\,\dd V$).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:2}
Pressure $M L^{-1} T^{-2}$; power $M L^2 T^{-3}$; charge $I\,T$;
voltage $= $ power/current $= M L^2 T^{-3} I^{-1}$.
$[G] = [F r^2/m^2] = M L T^{-2} L^2 M^{-2} = L^3 M^{-1} T^{-2}$;
$[h] = [E/\nu] = M L^2 T^{-2} \cdot T = M L^2 T^{-1}$;
$[k_B] = [E/T] = M L^2 T^{-2} \Theta^{-1}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:3}
$\sqrt{2gh}$: $\sqrt{L T^{-2} \cdot L} = L T^{-1}$, homogeneous.
$\tfrac12 mv$: $M L T^{-1} \neq M L^2 T^{-2}$, not an energy.
$\rho g h$: $M L^{-3} \cdot L T^{-2} \cdot L = M L^{-1} T^{-2}$, a pressure.
$v_0 t + \tfrac12 g t^2$: both terms $L$, homogeneous.
$2\pi\sqrt{g/\ell}$: $\sqrt{T^{-2}} = T^{-1}$ --- a frequency, not a period.
$I_0\eu^{-t/RC}$: $RC$ in $\unit{\ohm.F} = \unit{s}$, argument \omterm{def:b1:units-dimensions:dimension}{dimensionless}, homogeneous.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:4}
Ruler: $\delta = \qty{1}{mm}$, $u = 1/\sqrt{12} = \qty{0.29}{mm}$.
Balance: $u = 0.01/\sqrt{12} = \qty{0.003}{g}$.
Voltmeter: $a = 0.005 \times 4.80 = \qty{0.024}{V}$,
$u = 0.024/\sqrt3 = \qty{0.014}{V}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:5}
$v = k F^\alpha \mu^\beta$: $L T^{-1} = (M L T^{-2})^\alpha (M L^{-1})^\beta$,
so $\alpha + \beta = 0$, $\alpha - \beta = 1$, $-2\alpha = -1$:
$\alpha = \tfrac12$, $\beta = -\tfrac12$, $v = k\sqrt{F/\mu}$ (in fact $k = 1$).
Sound: $P/\rho$ has \omterm{def:b1:units-dimensions:dimension}{dimension} $M L^{-1} T^{-2} / (M L^{-3}) = L^2 T^{-2}$,
so $v = k\sqrt{P/\rho}$ ($k = \sqrt\gamma \approx 1.2$ for air).
\end{solution}

\begin{solution}{exo:b1:units-dimensions:6}
Sum $= \qty{3620}{ms}$, $\bar t = \qty{452.5}{ms}$. Deviations
$-0.5, -5.5, 8.5, 2.5, -3.5, 5.5, -8.5, 1.5$; squares sum to $226$;
$s = \sqrt{226/7} = \qty{5.7}{ms}$; $u(\bar t) = 5.7/\sqrt8 = \qty{2.0}{ms}$.
Result: $t = \qty{452.5\pm2.0}{ms}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:7}
$R = 4.80/0.0124 = \qty{387}{\ohm}$. Relative uncertainties $0.03/4.80 =
0.63\%$ and $0.2/12.4 = 1.6\%$; combined $\sqrt{0.63^2 + 1.6^2} = 1.7\%$,
$u(R) = \qty{7}{\ohm}$: $R = \qty{387\pm7}{\ohm}$. The current
dominates --- improve the ammeter (or its range) first.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:8}
$z = 7/\sqrt{16 + 9} = 1.4 \leq 2$: compatible.
Third vs first: $z = 9/\sqrt{4 + 16} = 2.0$: at the limit, barely
compatible. Third vs second: $z = 16/\sqrt{4 + 9} = 4.4$: incompatible
--- at least one of the two has a systematic error or an
underestimated uncertainty.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:9}
$V = \pi d^3/6 = \pi \times 25.40^3/6 = \qty{8580}{mm^3}$.
$u(V)/V = 3\,u(d)/d = 3 \times 0.02/25.40 = 0.24\%$,
$u(V) = \qty{20}{mm^3}$: $V = \qty{8.58\pm0.02}{cm^3}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:10}
$F = k\eta^a R^b v^c$: $M$: $a = 1$; $T$: $-a - c = -2$, $c = 1$;
$L$: $-a + b + c = 1$, $b = 1$: $F = k\eta R v$ (Stokes, $k = 6\pi$).
With $\rho$ instead: $M$: $a = 1$; $T$: $-c = -2$, $c = 2$; $L$:
$-3 + b + 2 = 1$, $b = 2$: $F = k\rho R^2 v^2$.
Raindrop: $\eta R v = 1.8\times10^{-5} \times 10^{-3} \times 5 \approx
\qty{1e-7}{N}$; $\rho R^2 v^2 = 1.2 \times 10^{-6} \times 25 =
\qty{3e-5}{N}$, three hundred times larger: the inertial (high-speed)
regime applies --- the ratio $\rho v R/\eta \approx 330$ is the Reynolds
number.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:11}
$\bar I = \qty{5.0}{mA}$, $\bar U = \qty{1.000}{V}$; deviations in $I$:
$-3, -1, 1, 3$; in $U$: $-0.59, -0.22, 0.22, 0.59$.
$\sum (I_i - \bar I)(U_i - \bar U) = 3.98$, $\sum (I_i - \bar I)^2 = 20$:
$a = \qty{0.199}{V/mA} = \qty{199}{\ohm}$, $b = 1.000 - 0.199 \times 5 =
\qty{0.005}{V}$. With $u(U) = \qty{0.02}{V}$ per point, an intercept of
\qty{0.005}{V} is well within uncertainty: compatible with zero, the
resistor is ohmic, $R \approx \qty{199}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:units-dimensions:12}
$\ell = G^a h^b c^c$: $M$: $-a + b = 0$; $T$: $-2a - b - c = 0$; $L$:
$3a + 2b + c = 1$. So $b = a$, $c = -3a$, $2a = 1$:
$\ell_P = \sqrt{Gh/c^3} = \sqrt{4.42\times10^{-44}/2.7\times10^{25}}
= \qty{4.1e-35}{m}$; $t_P = \ell_P/c = \qty{1.4e-43}{s}$;
$m_P = \sqrt{hc/G} = \sqrt{1.99\times10^{-25}/6.67\times10^{-11}}
= \qty{5.5e-8}{kg}$. The Planck length is twenty orders of magnitude
below a proton; the Planck mass, \qty{55}{\micro g}, is a speck of dust
--- huge for a particle, tiny for a person ($10^9$ times lighter).
\end{solution}

\begin{solution}{pb:b1:units-dimensions:1}
\textbf{1.} $\sqrt{L/(L T^{-2})} = \sqrt{T^2} = T$.

\textbf{2.} $\tau = k\ell^\alpha m^\beta g^\gamma$ gives
$T = L^{\alpha+\gamma} M^\beta T^{-2\gamma}$, so $\beta = 0$: no
combination of $\ell$ and $g$ can cancel a mass.

\textbf{3.} $\theta_0$ is \omterm{def:b1:units-dimensions:dimension}{dimensionless}, so any factor $f(\theta_0)$ is
invisible to \omterm{def:b1:units-dimensions:dimension}{dimensions}. Keep the amplitude small (below about
$10^\circ$), where $f \approx 1$ to better than $0.2\%$.

\textbf{4.} $g = 4\pi^2\ell/\tau^2$.

\textbf{5.} $\ell = 99.0 + 1.00 = \qty{100.0}{cm} = \qty{1.000}{m}$.

\textbf{6.} $u = \qty{1}{mm}/\sqrt{12} = \qty{0.29}{mm}$.

\textbf{7.} Alignment: $5/\sqrt3 = \qty{2.9}{mm}$; combined
$\sqrt{0.29^2 + 2.9^2} = \qty{2.9}{mm}$ --- the tape reading is
negligible. $u(\ell)/\ell = 0.29\%$.

\textbf{8.} The reaction-time error (about \qty{0.1}{s}) is the same
whether one or ten periods are timed, so its effect on one period is
ten times smaller. Repeating reveals the scatter (hence $s$) and
divides the uncertainty of the mean by $\sqrt8$.

\textbf{9.} Sum $= \qty{160.80}{s}$, $\overline{t_{10}} = \qty{20.100}{s}$.
Deviations $0.02, -0.05, 0.08, -0.01, 0.05, -0.08, 0.01, -0.02$; squares
sum to $\num{188e-4}$; $s = \sqrt{188\times10^{-4}/7} = \qty{0.052}{s}$.

\textbf{10.} $u(\overline{t_{10}}) = 0.052/\sqrt8 = \qty{0.018}{s}$;
$\tau = \qty{2.0100}{s}$, $u(\tau) = \qty{0.0018}{s}$.

\textbf{11.} $u(\tau)/\tau = 0.09\%$, three times smaller than
$u(\ell)/\ell = 0.29\%$.

\textbf{12.} $g = 4\pi^2 \times 1.000/2.0100^2 = 39.48/4.040 =
\qty{9.772}{m/s^2}$.

\textbf{13.} $g \propto \ell\,\tau^{-2}$:
$\dfrac{u(g)}{g} = \sqrt{\left(\dfrac{u(\ell)}{\ell}\right)^2
+ \left(2\dfrac{u(\tau)}{\tau}\right)^2} = \sqrt{0.29^2 + 0.18^2}\,\%
= 0.34\%$.

\textbf{14.} $u(g) = 0.0034 \times 9.772 = \qty{0.033}{m/s^2}$:
$g = \qty{9.77\pm0.03}{m/s^2}$.

\textbf{15.} $z = \abs{9.772 - 9.81}/0.033 = 1.2 \leq 2$: compatible.

\textbf{16.} The length ($0.29\%$ against $0.18\%$ for the period term).
Locating both ends to $\pm\qty{1}{mm}$ brings $u(\ell)/\ell$ to
$0.06\%$ and $u(g)/g$ to $\sqrt{0.06^2 + 0.18^2} = 0.19\%$; doubling
the length halves $u(\ell)/\ell$ and lengthens $\tau$, helping both
terms. Timing twenty periods instead of ten halves the period term
only: $u(g)/g = \sqrt{0.29^2 + 0.09^2} = 0.30\%$ --- a small gain while
the length is the bottleneck.

\textbf{17.} No: a shift common to every reading is a
\emph{systematic error} (a bias); the budget covers random scatter and
declared tolerances, not a consistent offset. It is detected by
changing method or instrument, or --- Part V --- by a non-zero
intercept of the $\tau^2$--$\ell$ line.

\textbf{18.} $\tau^2 = (4\pi^2/g)\,\ell$ is a straight line through the
origin: linearity and the zero intercept are both testable by eye and
by \omterm{prop:b1:units-dimensions:regression}{least squares}; $\tau \propto \sqrt\ell$ is not.

\textbf{19.} $\bar\ell = \qty{0.800}{m}$; $\overline{\tau^2} = 16.16/5 =
\qty{3.232}{s^2}$.

\textbf{20.} Deviations in $\ell$: $-0.4, -0.2, 0, 0.2, 0.4$; in
$\tau^2$: $-1.612, -0.822, 0.018, 0.808, 1.608$. Cross sum $= 1.614$,
$\sum(\ell_i - \bar\ell)^2 = 0.400$: $a = \qty{4.035}{s^2/m}$,
$b = 3.232 - 4.035 \times 0.800 = \qty{0.004}{s^2}$.

\textbf{21.} $g = 4\pi^2/a = 39.48/4.035 = \qty{9.78}{m/s^2}$.

\textbf{22.} $b \approx 0$, as the model predicts. A clearly non-zero
$b$ would betray a systematic offset $\delta$ in every length
($\tau^2 = a(\ell_{\text{meas}} - \delta)$ gives $b = -a\delta$), e.g.\ a
mislocated ball center --- or an amplitude effect.

\textbf{23.} $u(g)/g = u(a)/a = 0.05/4.035 = 1.2\%$, $u(g) =
\qty{0.12}{m/s^2}$: $g = \qty{9.78\pm0.12}{m/s^2}$.

\textbf{24.} $z = \abs{9.772 - 9.784}/\sqrt{0.033^2 + 0.12^2} = 0.1$:
fully compatible. The single careful measurement is more precise; the
line validates the model and guards against a length offset.

\textbf{25.} $\tau = 2\pi\sqrt{10/9.8} = \qty{6.3}{s}$; one swing timed
with a reaction time of about \qty{0.2}{s} gives $u(\tau)/\tau \approx
3\%$, hence $u(g)/g \approx 6\%$, while $u(\ell)/\ell$ drops to
$0.03\%$: twenty times worse than Part IV. The length gain is wasted by
timing a single period. Best strategy: long pendulum \emph{and} many
periods, repeated --- ten swings of the \qty{10}{m} pendulum, eight
times, gives $u(\tau)/\tau \approx 0.03\%$ and $u(g)/g \approx 0.07\%$.
\end{solution}
