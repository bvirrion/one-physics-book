\section*{Chapter \ref{ch:b1:second-law-entropy} --- The Second Law: Entropy}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$: melting $334000/273 = \qty{1.22}{kJ/K}$; boiling $2260000/
373 = \qty{6.06}{kJ/K}$. Vaporization frees the molecules from each other
entirely --- far more microscopic disorder, and far more heat per
\omterm{def:b1:kinetic-theory:temperature}{kelvin}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- a state
function: independent of the method (the entropy \emph{created} is not).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$. Reversible isothermal: $Q = nRT\ln2$
from the bath at $T$, $S_{\mathrm{exch}} = nR\ln2$, created $0$. Joule:
$Q = 0$, $S_{\mathrm{exch}} = 0$, created \qty{5.8}{J/K}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
$S = nC_{V,m}\ln T + nR\ln V$ constant: $T^{C_{V,m}}V^R$ constant, i.e.\
$TV^{R/C_{V,m}} = TV^{\gamma - 1}$ constant. Sudden compression: $Q = 0$, so
$S_{\mathrm{exch}} = 0$ and $\Delta S = S_{\mathrm{created}}$; with $T$: $300 \to
643$ K, $V$: $24.9 \to 10.7$ L: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ created.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
Block: $450\ln(293/373) = \qty{-109}{J/K}$; lake: $+36000/293 = \qty{+123}{J/K}$;
universe: $+\qty{14}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$; $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$, all created. $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$, so the logarithm's argument is $\geq 1$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$: $1000(1/280 - 1/300) = \qty{0.24}{J/K}$; furnace to
room: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$, ten times more: the larger
the temperature gap a heat flow crosses, the more irreversible --- the
furnace-to-room drop is where the high-grade energy is squandered.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
Cycle: $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$ with $S_{\mathrm{created}} \geq 0$,
so $Q \leq 0$ and $W = -Q \geq 0$: the machine cannot deliver work. The
sea is a single thermostat: its heat cannot be turned into work without
a colder body to dump entropy into.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
Each gas expands from $V$ to $2V$ at constant $T$: $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$, created (no heat exchanged, the box is isolated). Two
halves of nitrogen: removing the wall changes nothing macroscopic ---
$\Delta S = 0$ (Gibbs's paradox: identical molecules are not ``mixed'').
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
$Q = \qty{3.6e5}{J}$ into the room at \qty{293}{K}: $S = \qty{1230}{J/K}$
created --- in the filament (electrical work turned into heat) and in the
flow of that heat from the hot filament to the cool room; the electrical
work itself carried no entropy.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
$\Delta S_{\text{body}} = C\ln(T_0/T_1)$ in every case. (a) exchanged
$C(T_0 - T_1)/T_0$: created $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$. (b) two steps
with ratio $r = \sqrt{T_1/T_0}$ each: created $2C[r - 1 - \ln r]$, smaller
(for $T_1/T_0 = 2$: $0.193C$ against $0.307C$). (c) infinitely many steps:
each creates $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$,
summing to zero as $\epsilon \to 0$: reversible cooling needs a thermostat
at every intermediate temperature --- heat must never cross a finite gap.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$; $2^{-100} = \num{8e-31}$; $2^{-10^{22}} = 10^{-3\times10^{21}}$.
$k_B\ln2^N = Nk_B\ln2 = nR\ln2$. ``Impossible'' strictly means probability
zero; here it is $10^{-3\times10^{21}}$ --- not zero, but a number so small that
no difference can ever be observed: irreversibility is statistics at
the scale of $10^{23}$.
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$
for the coffee: the room gains \qty{73}{kJ}, $\Delta S_{\text{room}} = 73150/293
= \qty{+250}{J/K}$.

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$.

\textbf{3.} $S_{\mathrm{created}} = -224 - (-250) = \qty{+26}{J/K} > 0$.

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$; $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$, $S_{\mathrm{exch}} = \qty{-117.7}{J/K}$: created
\qty{18}{J/K} so far --- most of the irreversibility happens while the
coffee is hottest.

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$ with $x = T_1/T_0$; $f'(x) = 1 - 1/x$ vanishes at $x = 1$ where
$f = 0$, and $f'' = 1/x^2 > 0$: minimum $0$.

\textbf{6.} No: initial and final states are the same, the heat is the
same, the room is the same thermostat --- the created entropy is fixed by
the states, not by the speed.

\textbf{7.} $T_f = \qty{350}{K}$; created $4180\ln(350^2/120000) = \qty{86}{J/K}$.

\textbf{8.} Reversible: total entropy conserved: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$, so $T_f'^2 = T_1T_2$.

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$; the waters lose $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$, delivered as
work.

\textbf{10.} Hot: $4180\ln(346.4/400) = \qty{-601}{J/K}$; cold: $4180\ln(346.4/
300) = \qty{+601}{J/K}$: sum zero, as reversibility demands.

\textbf{11.} $T_0S_{\mathrm{created}} = 300 \times 86 = \qty{26}{kJ}$, close to the
\qty{30}{kJ} recoverable: the created entropy, times the reference
temperature, is the work that the irreversible path threw away.

\textbf{12.} It would need an engine working between two lukewarm
waters whose temperatures keep changing --- a few tens of kilojoules for
a large machine: not worth it, but the second law says the work was
there.

\textbf{13.} Received from the interior $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$;
given to the room $Q_0/T_0 = (40 + W)/293$; balance over a cycle:
$0 = 0.144 - (40 + W)/293 + S_c$.

\textbf{14.} $S_c \geq 0$: $(40 + W)/293 \geq 0.144$, $W \geq 42.2 - 40 =
\qty{2.2}{W}$; COP $\leq 40/2.2 = 18$ ($= T_c/(T_0 - T_c)$).

\textbf{15.} $\mathrm{COP}_{\max} = T_c/(T_0 - T_c)$: for a freezer $255/38 = 6.7$
--- the colder the box, the dearer each joule removed.

\textbf{16.} With $W = 15$: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ created
every second.

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$; the machine dumps
\qty{200}{W} plus its own work into the room while the interior stays
cold: the room's net gain is $W$ --- an open refrigerator is a heater.

\textbf{18.} $W = 0$ would give $S_c = Q_c(1/T_0 - 1/T_c) < 0$: forbidden ---
Clausius's statement.

\textbf{19.} Before: $1$ (all left); after: $2^N$; $\Delta S = k_B\ln2^N =
Nk_B\ln2$.

\textbf{20.} $Nk_B = nR$: $\Delta S = nR\ln2$, the Joule-expansion result.

\textbf{21.} One bit per molecule: $N$ bits; and $\Delta S/(k_B\ln2) = N$ ---
entropy counts the information one would need to pin the molecules
down.

\textbf{22.} $2^{-50} = \num{9e-16}$; for a \omterm{def:b1:kinetic-theory:scales}{mole} $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$.

\textbf{23.} No heat ($Q = 0$): exchanged $0$, created $nR\ln2$ --- the
purest case: no work, no heat, only the spreading of the gas.

\textbf{24.} $W = nRT\ln2$ received, $Q = -nRT\ln2$ given to the room,
$S_{\mathrm{exch}} = -nR\ln2$ (the gas's entropy returns to its initial
value, reversibly): undoing the free expansion costs the work $nRT\ln2$
and dumps $nR\ln2$ of entropy into the room --- the universe keeps the
$nR\ln2$ created once and for all.

\textbf{25.} Entropy is conserved in nothing real --- only in the
idealized reversible limit; the law forbids every process that would
destroy it (heat uphill, work from one thermostat, unmixing for free);
and every bit created is work lost, $T_0S_{\mathrm{created}}$ of it.
\end{solution}
