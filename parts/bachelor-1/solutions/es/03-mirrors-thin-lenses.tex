\section*{Capítulo \ref{ch:b1:mirrors-thin-lenses} --- Espejos y lentes delgadas}

\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$; $1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. A \qty{60}{cm}: $\overline{SA'} = \qty{-30}{cm}$,
real, $\gamma = -0.5$. A \qty{30}{cm}: $\overline{SA'} = \qty{-60}{cm}$,
real, $\gamma = -2$. A \qty{10}{cm}: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$, $\overline{SA'} = \qty{+20}{cm}$ detrás del espejo, virtual,
$\gamma = +2$ (derecha, agrandada).
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}

$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ detrás

\begin{solution}{exo:b1:mirrors-thin-lenses:6}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$. La
\omterm{def:b1:mirrors-thin-lenses:image}{imagen} está invertida ($\gamma < 0$), arriba--abajo e izquierda--derecha:

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
Objeto en el plano focal $\Rightarrow$ los rayos de cada punto salen de la
\omterm{def:b1:mirrors-thin-lenses:lens}{lente} paralelos; el \omterm{prop:b1:rays-reflection-refraction:mirror}{espejo plano} los devuelve paralelos (en la dirección
simétrica); la \omterm{def:b1:mirrors-thin-lenses:lens}{lente} enfoca un haz paralelo en su plano focal, en el punto
simétrico del punto fuente respecto del eje: \omterm{def:b1:mirrors-thin-lenses:image}{imagen} en el plano objeto,
invertida, $\gamma = -1$. Inclinar el espejo un ángulo $\alpha$ gira el
haz de vuelta $2\alpha$ y desplaza la \omterm{def:b1:mirrors-thin-lenses:image}{imagen} lateralmente
$2\alpha f'$, que sigue nítida en el mismo plano.
\end{solution}

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ conjuga el infinito en $F_1'$, \qty{20}{cm} más allá de $L_1$, es decir \qty{5}{cm}
más allá de $L_2$: un objeto virtual, $\overline{O_2A_1} = \qty{+5}{cm}$.
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: \omterm{def:b1:mirrors-thin-lenses:image}{imagen} final \qty{10}{cm}
más allá de $L_2$. Altura de la \omterm{def:b1:mirrors-thin-lenses:image}{imagen} intermedia $h_1 = f_1'\alpha = 20\alpha$ (cm);
$\gamma_2 = 10/5 = 2$, luego $h = 40\alpha$: \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} equivalente
\qty{40}{cm}, en un tubo de $15 + 10 = \qty{25}{cm}$ --- un teleobjetivo.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ da $\overline{SA'} =
3\overline{SA}$; ambos reales y por delante: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, y la distancia pantalla--lámpara $2x = \qty{80}{cm}$:
$x = \qty{40}{cm}$. Espejo a \qty{40}{cm} de la lámpara y \qty{120}{cm}
de la pantalla; $1/\overline{SF} = -1/40 - 1/120 = -1/30$:
$R = 2 \times 30 = \qty{60}{cm}$.

\begin{solution}{exo:b1:mirrors-thin-lenses:12}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$; eje orientado según la luz,
distancias algebraicas desde $O$.

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$, es decir $p^2 - Dp + Df_1' = 0$; raíces reales
si y solo si $D^2 - 4Df_1' \geq 0$, es decir $D \geq 4f_1'$.

\textbf{3.} Las raíces suman $D$: $p_1 + p_2 = D$, luego $p_2 = D - p_1$ ---
las distancias objeto e \omterm{def:b1:mirrors-thin-lenses:image}{imagen} se intercambian y las posiciones son
simétricas respecto del punto medio. $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$,
de donde $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} Solo intervienen el desplazamiento $d$ del tubo entre los dos
ajustes nítidos y la distancia objeto--pantalla $D$: ninguno de los dos
exige saber dónde se sitúa $O$ dentro del tubo.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$; luego
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ y $u(f_1') = \qty{0.14}{cm}$:
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ o \qty{60}{cm}: $\gamma =
-60/20 = -3$ (diapositiva cerca de la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, agrandada: el proyector) y
$\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: la \omterm{def:b1:mirrors-thin-lenses:image}{imagen real} está girada $180^\circ$
alrededor del eje, así que la diapositiva se introduce girada $180^\circ$
(boca abajo e invertida derecha--izquierda).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$: la \omterm{def:b1:mirrors-thin-lenses:lens}{lente} se acerca \qty{0.7}{mm}
a la diapositiva. $\gamma = 1000/(-15.23) = -65.7$: \omterm{def:b1:mirrors-thin-lenses:image}{imagen} de
\qty{2.36}{m} de ancho.

\textbf{13.} Derivando Descartes: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, luego $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: la \omterm{def:b1:mirrors-thin-lenses:image}{imagen} se
desenfoca a medida que la película se calienta --- de ahí las diapositivas
montadas entre vidrios, o un autofoco que siga a la película.

\textbf{15.} Triángulos semejantes desde la abertura hasta el punto
\omterm{def:b1:mirrors-thin-lenses:image}{imagen}: mancha $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$; desde \qty{5}{m}
subtiende $\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$: no se aprecia.

\textbf{16.} Área de la diapositiva $8.64\times10^{-4}\ \unit{m^2}$: $E_{\text{diap}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$; el área de la pantalla es $\gamma^2$
veces mayor, $\qty{2.16}{m^2}$: $E_{\text{pant}} = \qty{0.93}{W/m^2}$;
misma potencia, áreas en la razón $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$: la \omterm{def:b1:mirrors-thin-lenses:image}{imagen}
intermedia queda \qty{29.3}{cm} \emph{más allá} de $L_2$ --- un objeto
virtual para $L_2$ (los rayos que salen de $L_1$ convergen hacia ella).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$; $\gamma_2 = 1200/29.3 =
41.0$; $\gamma = -60.9$.

\textbf{20.} Anchura $60.9 \times 36 = \qty{2.19}{m}$. $L_1$ sola:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, anchura
\qty{2.86}{m}.

\textbf{21.} Haz con ángulo $\alpha$: $L_1$ lo enfoca a la altura $h_1 = f_1'\alpha$
en su plano focal, que dista $\overline{O_2A_1} = f_1' - e$ de $L_2$;
$L_2$ conjuga ese punto en $\overline{O_2A'}$ con $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ y lo aumenta en $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$, luego $h = \gamma_2 f_1'\alpha$ y $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, cuyo inverso es la fórmula
enunciada. $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ y $15/19.6 = 0.77$. Solo aproximadamente,
porque la diapositiva está a distancia finita y los dos tiros se miden
desde \omterm{def:b1:mirrors-thin-lenses:lens}{lentes} distintas; para un objeto en el infinito el cociente sería
exacto.

\textbf{23.} Objeto virtual en $p > 0$ más allá de una \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente}:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, positivo (\omterm{def:b1:mirrors-thin-lenses:image}{imagen
real}) si y solo si $p < \abs{f_2'}$.

\textbf{24.} La misma potencia sobre una \omterm{def:b1:mirrors-thin-lenses:image}{imagen} de área $(2.19/2.86)^2 = 0.59$
veces menor: irradiancia $\times 1.7$.

\textbf{25.} El conversor multiplica la \omterm{def:b1:mirrors-thin-lenses:lens}{distancia focal} por
$f_{\mathrm{eq}}/f_1' = 1.3$; a un tiro dado la \omterm{def:b1:mirrors-thin-lenses:image}{imagen} es $1.3$
veces menor en cada dirección y $1.7$ veces más luminosa --- un objetivo
de tiro largo.
\end{solution}
