\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:2}
$\sin i_2 = \sin i_1/1.33$: $30^\circ \to 22.1^\circ$; $60^\circ \to
40.6^\circ$. Desde el agua a $50^\circ$: $1.33\sin 50^\circ = 1.02 > 1$,
no hay rayo refractado --- \omterm{thm:b1:rays-reflection-refraction:tir}{reflexión total} ($i_c = 48.8^\circ$).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:3}
$i_c = \arcsin(1/n)$: vidrio $41.8^\circ$, diamante $24.4^\circ$, agua
$48.8^\circ$. El disco de cielo abarca $2 \times 48.8^\circ = 97.5^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:4}
Ojos a \qty{1.60}{m}. El rayo que sale de los pies llega al ojo tras
reflejarse a media altura del ojo, \qty{0.80}{m}; el rayo que sale de la
coronilla se refleja a mitad de camino entre la coronilla y los ojos, a
\qty{1.65}{m}. Espejo de \qty{0.80}{m} a \qty{1.65}{m}: altura
\qty{0.85}{m}, la mitad de la estatura --- con independencia de la
distancia (la regla del punto medio vale a cualquier distancia).
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:5}
Con ángulos pequeños, la posición aparente es $h' = h/n$ mirando desde el
medio menos refringente y $h' = nh$ desde el más refringente. Moneda: $1.2/1.33 =
\qty{0.90}{m}$. Pescador: $1.33 \times 1.8 = \qty{2.4}{m}$. Justificación:
para un punto de la superficie a distancia horizontal $x$, $\tan i_2 \approx i_2 =
x/h$ y $\tan i_1 \approx i_1 = x/h'$ con $i_1 = n i_2$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:6}
$\sin\frac{A + D_{\min}}{2} = 1.52\sin 30^\circ = 0.760$, luego
$\frac{A + D_{\min}}{2} = 49.5^\circ$ y $D_{\min} = 38.9^\circ$.
La emergencia exige $r' \leq i_c$; además $\sin r = \sin i/n \leq 1/n$ da
$r \leq i_c$; por tanto $A = r + r' \leq 2i_c$. Aquí $i_c = \arcsin(1/1.52) =
41.1^\circ$ y $2i_c = 82.3^\circ > 60^\circ$: se cumple.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:7}
$\mathrm{NA} = \sqrt{1.50^2 - 1.48^2} = \sqrt{0.0596} = 0.244$,
$\theta_a = 14.1^\circ$. $\Delta t/L = (1.50/c)(1.50/1.48 - 1) =
5.0\times10^{-9} \times 0.0135 = \qty{68}{ns/km}$. En \qty{1}{km}:
$B_{\max} \approx 1/(2 \times \qty{68}{ns}) = \qty{7.4}{Mbit/s}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:8}
$D \approx (n - 1)A$: rojo $0.510 \times 10^\circ = 5.10^\circ$, azul
$5.30^\circ$; anchura del espectro $0.20^\circ$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:9}
Entrada: $\sin r = \sin 45^\circ/1.5 = 0.471$, $r = 28.1^\circ$. Las dos
caras son paralelas, así que la incidencia sobre la segunda cara vale $r$ y
$\sin i' = n\sin r = \sin i$: $i' = i$, el rayo emergente es paralelo al
incidente. A lo largo del trayecto interior, de longitud $e/\cos r$, el
desplazamiento perpendicular vale $(e/\cos r)\sin(i - r)$:
$d = 2.0 \times \sin 16.9^\circ/\cos 28.1^\circ = \qty{0.66}{cm}$.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:10}
Dos refracciones ($2(i - r)$) y dos reflexiones ($2(\pi - 2r)$):
$D = 2\pi + 2i - 6r$. $\dd D/\dd i = 2 - 6\cos i/(n\cos r) = 0$ da
$3\cos i = n\cos r$, $9\cos^2 i = n^2 - 1 + \cos^2 i$, $\cos^2 i =
(n^2 - 1)/8$. Con $n = 1.333$: $\cos^2 i = 0.0971$, $i = 71.8^\circ$,
$\sin r = 0.950/1.333$, $r = 45.5^\circ$, $D = 360^\circ + 143.7^\circ -
272.8^\circ = 230.9^\circ$: el arco está a $230.9^\circ - 180^\circ =
51^\circ$ del punto antisolar, por fuera del primario ($42^\circ$).
El rojo ($n = 1.331$) a $50.4^\circ$ y el violeta ($1.343$) a $53.6^\circ$:
los colores se invierten, el violeta por fuera.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:11}
$\sin i_c = 1.00026/1.00029 = 0.99997$, $i_c = 89.56^\circ$: los rayos
que no se apartan más de $0.44^\circ = \qty{7.7}{mrad}$ de la superficie
de la carretera se reflejan totalmente (en general, el ángulo rasante
vale $\approx \sqrt{2\Delta n/n}$). Desde \qty{1.2}{m} de altura, la
visual roza con ese ángulo más allá de $1.2/0.0077 \approx \qty{160}{m}$:
la carretera refleja allí el cielo --- el ``charco''.
\end{solution}

\begin{solution}{exo:b1:rays-reflection-refraction:12}
$t(x) = \dfrac{\sqrt{a^2 + x^2}}{v_1} + \dfrac{\sqrt{b^2 + (d - x)^2}}{v_2}$;
$t'(x) = \dfrac{x}{v_1\sqrt{a^2 + x^2}} - \dfrac{d - x}{v_2\sqrt{b^2 + (d - x)^2}}
= \dfrac{\sin i_1}{v_1} - \dfrac{\sin i_2}{v_2}$, nulo exactamente cuando
$\sin i_1/v_1 = \sin i_2/v_2$, es decir $n_1\sin i_1 = n_2\sin i_2$ con
$n = c/v$.
\end{solution}

\begin{solution}{pb:b1:rays-reflection-refraction:1}
\textbf{1.} $n = c/v$; $v_1 = 3.00\times10^8/1.480 = \qty{2.03e8}{m/s}$.

\textbf{2.} $t = Ln_1/c = 5.0\times10^4 \times 1.48/3.00\times10^8 =
\qty{247}{\micro s}$.

\textbf{3.} $\nu = c/\lambda_0 = \qty{1.94e14}{Hz}$, invariable en el vidrio;
$\lambda = \lambda_0/n_1 = \qty{1047}{nm}$.

\textbf{4.} Un \omterm{def:b1:rays-reflection-refraction:fiber}{núcleo} desnudo solo guía mientras su superficie esté
perfectamente limpia e intacta: el polvo, el agua o una mano sustituyen
localmente el aire por un índice mayor y la luz se escapa, y las rayaduras
la dispersan. El \omterm{def:b1:rays-reflection-refraction:fiber}{revestimiento} es una interfaz sellada y controlada, con
un salto de índice elegido a propósito.

\textbf{5.} $\sin i_c = 1.465/1.480 = 0.9899$, $i_c = 81.8^\circ$.

\textbf{6.} $90^\circ - 81.8^\circ = 8.2^\circ$.

\textbf{7.} $\sin\theta = n_1\sin\theta'$; guiado si y solo si $\cos\theta' \geq
n_2/n_1$, es decir $\sin\theta' \leq \sqrt{1 - n_2^2/n_1^2}$,
es decir $\sin\theta \leq \sqrt{n_1^2 - n_2^2} = \sqrt{2.1904 - 2.1462} = 0.210$:
$\theta_a = 12.1^\circ$.

\textbf{8.} $\mathrm{NA} = 0.210$. Un salto mayor ensancha el cono, así
que se capta más luz de una fuente divergente (un LED).

\textbf{9.} El rayo es tangente a la circunferencia de radio $R_b$ (borde
interior) y alcanza la de radio $R_b + 2a$ (borde exterior); en el
triángulo rectángulo centro--punto de tangencia--punto de impacto, el
ángulo en el punto de impacto entre el rayo y el radio (la normal) cumple
$\sin i = R_b/(R_b + 2a)$. Guiado si y solo si $\sin i \geq n_2/n_1$:
$R_b \geq 2a\,\dfrac{n_2/n_1}{1 - n_2/n_1} = \qty{50}{\micro m} \times 97.6
= \qty{4.9}{mm}$. Con curvas más cerradas la luz se escapa.

\textbf{10.} $t_0 = Ln_1/c$; el rayo más inclinado recorre $L/\cos\theta'_{\max} =
Ln_1/n_2$, luego $t_1 = Ln_1^2/(n_2 c)$.

\textbf{11.} $\Delta t/L = (n_1/c)(n_1/n_2 - 1) = 4.93\times10^{-9} \times
0.01024 = \qty{50.5}{ns/km}$; en \qty{2}{km}: \qty{101}{ns}.

\textbf{12.} $B_{\max} \approx 1/(2\Delta t) = 1/\qty{202}{ns} \approx
\qty{5}{Mbit/s}$.

\textbf{13.} $B_{\max} L = 1/(2\,\Delta t/L) = 1/(2 \times
\qty{50.5}{ns/km}) \approx \qty{10}{Mbit.s^{-1}.km}$.

\textbf{14.} Los rayos alejados del eje recorren caminos más largos, pero
en un índice menor y, por tanto, más deprisa; con un perfil bien elegido
los dos efectos se compensan y todos los rayos llegan juntos.

\textbf{15.} $\qty{9}{\micro m} \approx 6\lambda_0$: el \omterm{def:b1:rays-reflection-refraction:fiber}{núcleo} no es
grande frente a la longitud de onda, así que domina la difracción y los
rayos carecen de sentido. El tratamiento ondulatorio muestra un único
patrón guiado: ningún ensanchamiento intermodal; solo quedan la dispersión
cromática y la atenuación.

\textbf{16.} \qty{100}{km} $\times$ \qty{0.2}{dB/km} $= \qty{20}{dB}$:
$P/P_0 = 10^{-2}$, el uno por ciento.

\textbf{17.} $10\log(P_0/P_{\min}) = \qty{30}{dB}$, luego $L_{\max} =
30/0.20 = \qty{150}{km}$.

\textbf{18.} \qty{100}{km} cuestan \qty{20}{dB}: margen de \qty{10}{dB},
un factor $10$ en potencia.

\textbf{19.} LED: $\Delta t_c = 17 \times 40 \times 100 = \qty{68000}{ps}
= \qty{68}{ns}$, $B \approx 1/\qty{136}{ns} = \qty{7}{Mbit/s}$. Láser:
$\Delta t_c = \qty{170}{ps}$, $B \approx 1/\qty{340}{ps} =
\qty{2.9}{Gbit/s}$.

\textbf{20.} $30/2.0 = \qty{15}{km}$, diez veces menos. La sílice es más
transparente cerca de \qty{1550}{nm} (su mínimo de atenuación), y allí
existen amplificadores ópticos.

\textbf{21.} $30/30 = \qty{1}{km}$ entre amplificadores, frente a
\qty{150}{km}: la fibra gana por un factor $150$.

\textbf{22.} Intermodal: \qty{101}{ns} ($\to$ \qty{5}{Mbit/s});
cromático (láser): $17 \times 0.1 \times 2 = \qty{3.4}{ps}$;
atenuación \qty{0.4}{dB}, despreciable. Limita el ensanchamiento
intermodal: \qty{5}{Mbit/s}.

\textbf{23.} Sin ensanchamiento intermodal; cromático \qty{170}{ps} $\to$
\qty{2.9}{Gbit/s}; atenuación \qty{20}{dB}, dentro del balance. Limita la
dispersión cromática, a unos \qty{3}{Gbit/s}.

\textbf{24.} $80 \times 2.9 \approx \qty{230}{Gbit/s}$.

\textbf{25.} Índice escalonado: \qty{10}{Mbit.s^{-1}.km}; canal monomodo:
$2.9 \times 100 = \qty{290}{Gbit.s^{-1}.km}$ --- un factor de unos
$3\times10^4$. Un solo hecho: un único camino guiado y, por tanto, ningún
ensanchamiento geométrico (intermodal) de los tiempos de llegada.
\end{solution}
