\section*{Capítulo \ref{ch:b1:transient-regimes} --- Regímenes transitorios: primer y segundo orden}

\begin{solution}{exo:b1:transient-regimes:1}
$\tau = RC = 10^4 \times 47\times10^{-9} = \qty{0.47}{ms}$; $u_C(\tau)/E
= 1 - \eu^{-1} = 0.63$; $99\%$: $\eu^{-t/\tau} = 0.01$, $t = \tau\ln 100
= 4.6\tau = \qty{2.2}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:2}
$\tau = L/R = \qty{4.0}{ms}$; $i_\infty = E/R = \qty{2.0}{A}$; $i(\tau) =
0.63 \times 2.0 = \qty{1.26}{A}$; en $0^+$, $i = 0$ (continuidad), luego
$u_L = E - Ri = \qty{10}{V}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:3}
$\omega_0 = 1/\sqrt{LC} = \qty{3.16e4}{rad/s}$, $f_0 = \qty{5.03}{kHz}$;
$Q = \sqrt{L/C}/R = 316/100 = 3.2 > \tfrac12$: seudoperiódico;
$\tau = 2Q/\omega_0 = \qty{0.20}{ms}$; $\omega = \omega_0\sqrt{1 - 1/4Q^2}
= 0.987\omega_0$, $T = \qty{0.20}{ms}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:4}
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$, $Q = \tfrac12$. A \qty{2}{k\ohm},
$Q = 0.16$: aperiódico, una subida lenta gobernada por la raíz pequeña
$\abs{r_+} \approx \omega_0 Q = \qty{5e3}{s^{-1}}$ (escala de tiempo
\qty{0.2}{ms}, pero sin oscilación).
\end{solution}

\begin{solution}{exo:b1:transient-regimes:5}
$u = U_0\eu^{-t/\tau}$ se reduce a la mitad cuando $\eu^{-t/\tau} = \tfrac12$: $t_{1/2}
= \tau\ln 2$. $\tau = 3.5/0.693 = \qty{5.05}{ms}$, $R = \tau/C =
\qty{5.1}{k\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:6}
$i = (E/R)\eu^{-t/\tau}$: $\int_0^\infty Ri^2\,\dd t = (E^2/R)(\tau/2) =
\tfrac12 CE^2$, sin rastro de $R$. La fuente aporta $CE^2$: rendimiento
del $50\%$ con cualquier \omterm{def:b1:dc-circuits:resistor}{resistor}. Mejor: cargar a través de una bobina
(la corriente ya no es proporcional a la diferencia de tensión) o en
varios escalones de tensión --- cada escalón de $E/n$ solo pierde
$\tfrac12 C(E/n)^2$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:7}
$\delta = \ln(2.0/1.5) = 0.29$ (y $\ln(1.5/1.12) = 0.29$: coherente);
$Q \approx \pi/\delta = 11$. $T = \qty{0.40}{ms}$, luego $\omega_0 \approx
2\pi/T = \qty{1.57e4}{rad/s}$ (la corrección $1/8Q^2$ es del $0.1\%$).
$L = 1/(\omega_0^2 C) = \qty{4.1}{mH}$; $R = \omega_0 L/Q = \qty{5.8}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:8}
$u_C \approx U_0\eu^{-t/\tau}\cos\omega_0 t$; $i = -C\,\dd u_C/\dd t
\approx CU_0\omega_0\sin\omega_0 t$ (para $t \ll \tau$), con máximo $U_0\sqrt{C/L}
= 100 \times 0.1 = \qty{10}{A}$, alcanzado un cuarto de periodo después de
cerrar, cuando $u_C = 0$: toda la energía ($\tfrac12 CU_0^2 = \qty{50}{mJ}$)
es entonces magnética, $\tfrac12 Li^2 = \qty{50}{mJ}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:9}
$\omega_0 = \sqrt{k/m} = \qty{7.1}{rad/s}$, $f_0 = \qty{1.1}{Hz}$;
$\alpha_c = 2\sqrt{km} = 2\sqrt{8\times10^6} = \qty{5.7e3}{N.s/m}$.
$\alpha = 2000$: $Q = \sqrt{km}/\alpha = 2828/2000 = 1.4$, seudoperiódico;
$T = 2\pi/(\omega_0\sqrt{1 - 1/4Q^2}) = 0.89/0.935 = \qty{0.95}{s}$;
$\tau = 2Q/\omega_0 = \qty{0.40}{s}$.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:10}
$i = I_0\eu^{-t/\tau}$, $\tau = L/R = \qty{5.0}{ms}$; tensión máxima
$RI_0 = \qty{200}{V}$ en $t = 0^+$; energía $\tfrac12 LI_0^2 = \qty{1.0}{J}$
disipada en $R$. Sin el \omterm{def:b1:dc-circuits:resistor}{resistor} la corriente no tiene camino:
$\dd i/\dd t$ es enorme, la tensión de la bobina llega a los kilovoltios y
salta un arco en el interruptor.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:11}
Visto desde $C$: $E_{\mathrm{Th}} = ER_2/(R_1 + R_2) = \qty{8.0}{V}$,
$R_{\mathrm{Th}} = R_1 \parallel R_2 = \qty{6.67}{k\ohm}$; $\tau =
R_{\mathrm{Th}}C = \qty{6.7}{ms}$; $u_C = 8.0(1 - \eu^{-t/\tau})$ V.
\end{solution}

\begin{solution}{exo:b1:transient-regimes:12}
Raíz doble $-\omega_0$: $x = x_\infty + (A + Bt)\eu^{-\omega_0 t}$;
$x(0) = 0$ da $A = -x_\infty$ y $x'(0) = 0$ da $B = \omega_0 A$:
$x = x_\infty[1 - (1 + \omega_0 t)\eu^{-\omega_0 t}]$. Para el $95\%$:
$(1 + s)\eu^{-s} = 0.05$, $s \approx 4.7$: $t_{95} = 4.7/\omega_0$. Para
$Q = 5$ la envolvente $\eu^{-t/\tau}$ con $\tau = 10/\omega_0$ necesita
$3\tau = 30/\omega_0$; para $Q = 0.1$ la raíz lenta es $\abs{r_+} \approx
\omega_0 Q$ y $3/\abs{r_+} = 30/\omega_0$. Ambos son seis veces más
lentos: el amortiguamiento crítico es el retorno sin sobreoscilación más
rápido, que es lo que quiere un aparato de medida, un cierrapuertas o una
suspensión.
\end{solution}

\begin{solution}{pb:b1:transient-regimes:1}
\textbf{1.} $E = L\,\dd i/\dd t + Ri + u_C$, $i = C\,\dd u_C/\dd t$:
$LC\,u_C'' + RC\,u_C' + u_C = E$.

\textbf{2.} $u_C'' + (\omega_0/Q)u_C' + \omega_0^2u_C = \omega_0^2E$ con
$\omega_0 = 1/\sqrt{LC}$, $Q = \sqrt{L/C}/R$.

\textbf{3.} $\omega_0 = \qty{3.16e4}{rad/s}$, $f_0 = \qty{5.03}{kHz}$,
$R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$.

\textbf{4.} $r^2 + (\omega_0/Q)r + \omega_0^2 = 0$, $\Delta = \omega_0^2(1/Q^2
- 4)$: $Q < \tfrac12$ aperiódico, $Q = \tfrac12$ crítico, $Q > \tfrac12$
seudoperiódico.

\textbf{5.} $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$,
$u_C = E + \eu^{-t/\tau}(A\cos\omega t + B\sin\omega t)$.

\textbf{6.} $Q = 3.16$, $\tau = \qty{0.20}{ms}$, $T = 2\pi/\omega =
\qty{0.20}{ms}$, unas tres oscilaciones.

\textbf{7.} $u_C(0^+) = U_0$ ($u_C$ continua: corriente finita);
$u_C'(0^+) = i(0^+)/C = 0$ ($i_L$ continua y nula antes).

\textbf{8.} $m\ddot x = -k(x + x_{\text{eq}}) - \alpha\dot x + mg$ con
$kx_{\text{eq}} = mg$: $m\ddot x + \alpha\dot x + kx = 0$.

\textbf{9.} $\omega_0 = \sqrt{k/m}$, $Q = \sqrt{km}/\alpha$; $x \leftrightarrow
q$, $\dot x \leftrightarrow i$, $m \leftrightarrow L$, $\alpha \leftrightarrow
R$, $k \leftrightarrow 1/C$.

\textbf{10.} $\omega_0 = \sqrt{22000/350} = \qty{7.93}{rad/s}$, $f_0 =
\qty{1.26}{Hz}$; $\alpha_c = 2\sqrt{km} = \qty{5.55e3}{N.s/m}$.

\textbf{11.} $Q = 2775/4000 = 0.69$, $\xi = 0.72$: seudoperiódico, justo
por debajo del crítico.

\textbf{12.} $\tau = 2Q/\omega_0 = \qty{0.18}{s}$; $\omega = \omega_0\sqrt{1
- \xi^2} = \qty{5.5}{rad/s}$, $T = \qty{1.1}{s}$.

\textbf{13.} Sobreoscilación $\exp(-\pi \times 0.72/0.69) = \exp(-3.3) = 3.8\%$:
una pequeña subida por encima del equilibrio y se acabó --- cómodo.

\textbf{14.} $R = \sqrt{L/C}/Q = 316/0.69 = \qty{455}{\ohm}$.

\textbf{15.} $Q = 2775/1200 = 2.3$, $\xi = 0.22$; $\omega = 0.976\omega_0$,
$T = \qty{0.81}{s}$; $\tau = 2Q/\omega_0 = \qty{0.58}{s}$.

\textbf{16.} $\eu^{-T/\tau} = \eu^{-1.39} = 0.25$: cada rebote es la cuarta
parte del anterior; $0.25^3 = 1.6\%$: tres rebotes visibles.

\textbf{17.} $x(0) = -x_0$, $\dot x(0) = 0$; energía $\tfrac12 kx_0^2 =
0.5 \times 22000 \times 0.0025 = \qty{28}{J}$, disipada en forma de calor
en el aceite del amortiguador.

\textbf{18.} $v \approx x_0\omega_0 = 0.05 \times 7.93 = \qty{0.40}{m/s}$;
fuerza del amortiguador $\alpha v \approx 1200 \times 0.40 = \qty{480}{N}$.

\textbf{19.} Con $Q = \tfrac12$ el retorno es $(1 + \omega_0 t)\eu^{-\omega_0
t}$, que se aproxima asintóticamente: los últimos milímetros tardan mucho.
Un sistema ligeramente subamortiguado ($\xi \approx 0.7$) sobrepasa unos
pocos por ciento, pero entra antes en una banda de tolerancia y se queda.

\textbf{20.} $\exp(-\pi \times 0.22/0.976) = \exp(-0.70) = 50\%$, frente al
$3.8\%$: el coche gastado rebota media \omterm{def:b1:wave-propagation:signal}{amplitud} por encima del equilibrio.

\textbf{21.} $\delta = \ln 4 = 1.39$ (dos veces, coherente); $\xi =
\delta/\sqrt{4\pi^2 + \delta^2} = 1.39/6.43 = 0.215$, $Q = 1/2\xi = 2.3$:
el valor del amortiguador gastado.

\textbf{22.} $-\ln 0.038 = 3.27 = \pi\xi/\sqrt{1 - \xi^2}$, luego $\xi =
3.27/\sqrt{\pi^2 + 3.27^2} = 0.72$, $Q = 0.69$.

\textbf{23.} La energía es $\propto$ \omterm{def:b1:wave-propagation:signal}{amplitud}$^2$: fracción restante por
periodo $0.25^2 = 6\%$, luego se disipa el $94\%$ por seudoperiodo.

\textbf{24.} $R = 316/2.3 = \qty{137}{\ohm}$. En la variable $\omega_0 t$
la ecuación canónica solo contiene $Q$: dos sistemas con el mismo $Q$
trazan la misma curva.

\textbf{25.} El \omterm{def:b1:transient-regimes:secondorder}{factor de calidad} $Q$ (o $\xi$) fija por sí solo la forma;
un diseñador apunta a $Q \approx 0.7$ ($\xi \approx 0.7$); más de un
rebote visible significa un $Q$ bastante mayor que $1$: el amortiguador
está gastado.
\end{solution}
