\section*{Capítulo \ref{ch:b1:systems-of-points} --- Sistemas de puntos; introducción al sólido rígido}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$. Tierra--Luna:
$384000/82 = \qty{4700}{km}$ desde el centro de la Tierra, dentro de ella
(\qty{6370}{km}): la Tierra bambolea en torno a un punto situado
\qty{1700}{km} bajo su superficie.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: $v = \qty{2.0}{m/s}$. Bala $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$, fusil \qty{8}{J}: de la energía química de
la pólvora; el cuerpo ligero se lleva casi toda ($E_k = p^2/2m$ con la
misma $p$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$; $E_k$: \qty{200}{kJ} antes y $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$ después: se pierde el $60\%$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
Extremo: $\int_0^L x^2(M/L)\dd x = ML^2/3$; centro: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$; Huygens con $d = L/2$: $ML^2/12 + ML^2/4 = ML^2/3$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: cilindro $1.5$ y aro $2$ veces $\tfrac12 Mv^2$.
A igual rapidez, más energía para el aro: rueda más lejos en un suelo
plano (mismo rozamiento) y sube más alto ($h = v^2(1 + k)/2g$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
HCl: $\mu = 35m_H/36 = 0.972m_H$; átomo: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$. Las frecuencias son $\propto 1/\sqrt\mu$: $+1.4\%$ para el HCl y
$+0.027\%$ para el hidrógeno (en ello se apoya el desplazamiento isotópico
entre los espectros de H y de D).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$, $d = L/2$: $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$; longitud equivalente $2L/3 = \qty{0.67}{m}$. Un golpe en
el centro de percusión pone la varilla a girar en torno al pivote sin
reacción impulsiva allí: golpear la pelota en el ``punto dulce'' del bate
y las manos no sienten el rebote.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$; $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$, $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$;
$\Gamma = J\omega/t = \qty{314}{N.m}$; vueltas: $\tfrac12\omega t/2\pi = 250$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: esfera ($k = 2/5$) $0.714$, cilindro ($1/2$)
$0.667$ y aro ($1$) $0.500$ veces $g\sin\alpha$: la esfera primero y el aro
el último; ni masa ni radio por ninguna parte.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
Cantidad de movimiento: $0.010 \times 400 = 2.01\,v$: $v = \qty{2.0}{m/s}$;
después, energía: $h = v^2/2g = \qty{0.20}{m}$. $E_k$: \qty{800}{J} antes y
$\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$ después: un $99.5\%$ perdido en deformación
y calor. El impacto es inelástico --- la energía no se conserva a través
de él, la cantidad de movimiento sí (la tensión del hilo es vertical y el
impacto brevísimo).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ y $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ dan
$v_1 + v_1' = v_2'$ (divídase la ecuación de la energía por la de la
cantidad de movimiento), de donde las fórmulas. Fracción de energía cedida
al blanco $4m_1m_2/(m_1 + m_2)^2$: protón, $100\%$; carbono,
$4 \times 12/169 = 28\%$; plomo, $4 \times 207/208^2 =
1.9\%$. Los núcleos ligeros se quedan con la energía del neutrón en unos
pocos choques (agua, grafito); el plomo necesitaría centenares.
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} Sobre el eje, en el centro del disco, por simetría.

\textbf{2.} König: $E_k = \tfrac12 Mv_G^2 + E_k^*$, y en el \omterm{def:b1:systems-of-points:barycentric}{sistema
baricéntrico} el yoyó gira sobre su eje: $E_k^* = \tfrac12 J\omega^2$.

\textbf{3.} Anillo de radio $\rho$ y masa $2M\rho\,\dd\rho/R^2$: $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$.

\textbf{4.} La cuerda está fija y se desenrolla del eje: el punto del eje
en contacto con ella está momentáneamente en reposo, luego $v_G = r\omega$.

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$; $R^2/2r^2 = 9/(2 \times 0.16) = 28$: factor $29$ ---
casi toda la energía está en el giro.

\textbf{6.} Peso $Mg$ hacia abajo y tensión $T$ hacia arriba: $Ma_G = Mg - T$.

\textbf{7.} Solo la tensión tiene momento respecto del eje (\omterm{def:b1:angular-momentum:moment}{brazo de
palanca} $r$): $J\dot\omega = Tr$.

\textbf{8.} $\dot\omega = a_G/r$: $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$; entonces
$Ma_G = Mg - (R^2/2r^2)Ma_G$, $a_G = g/29 = \qty{0.34}{m/s^2}$.

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$, el $97\%$
del peso: la cuerda casi lo sostiene.

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$; $v_G = a_Gt = \qty{0.82}{m/s}$.
Una piedra: \qty{0.45}{s} y \qty{4.4}{m/s}.

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$.

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ frente a $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: $28$ veces más energía en el giro que en la caída.

\textbf{13.} $Mgh = \qty{0.49}{J}$; $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$. La tensión actúa en el punto de la
cuerda que está en reposo (la cuerda no se mueve): potencia nula.

\textbf{14.} La fuerza que lo detiene actúa según la cuerda, pasando por
el eje: no tiene momento respecto de él y el \omterm{def:b1:angular-momentum:L}{momento angular} $J\omega$ queda
intacto. Solo una fuerza tangencial (rozamiento en el eje) podría frenar
el giro.

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$, dieciocho veces el peso --- de ahí el tirón que siente el
dedo.

\textbf{16.} $J\dot\omega = -\Gamma$: $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$.

\textbf{17.} Toda la energía de giro $\tfrac12 J\omega^2 = \qty{0.47}{J}$ se
convierte en $Mgh'$: $h' = \qty{0.96}{m}$, casi la altura entera; menos en
la práctica por el rozamiento del eje durante el sueño y por las pérdidas
del tirón.

\textbf{18.} $a_G$ apunta hacia abajo (el yoyó frena al subir) mientras
$v_G$ apunta hacia arriba: $T = M(g - a_G) < Mg$ --- menor que el peso.

\textbf{19.} $r = R$: factor $1 + 1/2 = 1.5$, $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- caída rápida y poco giro: un eje fino es lo que hace a un yoyó.

\textbf{20.} \omterm{ex:b1:systems-of-points:rolling}{Rodadura sin deslizamiento}: el \omterm{def:b1:point-kinematics:frame}{punto material} en $C$ tiene
velocidad nula; el campo de velocidades de un \omterm{def:b1:systems-of-points:solid}{sólido rígido} con un punto
en reposo es una rotación en torno a ese punto (eje por $C$, el mismo $\omega$).

\textbf{21.} $F$ actúa en la parte baja del eje, a una altura $R - r$ sobre
la mesa y en horizontal hacia la mano: momento $F(R - r)$ respecto de $C$
en el sentido que hace rodar el carrete hacia la mano; el peso y la
reacción pasan por $C$ (sin momento).

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$, $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: hacia la mano.

\textbf{23.} La recta de acción de $F$ pasa por $C$ cuando el hilo que sale
de la parte baja del eje encuentra la mesa en $C$: $\cos\beta = r/R$,
$\beta = 82^\circ$. Más inclinado: el momento respecto de $C$ se invierte y
el carrete rueda alejándose (y, con $\beta$ bastante grande, se levanta).

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$; el rozamiento sale de $Ma_G = F - f$ (se opone a $F$):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$; ha de mantenerse por debajo
de $\mu_sMg
\approx \qty{0.2}{N}$ (con $\mu_s = 0.4$), es decir $F \lesssim \qty{0.5}{N}$.

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{Centro de masas}: la traslación de $G$ y la tensión; \omterm{def:b1:angular-momentum:L}{momento
angular} respecto de un eje: el giro y el sentido de rodadura; energía: la
rapidez abajo y la altura de la subida.
\end{solution}
