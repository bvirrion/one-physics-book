\section*{Capítulo \ref{ch:b1:operational-amplifier} --- El amplificador operacional}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$; $V_s = \qty{5.0}{V}$; satura en $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$; \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} de entrada $R_1 = \qty{10}{k\ohm}$; $i = 1.0/10^4 =
\qty{0.10}{mA}$; $V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
Sin él: divisor $1/(1 + 10) = 0.091$, \qty{91}{mV}. Con el seguidor:
\qty{1.0}{V} --- el seguidor no toma nada del sensor y suministra el
\qty{1}{mA} que necesita la carga.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$; en $V_-$: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
luego $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$; con $V_- = V_+$:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. Datos: $10 \times 0.05 = \qty{0.50}{V}$;
el \qty{1}{V} común no aporta nada --- rechazo del modo común.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: pendiente $-1000 \times 1.0 = \qty{-1000}{V/s}$;
satura al cabo de $15/1000 = \qty{15}{ms}$. \omterm{def:b1:wave-propagation:signal}{Señal} cuadrada: triángulo de
\omterm{def:b1:wave-propagation:signal}{amplitud} $ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(más la constante que deje la carga inicial).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: ganancia en continua $-100$;
corte $1/(2\pi R'C) = \qty{1.6}{Hz}$; a \qty{1}{kHz}, $R'C\omega = 628$, $G =
100/628 = 0.16$ --- el valor $1/RC\omega$ del integrador. Integra para
$f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: $R_2 = \qty{10}{k\ohm}$; $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
Con $\mu$ finita: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, luego
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. \omterm{thm:b1:sinusoidal-impedance:resonance}{Ancho de banda} $f_T/G =
\qty{10}{kHz}$; seguidor: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
Cada excursión del ruido a través de \qty{2.0}{V} hace bascular la salida:
una ráfaga de conmutaciones rápidas en torno al cruce. Remedio: una
histéresis más ancha que el ruido, es decir un \omterm{prop:b1:operational-amplifier:schmitt}{disparador de Schmitt}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: umbrales $\pm\qty{1.5}{V}$. La salida
vale $+\qty{15}{V}$ mientras $V_e$ sube hasta \qty{1.5}{V}, bascula a
$-\qty{15}{V}$, se queda ahí hasta que $V_e$ baja de \qty{-1.5}{V} y
vuelve a bascular: una \omterm{def:b1:wave-propagation:signal}{señal} cuadrada de la frecuencia del triángulo, con
sus flancos en los cruces de umbral.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
Con $V_s = +V_{\mathrm{sat}}$, $V_C$ se carga desde $-\beta V_{\mathrm{sat}}$
hacia $+V_{\mathrm{sat}}$: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, y llega a $+\beta V_{\mathrm{sat}}$ cuando $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$: semiperiodo $RC\ln\frac{1 + \beta}{1 - \beta}$,
de donde $T$. Con $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$. $V_s$: \omterm{def:b1:wave-propagation:signal}{señal} cuadrada de
$\pm\qty{15}{V}$; $V_C$: arcos exponenciales entre $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} Dos divisores: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone}); el cableado
fija el signo.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ siempre; $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ a fondo de escala. Los
\qty{2.5}{V} comunes a ambos son la tensión de modo común.

\textbf{4.} Cada punto intermedio: $1 \parallel 1 = \qty{500}{\ohm}$; entre
los dos puntos intermedios, \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$: una pérdida del $9\%$.

\textbf{6.} \omterm{def:b1:sinusoidal-impedance:impedance}{Impedancia} de entrada infinita (no toma corriente del puente,
así que los puntos intermedios conservan sus potenciales en circuito
abierto) e \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} de salida nula (la etapa siguiente puede pedir lo que necesite).

\textbf{7.} $V_A$ y $V_B$ exactamente; corriente nula.

\textbf{8.} Como en \cref{prop:b1:operational-amplifier:sumdiff}: $V_+ =
V_BR_2/(R_1 + R_2)$, \omterm{thm:b1:dc-circuits:kirchhoff}{ley de los nudos} en $V_-$, $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$: por ejemplo \qty{1}{k\ohm} y \qty{9}{k\ohm}.
Ganancia total $1000$; a fondo de escala $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} Equilibrado: los \qty{2.5}{V} comunes se cancelan exactamente
(solo sobrevive $V_B - V_A$). Un desajuste del $1\%$ deja pasar en torno
al $1\%$ de ellos: un error referido a la entrada del orden de
\qty{25}{mV}, diez veces la \omterm{def:b1:wave-propagation:signal}{señal} a fondo de escala --- los cuatro
\omterm{def:b1:dc-circuits:resistor}{resistores} han de estar apareados mejor que un $0.01\%$ (o hay que
eliminar antes la tensión de modo común).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, el cuarenta por
ciento del fondo de escala: inservible sin ajuste de la desviación, sin un
amplificador de desviación baja (tipo chopper) o sin un calibrado a
cero con la viga descargada.

\textbf{13.} Etapa 1: $f_T/100 = \qty{10}{kHz}$; etapa 2: \qty{100}{kHz};
muy por encima de \qty{5}{Hz}.

\textbf{14.} Se necesitan $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
dos mil veces por debajo de la \omterm{rem:b1:operational-amplifier:real}{velocidad de subida}: no.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$: $G = 1/\sqrt{26}$, \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$; retraso $\varphi/\omega =
\qty{15}{ms}$ --- nada frente al paso de un camión.

\textbf{18.} No carga a la etapa siguiente (\omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} de salida nula) y
ofrece ganancia; además no atenúa la \omterm{def:b1:wave-propagation:signal}{señal} útil por debajo del corte.

\textbf{19.} \qty{-28}{dB} (las ganancias se multiplican y los \omterm{def:b1:filters-transfer-functions:bode}{decibelios}
se suman), exacto porque la \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} de salida de la primera etapa es
nula: la segunda no la carga.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$; salida $+V_{\mathrm{sat}}$ cuando la \omterm{def:b1:wave-propagation:signal}{señal} supera
\qty{2.0}{V} y $-V_{\mathrm{sat}}$ por debajo (o al revés, según el cableado).

\textbf{21.} Castañetea: la lámpara parpadea cada vez que el ruido cruza
la referencia.

\textbf{22.} Con $V_{\mathrm{ref}}$ a través de $R_2$ y la salida a través
de $R_1$\dots{} en la forma estándar, la entrada no inversora está a
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$; tomando la referencia
desplazada en consecuencia, los niveles de conmutación son $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
Con $\beta V_{\mathrm{sat}} = \qty{0.10}{V}$: $\beta = 1/150$, por ejemplo $R_1 =
\qty{1.0}{k\ohm}$, $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (disparo) y \qty{1.9}{V} (rearme): deformaciones
\num{1.68e-3} y \num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. En
$-V_{\mathrm{sat}}$ el LED queda polarizado en inversa con \qty{15}{V},
más de lo que admite: lo protege un diodo corriente en antiparalelo (o en serie).

\textbf{25.} Puente (\qty{2.5}{mV} a fondo de escala) $\to$ dos seguidores
(sin carga) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{amplificador diferencia} $\times100$ $\to$ no inversor
$\times10$ $\to$ paso bajo activo de \qty{10}{Hz} $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{disparador de
Schmitt} en $\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ LED. Los límites reales:
la \omterm{rem:b1:operational-amplifier:real}{tensión de desviación de entrada} y el apareamiento de los \omterm{def:b1:dc-circuits:resistor}{resistores}
del \omterm{prop:b1:operational-amplifier:sumdiff}{amplificador diferencia}, ambos comparables a la \omterm{def:b1:wave-propagation:signal}{señal} de milivoltios.
\end{solution}
