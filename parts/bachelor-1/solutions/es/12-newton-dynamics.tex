\section*{Capítulo \ref{ch:b1:newton-dynamics} --- Las leyes de Newton de la dinámica}

\begin{solution}{exo:b1:newton-dynamics:1}
Lectura de la báscula $N = m(g + a)$, con $a$ la aceleración hacia arriba:
hacia arriba, $70 \times 11.81 = \qty{827}{N}$ (``\qty{84}{kg}''); hacia
abajo, $70 \times 7.81 =
\qty{547}{N}$ (``\qty{56}{kg}''); a velocidad constante, \qty{687}{N}
(\qty{70}{kg}); en caída libre, $0$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:2}
La misma aceleración $a$ para ambos (hilo inextensible): $m_1a = T$,
$m_2a = m_2g - T$, luego $a = m_2g/(m_1 + m_2) = \qty{3.3}{m/s^2}$ y
$T = m_1a = \qty{6.5}{N}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:3}
$a = g(\sin\alpha - \mu_d\cos\alpha) = 9.81(0.50 - 0.17) = \qty{3.2}{m/s^2}$;
$v = \sqrt{2 \times 3.2 \times 5.0} = \qty{5.7}{m/s}$. La \omterm{def:b1:newton-dynamics:momentum}{masa} no aparece.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:4}
$k\Delta\ell = mg$: $\Delta\ell = 0.50 \times 9.81/200 = \qty{2.5}{cm}$.
Con $x$ el desplazamiento respecto del equilibrio, $m\ddot x = -kx$ (el
peso y el alargamiento estático se cancelan): $\omega_0 = \sqrt{k/m} = \qty{20}{rad/s}$,
$T = \qty{0.31}{s}$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:5}
$\mu_s = \tan 22^\circ = 0.40$. Curva: $v \leq \sqrt{\mu_sgR}$: $\sqrt{0.8
\times 9.81 \times 50} = \qty{20}{m/s}$ (\qty{71}{km/h}); en mojado,
\qty{14}{m/s} (\qty{50}{km/h}).
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:6}
Paracaidista: $v_\infty = \sqrt{2 \times 785/(1.2 \times 0.70)} = \qty{43}{m/s}$;
$\mathrm{Re} = \rho vL/\eta \approx 1.2 \times 43 \times 0.5/1.8\times10^{-5}
\approx 10^6$: cuadrático. Gotita: $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-12}{kg}$,
$v_\infty = mg/6\pi\eta r = 4.1\times10^{-11}/3.4\times10^{-9} = \qty{1.2}{cm/s}$;
$\mathrm{Re} = 1.2 \times 0.012 \times 2\times10^{-5}/1.8\times10^{-5} = 0.02$:
lineal.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:7}
Abajo: $v^2 = 2g\ell(1 - \cos 60^\circ) = g\ell$; $T = mg + mv^2/\ell =
2mg$. Punto de suelta: $v = 0$, $T = mg\cos\theta_0 = 0.5mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:8}
$v = \sqrt{gR\tan\beta} = \sqrt{9.81 \times 200 \times 0.268} = \qty{23}{m/s}$
(\qty{82}{km/h}). Más despacio: el coche tiende a deslizar peralte abajo y
el rozamiento debe apuntar pendiente arriba; más deprisa: el rozamiento
apunta pendiente abajo para añadir \omterm{prop:b1:newton-dynamics:centripetal}{fuerza centrípeta}.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:9}
$T\cos\alpha = mg$, $T\sin\alpha = m\omega^2\ell\sin\alpha$: $\omega =
\sqrt{g/\ell\cos\alpha} = \sqrt{9.81/0.866} = \qty{3.4}{rad/s}$, periodo
\qty{1.9}{s}; $T = mg/\cos\alpha = 1.15mg$.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:10}
$a = 27.8/8.0 = \qty{3.5}{m/s^2}$, $F = \qty{3.5}{kN}$; $F \leq \mu_smg$
da $\mu_s \geq a/g = 0.35$. En calzada mojada, $\mu_s \approx 0.4$ limita
la aceleración a unos $0.4g$ por mucho motor que haya: adherencia, no potencia.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:11}
$v = v_\infty(1 - \eu^{-t/\tau})$, $v_\infty = g\tau = \qty{4.9}{m/s}$;
$z = \int v = v_\infty[t - \tau(1 - \eu^{-t/\tau})]$. A los \qty{2.0}{s}:
$4.9 \times (2.0 - 0.5 \times 0.98) = \qty{7.4}{m}$, frente a $\tfrac12 gt^2
= \qty{20}{m}$ en el vacío.
\end{solution}

\begin{solution}{exo:b1:newton-dynamics:12}
Proyección radial (hacia el centro): $N - mg\cos\theta = mv^2/R$, luego
$N = m(v^2/R + g\cos\theta)$. Arriba ($\theta = \pi$): $N = m(v^2/R - g)
\geq 0$ si y solo si $v \geq \sqrt{gR}$. Con la ley de rapidez, $v_{\text{arr}}^2 =
v_0^2 - 4gR \geq gR$: $v_0 \geq \sqrt{5gR}$; y entonces abajo $N =
m(5g + g) = 6mg$.
\end{solution}

\begin{solution}{pb:b1:newton-dynamics:1}
\textbf{1.} $m\dot v = mg$: $v = gt$, $z = \tfrac12 gt^2$; a los \qty{10}{s}:
\qty{98}{m/s}, \qty{490}{m}.

\textbf{2.} $mg = \qty{785}{N}$.

\textbf{3.} En el suelo, el piso empuja hacia arriba con $N = mg$ --- ese
empuje es lo que se siente como peso. En caída libre no hay \omterm{def:b1:newton-dynamics:momentum}{fuerza} de
contacto: la gravedad actúa igual sobre todas las partes y nada aprieta.

\textbf{4.} Sí, con un error de una fracción de por ciento en un minuto;
se desprecian la rotación terrestre (términos centrífugo y de Coriolis,
\cref{ch:b1:non-inertial-frames}) y el viento.

\textbf{5.} $F_d = \tfrac12\rho C_xSv^2 = 0.48\,v^2$ (SI).

\textbf{6.} $m\,\dd v/\dd t = mg - 0.48v^2$.

\textbf{7.} $v_\infty = \sqrt{mg/0.48} = \sqrt{1635} = \qty{40}{m/s} =
\qty{146}{km/h}$.

\textbf{8.} Divídase por $mg$: $\dd u/\dd t\,(v_\infty/g) = 1 - u^2$;
$\tau = 40.4/9.81 = \qty{4.1}{s}$.

\textbf{9.} $\int_0^u\dd u'/(1 - u'^2) = \operatorname{artanh}u = t/\tau$,
luego $u = \tanh(t/\tau)$.

\textbf{10.} $t = \tau\operatorname{artanh}(0.9) = 1.47\tau = \qty{6.1}{s}$;
para $0.99$: $2.65\tau = \qty{11}{s}$.

\textbf{11.} $z = \int v_\infty\tanh(t/\tau)\dd t = v_\infty\tau\ln\cosh(t/\tau)$;
a los \qty{10}{s}: $166 \times \ln\cosh(2.43) = 166 \times 1.74 = \qty{290}{m}$,
frente a \qty{490}{m}.

\textbf{12.} $\mathrm{Re} = 1.2 \times 40 \times 0.5/1.8\times10^{-5} \approx
\num{1.3e6}$: muy por encima de $1000$, cuadrático.

\textbf{13.} $v_\infty' = \sqrt{785/(0.6 \times 25)} = \sqrt{52} = \qty{7.2}{m/s}$.

\textbf{14.} Arrastre $= 15 \times 1600 = \qty{24}{kN}$; $a = (24000 - 785)/80
= \qty{290}{m/s^2} \approx 30g$: se rompería el arnés, o el saltador.

\textbf{15.} Media $a \approx (40 - 7)/3 = \qty{11}{m/s^2} \approx 1.1g$;
\omterm{def:b1:newton-dynamics:momentum}{fuerza} del arnés $\approx m(g + a) \approx \qty{1.7}{kN}$.

\textbf{16.} $\tau' = 7.2/9.81 = \qty{0.73}{s}$: a los pocos segundos de
la apertura completa el descenso es estacionario.

\textbf{17.} $\vect a = \vect 0$: \omterm{def:b1:newton-dynamics:momentum}{fuerza} del arnés $= mg = \qty{785}{N}$.

\textbf{18.} El equilibrio $mg = \tfrac12\rho C_xSv^2$ da $v \propto (C_xS)^{-1/2}$:
con la mitad de área, $\sqrt2$ veces la rapidez (\qty{10}{m/s}).

\textbf{19.} $v \propto \sqrt m$: $7.2 \times \sqrt{100/80} = \qty{8.1}{m/s}$.

\textbf{20.} $h = v^2/2g = 52/19.6 = \qty{2.6}{m}$: un salto desde un
primer piso.

\textbf{21.} $a = v^2/2d$: \qty{52}{m/s^2} ($5.3g$) en \qty{0.5}{m},
\omterm{def:b1:newton-dynamics:momentum}{fuerza} $m(g + a) \approx \qty{5}{kN}$; \qty{17}{m/s^2} ($1.8g$) en
\qty{1.5}{m}, \qty{2.2}{kN}. Mejor la voltereta.

\textbf{22.} Un arrastre mayor (y algo de sustentación) supera
momentáneamente al peso: el saltador decelera y toma tierra por debajo de $v_\infty'$.

\textbf{23.} $m = \tfrac43\pi r^3\rho_w = \qty{4.2e-6}{kg}$, $S = \pi r^2 =
\qty{3.1e-6}{m^2}$: $v_\infty = \sqrt{2mg/\rho C_xS} = \sqrt{8.2\times10^{-5}/
1.9\times10^{-6}} = \qty{6.6}{m/s}$; $\mathrm{Re} = 1.2 \times 6.6 \times
2\times10^{-3}/1.8\times10^{-5} \approx 900$: cuadrático (a grandes rasgos).

\textbf{24.} $v_\infty = \qty{1.2}{cm/s}$ (\cref{exo:b1:newton-dynamics:6});
$\mathrm{Re} \approx 0.02 \ll 1$: Stokes se aplica, de forma coherente.

\textbf{25.} $m\dot{\vect v} = m\vect g + \vect F_{\text{arr}}$; el
arrastre es lineal en $v$ para lo pequeño y lento, y cuadrático para lo
grande y rápido. Los números: \qty{40}{m/s} para un cuerpo en caída,
\qty{7}{m/s} bajo una vela (o para una gota de lluvia) y \qty{1}{cm/s}
para la niebla --- la misma ley a lo largo de cuatro órdenes de magnitud.
\end{solution}
