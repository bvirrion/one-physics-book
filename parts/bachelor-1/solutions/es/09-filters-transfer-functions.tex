\section*{Capítulo \ref{ch:b1:filters-transfer-functions} --- Filtros y funciones de transferencia}

\begin{solution}{exo:b1:filters-transfer-functions:1}
$0.5 \to \qty{-6.0}{dB}$; $10 \to \qty{20}{dB}$; $1/\sqrt2 \to \qty{-3.0}{dB}$;
$100 \to \qty{40}{dB}$. $\qty{-40}{dB} \to 0.01$; $\qty{+6}{dB} \to 2.0$.
Cascada: $-6 - 20 = \qty{-26}{dB}$, una razón de $0.05$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:2}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 10^4 \times 1.6\times10^{-8}) =
\qty{1.0}{kHz}$. \omterm{def:b1:filters-transfer-functions:H}{Ganancias} $-10\log(1 + (f/f_c)^2)$: \qty{100}{Hz}:
\qty{-0.04}{dB}; \qty{1}{kHz}: \qty{-3}{dB}; \qty{10}{kHz}: \qty{-20}{dB};
\qty{100}{kHz}: \qty{-40}{dB}.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:3}
(a) paso bajo ($C$ cortocircuita la salida a $f$ alta); (b) paso alto ($C$
se abre a $f$ baja); (c) paso bajo de segundo orden ($L$ se abre a $f$
alta y $C$ cortocircuita); (d) paso banda ($C$ bloquea $f$ baja y $L$
bloquea $f$ alta).
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:4}
$4E/n\pi$: \qty{6.37}{V} a \qty{1}{kHz}, \qty{2.12}{V} a \qty{3}{kHz} y
\qty{1.27}{V} a \qty{5}{kHz}. La \omterm{def:b1:wave-propagation:signal}{señal} cuadrada es antisimétrica al
desplazarla medio periodo ($s(t + T/2) = -s(t)$), cosa que los armónicos
pares no son.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:5}
A \qty{100}{Hz} y en sus armónicos útiles, $\omega \ll \omega_c$:
$\underline H \approx j\omega/\omega_c$, una derivada dividida por
$\omega_c$. La pendiente del triángulo vale $\pm 4Ef = \pm\qty{400}{V/s}$,
así que la salida es una \omterm{def:b1:wave-propagation:signal}{señal} cuadrada de \omterm{def:b1:wave-propagation:signal}{amplitud} $400/(2\pi \times 10^4) =
\qty{6.4}{mV}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:6}
$\omega \gg \omega_c$: $u_s \approx (1/\tau)\int u_e\,\dd t$, con $\tau =
1/(2\pi \times 10) = \qty{16}{ms}$. La integral de $\pm E$ es un triángulo
que sube $ET/2$ por medio periodo: excursión $ET/2\tau$, \omterm{def:b1:wave-propagation:signal}{amplitud}
$ET/4\tau = 1.0 \times 10^{-3}/(4 \times 0.016) = \qty{16}{mV}$ en torno a
cero.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:7}
$f_0 = \qty{5.03}{kHz}$. $Q = \sqrt{L/C}/R = 1/\sqrt2$: $R = \sqrt2 \times
316 = \qty{447}{\ohm}$. En $f_0$: \qty{-3}{dB}; en $10f_0$: $1/\sqrt{1 +
10^4} \to \qty{-40}{dB}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:8}
$G = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ con $x = f/3$: \qty{1}{kHz}: $x = 1/3$,
$Q(x - 1/x) = -53$, $G = 0.019$; \qty{3}{kHz}: $G = 1$; \qty{5}{kHz}:
$x = 5/3$, $21$, $G = 0.047$. Salida: \qty{2.1}{V} a \qty{3}{kHz}, con
\qty{0.12}{V} a \qty{1}{kHz} y \qty{0.06}{V} a \qty{5}{kHz} --- una
sinusoide de \qty{3}{kHz} casi pura.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:9}
Nudos: con $a = jRC\omega$, el primer condensador ve $R$ aguas arriba y
$R + 1/jC\omega$ aguas abajo: $\underline H = 1/(1 + 3a + a^2)$, es decir
$1/[1 - (\omega/\omega_c)^2 + 3j\omega/\omega_c]$. En $\omega_c$: $1/3j$,
$G = 1/3$, \qty{-9.5}{dB}; el producto de las dos \omterm{def:b1:filters-transfer-functions:H}{ganancias} por separado
daría $1/2$, \qty{-6}{dB}. $f_c = \qty{1.0}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:10}
Por debajo de $\omega_1$: \qty{0}{dB}; entre $\omega_1$ y $\omega_2$ sube
\qty{20}{dB} por década; por encima de $\omega_2$, plana en
$20\log(\omega_2/\omega_1) = \qty{40}{dB}$. Fase
$\arctan(\omega/\omega_1) - \arctan(\omega/\omega_2)$: en $\omega = 10^3$,
$\arctan 10 - \arctan 0.1 = 84^\circ - 6^\circ = 78^\circ$.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:11}
Muy por encima del corte, $G \approx f_c/f$; la razón de rizado tras el
filtrado vale
\[
\frac{(4E/3\pi)(f_c/100)}{2E/\pi} = \frac{2}{3}\,\frac{f_c}{100} < 0.01,
\]
luego $f_c < \qty{1.5}{Hz}$ y $\tau = 1/2\pi f_c > \qty{0.1}{s}$. La propia carga hace de
$R$: un condensador en paralelo con ella no cuesta nada ni en caída de
tensión ni en potencia, mientras que un \omterm{def:b1:dc-circuits:resistor}{resistor} en serie desperdiciaría ambas.
\end{solution}

\begin{solution}{exo:b1:filters-transfer-functions:12}
$f_c = 1/(2\pi RC) = 1/(2\pi \times 1.6\times10^5 \times 10^{-7}) =
\qty{10}{Hz}$, $\tau = RC = \qty{16}{ms}$. Durante un techo plano el
condensador se carga a través de $R$: la salida decae como $\eu^{-t/\tau}$
desde cada flanco. Hundimiento en $T/2$: $1 - \eu^{-T/2\tau}$: a \qty{50}{Hz}
($T/2 = \qty{10}{ms}$), $1 - \eu^{-0.63} = 47\%$; a \qty{1}{kHz}
($T/2 = \qty{0.5}{ms}$), $3\%$.
\end{solution}

\begin{solution}{pb:b1:filters-transfer-functions:1}
\textbf{1.} $\underline H = \underline U_{\text{altavoz}}/\underline U_e$;
$G_{\mathrm{dB}} = 20\log\abs{\underline H}$.

\textbf{2.} $\underline H_L = R/(R + jL\omega) = 1/(1 + j\omega/\omega_c)$,
$\omega_c = R/L$.

\textbf{3.} $L = R/\omega_c = 8.0/(2\pi \times 2000) = \qty{0.64}{mH}$.

\textbf{4.} $\underline H_H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$,
$\omega_c = 1/RC$; $C = 1/(R\omega_c) = \qty{9.9}{\micro F} \approx
\qty{10}{\micro F}$.

\textbf{5.} En $f_c$: \qty{-3}{dB} cada uno. \qty{200}{Hz} ($x = 0.1$):
graves \qty{-0.04}{dB}, agudos \qty{-20}{dB}. \qty{20}{kHz} ($x = 10$):
graves \qty{-20}{dB}, agudos \qty{-0.04}{dB}.

\textbf{6.} $G_L^2 + G_H^2 = (1 + x^2)/(1 + x^2) = 1$: con cargas iguales,
las dos potencias suman la que tomaría un único altavoz de \qty{8}{\ohm}
--- repartida, no perdida.

\textbf{7.} $-45^\circ$ (graves) y $+45^\circ$ (agudos): $90^\circ$ de
diferencia; en $f_c$ los dos conos no se mueven juntos.

\textbf{8.} \qty{\pm 6}{dB} por octava (\qty{20}{dB} por década).

\textbf{9.} Divisor entre $jL\omega$ y $R \parallel 1/jC\omega =
R/(1 + jRC\omega)$: $\underline H = R/[R + jL\omega(1 + jRC\omega)] =
1/(1 - LC\omega^2 + jL\omega/R)$.

\textbf{10.} $\omega_0 = 1/\sqrt{LC}$; $L\omega/R = x/Q$ da $Q =
R/L\omega_0 = R\sqrt{C/L}$.

\textbf{11.} $\abs{\underline H}^{-2} = (1 - x^2)^2 + x^2/Q^2 = 1 + x^4 +
x^2(1/Q^2 - 2) = 1 + x^4$ para $Q^2 = \tfrac12$; en $x = 1$, $G = 1/\sqrt2$.

\textbf{12.} $L = R/(Q\omega_0) = 8\sqrt2/(1.257\times10^4) = \qty{0.90}{mH}$;
$C = 1/(L\omega_0^2) = \qty{7.0}{\micro F}$.

\textbf{13.} $x = 2$: $1/\sqrt{17}$, \qty{-12.3}{dB}; $x = 4$: $1/\sqrt{257}$,
\qty{-24.1}{dB}: \qty{12}{dB} por octava.

\textbf{14.} Intercambiar $L$ y $C$ intercambia $x \leftrightarrow 1/x$ (y
$Q$ no cambia): $\underline H_H = -x^2/(1 - x^2 + jx/Q)$.

\textbf{15.} En $x = 1$: $\underline H_L = Q/j = -jQ$ ($-90^\circ$) y
$\underline H_H = jQ$ ($+90^\circ$): los altavoces están en oposición de
fase en el corte y se cancelarían acústicamente; el de agudos se conecta
con la polaridad invertida.

\textbf{16.} Los armónicos por encima de \qty{2}{kHz}: de $n \geq 10$
(\qty{2.2}{kHz} en adelante) van sobre todo al altavoz de agudos.

\textbf{17.} $9 \times 220 = \qty{1980}{Hz} \approx f_c$: se reparte por
igual, \qty{-3}{dB} a cada uno.

\textbf{18.} Graves ($G_L$): $x = 0.5$: $0.89$; $x = 1.5$: $0.55$; $x = 2.5$:
$0.37$. Agudos ($G_H$): $0.45$, $0.83$, $0.93$.

\textbf{19.} El de agudos: su condensador en serie bloquea la continua. El
de graves se lleva la componente continua (\qty{1}{V} sobre
\qty{8}{\ohm}: \qty{125}{mA}, \qty{0.125}{W} de calor y un cono desplazado).

\textbf{20.} Primer orden, $x = 0.25$: $G_H = 0.25/\sqrt{1.0625} = 0.24$
(\qty{-12}{dB}), fracción de potencia $6\%$. Segundo orden: $0.0625/\sqrt{1 +
0.0039} = 0.06$ (\qty{-24}{dB}), $0.4\%$. Un altavoz de agudos de unos
pocos vatios recibe el $6\%$ de una nota grave fuerte con el diseño de
primer orden --- puede quemarse.

\textbf{21.} $\underline Z = 8 + jL_{vc}\omega$: a \qty{2}{kHz}: $8 + 6.3j$,
$\abs{\underline Z} = \qty{10.2}{\ohm}$; a \qty{8}{kHz}: $8 + 25.1j$,
\qty{26.3}{\ohm}.

\textbf{22.} $\underline H = \underline Z/(\underline Z + jL\omega)$: a
\qty{2}{kHz}: $10.2/\abs{8 + 14.3j} = 10.2/16.4 = 0.62$ (\qty{-4.2}{dB});
a \qty{8}{kHz}: $26.3/\abs{8 + 57.3j} = 0.45$ (\qty{-6.9}{dB}) en vez de
\qty{-12.3}{dB}: la \omterm{def:b1:sinusoidal-impedance:impedance}{impedancia} creciente de la bobina móvil se opone a la
bobina en serie y la caída se aplana a unos pocos dB por octava.

\textbf{23.} \omterm{def:b1:sinusoidal-impedance:impedance}{Admitancias}: $1/(R + jL_{vc}\omega) + jC_z\omega/(1 + jRC_z\omega)$;
con $C_z = L_{vc}/R^2$ el segundo término vale $jL_{vc}\omega/[R(R +
jL_{vc}\omega)]$, y la suma es $(R + jL_{vc}\omega)/[R(R + jL_{vc}\omega)]
= 1/R$. $C_z = 0.50\times10^{-3}/64 = \qty{7.8}{\micro F}$.

\textbf{24.} $U = \sqrt{PR} = \sqrt{400} = \qty{20}{V}$ eficaces. En $f_c$
cada altavoz recibe la mitad: \qty{25}{W}. A \qty{500}{Hz} el de agudos
recibe el $6\%$: \qty{3}{W}.

\textbf{25.} Primer orden: dos componentes, complementario en potencia,
pendientes suaves de \qty{6}{dB/oct}, $90^\circ$ entre altavoces y graves
que se cuelan en el de agudos. Segundo orden: cuatro componentes,
\qty{12}{dB/oct}, $180^\circ$ en el corte (hay que invertir el de agudos)
y una protección mucho mejor del altavoz de agudos. En ambos casos la
inductancia del altavoz de graves debe domarse con una red de Zobel para
que el diseño se sostenga.
\end{solution}
