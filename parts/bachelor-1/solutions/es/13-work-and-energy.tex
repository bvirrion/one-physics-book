\section*{Capítulo \ref{ch:b1:work-and-energy} --- Trabajo, energía y energía potencial}

\begin{solution}{exo:b1:work-and-energy:1}
Fuerza que eleva: $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$; peso,
$-\qty{589}{J}$; potencia $589/5.0 = \qty{118}{W}$. Rozamiento: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$; $d = E_k/F =
\qty{55}{m}$; duplicar $v$ cuadruplica $E_k$ y $d$: \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$; $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$: $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
Tensión: $T - mg = mv^2/\ell = mg$, $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
El rozamiento trabaja $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$, luego
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$; se pierde el $35\%$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
$E_m = \tfrac12 mv^2 - GMm/R \geq 0$ para que el cuerpo alcance el infinito
(donde $E_p = 0$ y $v \to 0$): $v_e = \sqrt{2GM/R}$, es decir
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
La misma densidad da $M \propto R^3$: $v_e \propto R$, luego \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
Punto más bajo: $mg(\ell_0 + x) = \tfrac12 kx^2$, $25x^2 - 687x - 13734 = 0$,
$x = \qty{41}{m}$; caída total \qty{61}{m}; $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (neto \qty{2}{g} hacia arriba). La rapidez es máxima donde se
anula la fuerza neta, $kx = mg$: $x = \qty{14}{m}$, \qty{34}{m} por debajo
del punto de salto.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
Gravedad: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$;
arrastre: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$; en total \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$: $x = 0, \pm a$. $E_p'' = E_0(12x^2/a^4 -
4/a^2)$: $-4E_0/a^2$ en $0$ (máximo inestable, $E_p = 0$) y $+8E_0/a^2$
en $\pm a$ (mínimos estables, $E_p = -E_0$): pozos de profundidad $E_0$;
$\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$: una elipse de semiejes $x_{\max}
= \sqrt{2E/k}$ y $\dot x_{\max} = \sqrt{2E/m}$. El movimiento es armónico,
de periodo $2\pi\sqrt{m/k}$ sea cual sea la \omterm{def:b1:wave-propagation:signal}{amplitud} (isocronismo), así
que todas las elipses se recorren en el mismo tiempo.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
Energía: $v^2 = 2gR(1 - \cos\theta)$. Radial: $mg\cos\theta - N = mv^2/R$,
luego $N = mg(3\cos\theta - 2)$, que se anula en $\cos\theta = 2/3$ ($\theta =
48^\circ$): altura $2R/3$, rapidez $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$: $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
Calor en el contacto neumático--calzada (y una marca de derrape). Las
ruedas que ruedan usan rozamiento estático, $\mu_s > \mu_d$: mayor fuerza
de frenado, menor distancia y dirección conservada.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} $r \to 0$: $+\infty$ (el término en $r^{-12}$); $r \to \infty$:
$0^-$. Una pared abrupta, un pozo y una cola larga.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ si y
solo si $(\sigma/r)^6 = 1$: $r = \sigma$, $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$: $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$, mínimo en $u = 1$ ($r = \sigma$) con valor $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$:
positiva (repulsiva) para $r < \sigma$ y negativa (atractiva) más allá.

\textbf{5.} $(1/1.2)^6 = 0.335$, $(1/1.2)^{12} = 0.112$: $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. $(1/1.2)^{13} = 0.0935$,
$(1/1.2)^7 = 0.279$: $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$,
cerca de los \qty{431}{kJ/mol} medidos.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$; en
$\sigma$: $72\epsilon/\sigma^2 > 0$: estable.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$:
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$; $f = \qty{2.2e14}{Hz}$; $\lambda = c/f = \qty{1.4}{\micro m}$;
$1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} Un factor $2.4$ de más: las paredes del modelo son demasiado
abruptas (el núcleo en $r^{-12}$ es un sustituto tosco de un enlace
covalente), pero el \omterm{def:b1:units-dimensions:oom}{orden de magnitud} --- una vibración infrarroja a
$10^{14}$ Hz --- es el correcto, y el método (la curvatura del pozo) es
exactamente el que se usa con potenciales mejores.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$: $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$: el enlace apenas
tiembla.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$: curvas cerradas en torno a $(\sigma, 0)$
--- vibración, ligada; $E_m > 0$: curvas abiertas --- los átomos se
separan; $E_m = 0$: la separatriz, un átomo que apenas alcanza el infinito.

\textbf{15.} $u^2 - 2u = -\tfrac12$ con $u = (\sigma/r)^6$: $u = 1 \pm 1/\sqrt2
= 1.71$ o $0.293$; $r = \sigma u^{-1/6}$: $0.915\sigma$ y $1.23\sigma$.

\textbf{16.} Punto medio $1.07\sigma > \sigma$: la pared exterior es más
blanda que la interior, así que el átomo dispone de más espacio por fuera;
la distancia promediada en el tiempo crece con la energía --- los enlaces
calientes se alargan y los sólidos se dilatan.

\textbf{17.} Mayor: la fuerza recuperadora en el lado exterior, más
blando, es menor que la que supone la aproximación armónica, así que el
regreso tarda más.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$ de \omterm{thm:b1:work-and-energy:ke}{energía
cinética} en $\sigma$: $v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}
= \qty{2.0e4}{m/s}$.

\textbf{19.} Conservación de la energía: si $E_m > 0$ sigue siendo $> 0$,
la curva es abierta y los átomos se separan tras un solo encuentro. Un
tercer cuerpo (otra molécula, una pared, un fotón) ha de llevarse el exceso.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$, positivo: la atracción tira del
átomo hacia dentro, es motora.

\textbf{21.} $W_{\text{ext}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$, $(1/0.9)^7 = 2.09$: $F = +22\epsilon/\sigma
= \qty{+120}{nN}$, diez veces la atracción en $1.2\sigma$: comprimir un
enlace un $10\%$ cuesta mucho más que estirarlo un $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. Los gases reales se disocian muy por debajo porque las
energías están repartidas: una fracción de las moléculas lleva muchas
veces $k_BT$ (\cref{ch:b1:kinetic-theory}).

\textbf{24.} Sobreviven: el equilibrio en el mínimo, $k = E_p''$ en el
mínimo, la energía de disociación $=$ la profundidad, los \omterm{def:b1:work-and-energy:turning}{puntos de
retroceso} a partir de $E_p = E_m$ y la asimetría $\Rightarrow$ dilatación.
Cambian: los valores de $k$, de la frecuencia y de los \omterm{def:b1:work-and-energy:turning}{puntos de retroceso}.

\textbf{25.} Equilibrio: $E_p' = 0$; vibración: $\omega_0^2 = E_p''/m$;
escape: $E_m$ comparada con el borde del pozo.
\end{solution}
