\section*{Capítulo \ref{ch:b1:sinusoidal-impedance} --- Régimen sinusoidal permanente e impedancia}

\begin{solution}{exo:b1:sinusoidal-impedance:1}
$\underline U = 5\eu^{j\pi/3} = 2.5 + 4.33j$ (V). $3\cos\omega t +
4\sin\omega t$: $3 + 4\eu^{-j\pi/2} = 3 - 4j$, de módulo $5$ y argumento
$-\arctan(4/3) = \qty{-0.927}{rad}$: $5\cos(\omega t - 0.927)$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:2}
$R$: \qty{100}{\ohm} en ambos casos. $L\omega$: \qty{31.4}{\ohm} (\qty{50}{Hz}),
\qty{628}{\ohm} (\qty{1}{kHz}). $1/C\omega$: \qty{318}{\ohm}, \qty{15.9}{\ohm}.
$RL$ serie a \qty{50}{Hz}: $\abs{\underline Z} = \sqrt{100^2 + 31.4^2} =
\qty{105}{\ohm}$, fase $\arctan(0.314) = 17^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:3}
$U = 230\sqrt2 = \qty{325}{V}$. $I_{\mathrm{rms}} = 2000/230 = \qty{8.7}{A}$,
pico \qty{12.3}{A}. $\langle Ri^2\rangle = R\langle i^2\rangle =
RI_{\mathrm{rms}}^2$ por definición de \omterm{def:b1:sinusoidal-impedance:rms}{valor eficaz}.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:4}
$f_0 = 1/(2\pi\sqrt{LC}) = \qty{5.03}{kHz}$; $Q = \sqrt{L/C}/R = 3.16$;
$\Delta f = f_0/Q = \qty{1.6}{kHz}$; $I_{\max} = E/R = \qty{10}{mA}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:5}
$1/\sqrt2$ en $\omega = 1/RC$: $f_c = 1/(2\pi RC) = \qty{1.0}{kHz}$. A
\qty{1.0}{kHz}: razón $0.71$, fase $-45^\circ$. A \qty{10}{kHz}:
$RC\omega = 10$, razón $1/\sqrt{101} = 0.10$, fase $-84^\circ$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:6}
$\underline Z = 100 + j(250 - 100) = 100 + 150j$: $Z = \qty{180}{\ohm}$,
$\arg = 56^\circ$. $I = 1.0/180 = \qty{5.6}{mA}$, retrasada $56^\circ$
respecto de la fuente. $U_R = \qty{0.56}{V}$, $U_L = \qty{1.39}{V}$, $U_C = \qty{0.56}{V}$;
\omterm{def:b1:sinusoidal-impedance:complex}{fasores}: $U_R$ a lo largo de $I$, $U_L$ hacia arriba, $U_C$ hacia abajo,
resultante \qty{1.0}{V}. $L\omega > 1/C\omega$: por encima de la resonancia (inductivo).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:7}
$f_0 = 1/(2\pi\sqrt{LC})$: de \qty{1.59}{MHz} (\qty{50}{pF}) a \qty{0.50}{MHz}
(\qty{500}{pF}). A \qty{1.00}{MHz}: $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$.
$\Delta f = \qty{10}{kHz}$. Emisora a \qty{1.009}{MHz}: $x = 1.009$,
$Q(x - 1/x) = 1.8$, razón $1/\sqrt{1 + 3.2} = 0.49$. Tensión del
condensador $QE = \qty{1.0}{mV}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:8}
$Q(x - 1/x) = \pm1 \Leftrightarrow x^2 \mp x/Q - 1 = 0$; raíces positivas
$x_{1,2} = \mp 1/2Q + \sqrt{1 + 1/4Q^2}$. La diferencia vale $1/Q$, luego
$\Delta\omega = \omega_0/Q$; el producto $(1 + 1/4Q^2) - 1/4Q^2 = 1$, luego
$\omega_1\omega_2 = \omega_0^2$ (la banda es simétrica en escala logarítmica).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:9}
$I = P/(U\cos\varphi) = 2000/(230 \times 0.70) = \qty{12.4}{A}$. \omterm{thm:b1:sinusoidal-impedance:power}{Potencia
reactiva} $P\tan\varphi = 2000 \times 1.02 = \qty{2.0}{kvar}$. $C = 2040/(314
\times 230^2) = \qty{123}{\micro F}$. Nueva corriente $2000/230 = \qty{8.7}{A}$.
Pérdidas $0.50 \times 12.4^2 = \qty{77}{W}$ antes y $0.50 \times 8.7^2 =
\qty{38}{W}$ después.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:10}
$\underline Y = 1/R + j(C\omega - 1/L\omega)$; $\underline U = \underline I/
\underline Y$ es máxima cuando se anula la susceptancia, en $\omega_0$,
donde $U = RI$. Potencia mitad: $\abs{C\omega - 1/L\omega} = 1/R$, es decir
$RC\omega_0(x - 1/x) = \pm1$: $\Delta\omega = \omega_0/(RC\omega_0) = 1/RC$
y $Q = RC\omega_0 = R\sqrt{C/L}$. Datos: $\omega_0 = \qty{1.0e6}{rad/s}$
($f_0 = \qty{159}{kHz}$), $Q = 10^4\sqrt{10^{-6}} = 10$, $\Delta f =
\qty{16}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:11}
$U_C = I/C\omega = E/[C\omega\sqrt{R^2 + (L\omega - 1/C\omega)^2}]$;
multiplíquese por $C\omega$: $\sqrt{(RC\omega)^2 + (LC\omega^2 - 1)^2} =
\sqrt{x^2/Q^2 + (1 - x^2)^2}$. Con $s = x^2$: $(1 - s)^2 + s/Q^2$ es
mínimo en $s = 1 - 1/2Q^2$ (si es positivo), con mínimo $1/Q^2 - 1/4Q^4$:
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$. Para $Q = 3.16$: $x_r = 0.975$,
$U_{C,\max} = 3.2E$. Para $Q < 1/\sqrt2$ el mínimo está en $s = 0$: $U_C$
decrece de forma monótona desde $E$, sin máximo.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:12}
En la resonancia $i = I\cos\omega_0 t$ y $u_C = (I/C\omega_0)\sin\omega_0 t$,
de modo que, usando $1/C\omega_0^2 = L$,
\[
\tfrac12 Li^2 + \tfrac12 Cu_C^2 = \tfrac12 LI^2\cos^2\omega_0 t
+ \tfrac12 LI^2\sin^2\omega_0 t = \tfrac12 LI^2 .
\]
Disipada por periodo: $\tfrac12 RI^2T$. El cociente $\times 2\pi$: $2\pi L/(RT)
= L\omega_0/R = Q$.
\end{solution}

\begin{solution}{pb:b1:sinusoidal-impedance:1}
\textbf{1.} $\underline Z = R + j(L\omega - 1/C\omega)$.

\textbf{2.} $\underline I = E/\underline Z$; $I = E/\sqrt{R^2 + (L\omega -
1/C\omega)^2}$; $\varphi_i = -\arctan[(L\omega - 1/C\omega)/R]$.

\textbf{3.} La \omterm{def:b1:sinusoidal-impedance:impedance}{reactancia} se anula en $\omega_0 = 1/\sqrt{LC}$: $I_{\max}
= E/R$, $\varphi_i = 0$.

\textbf{4.} $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$; $L\omega_0 = 2\times10^{-4}
\times 6.28\times10^6 = \qty{1257}{\ohm}$.

\textbf{5.} $Q = 1257/12.6 = 100$.

\textbf{6.} $L\omega - 1/C\omega = L\omega_0(x - 1/x)$, divídase por $R$.

\textbf{7.} $x < 1$: \omterm{def:b1:sinusoidal-impedance:impedance}{reactancia} negativa, $\varphi_i > 0$, la corriente se
adelanta (capacitivo); $x > 1$: se retrasa (inductivo).

\textbf{8.} $Q(x - 1/x) = \pm1$.

\textbf{9.} $x^2 \mp x/Q - 1 = 0$, $x_{1,2} = \mp1/2Q + \sqrt{1 + 1/4Q^2}$,
$x_2 - x_1 = 1/Q$: $\Delta\omega = \omega_0/Q$.

\textbf{10.} $\Delta f = \qty{10}{kHz}$.

\textbf{11.} \qty{1.010}{MHz}: $Q(x - 1/x) = 100 \times 0.0199 = 2.0$,
razón $1/\sqrt{5.0} = 0.45$, potencia $0.20$: \qty{-7}{dB}. \qty{1.020}{MHz}:
$3.96$, razón $0.25$, potencia $0.06$: \qty{-12}{dB}.

\textbf{12.} Justo en el límite: la vecina queda solo \qty{7}{dB} por
debajo. Un $Q$ mayor (una $R$ menor), o varias etapas sintonizadas en
cascada, afinan la selección.

\textbf{13.} De \qty{0.50}{MHz} a \qty{1.59}{MHz}: la banda de AM.

\textbf{14.} $\underline U_C = \underline I/jC\omega$; en $\omega_0$,
$U_C = I_{\max}/C\omega_0 = E/(RC\omega_0) = QE$.

\textbf{15.} $I_{\max} = 10^{-5}/12.6 = \qty{0.79}{\micro A}$; $U_C = QE =
\qty{1.0}{mV}$: una ganancia de tensión de cien veces sin amplificador.

\textbf{16.} $\underline U_L = jL\omega_0\underline I = jQE$, $\underline U_C
= -jQE$: iguales y opuestas, se cancelan; la fuente ve solo $R$.

\textbf{17.} Almacenada $\tfrac12 LI_{\max}^2 = \qty{6.2e-17}{J}$; disipada
por periodo $\tfrac12 RI_{\max}^2T = \qty{3.9e-18}{J}$; $2\pi \times 6.2/0.39
\approx 100 = Q$.

\textbf{18.} $\tau = 2Q/\omega_0 = \qty{32}{\micro s}$. El circuito no
puede seguir cambios más rápidos que $\sim\tau$: un \omterm{thm:b1:sinusoidal-impedance:resonance}{ancho de banda}
$\Delta f$ solo deja pasar modulaciones más lentas que $\Delta f$; con
$Q = 1000$, $\Delta f =
\qty{1}{kHz}$ y se perderían los agudos del programa.

\textbf{19.} $Q$ se reduce a la mitad, $50$; $\Delta f$ se duplica hasta
\qty{20}{kHz}; $U_C$ baja a la mitad, \qty{0.5}{mV}.

\textbf{20.} $R_{\text{tot}} = \qty{62.6}{\ohm}$: $Q = 1257/62.6 = 20$,
$\Delta f = \qty{50}{kHz}$, $U_C = 20 \times \qty{10}{\micro V} = \qty{0.2}{mV}$:
se hunden a la vez la sensibilidad y la selectividad.

\textbf{21.} $P = E^2/(2R_{\text{tot}}) = 10^{-10}/125 = \qty{8e-13}{W}$;
fracción en $R_a$: $50/62.6 = 80\%$.

\textbf{22.} La adaptación pide una resistencia de carga igual a $R_a$, es
decir una resistencia total grande; la selectividad pide la resistencia
total más pequeña posible ($Q = L\omega_0/R_{\text{tot}}$). Con una
conexión directa no se pueden tener las dos cosas.

\textbf{23.} Una toma intermedia o un pequeño condensador de acoplamiento
presentan a la antena solo una fracción del circuito resonante: el
circuito ``ve'' muy reducida la resistencia de la antena, con lo que $Q$
sigue alto y la antena entrega potencia cerca de su óptimo.

\textbf{24.} En las frecuencias de potencia mitad la \omterm{def:b1:sinusoidal-impedance:impedance}{reactancia} vale
$\pm R$, luego $\varphi = \pm45^\circ$ y $\cos\varphi = 1/\sqrt2$; como
además $I =
I_{\max}/\sqrt2$, $P = \tfrac12 EI\cos\varphi$ es la mitad de su valor en
la resonancia.

\textbf{25.} $Q$ multiplica la \omterm{def:b1:wave-propagation:signal}{señal} por $Q$ y estrecha la banda a
$f_0/Q$, a costa de una respuesta más lenta ($\tau = 2Q/\omega_0$) y de un
acoplamiento frágil con la antena; para canales de \qty{10}{kHz} a
\qty{1}{MHz}, $Q \approx 100$ es el número --- y la antena debe acoplarse
débilmente para conservarlo.
\end{solution}
