\section*{Capítulo \ref{ch:b1:charged-particles} --- Partículas cargadas en campos E y B}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$: electrón
\qty{1.9e7}{m/s}; protón \qty{4.4e5}{m/s} ($\sqrt{1836}$ veces más lento).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. A $30^\circ$: una hélice de radio $R\sin 30^\circ
= \qty{10}{m}$ y paso $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, que avanza
a lo largo de la línea de campo.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Iones más rápidos: gana la
fuerza magnética y se curvan hacia ese lado; con carga opuesta y la misma
rapidez, ambas fuerzas se invierten y siguen equilibrándose --- el
selector es ciego al signo y a la masa.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; desviación
$0.30 \times 0.10 = \qty{3.0}{cm}$; sensibilidad \qty{0.30}{mm/V}. Los
electrones más rápidos pasan menos tiempo entre las placas: $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; separación $2\Delta R =
\qty{1.2}{cm}$ entre los puntos de llegada (los radios difieren en
\qty{5.8}{mm} y los diámetros en el doble).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; con \qty{100}{keV} por vuelta: $270$ vueltas; tiempo $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
periodo $2\pi m/eB = \qty{18}{ns}$, paso $v_\parallel T = \qty{1.5}{cm}$:
unas $64$ vueltas por metro.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: por debajo del microvoltio. Con una $n$ diez millones
de veces menor, un semiconductor da milivoltios --- un sensor utilizable.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; multiplíquese escalarmente por $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ sea cual sea $\vect B(\vect r, t)$, siempre que no haya $\vect E$.
La dirección de la velocidad, el \omterm{def:b1:point-kinematics:frenet}{radio de curvatura} y el paso pueden
cambiar todos; la rapidez y la \omterm{thm:b1:work-and-energy:ke}{energía cinética} (\qty{10}{keV}), no.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. Con $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- una
rotación ciclotrónica pura de $\vect w$ a $\omega_c$. Partiendo del reposo,
$\vect w(0) = -(E/B)\vect e_x$: $\vect w$ gira con norma constante $E/B$, así
que la velocidad es la deriva $E/B$ más un vector giratorio de la misma
norma --- una cicloide, con \omterm{def:b1:work-and-energy:turning}{puntos de retroceso} donde $\vect v = \vect 0$.
Electrón: deriva $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, altura de los arcos $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (unos $15^\circ$):
desviación $\approx 0.30 \times 0.27 = \qty{8}{cm}$ con un modesto
militesla. Conseguir ángulos grandes con placas exigiría kilovoltios y
placas largas; una bobina desvía el haz decenas de grados sin tensión alguna.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: no hay potencia,
la rapidez es constante.

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, libre de
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ por cruce, \qty{100}{keV} por vuelta.

\textbf{5.} El hueco ha de estar en la polaridad aceleradora cada vez que
llega el protón, a medio periodo de distancia; fuera de frecuencia, la
fase se desliza y los impulsos acaban frenándolo.

\textbf{6.} Tras una vuelta, $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; tras $n$ vueltas, $E_k =
n \times \qty{100}{keV}$ y $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ vueltas; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} No: $E_k$ depende solo de $B$ y de $R_{\max}$; $U$ fija el
número de vueltas y el tiempo.

\textbf{11.} Camino de \qty{560}{m}: hace falta $\lambda \gtrsim \qty{1}{km}$, es decir
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
un vacío alto.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; el periodo crece un $2.9\%$. A
lo largo de $270$ vueltas el desfase acumula varios periodos --- las
últimas vueltas quedarían fuera de fase; en la práctica se ajustan $B$ y
la fase de inyección, y \qty{27}{MeV} está cerca del límite de la máquina sencilla.

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; $R =
\qty{1.1}{m}$ con \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ crece un $24\%$ del centro al borde: una
frecuencia fija se desfasa sin remedio.

\textbf{15.} Sincrociclotrón: bajar la frecuencia a medida que el paquete
gana energía --- solo un paquete cada vez, corriente media baja.
Isócrono: dar al campo la forma $B(R) \propto \gamma$ para que el periodo
se mantenga constante --- haz continuo, la opción moderna.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, algo más que los
\qty{1.1}{m} no relativistas.

\textbf{17.} Impulso transversal $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; ángulo $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: inútil para desviar.

\textbf{18.} Magnético: $R = p/eB = \qty{2.3}{m}$, ángulo $L/R = \qty{0.43}{rad}
\approx 25^\circ$: mil veces más --- imanes, pues.

\textbf{19.} Energía por protón $\qty{3.68e-11}{J}$; dosis $2.0 \times 0.50 =
\qty{1.0}{J}$: $2.7\times10^{10}$ protones.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \qty{6.2e9}{protones/s}$: unos
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
una lamparilla, entregada exactamente donde se quiere.

\textbf{22.} $\qty{58}{MeV}$ en \qty{1}{cm}: unos \qty{6}{keV/\micro m} ---
una ionización densa a lo largo del último tramo.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionizaciones por protón ---
el daño molecular que mata la célula.

\textbf{24.} Los rayos X depositan energía a lo largo de todo su camino,
sobre todo cerca de la entrada; los protones depositan casi toda la suya
al final de su alcance, respetando el tejido anterior y sin dejar nada detrás.

\textbf{25.} $R = mv/eB$ y $T = 2\pi m/eB$ (la circunferencia), $E_k = (eBR)^2/2m$
(la energía), $E_k/2eU$ (las vueltas), dosis $=$ (protones $\times$ energía)/masa
y corriente $= e \times$ ritmo (el tiempo).
\end{solution}
