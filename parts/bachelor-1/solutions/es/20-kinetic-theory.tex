\section*{Capítulo \ref{ch:b1:kinetic-theory} --- Teoría cinética y gas perfecto}

\begin{solution}{exo:b1:kinetic-theory:1}
$N = PV/k_BT = 1.013\times10^5 \times 10^{-3}/(1.38\times10^{-23} \times 273) =
\num{2.7e22}$; $n^{*-1/3} = (3.7\times10^{-26})^{1/3} = \qty{3.3}{nm}$, diez
veces el tamaño molecular.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:2}

\qty{484}{m/s}, CO$_2$ \qty{412}{m/s}; N$_2$ a \qty{77}{K}: \qty{262}{m/s}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:3}
$P = \qty{3.5}{bar}$ absolutos: $n = PV/RT = 3.5\times10^5 \times 0.030/(8.314
\times 293) = \qty{4.3}{mol}$, $\qty{125}{g}$. A \qty{323}{K}, con $P \propto T$:
\qty{3.86}{\omterm{def:b1:kinetic-theory:pressure}{bar}} (manométrica \qty{2.85}{\omterm{def:b1:kinetic-theory:pressure}{bar}}).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:4}
$n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$; $\lambda = 1/(\sqrt2\pi \times 1.37\times
10^{-19} \times 2.4\times10^{25}) = \qty{6.8e-8}{m}$; frecuencia $u/\lambda \approx
500/7\times10^{-8} = \qty{7e9}{s^{-1}}$. Como $\lambda \propto 1/P$: \qty{1}{m} para
$P \approx 10^5 \times 7\times10^{-8} = \qty{7e-3}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:5}
$\rho = PM/RT$: \qty{1.20}{kg/m^3} a \qty{293}{K} y \qty{0.946}{kg/m^3} a
\qty{373}{K}. Fuerza ascensional $(1.20 - 0.95) \times 2800 \times 9.81 = \qty{7.0}{kN}$:
unos \qty{720}{kg}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:6}
$P_{\mathrm{O}_2} = 0.21P$: \qty{0.21}{atm}; \qty{0.11}{atm} a \qty{5000}{m};
\qty{0.07}{atm} en el Everest --- un tercio del nivel del mar. El oxígeno
entra en la sangre en proporción a su \omterm{def:b1:kinetic-theory:pressure}{presión} parcial; a un tercio, los
pulmones no logran saturar la hemoglobina.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:7}
\omterm{def:b1:kinetic-theory:model}{Gas perfecto}: $P = RT/V = 2494/10^{-3} = \qty{24.9}{bar}$. Van der Waals:
$RT/(V - b) - a/V^2 = 2494/9.57\times10^{-4} - 0.364/10^{-6} = 26.1 - 3.6 =
\qty{22.4}{bar}$: la atracción ($-3.6$ \omterm{def:b1:kinetic-theory:pressure}{bar}) pesa más que el volumen
excluido ($+1.2$ \omterm{def:b1:kinetic-theory:pressure}{bar}); el CO$_2$ real queda un $10\%$ por debajo del valor ideal.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:8}
$n = PV/RT = 10^5 \times 50/(8.314 \times 300) = \qty{2000}{mol}$; $U = \tfrac52
nRT = \qty{1.25e7}{J}$; $\Delta U = \tfrac52 nR \times 5 = \qty{208}{kJ}$;
\qty{104}{s} con \qty{2}{kW}.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:9}
$\Delta L = \alpha L\Delta T = 1.2\times10^{-5} \times 30 \times 50 = \qty{18}{mm}$.
Anillo: $\Delta d/d = \alpha\Delta T = 0.05/49.95 = 10^{-3}$: $\Delta T = 10^{-3}/
1.9\times10^{-5} = \qty{53}{K}$ (el agujero se dilata como el metal que lo rodea).
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:10}
$\tfrac32 k_BT = \qty{6.2e-21}{J} = \qty{0.039}{eV}$. Argón, $\tfrac32 R =
\qty{12.5}{J/(mol.K)}$; nitrógeno, $\tfrac52 R = \qty{20.8}{J/(mol.K)}$: la
molécula diatómica almacena energía también en la rotación (dos grados de
libertad más), sea cual sea su masa.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:11}
Modelo de las seis direcciones: como en el texto, $P = \tfrac13 n^*mu^2$.
Isotropía: $\langle v_x^2\rangle = \langle v_y^2\rangle = \langle v_z^2\rangle$ y
su suma vale $\langle v^2\rangle$, luego cada uno es
$\tfrac13\langle v^2\rangle$. El argumento del flujo con una distribución da
$P = n^*m\langle v_x^2\rangle$ (las moléculas con $v_x > 0$ golpean la pared,
con cantidad de movimiento $2mv_x$ cada una y flujo $n^*v_x/2$ por clase de
rapidez, promediado), es decir $\tfrac13 n^*m\langle v^2\rangle$: lo mismo.
\end{solution}

\begin{solution}{exo:b1:kinetic-theory:12}
H$_2$: \qty{1.9}{km/s}, O$_2$: \qty{0.48}{km/s}; Tierra \qty{11.2}{km/s},
Luna \qty{2.4}{km/s}. Para el O$_2$ en la Tierra, $v_e/u = 23$: la cola de
la distribución más allá de eso es del todo despreciable; para el H$_2$,
$v_e/u \approx 6$: una fracción ínfima pero constante escapa cada año y, en
miles de millones de años, el hidrógeno desaparece. En la Luna, $v_e/u$
vale $5$ para el O$_2$ y $1.2$ para el H$_2$: todo se fugó.
\end{solution}

\begin{solution}{pb:b1:kinetic-theory:1}
\textbf{1.} $V = \qty{200}{m^3}$: $n = PV/RT = 1.013\times10^5 \times 200/(8.314
\times 293) = \qty{8320}{mol}$; $N = nN_A = \num{5.0e27}$.

\textbf{2.} $m = nM = \qty{241}{kg}$: tres personas.

\textbf{3.} $n^* = N/V = \qty{2.5e25}{m^{-3}}$; distancia $n^{*-1/3} = \qty{3.4}{nm}$,
nueve diámetros moleculares.

\textbf{4.} N$_2$: $\sqrt{3 \times 8.314 \times 293/0.028} = \qty{511}{m/s}$;
O$_2$: \qty{478}{m/s}. El mismo $\tfrac32 k_BT$ por molécula: $\tfrac12 mu^2$
fijo, luego $u \propto 1/\sqrt m$.

\textbf{5.} $\tfrac32 k_BT = \qty{6.1e-21}{J}$; en total $N \times 6.1\times10^{-21}
= \qty{3.0e7}{J}$; un coche de \qty{1000}{kg} a \qty{100}{km/h}: \qty{3.9e5}{J} ---
el movimiento molecular del aula equivale a ochenta coches.

\textbf{6.} $\lambda = 1/(\sqrt2\pi \times 1.37\times10^{-19} \times 2.5\times10^{25})
= \qty{66}{nm}$; tiempo $\lambda/u = 66\times10^{-9}/500 = \qty{1.3e-10}{s}$.

\textbf{7.} Impactos por segundo sobre $S$: $\tfrac16 n^*Su = \tfrac16 \times 2.5\times
10^{25} \times 10^{-4} \times 500 = \qty{2.1e23}{s^{-1}}$.

\textbf{8.} Cada uno entrega $2mu = 2 \times 4.8\times10^{-26} \times 500 = \qty{4.8e-23}{kg.m/s}$
(con $m = M/N_A = \qty{4.8e-26}{kg}$): fuerza $2.1\times10^{23} \times 4.8\times10^{-23}
= \qty{10}{N}$ sobre \qty{1}{cm^2}, es decir \qty{1.0e5}{Pa}. El modelo
reproduce la \omterm{def:b1:kinetic-theory:pressure}{presión} atmosférica.

\textbf{9.} $F = PS = 1.013\times10^5 \times 20 = \qty{2.0e6}{N}$ --- doscientas
toneladas; la misma \omterm{def:b1:kinetic-theory:pressure}{presión} actúa por el otro lado.

\textbf{10.} $P' = P \times 303/293 = \qty{1.048e5}{Pa}$; neto $\Delta P\,S = 3.5
\times10^3 \times 20 = \qty{70}{kN}$ hacia fuera: siete toneladas sobre la pared.

\textbf{11.} Sale aire hasta que se igualan las presiones: $n \propto 1/T$ a
$P$ y $V$ fijos: salen $8320 \times (1 - 293/303) = \qty{275}{mol}$,
\qty{8}{kg}.

\textbf{12.} $\rho = PM/RT$: \qty{1.205}{kg/m^3} a \qty{293}{K} y \qty{0.946}{kg/m^3}
a \qty{373}{K}.

\textbf{13.} Aire caliente $0.946 \times 2800 \times 9.81 = \qty{26.0}{kN}$; aire
frío desalojado $1.205 \times 2800 \times 9.81 = \qty{33.1}{kN}$; fuerza
ascensional \qty{7.1}{kN} (\qty{725}{kg}).

\textbf{14.} $725 - 300 = \qty{425}{kg}$: cinco pasajeros.

\textbf{15.} La fuerza ascensional es $\propto \rho_c - \rho_h = \rho_c(1 - 293/T)$:
duplicar $1 - 293/373 = 0.215$ exige $T = 293/0.57 = \qty{514}{K}$ (\qty{240}{\degreeCelsius}),
muy por encima de lo que tolera el nailon (unos \qty{120}{\degreeCelsius}).

\textbf{16.} Helio: $\rho = 1.205 \times 4/29 = \qty{0.166}{kg/m^3}$; fuerza
ascensional $(1.205 - 0.166) \times 2800 \times 9.81 = \qty{28.5}{kN}$ (\qty{2.9}{t}),
cuatro veces más --- pero el helio es caro y se pierde en cada aterrizaje,
mientras que el aire caliente es gratis y su empuje se regula con el quemador.

\textbf{17.} $0.21 \times 1.013\times10^5 = \qty{2.1e4}{Pa}$.

\textbf{18.} $n^*_{\mathrm{O}_2} = P_{\mathrm{O}_2}/k_BT$: aula, $2.1\times10^4/
(1.38\times10^{-23} \times 293) = \qty{5.3e24}{m^{-3}}$; Everest, $0.21 \times 0.33
\times 1.013\times10^5/(1.38\times10^{-23} \times 250) = \qty{2.0e24}{m^{-3}}$:
un $38\%$ del aula.

\textbf{19.} $H = RT/Mg = 8.314 \times 260/(0.029 \times 9.81) = \qty{7.6}{km}$;
$P_0/2$ a $H\ln 2 = \qty{5.3}{km}$.

\textbf{20.} $c_s = \sqrt{1.4 \times 8.314 \times 293/0.029} = \qty{343}{m/s}$,
frente a $u = \qty{503}{m/s}$ para la mezcla: $c_s/u = \sqrt{\gamma/3} = 0.68$.
El sonido es una perturbación que transportan las propias moléculas, así
que viaja a una velocidad del orden de la suya --- algo menor, porque solo
cuenta la componente en la dirección de propagación.

\textbf{21.} $c_s \propto \sqrt T$: $343\sqrt{303/293} = \qty{349}{m/s}$.
Helio: $\sqrt{(5/3) \times 8.314 \times 293/0.004} = \qty{1010}{m/s}$: las
resonancias del tracto vocal escalan con $c_s$, de modo que todos los
formantes suben tres veces.

\textbf{22.} Más lento: $343\sqrt{250/293} = \qty{317}{m/s}$, un $8\%$ menos;
la \omterm{def:b1:kinetic-theory:pressure}{presión} no interviene.

\textbf{23.} El mismo $N$ (mismos $P$, $V$, $T$); masa $\times 40/29 = \qty{333}{kg}$;
rapideces $\times\sqrt{29/40}$: $u = \qty{428}{m/s}$; $U = \tfrac32 nRT$ en vez de
$\tfrac52$: un $60\%$.

\textbf{24.} $U = \tfrac52 \times 8320 \times 8.314 \times 293 = \qty{5.1e7}{J}$;
$\Delta U = \tfrac52 nR \times 10 = \qty{1.7e6}{J}$.

\textbf{25.} Un solo promedio, $\langle v^2\rangle$, fija la \omterm{def:b1:kinetic-theory:pressure}{presión}
($\tfrac13 n^*m\langle v^2\rangle$), define la \omterm{def:b1:kinetic-theory:temperature}{temperatura}
($\tfrac12 m\langle v^2\rangle = \tfrac32 k_BT$), da la ecuación de estado y
la \omterm{def:b1:kinetic-theory:U}{energía interna}, y marca la escala de la velocidad del sonido.
\end{solution}
