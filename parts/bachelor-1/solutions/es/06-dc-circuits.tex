\section*{Capítulo \ref{ch:b1:dc-circuits} --- Circuitos en corriente continua: leyes de Kirchhoff y teoremas}

\begin{solution}{exo:b1:dc-circuits:1}
$q = 1.5 \times 3600 = \qty{5400}{C}$, $N = 5400/1.6\times10^{-19} =
\num{3.4e22}$ electrones. $c/f$: \qty{6000}{km} a \qty{50}{Hz} (vivienda:
sí); \qty{300}{m} a \qty{1}{MHz} (circuito de \qty{10}{cm}: sí);
\qty{30}{cm} a \qty{1}{GHz}, comparable a \qty{10}{cm}: no.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:2}
$I = 2000/230 = \qty{8.7}{A}$; $R = U^2/P = \qty{26.5}{\ohm}$; $E =
\qty{6.0}{kWh} = \qty{2.2e7}{J}$. Cable: un conductor $R = 1.7\times10^{-8}
\times 5.0/1.5\times10^{-6} = \qty{57}{m\ohm}$, dos conductores
\qty{0.11}{\ohm}: $P = 0.11 \times 8.7^2 = \qty{8.6}{W}$.
\end{solution}

$10 \parallel 20 = \qty{6.7}{\ohm}$, más $30$: \qty{36.7}{\ohm}.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:4}
$u_2 = 12 \times 2.2/6.9 = \qty{3.83}{V}$. Con carga: $2.2 \parallel 2.2 =
\qty{1.1}{k\ohm}$, $u_2 = 12 \times 1.1/5.8 = \qty{2.28}{V}$: la carga
hunde el divisor un $40\%$ --- un divisor solo ``divide'' si alimenta algo
de resistencia mucho mayor.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:5}
$r = (12.6 - 11.4)/120 = \qty{10}{m\ohm}$; $I_N = 12.6/0.010 =
\qty{1260}{A}$. Al motor de arranque $11.4 \times 120 = \qty{1.37}{kW}$; se
pierden $0.010 \times 120^2 = \qty{144}{W}$; rendimiento $1368/1512 = 90\%$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:6}
$E_{\mathrm{Th}} = \qty{3.83}{V}$ (divisor en circuito abierto); $R_{\mathrm{Th}}
= 4.7 \parallel 2.2 = \qty{1.50}{k\ohm}$ (fuente cortocircuitada). Con carga:
$u = 3.83 \times 2.2/(1.50 + 2.2) = \qty{2.28}{V}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:7}
Superposición: $E_1$ sola ve $r_2 \parallel R = \qty{1.0}{\ohm}$,
$u = 10 \times 1/(1 + 1) = \qty{5.0}{V}$; $E_2$ sola ve $r_1 \parallel R
= \qty{0.667}{\ohm}$, $u = 5 \times 0.667/2.667 = \qty{1.25}{V}$; en total
\qty{6.25}{V}. Millman: $(10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = \qty{6.25}{V}$.
Batería 1: $(10 - 6.25)/1 = \qty{3.75}{A}$ que salen; batería 2: $(5 - 6.25)/2
= \qty{-0.63}{A}$, es decir \qty{0.63}{A} \emph{hacia} ella --- la está
cargando la más fuerte; carga $6.25/2 = \qty{3.1}{A}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:8}
$i = 9/(R + 3)$, $P = Ri^2$, $\eta = R/(R + 3)$: $\qty{1}{\ohm}$:
\qty{5.1}{W}, $25\%$; $\qty{3}{\ohm}$: \qty{6.75}{W}, $50\%$;
$\qty{12}{\ohm}$: \qty{4.3}{W}, $80\%$. Un aparato a batería busca
rendimiento (autonomía), luego $R \gg r$; la carga adaptada malgasta la
mitad de la energía en calor dentro de la batería.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:9}
$R = (5.0 - 2.0)/0.015 = \qty{200}{\ohm}$; $P_R = 3.0 \times 0.015 =
\qty{45}{mW}$, $P_{\mathrm{LED}} = 2.0 \times 0.015 = \qty{30}{mW}$. La
\omterm{met:b1:dc-circuits:loadline}{recta de carga}, de $(0, \qty{25}{mA})$ a $(\qty{5}{V}, 0)$, corta la
característica vertical en \qty{2.0}{V}. A \qty{3.3}{V} el mismo \omterm{def:b1:dc-circuits:resistor}{resistor}
da solo \qty{6.5}{mA}; para \qty{15}{mA} hace falta $R = 1.3/0.015 =
\qty{87}{\ohm}$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:10}
$u = E\left[\dfrac12 - \dfrac{1 + \epsilon}{2 + \epsilon}\right] =
-E\dfrac{\epsilon}{2(2 + \epsilon)} \approx -\dfrac{E\epsilon}{4} =
\qty{-1.25}{mV}$. En un divisor simple esa misma variación se monta sobre
un desplazamiento de \qty{2.5}{V} (\qty{2.50125}{V}); el puente entrega
los \qty{1.25}{mV} en torno a cero, que pueden amplificarse mil veces sin
saturar nada.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:11}
Por simetría, los tres vecinos del vértice de entrada comparten un
potencial y los tres vecinos del vértice de salida otro. La corriente
$I$ se reparte en $I/3$ por las tres aristas de entrada, cada una después
en dos de $I/6$ por las seis aristas centrales, que se recombinan en $I/3$
en las tres aristas de salida. \omterm{def:b1:dc-circuits:voltage}{Tensión}: $RI/3 + RI/6 + RI/3 = 5RI/6$, luego
$R_{\mathrm{eq}} = 5R/6$.
\end{solution}

\begin{solution}{exo:b1:dc-circuits:12}
Thévenin: $E = I_N r = \qty{8.0}{V}$ con \qty{4.0}{\ohm} en serie.
Malla: $8.0 - 6.0 = (4.0 + 2.0)i$, $i = \qty{0.33}{A}$ hacia la batería.
La batería recibe $6.0 \times 0.33 = \qty{2.0}{W}$ (se carga); el \omterm{def:b1:dc-circuits:resistor}{resistor}
de \qty{2}{\ohm} disipa \qty{0.22}{W}; la \omterm{def:b1:dc-circuits:voltage}{tensión} en bornes del conjunto
Norton vale $8.0 - 4.0 \times 0.33 = \qty{6.67}{V}$, así que su
\qty{4}{\ohm} lleva \qty{1.67}{A} y disipa \qty{11.1}{W}, y la \omterm{def:b1:dc-circuits:sources}{fuente de
corriente} entrega $2.0 \times 6.67 = \qty{13.3}{W}$ --- todo cuadra.
\end{solution}

\begin{solution}{pb:b1:dc-circuits:1}
\textbf{1.} Fem ideal $E$ en serie con $r$: $u = E - ri$.

\textbf{2.} $u(0) = \qty{12.6}{V}$; $I_N = E/r = \qty{1260}{A}$. Una llave
apoyada sobre los bornes la llevaría: $\qty{16}{kW}$ en unos gramos de
acero --- se funde y salpica.

\textbf{3.} \omterm{def:b1:dc-circuits:sources}{Fuente de corriente} $I_N = \qty{1260}{A}$ en paralelo con
\qty{10}{m\ohm}.

\textbf{4.} $i = E/(R + r)$; $u = ER/(R + r)$; $P_R = E^2R/(R + r)^2$;
$P_r = E^2r/(R + r)^2$.

\textbf{5.} $\dd P_R/\dd R \propto (r - R)$: máximo en $R = r$,
$P_{\max} = E^2/4r = 12.6^2/0.040 = \qty{4.0}{kW}$; rendimiento
$R/(R + r) = 50\%$.

\textbf{6.} $12.6 - 0.010 \times 12 = \qty{12.48}{V}$; $12.6 - 0.010 \times
120 = \qty{11.40}{V}$. Representar $u$ frente a $i$ da una recta de
ordenada en el origen $E$ y pendiente $-r$ (\omterm{prop:b1:units-dimensions:regression}{mínimos cuadrados}, \cref{ch:b1:units-dimensions}).

\textbf{7.} Dos lámparas: \qty{110}{W} a \qty{12}{V}, \qty{9.2}{A};
$60/9.2 = \qty{6.5}{h}$. Energía $60 \times 3600 \times 12 = \qty{2.6}{MJ}$.

\textbf{8.} $R_h = 12^2/55 = \qty{2.62}{\ohm}$; en paralelo \qty{1.31}{\ohm}.

\textbf{9.} $i = 12.6/0.090 = \qty{140}{A}$; $u = 12.6 - 1.4 = \qty{11.2}{V}$;
$P_s = 0.080 \times 140^2 = \qty{1.57}{kW}$; se pierden $0.010 \times 140^2 =
\qty{196}{W}$.

\textbf{10.} $R = 0.080 \parallel 1.31 = \qty{75.4}{m\ohm}$; $i = 12.6/0.0854
= \qty{148}{A}$; $u = 12.6 - 1.48 = \qty{11.1}{V}$.

\textbf{11.} $i_s = i \times 1.31/(1.31 + 0.080) = \qty{139}{A}$; $i_h =
\qty{8.5}{A}$ para las dos lámparas.

\textbf{12.} Cada lámpara: $u^2/R_h = 11.1^2/2.62 = \qty{47}{W}$ frente a
\qty{55}{W}: un $14\%$ menos.

\textbf{13.} \omterm{thm:b1:dc-circuits:kirchhoff}{Ley de las mallas}: la \omterm{def:b1:dc-circuits:voltage}{tensión} en bornes vale $u = E - ri$; los
\qty{140}{A} del motor de arranque producen una caída de \qty{1.4}{V}
dentro de la batería, y las lámparas, en paralelo sobre esos mismos
bornes, reciben esa \omterm{def:b1:dc-circuits:voltage}{tensión} reducida.

\textbf{14.} Solo en parte: unos cables más gruesos reducen la caída en los
cables, no en $r$; las lámparas alimentadas desde los bornes siguen viendo $E - ri$.

\textbf{15.} Tres ramas entre los dos mismos nudos: $(E, r)$,
$(E_a, r_a)$ y $R_h = \qty{1.31}{\ohm}$.

\textbf{16.} $V = \dfrac{12.6/0.010 + 14.2/0.050 + 0/1.31}{100 + 20 + 0.763}
= \dfrac{1544}{120.8} = \qty{12.79}{V}$.

\textbf{17.} Alternador $(14.2 - 12.79)/0.050 = \qty{28.3}{A}$; batería
$(12.6 - 12.79)/0.010 = \qty{-18.6}{A}$ (que circulan \emph{hacia} ella);
lámparas $12.79/1.31 = \qty{9.8}{A}$; $28.3 = 18.6 + 9.8$.

\textbf{18.} Se carga. Recibe $12.79 \times 18.6 = \qty{238}{W}$
(\qty{234}{W} almacenados, \qty{3.5}{W} de calor); el alternador suministra
$12.79 \times 28.3 = \qty{362}{W}$ en sus bornes.

\textbf{19.} Batería sola (alternador $\to$ \qty{50}{m\ohm}): carga
$0.050 \parallel 1.31 = \qty{48.2}{m\ohm}$, $u_1 = 12.6 \times 48.2/58.2 =
\qty{10.44}{V}$. Alternador solo (batería $\to$ \qty{10}{m\ohm}): carga
$0.010 \parallel 1.31 = \qty{9.92}{m\ohm}$, $u_2 = 14.2 \times 9.92/59.9 =
\qty{2.35}{V}$. Suma \qty{12.79}{V}.

\textbf{20.} $u = 5.0\,R_s/(R_0 + R_s)$: \qty{3.60}{V} a
\qty{20}{\degreeCelsius} y \qty{1.40}{V} a \qty{90}{\degreeCelsius}.

\textbf{21.} $\Delta u = 5[R_a/(R_0 + R_a) - R_b/(R_0 + R_b)]$;
$\dd\Delta u/\dd R_0 = 0 \Leftrightarrow R_a/(R_0 + R_a)^2 = R_b/(R_0 +
R_b)^2 \Leftrightarrow R_0^2 = R_aR_b$: $R_0 = \sqrt{2000 \times 300} =
\qty{775}{\ohm}$.

\textbf{22.} $u = 5[\tfrac12 - R_s/(775 + R_s)]$, nula cuando $R_s =
\qty{775}{\ohm}$, es decir a la temperatura intermedia en la que el
termistor alcanza ese valor.

\textbf{23.} $R = 1.7\times10^{-8} \times 1.5/16\times10^{-6} =
\qty{1.6}{m\ohm}$ por conductor, \qty{3.2}{m\ohm} en total; caída $0.0032
\times 140 = \qty{0.45}{V}$; potencia \qty{63}{W}.

\textbf{24.} \qty{4}{mm^2}: \qty{6.4}{m\ohm} cada uno, \qty{12.8}{m\ohm}
en total: caída \qty{1.8}{V} y \qty{250}{W} desperdiciados --- el motor de
arranque perdería un sexto de su \omterm{def:b1:dc-circuits:voltage}{tensión}. Un cable grueso mantiene
$R_{\text{cable}} \ll R_s$.

\textbf{25.} Unos \qty{11.1}{V} en bornes durante el arranque; el motor de
arranque recibe $1568/(1568 + 196) \approx 89\%$ de la potencia de la
batería; la atenuación de las luces es la \omterm{thm:b1:dc-circuits:kirchhoff}{ley de las mallas}, $u = E - ri$.
\end{solution}
