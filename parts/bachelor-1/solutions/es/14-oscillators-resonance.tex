\begin{solution}{exo:b1:oscillators-resonance:1}

\qty{0.31}{s}$; $v_{\max} = A\omega_0 = \qty{1.0}{m/s}$; $E = \tfrac12 kA^2 =

\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:2}
$T = 2\pi\sqrt{\ell/g}$: \qty{2.0}{s} en la Tierra y \qty{4.9}{s} en la
Luna. El periodo del conjunto masa--resorte, $2\pi\sqrt{m/k}$, no contiene
$g$: no cambia.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:3}
$\tau = 2Q/\omega_0 = 100/(4\pi) = \qty{8.0}{s}$; la energía, $\propto \eu^{-2t/\tau}$,
se reduce a la mitad en $t = \tau\ln2/2 = \qty{2.8}{s}$; el $5\%$ tras
$3\tau = \qty{24}{s}$, unas $48 \approx Q$ oscilaciones.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:4}

en $3\omega_0$: $X_0/\sqrt{64 + 0.09} = \qty{0.13}{cm}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:5}
$v = Lf_0 = 8.0 \times 1.3 = \qty{10}{m/s}$ (\qty{37}{km/h}). A
\qty{60}{km/h}: $f = 16.7/8.0 = \qty{2.1}{Hz}$, $x_r = 1.6$: $X/X_g =
\sqrt{(1 + 5.2)/(2.4 + 5.2)} = 0.90$ --- apenas aislado; el aislamiento
exige $x_r \gg 1$ (a \qty{120}{km/h}, $x_r = 3.2$: $0.45$).
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:6}
El trabajo de un empujón es positivo solo cuando actúa en el sentido del
movimiento; empujando cada periodo, todos los empujones se suman; a
cualquier otro ritmo se alternan. Régimen estacionario:
$2\pi E/Q = \qty{25}{J}$, $E = \qty{80}{J}$;
$mg\ell(1 - \cos\theta_{\max}) = 80$: $1 - \cos\theta_{\max} = 0.109$,
$\theta_{\max} = 27^\circ$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:7}
$\delta = \ln(1/0.9) = 0.105$, $Q = \pi/\delta = 30$; $\Delta f = f_0/Q =
\qty{0.067}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:8}
Desplazamiento: en fase ($\varphi = 0$) muy por debajo, $-\pi/2$ en la
resonancia y $-\pi$ (en oposición) muy por encima. Velocidad en la
resonancia: en fase con la fuerza. La potencia es $\vect F\cdot\vect v$:
es la fase de la velocidad, y no la del desplazamiento, la que decide
cuánto trabajo realiza la excitación.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:9}
$V = \omega X = \omega(F_0/m)/\sqrt{(\omega_0^2 - \omega^2)^2 + (\omega\omega_0/Q)^2}$;
divídanse numerador y denominador por $\omega\omega_0/Q$: $V = (F_0Q/m\omega_0)/
\sqrt{1 + Q^2(\omega_0/\omega - \omega/\omega_0)^2}$, y $F_0Q/m\omega_0 = F_0/\alpha$.
La raíz es $\geq 1$, e igual a $1$ exactamente en $\omega = \omega_0$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:10}
$P = F_0^2/2\alpha = \qty{10}{W}$; $V_{\max} = F_0/\alpha = \qty{20}{m/s}$,
$E = \tfrac12 mV_{\max}^2 = \qty{40}{J}$; $E/P = \qty{4.0}{s}$ y $Q/\omega_0 =
m/\alpha = \qty{4.0}{s}$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:11}
$3\tau = \qty{10}{s}$: $\tau = \qty{3.3}{s}$, $Q = \omega_0\tau/2 = 2\pi \times
440 \times 3.3/2 \approx 4600$; $\Delta f = f_0/Q = \qty{0.1}{Hz}$. Responde
a una sola frecuencia, y emite una sola: un patrón bueno hasta unas pocas
partes en $10^4$.
\end{solution}

\begin{solution}{exo:b1:oscillators-resonance:12}
$m\ddot x = mg - \rho_wgS(h + x) = -\rho_wgSx$ (puesto que $mg = \rho_wgSh$):
$\omega_0^2 = \rho_wgS/m = g/h$; $T = 2\pi\sqrt{0.10/9.81} = \qty{0.63}{s}$.
Tanto la masa como la densidad del líquido entran solo a través de la
profundidad de equilibrio $h$.
\end{solution}

\begin{solution}{pb:b1:oscillators-resonance:1}
\textbf{1.} Fuerzas: el peso (equilibrado por el alargamiento estático),
el resorte $-ku$ y el amortiguador $-\alpha\dot u$; $m\ddot x = -ku - \alpha\dot u$ con $\ddot x =
\ddot x_g + \ddot u$.

\textbf{2.} $\ddot u + (\omega_0/Q)\dot u + \omega_0^2u = -\ddot x_g$, $\omega_0 =
\sqrt{k/m}$, $Q = \sqrt{km}/\alpha$.

\textbf{3.} $k = m\omega_0^2 = \pi^2 = \qty{9.9}{N/m}$; $\alpha = m\omega_0/Q =
3.14/0.70 = \qty{4.5}{N.s/m}$.

\textbf{4.} $u = -a_g/\omega_0^2$: la masa se desplaza una cantidad fija ---
el instrumento es un acelerómetro a frecuencia nula (y un inclinómetro).

\textbf{5.} El desplazamiento relativo $u(t)$.

\textbf{6.} $-\omega^2\underline U + j\omega(\omega_0/Q)\underline U + \omega_0^2\underline U
= \omega^2X_g$; divídase por $\omega_0^2$.

\textbf{7.} $U/X_g = x_r^2/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$.

\textbf{8.} $U \to X_g$: la masa no se mueve en el \omterm{thm:b1:newton-dynamics:laws}{sistema inercial} (su
inercia la mantiene quieta, porque el resorte es demasiado blando para
arrastrarla); el bastidor se mueve a su alrededor y la plumilla registra
el desplazamiento del suelo --- un sismómetro.

\textbf{9.} $U \approx x_r^2X_g = \omega^2X_g/\omega_0^2 = a_g/\omega_0^2$: un
acelerómetro.

\textbf{10.} $U = QX_g = 0.7X_g$: sin pico; los dos regímenes se enlazan
con suavidad y el instrumento no resuena.

\textbf{11.} Para $Q = 0.7$: una recta de pendiente $2$ que sube hasta $1$
cerca de $x_r = 1$ y después se mantiene plana en $1$; para $Q = 5$: lo
mismo con un pico de altura $5$ en $x_r = 1$.

\textbf{12.} $\underline U \to -X_g$: $u = -x_g$, la masa quieta mientras
el bastidor se mueve.

\textbf{13.} $\Delta\ell = g/\omega_0^2 = 9.81/(2\pi \times 0.02)^2 = \qty{620}{m}$:
absurdo.

\textbf{14.} $\ell = g/\omega_0^2 = \qty{620}{m}$.

\textbf{15.} Puerta de jardín: el péndulo oscila en torno a un eje casi
vertical, así que solo una fracción minúscula de $g$ lo devuelve al
equilibrio y un brazo corto da un periodo largo. Realimentación de fuerza:
una bobina mantiene la masa fija, y la corriente necesaria es proporcional
a la aceleración del suelo --- la rigidez es electrónica y ajustable.

\textbf{16.} $\Delta\ell = 9.81/\pi^2 = \qty{0.99}{m}$: viable.

\textbf{17.} $x_r = 0.2$: $U/X_g = 0.04/\sqrt{0.92 + 0.08} = 0.040$:
\qty{40}{\micro m} --- régimen de acelerómetro, hace falta amplificar
(palanca óptica o electrónica).

\textbf{18.} $x_r = 10$: $100/\sqrt{9801 + 204} \approx 1.00$: \qty{1.0}{mm},
un registro fiel del desplazamiento.

\textbf{19.} $\omega_0 = \qty{6.3e3}{rad/s}$: $U = g/\omega_0^2 = \qty{2.5e-7}{m}$
--- se lee como la variación del espacio de un condensador.

\textbf{20.} $U/(a_g/\omega_0^2) = 1/\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}$; con
$Q = 0.7$ decrece de forma monótona y se mantiene por encima de $0.9$
hasta $x_r \approx 0.7$: unos \qty{700}{Hz}.

\textbf{21.} $\tau = 2Q/\omega_0 = \qty{0.22}{ms}$: sigue una caída (de
decenas de milisegundos) sin retraso.

\textbf{22.} $\tau = 1.4/3.14 = \qty{0.45}{s}$: tras un escalón del suelo
el registro se estabiliza en medio segundo y sin oscilar.

\textbf{23.} Un $Q$ alto haría que el instrumento resonase en $f_0$ tras
cada sacudida (registrándose a sí mismo en vez de al suelo) y presentase
un pico cerca de $f_0$; $Q \approx 0.7$ es plano y rápido.

\textbf{24.} $U = QX_g = 0.7X_g$ frente al ideal $X_g$ (o $a_g/\omega_0^2
= X_g$): un $30\%$ por debajo.

\textbf{25.} Compárese la frecuencia de la \omterm{def:b1:wave-propagation:signal}{señal} con $f_0$: muy por
encima, la masa es una referencia fija y $u$ es el desplazamiento; muy por
debajo, $u$ es la aceleración dividida por $\omega_0^2$.
\end{solution}
