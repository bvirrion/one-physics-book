\section*{Capítulo \ref{ch:b1:fluid-statics} --- Estática de fluidos}

\begin{solution}{exo:b1:fluid-statics:1}
$P = P_0 + \rho gh$: \qty{2.0e5}{Pa} (\qty{2.0}{bar}), \qty{1.1e6}{Pa}
(\qty{11}{bar}), \qty{1.0e7}{Pa} (\qty{101}{bar}). A \qty{300}{m}: $P =
\qty{3.1e6}{Pa}$, fuerza \qty{3.1}{MN} sobre \qty{1}{m^2} (neta, contra
\qty{1}{atm} en el interior: \qty{3.0}{MN}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:2}
$h = P/\rho g$: mercurio \qty{0.760}{m}; agua \qty{10.3}{m}. Una bomba puede
a lo sumo hacer el vacío sobre el agua: la atmósfera empuja entonces la
columna \qty{10.3}{m} hacia arriba, no más.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:3}
$F_2 = F_1S_2/S_1 = \qty{10}{kN}$ (una tonelada); $d_2 = d_1S_1/S_2 = \qty{2.0}{mm}$
(mismo volumen desalojado, mismo trabajo).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:4}
Misma presión al nivel de la interfaz aceite--agua en ambas ramas: $\rho_og
h_o = \rho_wgh_w$, $h_w = 0.85 \times 12 = \qty{10.2}{cm}$ de agua por encima
de ese nivel en la otra rama; la superficie del aceite queda
$12 - 10.2 = \qty{1.8}{cm}$ más alta.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:5}
$H = RT/Mg = 8.314 \times 273/(0.029 \times 9.81) = \qty{8.0}{km}$; $P/P_0 =
\eu^{-z/H}$: $0.54$ a \qty{5}{km}, $0.33$ a \qty{8848}{m}; la mitad a $H\ln2 =
\qty{5.5}{km}$ --- los números empleados en \cref{ch:b1:kinetic-theory}.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:6}
$F = \tfrac12\rho gh^2L = \qty{1.96e8}{N}$; \omterm{prop:b1:fluid-statics:wall}{centro de presiones} a \qty{13.3}{m}
de profundidad, es decir \qty{6.7}{m} sobre la base; momento respecto de la base $F \times
6.7 = \qty{1.3e9}{N.m}$. Curvar la presa hacia el agua convierte el empuje
en compresión del arco, que se transmite a las laderas.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:7}
$917/1025 = 89\%$; tronco $60\%$. Balsa: empuje con inmersión total $1000 \times
2.0 \times 9.81 = \qty{19.6}{kN}$, peso propio \qty{11.8}{kN}: \qty{7.8}{kN}
de sobra, diez personas\dots{} la décima con los pies mojados: nueve para no
mojarse (\qty{800}{kg}).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:8}
Volumen sumergido $m/\rho$: en agua \qty{20}{cm^3}; en salmuera $20/1.1 = \qty{18.2}{cm^3}$,
$\qty{1.8}{cm^3}$ menos, es decir $1.8/0.50 = \qty{3.6}{cm}$ más de vástago: emergen
\qty{7.6}{cm}. Sensibilidad $\approx m/(\rho^2S) \approx \qty{40}{cm}$ por unidad de
densidad relativa (\qty{3.6}{cm} por cada $0.1$).
\end{solution}

\begin{solution}{exo:b1:fluid-statics:9}
Volumen desalojado $2800/1.025 = \qty{2732}{m^3}$: emergen \qty{268}{m^3}
($9\%$). Neutra: masa $1025 \times 3000 = \qty{3075}{t}$: hay que tomar \qty{275}{t}.
Agua dulce: el empuje baja a \qty{3000}{t}: \qty{75}{t} de más --- se hunde
salvo que se bombeen fuera \qty{75}{t}.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:10}
Aire desalojado: $10^{-2} \times 1.20 = \qty{12.0}{g}$; helio dentro $12.0 \times
4/29 = \qty{1.7}{g}$; goma \qty{3.0}{g}: fuerza ascensional neta \qty{7.3}{g} --- siete metros
de cordel ligero.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:11}
Momento $\int_0^h\rho gx\cdot xL\,\dd x = \rho gLh^3/3$, fuerza $\rho gLh^2/2$:
brazo $2h/3$. De $h_1$ a $h_2$: momento $\rho gL(h_2^3 - h_1^3)/3$, fuerza
$\rho gL(h_2^2 - h_1^2)/2$: profundidad $\frac23(h_2^3 - h_1^3)/(h_2^2 - h_1^2)$.
\end{solution}

\begin{solution}{exo:b1:fluid-statics:12}
Una inmersión adicional $x$ añade el empuje $\rho gSx$ hacia arriba: $m\ddot x = -\rho gSx$,
$\omega_0^2 = \rho gS/m$, $T = 2\pi\sqrt{m/\rho gS} = 2\pi\sqrt{0.020/(1000 \times
9.81 \times 5\times10^{-5})} = \qty{1.3}{s}$.
\end{solution}

\begin{solution}{pb:b1:fluid-statics:1}
\textbf{1.} $P = 1.013\times10^5 + 1025 \times 9.81 \times 300 = \qty{3.12e6}{Pa}
= \qty{31}{bar}$.

\textbf{2.} $S = \pi(0.40)^2 = \qty{0.503}{m^2}$: fuerza neta $(P - P_0)S =
\qty{1.52e6}{N}$ --- 155 toneladas, un camión cargado sobre una escotilla.

\textbf{3.} La quilla, \qty{10}{m} más abajo: $+\rho g \times 10 = \qty{1.0e5}{Pa}$ más,
un $3\%$: el casco está cargado casi uniformemente.

\textbf{4.} $\Delta\rho/\rho = \chi_T\Delta P = 4.6\times10^{-10} \times 3.0\times10^6
= 0.14\%$: despreciable para la presión (unos kilopascales de tres
millones).

\textbf{5.} Treinta bares fuera, uno dentro: el casco es un recipiente a
presión cargado hacia dentro; un cilindro (o una esfera) convierte la
presión en compresión a lo largo de su pared, que el acero resiste bien,
mientras que una plancha plana se doblaría.

\textbf{6.} $\Delta P\,R/e = 3.0\times10^6 \times 5.0/0.030 = \qty{5.0e8}{Pa}
= \qty{500}{MPa}$: cerca del límite elástico --- \qty{300}{m} está próximo al
límite de este casco, y los buques que bajan más necesitan un acero más
grueso o más resistente (o titanio).

\textbf{7.} Volumen desalojado $m_0/\rho = 2800/1.025 = \qty{2732}{m^3}$: emergen \qty{268}{m^3}
($9\%$).

\textbf{8.} Neutra: $m = \rho V = \qty{3075}{t}$: hay que tomar \qty{275}{t}.

\textbf{9.} El empuje cae un $0.1\%$ de $\rho Vg$: $0.001 \times 3075 \times 9.81
\times10^3 = \qty{30}{kN}$ hacia abajo (tres toneladas). Más hondo $\Rightarrow$ más
compresión $\Rightarrow$ más pesado: inestable --- un submarino ha de
compensar continuamente (y un casco de acero se comprime menos que el agua,
lo que ayuda; un casco demasiado flexible se hundiría sin remedio).

\textbf{10.} Empuje $1000 \times 3000 \times 9.81 = \qty{29.4}{MN}$ frente a un
peso $3075 \times 9.81 \times 10^3 = \qty{30.2}{MN}$: \qty{0.74}{MN} hacia abajo
(\qty{75}{t}); hay que bombear fuera \qty{75}{t}.

\textbf{11.} Queda de inmediato \qty{20}{t} más ligero: hay que tomar \qty{20}{t} de agua
(\qty{19.5}{m^3}) en los tanques de compensación.

\textbf{12.} El agua solo puede expulsarse con algo a mayor presión que el
mar exterior; el aire comprimido de las botellas la expulsa --- pero solo
mientras la presión de la botella supere a la presión ambiente a esa
profundidad.

\textbf{13.} $PV$ constante: \qty{275}{m^3} a \qty{31}{bar} exigen $275 \times 31
= \qty{8500}{m^3}$ de aire a \qty{1}{atm} --- almacenados a \qty{200}{bar} en
\qty{43}{m^3} de botellas.

\textbf{14.} \qty{2.0}{bar} a \qty{10}{m}; \qty{4.0}{bar} a \qty{30}{m}.

\textbf{15.} $PV$ constante: $6.0/4.0 = \qty{1.5}{L}$ --- la caja torácica se
aplasta; los apneístas lo notan.

\textbf{16.} El aire tomado a \qty{4}{bar} se expande cuatro veces: \qty{24}{L} en unos
pulmones que contienen \qty{6}{L}: rotura. Nunca hay que contener la
respiración al subir; se espira sin parar.

\textbf{17.} $\Delta P = \rho g \times 1.0 = \qty{1.0e4}{Pa}$; fuerza $\qty{1.0e4}{Pa}
\times 0.05 = \qty{500}{N}$ sobre el pecho: los músculos respiratorios no
pueden levantar cincuenta kilogramos --- un tubo respirador solo sirve en los primeros decímetros.

\textbf{18.} $12 \times 200 = \qty{2400}{L}$ a \qty{1}{bar}, es decir \qty{600}{L}
a \qty{4}{bar}: \qty{30}{min}.

\textbf{19.} $V = 10^8/1025 = \qty{97600}{m^3}$.

\textbf{20.} En agua dulce ha de desalojar $10^5\ \unit{m^3}$: \qty{2400}{m^3}
más, es decir $2400/8000 = \qty{0.30}{m}$ más de calado.

\textbf{21.} $10^4/1.025 = \qty{9756}{m^3}$ más: \qty{1.2}{m} más de calado.

\textbf{22.} Vacío, el $G$ del buque queda alto y poco casco está sumergido:
el metacentro puede caer por debajo de $G$ y el buque volcar con marejada.
El agua de mar en el fondo baja $G$ y aumenta el calado.

\textbf{23.} $\rho gh = \qty{2.0e5}{Pa}$, \qty{0.2}{MN/m^2}: quince veces menos
que los \qty{3}{MN/m^2} del submarino.

\textbf{24.} No: la presión a la profundidad del submarino es $P_0 + \rho gh$,
fijada solo por la profundidad. El peso del petrolero lo sostiene el agua
que desaloja, que eleva el nivel del mar en todas partes en una cantidad
inmensurable; ninguna carga adicional llega al submarino.

\textbf{25.} $\dd P/\dd z = -\rho g$ --- de donde $P = P_0 + \rho gh$ para toda
profundidad, fuerza y volumen pulmonar de aquí --- y el \omterm{thm:b1:fluid-statics:archimedes}{teorema de
Arquímedes}, que es esa relación integrada sobre una superficie cerrada.
\end{solution}
