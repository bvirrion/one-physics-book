\section*{Capítulo \ref{ch:b1:optical-instruments} --- Instrumentos ópticos y el ojo}

\begin{solution}{exo:b1:optical-instruments:1}
Infinito: $f' = \qty{17}{mm}$, $V = 1/0.017 = \qty{58.8}{\delta}$.
\qty{25}{cm}: $V = 58.8 + 1/0.25 = \qty{62.8}{\delta}$, $f' = \qty{15.9}{mm}$.
Amplitud \qty{4.0}{\delta}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:2}
Miope: el infinito ha de conjugarse con el \omterm{def:b1:optical-instruments:punctum}{punto remoto}, $f' = \qty{-40}{cm}$,
$V = \qty{-2.5}{\delta}$. Hipermétrope: un objeto a \qty{25}{cm} ha de
conjugarse (virtualmente) a \qty{1.0}{m}: $V = 1/\overline{OA'} - 1/\overline{OA}
= -1 + 4 = \qty{+3.0}{\delta}$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:3}
$G = 25/4.0 = 6.3$; sello en el plano focal, a \qty{4.0}{cm} de la
\omterm{def:b1:mirrors-thin-lenses:lens}{lente}. Detalle de \qty{2}{mm}: $\theta' = 2/40 = \qty{0.05}{rad}$, frente a
$2/250 = \qty{0.008}{rad}$ a \qty{25}{cm}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:4}
$G = 900/25 = 36$ y $900/9 = 100$; longitudes del tubo $f_1' + f_2' =
\qty{925}{mm}$ y \qty{909}{mm}; Luna: $36 \times 0.52^\circ = 19^\circ$
y $52^\circ$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:5}
$\gamma_1 = -160/4 = -40$; $G_2 = 250/20 = 12.5$; $G = 500$.
$\overline{F_1A} = -f_1'^2/\Delta = -16/160 = \qty{-0.10}{mm}$: objeto a
\qty{4.1}{mm} del \omterm{def:b1:optical-instruments:microscope}{objetivo}. El objeto debe permanecer a esa distancia del
\omterm{def:b1:optical-instruments:microscope}{objetivo}, así que el tubo entero se desplaza \qty{10}{\micro m} --- el
mando de enfoque fino.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:6}
\omterm{def:b1:optical-instruments:eye}{Ojo} desnudo: $1.5/3\times10^{-4} = \qty{5}{km}$. Anteojo: $\theta_{\min}
= 1.22 \times 5.5\times10^{-7}/0.100 = \qty{6.7e-6}{rad}$; faros
separables hasta $1.5/6.7\times10^{-6} \approx \qty{220}{km}$ (en
principio: el aire y la curvatura terrestre se interponen antes).
$G_r = 3\times10^{-4}/
6.7\times10^{-6} = 45$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Exposición $\propto N^2$: $(8/2)^2 = 16$ veces
más larga, $16/500 \approx 1/30$~\unit{s}. Se gana \omterm{def:b1:optical-instruments:camera}{profundidad de campo} (y
menos aberraciones) al precio de trepidación o de un trípode.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$; nítido desde $H/2 =
\qty{1.9}{m}$ hasta el infinito. Prefijada en $H$, la \omterm{def:b1:optical-instruments:camera}{cámara} no necesita
enfocar: la foto se hace en el instante mismo en que sucede la escena.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$, toda
la luz entra en el \omterm{def:b1:optical-instruments:eye}{ojo}. $7\times50$: \qty{7.1}{mm}, que iguala la pupila
nocturna, y $(50/30)^2 = 2.8$ veces más luz recogida: los del atardecer.
$20\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} \qty{1.5}{mm}, imagen apagada, campo
estrecho y pulso de la mano amplificado veinte veces.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
\omterm{def:b1:mirrors-thin-lenses:lens}{Lente} de contacto: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$. Gafas:
el \omterm{def:b1:optical-instruments:punctum}{punto remoto} queda a $25 - 1.5 = \qty{23.5}{cm}$ de la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, luego $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- más fuerte. La \omterm{def:b1:mirrors-thin-lenses:lens}{lente} ha de
formar la imagen en el \omterm{def:b1:optical-instruments:punctum}{punto remoto} del \omterm{def:b1:optical-instruments:eye}{ojo}, cuya distancia a la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
depende de dónde se sitúe la \omterm{def:b1:mirrors-thin-lenses:lens}{lente}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afocal: $F_1' = F_2$, separación $f_1' + f_2' = \qty{20}{cm}$ (más corta
que los \qty{40}{cm} de un kepleriano del mismo $G$). $G = -30/(-10) =
+3$: derecha. \omterm{prop:b1:optical-instruments:telescope}{Pupila de salida} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$: virtual, dentro del tubo, donde el \omterm{def:b1:optical-instruments:eye}{ojo}
no puede situarse; el \omterm{def:b1:optical-instruments:eye}{ojo}, detrás del \omterm{def:b1:optical-instruments:microscope}{ocular}, solo recoge los haces que
por azar le llegan --- campo estrecho, la queja clásica de los gemelos de
teatro.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$; $\Delta m = 2.5\log_{10}820 = 7.3$: magnitud límite
$13.3$. El flujo es $\propto 1/r^2$, así que la lámpara se ve
$\sqrt{820} = 29$ veces más lejos.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$: separación $f_1' + f_2' = \qty{1225}{mm}$
(\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}) o \qty{1206}{mm} (el de \qty{6}{mm}).

\textbf{2.} $L_1$: $\overline{O_1A_1} = f_1'$ (imagen en $F_1'$). $L_2$:
$\overline{O_2A_1} = -f_2'$, luego $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$: imagen en el infinito.

\textbf{3.} El haz converge en el plano focal común a la altura $h =
f_1'\alpha$ (rayo por $O_1$); desde ahí, el rayo que pasa por $O_2$ fija
la dirección emergente, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ y $-200$; Luna: $25^\circ$ y $104^\circ$.

\textbf{5.} Arriba y abajo carecen de sentido en el cielo; un anteojo
terrestre añade un par de prismas enderezadores (o un relé de \omterm{def:b1:mirrors-thin-lenses:lens}{lentes}).

\textbf{6.} En el plano focal de $L_1$, de diámetro $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} Objeto $O_1$ en $\overline{O_2O_1} = -(f_1' + f_2')$:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$;
aumento $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
diámetro $D/\abs G$.

\textbf{8.} \qty{25}{mm}: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, diámetro $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}:
\qty{6.0}{mm} por detrás, diámetro \qty{0.75}{mm}.

\textbf{9.} Todo rayo que ha entrado por el \omterm{def:b1:optical-instruments:microscope}{objetivo} pasa por la \omterm{prop:b1:optical-instruments:telescope}{pupila de
salida}: un \omterm{def:b1:optical-instruments:eye}{ojo} situado allí recibe el campo entero. Más atrás solo
intercepta los haces centrales: el campo se reduce a una mirilla.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$;
\omterm{def:b1:optical-instruments:eye}{ojo} desnudo $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$; el \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}
($48$) es el más próximo, algo por debajo; el de \qty{6}{mm} ($200$) lo
triplica.

\textbf{14.} Con $G = 200$ la mancha de difracción subtiende $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$: visiblemente borrosa. La luz
de la Luna se reparte sobre un área $(200/48)^2 = 17$ veces mayor en la
\omterm{def:b1:optical-instruments:eye}{retina}: más apagada.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Por encima de ese diámetro, la turbulencia limita la
resolución desde el suelo (salvo óptica adaptativa); un $D$ mayor sigue
recogiendo más luz y alcanzando objetos más débiles.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$: magnitud $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
$460$ veces la del \omterm{def:b1:optical-instruments:eye}{ojo} desnudo. La imagen retiniana tiene un área
$\abs G^2$ veces mayor: cociente de brillos superficiales $460/G^2 = 0.20$
($G = 48$) y $0.012$ ($G = 200$) --- igual a $(D/\abs G)^2/d_p^2$, el
cuadrado del cociente entre \omterm{prop:b1:optical-instruments:telescope}{pupila de salida} y pupila del \omterm{def:b1:optical-instruments:eye}{ojo}.

\textbf{19.} Para un objeto extenso, la luz recogida crece como $D^2$ pero
el área de la imagen como $G^2 \geq (D/d_p)^2$: el brillo superficial, en
el mejor de los casos, iguala al de la visión a simple vista (\omterm{prop:b1:optical-instruments:telescope}{pupila de
salida} $=$ pupila del \omterm{def:b1:optical-instruments:eye}{ojo}). Una estrella no se resuelve: toda su luz cae
en una única mancha de difracción, y su brillo crece como $(D/d_p)^2$.

\textbf{20.} Un punto de ángulo $\beta$ se conjuga a la altura $f_1'\beta$
en el plano focal; el diafragma deja pasar $\abs{f_1'\beta} \leq \phi/2$:
campo total $\phi/f_1'$ (ángulos pequeños).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$: la Luna entera
cabe. $5/1200 = 0.24^\circ$: no cabe.

\textbf{22.} Campo aparente $\phi/f_2'$: $20/25 = \qty{0.8}{rad} = 46^\circ$;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$: $3420/15 = \qty{230}{s}$, casi cuatro
minutos; $0.24^\circ = 864''$: \qty{58}{s}.

\textbf{24.} \omterm{def:b1:optical-instruments:microscope}{Ocular} de \qty{25}{mm} ($G = 48$, cerca de $G_r$, luminoso,
Luna entera) para el disco; el de \qty{6}{mm} ($G = 200$) para acercarse a
los cráteres, más apagado y más borroso. El número del instrumento:
cráteres de hasta \qty{1.7}{km}.

\textbf{25.} Con la misma $f_1'$ y el mismo $D$: los mismos $G$, campos,
\omterm{prop:b1:optical-instruments:telescope}{pupilas de salida}, resolución y luz (descontada la obstrucción central).
Se gana: ninguna aberración cromática. Cambio práctico: el foco queda por
delante del espejo, así que un pequeño espejo secundario debe desviar el
haz hacia el \omterm{def:b1:optical-instruments:microscope}{ocular}.
\end{solution}
