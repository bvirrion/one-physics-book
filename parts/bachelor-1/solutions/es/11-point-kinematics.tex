\section*{Capítulo \ref{ch:b1:point-kinematics} --- Cinemática del punto}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{12}{m/s} en \qty{3.0}{s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ en $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). \omterm{cor:b1:point-kinematics:circular}{Movimiento circular} uniforme: rapidez
constante y \omterm{cor:b1:point-kinematics:circular}{aceleración centrípeta}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Vértice en $t = 7.71/9.81 = \qty{0.79}{s}$, altura $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Alcance $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
Llegada al suelo: $(9.19, -7.71)$, \qty{12}{m/s} a $40^\circ$ bajo la
horizontal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. Al cabo de \qty{5}{s}: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. El término $2\dot r\dot\theta$ tiene dos mitades iguales:
la dirección de la \omterm{def:b1:point-kinematics:velocity}{velocidad} radial gira a razón de $\dot\theta$ (lo que
da $\dot r\dot\theta\,\vect e_\theta$) y alejarse aumenta la rapidez
ortorradial $r\dot\theta$ (otro $\dot r\dot\theta$).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ hacia el eje; paso $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ se adelanta a $x$ en un cuarto de periodo y $a = -\omega^2x$ es
opuesta a $x$. $\abs v$ es máximo en $x = 0$, y $\abs a$ en los extremos $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Entonces
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, y derivando una
vez más con la regla del producto se obtiene la \omterm{def:b1:point-kinematics:velocity}{aceleración} de
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Vértice: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$.
Lanzamiento: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$.
El radio en el vértice es menor en un factor $\cos^3\alpha$: la parábola
se curva más donde va más despacio y donde toda $\vect g$ es normal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. En el suelo ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ y $\vect a = (0, v^2/R)$, hacia arriba. En lo alto
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: rapidez $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. En $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). El primero es el término centrípeto
de la rotación; el segundo, $2\dot r\dot\theta$, existe solo porque la
propia dirección radial gira --- en el suelo, caminar recto a rapidez
constante no da \omterm{def:b1:point-kinematics:velocity}{aceleración} alguna.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (movimiento con $\theta$ creciente),
$\vect N = -\vect e_r$ (hacia $C$), $\rho = R$.

\textbf{3.} Arriba: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$.
Laterales: $625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ y $v = R\dot\theta$:
$\dot v = -g\sin\theta$. Negativa para $0 < \theta < \pi$ (frena al
subir), positiva al bajar.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: abajo
\qty{62.5}{m/s^2}; laterales $62.5 - 19.6 = \qty{42.9}{m/s^2}$; arriba $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} Abajo: \qty{62.5}{m/s^2} (sin parte tangencial); laterales:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; arriba: \qty{23.3}{m/s^2},
dirigida verticalmente hacia $C$: ángulo $0$ con la vertical.

\textbf{7.} Perpendicular abajo y arriba ($a_T = 0$); nunca paralela
($a_N > 0$ en todas partes).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
abajo \qty{2.50}{rad/s}, arriba \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: la ecuación de un
péndulo simple de longitud $R$ --- el vagón sobre su vía es una lenteja de
péndulo que pasa por encima.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Media de $25$, $20.7$, $15.3$ y $20.7$: unos \qty{20}{m/s};
longitud $2\pi R = \qty{63}{m}$: aproximadamente \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: sí. Mínimo: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} Abajo: $\vect a = 62.5$ hacia arriba y $\vect g$ hacia abajo:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} Arriba: ambas hacia abajo, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
que aprieta al pasajero contra el asiento (hacia el centro, es decir hacia
arriba para el pasajero invertido).

\textbf{15.} Laterales: $\vect a = 42.9$ horizontal hacia $C$ más $9.8$
hacia abajo (tangencial); $\vect a - \vect g$ vale $42.9$ horizontal:
$4.4g$ de lado, hacia el centro.

\textbf{16.} No pasa: $7.4g$ abajo (arriba y en los laterales va bien).

\textbf{17.} Peso aparente nulo arriba: $v_{\text{arr}}^2 = gR$; entonces
$v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ abajo y la carga vale
$5g + g = 6g$, con independencia de $R$.

\textbf{18.} $a_N = v^2/\rho$: la altura fija $v$ (energía) y la geometría
de la vía fija $\rho$; un $\rho$ distinto a la misma altura cambia solo
$a_N$.

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{arr}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: crece linealmente
desde cero --- una tasa constante de variación de la \omterm{def:b1:point-kinematics:velocity}{aceleración} --- en
lugar de saltar de $0$ a $v^2/R$ en la entrada de una circunferencia, cosa
que el cuello percibe como un golpe.

\textbf{22.} Una lágrima: ancha y de curvatura suave abajo, donde el vagón
va deprisa, y estrecha arriba, donde va despacio. Mantiene la carga cerca
de $3g$ en todo el recorrido y la aplica de forma gradual.

\textbf{23.} Visual: distancia horizontal \qty{50}{m}, altura
\qty{20}{m}: $r = \qty{53.9}{m}$, elevación $\beta = 21.8^\circ$. La
componente transversal de la \omterm{def:b1:point-kinematics:velocity}{velocidad} horizontal vale $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Vista desde un punto descentrado, la \omterm{cor:b1:point-kinematics:circular}{velocidad angular} de la
visual no es constante (rápida cuando el vagón está cerca, lenta cuando
está lejos): una cámara a \omterm{def:b1:point-kinematics:velocity}{velocidad} constante se desvía del vagón.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Coordenadas polares} en $C$ para la \omterm{def:b1:point-kinematics:velocity}{velocidad} y la \omterm{def:b1:point-kinematics:velocity}{aceleración}
sobre la circunferencia; la descomposición de Frenet para separar
``acelerar'' de ``girar'' y para la clotoide; \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} en la
cámara para la \omterm{def:b1:point-kinematics:velocity}{velocidad} de seguimiento; la energía (capítulos
siguientes) para la ley de rapidez.
\end{solution}
