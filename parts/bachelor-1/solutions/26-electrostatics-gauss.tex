\section*{Chapter \ref{ch:b1:electrostatics-gauss} --- Electrostatics: Field and Gauss's Law}

\begin{solution}{exo:b1:electrostatics-gauss:1}
$F_e = 8.99 \times 10^9 \times (1.6 \times 10^{-19})^2/(2 \times 10^{-15})^2 = \qty{58}{N}$;
$F_g = 6.67 \times 10^{-11} \times (1.67 \times 10^{-27})^2/(4 \times 10^{-30}) =
\qty{4.6e-35}{N}$; ratio $1.2 \times 10^{36}$. Nuclei hold because the strong
interaction, at femtometre range, beats even this repulsion.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:2}
$E = 8.99 \times 10^9 \times 10^{-6}/1 = \qty{9.0}{kV/m}$. A downward field is
that of a negative charge: $Q = -4\pi\varepsilon_0R^2E = -1.11 \times 10^{-10} \times
(6.37 \times 10^6)^2 \times 100 = \qty{-4.5e5}{C}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:3}
On the axis, $|x| > a$: $E_x = \dfrac{q}{4\pi\varepsilon_0}\Bigl[\dfrac{1}{(x - a)^2} +
\dfrac{1}{(x + a)^2}\Bigr]$, outward. On the $y$ axis the $x$ components
cancel: $E_y = \dfrac{2qy}{4\pi\varepsilon_0(y^2 + a^2)^{3/2}}$; $\dd E_y/\dd y \propto
(y^2 + a^2) - 3y^2 = 0$ at $y = a/\sqrt2$. At the origin the field
vanishes; lines leaving each charge toward the other bend away along
$\pm y$ --- a saddle.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:4}
Between the charges both fields point toward $-q$: no zero. Left of
$+2q$, $2q/x^2 > q/(x + a)^2$ always. Beyond $-q$, at distance $d$ from
it: $2q/(a + d)^2 = q/d^2$, $a + d = \sqrt2\,d$, $d = a/(\sqrt2 - 1) = 2.4a$.
Two lines leave $+2q$ for each one ending on $-q$; the other half go to
infinity, where the pair looks like a charge $+q$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:5}
Ring of radius $a$, width $\dd a$: $\dd q = 2\pi\sigma a\,\dd a$, $\dd E_z =
\dfrac{\sigma z}{2\varepsilon_0}\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}}$; integrate:
$\dfrac{\sigma z}{2\varepsilon_0}\Bigl[-\dfrac{1}{\sqrt{a^2 + z^2}}\Bigr]_0^R = \dfrac{\sigma}{2\varepsilon_0}
\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$. $z \ll R$: $\sigma/2\varepsilon_0$, the
infinite plane. $z \gg R$: $z/\sqrt{z^2 + R^2} \approx 1 - R^2/2z^2$, so $E_z
\approx \sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, the point charge.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:6}
$Q = 79e = \qty{1.26e-17}{C}$; $E(R) = 8.99 \times 10^9 \times 1.26 \times 10^{-17}/
(7 \times 10^{-15})^2 = \qty{2.3e21}{V/m}$; at $R/2$, half: \qty{1.2e21}{V/m}; at
\qty{1}{pm}: $1.14 \times 10^{-7}/10^{-24} = \qty{1.1e17}{V/m}$ --- $10^{15}$ times
the breakdown field of air, at a distance where the electrons live.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:7}
Cylinder of radius $r$, length $h$: $2\pi rh\,E = \lambda h/\varepsilon_0$, $E =
\lambda/2\pi\varepsilon_0r$. Corona when $\lambda = 2\pi\varepsilon_0 rE = 2\pi \times 8.85 \times
10^{-12} \times 0.01 \times 3 \times 10^6 = \qty{1.7}{\micro C/m}$; at \qty{5}{m},
$E = 3 \times 10^6 \times 0.01/5 = \qty{6}{kV/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:8}
$E = \sigma/\varepsilon_0 = 10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$. A plate
exerts no net force on itself (its own field's contributions cancel
pairwise); the other plate's field at its location is $\sigma/2\varepsilon_0$, so
$F = Q\sigma/2\varepsilon_0 = 10^{-6} \times 10^{-6}/(2 \times 8.85 \times 10^{-12}) =
\qty{5.6e-2}{N}$, attractive; pressure $\sigma^2/2\varepsilon_0 = \qty{0.056}{Pa}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:9}
$r < a$: the inner conductor's charge sits on its surface, no charge
inside, $E = 0$. $a < r < b$: $E = \lambda/2\pi\varepsilon_0r$. $r > b$: enclosed
charge $\lambda - \lambda = 0$, $E = 0$ --- the cable radiates no static
field. At $r = a$: $2 \times 8.99 \times 10^9 \times 10^{-8}/(5 \times 10^{-4}) =
\qty{3.6e5}{V/m}$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:10}
$\rho = 3Q/4\pi(R_2^3 - R_1^3)$. $r < R_1$: $E = 0$. $R_1 < r < R_2$: $4\pi r^2E =
\rho\,\tfrac43\pi(r^3 - R_1^3)/\varepsilon_0$, so $E = \dfrac{Q}{4\pi\varepsilon_0r^2}\,
\dfrac{r^3 - R_1^3}{R_2^3 - R_1^3}$. $r > R_2$: $Q/4\pi\varepsilon_0r^2$. At $R_1$ both
sides give $0$; at $R_2$ both give $Q/4\pi\varepsilon_0R_2^2$. In the shell $E
\propto r - R_1^3/r^2$ increases, so the maximum is at $R_2$, after
which it falls as $1/r^2$. In the cavity the field is zero: a charge
there feels no force --- wherever it sits.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:11}
(a) Core $\tfrac43\pi(3.48 \times 10^6)^3 \times 11000 = \qty{1.94e24}{kg}$; mantle
$\tfrac43\pi[(6.37 \times 10^6)^3 - (3.48 \times 10^6)^3] \times 4500 = \qty{4.08e24}{kg}$;
total \qty{6.0e24}{kg}, within $1\%$. (b) Surface $GM/R^2 = \qty{9.9}{m/s^2}$;
boundary $6.67 \times 10^{-11} \times 1.94 \times 10^{24}/(3.48 \times 10^6)^2 =
\qty{10.7}{m/s^2}$. (c) $g = GM(r)/r^2$ with $\dd M = 4\pi r^2\rho_m\,\dd r$:
$\dd g/\dd r = 4\pi G\rho_m - 2GM(r)/r^3 = 4\pi G[\rho_m - \tfrac23\bar\rho(r)]$,
where $\bar\rho(r) = M(r)/\tfrac43\pi r^3$ is the mean density below $r$.
$g$ rises with depth ($\dd g/\dd r < 0$) iff $\rho_m < \tfrac23\bar\rho$. Just
below the surface $\bar\rho = 5500$, $\tfrac23\bar\rho = 3700 < 4500$: $g$
first dips slightly; at the core boundary $\bar\rho = 11000$, $7300 >
4500$: it rises there, to $10.7$.
\end{solution}

\begin{solution}{exo:b1:electrostatics-gauss:12}
(a) Inside, $E = er/4\pi\varepsilon_0R^3$ outward; on the electron $\vect F =
-\dfrac{e^2}{4\pi\varepsilon_0R^3}\,\vect r$: a spring. (b) $\omega = \sqrt{e^2/4\pi\varepsilon_0mR^3}
= \sqrt{2.31 \times 10^{-28}/(9.11 \times 10^{-31} \times 10^{-30})} = \qty{1.6e16}{rad/s}$,
$f = \qty{2.5e15}{Hz}$, $\lambda = c/f = \qty{120}{nm}$. (c) Startlingly close
to the \qty{122}{nm} line: the size and strength of the atom are right.
But the model gives one frequency, not a series, and the nucleus is
not a \qty{0.1}{nm} ball but a $10^{-15}$~m point --- and a classical
orbiting electron would radiate its energy away in nanoseconds. The
fix is \cref{ch:b1:quantum-introduction}.
\end{solution}

\begin{solution}{pb:b1:electrostatics-gauss:1}
\textbf{1.} $\vect F = -e\vect E$: upward. A downward field at the surface
is the field of a negative charge: the Earth is negative.

\textbf{2.} Sphere of radius $R_E$, $E_n = -100$: $Q = 4\pi\varepsilon_0R_E^2E_n =
1.11 \times 10^{-10} \times 4.06 \times 10^{13} \times (-100) = \qty{-4.5e5}{C}$.

\textbf{3.} $\sigma = \varepsilon_0E_n = \qty{-8.9e-10}{C/m^2}$; $8.9 \times 10^{-10}/1.6 \times
10^{-19} = 5.5 \times 10^9$ electrons per square metre, $5.5 \times 10^5$ per
square centimetre.

\textbf{4.} $F = 100e \times 100 = \qty{1.6e-15}{N}$; weight $10^{-9} \times 9.8 =
\qty{9.8e-9}{N}$: six million times larger.

\textbf{5.} Outward normals: top $E_n = -5$, bottom $E_n = +100$, sides
zero: $\Phi = \qty{95}{V.m}$, so $Q = 95\varepsilon_0 = \qty{+8.4e-10}{C}$ per square
metre --- positive space charge, almost cancelling the ground's
$-8.9 \times 10^{-10}$.

\textbf{6.} $\rho = 8.4 \times 10^{-10}/10^4 = \qty{8.4e-14}{C/m^3}$: $5 \times 10^5$
\omterm{def:b1:electrostatics-gauss:charge}{elementary charges} per cubic metre; against $10^9$ ions of each sign,
an imbalance of $5 \times 10^{-4}$.

\textbf{7.} $J = \gamma E_0 = \qty{2e-12}{A/m^2}$; $I = J \times 5.1 \times 10^{14} =
\qty{1.0}{kA}$; $Q/I = 4.5 \times 10^5/1000 = \qty{450}{s}$. The Earth would be
neutral in eight minutes: something recharges it continuously.

\textbf{8.} The axis lies in every plane containing it, all planes of
symmetry: $\vect E$ is axial. Each $\dd q$ gives $\dd q/4\pi\varepsilon_0(a^2 + z^2)$
along its own direction, of which the axial part is the fraction
$z/\sqrt{a^2 + z^2}$: $E_z = qz/4\pi\varepsilon_0(a^2 + z^2)^{3/2}$.

\textbf{9.} $\dd q = 2\pi\sigma a\,\dd a$: $E_z = \dfrac{\sigma z}{2\varepsilon_0}\displaystyle\int_0^R
\dfrac{a\,\dd a}{(a^2 + z^2)^{3/2}} = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2
+ R^2}}\Bigr)$.

\textbf{10.} $z \ll R$: $\sigma/2\varepsilon_0$, the infinite plane (Gauss's value);
$z \gg R$: $\sigma R^2/4\varepsilon_0z^2 = Q/4\pi\varepsilon_0z^2$, the point charge.

\textbf{11.} $\sigma = -40/(\pi \times 9 \times 10^6) = \qty{-1.4e-6}{C/m^2}$; $\sigma/2\varepsilon_0 =
\qty{8.0e4}{V/m}$; at $z = \qty{3}{km}$, $1 - 3/\sqrt{18} = 0.29$: $E =
\qty{2.3e4}{V/m}$, pointing \emph{up} toward the negative charge.

\textbf{12.} Upper disk, $z = \qty{9}{km}$: $8.0 \times 10^4 \times (1 - 9/\sqrt{90}) =
\qty{4.1e3}{V/m}$ downward. Net $\qty{1.9e4}{V/m}$ upward, about $4 \times 10^4$
with the ground's induced charges: the right order of the \qty{10}{kV/m}
measured (the real charge is more spread out and partly screened);
reversed and a hundred times the fair-weather field.

\textbf{13.} Whole cloud: net charge zero, \omterm{def:b1:electrostatics-gauss:flux}{flux} zero. Lower disk alone:
$-Q/\varepsilon_0 = -40/8.85 \times 10^{-12} = \qty{-4.5e12}{V.m}$.

\textbf{14.} Each disk is \qty{3}{km} away and gives $2.3 \times 10^4$; the
upper ($+$) pushes down, the lower ($-$) pulls down: \qty{4.7e4}{V/m}
downward --- thirty times below breakdown.

\textbf{15.} $Q/4\pi\varepsilon_0R^2 = 8.99 \times 10^9 \times 40/(2.5 \times 10^5) =
\qty{1.4e6}{V/m}$: only a charge packed into half a kilometre reaches
breakdown, and measured in-cloud fields are ten times too weak. Local
enhancement at drops and ice, or runaway electrons seeded by cosmic
rays, are the candidates --- how a flash starts is still debated.

\textbf{16.} $I = 5/(5 \times 10^{-5}) = \qty{1e5}{A}$.

\textbf{17.} $W = QEd = 5 \times 10^5 \times 3000 = \qty{1.5e9}{J}$ (\qty{420}{kW h});
$P = 1.5 \times 10^9/5 \times 10^{-5} = \qty{3e13}{W}$ --- one and a half times
the power of all humanity, for fifty microseconds.

\textbf{18.} Volume $\pi(0.02)^2 \times 3000 = \qty{3.8}{m^3}$, mass \qty{4.5}{kg};
$\Delta T = 1.5 \times 10^9/(4.5 \times 720) = \qty{4.6e5}{K}$ if nothing escaped.
The channel explodes outward (the shock is the thunder), radiates light
and radio waves, ionizes and dissociates the air: heating the final
channel to $3 \times 10^4$~K is a small part of the bill.

\textbf{19.} \qty{2.9}{s} per kilometre. Sound leaves every point of a
\qty{3}{km} channel at once, from distances that differ by kilometres: it
arrives spread over several seconds --- the rumble.

\textbf{20.} $50 \times 5 = \qty{250}{A}$ carried to the ground; with the
quieter storm currents (point discharge under clouds, charged rain) the
total reaches the kiloampere of Part~I: storms are the generators of a
global circuit whose return path is the fair-weather air.

\textbf{21.} $\Phi_{\text{out}}(\vect g) = -4\pi GM_{\text{in}}$: for a point mass
$g = -GM/r^2$ on a sphere of area $4\pi r^2$, \omterm{def:b1:electrostatics-gauss:flux}{flux} $-4\pi GM$, independent
of $r$; the rest follows as for Coulomb.

\textbf{22.} $g(r) = g_0r/R_E$; at $R_E/2$: \qty{4.9}{m/s^2}.

\textbf{23.} $M_{\text{core}} = \qty{1.94e24}{kg}$, $g = GM_{\text{core}}/R_c^2 =
\qty{10.7}{m/s^2}$ --- greater than at the surface: the mantle above is
lighter than the average of what is beneath.

\textbf{24.} $m\ddot x = -mg_0x/R_E$: $\omega = \sqrt{g_0/R_E} = \qty{1.24e-3}{rad/s}$,
$T = 2\pi/\omega = \qty{5060}{s} = \qty{84}{min}$, $v_{\max} = \omega R_E =
\qty{7.9}{km/s}$ --- the \omterm{thm:b1:central-forces:circular}{orbital speed} of a grazing satellite, whose
period is the same.

\textbf{25.} It replaced the sum over every charge of the Earth (I),
over the ions of the air (I), and over every shell of rock (IV) --- and
in~II it fixed the limits of the disk sum. Keep: \qty{-5e5}{C} and
\qty{1}{kA} for the fair-weather Earth; \qty{40}{C} and \qty{e4}{V/m} under a
storm; \qty{e5}{A} and \qty{e9}{J} per stroke; \qty{10.7}{m/s^2} at the core
boundary.
\end{solution}
