\begin{solution}{exo:b1:heat-engines:5}
Adiabats: $V_C = V_B(T_h/T_c)^{1/(\gamma-1)} = 3.0 \times (5/3)^{2.5} =

\begin{solution}{exo:b1:heat-engines:6}

\begin{solution}{exo:b1:heat-engines:8}
$Q_h = RT_h\ln2 = 8.314 \times 600 \times 0.693 = \qty{3.46}{kJ}$, $Q_c = -8.314

\begin{solution}{exo:b1:heat-engines:9}
Entropy of the two blocks: $\Delta S = C\ln(T_f/T_1) + C\ln(T_f/T_2) =
C\ln\bigl(T_f^2/T_1T_2\bigr) = S_{\mathrm{created}} \geq 0$ (the engine's own
entropy returns to its value each cycle), so $T_f \geq \sqrt{T_1T_2}$, and
$W = C(T_1 + T_2 - 2T_f)$ (first law) is largest for the smallest $T_f$:
the reversible case, $T_f = \sqrt{400 \times 300} = \qty{346.4}{K}$,
$W_{\max} = 4000 \times (700 - 692.8) = \qty{29}{kJ}$. Direct contact: $T_f =
\qty{350}{K}$, no work, $S_{\mathrm{created}} = 4000\ln(350^2/120000) =
\qty{82}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:heat-engines:10}
(a) $P = \dot Q_h - |\dot Q_c| = K(T_h - x) - K(y - T_c)$; the reversible core
exchanges no net entropy: $K(T_h - x)/x = K(y - T_c)/y$. (b) Cross-multiply:
$y(T_h - x) = x(y - T_c)$, so $y(T_h - 2x) = -xT_c$ and $y = xT_c/(2x - T_h)$.
Then $P = K(T_h - x)(1 - y/x) = K(T_h - x)(2x - T_h - T_c)/(2x - T_h)$; with
$u = 2x - T_h$, $T_h - x = (T_h - u)/2$ and $P = K(T_h - u)(u - T_c)/(2u) =
\tfrac{K}{2}[(T_h + T_c) - u - T_hT_c/u]$. (c) $\dd P/\dd u = \tfrac K2(-1 +
T_hT_c/u^2) = 0$ at $u = \sqrt{T_hT_c}$, a maximum; then $x = (T_h +
\sqrt{T_hT_c})/2$, $y = xT_c/u = (\sqrt{T_hT_c} + T_c)/2$, and $\eta^\ast = 1 - y/x
= 1 - \sqrt{T_c}(\sqrt{T_h} + \sqrt{T_c})/[\sqrt{T_h}(\sqrt{T_h} + \sqrt{T_c})] = 1 -
\sqrt{T_c/T_h}$; $P_{\max} = \tfrac K2(\sqrt{T_h} - \sqrt{T_c})^2$. (d)
$\eta^\ast = 1 - \sqrt{0.375} = 0.39$, against $\eta_C = 0.625$: the real $0.40$
of \cref{exo:b1:heat-engines:1} is an engine built for power, not for
\omterm{def:b1:heat-engines:efficiency}{efficiency}.
\end{solution}

\begin{solution}{exo:b1:heat-engines:11}
Cycle: $Q_h + Q_c + Q_0 = 0$ and $Q_h/T_h + Q_c/T_c + Q_0/T_0 \leq 0$.
Eliminate $Q_0 = -(Q_h + Q_c)$: $Q_h(1/T_h - 1/T_0) + Q_c(1/T_c - 1/T_0) \leq 0$,
i.e.\ $Q_c(T_0 - T_c)/(T_cT_0) \leq Q_h(T_h - T_0)/(T_hT_0)$, whence
$Q_c/Q_h \leq (1 - T_0/T_h)\,T_c/(T_0 - T_c)$. The first factor is the
\omterm{def:b1:heat-engines:efficiency}{efficiency} of a Carnot engine between burner and room, the second the
COP of a Carnot \omterm{def:b1:heat-engines:machine}{refrigerator} between chamber and room: the bound is what
one gets by coupling the two ideally. Numbers: $0.25 \times 11 = 2.75$
(real absorption fridges: about $0.6$).
\end{solution}

\begin{solution}{exo:b1:heat-engines:12}
(a) \qty{0.40}{J}. (b) $0.40 \times 3.5 = \qty{1.4}{J}$. (c) \qty{0.90}{J}.
(d) Engine $\eta = 1 - 293/800 = 0.634$, pump $e_p = 293/20 = 14.65$:

\end{solution}

\begin{solution}{pb:b1:heat-engines:1}
\textbf{1.} $P = 250 \times 25 = \qty{6.25}{kW}$.

\textbf{2.} Mean loss $250 \times 13 = \qty{3.25}{kW}$ for $200 \times 86400 =
\qty{1.73e7}{s}$: $E = \qty{5.6e10}{J} = \qty{15600}{kW h}$.

\textbf{3.} $15600 \times 0.25 = 3900$~\texteuro.

\textbf{4.} Gas $15600/0.90 = \qty{17300}{kW h}$: $1730$~\texteuro.

\textbf{5.} \omterm{def:b1:dc-circuits:resistor}{Resistors}: $15600/0.40 = \qty{39000}{kW h}$ of fuel; boiler
\qty{17300}{kW h}: $2.25$ times less. Electricity is work, the form that can
do anything; turning it into \qty{20}{\degreeCelsius} heat in a \omterm{def:b1:dc-circuits:resistor}{resistor}
discards the $60\%$ of the fuel already lost at the plant and the
possibility of pumping heat.

\textbf{6.} Over a cycle $W + Q_h + Q_c = 0$ and $Q_h/T_{\mathrm{in}} +
Q_c/T_{\mathrm{out}} \leq 0$ with $Q_h < 0$, $Q_c > 0$: $Q_c \leq
|Q_h|T_{\mathrm{out}}/T_{\mathrm{in}}$, so $W = |Q_h| - Q_c \geq |Q_h|(1 -
T_{\mathrm{out}}/T_{\mathrm{in}})$ and $e_p = |Q_h|/W \leq T_{\mathrm{in}}/(T_{\mathrm{in}} -
T_{\mathrm{out}})$.

\textbf{7.} $293/25 = 11.7$; $6.25/11.7 = \qty{0.53}{kW}$.

\textbf{8.} $293/13 = 22.5$; $15600/22.5 = \qty{690}{kW h}$ (about
$170$~\texteuro).

\textbf{9.} The work per joule lifted is $(T_{\mathrm{in}} - T_{\mathrm{out}})/
T_{\mathrm{in}}$, the ``height'' of the lift; the colder it is outside, the
higher each joule must be pumped --- and the more joules are needed, since
the loss is also proportional to $T_{\mathrm{in}} - T_{\mathrm{out}}$: the
electrical power grows as the square of the temperature difference, worst
exactly when heating matters most.

\textbf{10.} From the outdoor air, which is cooled: $Q_c = 6.25 - 0.53 =
\qty{5.7}{kW}$. The second law forbids heat flowing \emph{spontaneously}
from cold to hot; here work pays the lift. Check: entropy taken from the
air $5.72/268 = \qty{21.3}{W/K}$, entropy given to the house $6.25/293 =
\qty{21.3}{W/K}$ --- nothing destroyed, nothing created: the ideal pump.

\textbf{11.} Compressor (receives $W$), condenser (gives $|Q_h|$),
expansion valve (neither), evaporator (receives $Q_c$). Heat only flows
down a temperature slope: the \omterm{def:b1:heat-engines:vapourcycle}{refrigerant} must be colder than the air to
take heat from it and hotter than the house water to give heat to it.

\textbf{12.} $T_{\mathrm{ev}} = 268 - 8 = \qty{260}{K}$; $e_{p,C} = 313/(313 - 260)
= 5.9$.

\textbf{13.} $e_p = 0.55 \times 5.9 = 3.25$; $W = 6.25/3.25 = \qty{1.92}{kW}$;
$Q_c = 6.25 - 1.92 = \qty{4.33}{kW}$ from the air.

\textbf{14.} $\dot m = 4330/200000 = \qty{22}{g/s}$, \qty{1.3}{kg} a minute.

\textbf{15.} $T_{\mathrm{ev}} = \qty{272}{K}$, $313/41 = 7.6$, $e_p = 4.2$;

\textbf{16.} $e_p = 0.55 \times 313/(321 - T_{\mathrm{out}}) = 172/(321 -

\textbf{18.} Demand $250 \times 35 = \qty{8.75}{kW}$; $T_{\mathrm{ev}} = \qty{250}{K}$,

S_{\mathrm{created}} = 0$ gives $Q_c = T_{\mathrm{out}}|Q_h|/T_{\mathrm{in}} -

\textbf{24.} $T_{\mathrm{cd}} = \qty{303}{K}$, $T_{\mathrm{ev}} = \qty{264}{K}$:

\textbf{25.} $15600/3700 = 4.2$ kilowatt-hours of heat per kilowatt-hour
of electricity; $930$~\texteuro\ against $1730$ for gas; carbon: $3700
