\section*{Capítulo \ref{ch:b1:sinusoidal-impedance} --- Regime senoidal permanente e impedância}

\begin{solution}{exo:b1:sinusoidal-impedance:1}
$\underline U = 5\eu^{j\pi/3} = 2.5 + 4.33j$ (V). $3\cos\omega t +
4\sin\omega t$: $3 + 4\eu^{-j\pi/2} = 3 - 4j$, módulo $5$, argumento
$-\arctan(4/3) = \qty{-0.927}{rad}$: $5\cos(\omega t - 0.927)$.
\end{solution}

\qty{628}{\ohm} (\qty{1}{kHz}). $1/C\omega$: \qty{318}{\ohm}, \qty{15.9}{\ohm}.

\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:5}

\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:6}
$\underline Z = 100 + j(250 - 100) = 100 + 150j$: $Z = \qty{180}{\ohm}$,
$\arg = 56^\circ$. $I = 1.0/180 = \qty{5.6}{mA}$, atrasada em relação à fonte de
$56^\circ$. $U_R = \qty{0.56}{V}$, $U_L = \qty{1.39}{V}$, $U_C = \qty{0.56}{V}$;
\omterm{def:b1:sinusoidal-impedance:complex}{fasores}: $U_R$ ao longo de $I$, $U_L$ para cima, $U_C$ para baixo, resultante \qty{1.0}{V}.
$L\omega > 1/C\omega$: acima da ressonância (indutivo).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:7}
$f_0 = 1/(2\pi\sqrt{LC})$: \qty{1.59}{MHz} (\qty{50}{pF}) a \qty{0.50}{MHz}
(\qty{500}{pF}). Em \qty{1.00}{MHz}: $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$.
$\Delta f = \qty{10}{kHz}$. Estação em \qty{1.009}{MHz}: $x = 1.009$,
$Q(x - 1/x) = 1.8$, razão $1/\sqrt{1 + 3.2} = 0.49$. \omterm{def:b1:dc-circuits:voltage}{Tensão} no capacitor
$QE = \qty{1.0}{mV}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:8}
$Q(x - 1/x) = \pm1 \Leftrightarrow x^2 \mp x/Q - 1 = 0$; raízes positivas
$x_{1,2} = \mp 1/2Q + \sqrt{1 + 1/4Q^2}$. Diferença $1/Q$, logo
$\Delta\omega = \omega_0/Q$; produto $(1 + 1/4Q^2) - 1/4Q^2 = 1$, logo
$\omega_1\omega_2 = \omega_0^2$ (a banda é simétrica em escala logarítmica).
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:9}
$I = P/(U\cos\varphi) = 2000/(230 \times 0.70) = \qty{12.4}{A}$. \omterm{thm:b1:sinusoidal-impedance:power}{Potência
reativa} $P\tan\varphi = 2000 \times 1.02 = \qty{2.0}{kvar}$. $C = 2040/(314
\times 230^2) = \qty{123}{\micro F}$. Nova corrente $2000/230 = \qty{8.7}{A}$.
Perdas $0.50 \times 12.4^2 = \qty{77}{W}$ antes e $0.50 \times 8.7^2 =
\qty{38}{W}$ depois.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:10}
$\underline Y = 1/R + j(C\omega - 1/L\omega)$; $\underline U = \underline I/
\underline Y$ é máxima quando a susceptância se anula, em $\omega_0$,
onde $U = RI$. Meia potência: $\abs{C\omega - 1/L\omega} = 1/R$, isto é,\
$RC\omega_0(x - 1/x) = \pm1$: $\Delta\omega = \omega_0/(RC\omega_0) = 1/RC$
e $Q = RC\omega_0 = R\sqrt{C/L}$. Números: $\omega_0 = \qty{1.0e6}{rad/s}$
($f_0 = \qty{159}{kHz}$), $Q = 10^4\sqrt{10^{-6}} = 10$, $\Delta f =
\qty{16}{kHz}$.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:11}
$U_C = I/C\omega = E/[C\omega\sqrt{R^2 + (L\omega - 1/C\omega)^2}]$.
Multiplicando por $C\omega$ dentro da raiz, o denominador vira
$\sqrt{(RC\omega)^2 + (LC\omega^2 - 1)^2} =
\sqrt{x^2/Q^2 + (1 - x^2)^2}$. Com $s = x^2$: $(1 - s)^2 + s/Q^2$ é
mínimo em $s = 1 - 1/2Q^2$ (se positivo), com mínimo $1/Q^2 - 1/4Q^4$:
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$. Para $Q = 3.16$: $x_r = 0.975$,
$U_{C,\max} = 3.2E$. Para $Q < 1/\sqrt2$ o mínimo fica em $s = 0$: $U_C$
decresce monotonicamente a partir de $E$, sem pico.
\end{solution}

\begin{solution}{exo:b1:sinusoidal-impedance:12}
Na ressonância $i = I\cos\omega_0 t$ e $u_C = (I/C\omega_0)\sin\omega_0 t$,
de modo que, usando $1/C\omega_0^2 = L$,
\[
\tfrac12 Li^2 + \tfrac12 Cu_C^2 = \tfrac12 LI^2\cos^2\omega_0 t
+ \tfrac12 LI^2\sin^2\omega_0 t = \tfrac12 LI^2 .
\]
Dissipada por período: $\tfrac12 RI^2T$. Razão $\times 2\pi$: $2\pi L/(RT)
= L\omega_0/R = Q$.
\end{solution}

\begin{solution}{pb:b1:sinusoidal-impedance:1}
\textbf{1.} $\underline Z = R + j(L\omega - 1/C\omega)$.

\textbf{2.} $\underline I = E/\underline Z$; $I = E/\sqrt{R^2 + (L\omega -
1/C\omega)^2}$; $\varphi_i = -\arctan[(L\omega - 1/C\omega)/R]$.

\textbf{3.} A \omterm{def:b1:sinusoidal-impedance:impedance}{reatância} se anula em $\omega_0 = 1/\sqrt{LC}$: $I_{\max}
= E/R$, $\varphi_i = 0$.

\textbf{4.} $C = 1/(4\pi^2f_0^2L) = \qty{127}{pF}$; $L\omega_0 = 2\times10^{-4}
\times 6.28\times10^6 = \qty{1257}{\ohm}$.

\textbf{5.} $Q = 1257/12.6 = 100$.

\textbf{6.} $L\omega - 1/C\omega = L\omega_0(x - 1/x)$; divida por $R$.

\textbf{7.} $x < 1$: \omterm{def:b1:sinusoidal-impedance:impedance}{reatância} negativa, $\varphi_i > 0$, a corrente
adianta (capacitivo); $x > 1$: atrasa (indutivo).

\textbf{8.} $Q(x - 1/x) = \pm1$.

\textbf{9.} $x^2 \mp x/Q - 1 = 0$, $x_{1,2} = \mp1/2Q + \sqrt{1 + 1/4Q^2}$,
$x_2 - x_1 = 1/Q$: $\Delta\omega = \omega_0/Q$.

\textbf{10.} $\Delta f = \qty{10}{kHz}$.

\textbf{11.} \qty{1.010}{MHz}: $Q(x - 1/x) = 100 \times 0.0199 = 2.0$,
razão $1/\sqrt{5.0} = 0.45$, potência $0.20$: \qty{-7}{dB}. \qty{1.020}{MHz}:
$3.96$, razão $0.25$, potência $0.06$: \qty{-12}{dB}.

\textbf{12.} No limite: a vizinha está apenas \qty{7}{dB} abaixo. Um
$Q$ maior (um $R$ menor), ou vários estágios sintonizados em cascata, aguça a
seleção.

\textbf{13.} De \qty{0.50}{MHz} a \qty{1.59}{MHz}: a faixa AM.

\textbf{14.} $\underline U_C = \underline I/jC\omega$; em $\omega_0$,
$U_C = I_{\max}/C\omega_0 = E/(RC\omega_0) = QE$.

\textbf{15.} $I_{\max} = 10^{-5}/12.6 = \qty{0.79}{\micro A}$; $U_C = QE =
\qty{1.0}{mV}$: um ganho de \omterm{def:b1:dc-circuits:voltage}{tensão} de cem vezes, sem amplificador.

\textbf{16.} $\underline U_L = jL\omega_0\underline I = jQE$, $\underline U_C
= -jQE$: iguais e opostas, elas se cancelam; a fonte enxerga apenas $R$.

\textbf{17.} Armazenada $\tfrac12 LI_{\max}^2 = \qty{6.2e-17}{J}$; dissipada
por período $\tfrac12 RI_{\max}^2T = \qty{3.9e-18}{J}$; $2\pi \times 6.2/0.39
\approx 100 = Q$.

\textbf{18.} $\tau = 2Q/\omega_0 = \qty{32}{\micro s}$. O circuito não consegue
acompanhar variações mais rápidas que $\sim\tau$: uma \omterm{thm:b1:sinusoidal-impedance:resonance}{largura de banda} $\Delta f$ deixa passar
apenas modulações mais lentas que $\Delta f$; com $Q = 1000$, $\Delta f =
\qty{1}{kHz}$ e os agudos do programa se perderiam.

\textbf{19.} $Q$ cai à metade, $50$, $\Delta f$ dobra para \qty{20}{kHz} e $U_C$
cai à metade, \qty{0.5}{mV}.

\textbf{20.} $R_{\text{tot}} = \qty{62.6}{\ohm}$: $Q = 1257/62.6 = 20$,
$\Delta f = \qty{50}{kHz}$, $U_C = 20 \times \qty{10}{\micro V} = \qty{0.2}{mV}$:
sensibilidade e seletividade desabam juntas.

\textbf{21.} $P = E^2/(2R_{\text{tot}}) = 10^{-10}/125 = \qty{8e-13}{W}$;
fração em $R_a$: $50/62.6 = 80\%$.

\textbf{22.} O casamento pede uma resistência de carga igual a $R_a$, isto é,\ uma
resistência total grande; a seletividade pede a menor resistência total
possível ($Q = L\omega_0/R_{\text{tot}}$). Não se pode ter as duas com uma
ligação direta.

\textbf{23.} Uma derivação da bobina ou um pequeno capacitor de acoplamento apresentam à antena
apenas uma fração do circuito ressonante: a resistência da antena
é ``vista'' pelo circuito muito reduzida, o que mantém $Q$ alto, enquanto a
antena continua entregando potência perto do seu ótimo.

\textbf{24.} Nas frequências de meia potência a \omterm{def:b1:sinusoidal-impedance:impedance}{reatância} vale $\pm R$,
de modo que $\varphi = \pm45^\circ$ e $\cos\varphi = 1/\sqrt2$; como também $I =
I_{\max}/\sqrt2$, a potência $P = \tfrac12 EI\cos\varphi$ vale metade do seu valor
na ressonância.

\textbf{25.} $Q$ multiplica o sinal por $Q$ e estreita a banda para
$f_0/Q$, ao custo de uma resposta mais lenta ($\tau = 2Q/\omega_0$) e de um
acoplamento frágil com a antena; para canais de \qty{10}{kHz} em
\qty{1}{MHz}, $Q \approx 100$ é o número --- e a antena precisa ser
acoplada de leve para preservá-lo.
\end{solution}
