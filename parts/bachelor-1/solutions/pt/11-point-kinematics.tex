\section*{Capítulo \ref{ch:b1:point-kinematics} --- Cinemática do ponto}

\begin{solution}{exo:b1:point-kinematics:1}
$v = 4.0t$: \qty{12}{m/s} em \qty{3.0}{s}; $a = \qty{4.0}{m/s^2}$;
$x(5) = \qty{50}{m}$; $\qty{100}{km/h} = \qty{27.8}{m/s}$ em $t = 27.8/4.0
= \qty{6.9}{s}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:2}
$\dot r = 0$, $\dot\theta = \qty{3.0}{rad/s}$: $\vect v = r\dot\theta\,\vect e_\theta
= 6.0\,\vect e_\theta$ (\unit{m/s}); $\vect a = -r\dot\theta^2\vect e_r =
-18\,\vect e_r$ (\unit{m/s^2}). \omterm{cor:b1:point-kinematics:circular}{Movimento circular} uniforme: \omterm{def:b1:point-kinematics:velocity}{velocidade} escalar constante,
\omterm{cor:b1:point-kinematics:circular}{aceleração centrípeta}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:3}
$v = 2\pi R/T = 2\pi \times 6.37\times10^6/86164 = \qty{465}{m/s}$;
$a = v^2/R = \qty{0.034}{m/s^2}$, ou seja, $g/290$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:4}
$v_0\sin\alpha = \qty{7.71}{m/s}$, $v_0\cos\alpha = \qty{9.19}{m/s}$.
Ápice em $t = 7.71/9.81 = \qty{0.79}{s}$, altura $7.71^2/(2 \times 9.81) =
\qty{3.0}{m}$. Alcance $2 \times 9.19 \times 0.786 = \qty{14.5}{m}$.
Na chegada: $(9.19, -7.71)$, \qty{12}{m/s} a $40^\circ$ abaixo da
horizontal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:5}
$a_T = \qty{1.0}{m/s^2}$, $a_N = 10^2/25 = \qty{4.0}{m/s^2}$, $a =
\qty{4.1}{m/s^2}$. Após \qty{5}{s}: $v = \qty{15}{m/s}$, $a_N = 225/25 =
\qty{9.0}{m/s^2}$, $a = \qty{9.1}{m/s^2}$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:6}
$r = b\omega t$, $\dot r = b\omega$, $\ddot r = 0$, $\dot\theta = \omega$:
$\vect v = b\omega\,\vect e_r + b\omega^2 t\,\vect e_\theta$, $v = b\omega
\sqrt{1 + \omega^2t^2}$; $\vect a = -b\omega^3 t\,\vect e_r + 2b\omega^2\,
\vect e_\theta$. O termo $2\dot r\dot\theta$ tem duas metades iguais: a
direção da \omterm{def:b1:point-kinematics:velocity}{velocidade} radial gira com taxa $\dot\theta$ (dando
$\dot r\dot\theta\,\vect e_\theta$), e afastar-se do centro aumenta a
\omterm{def:b1:point-kinematics:velocity}{velocidade} ortorradial $r\dot\theta$ (mais um $\dot r\dot\theta$).
\end{solution}

\begin{solution}{exo:b1:point-kinematics:7}
$v = \sqrt{(2.0 \times 1.0)^2 + 0.5^2} = \qty{2.1}{m/s}$; $a = R\omega^2 =
\qty{2.0}{m/s^2}$ rumo ao eixo; passo $v_z \times 2\pi/\omega =
\qty{3.1}{m}$; $\rho = v^2/a = 4.25/2.0 = \qty{2.1}{m} > R$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:8}
$v_{\max} = A\omega = \qty{1.0}{m/s}$; $a_{\max} = A\omega^2 = \qty{10}{m/s^2}$;
$v$ adianta $x$ em um quarto de período, e $a = -\omega^2x$ é oposta a $x$.
$\abs v$ é máximo em $x = 0$, e $\abs a$ nos extremos $x = \pm A$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:9}
$\dot{\vect e}_r = \dot\theta(-\sin\theta\,\vect e_x + \cos\theta\,\vect e_y)
= \dot\theta\,\vect e_\theta$; $\dot{\vect e}_\theta = \dot\theta(-\cos\theta\,
\vect e_x - \sin\theta\,\vect e_y) = -\dot\theta\,\vect e_r$. Então
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$, e derivar
mais uma vez com a regra do produto dá a \omterm{def:b1:point-kinematics:velocity}{aceleração} do
\cref{thm:b1:point-kinematics:cylindrical}.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:10}
Ápice: $v = v_0\cos\alpha$, $a_N = g$: $\rho = v_0^2\cos^2\alpha/g$.
Lançamento: $v = v_0$, $a_N = g\cos\alpha$: $\rho = v_0^2/(g\cos\alpha)$.
O raio no ápice é menor por um fator $\cos^3\alpha$: a parábola se curva mais
onde é mais lenta e onde todo o $\vect g$ é normal.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:11}
$\dot x = v - v\cos(vt/R)$, $\dot y = v\sin(vt/R)$; $\ddot x = (v^2/R)
\sin(vt/R)$, $\ddot y = (v^2/R)\cos(vt/R)$. No solo ($vt/R = 2\pi n$):
$\dot x = \dot y = 0$ e $\vect a = (0, v^2/R)$, para cima. No topo
($vt/R = \pi$): $\dot x = 2v$, $\dot y = 0$: \omterm{def:b1:point-kinematics:velocity}{velocidade} escalar $2v$.
\end{solution}

\begin{solution}{exo:b1:point-kinematics:12}
$\ddot r = 0$, $\dot r = u$, $\dot\theta = \omega$: $\vect a = -ut\omega^2\,
\vect e_r + 2u\omega\,\vect e_\theta$. Em $t = \qty{2.0}{s}$: $-2.0\,\vect e_r
+ 2.0\,\vect e_\theta$ (\unit{m/s^2}). O primeiro é o termo centrípeto
da rotação; o segundo, $2\dot r\dot\theta$, existe apenas porque a
própria direção radial gira --- no solo, caminhar em linha reta com
\omterm{def:b1:point-kinematics:velocity}{velocidade} constante não dá \omterm{def:b1:point-kinematics:velocity}{aceleração} alguma.
\end{solution}

\begin{solution}{pb:b1:point-kinematics:1}
\textbf{1.} $\vect v = R\dot\theta\,\vect e_\theta$; $\vect a = -R\dot\theta^2
\,\vect e_r + R\ddot\theta\,\vect e_\theta$.

\textbf{2.} $\vect T = \vect e_\theta$ (movimento com $\theta$ crescente),
$\vect N = -\vect e_r$ (rumo a $C$), $\rho = R$.

\textbf{3.} Alto: $v^2 = 625 - 392 = 233$, $v = \qty{15.3}{m/s}$. Lados:
$625 - 196 = 429$, $v = \qty{20.7}{m/s}$.

\textbf{4.} $2v\dot v = -2gR\sin\theta\,\dot\theta$ e $v = R\dot\theta$:
$\dot v = -g\sin\theta$. Negativa para $0 < \theta < \pi$ (perde \omterm{def:b1:point-kinematics:velocity}{velocidade} na
subida) e positiva na descida.

\textbf{5.} $a_N = v^2/R = v_0^2/R - 2g(1 - \cos\theta)$: embaixo
\qty{62.5}{m/s^2}; lados $62.5 - 19.6 = \qty{42.9}{m/s^2}$; alto $62.5 -
39.2 = \qty{23.3}{m/s^2}$.

\textbf{6.} Embaixo: \qty{62.5}{m/s^2} (sem parte tangencial); lados:
$\sqrt{42.9^2 + 9.81^2} = \qty{44.0}{m/s^2}$; alto: \qty{23.3}{m/s^2},
apontando verticalmente para baixo, rumo a $C$: ângulo $0$ com a vertical.

\textbf{7.} Perpendicular embaixo e no alto ($a_T = 0$); nunca
paralelo ($a_N > 0$ em toda parte).

\textbf{8.} $\dot\theta = v/R = \sqrt{v_0^2 - 2gR(1 - \cos\theta)}/R$:
embaixo \qty{2.50}{rad/s}, no alto \qty{1.53}{rad/s}.

\textbf{9.} $\ddot\theta = \dot v/R = -(g/R)\sin\theta$: a equação de um
pêndulo simples de comprimento $R$ --- o carro na sua pista é um pêndulo
que passa por cima do ponto de suspensão.

\textbf{10.} $2\pi\sqrt{R/g} = 2\pi\sqrt{10/9.81} = \qty{6.3}{s}$.

\textbf{11.} Média de $25$, $20.7$, $15.3$, $20.7$: cerca de \qty{20}{m/s};
comprimento $2\pi R = \qty{63}{m}$: aproximadamente \qty{3}{s}.

\textbf{12.} $\sqrt{gR} = \qty{9.9}{m/s} < 15.3$: sim. Mínimo: $v_0^2 \geq
gR + 4gR = 5gR = 490$, $v_0 \geq \qty{22.1}{m/s}$.

\textbf{13.} Embaixo: $\vect a = 62.5$ para cima, $\vect g$ para baixo:
$\abs{\vect a - \vect g} = 62.5 + 9.8 = \qty{72.3}{m/s^2} = 7.4g$.

\textbf{14.} No alto: ambos para baixo, $23.3 - 9.8 = \qty{13.5}{m/s^2} = 1.4g$,
pressionando o passageiro contra o assento (rumo ao centro, isto é,\ para cima
para o passageiro de cabeça para baixo).

\textbf{15.} Lados: $\vect a = 42.9$ horizontal rumo a $C$ mais $9.8$
para baixo (tangencial); $\vect a - \vect g$ vale $42.9$ na horizontal: $4.4g$
lateralmente, rumo ao centro.

\textbf{16.} Reprovado: $7.4g$ embaixo (o alto e os lados estão bem).

\textbf{17.} Peso aparente nulo no alto: $v_{\text{alto}}^2 = gR$;
então $v_0^2 = gR + 4gR = 5gR$, $a_N = 5g$ embaixo e a carga vale
$5g + g = 6g$, independentemente de $R$.

\textbf{18.} $a_N = v^2/\rho$: $v$ é fixado pela altura (energia), e $\rho$
pela geometria da pista; um $\rho$ diferente na mesma altura muda
$a_N$ sozinho.

\textbf{19.} $a_N + g = 3g$: $a_N = 2g$, $\rho = v_0^2/2g = 625/19.6 =
\qty{32}{m}$.

\textbf{20.} $v_{\text{alto}} = \qty{15.3}{m/s}$; $a_N - g = 1.5g$: $a_N =
\qty{24.5}{m/s^2}$, $\rho = 233/24.5 = \qty{9.5}{m}$.

\textbf{21.} $a_N = v^2/\rho = v^2s/A^2 = v^3t/A^2$: ela cresce linearmente
a partir de zero --- uma taxa constante de variação da \omterm{def:b1:point-kinematics:velocity}{aceleração} --- em vez de
saltar de $0$ a $v^2/R$ na entrada de um círculo, o que o pescoço
sente como um golpe.

\textbf{22.} Uma gota: larga e de curvatura suave embaixo, onde o
carro é rápido, e apertada no alto, onde ele é lento. Ela mantém a carga perto de
$3g$ ao longo de todo o percurso e a impõe de modo gradual.

\textbf{23.} Linha de visada: distância horizontal \qty{50}{m}, altura
\qty{20}{m}: $r = \qty{53.9}{m}$, elevação $\beta = 21.8^\circ$. A
componente transversal da \omterm{def:b1:point-kinematics:velocity}{velocidade} horizontal vale $v\sin\beta = 15 \times
0.371 = \qty{5.6}{m/s}$; $\dot\theta = 5.6/53.9 = \qty{0.10}{rad/s}$.

\textbf{24.} Visto de um ponto fora do centro, a taxa angular da linha
de visada não é constante (rápida quando o carro está perto, lenta quando está longe):
uma \omterm{def:b1:optical-instruments:camera}{câmera} de taxa constante se desvia do carro.

\textbf{25.} \omterm{def:b1:point-kinematics:cylindrical}{Coordenadas polares} em $C$ para a \omterm{def:b1:point-kinematics:velocity}{velocidade} e a \omterm{def:b1:point-kinematics:velocity}{aceleração} no
círculo; a decomposição de Frenet para ``acelerar'' contra ``girar'' e
para a clotoide; \omterm{def:b1:point-kinematics:cylindrical}{coordenadas polares} na \omterm{def:b1:optical-instruments:camera}{câmera} para a taxa de
acompanhamento; a energia (próximos capítulos) para a lei da \omterm{def:b1:point-kinematics:velocity}{velocidade}.
\end{solution}
