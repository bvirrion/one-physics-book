\section*{Capítulo \ref{ch:b1:mirrors-thin-lenses} --- Espelhos e lentes delgadas}

\begin{solution}{exo:b1:mirrors-thin-lenses:1}
$\overline{SF} = \qty{-20}{cm}$; $1/\overline{SA'} = 1/\overline{SF} -
1/\overline{SA}$. A \qty{60}{cm}: $\overline{SA'} = \qty{-30}{cm}$,
real, $\gamma = -0.5$. A \qty{30}{cm}: $\overline{SA'} = \qty{-60}{cm}$,
real, $\gamma = -2$. A \qty{10}{cm}: $1/\overline{SA'} = -1/20 + 1/10
= 1/20$, $\overline{SA'} = \qty{+20}{cm}$ atrás do espelho, virtual,
$\gamma = +2$ (direita, ampliada).
\end{solution}

$\overline{OA} = \qty{-3.0}{cm}$: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$,

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:4}
$\overline{FA} = \qty{-15}{cm}$: $\overline{F'A'} = -f'^2/\overline{FA}
= +100/15 = \qty{6.7}{cm}$ depois de $F'$, isto é,\ $\overline{OA'} = \qty{16.7}{cm}$;
$\gamma = f'/\overline{FA} = -0.67$. Descartes: $\overline{OA} =
\qty{-25}{cm}$, $1/\overline{OA'} = 1/10 - 1/25 = 3/50$,

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:5}
$\overline{SF} = \qty{+1.0}{m}$, $\overline{SA} = \qty{-20}{m}$:
$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ atrás
do espelho (virtual, direita); $\gamma = -0.95/(-20) = +0.048$, altura
da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} \qty{7.1}{cm}. O cérebro julga a distância pelo tamanho aparente: um

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:6}
$1/\overline{OA'} = 1/50 - 1/2000 = \qty{0.0195}{mm^{-1}}$,
$\overline{OA'} = \qty{51.3}{mm}$: a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} se afasta \qty{1.3}{mm} do
sensor. $\gamma = 51.3/(-2000) = -0.026$. Para $\abs\gamma = 24/1800

\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:7}
$\gamma = -1.8/0.036 = -50$; $\overline{OA'} = \qty{5.0}{m}$,
$\overline{OA} = \overline{OA'}/\gamma = \qty{-10}{cm}$;
$1/f' = 1/5.0 + 1/0.10 = \qty{10.2}{m^{-1}}$, $f' = \qty{98}{mm}$. A
\omterm{def:b1:mirrors-thin-lenses:image}{imagem} é invertida ($\gamma < 0$), de cima para baixo e da esquerda para a direita: o slide
entra girado de $180^\circ$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:8}
Objeto no plano focal $\Rightarrow$ os raios de cada ponto deixam a
\omterm{def:b1:mirrors-thin-lenses:lens}{lente} paralelos; o \omterm{prop:b1:rays-reflection-refraction:mirror}{espelho plano} os devolve paralelos (direção
simétrica); a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} focaliza um feixe paralelo no seu plano focal, no
ponto simétrico do ponto-fonte em relação ao eixo: \omterm{def:b1:mirrors-thin-lenses:image}{imagem} no plano do
objeto, invertida, $\gamma = -1$. Inclinar o espelho de $\alpha$ gira o
feixe de volta de $2\alpha$ e desloca a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} lateralmente de
$2\alpha f'$, ainda nítida no mesmo plano.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:9}
$f' = (100^2 - 40^2)/(4 \times 100) = \qty{21}{cm}$. Menor distância
$4f' = \qty{84}{cm}$, com $\gamma = -1$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:10}
$L_1$ conjuga o infinito em $F_1'$, \qty{20}{cm} depois de $L_1$, isto é,\ \qty{5}{cm}
depois de $L_2$: um objeto virtual, $\overline{O_2A_1} = \qty{+5}{cm}$.
$1/\overline{O_2A'} = -1/10 + 1/5 = 1/10$: \omterm{def:b1:mirrors-thin-lenses:image}{imagem} final \qty{10}{cm}
depois de $L_2$. Altura da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} intermediária $h_1 = f_1'\alpha = 20\alpha$ (cm);
$\gamma_2 = 10/5 = 2$, logo $h = 40\alpha$: \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} equivalente
\qty{40}{cm}, num barrilete de $15 + 10 = \qty{25}{cm}$ --- uma teleobjetiva.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:11}
$\gamma = -3 = -\overline{SA'}/\overline{SA}$ dá $\overline{SA'} =
3\overline{SA}$; ambos reais, à frente: $\overline{SA} = -x$,
$\overline{SA'} = -3x$, e a distância anteparo--lâmpada $2x = \qty{80}{cm}$:
$x = \qty{40}{cm}$. Espelho a \qty{40}{cm} da lâmpada e a \qty{120}{cm}
do anteparo; $1/\overline{SF} = -1/40 - 1/120 = -1/30$:
$R = 2 \times 30 = \qty{60}{cm}$.
\end{solution}

\begin{solution}{exo:b1:mirrors-thin-lenses:12}
$\theta = 0.53^\circ = \qty{9.3e-3}{rad}$; diâmetro da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} $f'\theta =
\qty{4.6}{mm}$. Potência coletada $10^3 \times \pi(0.05)^2 = \qty{7.9}{W}$;

\end{solution}

\begin{solution}{pb:b1:mirrors-thin-lenses:1}
\textbf{1.} $1/\overline{OA'} - 1/\overline{OA} = 1/f'$,
$\gamma = \overline{OA'}/\overline{OA}$; eixo orientado no sentido da luz,
distâncias algébricas a partir de $O$.

\textbf{2.} $p = -\overline{OA}$, $\overline{OA'} = D - p$:
$1/(D - p) + 1/p = 1/f_1'$, isto é,\ $p^2 - Dp + Df_1' = 0$; raízes reais se e só se
$D^2 - 4Df_1' \geq 0$, isto é,\ $D \geq 4f_1'$.

\textbf{3.} As raízes somam $D$: $p_1 + p_2 = D$, logo $p_2 = D - p_1$ ---
as distâncias do objeto e da \omterm{def:b1:mirrors-thin-lenses:image}{imagem} se trocam e as posições são simétricas em relação ao
ponto médio. $d = p_2 - p_1 = \sqrt{D^2 - 4Df_1'}$, donde $f_1' = (D^2 -
d^2)/4D$.

\textbf{4.} $f_1' = (6400 - 1600)/320 = \qty{15.0}{cm}$.

\textbf{5.} Só entram o deslocamento $d$ do barrilete entre os dois
ajustes nítidos e a distância objeto--anteparo $D$: nenhum dos dois
exige saber onde $O$ fica dentro do barrilete.

\textbf{6.} $\partial f_1'/\partial D = 1/4 + d^2/4D^2 = 0.3125$,
$\partial f_1'/\partial d = -d/2D = -0.25$; logo
$u(f_1')^2 = (0.3125 \times 0.2)^2 + (0.25 \times 0.5)^2 =
\qty{0.0195}{cm^2}$ e $u(f_1') = \qty{0.14}{cm}$:
$f_1' = \qty{15.00\pm0.14}{cm}$.

\textbf{7.} $p = (80 \mp 40)/2 = 20$ ou \qty{60}{cm}: $\gamma =
-60/20 = -3$ (slide perto da \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, ampliado: o projetor) e
$\gamma = -1/3$.

\textbf{8.} $D_{\min} = 4f_1' = \qty{60.0}{cm}$, $\gamma = -1$.

\textbf{9.} $\gamma = -1.80/0.036 = -50$.

\textbf{10.} $\overline{OA'} = f_1'(1 - \gamma) = 15 \times 51 =
\qty{765}{cm} = \qty{7.65}{m}$; $\overline{OA} = \overline{OA'}/\gamma =
\qty{-15.3}{cm}$.

\textbf{11.} $\gamma < 0$: a \omterm{def:b1:mirrors-thin-lenses:image}{imagem real} é girada de $180^\circ$
em torno do eixo, de modo que o slide é inserido girado de $180^\circ$ (de cabeça
para baixo e invertido da esquerda para a direita).

\textbf{12.} $\overline{OA'} = \qty{1000}{cm}$: $1/\overline{OA} = 1/1000
- 1/15$, $\overline{OA} = \qty{-15.23}{cm}$: a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} se desloca \qty{0.7}{mm}
em direção ao slide. $\gamma = 1000/(-15.23) = -65.7$: \omterm{def:b1:mirrors-thin-lenses:image}{imagem} de
\qty{2.36}{m} de largura.

\textbf{13.} Derivando Descartes: $-\dd\overline{OA'}/\overline{OA'}^2
+ \dd\overline{OA}/\overline{OA}^2 = 0$, logo $\dd\overline{OA'} =
(\overline{OA'}/\overline{OA})^2\,\dd\overline{OA} = \gamma^2\,\dd\overline{OA}$.

\textbf{14.} $2500 \times \qty{0.10}{mm} = \qty{25}{cm}$: a \omterm{def:b1:mirrors-thin-lenses:image}{imagem}
sai de \omterm{def:b1:mirrors-thin-lenses:mirror}{foco} à medida que o filme esquenta --- daí os slides montados em vidro ou
um autofoco que acompanha o filme.

\textbf{15.} Triângulos semelhantes da abertura ao \omterm{def:b1:mirrors-thin-lenses:image}{ponto-imagem}:
borrão $= 30 \times 0.20/7.65 = \qty{0.78}{mm}$; de \qty{5}{m} isso
subtende $\qty{1.6e-4}{rad} < \qty{3e-4}{rad}$: não é percebido.

\textbf{16.} Área do slide $8.64\times10^{-4}\ \unit{m^2}$: $E_{\text{slide}}
= 2.0/8.64\times10^{-4} = \qty{2.3e3}{W/m^2}$; a área da tela é $\gamma^2$
vezes maior, $\qty{2.16}{m^2}$: $E_{\text{tela}} = \qty{0.93}{W/m^2}$;
mesma potência, áreas na razão $\gamma^2 = 2500$.

\textbf{17.} $\overline{O_2A'} = \qty{1200}{cm}$: $1/\overline{O_2A} =
1/1200 + 1/30 = 0.03417$, $\overline{O_2A} = \qty{+29.3}{cm}$: a
\omterm{def:b1:mirrors-thin-lenses:image}{imagem} intermediária fica \qty{29.3}{cm} \emph{além} de $L_2$ --- um objeto
virtual para $L_2$ (os raios vindos de $L_1$ convergem para ela).

\textbf{18.} $\overline{O_1A_1'} = 8.0 + 29.3 = \qty{37.3}{cm}$;
$1/\overline{O_1A} = 1/37.3 - 1/15 = -0.0398$, $\overline{O_1A} =
\qty{-25.1}{cm}$.

\textbf{19.} $\gamma_1 = 37.3/(-25.1) = -1.49$; $\gamma_2 = 1200/29.3 =
41.0$; $\gamma = -60.9$.

\textbf{20.} Largura $60.9 \times 36 = \qty{2.19}{m}$. $L_1$ sozinha:
$\gamma = 1 - \overline{OA'}/f_1' = 1 - 1208/15 = -79.5$, largura
\qty{2.86}{m}.

\textbf{21.} Feixe a $\alpha$: $L_1$ o focaliza à altura $h_1 = f_1'\alpha$
no seu plano focal, que dista $\overline{O_2A_1} = f_1' - e$ de $L_2$;
$L_2$ conjuga esse ponto em $\overline{O_2A'}$ com $1/\overline{O_2A'}
= 1/f_2' + 1/(f_1' - e)$ e o amplia por $\gamma_2 = \overline{O_2A'}/(f_1'
- e)$, de modo que $h = \gamma_2 f_1'\alpha$ e $f_{\mathrm{eq}} = \gamma_2 f_1'
= \dfrac{f_1' f_2'}{f_1' + f_2' - e}$, cujo inverso é a fórmula
enunciada. $1/f_{\mathrm{eq}} = 1/15 - 1/30 + 8/450 = \qty{0.0511}{cm^{-1}}$,
$f_{\mathrm{eq}} = \qty{19.6}{cm}$.

\textbf{22.} $2.19/2.86 = 0.77$ e $15/19.6 = 0.77$. Apenas aproximadamente,
porque o slide está a distância finita e as duas projeções são medidas
a partir de \omterm{def:b1:mirrors-thin-lenses:lens}{lentes} diferentes; para um objeto no infinito a razão seria
exata.

\textbf{23.} Objeto virtual em $p > 0$ além de uma \omterm{def:b1:mirrors-thin-lenses:lens}{lente divergente}:
$1/\overline{O_2A'} = 1/p + 1/f_2' = 1/p - 1/\abs{f_2'}$, positivo (\omterm{def:b1:mirrors-thin-lenses:image}{imagem
real}) se e só se $p < \abs{f_2'}$.

\textbf{24.} Mesma potência sobre uma \omterm{def:b1:mirrors-thin-lenses:image}{imagem} de área $(2.19/2.86)^2 = 0.59$
vezes menor: irradiância $\times 1.7$.

\textbf{25.} O conversor multiplica a \omterm{def:b1:mirrors-thin-lenses:lens}{distância focal} por
$f_{\mathrm{eq}}/f_1' = 1.3$; a uma dada distância de projeção a \omterm{def:b1:mirrors-thin-lenses:image}{imagem} fica $1.3$
vezes menor em cada direção e $1.7$ vezes mais brilhante --- uma
objetiva de longo alcance.
\end{solution}
