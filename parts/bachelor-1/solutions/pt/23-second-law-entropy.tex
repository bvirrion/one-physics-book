\section*{Capítulo \ref{ch:b1:second-law-entropy} --- A segunda lei: a entropia}

\begin{solution}{exo:b1:second-law-entropy:1}
$\Delta S = mL/T$: fusão $334000/273 = \qty{1.22}{kJ/K}$; ebulição $2260000/
373 = \qty{6.06}{kJ/K}$. A vaporização liberta completamente as moléculas
umas das outras --- muito mais desordem microscópica e muito mais calor por
\omterm{def:b1:kinetic-theory:temperature}{kelvin}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:2}
$\Delta S = mc\ln(T_2/T_1) = 4180\ln(353/293) = \qty{0.78}{kJ/K}$ --- uma função
de estado: independente do método (a entropia \emph{criada} não é).
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:3}
$\Delta S = nR\ln2 = \qty{5.8}{J/K}$. Isotérmica reversível: $Q = nRT\ln2$
vindo do banho a $T$, $S_{\mathrm{exch}} = nR\ln2$, criada $0$. Joule:
$Q = 0$, $S_{\mathrm{exch}} = 0$, criada \qty{5.8}{J/K}.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:4}
$S = nC_{V,m}\ln T + nR\ln V$ constante: $T^{C_{V,m}}V^R$ constante, isto é,
$TV^{R/C_{V,m}} = TV^{\gamma - 1}$ constante. Compressão súbita: $Q = 0$, logo
$S_{\mathrm{exch}} = 0$ e $\Delta S = S_{\mathrm{created}}$; com $T$: $300 \to
643$ K, $V$: $24.9 \to 10.7$ L: $\Delta S = 20.8\ln(643/300) + 8.314\ln(10.7/
24.9) = 15.9 - 7.0 = \qty{+8.9}{J/K}$ criados.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:5}
Bloco: $450\ln(293/373) = \qty{-109}{J/K}$; lago: $+36000/293 = \qty{+123}{J/K}$;
universo: $+\qty{14}{J/K}$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:6}
$T_f = \qty{350}{K}$; $S = 4180[\ln(350/400) + \ln(350/300)] = 4180 \times
0.0206 = \qty{86}{J/K}$, tudo criado. $(T_1 + T_2)^2 - 4T_1T_2 = (T_1 - T_2)^2
\geq 0$, de modo que o argumento do logaritmo é $\geq 1$.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:7}
$S = Q(1/T_c - 1/T_h)$: $1000(1/280 - 1/300) = \qty{0.24}{J/K}$; da fornalha
para a sala: $1000(1/300 - 1/1000) = \qty{2.3}{J/K}$, dez vezes mais: quanto
maior a diferença de temperatura que um fluxo de calor atravessa, mais
irreversível --- é na queda da fornalha à sala que a energia nobre se desperdiça.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:8}
Ciclo: $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$ com $S_{\mathrm{created}} \geq 0$,
logo $Q \leq 0$ e $W = -Q \geq 0$: a máquina não pode fornecer trabalho. O
mar é um único termostato: o seu calor não pode virar trabalho sem um
corpo mais frio onde despejar entropia.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:9}
Cada gás se expande de $V$ a $2V$ à mesma $T$: $\Delta S = 2 \times R\ln2
= \qty{11.5}{J/K}$, criados (sem troca de calor, pois a caixa é isolada). Duas
metades de nitrogênio: retirar a parede não muda nada macroscópico ---
$\Delta S = 0$ (paradoxo de Gibbs: moléculas idênticas não se ``misturam'').
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:10}
$Q = \qty{3.6e5}{J}$ para a sala a \qty{293}{K}: $S = \qty{1230}{J/K}$
criados --- no filamento (trabalho elétrico virado calor) e no escoamento
desse calor do filamento quente para a sala fria; o próprio trabalho
elétrico não carregava entropia.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:11}
$\Delta S_{\text{corpo}} = C\ln(T_0/T_1)$ em todos os casos. (a) trocada
$C(T_0 - T_1)/T_0$: criada $C[T_1/T_0 - 1 - \ln(T_1/T_0)]$. (b) duas etapas
com razão $r = \sqrt{T_1/T_0}$ cada: criada $2C[r - 1 - \ln r]$, menor
(para $T_1/T_0 = 2$: $0.193C$ contra $0.307C$). (c) infinitas etapas:
cada uma cria $C[(1 + \epsilon) - 1 - \ln(1 + \epsilon)] \approx C\epsilon^2/2$,
cuja soma tende a zero quando $\epsilon \to 0$: um resfriamento reversível exige um termostato
a cada temperatura intermediária --- o calor nunca pode atravessar uma diferença finita.
\end{solution}

\begin{solution}{exo:b1:second-law-entropy:12}
$2^{-10} = 10^{-3}$; $2^{-100} = \num{8e-31}$; $2^{-10^{22}} = 10^{-3\times10^{21}}$.
$k_B\ln2^N = Nk_B\ln2 = nR\ln2$. ``Impossível'' significa, a rigor,
probabilidade nula; aqui ela vale $10^{-3\times10^{21}}$ --- não é zero, mas é um
número tão pequeno que nenhuma diferença poderá jamais ser observada: a
irreversibilidade é estatística na escala de $10^{23}$.
\end{solution}

\begin{solution}{pb:b1:second-law-entropy:1}
\textbf{1.} $Q = mc(T_0 - T_1) = 0.25 \times 4180 \times (-70) = \qty{-73}{kJ}$
para o café: a sala ganha \qty{73}{kJ}, $\Delta S_{\text{sala}} = 73150/293
= \qty{+250}{J/K}$.

\textbf{2.} $mc\ln(T_0/T_1) = 1045\ln(293/363) = \qty{-224}{J/K}$.

\textbf{3.} $S_{\mathrm{created}} = -224 - (-250) = \qty{+26}{J/K} > 0$.

\textbf{4.} $\Delta S = 1045\ln(330/363) = \qty{-99.6}{J/K}$; $Q = 1045 \times
(-33) = \qty{-34.5}{kJ}$, $S_{\mathrm{exch}} = \qty{-117.7}{J/K}$: criados
\qty{18}{J/K} até aqui --- a maior parte da irreversibilidade acontece
enquanto o café está mais quente.

\textbf{5.} $\Delta S - Q/T_0 = mc[\ln(T_0/T_1) - (T_0 - T_1)/T_0] = mc[x - 1 -
\ln x]$ com $x = T_1/T_0$; $f'(x) = 1 - 1/x$ se anula em $x = 1$, onde
$f = 0$, e $f'' = 1/x^2 > 0$: mínimo $0$.

\textbf{6.} Não: os estados inicial e final são os mesmos, o calor é o
mesmo, a sala é o mesmo termostato --- a entropia criada está fixada pelos
estados, e não pela velocidade.

\textbf{7.} $T_f = \qty{350}{K}$; criados $4180\ln(350^2/120000) = \qty{86}{J/K}$.

\textbf{8.} Reversível: entropia total conservada: $mc\ln(T_f'/T_1) + mc\ln(T_f'/
T_2) = 0$, logo $T_f'^2 = T_1T_2$.

\textbf{9.} $T_f' = \sqrt{120000} = \qty{346.4}{K}$; as águas perdem $mc[(T_1 -
T_f') + (T_2 - T_f')] = 4180 \times (53.6 - 46.4) = \qty{30}{kJ}$, fornecidos como
trabalho.

\textbf{10.} Quente: $4180\ln(346.4/400) = \qty{-601}{J/K}$; fria: $4180\ln(346.4/
300) = \qty{+601}{J/K}$: soma nula, como a reversibilidade exige.

\textbf{11.} $T_0S_{\mathrm{created}} = 300 \times 86 = \qty{26}{kJ}$, perto dos
\qty{30}{kJ} recuperáveis: a entropia criada, vezes a temperatura de
referência, é o trabalho que o caminho irreversível jogou fora.

\textbf{12.} Seria preciso uma máquina funcionando entre duas águas mornas
cujas temperaturas mudam sem parar --- algumas dezenas de quilojoules para
uma máquina grande: não compensa, mas a segunda lei diz que o trabalho
estava ali.

\textbf{13.} Recebida do interior $Q_c/T_c = 40/278 = \qty{0.144}{W/K}$;
cedida à sala $Q_0/T_0 = (40 + W)/293$; balanço ao longo de um ciclo:
$0 = 0.144 - (40 + W)/293 + S_c$.

\textbf{14.} $S_c \geq 0$: $(40 + W)/293 \geq 0.144$, $W \geq 42.2 - 40 =
\qty{2.2}{W}$; COP $\leq 40/2.2 = 18$ ($= T_c/(T_0 - T_c)$).

\textbf{15.} $\mathrm{COP}_{\max} = T_c/(T_0 - T_c)$: para um congelador $255/38 = 6.7$
--- quanto mais frio o compartimento, mais caro cada joule retirado.

\textbf{16.} Com $W = 15$: $S_c = (55/293) - 0.144 = \qty{0.044}{W/K}$ criados
a cada segundo.

\textbf{17.} $W \geq 200 \times 15/278 = \qty{10.8}{W}$; a máquina despeja
\qty{200}{W} mais o seu próprio trabalho na sala enquanto o interior fica
frio: o ganho líquido da sala é $W$ --- uma geladeira aberta é um aquecedor.

\textbf{18.} $W = 0$ daria $S_c = Q_c(1/T_0 - 1/T_c) < 0$: proibido ---
é o \omterm{prop:b1:second-law-entropy:clausiuskelvin}{enunciado de Clausius}.

\textbf{19.} Antes: $1$ (todas à esquerda); depois: $2^N$; $\Delta S = k_B\ln2^N =
Nk_B\ln2$.

\textbf{20.} $Nk_B = nR$: $\Delta S = nR\ln2$, o resultado da \omterm{rem:b1:first-law:irreversible}{expansão de Joule}.

\textbf{21.} Um bit por molécula: $N$ bits; e $\Delta S/(k_B\ln2) = N$ ---
a entropia conta a informação de que se precisaria para fixar a posição
das moléculas.

\textbf{22.} $2^{-50} = \num{9e-16}$; for a mole $2^{-6\times10^{23}} =
10^{-1.8\times10^{23}}$.

\textbf{23.} Nenhum calor ($Q = 0$): trocada $0$, criada $nR\ln2$ --- o caso
mais puro: sem trabalho, sem calor, só o espalhamento do gás.

\textbf{24.} $W = nRT\ln2$ recebidos, $Q = -nRT\ln2$ cedidos à sala,
$S_{\mathrm{exch}} = -nR\ln2$ (a entropia do gás volta ao seu valor
inicial, reversivelmente): desfazer a expansão livre custa o trabalho $nRT\ln2$
e despeja $nR\ln2$ de entropia na sala --- o universo fica com os
$nR\ln2$ criados de uma vez por todas.

\textbf{25.} A entropia não se conserva em nada de real --- só no limite
reversível idealizado; a lei proíbe todo processo que a destruiria (calor
subindo, trabalho a partir de um só termostato, separação gratuita de uma
mistura); e cada porção criada é trabalho perdido, $T_0S_{\mathrm{created}}$ dele.
\end{solution}
