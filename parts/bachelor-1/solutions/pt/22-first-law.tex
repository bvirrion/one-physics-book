\section*{Capítulo \ref{ch:b1:first-law} --- A primeira lei da termodinâmica}

\begin{solution}{exo:b1:first-law:1}
$Q = C_{P,m}\Delta T = 29.1 \times 100 = \qty{2.91}{kJ}$; $W = -nR\Delta T =
\qty{-0.83}{kJ}$; $\Delta U = Q + W = \qty{2.08}{kJ} = C_{V,m}\Delta T$.
\end{solution}

\begin{solution}{exo:b1:first-law:2}
$W = -nRT\ln(V_2/V_1) = -2494\ln(0.25) = \qty{+3.46}{kJ}$; $\Delta U = 0$, logo
$Q = \qty{-3.46}{kJ}$ (cedido ao banho térmico).
\end{solution}

\begin{solution}{exo:b1:first-law:3}
$T_2 = T_1 \times 10^{\gamma - 1} = 300 \times 10^{0.4} = \qty{754}{K}$; $P_2 =
10^{1.4} = \qty{25}{bar}$.
\end{solution}

\begin{solution}{exo:b1:first-law:4}
Argônio: $C_{V,m} = \tfrac32 R = 12.5$, $C_{P,m} = \tfrac52 R = 20.8$ J/(\omterm{def:b1:kinetic-theory:scales}{mol} K),
$\gamma = 1.67$; nitrogênio: $20.8$, $29.1$, $\gamma = 1.40$. Para a água, a
dilatação com o aquecimento é ínfima, de modo que o trabalho $P\,\dd V$ a pressão constante
é desprezível diante do \omterm{def:b1:first-law:heat}{calor}: $c_P - c_V \approx 0$.
\end{solution}

\begin{solution}{exo:b1:first-law:5}
$T_f = (200 \times 80 + 300 \times 20)/500 = \qty{44}{\degreeCelsius}$. Com o
copo: $(200 \times 4.18 \times 80 + (300 \times 4.18 + 80) \times 20)/(500 \times
4.18 + 80) = \qty{43}{\degreeCelsius}$.
\end{solution}

\begin{solution}{exo:b1:first-law:6}
Gelo: aquecer até $0$: $0.050 \times 2100 \times 10 = \qty{1.05}{kJ}$; derreter:
\qty{16.7}{kJ}; total \qty{17.8}{kJ}. A água pode dar \qty{25.1}{kJ} até
$0$: tudo derrete; então $0.250 \times 4180\,T_f = 25.1 - 17.8$ kJ: $T_f =
\qty{7}{\degreeCelsius}$. Com \qty{150}{g}: seriam precisos \qty{53.3}{kJ} $>$ \qty{25.1}{kJ}:
só $(25.1 - 3.15)/334 = \qty{66}{g}$ de gelo derretem, $T_f = \qty{0}{\degreeCelsius}$
e restam \qty{84}{g} de gelo.
\end{solution}

\begin{solution}{exo:b1:first-law:7}
$W = -P_1(V_2 - V_1) + 0 - P_2(V_1 - V_2) + 0 = (P_2 - P_1)(V_2 - V_1) > 0$:
o ciclo é percorrido no sentido anti-horário (expansão a baixa pressão,
compressão a alta), de modo que o gás \emph{recebe} trabalho líquido, igual à
área encerrada; no sentido horário ele o forneceria --- uma máquina térmica.
\end{solution}

\begin{solution}{exo:b1:first-law:8}
$T_2 = 300 \times 7^{0.286} = \qty{523}{K}$. O ar quente e comprimido aquece o
corpo da bomba por condução a cada curso. No pneu o ar esfria até a
temperatura ambiente a volume fixo: $P \propto T$, e a pressão cai na razão
das temperaturas --- complete o enchimento.
\end{solution}

\begin{solution}{exo:b1:first-law:9}
$W = 0$ (não há pressão externa contra a qual empurrar), $Q = 0$: $\Delta U = 0$,
donde $\Delta T = 0$ para um \omterm{def:b1:kinetic-theory:model}{gás perfeito}. Irreversível: o gás nunca volta
sozinho. Um gás real, cujas moléculas se atraem, esfria um pouco (parte da
\omterm{thm:b1:work-and-energy:ke}{energia cinética} vira \omterm{def:b1:work-and-energy:conservative}{energia potencial} quando elas se afastam).
\end{solution}

\begin{solution}{exo:b1:first-law:10}
$Q = L_v = \qty{2.26}{MJ}$; $W = -P(V_g - V_l) = -1.013\times10^5 \times 1.67 =
\qty{-0.17}{MJ}$; $\Delta U = \qty{2.09}{MJ}$: $93\%$ do \omterm{def:b1:first-law:heat}{calor} vai para
separar as moléculas (\omterm{def:b1:kinetic-theory:U}{energia interna}) e $7\%$ para empurrar a
atmosfera.
\end{solution}

\begin{solution}{exo:b1:first-law:11}
$W = -P_2(V_2 - V_1)$, $\Delta U = nC_{V,m}(T_2 - T_1)$, $V_2 = nRT_2/P_2$, $V_1
= nRT_1/P_1$: $\tfrac52(T_2 - 300) = -T_2 + 300 \times 5$, $3.5T_2 = 2250$,
$T_2 = \qty{643}{K}$, $V_2 = \qty{10.7}{L}$. Reversível: $T_2 = 300 \times
5^{0.286} = \qty{475}{K}$, $V_2 = \qty{7.9}{L}$. A compressão súbita realiza
mais trabalho sobre o gás (os \qty{5}{bar} inteiros agem desde o início) e
termina mais quente e menos comprimida.
\end{solution}

\begin{solution}{exo:b1:first-law:12}
$\rho = PM/RT = \qty{1.20}{kg/m^3}$: $\sqrt{\gamma P/\rho} = \sqrt{1.4 \times
1.013\times10^5/1.20} = \qty{343}{m/s}$; $\sqrt{P/\rho} = \qty{290}{m/s}$, $15\%$
abaixo. Com $P/\rho = RT/M$: $c = \sqrt{\gamma RT/M}$; contra $u = \sqrt{3RT/M}$
= \qty{502}{m/s}: $c/u = \sqrt{\gamma/3} = 0.68$.
\end{solution}

\begin{solution}{pb:b1:first-law:1}
\textbf{1.} $n = P_1V_1/RT_1 = 10^5 \times 2.12\times10^{-4}/2494 = \qty{8.5e-3}{mol}$.

\textbf{2.} $PV^\gamma$ constante: $V_2 = V_1(P_1/P_2)^{1/\gamma} = 212 \times
7^{-0.714} = \qty{53}{cm^3}$.

\textbf{3.} $T_2 = T_1(P_2/P_1)^{(\gamma - 1)/\gamma} = 300 \times 7^{0.286} =
\qty{523}{K}$.

\textbf{4.} $W = nC_{V,m}(T_2 - T_1) = 8.5\times10^{-3} \times 20.8 \times 223 =
\qty{39}{J}$.

\textbf{5.} Ar a acrescentar: $\Delta n = \Delta P\,V/RT = 6\times10^5 \times 2\times10^{-3}/
2494 = \qty{0.48}{mol}$: $0.48/8.5\times10^{-3} = 57$ cursos, cerca de
\qty{2.2}{kJ} --- um minuto de esforço honesto.

\textbf{6.} A volume constante $P \propto T$: a pressão cai enquanto o ar
esfria (até $300/523$ para o ar do último curso); o ciclista dá mais alguns
cursos depois de uma pausa.

\textbf{7.} $W_{\text{iso}} = nRT\ln(P_2/P_1) = 8.5\times10^{-3} \times 2494 \times
1.95 = \qty{41}{J} > \qty{39}{J}$: o caminho isotérmico comprime o gás
ainda mais (até \qty{30}{cm^3} em vez de \qty{53}{cm^3}) para chegar a \qty{7}{bar},
porque ele fica frio --- mais volume varrido, mais trabalho.

\textbf{8.} $n = 10^5 \times 5\times10^{-4}/2494 = \qty{0.020}{mol}$; $T_2 =
300 \times 20^{0.4} = \qty{994}{K}$.

\textbf{9.} $P_2 = 20^{1.4} = \qty{66}{bar}$.

\textbf{10.} \qty{994}{K} está bem acima do ponto de autoignição do
combustível: o combustível injetado acende ao contato. Um motor a gasolina
comprime a sua mistura ar--combustível só até uns \qty{750}{K}, para que
ela não acenda antes da faísca; uma taxa alta demais provoca ``detonação''.

\textbf{11.} $W = nC_{V,m}(T_2 - T_1) = 0.020 \times 20.8 \times 694 = \qty{289}{J}$.

\textbf{12.} $25 \times 289 = \qty{7.2}{kW}$ por cilindro, armazenados no gás
quente e devolvidos na expansão.

\textbf{13.} A irreversibilidade (o gás não segue o caminho quase-estático,
e um trabalho extra é dissipado) eleva $T_2$; o \omterm{def:b1:first-law:heat}{calor} perdido para as
paredes o abaixa. Num motor real o segundo efeito costuma vencer: $T_2$
fica um pouco abaixo do valor ideal.

\textbf{14.} $Q = 20\times10^{-6} \times 43\times10^6 = \qty{860}{J}$.

\textbf{15.} Isobárica: $Q = nC_{P,m}(T_3 - T_2)$: $T_3 - T_2 = 860/(0.020 \times
29.1) = \qty{1480}{K}$, $T_3 = \qty{2470}{K}$; $V_3 = V_2T_3/T_2 = 25 \times
2.48 = \qty{62}{cm^3}$.

\textbf{16.} $W = -P_2(V_3 - V_2) = -66\times10^5 \times 37\times10^{-6} =
\qty{-244}{J}$: trabalho fornecido pelo gás.

\textbf{17.} $T_4 = T_3(V_3/V_1)^{\gamma - 1} = 2470 \times (0.124)^{0.4} =
\qty{1070}{K}$; $W = nC_{V,m}(T_4 - T_3) = 0.020 \times 20.8 \times (-1400) =
\qty{-582}{J}$.

\textbf{18.} Trabalho líquido fornecido $= 244 + 582 - 289 = \qty{537}{J}$; razão
$537/860 = 62\%$ (o rendimento ideal do ciclo Diesel nestes ajustes; os
motores reais chegam a uns $40\%$).

\textbf{19.} $1 - 300/2470 = 88\%$: o ciclo Diesel ideal fica bem abaixo do
limite de Carnot, porque recebe o seu \omterm{def:b1:first-law:heat}{calor} ao longo de um intervalo de
temperaturas, e não todo na mais alta.

\textbf{20.} Ao longo de um ciclo $\Delta U = 0$: $Q_{\text{saída}} = -(860 - 537) =
\qty{-323}{J}$, isto é, \qty{323}{J} saem com o escapamento; verificação:
$nC_{V,m}(T_4 - T_1) = 0.020 \times 20.8 \times 770 = \qty{320}{J}$.

\textbf{21.} A oscilação é rápida (um segundo) comparada com a condução de
\omterm{def:b1:first-law:heat}{calor} através de \qty{10}{L} de ar: adiabática. Linearizando
$PV^\gamma$: $\dd P/P = -\gamma\,\dd V/V = -\gamma Sx/V$.

\textbf{22.} Força sobre a esfera $S\,\dd P = -\gamma PS^2x/V$: $m\ddot x =
-(\gamma PS^2/V)x$, harmônica, $\omega_0^2 = \gamma PS^2/mV$.

\textbf{23.} $\omega_0^2 = 1.4 \times 10^5 \times 4\times10^{-8}/(0.0165 \times 0.010)
= \qty{33.9}{s^{-2}}$: $T = 2\pi/5.83 = \qty{1.08}{s}$. Com $T = \qty{1.10}{s}$:
$\gamma = 4\pi^2mV/(T^2PS^2) = 4\pi^2 \times 0.0165 \times 0.010/(1.21 \times 10^5
\times 4\times10^{-8}) = 1.35$.

\textbf{24.} O atrito amortece o movimento e, se for seco, desloca o
equilíbrio; o vazamento ao redor da esfera deixa o ar escapar durante uma
compressão, baixando a força de restituição: os dois alongam o período e
enviesam $\gamma$ para baixo --- como o $1.35$ acima.

\textbf{25.} Bomba e diesel: $Q = 0$, logo $\Delta U = W$ --- o trabalho virou
\omterm{def:b1:kinetic-theory:U}{energia interna} e temperatura, pela \omterm{prop:b1:first-law:transformations}{lei de Laplace} com expoente
$\gamma$. Esfera: uma compressão rápida é adiabática, e $\gamma$ fixa a
rigidez de uma mola de ar.
\end{solution}
