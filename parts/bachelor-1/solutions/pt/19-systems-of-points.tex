\section*{Capítulo \ref{ch:b1:systems-of-points} --- Sistemas de pontos; introdução aos corpos rígidos}

\begin{solution}{exo:b1:systems-of-points:1}
$x_G = (2.0 \times 0 + 3.0 \times 1.0)/5.0 = \qty{0.60}{m}$. Terra--Lua:
$384000/82 = \qty{4700}{km}$ a partir do centro da Terra, dentro da Terra
(\qty{6370}{km}): a Terra oscila em torno de um ponto \qty{1700}{km} abaixo
da sua superfície.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:2}
$4.0v = 0.010 \times 800$: $v = \qty{2.0}{m/s}$. Bala $\tfrac12 \times 0.010
\times 800^2 = \qty{3.2}{kJ}$, fuzil \qty{8}{J}: vêm da energia química da
pólvora; o corpo leve fica com quase tudo ($E_k = p^2/2m$
com $p$ igual).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:3}
$v = 20000/2500 = \qty{8.0}{m/s}$; $E_k$: \qty{200}{kJ} antes, $\tfrac12 \times
2500 \times 64 = \qty{80}{kJ}$ depois: $60\%$ perdidos.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:4}
Extremidade: $\int_0^L x^2(M/L)\dd x = ML^2/3$; centro: $\int_{-L/2}^{L/2}x^2(M/L)
\dd x = ML^2/12$; Huygens com $d = L/2$: $ML^2/12 + ML^2/4 = ML^2/3$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:5}
$E_k = \tfrac12 Mv^2(1 + k)$: cilindro $1.5$, aro $2$ vezes $\tfrac12 Mv^2$.
À mesma velocidade, mais energia para o aro: ele rola mais longe num piso plano
(mesmo atrito) e sobe mais alto ($h = v^2(1 + k)/2g$).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:6}
HCl: $\mu = 35m_H/36 = 0.972m_H$; átomo: $\mu = m_e/(1 + 1/1836) =
0.99946m_e$. As frequências são $\propto 1/\sqrt\mu$: $+1.4\%$ para o HCl, $+0.027\%$
para o hidrogênio (o desvio isotópico entre os espectros do H e do D repousa nisso).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:7}
$J = ML^2/3$, $d = L/2$: $T = 2\pi\sqrt{(ML^2/3)/(MgL/2)} = 2\pi\sqrt{2L/3g}
= \qty{1.64}{s}$; comprimento equivalente $2L/3 = \qty{0.67}{m}$. Um golpe no
centro de percussão põe a barra a girar em torno do pivô sem nenhuma reação
impulsiva ali: acerte a bola no ``ponto doce'' do bastão e as mãos não
sentem o repuxo.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:8}
$J = \tfrac12 MR^2 = \qty{1.0e-3}{kg.m^2}$; $C = 4\pi^2J/T^2 = \qty{9.9e-3}{N.m/rad}$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:9}
$\omega = \qty{314}{rad/s}$, $E = \tfrac12 J\omega^2 = \qty{4.9e5}{J}$;
$\Gamma = J\omega/t = \qty{314}{N.m}$; voltas: $\tfrac12\omega t/2\pi = 250$.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:10}
$a = g\sin\alpha/(1 + k)$: esfera ($k = 2/5$) $0.714$, cilindro ($1/2$)
$0.667$, aro ($1$) $0.500$ vezes $g\sin\alpha$: a esfera primeiro, o aro por
último; nem massa nem raio aparecem.
\end{solution}

\begin{solution}{exo:b1:systems-of-points:11}
Quantidade de movimento: $0.010 \times 400 = 2.01\,v$: $v = \qty{2.0}{m/s}$; depois, a energia:
$h = v^2/2g = \qty{0.20}{m}$. $E_k$: \qty{800}{J} antes, $\tfrac12 \times
2.01 \times 4.0 = \qty{4.0}{J}$ depois: $99.5\%$ perdidos em deformação e
calor. O impacto é inelástico --- a energia não se conserva nele, a
quantidade de movimento sim (a tração do fio é vertical e o impacto é breve).
\end{solution}

\begin{solution}{exo:b1:systems-of-points:12}
$m_1v_1 = m_1v_1' + m_2v_2'$ e $m_1v_1^2 = m_1v_1'^2 + m_2v_2'^2$ dão
$v_1 + v_1' = v_2'$ (divida a equação da energia pela da quantidade de movimento),
donde as fórmulas. Fração da energia cedida ao alvo $4m_1m_2/(m_1 + m_2)^2$:
próton $100\%$; carbono $4 \times 12/169 = 28\%$; chumbo $4 \times 207/208^2 =
1.9\%$. Os núcleos leves tomam a energia do nêutron em poucas colisões
(água, grafite); o chumbo precisaria de centenas.
\end{solution}

\begin{solution}{pb:b1:systems-of-points:1}
\textbf{1.} No eixo, no centro do disco, por simetria.

\textbf{2.} Koenig: $E_k = \tfrac12 Mv_G^2 + E_k^*$, e no \omterm{def:b1:systems-of-points:barycentric}{referencial
baricêntrico} o ioiô gira em torno do seu eixo: $E_k^* = \tfrac12 J\omega^2$.

\textbf{3.} Anel de raio $\rho$, de massa $2M\rho\,\dd\rho/R^2$: $J = \int_0^R
\rho^2\cdot 2M\rho\,\dd\rho/R^2 = MR^2/2 = 0.5 \times 0.050 \times 9\times10^{-4}
= \qty{2.25e-5}{kg.m^2}$.

\textbf{4.} O cordão está fixo e se desenrola do eixo: o ponto do eixo em
contato com ele está momentaneamente em repouso, logo $v_G = r\omega$.

\textbf{5.} $E_k = \tfrac12 Mv_G^2 + \tfrac12(\tfrac12 MR^2)(v_G/r)^2 = \tfrac12
Mv_G^2(1 + R^2/2r^2)$; $R^2/2r^2 = 9/(2 \times 0.16) = 28$: fator $29$ ---
quase toda a energia está na rotação.

\textbf{6.} Peso $Mg$ para baixo, tração $T$ para cima: $Ma_G = Mg - T$.

\textbf{7.} Só a tração tem momento em relação ao eixo (\omterm{def:b1:angular-momentum:moment}{braço de alavanca} $r$):
$J\dot\omega = Tr$.

\textbf{8.} $\dot\omega = a_G/r$: $T = Ja_G/r^2 = (R^2/2r^2)Ma_G$; então
$Ma_G = Mg - (R^2/2r^2)Ma_G$, $a_G = g/29 = \qty{0.34}{m/s^2}$.

\textbf{9.} $T = M(g - a_G) = 0.050 \times 9.47 = \qty{0.47}{N}$, $97\%$ do
peso: o cordão quase o segura.

\textbf{10.} $t = \sqrt{2h/a_G} = \qty{2.4}{s}$; $v_G = a_Gt = \qty{0.82}{m/s}$.
Uma pedra: \qty{0.45}{s} e \qty{4.4}{m/s}.

\textbf{11.} $\omega = v_G/r = \qty{205}{rad/s} = \qty{1960}{rpm}$.

\textbf{12.} $\tfrac12 Mv_G^2 = \qty{0.017}{J}$ contra $\tfrac12 J\omega^2 =
\qty{0.47}{J}$: $28$ vezes mais energia na rotação do que na queda.

\textbf{13.} $Mgh = \qty{0.49}{J}$; $\tfrac12 Mv_G^2 \times 29 = 0.5 \times 0.050
\times 0.67 \times 29 = \qty{0.49}{J}$. A tração age no ponto do cordão que
está em repouso (o cordão não se move): potência nula.

\textbf{14.} A força que para o ioiô age ao longo do cordão, passando pelo
eixo: nenhum momento em torno dele, e o \omterm{def:b1:angular-momentum:L}{momento angular} $J\omega$ fica intacto.
Só uma força tangencial (atrito no eixo) poderia frear a rotação.

\textbf{15.} $T \approx Mv_G/\Delta t + Mg = 0.050 \times 0.82/0.005 + 0.49 =
\qty{8.7}{N}$, dezoito vezes o peso --- daí o tranco sentido no
dedo.

\textbf{16.} $J\dot\omega = -\Gamma$: $t = J\omega/\Gamma = 2.25\times10^{-5} \times
205/10^{-4} = \qty{46}{s}$.

\textbf{17.} Toda a energia de rotação $\tfrac12 J\omega^2 = \qty{0.47}{J}$ se
converte em $Mgh'$: $h' = \qty{0.96}{m}$, quase toda a altura; menos na
prática, por causa do atrito no eixo durante o sono e das perdas no
tranco.

\textbf{18.} $a_G$ aponta para baixo (o ioiô desacelera na subida) enquanto
$v_G$ aponta para cima: $T = M(g - a_G) < Mg$ --- menor do que o peso.

\textbf{19.} $r = R$: fator $1 + 1/2 = 1.5$, $a_G = 2g/3 = \qty{6.5}{m/s^2}$
--- queda rápida, pouca rotação: é o eixo fino que faz um ioiô.

\textbf{20.} \omterm{ex:b1:systems-of-points:rolling}{Rolamento sem deslizamento}: o \omterm{def:b1:point-kinematics:frame}{ponto material} em $C$ tem
velocidade nula; o campo de velocidades de um \omterm{def:b1:systems-of-points:solid}{corpo rígido} com um ponto em
repouso é uma rotação em torno desse ponto (eixo por $C$, mesmo $\omega$).

\textbf{21.} $F$ age por baixo do eixo, a uma altura $R - r$ acima da
mesa, horizontalmente na direção da mão: momento $F(R - r)$ em relação a $C$
no sentido que rola o carretel para a mão; o peso e a reação passam por $C$
(sem momento).

\textbf{22.} $(J + MR^2)\dot\omega = F(R - r)$, $a_G = R\dot\omega = FR(R - r)/
(\tfrac32 MR^2) = 2F(R - r)/3MR > 0$: na direção da mão.

\textbf{23.} A linha de ação de $F$ passa por $C$ quando o fio que sai de
baixo do eixo encontra a mesa em $C$: $\cos\beta = r/R$,
$\beta = 82^\circ$. Mais inclinado: o momento em relação a $C$ se inverte e o carretel
se afasta (e, para $\beta$ grande o bastante, levanta).

\textbf{24.} $a_G = 2 \times 0.20 \times 0.026/(3 \times 0.050 \times 0.030) =
\qty{2.3}{m/s^2}$; o atrito sai de $Ma_G = F - f$ (o atrito se opõe a $F$):
$f = 0.20 - 0.050 \times 2.3 = \qty{0.085}{N}$; ele deve ficar abaixo de $\mu_sMg
\approx \qty{0.2}{N}$ (para $\mu_s = 0.4$), isto é, $F \lesssim \qty{0.5}{N}$.

\textbf{25.} \omterm{def:b1:systems-of-points:cm}{Centro de massa}: a translação de $G$ e a tração; \omterm{def:b1:angular-momentum:L}{momento
angular} em relação a um eixo: a rotação e o sentido do rolamento;
energia: a velocidade lá embaixo e a altura da subida.
\end{solution}
