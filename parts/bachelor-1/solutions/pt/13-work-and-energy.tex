\section*{Capítulo \ref{ch:b1:work-and-energy} --- Trabalho, energia e energia potencial}

\begin{solution}{exo:b1:work-and-energy:1}
Força que ergue $+mgh = 20 \times 9.81 \times 3.0 = \qty{589}{J}$; peso
$-\qty{589}{J}$; potência $589/5.0 = \qty{118}{W}$. Atrito: $-\mu_dmgd =
-0.30 \times 196 \times 3.0 = \qty{-177}{J}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:2}
$E_k = \tfrac12 \times 1000 \times 27.8^2 = \qty{386}{kJ}$; $d = E_k/F =
\qty{55}{m}$; dobrar $v$ quadruplica $E_k$ e $d$: \qty{220}{m}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:3}
$E_p = \tfrac12 \times 500 \times 0.10^2 = \qty{2.5}{J}$; $v = \sqrt{2 \times
2.5/0.050} = \qty{10}{m/s}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:4}
$\tfrac12 mv^2 = mg\ell(1 - \cos 60^\circ)$: $v = \sqrt{g\ell} = \qty{3.1}{m/s}$.
Tensão: $T - mg = mv^2/\ell = mg$, $T = 2mg$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:5}
O atrito realiza $-\mu_dmg\cos\alpha\,(h/\sin\alpha)$, de modo que
\[
\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha) = mgh(1 - 0.35),
\]
$v = \sqrt{2 \times 9.81 \times 2.0 \times 0.65} = \qty{5.1}{m/s}$; $35\%$ perdidos.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:6}
$E_m = \tfrac12 mv^2 - GMm/R \geq 0$ para que o corpo alcance o infinito
(onde $E_p = 0$ e $v \to 0$): $v_e = \sqrt{2GM/R}$, isto é,
\[
v_e = \sqrt{\frac{2 \times 6.67\times10^{-11} \times 5.97\times10^{24}}{6.37\times10^6}}
= \qty{11.2}{km/s}.
\]
Mesma densidade, $M \propto R^3$: $v_e \propto R$, logo \qty{22.4}{km/s}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:7}
Ponto mais baixo: $mg(\ell_0 + x) = \tfrac12 kx^2$, $25x^2 - 687x - 13734 = 0$,
$x = \qty{41}{m}$; queda total \qty{61}{m}; $T_{\max} = kx = \qty{2.0}{kN}
= 3.0\,mg$ (resultante de \qty{2}{g} para cima). A velocidade é máxima onde a força resultante
se anula, $kx = mg$: $x = \qty{14}{m}$, \qty{34}{m} abaixo do ponto de salto.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:8}
Gravidade: $mgv\sin\alpha = 80 \times 9.81 \times 5.0 \times 0.05 = \qty{196}{W}$;
arrasto: $\tfrac12\rho C_xSv^3 = 0.5 \times 1.2 \times 0.50 \times 125 =
\qty{38}{W}$; total \qty{234}{W}.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:9}
$E_p' = 4E_0(x^3/a^4 - x/a^2) = 0$: $x = 0, \pm a$. $E_p'' = E_0(12x^2/a^4 -
4/a^2)$: $-4E_0/a^2$ em $0$ (máximo instável, $E_p = 0$), $+8E_0/a^2$
em $\pm a$ (mínimos estáveis, $E_p = -E_0$): poços de profundidade $E_0$;
$\omega_0^2 = 8E_0/ma^2$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:10}
$\tfrac12 m\dot x^2 + \tfrac12 kx^2 = E$: uma elipse de semieixos $x_{\max}
= \sqrt{2E/k}$ e $\dot x_{\max} = \sqrt{2E/m}$. O movimento é harmônico,
de período $2\pi\sqrt{m/k}$ qualquer que seja a \omterm{def:b1:wave-propagation:signal}{amplitude} (isocronismo), de modo que
toda elipse leva o mesmo tempo.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:11}
Energia: $v^2 = 2gR(1 - \cos\theta)$. Radial: $mg\cos\theta - N = mv^2/R$,
logo $N = mg(3\cos\theta - 2)$, nula em $\cos\theta = 2/3$ ($\theta =
48^\circ$): altura $2R/3$, velocidade $\sqrt{2gR/3}$.
\end{solution}

\begin{solution}{exo:b1:work-and-energy:12}
$\tfrac12 mv^2 = \mu_dmgd$: $\mu_d = 625/(2 \times 9.81 \times 80) = 0.40$.
Calor no contato pneu--pista (e uma marca de derrapagem). Rodas que giram usam
o atrito estático, com $\mu_s > \mu_d$: força de frenagem maior, distância
menor e direção preservada.
\end{solution}

\begin{solution}{pb:b1:work-and-energy:1}
\textbf{1.} $r \to 0$: $+\infty$ (o termo em $r^{-12}$); $r \to \infty$:
$0^-$. Uma parede íngreme, um poço, uma longa cauda.

\textbf{2.} $E_p' = \epsilon[-12\sigma^{12}/r^{13} + 12\sigma^6/r^7] = 0$ se e só se
$(\sigma/r)^6 = 1$: $r = \sigma$, $E_p = \epsilon(1 - 2) = -\epsilon$.

\textbf{3.} $(\sigma/r)^{12} = u^2$: $E_p = \epsilon(u^2 - 2u) = \epsilon[(u - 1)^2
- 1]$, mínimo em $u = 1$ ($r = \sigma$) com valor $-\epsilon$.

\textbf{4.} $F = -E_p' = (12\epsilon/\sigma)[(\sigma/r)^{13} - (\sigma/r)^7]$:
positiva (repulsiva) para $r < \sigma$, negativa (atrativa) além disso.

\textbf{5.} $(1/1.2)^6 = 0.335$, $(1/1.2)^{12} = 0.112$: $E_p = \epsilon(0.112
- 0.670) = -0.56\epsilon = \qty{-2.5}{eV}$. $(1/1.2)^{13} = 0.0935$,
$(1/1.2)^7 = 0.279$: $F = (12\epsilon/\sigma)(-0.186) = -2.2\epsilon/\sigma =
-2.2 \times 7.0\times10^{-19}/1.27\times10^{-10} = \qty{-12}{nN}$.

\textbf{6.} $\epsilon = \qty{4.4}{eV} = 4.4 \times 96.5 = \qty{425}{kJ/mol}$,
próximo dos \qty{431}{kJ/mol} medidos.

\textbf{7.} $E_p'' = \epsilon[156\sigma^{12}/r^{14} - 84\sigma^6/r^8]$; em
$\sigma$: $72\epsilon/\sigma^2 > 0$: estável.

\textbf{8.} $E_p \approx -\epsilon + \tfrac12(72\epsilon/\sigma^2)(r - \sigma)^2$:
$k = 72\epsilon/\sigma^2$.

\textbf{9.} $k = 72 \times 7.0\times10^{-19}/(1.27\times10^{-10})^2 =
\qty{3.1e3}{N/m}$.

\textbf{10.} $\omega_0 = \sqrt{k/m} = \sqrt{3.1\times10^3/1.67\times10^{-27}} =
\qty{1.4e15}{rad/s}$; $f = \qty{2.2e14}{Hz}$; $\lambda = c/f = \qty{1.4}{\micro m}$;
$1/\lambda = \qty{7300}{cm^{-1}}$.

\textbf{11.} Um fator $2.4$ alto demais: as paredes do modelo são íngremes demais
(o caroço em $r^{-12}$ é um substituto grosseiro de uma ligação covalente), mas a
\omterm{def:b1:units-dimensions:oom}{ordem de grandeza} --- uma vibração no infravermelho a $10^{14}$ Hz --- está
certa, e o método (curvatura do poço) é exatamente o que se usa
com potenciais melhores.

\textbf{12.} $\tfrac12 kA^2 \approx k_BT$: $A = \sqrt{2k_BT/k} = \sqrt{8.3\times
10^{-21}/3.1\times10^3} = \qty{1.6e-12}{m} = 0.013\sigma$: a ligação
apenas estremece.

\textbf{13.} $v_{\max} = A\omega_0 = 1.6\times10^{-12} \times 1.4\times10^{15}
= \qty{2.2}{km/s}$.

\textbf{14.} $-\epsilon < E_m < 0$: curvas fechadas em torno de $(\sigma, 0)$ ---
vibração, estado ligado; $E_m > 0$: curvas abertas --- os átomos se separam;
$E_m = 0$: a separatriz, um átomo que apenas consegue alcançar o infinito.

\textbf{15.} $u^2 - 2u = -\tfrac12$ com $u = (\sigma/r)^6$: $u = 1 \pm 1/\sqrt2
= 1.71$ ou $0.293$; $r = \sigma u^{-1/6}$: $0.915\sigma$ e $1.23\sigma$.

\textbf{16.} Ponto médio $1.07\sigma > \sigma$: a parede externa é mais macia que
a interna, de modo que o átomo dispõe de mais espaço do lado de fora; a
distância média no tempo cresce com a energia --- ligações aquecidas se alongam, e os sólidos se dilatam.

\textbf{17.} Maior: a força restauradora do lado externo, mais macio, é
mais fraca do que a aproximação harmônica supõe, de modo que o retorno leva
mais tempo.

\textbf{18.} $\epsilon/2 = \qty{2.2}{eV} = \qty{3.5e-19}{J}$. Tomada como
\omterm{thm:b1:work-and-energy:ke}{energia cinética} em $\sigma$, essa energia dá
$v = \sqrt{2 \times 3.5\times10^{-19}/1.67\times10^{-27}}
= \qty{2.0e4}{m/s}$.

\textbf{19.} Conservação da energia: $E_m > 0$ continua $> 0$, a curva é
aberta, e os átomos se afastam após um único encontro. Um terceiro corpo (outra
molécula, uma parede, um fóton) precisa levar embora o excesso.

\textbf{20.} $W = E_p(2\sigma) - E_p(\sigma) = \epsilon(2^{-12} - 2 \times 2^{-6})
+ \epsilon = 0.97\epsilon = \qty{4.3}{eV}$, positivo: a atração puxa
o átomo para dentro, é motora.

\textbf{21.} $W_{\text{ext}} = E_p(1.5\sigma) - E_p(\sigma) = \epsilon(0.0077 -
0.176) + \epsilon = 0.83\epsilon = \qty{3.7}{eV}$.

\textbf{22.} $(1/0.9)^{13} = 3.93$, $(1/0.9)^7 = 2.09$: $F = +22\epsilon/\sigma
= \qty{+120}{nN}$, dez vezes a atração em $1.2\sigma$: comprimir uma
ligação em $10\%$ custa muito mais do que esticá-la em $20\%$.

\textbf{23.} $T = \epsilon/k_B = 7.0\times10^{-19}/1.38\times10^{-23} =
\qty{5.1e4}{K}$. Os gases reais se dissociam bem abaixo disso porque as energias são
distribuídas: uma fração das moléculas carrega muitas vezes $k_BT$
(\cref{ch:b1:kinetic-theory}).

\textbf{24.} Sobrevivem: o equilíbrio no mínimo, $k = E_p''$ no
mínimo, a energia de dissociação $=$ profundidade, os \omterm{def:b1:work-and-energy:turning}{pontos de retorno} a partir de $E_p = E_m$,
a assimetria $\Rightarrow$ dilatação. Mudam: os valores de $k$, da
frequência e dos \omterm{def:b1:work-and-energy:turning}{pontos de retorno}.

\textbf{25.} Equilíbrio: $E_p' = 0$; vibração: $\omega_0^2 = E_p''/m$;
escape: $E_m$ comparada com a borda do poço.
\end{solution}
