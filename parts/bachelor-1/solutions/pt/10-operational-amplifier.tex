\section*{Capítulo \ref{ch:b1:operational-amplifier} --- O amplificador operacional}

\begin{solution}{exo:b1:operational-amplifier:1}
$G = 1 + 9 = 10$; $V_s = \qty{5.0}{V}$; saturação em $V_e = 15/10 =
\qty{1.5}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:2}
$G = -10$; \omterm{def:b1:sinusoidal-impedance:impedance}{impedância} de entrada $R_1 = \qty{10}{k\ohm}$; $i = 1.0/10^4 =
\qty{0.10}{mA}$; $V_s = \qty{-10}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:3}
Sem ele: divisor $1/(1 + 10) = 0.091$, \qty{91}{mV}. Com o seguidor:
\qty{1.0}{V} --- o seguidor nada drena do sensor e fornece
o \qty{1}{mA} de que a carga precisa.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:4}
$V_s = -R_f(V_1/R_1 + V_2/R_2) = -(V_1 + V_2/2) = -(1.0 + 1.0) = \qty{-2.0}{V}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:5}
$V_+ = V_2R_2/(R_1 + R_2)$; em $V_-$: $(V_1 - V_-)/R_1 = (V_- - V_s)/R_2$,
logo $V_s = V_-(1 + R_2/R_1) - V_1R_2/R_1$; com $V_- = V_+$:
$V_s = V_2R_2/R_1 - V_1R_2/R_1$. Números: $10 \times 0.05 = \qty{0.50}{V}$;
o \qty{1}{V} comum não contribui em nada --- é a rejeição de modo comum.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:6}
$V_s = -(1/RC)\int V_e$: inclinação $-1000 \times 1.0 = \qty{-1000}{V/s}$;
saturação após $15/1000 = \qty{15}{ms}$. Onda quadrada: triângulo de
\omterm{def:b1:wave-propagation:signal}{amplitude} $ET/4RC = 1.0 \times 10^{-3}/(4 \times 10^{-3}) = \qty{0.25}{V}$
(mais a constante que a carga inicial tenha deixado).
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:7}
$\underline H = -(R'/R)/(1 + jR'C\omega)$: ganho contínuo $-100$; corte em
$1/(2\pi R'C) = \qty{1.6}{Hz}$; a \qty{1}{kHz}, $R'C\omega = 628$, $G =
100/628 = 0.16$ --- o valor $1/RC\omega$ do integrador. Ele integra para
$f \gg \qty{1.6}{Hz}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:8}
$R_2/R_1 = 10$: $R_2 = \qty{10}{k\ohm}$; $C = 1/(2\pi R_2f_c) = 1/(2\pi
\times 10^4 \times 500) = \qty{32}{nF}$.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:9}
Com $\mu$ finito: $V_s = \mu(V_e - \beta V_s)$, $\beta = 1/100$, logo
$G = \mu/(1 + \mu\beta) = 10^5/(1 + 10^3) = 99.9$. \omterm{thm:b1:sinusoidal-impedance:resonance}{Largura de banda} $f_T/G =
\qty{10}{kHz}$; seguidor: \qty{1}{MHz}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:10}
Cada excursão de ruído através de \qty{2.0}{V} faz a saída virar: uma rajada de
chaveamentos rápidos (``trepidação'') em torno do cruzamento. Remédio: uma histerese
mais larga que o ruído, isto é,\ um \omterm{prop:b1:operational-amplifier:schmitt}{gatilho de Schmitt}.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:11}
$V_+ = V_sR_1/(R_1 + R_2) = V_s/10$: limiares $\pm\qty{1.5}{V}$. A
saída vale $+\qty{15}{V}$ enquanto $V_e$ sobe até \qty{1.5}{V}, vira para
$-\qty{15}{V}$, ali permanece até que $V_e$ caia abaixo de \qty{-1.5}{V} e vira
de volta: uma onda quadrada na frequência do triângulo, com as bordas nos
cruzamentos dos limiares.
\end{solution}

\begin{solution}{exo:b1:operational-amplifier:12}
Com $V_s = +V_{\mathrm{sat}}$, $V_C$ se carrega de $-\beta V_{\mathrm{sat}}$
rumo a $+V_{\mathrm{sat}}$: $V_C = V_{\mathrm{sat}} - (1 + \beta)V_{\mathrm{sat}}
\eu^{-t/RC}$, atingindo $+\beta V_{\mathrm{sat}}$ quando $\eu^{-t/RC} =
(1 - \beta)/(1 + \beta)$: meio período $RC\ln\frac{1 + \beta}{1 - \beta}$,
donde $T$. Para $\beta = \tfrac12$: $T = 2RC\ln 3 = 2.2RC$; $RC = 1/(2.2 \times
10^3) = \qty{0.455}{ms}$, $R = \qty{4.6}{k\ohm}$. $V_s$: onda quadrada
$\pm\qty{15}{V}$; $V_C$: arcos exponenciais entre $\pm\qty{7.5}{V}$.
\end{solution}

\begin{solution}{pb:b1:operational-amplifier:1}
\textbf{1.} Dois divisores: $u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)]
\approx -E\epsilon/4$ (\cref{prop:b1:dc-circuits:wheatstone}); a ligação
fixa o sinal.

\textbf{2.} $u = 5.00 \times 2.0\times10^{-3}/4 = \qty{2.5}{mV}$.

\textbf{3.} $V_A = E/2 = \qty{2.500}{V}$ sempre; $V_B = E(1 + \epsilon)/(2 +
\epsilon) \approx 2.500 + 0.0025 = \qty{2.5025}{V}$ em fundo de escala. Os
\qty{2.5}{V} comuns aos dois são a \omterm{def:b1:dc-circuits:voltage}{tensão} de modo comum.

\textbf{4.} Cada ponto médio: $1 \parallel 1 = \qty{500}{\ohm}$; entre os
dois pontos médios, \qty{1.0}{k\ohm}.

\textbf{5.} $10/(10 + 1) = 0.91$: uma perda de $9\%$.

\textbf{6.} \omterm{def:b1:sinusoidal-impedance:impedance}{Impedância} de entrada infinita (nenhuma corrente sai da ponte, de modo que
os pontos médios conservam os seus potenciais em circuito aberto) e \omterm{def:b1:sinusoidal-impedance:impedance}{impedância} de
saída nula (o estágio seguinte pode drenar o que precisar).

\textbf{7.} Exatamente $V_A$ e $V_B$; corrente nula.

\textbf{8.} Como em \cref{prop:b1:operational-amplifier:sumdiff}: $V_+ =
V_BR_2/(R_1 + R_2)$, \omterm{thm:b1:dc-circuits:kirchhoff}{lei dos nós} em $V_-$, $V_- = V_+$.

\textbf{9.} $R_2 = \qty{1.0}{M\ohm}$.

\textbf{10.} $1 + R_2/R_1 = 10$: por exemplo, \qty{1}{k\ohm} e \qty{9}{k\ohm}.
Ganho total $1000$; fundo de escala $2.5 \times 10^{-3} \times 1000 = \qty{2.5}{V}$.

\textbf{11.} Casado: os \qty{2.5}{V} comuns se cancelam exatamente (só
$V_B - V_A$ sobrevive). Um descasamento de $1\%$ deixa passar cerca de $1\%$ deles: um
erro referido à entrada da ordem de \qty{25}{mV}, dez vezes o sinal de
fundo de escala --- os quatro \omterm{def:b1:dc-circuits:resistor}{resistores} precisam ser casados a melhor que $0.01\%$
(ou a \omterm{def:b1:dc-circuits:voltage}{tensão} de modo comum removida antes).

\textbf{12.} $1\ \unit{mV} \times 1000 = \qty{1}{V}$, quarenta por cento do fundo
de escala: inutilizável sem ajuste de offset, sem um amplificador de baixo offset
(chopper) ou sem uma calibração de zero com a viga descarregada.

\textbf{13.} Estágio 1: $f_T/100 = \qty{10}{kHz}$; estágio 2: \qty{100}{kHz};
muito acima de \qty{5}{Hz}.

\textbf{14.} Exige-se $2.5/10^{-2} = \qty{250}{V/s} = \qty{0.25}{mV/\micro s}$,
duas mil vezes abaixo da \omterm{rem:b1:operational-amplifier:real}{slew rate}: não.

\textbf{15.} $C = 1/(2\pi R_2f_c) = 1/(2\pi \times 1.6\times10^4 \times 10)
= \qty{1.0}{\micro F}$.

\textbf{16.} $x = 5$: $G = 1/\sqrt{26}$, \qty{-14}{dB}.

\textbf{17.} $\varphi = -\arctan 0.5 = -27^\circ$; atraso $\varphi/\omega =
\qty{15}{ms}$ --- nada diante da passagem de um caminhão.

\textbf{18.} Não carrega o estágio seguinte (\omterm{def:b1:sinusoidal-impedance:impedance}{impedância} de saída nula)
e há ganho disponível; além disso, não atenua o sinal desejado abaixo do
corte.

\textbf{19.} \qty{-28}{dB} (os ganhos se multiplicam, os decibéis se somam), exata porque
a \omterm{def:b1:sinusoidal-impedance:impedance}{impedância} de saída do primeiro estágio é nula: o segundo estágio não o
carrega.

\textbf{20.} $\epsilon = 1.6\times10^{-3} \to u = \qty{2.0}{mV} \to
\qty{2.0}{V}$; saída $+V_{\mathrm{sat}}$ quando o sinal ultrapassa
\qty{2.0}{V}, e $-V_{\mathrm{sat}}$ abaixo (ou o inverso, conforme a ligação).

\textbf{21.} Ela trepida: a lâmpada pisca à medida que o ruído cruza a
referência.

\textbf{22.} Com $V_{\mathrm{ref}}$ através de $R_2$ e a saída através de
$R_1$: na forma padrão, a entrada não inversora fica em
$V_{\mathrm{ref}}(1 - \beta) + \beta V_s$; tomando a referência deslocada
de acordo, os níveis de comutação valem $V_{\mathrm{ref}} \pm \beta V_{\mathrm{sat}}$.
De $\beta V_{\mathrm{sat}} = \qty{0.10}{V}$: $\beta = 1/150$, por exemplo $R_1 =
\qty{1.0}{k\ohm}$, $R_2 = \qty{149}{k\ohm}$.

\textbf{23.} \qty{2.1}{V} (disparo) e \qty{1.9}{V} (rearme): deformações
\num{1.68e-3} e \num{1.52e-3}.

\textbf{24.} $R = (15 - 2.0)/0.013 = \qty{1.0}{k\ohm}$. Em
$-V_{\mathrm{sat}}$ o LED fica polarizado reversamente com \qty{15}{V}, além da sua
especificação: um diodo comum em antiparalelo (ou em série) o protege.

\textbf{25.} Ponte (\qty{2.5}{mV} de fundo de escala) $\to$ dois seguidores (sem
carregamento) $\to$ \omterm{prop:b1:operational-amplifier:sumdiff}{amplificador diferencial} $\times100$ $\to$ não inversor
$\times10$ $\to$ passa-baixa ativo de \qty{10}{Hz} $\to$ \omterm{prop:b1:operational-amplifier:schmitt}{gatilho de Schmitt} em
$\qty{2.0}{V} \pm \qty{0.1}{V}$ $\to$ LED. Os limites reais: a \omterm{rem:b1:operational-amplifier:real}{tensão de
offset de entrada} e o casamento dos \omterm{def:b1:dc-circuits:resistor}{resistores} do \omterm{prop:b1:operational-amplifier:sumdiff}{amplificador
diferencial}, ambos comparáveis ao sinal de milivolts.
\end{solution}
