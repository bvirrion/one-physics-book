\section*{Capítulo \ref{ch:b1:charged-particles} --- Partículas carregadas em campos E e B}

\begin{solution}{exo:b1:charged-particles:1}
$\qty{1}{eV} = \qty{1.60e-19}{J}$, $\qty{13.6}{eV} = \qty{2.18e-18}{J}$,
$\qty{1}{MeV} = \qty{1.60e-13}{J}$. $v = \sqrt{2eU/m}$: elétron
\qty{1.9e7}{m/s}; próton \qty{4.4e5}{m/s} ($\sqrt{1836}$ vezes mais lento).
\end{solution}

\begin{solution}{exo:b1:charged-particles:2}
$v = \qty{1.88e7}{m/s}$; $R = mv/eB = 9.11\times10^{-31} \times 1.88\times10^7/
(1.6\times10^{-19} \times 10^{-3}) = \qty{0.11}{m}$; $\omega_c = eB/m =
\qty{1.8e8}{rad/s}$; $T = 2\pi/\omega_c = \qty{36}{ns}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:3}
$R = mv/qB = 1.67\times10^{-27} \times 10^5/(1.6\times10^{-19} \times 5\times
10^{-5}) = \qty{21}{m}$. A $30^\circ$: uma hélice de raio $R\sin 30^\circ
= \qty{10}{m}$ e passo $2\pi(m/qB)v\cos 30^\circ = \qty{114}{m}$, que avança
ao longo da linha de campo.
\end{solution}

\begin{solution}{exo:b1:charged-particles:4}
$v = E/B = 10^4/0.020 = \qty{5.0e5}{m/s}$. Íons mais rápidos: a força magnética
vence e eles se curvam para o lado magnético; carga oposta com a mesma
velocidade: as duas forças se invertem e continuam equilibradas --- o seletor é cego ao
sinal e à massa.
\end{solution}

\begin{solution}{exo:b1:charged-particles:5}
$\tan\alpha = EL/2U_0 = 10^4 \times 0.04/4000 = 0.10$; deflexão
$0.30 \times 0.10 = \qty{3.0}{cm}$; sensibilidade \qty{0.30}{mm/V}. Elétrons
mais rápidos passam menos tempo entre as placas: $\tan\alpha \propto 1/U_0$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:6}
$R = \sqrt{2mU/e}/B$: $m_{12} = \qty{1.99e-26}{kg}$, $R_{12} = \sqrt{2 \times
1.99\times10^{-26} \times 2\times10^4/1.6\times10^{-19}}/0.5 = \qty{14.1}{cm}$;
$R_{13} = R_{12}\sqrt{13/12} = \qty{14.7}{cm}$; separação $2\Delta R =
\qty{1.2}{cm}$ entre os pontos de chegada (os raios diferem de \qty{5.8}{mm},
e os diâmetros do dobro disso).
\end{solution}

\begin{solution}{exo:b1:charged-particles:7}
$f_c = eB/2\pi m = \qty{23}{MHz}$; $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} =
\qty{27}{MeV}$; \qty{100}{keV} por volta: $270$ voltas; tempo $270/f_c =
\qty{12}{\micro s}$; $\gamma - 1 = 27/938 = 2.9\%$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:8}
$v_\perp = \qty{5.0e5}{m/s}$, $v_\parallel = \qty{8.7e5}{m/s}$: $R = mv_\perp/eB
= 9.11\times10^{-31} \times 5\times10^5/(3.2\times10^{-22}) = \qty{1.4}{mm}$;
período $2\pi m/eB = \qty{18}{ns}$, passo $v_\parallel T = \qty{1.5}{cm}$:
cerca de $64$ voltas por metro.
\end{solution}

\begin{solution}{exo:b1:charged-particles:9}
$U_H = IB/nqd = 5.0/(8.5\times10^{28} \times 1.6\times10^{-19} \times 10^{-3})
= \qty{3.7e-7}{V}$: menos de um microvolt. Com $n$ dez milhões de vezes menor, um
semicondutor dá milivolts --- um sensor utilizável.
\end{solution}

\begin{solution}{exo:b1:charged-particles:10}
$m\,\dd\vect v/\dd t = q\vect v\wedge\vect B$; multiplique escalarmente por $\vect v$: $\dd(\tfrac12 mv^2)/
\dd t = 0$ qualquer que seja $\vect B(\vect r, t)$, desde que não haja $\vect E$. A
direção da velocidade, o \omterm{def:b1:point-kinematics:frenet}{raio de curvatura} e o passo podem todos
mudar; o módulo da velocidade e a \omterm{thm:b1:work-and-energy:ke}{energia cinética} (\qty{10}{keV}) não.
\end{solution}

\begin{solution}{exo:b1:charged-particles:11}
$m\dot v_x = qv_yB$, $m\dot v_y = qE - qv_xB$, $v_z = 0$. Com $\vect v =
(E/B)\vect e_x + \vect w$: $m\dot w_x = qw_yB$, $m\dot w_y = -qw_xB$ --- uma
rotação ciclotrônica pura de $\vect w$ em $\omega_c$. Partindo do repouso,
$\vect w(0) = -(E/B)\vect e_x$: $\vect w$ gira com norma constante $E/B$, de modo que a
velocidade é a deriva $E/B$ mais um vetor girante de mesma norma ---
uma cicloide, com cúspides onde $\vect v = \vect 0$. Elétron: deriva $E/B =
\qty{1.0e5}{m/s}$, $\omega_c = \qty{1.8e10}{rad/s}$, altura do arco $2mE/qB^2 =
\qty{11}{\micro m}$.
\end{solution}

\begin{solution}{exo:b1:charged-particles:12}
$R = mv_0/eB = 9.11\times10^{-31} \times 2.65\times10^7/(1.6\times10^{-22}) =
\qty{0.15}{m}$; $\alpha \approx L/R = 0.04/0.15 = 0.27$ (cerca de $15^\circ$):
deflexão $\approx 0.30 \times 0.27 = \qty{8}{cm}$ para um modesto
militesla. Ângulos grandes com placas exigiriam quilovolts e placas
longas; uma bobina curva o feixe em dezenas de graus sem \omterm{def:b1:dc-circuits:voltage}{tensão}
alguma.
\end{solution}

\begin{solution}{pb:b1:charged-particles:1}
\textbf{1.} $\vect F = e\vect v\wedge\vect B \perp \vect v$: sem potência, módulo da
velocidade constante.

\textbf{2.} $mv^2/R = evB$: $R = mv/eB$; $T = 2\pi R/v = 2\pi m/eB$, livre de
$v$.

\textbf{3.} $f_c = eB/2\pi m = 1.6\times10^{-19} \times 1.5/(2\pi \times 1.67
\times10^{-27}) = \qty{23}{MHz}$.

\textbf{4.} $eU = \qty{50}{keV}$ por travessia, \qty{100}{keV} por volta.

\textbf{5.} O intervalo deve estar na polaridade aceleradora cada vez que o
próton chega, a meio período de distância; fora da frequência, a fase escorrega e
os impulsos acabam por desacelerar.

\textbf{6.} Após uma volta, $E_k = \qty{100}{keV}$, $v = \sqrt{2E_k/m} =
\qty{4.4e6}{m/s}$, $R = mv/eB = \qty{3.0}{cm}$; após $n$ voltas, $E_k =
n \times \qty{100}{keV}$ e $R = \qty{3.0}{cm}\sqrt n$.

\textbf{7.} $E_k = e^2B^2R^2/2m = \qty{4.3e-12}{J} = \qty{27}{MeV}$; $v =
\sqrt{2E_k/m} = \qty{7.2e7}{m/s} = 0.24c$.

\textbf{8.} $270$ voltas; $270 \times \qty{43.5}{ns} = \qty{12}{\micro s}$.

\textbf{9.} $\sum 2\pi R_n = 2\pi \times 0.030\sum\sqrt n \approx 2\pi \times 0.030
\times \tfrac23 \times 270^{3/2} = \qty{560}{m}$.

\textbf{10.} Não: $E_k$ depende apenas de $B$ e de $R_{\max}$; $U$ fixa o
número de voltas e o tempo.

\textbf{11.} Trajeto de \qty{560}{m}: é preciso $\lambda \gtrsim \qty{1}{km}$, isto é,\
$p \lesssim 10^5\ \unit{Pa} \times 7\times10^{-8}/10^3 = \qty{7e-6}{Pa}$ ---
um alto-vácuo.

\textbf{12.} $\gamma - 1 = 27/938 = 0.029$; o período cresce $2.9\%$.
Ao longo de $270$ voltas o escorregamento acumula vários períodos --- as últimas
voltas ficariam fora de fase; na prática ajustam-se $B$ e a fase de injeção,
e os \qty{27}{MeV} estão perto do limite da máquina simples.

\textbf{13.} $E_k = (eBR)^2/2m$: $BR = \sqrt{2mE_k}/e = \sqrt{2 \times 1.67\times
10^{-27} \times 3.68\times10^{-11}}/1.6\times10^{-19} = \qty{2.2}{T.m}$; $R =
\qty{1.1}{m}$ a \qty{2}{T}.

\textbf{14.} $T = 2\pi\gamma m/eB$ cresce $24\%$ do centro à borda: uma
frequência fixa fica irremediavelmente fora de passo.

\textbf{15.} Sincrocíclotron: baixar a frequência à medida que o pacote ganha
energia --- um único pacote por vez, corrente média baixa. Isócrono:
moldar $B(R) \propto \gamma$ para que o período fique constante --- feixe
contínuo, a escolha moderna.

\textbf{16.} $p = \gamma mv = 1.245 \times 1.67\times10^{-27} \times 1.8\times10^8
= \qty{3.7e-19}{kg.m/s}$; $R = p/eB = \qty{1.2}{m}$, um pouco mais que
os \qty{1.1}{m} não relativísticos.

\textbf{17.} Impulso transversal $eEL/v = 1.6\times10^{-19} \times 10^5 \times
1.0/1.8\times10^8 = \qty{8.9e-23}{kg.m/s}$; ângulo $\approx 8.9\times10^{-23}/
3.7\times10^{-19} = \qty{2.4e-4}{rad}$: inútil para desviar o feixe.

\textbf{18.} Magnético: $R = p/eB = \qty{2.3}{m}$, ângulo $L/R = \qty{0.43}{rad}
\approx 25^\circ$: mil vezes mais --- ímãs, portanto.

\textbf{19.} Energia por próton $\qty{3.68e-11}{J}$; dose $2.0 \times 0.50 =
\qty{1.0}{J}$: $2.7\times10^{10}$ prótons.

\textbf{20.} $10^{-9}/1.6\times10^{-19} = \num{6.2e9}$ prótons por segundo: cerca de
\qty{4.4}{s}.

\textbf{21.} $P = \qty{230}{MeV} \times \qty{6.2e9}{s^{-1}} = \qty{0.23}{W}$:
uma luz de vigília, entregue exatamente onde se quer.

\textbf{22.} $\qty{58}{MeV}$ em \qty{1}{cm}: cerca de \qty{6}{keV/\micro m} ---
ionização densa ao longo do último trecho.

\textbf{23.} $230\times10^6/30 \approx \num{8e6}$ ionizações por próton ---
o dano molecular que mata a célula.

\textbf{24.} Os raios X depositam energia ao longo de todo o seu caminho, sobretudo perto da
entrada; os prótons depositam a maior parte da sua no fim do alcance,
poupando o tecido à frente e não deixando nada atrás.

\textbf{25.} $R = mv/eB$ e $T = 2\pi m/eB$ (o círculo), $E_k = (eBR)^2/2m$
(a energia), $E_k/2eU$ (as voltas), dose $=$ (prótons $\times$ energia)/massa
e corrente $= e \times$ taxa (o tempo).
\end{solution}
