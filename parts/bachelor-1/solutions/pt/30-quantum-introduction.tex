\begin{solution}{exo:b1:quantum-introduction:1}

\end{solution}

\begin{solution}{exo:b1:quantum-introduction:3}

\end{solution}

\begin{solution}{exo:b1:quantum-introduction:4}

\end{solution}

\begin{solution}{exo:b1:quantum-introduction:5}

\begin{solution}{exo:b1:quantum-introduction:6}
$\Delta\lambda = \qty{2.43}{pm}$: $\lambda' = \qty{12.4}{pm}$. Antes: $1240/0.010 =
\qty{124}{keV}$; depois: $1240/0.0124 = \qty{100}{keV}$; o elétron leva

\end{solution}

\begin{solution}{exo:b1:quantum-introduction:7}
$p = \sqrt{2 \times 9.11 \times 10^{-31} \times 5 \times 10^4 \times 1.6 \times 10^{-19}} =
\qty{1.2e-22}{kg.m/s}$, $\lambda = \qty{5.5}{pm}$ (a relatividade corrige para
\qty{5.4}{pm}); espaçamento das franjas $\lambda D/d = 5.5 \times 10^{-12} \times 0.35/2 \times 10^{-6}
= \qty{1}{\micro m}$: invisível a \omterm{def:b1:optical-instruments:eye}{olho} nu, daí a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} de aumento. Dez
elétrons: dez pontos, aparentemente ao acaso; $10^5$: as franjas. Cada
elétron chega num ponto; a \emph{probabilidade} de chegada segue
a onda de duas fendas --- a onda de um único elétron passa pelas duas fendas.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:8}
Átomo: $\Delta p \geq \hbar/2\Delta x = \qty{5.3e-25}{kg.m/s}$, $E \sim \Delta p^2/2m_e =
\qty{1.5e-19}{J} \approx \qty{1}{eV}$: a escala certa. Núcleo: $\Delta p \geq
\qty{1.1e-20}{kg.m/s}$; $\Delta p^2/2m_e$ daria \qty{400}{MeV}, muito acima de
$m_ec^2$, de modo que se usa $E \approx \Delta p\,c = \qty{3e-12}{J} = \qty{20}{MeV}$: força
nenhuma dentro do núcleo poderia segurar um elétron tão energético --- os núcleos não contêm
elétrons (os elétrons beta são criados na emissão). Grão: $\Delta v \geq
\hbar/2m\Delta x = 1.05 \times 10^{-34}/(2 \times 10^{-9} \times 10^{-6}) = \qty{5e-20}{m/s}$:
irrelevante.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:9}
(a) $m_ev^2/r = e^2/4\pi\varepsilon_0r^2$: $v = \sqrt{e^2/4\pi\varepsilon_0m_er}$, $E =
\tfrac12m_ev^2 - e^2/4\pi\varepsilon_0r = -e^2/8\pi\varepsilon_0r$. (b) $m_evr = n\hbar$ e
$m_ev^2r = e^2/4\pi\varepsilon_0$: divida o quadrado da primeira pela segunda:
$m_er = n^2\hbar^2 \cdot 4\pi\varepsilon_0/e^2$, $r_n = n^2a_0$, e $E_n = -e^2/8\pi\varepsilon_0n^2a_0
= E_1/n^2$ com $a_0 = \qty{52.9}{pm}$, $E_1 = \qty{-13.6}{eV}$. (c) $v_1 =
\hbar/m_ea_0 = \qty{2.19e6}{m/s}$, $v_1/c = 1/137$. (d) $2 \to 1$: $\tfrac34 \times
13.6 = \qty{10.2}{eV}$, \qty{122}{nm}; $3 \to 2$: $(\tfrac14 - \tfrac19) \times 13.6 =
\qty{1.89}{eV}$, \qty{656}{nm}. (e) $\lambda_n = h/m_ev_n = nh/m_ev_1 = 2\pi na_0$
(pois $m_ev_1a_0 = \hbar$), e $2\pi r_n = 2\pi n^2a_0 = n\lambda_n$.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:10}

\qty{54.4}{eV}. U$^{91+}$: $92^2 \times 13.6 = \qty{115}{keV}$ --- mas $v_1 = Z\alpha c

cada energia: $E_1 = \qty{-6.8}{eV}$, Lyman $\alpha$ a \qty{5.1}{eV}, \qty{243}{nm}.
\end{solution}

\begin{solution}{exo:b1:quantum-introduction:12}
(a) $E(\Delta x) = \hbar^2/8m\Delta x^2 + \tfrac12m\omega^2\Delta x^2$; derivada nula em
$\Delta x^2 = \hbar/2m\omega$, onde os dois termos valem $\hbar\omega/4$: $E_{\min} =
\hbar\omega/2$. (b) $\tfrac12 \times 1.055 \times 10^{-34} \times 8.3 \times 10^{14} = \qty{4.4e-20}{J}
= \qty{0.27}{eV}$; $T \sim \hbar\omega/k_B = \qty{6300}{K}$ --- à temperatura ambiente
a vibração está congelada no seu \omterm{prop:b1:quantum-introduction:box}{estado fundamental}. (c) $\Delta p = \hbar/2\Delta x =
\qty{1.05e-24}{kg.m/s}$, $E = \Delta p^2/2m_{\mathrm{He}} = 1.1 \times 10^{-48}/1.33 \times
10^{-26} = \qty{8e-23}{J} = \qty{0.5}{meV}$: metade da ligação --- os átomos não conseguem
ficar quietos o bastante para cristalizar; o hélio permanece líquido até o zero
absoluto, a menos que seja comprimido (\qty{25}{bar}). (d) $E_1 = \pi^2\hbar^2/2mL^2$ contra
$\hbar^2/2mL^2$: dez vezes maior, como tem de ser --- o limite é um limite
inferior, e o estado da caixa tem $\Delta x \approx 0.18L$, e não $L/2$.
\end{solution}

\begin{solution}{pb:b1:quantum-introduction:1}
\textbf{1.} A luz liberta elétrons do cátodo com energias
variadas; mantido a $-V_s$, o ânodo os repele e só chegam os que têm
$E_k > eV_s$. A corrente se anula quando $eV_s = E_{k,\max}$.

\textbf{2.} $eV_s = h\nu - W$: inclinação $h/e$, intercepto $-W/e$; $V_s = 0$ no
limiar $\nu_0 = W/h$.

\textbf{3.} Pontos extremos: $(2.36 - 0.30)/(5.0 \times 10^{14}) = \qty{4.12e-15}{V.s}$,
$h = 1.602 \times 10^{-19} \times 4.12 \times 10^{-15} = \qty{6.60e-34}{J.s}$ (um ajuste sobre
os cinco dá $6.62$): a $0.5\%$ de $6.626 \times 10^{-34}$.

\textbf{4.} $W = h\nu - eV_s = 2.47 - 0.30 = \qty{2.2}{eV}$; $\nu_0 = W/h =
\qty{5.3e14}{Hz}$, $\lambda_0 = 1240/2.2 = \qty{560}{nm}$. Uma caneta vermelha
(\qty{650}{nm}, \qty{1.9}{eV}) não arranca nada.

\textbf{5.} $V_s$ inalterado, corrente dobrada: o dobro de fótons, cada um com
a mesma energia. Uma onda daria mais energia a cada elétron com mais
intensidade, e um limiar em intensidade em vez de em frequência.

\textbf{6.} $10^{-6} \times 10^{-20} = \qty{1e-26}{W}$ sobre o átomo; \qty{2.2}{eV} $=
\qty{3.5e-19}{J}$ levam $3.5 \times 10^7$~s --- um ano. A emissão é observada
em nanossegundos: a energia chega em quanta inteiros.

\textbf{7.} $m_ev^2/r = e^2/4\pi\varepsilon_0r^2$: $v = \sqrt{e^2/4\pi\varepsilon_0m_er}$; $E =
\tfrac12m_ev^2 - e^2/4\pi\varepsilon_0r = -e^2/8\pi\varepsilon_0r$.

\textbf{8.} $(m_evr)^2 = n^2\hbar^2$ and $m_ev^2r = e^2/4\pi\varepsilon_0$: $r_n = n^2 \cdot
4\pi\varepsilon_0\hbar^2/m_ee^2 = n^2a_0$, $a_0 = (1.055 \times 10^{-34})^2/(8.99 \times 10^9
\times 9.11 \times 10^{-31} \times 2.57 \times 10^{-38}) = \qty{52.9}{pm}$.

\textbf{9.} $E_n = -e^2/8\pi\varepsilon_0r_n = -m_ee^4/2(4\pi\varepsilon_0)^2\hbar^2n^2$; $E_1 =
-2.31 \times 10^{-28}/(2 \times 5.29 \times 10^{-11}) = \qty{-2.18e-18}{J} = \qty{-13.6}{eV}$.

\textbf{10.} $v_1 = \hbar/m_ea_0 = \qty{2.19e6}{m/s}$; $v_1/c = 7.3 \times 10^{-3} =
1/137$.

\textbf{11.} $2 \to 1$: \qty{10.2}{eV}, \qty{122}{nm} (ultravioleta); $3 \to 2$:
\qty{1.89}{eV}, \qty{656}{nm} (vermelho); $4 \to 2$: \qty{2.55}{eV}, \qty{486}{nm}
(azul-esverdeado): as linhas de Balmer são as visíveis.

\textbf{12.} $hc/\lambda = |E_1|(1/n^2 - 1/p^2)$: $R_H = |E_1|/hc = 13.6/1240 =
\qty{1.097e-2}{nm^{-1}} = \qty{1.097e7}{m^{-1}}$. Limite de Balmer ($p \to \infty$):
$\lambda = 4/R_H = \qty{365}{nm}$.

\textbf{13.} \qty{13.6}{eV}; $\lambda = 1240/13.6 = \qty{91}{nm}$.

\textbf{14.} $v_n = v_1/n$, $\lambda_n = h/m_ev_n = nh/m_ev_1 = 2\pi na_0$ (pois $m_ev_1a_0
= \hbar$); $2\pi r_n = 2\pi n^2a_0 = n\lambda_n$: a órbita carrega $n$ ondas
inteiras --- a regra de Bohr é uma regra de \omterm{thm:b1:wave-propagation:standing}{onda estacionária}.

\textbf{15.} Errado: o elétron não tem órbita (ele irradiaria; o
\omterm{prop:b1:quantum-introduction:box}{estado fundamental} tem \omterm{def:b1:angular-momentum:L}{momento angular} nulo) e o modelo falha para qualquer átomo
com dois elétrons. Certo: os níveis do hidrogênio, e portanto cada linha do seu
espectro. A órbita é substituída por uma \omterm{def:b1:quantum-introduction:wavefunction}{função de onda}, uma nuvem de probabilidade
de tamanho $\sim n^2a_0$.

\textbf{16.} \omterm{thm:b1:wave-propagation:standing}{Ondas estacionárias}: $L = n\lambda/2$, $p_n = h/\lambda_n = nh/2L$, $E_n =
p_n^2/2m = n^2h^2/8mL^2$. A fórmula da esfera é a mesma com $L \to R$:
o \omterm{prop:b1:quantum-introduction:box}{estado fundamental} encaixa meio \omterm{prop:b1:wave-propagation:sinusoidal}{comprimento de onda} no raio.

\textbf{17.} $E_e = 6.02 \times 10^{-38}/(0.13 \times 4 \times 10^{-18}) = \qty{0.72}{eV}$;
$E_h = 0.13/0.45 \times 0.72 = \qty{0.21}{eV}$.

\textbf{18.} $R = \qty{2}{nm}$: $1.74 + 0.93 = \qty{2.67}{eV}$, \qty{464}{nm}, azul;
\qty{3}{nm}: \qty{2.15}{eV}, \qty{577}{nm}, amarelo; \qty{4}{nm}: \qty{1.97}{eV},
\qty{629}{nm}, vermelho.

\textbf{19.} A energia de confinamento cresce como $1/R^2$: uma caixa menor eleva
os níveis, o fóton fica mais energético e a luz, mais azul.

\textbf{20.} $\Delta p \sim \hbar/R = \qty{5.3e-26}{kg.m/s}$, $E \sim \Delta p^2/2m_e^\ast =
\qty{1.2e-20}{J} = \qty{0.07}{eV}$: a ordem certa (o coeficiente exato
$\pi^2/2$ e o $\Delta p$ mais grosseiro explicam o fator dez).

\textbf{21.} $1240/577 = \qty{2.15}{eV} = \qty{3.4e-19}{J}$: $10^{-9}/3.4 \times 10^{-19}
= 2.9 \times 10^9$ fótons por segundo.

\textbf{22.} $10^{-3}/(1.96 \times 1.6 \times 10^{-19}) = 3.2 \times 10^{15}$ por segundo.

\textbf{23.} $100 \times 2.48 \times 1.6 \times 10^{-19} = \qty{4e-17}{J}$ em \qty{0.1}{s}:
\qty{4e-16}{W}.

\textbf{24.} $h\nu = \qty{6.6e-26}{J}$: $1.5 \times 10^{13}$ fótons por segundo ---
e cada um é $10^5$ vezes menor que a energia térmica $k_BT$ da
antena: sem esperança, e sem necessidade; ali a imagem ondulatória é exata.

\textbf{25.} O tamanho dos átomos, $a_0 = \qty{53}{pm}$; a sua escala de energia,
\qty{13.6}{eV} (donde os fótons de alguns eV, a química, o espectro visível); e
a velocidade dos seus elétrons, $\alpha c \approx \qty{2e6}{m/s}$ --- os três
a partir de $h$, $e$ e $m_e$ (e $\varepsilon_0$), sem nada ajustado.
\end{solution}
