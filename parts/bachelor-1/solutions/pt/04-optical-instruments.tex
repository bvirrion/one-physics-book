\section*{Capítulo \ref{ch:b1:optical-instruments} --- Instrumentos ópticos e o olho}

\begin{solution}{exo:b1:optical-instruments:1}
Infinito: $f' = \qty{17}{mm}$, $V = 1/0.017 = \qty{58.8}{\delta}$.
\qty{25}{cm}: $V = 58.8 + 1/0.25 = \qty{62.8}{\delta}$, $f' = \qty{15.9}{mm}$.
Amplitude \qty{4.0}{\delta}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:2}
Míope: o infinito deve ser conjugado no \omterm{def:b1:optical-instruments:punctum}{ponto remoto}, $f' = \qty{-40}{cm}$,
$V = \qty{-2.5}{\delta}$. Hipermetrope: um objeto a \qty{25}{cm} deve ser
conjugado (virtualmente) a \qty{1.0}{m}: $V = 1/\overline{OA'} - 1/\overline{OA}
= -1 + 4 = \qty{+3.0}{\delta}$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:3}
$G = 25/4.0 = 6.3$; selo no plano focal, a \qty{4.0}{cm} da
\omterm{def:b1:mirrors-thin-lenses:lens}{lente}. Detalhe de \qty{2}{mm}: $\theta' = 2/40 = \qty{0.05}{rad}$, contra
$2/250 = \qty{0.008}{rad}$ a \qty{25}{cm}.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:4}
$G = 900/25 = 36$ e $900/9 = 100$; comprimentos de tubo $f_1' + f_2' =
\qty{925}{mm}$ e \qty{909}{mm}; Lua: $36 \times 0.52^\circ = 19^\circ$
e $52^\circ$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:5}
$\gamma_1 = -160/4 = -40$; $G_2 = 250/20 = 12.5$; $G = 500$.
$\overline{F_1A} = -f_1'^2/\Delta = -16/160 = \qty{-0.10}{mm}$: objeto
a \qty{4.1}{mm} da \omterm{def:b1:optical-instruments:microscope}{objetiva}. O objeto deve permanecer a essa distância
da \omterm{def:b1:optical-instruments:microscope}{objetiva}, de modo que o tubo inteiro se desloca \qty{10}{\micro m} ---
é o botão de foco fino.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:6}
\omterm{def:b1:optical-instruments:eye}{Olho} nu: $1.5/3\times10^{-4} = \qty{5}{km}$. Luneta: $\theta_{\min}
= 1.22 \times 5.5\times10^{-7}/0.100 = \qty{6.7e-6}{rad}$; faróis
separados até $1.5/6.7\times10^{-6} \approx \qty{220}{km}$ (em
princípio --- o ar e a curvatura da Terra intervêm antes). $G_r = 3\times10^{-4}/
6.7\times10^{-6} = 45$.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:7}
$D = 50/2 = \qty{25}{mm}$. Exposição $\propto N^2$: $(8/2)^2 = 16$ vezes
mais longa, $16/500 \approx 1/30$~\unit{s}. Ela traz \omterm{def:b1:optical-instruments:camera}{profundidade de campo} (e
menos aberrações) ao preço de borrão de movimento ou de um tripé.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:8}
$H = 35^2/(11 \times 0.03) = \qty{3.7}{m}$; nítido de $H/2 =
\qty{1.9}{m}$ ao infinito. Pré-ajustada em $H$, a \omterm{def:b1:optical-instruments:camera}{câmera} dispensa focalizar:
a foto pode ser feita no instante em que a cena acontece.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:9}
$8\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} $30/8 = \qty{3.8}{mm} < \qty{7}{mm}$, toda a luz
entra no \omterm{def:b1:optical-instruments:eye}{olho}. $7\times50$: \qty{7.1}{mm}, casando com a pupila noturna,
e $(50/30)^2 = 2.8$ vezes mais luz recolhida: o par para o anoitecer.
$20\times30$: \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} \qty{1.5}{mm}, imagem fraca, campo estreito e
tremor de mão ampliado vinte vezes.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:10}
\omterm{def:b1:mirrors-thin-lenses:lens}{Lente} de contato: $f' = \qty{-25}{cm}$, $V = \qty{-4.0}{\delta}$. Óculos:
o \omterm{def:b1:optical-instruments:punctum}{ponto remoto} fica a $25 - 1.5 = \qty{23.5}{cm}$ da \omterm{def:b1:mirrors-thin-lenses:lens}{lente}, logo $f' =
\qty{-23.5}{cm}$, $V = \qty{-4.3}{\delta}$ --- mais forte. A \omterm{def:b1:mirrors-thin-lenses:lens}{lente} deve
formar a imagem no \omterm{def:b1:optical-instruments:punctum}{ponto remoto} do \omterm{def:b1:optical-instruments:eye}{olho}, cuja distância à \omterm{def:b1:mirrors-thin-lenses:lens}{lente}
depende de onde a \omterm{def:b1:mirrors-thin-lenses:lens}{lente} se encontra.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:11}
Afocal: $F_1' = F_2$, separação $f_1' + f_2' = \qty{20}{cm}$ (mais curta
que os \qty{40}{cm} de uma kepleriana de mesmo $G$). $G = -30/(-10) =
+3$: direita. \omterm{prop:b1:optical-instruments:telescope}{Pupila de saída} $\overline{O_2P'} = f_2'(f_1' + f_2')/f_1' =
-10 \times 20/30 = \qty{-6.7}{cm}$: virtual, dentro do tubo, onde o
\omterm{def:b1:optical-instruments:eye}{olho} não pode estar; o \omterm{def:b1:optical-instruments:eye}{olho} atrás da \omterm{def:b1:optical-instruments:microscope}{ocular} capta apenas os feixes que
por acaso o alcançam --- campo estreito, a queixa clássica contra os binóculos
de teatro.
\end{solution}

\begin{solution}{exo:b1:optical-instruments:12}
$(200/7)^2 = 820$; $\Delta m = 2.5\log_{10}820 = 7.3$: magnitude
limite $13.3$. Como o fluxo $\propto 1/r^2$, a lâmpada é vista
$\sqrt{820} = 29$ vezes mais longe.
\end{solution}

\begin{solution}{pb:b1:optical-instruments:1}
\textbf{1.} $F_2 = F_1'$: separação $f_1' + f_2' = \qty{1225}{mm}$
(\omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}) ou \qty{1206}{mm} (\qty{6}{mm}).

\textbf{2.} $L_1$: $\overline{O_1A_1} = f_1'$ (imagem em $F_1'$). $L_2$:
$\overline{O_2A_1} = -f_2'$, logo $1/\overline{O_2A'} = 1/f_2' + 1/(-f_2')
= 0$: imagem no infinito.

\textbf{3.} O feixe focaliza no plano focal comum à altura $h =
f_1'\alpha$ (raio que passa por $O_1$); dali, o raio que passa por $O_2$ fixa
a direção emergente, $\alpha' = -h/f_2' = -(f_1'/f_2')\alpha$.

\textbf{4.} $G = -48$ e $-200$; Lua: $25^\circ$ e $104^\circ$.

\textbf{5.} Em cima e embaixo nada significam no céu; uma luneta terrestre
acrescenta um par de prismas endireitadores (ou um relé de \omterm{def:b1:mirrors-thin-lenses:lens}{lentes}).

\textbf{6.} No plano focal de $L_1$, diâmetro $f_1'\theta = 1200 \times
9.1\times10^{-3} = \qty{11}{mm}$.

\textbf{7.} Objeto $O_1$ em $\overline{O_2O_1} = -(f_1' + f_2')$:
$1/\overline{O_2P'} = 1/f_2' - 1/(f_1' + f_2') = f_1'/[f_2'(f_1' + f_2')]$;
aumento $\overline{O_2P'}/\overline{O_2O_1} = -f_2'/f_1' = 1/G$,
diâmetro $D/\abs G$.

\textbf{8.} \qty{25}{mm}: $\overline{O_2P'} = 25 \times 1225/1200 =
\qty{25.5}{mm}$, diâmetro $150/48 = \qty{3.1}{mm}$. \qty{6}{mm}:
\qty{6.0}{mm} atrás, diâmetro \qty{0.75}{mm}.

\textbf{9.} Todo raio que entrou na \omterm{def:b1:optical-instruments:microscope}{objetiva} passa pela \omterm{prop:b1:optical-instruments:telescope}{pupila de
saída}: um \omterm{def:b1:optical-instruments:eye}{olho} ali colocado recebe o campo inteiro. Mais atrás, ele
intercepta apenas os feixes centrais: o campo encolhe a um buraco de fechadura.

\textbf{10.} $D/\abs G \leq \qty{7}{mm}$: $\abs G \geq 150/7 \approx 21$.

\textbf{11.} $\theta_{\min} = 1.22 \times 5.5\times10^{-7}/0.150 =
\qty{4.5e-6}{rad} = 0.92''$.

\textbf{12.} $4.5\times10^{-6} \times 3.84\times10^5 = \qty{1.7}{km}$;
a \omterm{def:b1:optical-instruments:eye}{olho} nu $3\times10^{-4} \times 3.84\times10^5 = \qty{115}{km}$.

\textbf{13.} $G_r = 3\times10^{-4}/4.5\times10^{-6} = 67$; a \omterm{def:b1:optical-instruments:microscope}{ocular} de \qty{25}{mm}
($48$) é a mais próxima, ligeiramente abaixo; a de \qty{6}{mm} ($200$) fica
três vezes acima.

\textbf{14.} Em $G = 200$ a mancha de difração subtende $200 \times 4.5
\times10^{-6} = \qty{9e-4}{rad} > \epsilon$: visivelmente difusa. A luz da
Lua se espalha sobre uma área $(200/48)^2 = 17$ vezes maior na
\omterm{def:b1:optical-instruments:eye}{retina}: mais fraca.

\textbf{15.} $1'' = \qty{4.85e-6}{rad}$: $D = 1.22\lambda/\theta =
\qty{140}{mm}$. Além disso, é a turbulência que limita a resolução a partir do solo
(salvo óptica adaptativa); um $D$ maior ainda recolhe mais luz,
alcançando objetos mais fracos.

\textbf{16.} $(150/7)^2 = 460$.

\textbf{17.} $\Delta m = 2.5\log_{10}460 = 6.7$: magnitude $12.7$.

\textbf{18.} $P = 2\times10^{-3} \times \pi(0.075)^2 = \qty{3.5e-5}{W}$,
$460$ vezes a do \omterm{def:b1:optical-instruments:eye}{olho} nu. A imagem retiniana tem área $\abs G^2$ vezes
maior: razão de brilho superficial $460/G^2 = 0.20$ ($G = 48$),
$0.012$ ($G = 200$) --- igual a $(D/\abs G)^2/d_p^2$, o quadrado da razão
entre a \omterm{prop:b1:optical-instruments:telescope}{pupila de saída} e a pupila do \omterm{def:b1:optical-instruments:eye}{olho}.

\textbf{19.} Para um objeto extenso a luz recolhida cresce como $D^2$,
mas a área da imagem cresce como $G^2 \geq (D/d_p)^2$: o brilho superficial pode,
no máximo, igualar o valor a \omterm{def:b1:optical-instruments:eye}{olho} nu (\omterm{prop:b1:optical-instruments:telescope}{pupila de saída} $=$ pupila do \omterm{def:b1:optical-instruments:eye}{olho}). Uma estrela
não é resolvida: toda a sua luz cai numa única mancha de difração, de modo que o seu
brilho cresce como $(D/d_p)^2$.

\textbf{20.} Um ponto a um ângulo $\beta$ é conjugado à altura $f_1'\beta$ no
plano focal; o diafragma deixa passar $\abs{f_1'\beta} \leq \phi/2$: campo
total $\phi/f_1'$ (ângulos pequenos).

\textbf{21.} $20/1200 = \qty{0.0167}{rad} = 0.95^\circ$: a Lua inteira
cabe. $5/1200 = 0.24^\circ$: não cabe.

\textbf{22.} Campo aparente $\phi/f_2'$: $20/25 = \qty{0.8}{rad} = 46^\circ$;
$5/6 = 48^\circ$.

\textbf{23.} $0.95^\circ = 3420''$: $3420/15 = \qty{230}{s}$, quase quatro
minutos; $0.24^\circ = 864''$: \qty{58}{s}.

\textbf{24.} \omterm{def:b1:optical-instruments:microscope}{Ocular} de \qty{25}{mm} ($G = 48$, perto de $G_r$, brilhante, Lua
inteira) para o disco; a de \qty{6}{mm} ($G = 200$) para olhar as crateras de perto,
mais fraca e mais difusa. O número do instrumento: crateras até
\qty{1.7}{km}.

\textbf{25.} Mesmos $f_1'$ e $D$: mesmos $G$, campos, \omterm{prop:b1:optical-instruments:telescope}{pupilas de saída},
resolução e luz (descontada uma eventual obstrução central). Ganhos: nenhuma
aberração cromática. Mudança prática: o foco fica à frente do
espelho, de modo que um pequeno espelho secundário precisa desviar o feixe para
a \omterm{def:b1:optical-instruments:microscope}{ocular}.
\end{solution}
