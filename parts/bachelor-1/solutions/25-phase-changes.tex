\section*{Chapter \ref{ch:b1:phase-changes} --- Phase Changes of a Pure Substance}

\begin{solution}{exo:b1:phase-changes:1}
(a) Vapour: \qty{0.5}{kPa} is below $P_s(\qty{20}{\degreeCelsius}) = \qty{2.3}{kPa}$.
(b) Solid. (c) Above $T_c = \qty{374}{\degreeCelsius}$: supercritical fluid, no
liquid--gas distinction. (d) The \omterm{prop:b1:phase-changes:PTdiagram}{triple point}: all three. The puddle
evaporates from its surface as long as the vapour pressure in the air
is below $P_s(\qty{20}{\degreeCelsius})$; boiling is only the case where
bubbles can form inside the liquid.
\end{solution}

\begin{solution}{exo:b1:phase-changes:2}
$0.1 \times 2257 + 0.1 \times 4.18\,(100 - T_f) = 1.0 \times 4.18\,(T_f - 20)$:
$351.1 = 4.60\,T_f$, $T_f = \qty{76}{\degreeCelsius}$. With hot water
instead: $100 - T_f = 10(T_f - 20)$, $T_f = \qty{27}{\degreeCelsius}$ --- the
\omterm{def:b1:phase-changes:phase}{latent heat} is worth \qty{540}{K} of sensible heat.
\end{solution}

\begin{solution}{exo:b1:phase-changes:3}
$v = \qty{2.0e-3}{m^3/kg}$; $x = (2.0 - 1.09) \times 10^{-3}/0.392 = 2.3 \times 10^{-3}$:
\qty{2.3}{g} of vapour filling \qty{0.91}{L}; the \qty{0.998}{kg} of liquid
occupy $0.998 \times 1.09 = \qty{1.09}{L}$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:4}
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, i.e.\ $\times 0.018 = \qty{109}{J/(mol.K)}$;
fusion $334/273 \times 0.018 = \qty{22}{J/(mol.K)}$. Water's $109$ is well
above Trouton's $88$: hydrogen bonds make liquid water more ordered
than an ordinary liquid, so vaporizing it gains more disorder.
\end{solution}

\begin{solution}{exo:b1:phase-changes:5}
$1/T = 1/373.15 - \ln(P/P_0)/4890$. Everest: $\ln(0.33/1.013) = -1.12$,
$1/T = 2.680 \times 10^{-3} + 2.29 \times 10^{-4}$, $T = \qty{344}{K} =
\qty{71}{\degreeCelsius}$. At \qty{0.80}{bar}: $\ln = -0.236$, $T = \qty{366}{K} =
\qty{93}{\degreeCelsius}$. At \qty{71}{\degreeCelsius} the white sets slowly
and the yolk hardly at all: the \qty{3}{min} egg becomes a
\qty{10}{min} egg or more.
\end{solution}

\begin{solution}{exo:b1:phase-changes:6}
$\ln(1.8/1.013) = 0.575$: $1/T = 2.680 \times 10^{-3} - 1.18 \times 10^{-4}$,
$T = \qty{390}{K} = \qty{117}{\degreeCelsius}$; at \qty{2.5}{bar}: $\ln = 0.903$,

\end{solution}

\begin{solution}{exo:b1:phase-changes:7}
$\dd P/\dd T = 3.34 \times 10^5/[273 \times (-9.1 \times 10^{-5})] = \qty{-1.3e7}{Pa/K}
= \qty{-130}{bar/K}$: ice melts at \qty{-1}{\degreeCelsius} under about
\qty{130}{bar}. Under \qty{3}{km} of ice, $P = 917 \times 9.8 \times 3000 =
\qty{270}{bar}$: melting at \qty{-2}{\degreeCelsius}; a glacier bed can be at
its melting point while the surface is far colder, and the film of
meltwater lets the ice sheet slide.
\end{solution}

\begin{solution}{exo:b1:phase-changes:8}
$W = -P(v_v - v_l) = -1.013 \times 10^5 \times 1.672 = \qty{-169}{kJ}$ (the water
pushes the atmosphere back); $Q = L_v = \qty{2257}{kJ}$; $\Delta U = 2257 - 169
= \qty{2088}{kJ}$; $\Delta H = \qty{2257}{kJ}$; $\Delta S = 2257/373 = \qty{6.05}{kJ/K}$.
From a flame at \qty{1500}{K}: $S_{\mathrm{exch}} = 2257/1500 = \qty{1.50}{kJ/K}$,
created \qty{4.5}{kJ/K} --- most of the flame's quality is wasted heating
water.
\end{solution}

\begin{solution}{exo:b1:phase-changes:9}
$P_s(\qty{22}{\degreeCelsius}) \approx 2.34 + 0.4 \times 0.83 = \qty{2.67}{kPa}$,
$P_v = 0.70 \times 2.67 = \qty{1.87}{kPa}$; \omterm{ex:b1:phase-changes:dew}{dew point} where $P_s = 1.87$: $15 +
5 \times 0.17/0.64 = \qty{16}{\degreeCelsius}$. Mass: $\rho_v = 1870 \times 0.018/
(8.314 \times 295) = \qty{13.7}{g/m^3}$, \qty{0.69}{kg} in the room. Pane at
\qty{12}{\degreeCelsius}: $P_s \approx \qty{1.42}{kPa} < 1.87$, it fogs. At
\qty{25}{\degreeCelsius} the humidity falls to $1.87/3.17 = 59\%$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:10}
(a) $h$ is a state function: going solid $\to$ liquid $\to$ vapour at the
\omterm{prop:b1:phase-changes:PTdiagram}{triple point} or solid $\to$ vapour directly gives the same $\Delta h$:
$L_s = L_f + L_v$. (b) With $v_s, v_l \ll v_v$ both slopes are $L/(Tv_v)$ at
the same $T$ and $v_v$: their ratio is $L_s/L_v > 1$. (c) $v_v = RT/(MP) =
8.314 \times 273.16/(0.018 \times 611) = \qty{206}{m^3/kg}$; vaporization
$2.50 \times 10^6/(273.16 \times 206) = \qty{44}{Pa/K}$, sublimation $2.83 \times 10^6/
(273.16 \times 206) = \qty{50}{Pa/K}$. (d) Below \qty{0}{\degreeCelsius} at
\qty{1}{bar} no liquid exists; if the air's vapour pressure is below
$P_s$ over ice (\qty{260}{Pa} at \qty{-10}{\degreeCelsius}) the snow
sublimates --- the dry cold wind eats it.
\end{solution}

\begin{solution}{exo:b1:phase-changes:11}
(a) $P_s(\qty{105}{\degreeCelsius}) \approx 1.013 + 5 \times 0.036 = \qty{1.19}{bar}$:
$\Delta P = \qty{0.18}{bar}$, $r^\ast = 2 \times 0.059/(1.8 \times 10^4) = \qty{6.6}{\micro m}$.
(b) $\Delta P = \qty{3.6}{kPa}$, $r^\ast = \qty{33}{\micro m}$. (c) $x = c\,\Delta T/L_v =
4.18 \times 5/2257 = 0.93\%$: \qty{9.3}{g} per litre, i.e.\ $9.3 \times 10^{-3}
\times 1.67 = \qty{16}{L}$ of steam produced in a fraction of a second inside
a cup --- the water is blown out, boiling. (d) Chips hold air in their
pores: bubbles of tens of micrometres already exist, so boiling starts
as soon as $P_s$ reaches $P_0$.
\end{solution}

\begin{solution}{exo:b1:phase-changes:12}
(a) After the valve $h = \qty{256}{kJ/kg}$ at \qty{-10}{\degreeCelsius}: $x =
(256 - 187)/(392 - 187) = 0.34$. (b) Evaporator $q_c = 392 - 256 =
\qty{136}{kJ/kg}$; compressor $w = 430 - 392 = \qty{38}{kJ/kg}$; condenser
$q_h = 256 - 430 = \qty{-174}{kJ/kg}$; $136 + 38 - 174 = 0$. (c) $e_p =
174/38 = 4.6$, $e_f = 136/38 = 3.6$; Carnot $313/50 = 6.3$ and $263/50 =
5.3$: about $0.7$ of the ideal. (d) $\dot m = 6000/174000 = \qty{34}{g/s}$,
\qty{2.1}{kg} a minute.
\end{solution}

\begin{solution}{pb:b1:phase-changes:1}
\textbf{1.} At \qty{1}{atm} water boils at \qty{100}{\degreeCelsius}; while
liquid remains, any heat supplied makes vapour at that temperature
(\omterm{def:b1:phase-changes:phase}{latent heat}) instead of warming the liquid: the temperature is pinned
by the pressure.

\textbf{2.} $\dd P_s/\dd T = L_v/[T(v_v - v_l)]$ with $v_l \ll v_v = RT/(MP_s)$
(perfect-gas vapour): $\dd P_s/P_s = (ML_v/R)\,\dd T/T^2$; with $L_v$ taken
constant, integrate: $\ln(P_s/P_0) = (ML_v/R)(1/T_0 - 1/T)$.

\textbf{3.} $1/T = 1/373.15 - \ln(1.8/1.013)/4890 = 2.562 \times 10^{-3}$: $T =
\qty{390}{K} = \qty{117}{\degreeCelsius}$.

\textbf{4.} $m = \Delta P\,A/g = 0.8 \times 10^5 \times 2.0 \times 10^{-5}/9.8 =
\qty{0.16}{kg}$.

\textbf{5.} $Q = 2.0 \times 4.18 \times 97 = \qty{811}{kJ}$; $t = 811/2.0 =
\qty{405}{s} \approx \qty{7}{min}$.

\textbf{6.} $2000/(2.2 \times 10^6) = \qty{0.91}{g/s} = \qty{55}{g/min}$ --- two
litres gone in \qty{37}{min}. Turn the plate down to the least that
keeps the valve hissing: the temperature is set by the pressure, not by
the flame.

\textbf{7.} $+\qty{17}{K}$: $2^{1.7} = 3.2$ times faster, \qty{30}{min}
$\to$ \qty{9}{min}. Everest, $-\qty{30}{K}$: $2^{-3}$, four hours.

\textbf{8.} \omterm{ex:b1:phase-changes:dew}{Relative humidity} $= P_v/P_s(T)$. $P_v = 0.60 \times 3.17 =
\qty{1.90}{kPa}$; $\rho_v = P_vM/(RT) = 1900 \times 0.018/(8.314 \times 298) =
\qty{13.8}{g/m^3}$.

\textbf{9.} $P_s(T_d) = \qty{1.90}{kPa}$: between \qty{15}{\degreeCelsius}
(\qty{1.70}) and \qty{20}{\degreeCelsius} (\qty{2.34}): $T_d = 15 + 5 \times 0.20/0.64
= \qty{16.6}{\degreeCelsius}$. Grass radiating to the night sky, or a
bottle from the \omterm{def:b1:heat-engines:machine}{refrigerator}, falls below $T_d$: the vapour in contact
with it is then above saturation and condenses.

\textbf{10.} $TP^{(1-\gamma)/\gamma} = \text{const}$: $\dd T/T = \frac{\gamma - 1}{\gamma}
\dd P/P$; $\dd P = -\rho g\,\dd z$ with $\rho = PM/(RT)$: $\dd T/T = -\frac{\gamma -
1}{\gamma}\frac{Mg}{RT}\dd z$, so $\dd T/\dd z = -\frac{\gamma - 1}{\gamma}\frac{Mg}{R} =
-g/c_p$ since $c_p = \gamma R/[(\gamma - 1)M]$. Value $9.8/1000 = \qty{9.8}{K/km}$.

\textbf{11.} $T - T_d$ closes at $9.8 - 2 = \qty{7.8}{K/km}$: $z = (25 - 16.6)/
7.8 = \qty{1.1}{km}$.

\textbf{12.} $1 \times 2.45 = \qty{2.45}{kJ}$ per kilogram of air: $\Delta T =
\qty{2.45}{K}$. The release offsets part of the adiabatic cooling: a
saturated parcel cools at only \qtyrange{5}{6}{K/km}, stays warmer and
lighter than its surroundings and keeps rising --- the \omterm{def:b1:phase-changes:phase}{latent heat} is
the fuel of the thunderstorm.

\textbf{13.} $10^9 \times \qty{1}{g} = \qty{1000}{t}$; $10^6 \times 2.45 \times 10^6 =
\qty{2.5e12}{J} = \qty{0.68}{GW h}$ --- forty minutes of a large power plant.

\textbf{14.} $P_v = 0.80 \times 3.17 = \qty{2.54}{kPa}$; $P_s(\qty{18}{\degreeCelsius})
\approx 1.70 + 0.6 \times 0.64 = \qty{2.08}{kPa} < 2.54$: it fogs, until the
humidity drops below $2.08/3.17 = 66\%$.

\textbf{15.} $P = P_s(\qty{20}{\degreeCelsius}) = \qty{2.3}{kPa}$; $v = \qty{2.0e-3}{m^3/kg}$;
$v_v = RT/(MP_s) = 8.314 \times 293/(0.018 \times 2340) = \qty{58}{m^3/kg}$; $x =
(2.0 - 1.0) \times 10^{-3}/58 = 1.7 \times 10^{-5}$: \qty{9}{mg} of vapour;
liquid $0.50 \times 10^{-3} = \qty{0.50}{L}$, half the vessel.

\textbf{16.} $v = \qty{2.0e-3}{m^3/kg} < v_c$: the isochore meets the liquid
branch of the dome --- the liquid expands faster than it evaporates, the
level \emph{rises}, and the vessel is full of liquid when $v_l(T) =
\qty{2.0e-3}{m^3/kg}$, a little above \qty{360}{\degreeCelsius} (about
\qty{190}{bar}). Beyond, a nearly incompressible liquid heated at constant
volume: the pressure climbs by tens of bar per \omterm{def:b1:kinetic-theory:temperature}{kelvin} --- the vessel
bursts. Never seal a vessel full of liquid.

\textbf{17.} $v = \qty{1.0e-2}{m^3/kg} > v_c$: the level \emph{falls}; the last
drop goes when $v_v(T) = 0.010$, between \qty{340}{\degreeCelsius} ($0.0108$)
and \qty{350}{\degreeCelsius} ($0.0088$): about \qty{344}{\degreeCelsius},
\qty{150}{bar}.

\textbf{18.} $v = \qty{3.2e-3}{m^3/kg} \approx v_c$: the meniscus stays near
mid-height, grows flatter and fainter as the two densities converge,
the fluid turns milky (critical opalescence), and at \qty{374}{\degreeCelsius},
\qty{221}{bar} the meniscus vanishes: one \omterm{def:b1:phase-changes:phase}{phase}.

\textbf{19.} Three vertical lines: $v = 2.0 \times 10^{-3}$ leaves the dome
through the liquid branch, left of C; $10^{-2}$ through the vapour branch,
right of C; $3.2 \times 10^{-3}$ through the summit.

\textbf{20.} Air: \qty{0.5}{L} at \qty{1.0}{bar}, \qty{293}{K}, same volume at
\qty{373}{K}: $P_{\text{air}} = 1.0 \times 373/293 = \qty{1.27}{bar}$; plus
$P_s = \qty{1.01}{bar}$: \qty{2.3}{bar} inside, \qty{1.3}{bar} above the
outside: $1.3 \times 10^5 \times 10^{-2} = \qty{1.3}{kN}$ on the lid --- the
weight of \qty{130}{kg}.

\textbf{21.} Supercooled, metastable: no \omterm{def:b1:phase-changes:metastable}{nucleation} site (clean water,
no dust, no motion) for the first crystal. A knock nucleates ice, which
grows through the bottle releasing $L_f$; the temperature rises to
\qty{0}{\degreeCelsius} and stops there, the only temperature at which ice
and water coexist at \qty{1}{atm}: the freezing halts as soon as the
released heat has brought the mixture to it.

\textbf{22.} $x = c\,\Delta T/L_f = 4.18 \times 8/334 = 0.10$.

\textbf{23.} $\Delta s = c\ln(273/265) - xL_f/273 = 124.3 - 122.3 =
\qty{+2.0}{J/(kg.K)}$: adiabatic, hence all created; positive, as a
spontaneous process must be.

\textbf{24.} $x = 4.18 \times 5/2257 = 0.93\%$: \qty{9.3}{g} of steam per litre,
$9.3 \times 10^{-3} \times 1.67 = \qty{16}{L}$ of vapour born in the bulk in a
fraction of a second: the cup erupts.

\textbf{25.} $T_0S_{\mathrm{created}} = 273 \times 2.0 = \qty{550}{J}$ per litre,
against $mgh = \qty{9.8}{J}$: fifty-six times more. A \omterm{def:b1:phase-changes:metastable}{metastable state}
stores work --- what a reversible path could have extracted --- and a
knock turns it all into entropy.
\end{solution}
