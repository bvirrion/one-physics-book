\chapter{The First Law of Thermodynamics}\label{ch:b1:first-law}

Pump up a bicycle tire and the barrel of the pump gets too hot to hold.
A diesel engine has no spark plug: it squeezes its air twenty times
smaller, and the air becomes hot enough to light the fuel by itself. A
cylinder of compressed gas frosts over as it empties. In each case
energy changes form --- work becomes \omterm{def:b1:first-law:heat}{heat}, \omterm{def:b1:first-law:heat}{heat} becomes work, both
become \omterm{def:b1:kinetic-theory:U}{internal energy} --- and the first law of thermodynamics is the
bookkeeping that makes the accounts balance. This chapter defines work
and \omterm{def:b1:first-law:heat}{heat} for a \omterm{def:b1:first-law:transformation}{thermodynamic system}, states the first law, introduces
\omterm{def:b1:kinetic-theory:U}{internal energy} and \omterm{def:b1:first-law:enthalpy}{enthalpy} as the quantities it conserves, and applies
it to the \omterm{def:b1:first-law:transformation}{transformations} of a \omterm{def:b1:kinetic-theory:model}{perfect gas} and to calorimetry.

\section{Transformations, work, and heat}

\begin{definition}[System, transformation]\label{def:b1:first-law:transformation}
\index{thermodynamic system}\index{transformation}\index{quasi-static transformation}\index{reversible transformation}
A \emph{thermodynamic system} is the matter inside a chosen boundary;
everything else is the \emph{surroundings}. A \emph{transformation} takes
it from an equilibrium state to another. It is \emph{quasi-static} if the
system passes through a succession of equilibrium states (slow enough
for $P$, $T$ to be defined throughout), and \emph{reversible} if, in
addition, reversing the external conditions reverses the path (no
friction, no finite temperature or pressure gap with the surroundings).
Names by what stays constant: \emph{isothermal} ($T$), \emph{isobaric}
($P$), \emph{isochoric} ($V$); \emph{adiabatic} if no \omterm{def:b1:first-law:heat}{heat} is exchanged;
\emph{monobaric} (\emph{monothermal}) if the external pressure
(temperature) is constant, whatever happens inside.
\end{definition}

\begin{proposition}[Work of pressure forces]\label{prop:b1:first-law:work}
\index{work!of pressure forces}
A system whose volume changes by $\dd V$ against an external pressure
$P_{\mathrm{ext}}$ receives the work
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
For a \omterm{def:b1:first-law:transformation}{quasi-static transformation} $P_{\mathrm{ext}} = P$ (the system's own
pressure) and $W = -\int P\,\dd V$ is minus the area under the path in the
Clapeyron diagram: positive when compressed, negative when expanding.
\end{proposition}

\begin{proof}
A piston of area $S$ pushed by the surroundings with the force
$P_{\mathrm{ext}}S$ moves by $\dd x$ outward: the surroundings do the work
$-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$ on the system. Any
boundary is a collection of such pistons. Quasi-static: the piston is in
equilibrium, $P_{\mathrm{ext}} = P$ up to a vanishing difference.
\end{proof}

\begin{definition}[Heat]\label{def:b1:first-law:heat}
\index{heat}\index{thermostat}
\emph{Heat} $Q$ is energy transferred to the system other than by
macroscopic work --- by molecular collisions at a wall (conduction),
by the circulation of a fluid (convection), by radiation. A
\emph{thermostat} is a body so large that its temperature does not change
whatever heat it exchanges. Sign convention throughout: $W$ and $Q$ are
\emph{received} by the system, positive when they enter.
\end{definition}

\section{The first law}

\begin{theorem}[First law of thermodynamics]\label{thm:b1:first-law:firstlaw}
\index{first law of thermodynamics}\index{internal energy}
A \omterm{def:b1:newton-dynamics:conservation}{closed system} possesses a state function, its \emph{\omterm{def:b1:kinetic-theory:U}{internal energy}}
$U$ (extensive), such that for any \omterm{def:b1:first-law:transformation}{transformation}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
$E_k$ being the macroscopic \omterm{thm:b1:work-and-energy:ke}{kinetic energy} (zero for a system at rest):
the energy received as work and \omterm{def:b1:first-law:heat}{heat} is stored as \omterm{def:b1:kinetic-theory:U}{internal energy}. $W$
and $Q$ each depend on the path; their sum does not.
\end{theorem}
\admitted

\begin{remark}[What it says]\label{rem:b1:first-law:meaning}
It is the conservation of energy, with \omterm{def:b1:first-law:heat}{heat} recognized as a transfer of
energy (Joule's paddle-wheel experiment: a measured work always raises
the temperature of water as much as a measured \omterm{def:b1:first-law:heat}{heat}). ``$U$ is a state
function'' means $\Delta U$ is fixed by the initial and final states ---
so a cycle has $\Delta U = 0$ and $W + Q = 0$: an engine that delivers
work must receive \omterm{def:b1:first-law:heat}{heat}. The microscopic content of $U$ is that of
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Enthalpy; heat capacities]\label{def:b1:first-law:enthalpy}
\index{enthalpy}\index{heat capacity at constant pressure}
The \emph{enthalpy} $H = U + PV$ is a state function adapted to
\omterm{def:b1:first-law:transformation}{transformations} at constant pressure. The \emph{\omterm{def:b1:first-law:heat}{heat} capacities at
constant volume and pressure} are
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
in \unit{J/K}; per \omterm{def:b1:kinetic-theory:scales}{mole} $C_{V,m}$, $C_{P,m}$; per kilogram the specific
\omterm{def:b1:first-law:heat}{heats} $c_V$, $c_P$.
\end{definition}

\begin{proposition}[Heat at constant volume and at constant pressure]\label{prop:b1:first-law:qvqp}
For a system at rest exchanging work only through pressure forces:
\begin{itemize}
  \item isochoric \omterm{def:b1:first-law:transformation}{transformation}: $W = 0$ and $Q_V = \Delta U$;
  \item monobaric \omterm{def:b1:first-law:transformation}{transformation} between two states at the external
        pressure $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isochoric: $\dd V = 0$. Monobaric: $W = -P_0(V_2 - V_1)$, so $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ since $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Perfect gas: $U$, $H$, Mayer's relation]\label{prop:b1:first-law:gas}
\index{Mayer's relation}\index{adiabatic index}\index{Joule's laws}
For a \omterm{def:b1:kinetic-theory:model}{perfect gas}, $U$ and $H$ depend on $T$ alone (\omterm{prop:b1:kinetic-theory:Ugas}{Joule's laws}):
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, and
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ for monatomic gases, $7/5$ for diatomic ones at ordinary
temperatures. For a \omterm{prop:b1:kinetic-theory:condensed}{condensed phase} in the model of
\cref{prop:b1:kinetic-theory:condensed}, $C_P \approx C_V = C$ and
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$; differentiate with respect to $T$. $C_{V,m} = \tfrac32 R$ or
$\tfrac52 R$ from \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Transformations of a perfect gas}

\begin{proposition}[The four standard transformations]\label{prop:b1:first-law:transformations}
\index{isothermal transformation}\index{adiabatic transformation}\index{Laplace's law}
For $n$ \omterm{def:b1:kinetic-theory:scales}{moles} of \omterm{def:b1:kinetic-theory:model}{perfect gas}:
\begin{itemize}
  \item isochoric: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$;
  \item isobaric (quasi-static): $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$;
  \item isothermal reversible: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$;
  \item adiabatic reversible: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        and along the path (\emph{\omterm{prop:b1:first-law:transformations}{Laplace's law}})
        \[
        PV^\gamma = \text{const}, \qquad TV^{\gamma - 1} = \text{const}, \qquad
        T^\gamma P^{1 - \gamma} = \text{const} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isochoric and isobaric: the previous propositions. Isothermal: $W =
-\int nRT\,\dd V/V$. Adiabatic: $\dd U = \delta W$, i.e.\ $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, so $\dd T/T = -(\gamma - 1)\dd V/V$ (using
$R/C_{V,m} = \gamma - 1$); integrate: $TV^{\gamma - 1}$ constant; the other
forms by $PV = nRT$.
\end{proof}

\begin{example}[Pump and diesel]\label{ex:b1:first-law:diesel}
Air ($\gamma = 1.4$) compressed adiabatically from \qty{1}{bar}, \qty{300}{K}
to \qty{7}{bar} (a bicycle pump): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- \qty{250}{\degreeCelsius} for an instant, which the barrel feels. A
diesel's compression ratio of $20$: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- above the self-ignition temperature of
the fuel: the engine needs no spark (the weekend problem runs the
whole cycle).
\end{example}

\begin{remark}[Irreversible transformations]\label{rem:b1:first-law:irreversible}
\index{Joule expansion}\index{irreversible transformation}
When the \omterm{def:b1:first-law:transformation}{transformation} is not quasi-static, \omterm{prop:b1:first-law:transformations}{Laplace's law} and $W =
-\int P\,\dd V$ do not apply; only $W = -\int P_{\mathrm{ext}}\,\dd V$ and the
first law do. Two classic cases: a gas expanding into vacuum (\omterm{rem:b1:first-law:irreversible}{Joule
expansion}) receives no work and, in adiabatic walls, no \omterm{def:b1:first-law:heat}{heat} --- $\Delta U
= 0$, and a \omterm{def:b1:kinetic-theory:model}{perfect gas} keeps its temperature; a gas suddenly compressed
by an external pressure jumping to $P_2$ receives $W = -P_2\Delta V$, more
than the reversible work to the same pressure, and ends hotter
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Calorimetry}

\begin{method}[Calorimetric balance]\label{met:b1:first-law:calorimetry}
\index{calorimetry}\index{latent heat}
Bodies put in thermal contact inside an insulated vessel at constant
(atmospheric) pressure exchange \omterm{def:b1:first-law:heat}{heat} only among themselves: the total
$\Delta H = \sum Q_P = 0$. For each body $\Delta H = mc\,\Delta T$ (no phase
change) or $\pm mL$ for a change of state at its transition temperature,
with $L$ the \emph{latent \omterm{def:b1:first-law:heat}{heat}} (\omterm{def:b1:first-law:enthalpy}{enthalpy} of fusion or vaporization per
unit mass: water, $L_f = \qty{334}{kJ/kg}$ at \qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ at \qty{100}{\degreeCelsius}). Write the balance, solve
for the final temperature, and check that each body's assumed phase is
consistent with it.
\end{method}

\begin{example}[Ice in water]\label{ex:b1:first-law:ice}
\qty{50}{g} of ice at \qty{0}{\degreeCelsius} into \qty{200}{g} of water at
\qty{30}{\degreeCelsius}: melting needs \qty{16.7}{kJ}; the water can give
\qty{25.1}{kJ} in cooling to \qty{0}{\degreeCelsius}, so all the ice melts
and the balance $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ gives
$T_f = \qty{8}{\degreeCelsius}$. Had there been \qty{100}{g} of ice,
\qty{33.4}{kJ} would be needed: only part melts and $T_f = \qty{0}{\degreeCelsius}$
--- the check matters.
\end{example}

\section{Problem: From the bicycle pump to the diesel engine}

\begin{problem}[{Weekend problem --- a hand pump that burns the fingers,
a cylinder that lights its own fuel, and a steel ball that bounces on a
cushion of air: the first law in three compressions}]
\label{pb:b1:first-law:1}
Air: perfect \omterm{prop:b1:kinetic-theory:Ugas}{diatomic gas}, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$; $R = \qty{8.314}{J/(mol.K)}$. Initial
state everywhere: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Part I --- The bicycle pump.}
The pump's cylinder has a volume $V_1 = \qty{212}{cm^3}$; the air in it
is compressed reversibly and adiabatically until it reaches the tire's
pressure, \qty{7.0}{bar}.
\begin{enumerate}
  \item Amount of air in the cylinder.
  \item Volume at the end of the compression.
  \item Temperature at the end of the compression.
  \item Work received by the air during one stroke.
  \item The tire (\qty{2.0}{L}) must be brought from \qty{1}{bar} to
        \qty{7}{bar} at \qty{300}{K}: how many strokes, and how much work
        in all (take the work per stroke as constant)?
  \item The hot air then cools in the tire to \qty{300}{K}: what happens
        to the tire's pressure, and what does the cyclist do?
  \item Compare the adiabatic work per stroke with the work of an
        isothermal compression to the same \qty{7}{bar}. Why is the
        isothermal work larger although the gas ends cooler?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The diesel compression.}
A cylinder of $V_1 = \qty{0.50}{L}$ compresses its air reversibly and
adiabatically by a factor $20$.
\begin{enumerate}[resume]
  \item Amount of air, and temperature $T_2$ after compression.
  \item Pressure $P_2$.
  \item The fuel self-ignites above about \qty{500}{K}: explain why a
        diesel needs no spark plug, and why a petrol engine (ratio
        $\approx 10$, see \cref{exo:b1:first-law:3}) must not self-ignite.
  \item Work received by the air during the compression.
  \item At \qty{3000}{rpm} one cylinder compresses $25$ times per second:
        power absorbed by the compression (it is returned later in the
        cycle).
  \item Real compressions are fast (irreversibility) and lose \omterm{def:b1:first-law:heat}{heat} to the
        walls: in which direction does each effect move $T_2$?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Combustion and expansion.}
At the top of the stroke \qty{20}{mg} of fuel (heating value
\qty{43}{MJ/kg}) burn while the piston starts down, at constant pressure
$P_2$; then the gas expands reversibly and adiabatically back to $V_1$.
Treat the gas as air throughout.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Heat} released by the fuel.
  \item Temperature $T_3$ and volume $V_3$ at the end of the isobaric

  \item Work exchanged during the isobaric phase (sign!).
  \item Temperature $T_4$ at the end of the adiabatic expansion, and the

\medskip
\noindent\textbf{Part IV --- Measuring $\gamma$ with a bouncing ball.}
A steel ball of mass $m = \qty{16.5}{g}$ fits a vertical glass tube of
cross-section $S = \qty{2.0}{cm^2}$ closed at the bottom by a flask of
volume $V = \qty{10}{L}$ of air at $P = \qty{1.0}{bar}$; displaced by $x$
from its equilibrium position, it oscillates.
\begin{enumerate}[resume]
  \item Why can the compressions and expansions of the air be taken as
        adiabatic? Write the relation between a small displacement $x$
        and the pressure change $\dd P$ (\omterm{prop:b1:first-law:transformations}{Laplace's law}, linearized).
  \item Deduce the restoring force on the ball and show that its motion
        is harmonic, with $\omega_0^2 = \gamma PS^2/mV$.
  \item Compute the period, and the value of $\gamma$ one would deduce
        from a measured period of \qty{1.10}{s}.
  \item Why does friction between ball and tube, and leakage of air, bias
        the result, and in which direction?
  \item Summarize the three compressions: which quantity the first law
        conserved in each, and what the adiabatic exponent $\gamma$
        governed.
\end{enumerate}
\end{problem}
