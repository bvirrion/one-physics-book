\chapter{Phase Changes of a Pure Substance}\label{ch:b1:phase-changes}

Water is the one substance everybody has seen in all three states,
and it is the working fluid of most of the world's power stations, the
\omterm{def:b1:heat-engines:vapourcycle}{refrigerant} of the first ice machines, and the engine of the weather.
Ice melts at a temperature that does not budge while the heat pours
in; a kettle boils at \qty{100}{\degreeCelsius} at the seaside and at
\qty{70}{\degreeCelsius} on Everest; a sealed bottle of very cold water
freezes solid at a tap. This chapter draws the map of the \omterm{def:b1:phase-changes:phase}{phases} of a
\omterm{def:b1:phase-changes:phase}{pure substance} --- the $(T, P)$ and $(v, P)$ diagrams --- and gives the
three tools that go with it: the lever rule (how much of each \omterm{def:b1:phase-changes:phase}{phase}),
the \omterm{def:b1:phase-changes:phase}{latent heat} (what a change costs, in energy and in entropy) and the
\omterm{thm:b1:phase-changes:clapeyron}{Clausius--Clapeyron relation} (how the boundary lines slope). It ends
where the map fails: \omterm{def:b1:phase-changes:metastable}{metastable states}, which are where the weather
and the pressure cooker's surprises come from.

\section{Phases and the $(T, P)$ diagram}

\begin{definition}[Pure substance, phase, phase change]\label{def:b1:phase-changes:phase}
\index{phase}\index{phase change}\index{pure substance}\index{latent heat}
A \emph{pure substance} is a body of a single chemical species. A
\emph{phase} is a part of it that is homogeneous in its physical
properties: solid, liquid, gas (\emph{vapour} when it coexists with its
liquid). A \emph{phase change} (melting/solidification, vaporization/
condensation, sublimation/deposition) is the passage from one phase to
another; at fixed pressure it occurs at a fixed temperature, with heat
exchanged but no temperature change: the \emph{latent heat}.
\end{definition}

\begin{proposition}[Equilibrium diagram]\label{prop:b1:phase-changes:PTdiagram}
\index{phase diagram}\index{triple point}\index{critical point}\index{saturation vapour pressure}
In the $(T, P)$ plane each \omterm{def:b1:phase-changes:phase}{phase} occupies a region; two \omterm{def:b1:phase-changes:phase}{phases} coexist
only along a curve (sublimation, fusion, vaporization), all three only
at one point, the \emph{\omterm{prop:b1:phase-changes:PTdiagram}{triple point}}. The vaporization curve
$P = P_s(T)$ --- the \emph{\omterm{prop:b1:phase-changes:PTdiagram}{saturation vapour pressure}} --- ends at the
\emph{\omterm{prop:b1:phase-changes:PTdiagram}{critical point}} $(T_c, P_c)$, beyond which liquid and gas are no
longer distinct. For water: \omterm{prop:b1:phase-changes:PTdiagram}{triple point} \qty{273.16}{K}, \qty{611}{Pa};
\omterm{prop:b1:phase-changes:PTdiagram}{critical point} \qty{647}{K}, \qty{221}{bar}. The fusion curve of water
leans to the left (ice is less dense than water); for almost every
other substance it leans to the right.
\end{proposition}
\admitted

\begin{remark}[Boiling, evaporation and the pressure cooker]\label{rem:b1:phase-changes:boiling}
A liquid \emph{evaporates} at any temperature from its surface, as long
as the partial pressure of its vapour above it is below $P_s(T)$; it
\emph{boils} when $P_s(T)$ reaches the ambient pressure, so that bubbles
of vapour can grow inside it. Water boils at \qty{100}{\degreeCelsius}
because $P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$; at
\qty{0.33}{bar} on Everest it boils near \qty{70}{\degreeCelsius}, in a
pressure cooker at \qty{1.8}{bar} near \qty{117}{\degreeCelsius}. The
temperature of a boiling liquid is pinned by the pressure --- a fact
used to define the Celsius scale for two centuries.
\end{remark}

\section{The $(v, P)$ diagram and the lever rule}

\begin{proposition}[Andrews isotherms]\label{prop:b1:phase-changes:andrews}
\index{Andrews isotherms}\index{saturation curve}
In the Clapeyron plane $(v, P)$ of the specific volume, an isotherm
$T < T_c$ has three parts: a steep branch (liquid, almost
incompressible), a horizontal \emph{plateau} at $P = P_s(T)$ where liquid
and vapour coexist, and a hyperbola-like branch (vapour). The plateau
ends are the specific volumes $v_l(T)$ and $v_v(T)$ of the saturated
liquid and vapour; their locus is the \emph{\omterm{prop:b1:phase-changes:andrews}{saturation curve}} (the
dome), whose summit is the \omterm{prop:b1:phase-changes:PTdiagram}{critical point}, where the isotherm $T_c$
has a horizontal inflection. Above $T_c$ no plateau exists.
\end{proposition}
\admitted

\begin{theorem}[Lever rule]\label{thm:b1:phase-changes:lever}
\index{lever rule}\index{quality (vapour mass fraction)}
A mass $m$ of specific volume $v$ on the plateau at $T$ splits into a
vapour mass fraction $x = m_v/m$ (the \emph{quality}) and a liquid
fraction $1 - x$ with
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{i.e.}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
the ratio of the segments cut on the plateau by the state point $M$.
The same rule holds for every extensive quantity per unit mass ($u$,
$h$, $s$): $h = (1 - x)h_l + xh_v$.
\end{theorem}

\begin{proof}
Volumes add: $V = m_lv_l + m_vv_v$; divide by $m$. Energy, \omterm{def:b1:first-law:enthalpy}{enthalpy} and
entropy are extensive as well.
\end{proof}

\begin{example}[A kettle and a boiler]\label{ex:b1:phase-changes:lever}
Water at \qty{100}{\degreeCelsius}, \qty{1}{atm}: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- a ratio of $1600$. A closed \qty{1.0}{L}
vessel holding \qty{0.50}{kg} of water at that temperature has $v =
\qty{2.0e-3}{m^3/kg}$ and $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$:
\qty{0.29}{g} of vapour occupying half the vessel. Doubling the vapour
mass in a sealed vessel barely moves the state point; the pressure is
fixed by the temperature alone.
\end{example}

\section{Energetics of a phase change}

\begin{proposition}[Latent heat, enthalpy and entropy of transition]\label{prop:b1:phase-changes:latent}
\index{enthalpy of transition}\index{entropy of transition}
At the temperature $T$ and pressure $P_s(T)$ of coexistence, the
transition of unit mass from \omterm{def:b1:phase-changes:phase}{phase} 1 to \omterm{def:b1:phase-changes:phase}{phase} 2 is reversible,
isothermal and isobaric, and
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
$L_{12} > 0$ when going toward the less ordered \omterm{def:b1:phase-changes:phase}{phase} (fusion,
vaporization, sublimation), and $L_{21} = -L_{12}$. For water at \qty{1}{atm}:
$L_f = \qty{334}{kJ/kg}$ at \qty{0}{\degreeCelsius}, $L_v = \qty{2257}{kJ/kg}$ at
\qty{100}{\degreeCelsius}.
\end{proposition}

\begin{proof}
At constant pressure the heat received is $\Delta H$
(\cref{prop:b1:first-law:qvqp}); reversible and isothermal, it is
$T\Delta S$ (\cref{prop:b1:second-law-entropy:identity}); and $U = H - PV$.
\end{proof}

\begin{example}[Where the heat of vaporization goes]\label{ex:b1:phase-changes:whereheat}
For water at \qty{100}{\degreeCelsius}: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, so $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$:
$93\%$ of the \omterm{def:b1:phase-changes:phase}{latent heat} goes into tearing the molecules apart, $7\%$
into pushing the atmosphere back. The entropy of vaporization,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, dwarfs that of fusion, $334/273 =
\qty{1.22}{kJ/(kg.K)}$: melting loosens the molecules, boiling frees them.
\end{example}

\begin{method}[Calorimetry with phase changes]\label{meth:b1:phase-changes:calorimetry}
In an insulated vessel at constant pressure, $\sum\Delta H_i = 0$. Guess
the final state (which \omterm{def:b1:phase-changes:phase}{phases} are present, at what temperature);
write each body's $\Delta H$ as sensible heats $mc\,\Delta T$ plus \omterm{def:b1:phase-changes:phase}{latent
heats} $\pm mL$ for the changes it undergoes; solve; \emph{check the
guess}: a body ending above its boiling point or below its melting
point, or a negative mass of a \omterm{def:b1:phase-changes:phase}{phase}, means the guess was wrong --- the
final state is then a mixture at the transition temperature, and the
unknown becomes the mass transformed (\cref{ex:b1:first-law:ice}).
\end{method}

\begin{example}[Steam burns]\label{ex:b1:phase-changes:steam}
\qty{10}{g} of steam at \qty{100}{\degreeCelsius} condensing on skin gives
$0.010 \times 2257 = \qty{22.6}{kJ}$ before it has cooled one degree ---
as much as \qty{10}{g} of boiling water gives in cooling by
\qty{540}{K}. This is why steam scalds so much worse than water at the
same temperature, and why a steam radiator needs so little mass flow.
\end{example}

\section{The Clausius--Clapeyron relation}

\begin{theorem}[Clausius--Clapeyron]\label{thm:b1:phase-changes:clapeyron}
\index{Clausius--Clapeyron relation}
Along a coexistence curve of \omterm{def:b1:phase-changes:phase}{phases} 1 and 2,
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
Take unit mass round an infinitesimal \omterm{def:b1:heat-engines:carnotcycle}{Carnot cycle} inside the dome:
complete transition $1 \to 2$ along the plateau at $T + \dd T$, pressure
$P_s + \dd P_s$ (heat received $L_{12}$ to first order); an infinitesimal
adiabatic step down to $T$; the reverse transition along the plateau at
$T$, $P_s$; an adiabatic step back. The cycle is reversible, ditherm
between $T + \dd T$ and $T$, so its \omterm{def:b1:heat-engines:efficiency}{efficiency} is $\dd T/T$
(\cref{thm:b1:heat-engines:carnot}); its work, the area of the thin
rectangle, is $\dd P_s\,(v_2 - v_1)$. Hence $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$.
\end{proof}

\begin{corollary}[Saturation vapour pressure of a liquid]\label{cor:b1:phase-changes:vapourpressure}
If the vapour is a \omterm{def:b1:kinetic-theory:model}{perfect gas} and $v_v \gg v_l$, so that $v_v - v_l \approx
RT/(MP_s)$, and if $L_v$ varies little,
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
the vapour pressure rises roughly exponentially with temperature. For
water near \qty{100}{\degreeCelsius}, $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ and $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$: each \omterm{def:b1:kinetic-theory:temperature}{kelvin} of extra boiling temperature costs
\qty{36}{mbar} of extra pressure.
\end{corollary}

\begin{proof}
Substitute $v_v - v_l \approx RT/(MP_s)$ in the theorem and separate
variables; integrate with $L_v$ constant.
\end{proof}

skater on one blade of \qty{30}{cm} by \qty{3}{mm} ($\qty{9}{cm^2}$) exerts
about \qty{8}{bar}, lowering the melting point by \qty{0.06}{K} --- pressure melting is not why skates glide (a thin
surface layer of ice is liquid-like on its own, and friction heats);
but the negative slope is real, and it is why a wire loaded with
weights cuts through a block of ice that refreezes behind it.
\end{example}

\section{Metastable states}

\begin{definition}[Supercooling, superheating]\label{def:b1:phase-changes:metastable}
\index{supercooling}\index{superheating}\index{metastable state}\index{nucleation}
A \omterm{def:b1:phase-changes:phase}{phase} can persist beyond its equilibrium boundary: liquid water
cooled below \qty{0}{\degreeCelsius} without freezing (\emph{supercooling}),
heated above its boiling point without boiling (\emph{superheating}),
vapour compressed beyond $P_s$ without condensing (supersaturation).
Such \emph{metastable states} require the absence of \emph{nucleation}
sites --- dust, scratches, dissolved gas, ice crystals --- on which the
new \omterm{def:b1:phase-changes:phase}{phase} can start; a disturbance tips them suddenly and irreversibly
to equilibrium.
\end{definition}

\begin{example}[The supercooled bottle]\label{ex:b1:phase-changes:supercooled}
Still water at \qty{-8}{\degreeCelsius} in a clean bottle, knocked: ice
crystals shoot through it and the temperature jumps to
\qty{0}{\degreeCelsius}. The process is adiabatic and fast, so the sensible
heat $c\,\Delta T$ freezes a fraction $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ of the water; the entropy created, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, is positive as it must be. Clouds are
full of supercooled droplets down to \qty{-40}{\degreeCelsius}; aircraft
icing and the seeding of clouds both exploit that.
\end{example}

\begin{remark}[Boiling chips and bubble chambers]\label{rem:b1:phase-changes:nucleation}
A bubble of radius $r$ in a liquid at pressure $P$ holds vapour at
$P + 2\sigma/r$ (surface tension $\sigma$, Year 2 volume); it can only
grow if $P_s(T) > P + 2\sigma/r$, so a clean liquid at its boiling point
has no bubble small enough to start and superheats by several \omterm{def:b1:kinetic-theory:temperature}{kelvin}
before erupting --- the ``bumping'' that boiling chips prevent by
offering cavities full of trapped gas. The bubble chamber of particle
physics ran on the same instability in reverse: a superheated liquid
boils first along the track of an ionizing particle.
\end{remark}

\begin{example}[Humidity and dew]\label{ex:b1:phase-changes:dew}
\index{relative humidity}\index{dew point}
Air holds water vapour at a partial pressure $P_v \leq P_s(T)$; the
\emph{\omterm{ex:b1:phase-changes:dew}{relative humidity}} is $P_v/P_s(T)$, the \emph{\omterm{ex:b1:phase-changes:dew}{dew point}} the
temperature at which $P_s$ falls to $P_v$. Air at \qty{25}{\degreeCelsius}
($P_s = \qty{3.17}{kPa}$) and $60\%$ humidity has $P_v = \qty{1.9}{kPa}$,
$\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$, and dews on any surface below
about \qty{17}{\degreeCelsius} --- the grass at dawn, the cold bottle, the
bathroom mirror. Lifted, the same air cools adiabatically at
\qty{9.8}{K/km} (\cref{pb:b1:phase-changes:1}) and reaches its \omterm{ex:b1:phase-changes:dew}{dew point} a
kilometre or so up: that is where the flat bottoms of the cumulus are.
\end{example}

\section{Problem: Water in four scenes}

\begin{problem}[{Weekend problem --- the pressure cooker, the cloud, the
sealed vessel and the supercooled bottle: one substance, four uses of
its phase diagram}]
\label{pb:b1:phase-changes:1}
Data for water: $L_v = \qty{2257}{kJ/kg}$ at \qty{100}{\degreeCelsius} and
about \qty{2.2}{MJ/kg} at \qty{117}{\degreeCelsius}, \qty{2.45}{MJ/kg} near
\qty{20}{\degreeCelsius}; $L_f = \qty{334}{kJ/kg}$; $c = \qty{4.18}{kJ/(kg.K)}$
(liquid); $M = \qty{18}{g/mol}$; $ML_v/R = \qty{4890}{K}$ near
\qty{100}{\degreeCelsius}. Saturation pressures: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23}{kPa} (\qty{10}{\degreeCelsius}), \qty{1.70}{kPa} (\qty{15}{\degreeCelsius}),
\qty{2.34}{kPa} (\qty{20}{\degreeCelsius}), \qty{3.17}{kPa} (\qty{25}{\degreeCelsius}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius}).
Saturated specific volumes: $v_l = \qty{1.0e-3}{m^3/kg}$ (cold water),
\qty{1.9e-3}{m^3/kg} at \qty{360}{\degreeCelsius}; $v_v = \qty{0.0217}{m^3/kg}$
at \qty{300}{\degreeCelsius}, $0.0108$ at \qty{340}{\degreeCelsius}, $0.0088$
at \qty{350}{\degreeCelsius}; \omterm{prop:b1:phase-changes:PTdiagram}{critical point} \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$. Air: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$; $g = \qty{9.8}{m/s^2}$; $R =
\qty{8.314}{J/(mol.K)}$.

\medskip
\noindent\textbf{Part I --- The pressure cooker.}
\begin{enumerate}
  \item Why can water in an open pan at sea level not be heated above
        \qty{100}{\degreeCelsius}, however strong the flame?
  \item Starting from the \omterm{thm:b1:phase-changes:clapeyron}{Clausius--Clapeyron relation}, derive
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$, stating the
        approximations.
  \item Temperature in a cooker whose valve opens at \qty{1.8}{bar}
        absolute.
  \item The valve is a weight resting on an orifice of \qty{20}{mm^2};
        atmospheric pressure \qty{1.0}{bar}: mass of the weight.
  \item \qty{2.0}{L} of water are brought from \qty{20}{\degreeCelsius} to the
        working temperature by a \qty{2.0}{kW} plate (neglect the pot):
        heat and time.
  \item Once there, the plate is left at \qty{2.0}{kW}: mass of steam
        leaving the valve per minute. What should the cook do?
  \item Cooking reactions roughly double in rate for every \qty{10}{K}:
        by what factor does the cooker shorten a \qty{30}{min} sea-level
        recipe? And how long would the recipe take on Everest, where
        water boils at \qty{70}{\degreeCelsius}?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The cloud.}
\begin{enumerate}[resume]
  \item Define \omterm{ex:b1:phase-changes:dew}{relative humidity}. Air at \qty{25}{\degreeCelsius} and $60\%$:
        partial pressure of vapour, and mass of vapour per cubic metre
        (\omterm{def:b1:kinetic-theory:model}{perfect gas}).
  \item \omterm{ex:b1:phase-changes:dew}{Dew point} of that air (interpolate the table). Explain the dew
        on the grass at dawn and the drops on a cold bottle.
  \item A parcel of dry air rises without exchanging heat. From the
        reversible adiabatic law and the \omterm{thm:b1:fluid-statics:fundamental}{hydrostatic relation}
        $\dd P/\dd z = -\rho g$, show that $\dd T/\dd z = -g/c_p$ and
        evaluate it.
  \item The \omterm{ex:b1:phase-changes:dew}{dew point} of the rising parcel itself falls by about
        \qty{2}{K/km} as the pressure drops. Height at which the parcel of
        question~8 becomes saturated --- the base of the cloud.
  \item Above the base, the vapour that condenses releases its \omterm{def:b1:phase-changes:phase}{latent
        heat} into the parcel. By how much does condensing \qty{1}{g} of
        water per kilogram of air warm that air? What does this do to the
        cooling rate of the parcel as it keeps rising, and why does it
        matter for thunderstorms?
  \item A cumulus of \qty{1}{km^3} holds \qty{1}{g} of liquid water per cubic
        metre: mass of water, and the energy released by its
        condensation, in joules and in \unit{GW h}.
  \item A bathroom mirror at \qty{18}{\degreeCelsius}, air at
        \qty{25}{\degreeCelsius} and $80\%$: does it fog? At what humidity
        would it stop?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The sealed vessel.}
A rigid steel vessel of \qty{1.0}{L} is filled with a mass $m$ of water
and sealed with no air, then heated slowly.
\begin{enumerate}[resume]
  \item $m = \qty{0.50}{kg}$ at \qty{20}{\degreeCelsius}: pressure inside,
        specific volume, mass of vapour (use the \omterm{def:b1:kinetic-theory:model}{perfect gas} for $v_v$),
        fraction of the volume filled by liquid.
  \item Heated: does the liquid level rise or fall? At roughly what
        temperature does the liquid fill the vessel, and what happens to
        the pressure after that? (Use $v_l$ at \qty{360}{\degreeCelsius}.)
  \item $m = \qty{0.10}{kg}$: does the liquid level rise or fall on
        heating? At roughly what temperature does the last drop
        evaporate (use the $v_v$ table)?
  \item $m = \qty{0.31}{kg}$: describe what an observer sees at the
        meniscus as the temperature approaches \qty{374}{\degreeCelsius}.
  \item Sketch the three heating paths in the $(v, P)$ diagram with the
        saturation dome, and say which side of the \omterm{prop:b1:phase-changes:PTdiagram}{critical point} each
        passes.
  \item The same \qty{1.0}{L} vessel with \qty{0.50}{kg} of water is now
        sealed \emph{with} the air it contained at \qty{20}{\degreeCelsius},
        \qty{1.0}{bar}, and heated to \qty{100}{\degreeCelsius}: total
        pressure inside (Dalton), and the force on a lid of \qty{100}{cm^2}.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The supercooled bottle.}
\begin{enumerate}[resume]
  \item A clean, still bottle of water has cooled to \qty{-8}{\degreeCelsius}
        without freezing. What is this state called, why is it possible,
        and what happens when the bottle is knocked? Why does the
        temperature then settle at exactly \qty{0}{\degreeCelsius}?
  \item Fraction of the water that freezes (the process is fast, hence
        adiabatic).
  \item Entropy created per kilogram; sign check.
  \item The reverse surprise: a cup of water heated to \qty{105}{\degreeCelsius}
        in a microwave oven without boiling, then disturbed. Fraction
        of the water that flashes to steam, volume of steam produced per
        litre ($v_v = \qty{1.67}{m^3/kg}$), and why the cup erupts.
  \item \omterm{thm:b1:heat-engines:lostwork}{Lost work}: the supercooled litre could have driven a reversible
        engine with the \qty{0}{\degreeCelsius} surroundings; taking
        $T_0 = \qty{273}{K}$, the work thrown away by letting it freeze on a
        knock is $T_0S_{\mathrm{created}}$. Evaluate it per litre and
        compare with lifting the bottle by one metre. Conclude: what does
        a \omterm{def:b1:phase-changes:metastable}{metastable state} store?
\end{enumerate}
\end{problem}
