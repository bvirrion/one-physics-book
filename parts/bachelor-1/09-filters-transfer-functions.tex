\chapter{Filters and Transfer Functions}\label{ch:b1:filters-transfer-functions}

Inside a loudspeaker cabinet a coil and a capacitor decide, without a
single transistor, that the bass goes to the big cone and the treble to
the small dome. An oscilloscope's ``AC'' button removes a slowly
drifting offset and keeps the wiggle on top of it. A power supply turns
a pulsating rectified mains into a steady \omterm{def:b1:dc-circuits:voltage}{voltage} with one capacitor.
All three are \emph{\omterm{def:b1:filters-transfer-functions:H}{filters}}: \omterm{def:b1:dc-circuits:linear}{linear circuits} that treat each frequency
differently. Because any periodic \omterm{def:b1:wave-propagation:signal}{signal} is a sum of sinusoids, knowing
what a circuit does to every sinusoid --- its \omterm{def:b1:filters-transfer-functions:H}{transfer function} --- is
knowing what it does to everything. This chapter defines \omterm{def:b1:filters-transfer-functions:H}{transfer
functions}, draws their \omterm{def:b1:filters-transfer-functions:bode}{Bode diagrams} for the first- and second-order
circuits of the previous chapters, and uses them to predict how a
square wave comes out of a \omterm{def:b1:filters-transfer-functions:H}{filter}.

\section{Transfer function and Bode diagram}

\begin{definition}[Transfer function]\label{def:b1:filters-transfer-functions:H}
\index{transfer function}\index{gain}\index{filter}
A \omterm{def:b1:dc-circuits:linear}{linear circuit} with an input \omterm{def:b1:dc-circuits:voltage}{voltage} $\underline U_e$ and an output
\omterm{def:b1:dc-circuits:voltage}{voltage} $\underline U_s$ (taken with nothing connected at the output) has
the \emph{transfer function}
\[
\underline H(j\omega) = \frac{\underline U_s}{\underline U_e},
\qquad G(\omega) = \abs{\underline H}, \qquad \varphi(\omega) = \arg\underline H,
\]
the \emph{gain} $G$ (ratio of \omterm{def:b1:wave-propagation:signal}{amplitudes}) and the phase shift $\varphi$
of the output relative to the input. A \emph{filter} is such a circuit
used for its frequency dependence: a sinusoid of \omterm{def:b1:wave-propagation:signal}{amplitude} $E$ at
$\omega$ comes out as $G(\omega)E\cos(\omega t + \varphi(\omega))$.
\end{definition}

\begin{definition}[Decibels; Bode diagram]\label{def:b1:filters-transfer-functions:bode}
\index{decibel}\index{Bode diagram}\index{cutoff frequency}\index{passband}
The \omterm{def:b1:filters-transfer-functions:H}{gain} in \emph{decibels} is $G_{\mathrm{dB}} = 20\log_{10}G$: a factor
$10$ is \qty{20}{dB}, a factor $2$ is \qty{6}{dB}, $1/\sqrt2$ is
\qty{-3}{dB}. The \emph{Bode diagram} plots $G_{\mathrm{dB}}$ and
$\varphi$ against $\log_{10}\omega$ (or $f$). The \emph{cutoff
frequency} is where $G$ falls to $G_{\max}/\sqrt2$ (\qty{-3}{dB} from
the maximum); the \emph{passband} is the range of frequencies within
\qty{3}{dB} of the maximum. A decade is a factor $10$ in frequency, an
octave a factor $2$.
\end{definition}

\begin{method}[Nature of a filter at a glance]\label{met:b1:filters-transfer-functions:nature}
Before computing, look at the limits: at very low frequency a capacitor
is an open circuit and a coil a wire; at very high frequency the
reverse. Replace each and read whether the output is the input
(passes) or zero (blocked). Passes low, blocks high: low-pass; the
reverse: high-pass; blocks both: band-pass; passes both: band-stop.
Then compute $\underline H$ by a \omterm{prop:b1:dc-circuits:dividers}{voltage divider}.
\end{method}

\section{First-order filters}

\begin{proposition}[First-order low-pass]\label{prop:b1:filters-transfer-functions:lowpass1}
\index{low-pass filter}\index{integrator}
The RC divider with output across $C$ (or the RL divider with output
across $R$) has
\[
\underline H = \frac{1}{1 + j\omega/\omega_c}, \qquad
\omega_c = \frac{1}{RC} \ \Big(\text{or } \frac{R}{L}\Big),
\qquad G = \frac{1}{\sqrt{1 + (\omega/\omega_c)^2}}, \qquad
\varphi = -\arctan\frac{\omega}{\omega_c} .
\]
Bode asymptotes: $G_{\mathrm{dB}} = 0$ for $\omega \ll \omega_c$, a
straight line falling \qty{20}{dB} per decade for $\omega \gg \omega_c$,
meeting at $\omega_c$ where the true curve is \qty{-3}{dB} and
$\varphi = -45^\circ$. For $\omega \gg \omega_c$, $\underline H \approx
\omega_c/(j\omega)$: the output is proportional to the \emph{integral}
of the input --- the \omterm{def:b1:filters-transfer-functions:H}{filter} is an integrator there.
\end{proposition}

\begin{proof}
\omterm{prop:b1:dc-circuits:dividers}{Voltage divider}:
\[
\underline H = \frac{1/jC\omega}{R + 1/jC\omega} = \frac{1}{1 + jRC\omega} .
\]
Far above cutoff, $1 + j\omega/\omega_c \approx j\omega/\omega_c$, and
dividing a \omterm{def:b1:sinusoidal-impedance:complex}{complex amplitude} by $j\omega$ is integrating
(\cref{prop:b1:sinusoidal-impedance:rules}): $u_s \approx \omega_c\int u_e\,\dd t$.
$20\log(\omega_c/\omega)$ loses \qty{20}{dB} per tenfold $\omega$.
\end{proof}

\begin{proposition}[First-order high-pass]\label{prop:b1:filters-transfer-functions:highpass1}
\index{high-pass filter}\index{differentiator}
The CR divider with output across $R$ (or LR with output across $L$)
has
\[
\underline H = \frac{j\omega/\omega_c}{1 + j\omega/\omega_c},
\qquad G = \frac{\omega/\omega_c}{\sqrt{1 + (\omega/\omega_c)^2}},
\qquad \varphi = \frac{\pi}{2} - \arctan\frac{\omega}{\omega_c},
\]
rising \qty{20}{dB} per decade below $\omega_c$, flat above, \qty{-3}{dB}
and $+45^\circ$ at $\omega_c$. For $\omega \ll \omega_c$, $\underline H
\approx j\omega/\omega_c$: the output is proportional to the
\emph{derivative} of the input.
\end{proposition}

\begin{proof}
$\underline H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$;
multiplying by $j\omega$ is differentiating.
\end{proof}

\begin{example}[AC coupling]\label{ex:b1:filters-transfer-functions:accoupling}
An oscilloscope's AC input is a high-pass with $f_c \approx \qty{10}{Hz}$:
a \qty{1}{kHz} \omterm{def:b1:wave-propagation:signal}{signal} riding on a \qty{5}{V} offset comes through at
\qty{-0.0002}{dB} with its offset removed. But a \qty{50}{Hz} square
wave is visibly distorted --- its flat tops sag --- because at
\qty{50}{Hz} the \omterm{def:b1:filters-transfer-functions:H}{filter} is only one decade above its cutoff and the
\omterm{def:b1:wave-propagation:signal}{signal}'s low harmonics are shifted in phase
(\cref{exo:b1:filters-transfer-functions:12}).
\end{example}

\section{Second-order filters}

\begin{proposition}[The series RLC as three filters]\label{prop:b1:filters-transfer-functions:rlc}
\index{band-pass filter}\index{second-order filter}
With $x = \omega/\omega_0$, $\omega_0 = 1/\sqrt{LC}$ and $Q =
\sqrt{L/C}/R$, the series RLC fed by $\underline U_e$ gives, according to
where the output is taken:
\begin{align*}
\text{across } C:&\quad \underline H = \frac{1}{1 - x^2 + jx/Q}, \\
\text{across } R:&\quad \underline H = \frac{jx/Q}{1 - x^2 + jx/Q} = \frac{1}{1 + jQ(x - 1/x)}, \\
\text{across } L:&\quad \underline H = \frac{-x^2}{1 - x^2 + jx/Q}:
\end{align*}
a low-pass, a band-pass and a high-pass of second order. Far from
$\omega_0$ the low-pass and high-pass fall by \qty{40}{dB} per decade,
the band-pass by \qty{20}{dB} per decade on each side; the band-pass
has its maximum $1$ at $\omega_0$ and the bandwidth $\Delta\omega =
\omega_0/Q$ of \cref{thm:b1:sinusoidal-impedance:resonance}.
\end{proposition}

\begin{proof}
\omterm{prop:b1:dc-circuits:dividers}{Voltage dividers} with the total \omterm{def:b1:sinusoidal-impedance:impedance}{impedance} $R + j(L\omega - 1/C\omega)$;
multiply numerator and denominator by $jC\omega$ and use
$LC\omega_0^2 = 1$, $RC\omega_0 = 1/Q$. The band-pass is the current
resonance of the previous chapter read as a \omterm{def:b1:dc-circuits:voltage}{voltage} across $R$.
\end{proof}

\begin{proposition}[Second-order low-pass: the role of $Q$]\label{prop:b1:filters-transfer-functions:lowpass2}
\index{resonance!of a filter}
For the low-pass, $G = 1/\sqrt{(1 - x^2)^2 + x^2/Q^2}$: for
$Q > 1/\sqrt2$ the \omterm{def:b1:filters-transfer-functions:H}{gain} rises above $1$ near $\omega_0$ (resonance, peak
$\approx Q$ for large $Q$); for $Q = 1/\sqrt2$ it is ``maximally flat'',
$G = 1/\sqrt{1 + x^4}$, exactly \qty{-3}{dB} at $\omega_0$; for smaller
$Q$ it droops early. The phase runs from $0$ to $-\pi$, through $-\pi/2$
at $\omega_0$.
\end{proposition}

\begin{proof}
$(1 - x^2)^2 + x^2/Q^2 = 1 + x^4 + x^2(1/Q^2 - 2)$: the middle term
vanishes for $Q^2 = \tfrac12$, is negative (peak) above, positive
(droop) below; the behavior near $x = 1$ was analyzed in
\cref{prop:b1:sinusoidal-impedance:overvoltage}.
\end{proof}

\begin{example}[A crossover]\label{ex:b1:filters-transfer-functions:crossover}
A loudspeaker's woofer and tweeter, each \qty{8}{\ohm}, share the
amplifier through a low-pass (a series coil) and a high-pass (a series
capacitor) cut at \qty{2}{kHz}: $L = R/\omega_c = \qty{0.64}{mH}$,
$C = 1/R\omega_c = \qty{10}{\micro F}$. At \qty{2}{kHz} each driver gets
\qty{-3}{dB}, i.e.\ half the power, and $G_L^2 + G_H^2 = 1$ at every
frequency: the power is shared, not lost. The weekend problem designs
a steeper version.
\end{example}

\section{Filtering a periodic signal}

\begin{theorem}[Fourier decomposition]\label{thm:b1:filters-transfer-functions:fourier}
\index{Fourier series}\index{harmonic}\index{spectrum!of a periodic signal}
A periodic \omterm{def:b1:wave-propagation:signal}{signal} of period $T = 2\pi/\omega$ (reasonably regular) is
the sum of a constant and of sinusoids at the multiples of $\omega$:
\[
s(t) = c_0 + \sum_{n = 1}^{\infty} c_n\cos(n\omega t + \varphi_n),
\]
where $c_0 = \langle s\rangle$ is its average value; the term $n = 1$ is
the \emph{fundamental}, the others the \emph{harmonics}, and the
\omterm{def:b1:wave-propagation:signal}{amplitudes} $c_n$ form the \emph{spectrum}. A square wave of \omterm{def:b1:wave-propagation:signal}{amplitude}
$\pm E$ has $c_n = 4E/(n\pi)$ for odd $n$ and $0$ for even $n$; a
triangle wave of \omterm{def:b1:wave-propagation:signal}{amplitude} $\pm E$ has $c_n = 8E/(n\pi)^2$ for odd $n$.
\end{theorem}
\admitted

\begin{remark}[What to take from it]\label{rem:b1:filters-transfer-functions:fourier}
The theorem and the computation of the $c_n$ belong to the Year 2
mathematics course. Here it is a tool: sharp corners need many
harmonics (the square wave's fall off only as $1/n$), smooth \omterm{def:b1:wave-propagation:signal}{signals}
need few (the triangle's as $1/n^2$). A \omterm{def:b1:wave-propagation:signal}{signal}'s harmonic content is
what an instrument's timbre is made of, and what a \omterm{def:b1:filters-transfer-functions:H}{filter} acts on.
\end{remark}

\begin{proposition}[Linear filtering of a periodic signal]\label{prop:b1:filters-transfer-functions:filtering}
Through a \omterm{def:b1:filters-transfer-functions:H}{filter} of \omterm{def:b1:filters-transfer-functions:H}{transfer function} $\underline H$, the \omterm{def:b1:wave-propagation:signal}{signal} of the
theorem becomes
\[
s_{\text{out}}(t) = G(0)\,c_0 + \sum_{n \geq 1} G(n\omega)\,c_n\cos\!\big(n\omega t + \varphi_n + \varphi(n\omega)\big):
\]
each harmonic is scaled by the \omterm{def:b1:filters-transfer-functions:H}{gain} and shifted by the phase at
\emph{its own} frequency, then the pieces are added. Hence:
\begin{itemize}
  \item a low-pass with $\omega_c \ll \omega$ keeps essentially $c_0$:
        it \emph{averages} (smoothing, ripple removal), and turns a
        square wave into a small triangle (integration);
  \item a high-pass with $\omega_c \ll \omega$ removes $c_0$ and passes
        the rest (AC coupling); with $\omega_c \gg \omega$ it
        differentiates (spikes at the edges of a square wave);
  \item a band-pass tuned to $n\omega$ with $Q \gg 1$ extracts the
        $n$-th harmonic alone: a sinusoid out of a square wave.
\end{itemize}
\end{proposition}

\begin{proof}
Superposition for a \omterm{def:b1:dc-circuits:linear}{linear circuit} (\cref{thm:b1:dc-circuits:superposition})
applied to the sum of sinusoidal inputs, each treated by
\cref{def:b1:filters-transfer-functions:H}.
\end{proof}

\section{Cascading filters}

\begin{proposition}[Cascade and loading]\label{prop:b1:filters-transfer-functions:cascade}
\index{cascade (of filters)}\index{loading (of a filter)}
Two \omterm{def:b1:filters-transfer-functions:H}{filters} in cascade have $\underline H = \underline H_1\underline H_2$
--- \omterm{def:b1:filters-transfer-functions:H}{gains} multiply, \omterm{def:b1:filters-transfer-functions:bode}{decibels} and phases add --- \emph{provided the second
does not load the first}, i.e.\ draws negligible current from its
output. Otherwise the divider relations must be rewritten for the
combined network. An operational-amplifier follower
(\cref{ch:b1:operational-amplifier}) between stages makes the product
rule exact.
\end{proposition}

\begin{proof}
$\underline H_1$ was defined with an open output; if the next stage's
input \omterm{def:b1:sinusoidal-impedance:impedance}{impedance} is much larger than the first stage's output \omterm{def:b1:sinusoidal-impedance:impedance}{impedance}
(its Thévenin resistance), the output \omterm{def:b1:dc-circuits:voltage}{voltage} is unchanged and the
second stage sees exactly $\underline U_{s1}$ as its input.
\end{proof}

\begin{example}[Two RC stages]\label{ex:b1:filters-transfer-functions:tworc}
Two identical RC low-passes back to back do \emph{not} give
$1/(1 + j\omega/\omega_c)^2$: the second stage's $R$ loads the first.
The exact result is $\underline H = 1/[1 - (\omega/\omega_c)^2 +
3j\omega/\omega_c]$, \qty{-9.5}{dB} at $\omega_c$ instead of the
\qty{-6}{dB} the product rule promises, though both reach the same
\qty{-40}{dB/decade} slope eventually.
\end{example}

\section{Problem: The loudspeaker crossover}

\begin{problem}[{Weekend problem --- a woofer, a tweeter, a coil and a
capacitor: how a passive crossover splits the music, why the simplest
one leaks bass into the tweeter, and what a real voice coil does to the
design}]
\label{pb:b1:filters-transfer-functions:1}
The woofer and the tweeter are first modeled as pure resistances
$R = \qty{8.0}{\ohm}$. The amplifier is an ideal \omterm{def:b1:dc-circuits:sources}{voltage source}
$\underline U_e$. The crossover frequency is $f_c = \qty{2.0}{kHz}$.

\medskip
\noindent\textbf{Part I --- First-order crossover.}
The woofer is fed through a series coil $L$, the tweeter through a
series capacitor $C$.
\begin{enumerate}
  \item Define the \omterm{def:b1:filters-transfer-functions:H}{transfer function} of each branch (\omterm{def:b1:dc-circuits:voltage}{voltage} across the
        driver over $\underline U_e$) and the \omterm{def:b1:filters-transfer-functions:H}{gain} in \omterm{def:b1:filters-transfer-functions:bode}{decibels}.
  \item Show that the woofer branch is a first-order low-pass and give
        its cutoff $\omega_c$ in terms of $R$ and $L$.
  \item Compute $L$ for $f_c = \qty{2.0}{kHz}$.
  \item Show that the tweeter branch is a first-order high-pass; give
        $\omega_c$ and compute $C$.
  \item Give each driver's \omterm{def:b1:filters-transfer-functions:H}{gain} in \omterm{def:b1:filters-transfer-functions:bode}{decibels} at $f_c$, at \qty{200}{Hz}
        and at \qty{20}{kHz}.
  \item Show that $G_L^2 + G_H^2 = 1$ at every frequency. What does
        this mean for the power drawn from the amplifier?
  \item Give the phase of each branch at $f_c$; by how much do the two
        drivers' \omterm{def:b1:dc-circuits:voltage}{voltages} differ in phase there?
  \item What are the slopes of the two responses, in dB per octave, far
        from $f_c$?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Second-order crossover.}
The woofer is now fed through a series coil $L$ with a capacitor $C$
in parallel with the driver.
\begin{enumerate}[resume]
  \item Show that $\underline H = 1/(1 - LC\omega^2 + jL\omega/R)$.
  \item Identify $\omega_0$ and $Q$ in terms of $L$, $C$, $R$.
  \item For $Q = 1/\sqrt2$ show that $G = 1/\sqrt{1 + (\omega/\omega_0)^4}$
        and that the \omterm{def:b1:filters-transfer-functions:H}{gain} is \qty{-3}{dB} at $\omega_0$.
  \item Compute $L$ and $C$ for $f_0 = \qty{2.0}{kHz}$ and $Q = 1/\sqrt2$.
  \item Compute the \omterm{def:b1:filters-transfer-functions:H}{gain} in \omterm{def:b1:filters-transfer-functions:bode}{decibels} at \qty{4}{kHz} and \qty{8}{kHz};
        deduce the slope in dB per octave.
  \item The tweeter branch swaps the roles of $L$ and $C$ (series $C$,
        shunt $L$): write its \omterm{def:b1:filters-transfer-functions:H}{transfer function} without new
        calculation.
  \item Compute the phase of each branch at $f_0$. What must be done to
        the tweeter's wiring, and why?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Music through the crossover.}
\begin{enumerate}[resume]
  \item A cello note at \qty{220}{Hz} carries harmonics up to the 20th.
        Which harmonics go mostly to the tweeter, with the first-order
        crossover?
  \item What happens to the 9th harmonic?
  \item A \qty{1.0}{kHz} square wave is used as a test \omterm{def:b1:wave-propagation:signal}{signal}. With the
        first-order crossover, give the \omterm{def:b1:filters-transfer-functions:H}{gain} of each branch for its
        fundamental, third and fifth harmonics.
  \item The amplifier develops a \qty{1}{V} DC offset by accident. Which
        driver is protected, and by what?
  \item With the first-order crossover, what fraction of the \omterm{def:b1:dc-circuits:voltage}{voltage}
        and of the power of a \qty{500}{Hz} tone reaches the tweeter?
        Same with the second-order. Why does this matter for the
        tweeter's survival?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- A real voice coil.}
The woofer is actually $R = \qty{8.0}{\ohm}$ in series with a voice-coil
inductance $L_{vc} = \qty{0.50}{mH}$.
\begin{enumerate}[resume]
  \item Compute the woofer's \omterm{def:b1:sinusoidal-impedance:impedance}{impedance} (modulus) at \qty{2}{kHz} and at
        \qty{8}{kHz}.
  \item Recompute the first-order woofer \omterm{def:b1:filters-transfer-functions:H}{gain} at \qty{2}{kHz} and
        \qty{8}{kHz} with this \omterm{def:b1:sinusoidal-impedance:impedance}{impedance}, and compare with the ideal
        \qty{-3}{dB} and \qty{-12.3}{dB}. What happened to the roll-off?
  \item A ``Zobel network'' --- $R_z = R$ in series with $C_z =
        L_{vc}/R^2$ --- is connected in parallel with the woofer. Show
        that the combination behaves as a pure resistance $R$ at all
        frequencies, and compute $C_z$.
  \item The amplifier delivers \qty{50}{W} into \qty{8}{\ohm}: what rms
        \omterm{def:b1:dc-circuits:voltage}{voltage} is that, and what power does each driver receive at
        $f_c$ (first-order crossover)? At \qty{500}{Hz}, how much reaches
        the tweeter?
  \item Summarize the trade-offs: slope, phase, component count,
        protection of the tweeter, and the need for a Zobel network.
\end{enumerate}
\end{problem}
