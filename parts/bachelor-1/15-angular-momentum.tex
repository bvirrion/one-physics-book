\chapter{Angular Momentum}\label{ch:b1:angular-momentum}

A figure skater spinning with arms outstretched pulls them in and, with
no push from anyone, whirls twice as fast. A door yields to a fingertip
at the handle and resists a shoulder at the hinge. Planets sweep out
equal areas in equal times --- Kepler noticed it four centuries ago, long
before anyone could say why. And the Moon, tugged by the tides it
raises, drifts away from the Earth by a few centimeters a year while our
days grow imperceptibly longer. Behind all four stands one vector
quantity, the \omterm{def:b1:angular-momentum:L}{angular momentum}, and one law, its rate of change equals
the \omterm{def:b1:angular-momentum:moment}{moment} of the forces. This chapter defines both, proves the law for
a particle and for a system, and puts it to work on pendulums, orbits
and spinning things.

\section{Moment of a force}

\begin{definition}[Moment about a point; about an axis]\label{def:b1:angular-momentum:moment}
\index{moment of a force}\index{torque}\index{lever arm}
The \emph{moment} (or \emph{torque}) about a point $O$ of a force
$\vect F$ applied at $M$ is the vector
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
perpendicular to the plane of $\vect{OM}$ and $\vect F$, of norm
$OM\cdot F\sin\alpha = F\,d$, where $d$ --- the \emph{lever arm} --- is the
distance from $O$ to the line of action of $\vect F$. The moment about an
oriented axis $\Delta$ (unit vector $\vect e_\Delta$) through $O$ is the
scalar $\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$, independent
of the choice of $O$ on $\Delta$; it vanishes if $\vect F$ is parallel to
$\Delta$ or meets it. Two opposite forces on different lines of action
form a \emph{couple}, whose moment $\vect F\wedge\vect{AB}$ is the same
about every point.
\end{definition}

\begin{example}[The door]\label{ex:b1:angular-momentum:door}
A \qty{20}{N} push perpendicular to a door at \qty{0.80}{m} from the
hinges has \omterm{def:b1:angular-momentum:moment}{moment} $\qty{16}{N.m}$ about the hinge axis; the same push
at \qty{0.10}{m}, \qty{2}{N.m}; along the door (line of action through
the hinges), zero. The hinge's own reaction, meeting the axis, has no
\omterm{def:b1:angular-momentum:moment}{moment} about it: that is what makes the axis the natural place to take
\omterm{def:b1:angular-momentum:moment}{moments}.
\end{example}

\section{Angular momentum of a particle}

\begin{definition}[Angular momentum]\label{def:b1:angular-momentum:L}
\index{angular momentum}
The \emph{angular momentum} about $O$ of a particle of mass $m$ at $M$
with velocity $\vect v$ is
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
and about an axis $\Delta$ through $O$, $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
For a motion in a plane perpendicular to $\Delta$, in \omterm{def:b1:point-kinematics:cylindrical}{polar coordinates}
about the axis, $L_\Delta = mr^2\dot\theta$; for a circle of radius $R$,
$L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$, $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[Angular momentum theorem]\label{thm:b1:angular-momentum:theorem}
\index{angular momentum theorem}
In an \omterm{thm:b1:newton-dynamics:laws}{inertial frame}, for a point $O$ fixed in that frame,
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
for a fixed axis $\Delta$. If the total \omterm{def:b1:angular-momentum:moment}{moment} about $O$ vanishes,
$\vect L_O$ is conserved.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$, using Newton's second law
($O$ fixed so that $\dd\vect{OM}/\dd t = \vect v$). Project on $\vect e_\Delta$.
\end{proof}

\begin{corollary}[Central forces: planar motion and the law of areas]\label{cor:b1:angular-momentum:central}
\index{central force}\index{law of areas}\index{areal velocity}
A particle subject only to a \emph{central force} (always directed
along $\vect{OM}$, toward or away from the fixed center $O$) keeps
$\vect L_O$ constant. Its motion stays in the plane through $O$
perpendicular to $\vect L_O$, and in that plane $r^2\dot\theta = C$
(constant): the radius vector $\vect{OM}$ sweeps out area at the
constant \emph{\omterm{cor:b1:angular-momentum:central}{areal velocity}}
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ for $\vect F \parallel \vect{OM}$: $\vect L_O$
is constant. $\vect{OM}\perp\vect L_O$ always, hence the plane; $L_O =
mr^2\dot\theta$ is constant. The area swept during $\dd t$ is the
triangle of sides $\vect{OM}$ and $\vect v\dd t$: $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[The pendulum by moments]\label{ex:b1:angular-momentum:pendulum}
\omterm{ex:b1:newton-dynamics:pendulum}{Simple pendulum}, axis $\Delta$ through the pivot perpendicular to the
plane of swing: $L_\Delta = m\ell^2\dot\theta$; the tension meets the
axis (no \omterm{def:b1:angular-momentum:moment}{moment}), the weight's \omterm{def:b1:angular-momentum:moment}{moment} is $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{lever arm}
$\ell\sin\theta$, restoring). The theorem gives $m\ell^2\ddot\theta =
-mg\ell\sin\theta$, the equation of \cref{ex:b1:newton-dynamics:pendulum}
without ever writing the tension --- the chief advantage of \omterm{def:b1:angular-momentum:moment}{moments}:
forces through the axis disappear.
\end{example}

\begin{example}[Pulling a puck inward]\label{ex:b1:angular-momentum:puck}
A puck slides on frictionless ice in a circle, held by a string through
a hole at the center; the string is pulled so that the radius halves.
The tension is central: $L = mr^2\omega$ is conserved, so $\omega$ is
multiplied by $4$, the speed $r\omega$ by $2$, the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} by $4$
--- the extra energy is the work of the pull, which the puck resists
all the way in. The skater's arms are the same physics with the mass
distributed (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{Systems of points and rotation about a fixed axis}

\begin{theorem}[Angular momentum of a system]\label{thm:b1:angular-momentum:system}
\index{angular momentum!of a system}
For a system of particles, with $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
and $O$ fixed in an \omterm{thm:b1:newton-dynamics:laws}{inertial frame},
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
only the \emph{external} forces' \omterm{def:b1:angular-momentum:moment}{moments} count; the internal forces,
which obey the action--reaction law along the line joining the
particles, contribute nothing.
\end{theorem}

\begin{proof}
Sum the particle theorems. For an internal pair, $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$ since the force is along
$\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[Rigid rotation about a fixed axis]\label{prop:b1:angular-momentum:inertia}
\index{moment of inertia}
If the particles keep fixed distances $r_i$ from an axis $\Delta$ and all
turn about it at the same \omterm{cor:b1:point-kinematics:circular}{angular velocity} $\omega$ (a rigid rotation),
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
$J_\Delta$ being the \emph{\omterm{prop:b1:angular-momentum:inertia}{moment of inertia}} about $\Delta$, and the
theorem reads $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
Each particle contributes $m_ir_i^2\omega$ to $L_\Delta$ and
$\tfrac12 m_i(r_i\omega)^2$ to $E_k$. \cref{ch:b1:systems-of-points}
extends this to continuous solids.
\end{proof}

\begin{example}[The skater]\label{ex:b1:angular-momentum:skater}
Arms out, a skater's body has $J \approx \qty{4}{kg.m^2}$ plus two
\qty{3}{kg} arms at \qty{0.8}{m}: $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. Arms in (\qty{0.2}{m}): $J_2 = \qty{4.2}{kg.m^2}$.
The ice's reaction and the weight have no \omterm{def:b1:angular-momentum:moment}{moment} about the vertical
axis: $J\omega$ is conserved and the spin rate rises by $J_1/J_2 = 1.85$
--- from $2$ to $3.7$ turns per second --- while the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} rises
by the same factor, paid by the muscles pulling the arms in.
\end{example}

\begin{remark}[Statics]\label{rem:b1:angular-momentum:statics}
\index{equilibrium of a solid}
A solid at rest has $\vect L_O = \vect 0$ constant: the external forces
must have zero resultant \emph{and} zero total \omterm{def:b1:angular-momentum:moment}{moment} about any point.
The second condition is the lever law of the balance, the seesaw and
the crowbar: a small force far from the pivot balances a large one close
to it.
\end{remark}

\section{Problem: Why the Moon is leaving}

\begin{problem}[{Weekend problem --- the tides the Moon raises on the
Earth pull it forward and the Earth back: how angular momentum flows
from our days into the Moon's orbit, centimeter by centimeter, and
where the energy goes}]
\label{pb:b1:angular-momentum:1}
Data: $G = \qty{6.67e-11}{SI}$; Earth $M = \qty{5.97e24}{kg}$, $R =
\qty{6.37e6}{m}$, spin $\Omega = 2\pi/(\qty{86164}{s})$, \omterm{prop:b1:angular-momentum:inertia}{moment of inertia}
$J = 0.33\,MR^2$; Moon $m = \qty{7.35e22}{kg}$, orbit treated as
circular, $r = \qty{3.84e8}{m}$; lunar laser ranging measures
$\dd r/\dd t = \qty{3.8}{cm/yr}$; the day lengthens by \qty{2.3}{ms} per
century; $1$ century $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{Part I --- The angular momenta.}
\begin{enumerate}
  \item Compute the orbital \omterm{cor:b1:point-kinematics:circular}{angular velocity} $\omega$ of the Moon from
        Newton's law (circular orbit), and check it against the
        \qty{27.3}{d} sidereal month.
  \item Compute the Moon's orbital \omterm{def:b1:angular-momentum:L}{angular momentum} $L_{\text{orb}} =
        mr^2\omega$.
  \item Show that for a circular orbit $L_{\text{orb}} = m\sqrt{GMr}$, and
        deduce $\dd L_{\text{orb}}/\dd r = L_{\text{orb}}/2r$.
  \item Compute the Earth's spin \omterm{def:b1:angular-momentum:L}{angular momentum} $L_{\text{spin}} =
        J\Omega$ and compare with $L_{\text{orb}}$.
  \item The Moon spins once per month: estimate its spin \omterm{def:b1:angular-momentum:L}{angular
        momentum} ($J_m \approx 0.4\,mR_m^2$, $R_m = \qty{1.74e6}{m}$) and
        justify neglecting it.
  \item What external \omterm{def:b1:angular-momentum:moment}{moments} act on the Earth--Moon system about its
        center of mass? Argue that, to a good approximation, the total
        $L = L_{\text{spin}} + L_{\text{orb}}$ is conserved.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The tidal \omterm{def:b1:angular-momentum:moment}{torque}.}
The Moon raises two bulges of water on the Earth; because the Earth
spins faster than the Moon orbits, friction drags the bulges ahead of
the Earth--Moon line by a small angle.
\begin{enumerate}[resume]
  \item Sketch the Earth, the leading bulge and the Moon. In which
        direction does the Moon's attraction on the nearer bulge act,
        relative to the Earth's spin?
  \item Deduce the sign of the \omterm{def:b1:angular-momentum:moment}{moment} on the Earth's spin: does the
        day lengthen or shorten?
  \item By \omterm{thm:b1:newton-dynamics:laws}{action and reaction}, what \omterm{def:b1:angular-momentum:moment}{moment} acts on the Moon's orbital
        motion? Does $L_{\text{orb}}$ grow or shrink?
  \item From question 3, does $r$ grow or shrink? And the Moon's orbital
        speed $v = \sqrt{GM/r}$?
  \item Resolve the apparent paradox: the Moon is pushed forward, yet
        ends up moving more slowly.
  \item Toward what final state does the exchange tend?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The numbers.}
\begin{enumerate}[resume]
  \item From the lengthening of the day, compute the relative change
        $\Delta\Omega/\Omega$ per century.
  \item Deduce the change of $L_{\text{spin}}$ per century.
  \item Deduce the change of $L_{\text{orb}}$ per century.
  \item Using question 3, convert it into a change of $r$ per century
        and per year. Compare with the laser-ranging value.
  \item By what fraction, and by how many seconds, does the sidereal
        month lengthen per century ($T \propto r^{3/2}$)?
  \item Compute the change of the Earth's rotational \omterm{thm:b1:work-and-energy:ke}{kinetic energy}
        per century ($\Delta E = L_{\text{spin}}\Delta\Omega$ to first
        order).
  \item Compute the change of the Moon's orbital \omterm{thm:b1:work-and-energy:em}{mechanical energy},
        $E_{\text{orb}} = -GMm/2r$, per century.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The energy.}
\begin{enumerate}[resume]
  \item Show that the Earth loses more energy than the Moon gains, and
        compute the difference per century and the corresponding power.
        Where does it go?
  \item Compare that power with humanity's consumption, about
        \qty{18}{TW}, and with the Earth's internal heat flow, about
        \qty{47}{TW}.
  \item In the final state of question 12, day and month are equal:
        $\Omega = \omega = \sqrt{GM/r^3}$. Writing the conservation of
        $L_{\text{spin}} + L_{\text{orb}}$, check that $r \approx
        \qty{5.5e8}{m}$ satisfies it, and give the common period in days.
  \item Why will that state in fact never be reached? (Think of the Sun's
        tides.)
  \item The Moon already shows the Earth one face. Explain with the same
        mechanism, and say why it happened to the Moon first.
  \item Summarize in three lines: what is conserved, what is exchanged,
        what is dissipated.
\end{enumerate}
\end{problem}
