\chapter{Charged Particles in E and B Fields}\label{ch:b1:charged-particles}

Inside the oscilloscope that drew the traces of \cref{ch:b1:transient-regimes},
a beam of electrons is steered across a screen by the \omterm{def:b1:dc-circuits:voltage}{voltage} between
two plates; inside the magnet of a \omterm{prop:b1:charged-particles:spectrometer}{mass spectrometer}, ions of two
isotopes separate by millimeters because one is eight percent heavier
than the other; inside a hospital's cyclotron, protons spiral outward
through two hundred turns before leaving at a third of the speed of
light to burn a tumor. All three are the same dynamics: a charge in a
uniform field, electric or magnetic, and Newton's law. This chapter
works it out --- the parabola in an \omterm{def:b1:charged-particles:lorentz}{electric field}, the circle and the
helix in a \omterm{def:b1:charged-particles:lorentz}{magnetic field} --- and turns it into the instruments that
sort, accelerate and measure charged particles.

\section{The Lorentz force}

\begin{definition}[Lorentz force]\label{def:b1:charged-particles:lorentz}
\index{Lorentz force}\index{electric field}\index{magnetic field}
A particle of charge $q$ moving at velocity $\vect v$ in an electric
field $\vect E$ and a magnetic field $\vect B$ receives the force
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
The electric part is along $\vect E$ (or opposite, for $q < 0$) and
independent of the motion; the magnetic part is perpendicular both to
$\vect v$ and to $\vect B$, proportional to the speed, and vanishes for a
particle at rest. Units: $\vect E$ in \unit{V/m} (or \unit{N/C}), $\vect B$
in teslas (\unit{T}): $\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[Work of the Lorentz force]\label{prop:b1:charged-particles:work}
The magnetic force does no work: its power $q(\vect v\wedge\vect B)\cdot\vect v$
is zero, so a \omterm{def:b1:charged-particles:lorentz}{magnetic field} alone changes the direction of the velocity
but never the speed or the \omterm{thm:b1:work-and-energy:ke}{kinetic energy}. The electric force has power
$q\vect E\cdot\vect v$, and between two points its work is $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors}): a particle of charge $q$
crossing a \omterm{def:b1:dc-circuits:voltage}{voltage} $U$ gains the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. The electrostatic field derives from
the potential, $\vect E = -\overrightarrow{\operatorname{grad}}V$, so $q\vect E$
is \omterm{def:b1:work-and-energy:conservative}{conservative} with $E_p = qV$.
\end{proof}

\begin{definition}[The electron-volt]\label{def:b1:charged-particles:eV}
\index{electron-volt}
The \emph{electron-volt} is the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} gained by an elementary
charge crossing one volt: $\qty{1}{eV} = \qty{1.60e-19}{J}$. Chemistry
lives at a few eV, X-ray tubes at tens of keV, nuclear physics at MeV,
particle physics at GeV and TeV.
\end{definition}

\begin{remark}[Why gravity is ignored]\label{rem:b1:charged-particles:gravity}
For an electron in a field of \qty{1}{kV/m} --- a modest laboratory
value --- the electric force is $eE = \qty{1.6e-16}{N}$, against a weight
$mg = \qty{9e-30}{N}$: $10^{13}$ times smaller. Weight is dropped from
every calculation of this chapter without a second thought. (The
relativistic limit is another matter: the formulas below hold for
$\tfrac12 mv^2 \ll mc^2$, i.e.\ below about \qty{50}{keV} for electrons
and \qty{100}{MeV} for protons; see the Year 3 volume.)
\end{remark}

\section{Uniform electric field}

\begin{proposition}[Acceleration and deflection]\label{prop:b1:charged-particles:efield}
\index{accelerating voltage}
In a uniform field $\vect E$ the acceleration $\vect a = q\vect E/m$ is
constant: the motion is that of a projectile, with $q\vect E/m$ in place
of $\vect g$.
\begin{itemize}
  \item Released from rest and accelerated through a \omterm{def:b1:dc-circuits:voltage}{voltage} $U$, the
        particle reaches $v = \sqrt{2\abs{qU}/m}$.
  \item Entering at speed $v_0$ along $x$ a region of length $L$ where
        $\vect E = E\vect e_y$, it follows the parabola $y = (qE/2mv_0^2)x^2$
        and leaves deflected by the angle $\alpha$ with $\tan\alpha =
        qEL/mv_0^2$, as if it had come in a straight line from the
        middle of the plates.
\end{itemize}
\end{proposition}

\begin{proof}
Energy: $\tfrac12 mv^2 = \abs{qU}$. Deflection: $x = v_0t$, $y =
\tfrac12(qE/m)t^2$; at the exit, $\dd y/\dd x = (qE/m)(x/v_0^2)$ evaluated
at $L$; the tangent at the exit, $y = (qEL/mv_0^2)(x - L/2)$, crosses the
axis at $x = L/2$.
\end{proof}

\section{Uniform magnetic field}

\begin{theorem}[Motion in a uniform magnetic field]\label{thm:b1:charged-particles:bfield}
\index{cyclotron motion}\index{Larmor radius}\index{cyclotron frequency}\index{helical motion}
A particle of charge $q$, mass $m$, velocity $\vect v_0$ perpendicular to
a uniform $\vect B$, describes a circle at constant speed $v_0$, of radius
and \omterm{def:b1:wave-propagation:signal}{angular frequency}
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
the \emph{\omterm{thm:b1:charged-particles:bfield}{cyclotron frequency}} $\omega_c$ being independent of the speed;
the sense of rotation depends on the sign of $q$. If $\vect v_0$ also has
a component $v_\parallel$ along $\vect B$, that component is unchanged and
the \omterm{def:b1:point-kinematics:frame}{trajectory} is a \emph{helix} of radius $mv_\perp/\abs qB$ and pitch
$2\pi mv_\parallel/\abs qB$ wound around the field lines.
\end{theorem}

\begin{proof}
The force $q\vect v\wedge\vect B$ is perpendicular to $\vect v$ and to $\vect B$;
the speed is constant (no work) and the component along $\vect B$ sees no
force. In the plane perpendicular to $\vect B$ the force has the constant
norm $\abs qv_\perp B$, always perpendicular to the velocity: uniform
\omterm{cor:b1:point-kinematics:circular}{circular motion} (\cref{thm:b1:point-kinematics:frenet}) with $mv_\perp^2/R =
\abs qv_\perp B$. Then $\omega_c = v_\perp/R$; the period $2\pi m/\abs qB$ times
$v_\parallel$ is the pitch.
\end{proof}

\begin{example}[Orders of magnitude]\label{ex:b1:charged-particles:orders}
A \qty{1}{keV} electron ($v = \qty{1.9e7}{m/s}$) in \qty{1}{mT}:
$R = \qty{11}{cm}$, $\omega_c = \qty{1.8e8}{rad/s}$, one turn in
\qty{36}{ns} --- the same period at any energy, which is what makes the
cyclotron possible. A proton of the solar wind (\qty{1}{keV},
\qty{4.4e5}{m/s}) in the Earth's \qty{50}{\micro T}: $R = \qty{92}{m}$;
it cannot cross the field lines, only spiral along them toward the
poles --- the aurora.
\end{example}

\section{Instruments}

\begin{proposition}[Velocity selector and mass spectrometer]\label{prop:b1:charged-particles:spectrometer}
\index{velocity selector}\index{mass spectrometer}
In crossed uniform fields $\vect E \perp \vect B$, a particle moving
perpendicular to both passes undeflected iff $v = E/B$, whatever its
charge and mass (\emph{\omterm{prop:b1:charged-particles:spectrometer}{velocity selector}}). In a \omterm{def:b1:charged-particles:lorentz}{magnetic field} alone a
particle of known speed describes a semicircle of diameter $2mv/\abs qB$:
measuring it gives $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{mass spectrometer}}). Ions accelerated
through a \omterm{def:b1:dc-circuits:voltage}{voltage} $U$ then bent in $B$ land at
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
two isotopes of masses $m$ and $m + \Delta m$ separate by $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
$q\vect E + q\vect v\wedge\vect B = \vect 0$ requires $vB = E$ with the right
orientation. $R = mv/qB$ with $v = \sqrt{2qU/m}$; differentiate
$\ln R = \tfrac12\ln m + \text{const}$.
\end{proof}

\begin{example}[Carbon isotopes]\label{ex:b1:charged-particles:isotopes}
Singly charged $^{12}$C and $^{13}$C ions accelerated through \qty{20}{kV}
in \qty{0.50}{T}: $R = \qty{14.1}{cm}$ and \qty{14.7}{cm}, \qty{5.8}{mm}
apart --- two separate spots on the detector, and the ratio of their
intensities is the isotopic abundance, the raw material of radiocarbon
dating and of every isotope analysis.
\end{example}

\begin{proposition}[Cyclotron]\label{prop:b1:charged-particles:cyclotron}
\index{cyclotron}
Two hollow half-cylinders (``dees'') in a uniform $\vect B$, with an
alternating \omterm{def:b1:dc-circuits:voltage}{voltage} $U$ between them at the \omterm{thm:b1:charged-particles:bfield}{cyclotron frequency}: the
particle circles inside each dee (no field there), is accelerated by
$\abs qU$ at each of the two gap crossings per turn, and spirals outward
with $R \propto v$; at the extraction radius $R_{\max}$ its \omterm{thm:b1:work-and-energy:ke}{kinetic
energy} is
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
independent of the \omterm{def:b1:dc-circuits:voltage}{voltage}, which only sets the number of turns.
\end{proposition}

\begin{proof}
Because $\omega_c$ does not depend on the speed, a fixed-frequency \omterm{def:b1:dc-circuits:voltage}{voltage}
stays in step with the particle turn after turn; $v = \abs qBR/m$ at radius
$R$.
\end{proof}

\begin{example}[A hospital cyclotron]\label{ex:b1:charged-particles:medical}
Protons, $B = \qty{1.5}{T}$, $R_{\max} = \qty{0.50}{m}$: $E_k =
(1.6\times10^{-19} \times 1.5 \times 0.5)^2/(2 \times 1.67\times10^{-27}) =
\qty{4.3e-12}{J} = \qty{27}{MeV}$, at $f_c = \omega_c/2\pi = \qty{23}{MHz}$.
With \qty{50}{kV} on the dees, \qty{100}{keV} per turn: $270$ turns in
\qty{12}{\micro s}. Above a few tens of MeV the relativistic increase of
the mass (the Year 3 volume) desynchronizes the protons; therapy machines
at \qty{230}{MeV} modulate the frequency or the field to compensate ---
the weekend problem's subject.
\end{example}

\begin{proposition}[Hall effect]\label{prop:b1:charged-particles:hall}
\index{Hall effect}
A ribbon of thickness $d$ carrying a current $I$ of carriers of charge
$q$ and density $n$, in a field $B$ perpendicular to the ribbon,
develops across its width the \emph{Hall \omterm{def:b1:dc-circuits:voltage}{voltage}}
\[
U_H = \frac{IB}{nqd} :
\]
proportional to $B$ (a magnetometer) and inversely to $n$ (a probe of
the carrier density and sign).
\end{proposition}

\begin{proof}
The magnetic force $qv_dB$ on the carriers (drift speed $v_d$) pushes
them to one edge until the transverse \omterm{def:b1:charged-particles:lorentz}{electric field} $E_H$ balances it:
$qE_H = qv_dB$, and $U_H = E_Hw$ across the width $w$; with $I = nqv_dwd$,
$E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\section{Problem: Protons against a tumor}

\begin{problem}[{Weekend problem --- the cyclotron behind the wall of a
hospital: how protons are spun up to a third of the speed of light,
why the simple machine stops working before the energy a therapist
needs, and how many protons it takes to treat a tumor}]
\label{pb:b1:charged-particles:1}
Protons: $m = \qty{1.67e-27}{kg}$, $q = e = \qty{1.60e-19}{C}$, $mc^2 =
\qty{938}{MeV}$; $c = \qty{3.00e8}{m/s}$. Cyclotron: $B = \qty{1.5}{T}$,
extraction radius $R_{\max} = \qty{0.50}{m}$, dee \omterm{def:b1:dc-circuits:voltage}{voltage} \omterm{def:b1:wave-propagation:signal}{amplitude}
$U = \qty{50}{kV}$.

\medskip
\noindent\textbf{Part I --- One turn.}
\begin{enumerate}
  \item Write the \omterm{def:b1:charged-particles:lorentz}{Lorentz force} on a proton moving at $\vect v$
        perpendicular to $\vect B$, and show that its speed cannot change
        inside a dee.
  \item Derive the radius of its circle and its period; show that the
        period does not depend on the speed.
  \item Compute the \omterm{thm:b1:charged-particles:bfield}{cyclotron frequency} $f_c$ for these protons.
  \item Energy gained at each gap crossing; per turn.
  \item Why must the dee \omterm{def:b1:dc-circuits:voltage}{voltage} alternate at exactly $f_c$? What happens
        if it is slightly off?
