\chapter{Systems of Points; Introduction to Rigid Bodies}\label{ch:b1:systems-of-points}

A diver leaves the board, tucks, somersaults twice and straightens out;
through all of it one point of her body --- her \omterm{def:b1:systems-of-points:cm}{center of mass} ---
traces the same plain parabola a stone would. A yo-yo dropped from the
hand takes three times longer to reach the end of its string than a
stone would, and arrives spinning at two thousand turns a minute. A
solid sphere beats a hollow one down any incline, whatever their sizes
or masses. The mechanics of a single point, carried through the previous
chapters, extends to systems of many points --- and to the rigid bodies
of everyday life --- through a handful of theorems: the \omterm{def:b1:systems-of-points:cm}{center of mass}
moves as a point, energy and \omterm{def:b1:angular-momentum:L}{angular momentum} split into a center-of-mass
part and a part ``around'' it, and a body turning about a fixed axis is
a one-degree-of-freedom system governed by its \omterm{prop:b1:angular-momentum:inertia}{moment of inertia}.

\section{Center of mass and momentum}

\begin{definition}[Center of mass; momentum of a system]\label{def:b1:systems-of-points:cm}
\index{center of mass}\index{barycenter}
For particles of masses $m_i$ at $M_i$, total mass $M = \sum m_i$, the
\emph{center of mass} (barycenter) $G$ is defined by
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{i.e.}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
for any origin $O$. The \emph{momentum} of the system is $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Theorem of the center of mass]\label{thm:b1:systems-of-points:cmtheorem}
\index{theorem of the center of mass}\index{conservation of momentum}
In an \omterm{thm:b1:newton-dynamics:laws}{inertial frame},
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
the \omterm{def:b1:systems-of-points:cm}{center of mass} moves as a \omterm{def:b1:point-kinematics:frame}{point particle} of mass $M$ subject to the
resultant of the \emph{external} forces alone. For a \omterm{def:b1:newton-dynamics:conservation}{closed system} with
no external force, $\vect P$ is conserved and $G$ moves uniformly.
\end{theorem}

\begin{proof}
Sum Newton's second law over the particles; the internal forces cancel
in action--reaction pairs; $\sum m_i\vect a_i = M\vect a_G$ by
differentiating the definition of $G$ twice.
\end{proof}

\begin{example}[Diver, recoil, collision]\label{ex:b1:systems-of-points:cm}
\index{collision}
The diver's $G$ follows a parabola under her weight alone, however she
twists --- her muscles are internal forces. A rifle of \qty{4.0}{kg}
firing a \qty{10}{g} bullet at \qty{800}{m/s}: $\vect P = \vect 0$ before
and after, so the rifle recoils at \qty{2.0}{m/s}. Two cars, \qty{1000}{kg}
at \qty{20}{m/s} and \qty{1500}{kg} at rest, that lock together: $v =
20000/2500 = \qty{8.0}{m/s}$ --- momentum conserved, while $60\%$ of the
\omterm{thm:b1:work-and-energy:ke}{kinetic energy} has gone into crumpled metal and heat. Momentum is
conserved in every collision; \omterm{thm:b1:work-and-energy:ke}{kinetic energy} only in \emph{elastic} ones
(\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{Koenig's theorems}

\begin{definition}[Barycentric frame]\label{def:b1:systems-of-points:barycentric}
\index{barycentric frame}
The \emph{barycentric frame} $\mathcal R^*$ has its origin at $G$ and is
in translation with respect to the \omterm{thm:b1:newton-dynamics:laws}{inertial frame} $\mathcal R$ (its axes
keep fixed directions). Quantities measured in it are starred: $\vect v_i^*
= \vect v_i - \vect v_G$, and $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[Koenig's theorems]\label{thm:b1:systems-of-points:koenig}
\index{Koenig's theorems}
The \omterm{def:b1:angular-momentum:L}{angular momentum} about a point $O$ and the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} of a
system split into the contribution of its \omterm{def:b1:systems-of-points:cm}{center of mass}, carrying the
whole mass, and the contributions in the \omterm{def:b1:systems-of-points:barycentric}{barycentric frame}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
where $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ and $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ do not depend on the frame's origin.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ and $\vect v_i = \vect v_G + \vect v_i^*$;
expand $\sum\vect{OM_i}\wedge m_i\vect v_i$ and $\sum\tfrac12 m_iv_i^2$: the cross
terms contain $\sum m_i\vect{GM_i} = \vect 0$ or $\sum m_i\vect v_i^* = \vect 0$
and vanish.
\end{proof}

\begin{example}[A rolling wheel]\label{ex:b1:systems-of-points:rolling}
\index{rolling without slipping}
A wheel of mass $M$, radius $R$, \omterm{prop:b1:angular-momentum:inertia}{moment of inertia} $J$ about its axle,
\omterm{ex:b1:systems-of-points:rolling}{rolling without slipping} at speed $v$: the contact point is at rest, so
$\omega = v/R$, and $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- three quarters of $Mv^2$ for a uniform disk ($J = \tfrac12 MR^2$),
$Mv^2$ for a hoop. The second term is why a bicycle's wheels are made
light: their rotation costs energy that the frame's mass does not.
\end{example}

\section{The two-body problem}

\begin{proposition}[Reduced mass]\label{prop:b1:systems-of-points:twobody}
\index{two-body problem}\index{reduced mass}
Two particles interacting only with each other (force $\vect F$ on $m_2$
from $m_1$, $-\vect F$ on $m_1$): $G$ moves uniformly, and the relative
position $\vect r = \vect{M_1M_2}$ obeys
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
the equation of a single fictitious particle of \emph{\omterm{prop:b1:systems-of-points:twobody}{reduced mass}} $\mu$
moving under $\vect F$ about a fixed center. In $\mathcal R^*$, $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ and $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. In $\mathcal R^*$, $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ and
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$; substitute into $E_k^*$ and $\vect L^*$.
\end{proof}

\begin{example}[When the center moves too]\label{ex:b1:systems-of-points:reduced}
In \cref{ch:b1:central-forces} the Sun was fixed: exact only in the
limit $m \ll M$; in general \omterm{thm:b1:central-forces:circular}{Kepler's third law} reads $T^2 = 4\pi^2a^3/G(M +
m)$. For the vibrating HCl molecule of \cref{pb:b1:work-and-energy:1},
the mass that oscillates in the well is $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, which raises the frequency by $1.4\%$; for the hydrogen
atom, the electron's $\mu$ differs from $m_e$ by one part in $1836$ ---
measurable in the spectrum.
\end{example}

\section{Rigid body rotating about a fixed axis}

\begin{definition}[Solid; rotation about a fixed axis]\label{def:b1:systems-of-points:solid}
\index{rigid body}\index{rotation about a fixed axis}
A \emph{rigid body} (solid) is a system whose points keep fixed mutual
distances. If it turns about an axis $\Delta$ fixed in the frame, every
point describes a circle centered on $\Delta$ at the same \omterm{cor:b1:point-kinematics:circular}{angular velocity}
$\omega = \dot\theta$; a point at distance $r_i$ from the axis has speed
$r_i\omega$.
\end{definition}

\begin{proposition}[Moment of inertia; angular momentum; kinetic energy]\label{prop:b1:systems-of-points:inertia}
\index{moment of inertia!of solids}\index{Huygens' theorem}
With the \emph{\omterm{prop:b1:angular-momentum:inertia}{moment of inertia}} $J_\Delta = \sum m_ir_i^2$ (an integral
$\int r^2\,\dd m$ for a continuous body):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Standard values (mass $M$): thin ring or hollow cylinder about its axis,
$MR^2$; uniform disk or solid cylinder, $\tfrac12 MR^2$; thin rod of
length $L$ about its center, $\tfrac{1}{12}ML^2$, about one end,
$\tfrac13 ML^2$; solid sphere about a diameter, $\tfrac25 MR^2$.
\emph{Huygens' theorem}: about an axis $\Delta$ parallel to an axis
$\Delta_G$ through $G$ at distance $d$, $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ and $E_k$: \cref{prop:b1:angular-momentum:inertia}. Rod about
its end: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$; about the center, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Disk: ring of radius $r$, width $\dd r$, mass
$(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$; the ring and the
sphere are admitted (the sphere needs a double integral). Huygens: with
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$, the cross term sums to zero by definition of $G$.
\end{proof}

\begin{theorem}[Dynamics of a solid about a fixed axis]\label{thm:b1:systems-of-points:rotation}
\index{pivot}\index{power of a moment}
For a solid rotating about a fixed axis $\Delta$ in an \omterm{thm:b1:newton-dynamics:laws}{inertial frame},
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
the moments of the external forces about the axis drive the angular
acceleration, and a moment $\mathcal M_\Delta$ delivers the power
$\mathcal M_\Delta\omega$. An ideal \emph{pivot} (frictionless bearing) exerts
no moment about its axis and so neither accelerates nor brakes the
rotation.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} projected on $\Delta$, with $L_\Delta
= J_\Delta\dot\theta$ and $J_\Delta$ constant for a \omterm{def:b1:systems-of-points:solid}{rigid body}; multiply by
$\dot\theta$ for the energy form (the internal forces of a \omterm{def:b1:systems-of-points:solid}{rigid body} do
no work: the distances do not change).
\end{proof}

\begin{example}[Compound pendulum; torsion pendulum]\label{ex:b1:systems-of-points:pendulums}
\index{compound pendulum}\index{torsion pendulum}
A solid of mass $M$ pivoting about a horizontal axis at distance $d$
from $G$: the weight's moment is $-Mgd\sin\theta$, so $J_\Delta\ddot\theta =
-Mgd\sin\theta$ and small swings have the period
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
that of a \omterm{ex:b1:newton-dynamics:pendulum}{simple pendulum} of length $J_\Delta/Md$ --- for a uniform rod
pivoted at one end, $2L/3$. A disk hung from a wire of torsion constant
$C$ (restoring moment $-C\theta$): $J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$
--- the torsion balance that measured $G$ and Coulomb's law.
\end{example}

\begin{proposition}[Rolling without slipping down an incline]\label{prop:b1:systems-of-points:rollingincline}
A solid of revolution ($J = kMR^2$ about its axis) released on an
incline of angle $\alpha$ rolls without slipping with the acceleration
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
$\tfrac57 g\sin\alpha$ for a solid sphere, $\tfrac23 g\sin\alpha$ for a disk,
$\tfrac12 g\sin\alpha$ for a hoop --- independent of mass and radius.
\end{proposition}

\begin{proof}
Energy: $Mgh = \tfrac12 Mv^2(1 + k)$ along the slope (the static friction
at the resting contact point does no work), so $v^2 = 2ah$ with $a$ as
stated; or Newton for $G$ and the moment theorem about $G$ with the
friction $f$: $Ma = Mg\sin\alpha - f$, $kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Deformable systems]\label{rem:b1:systems-of-points:deformable}
\index{deformable system}
For a system that is not rigid, the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} theorem reads
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: the internal forces do
work (muscles, springs, friction inside the system). The skater pulling
her arms in, the diver tucking, a cyclist standing on the pedals all
gain \omterm{thm:b1:work-and-energy:ke}{kinetic energy} with no external work --- the \omterm{def:b1:systems-of-points:cm}{center of mass}
theorem stays true, the energy comes from inside.
\end{remark}

\section{Problem: The yo-yo}

\begin{problem}[{Weekend problem --- a disk on a thin axle at the end
of a string: why it falls so slowly, how fast it spins at the bottom,
why it ``sleeps'' there, and which way a spool rolls when you pull its
thread}]
\label{pb:b1:systems-of-points:1}
A yo-yo is a uniform disk of mass $M = \qty{50}{g}$ and radius $R =
\qty{3.0}{cm}$ with a thin axle of radius $r = \qty{0.40}{cm}$ around
which a string of length $h = \qty{1.0}{m}$ is wound; the string's upper
end is held fixed. $g = \qty{9.81}{m/s^2}$; $J = \tfrac12 MR^2$ about the
axis.

\medskip
\noindent\textbf{Part I --- Kinematics and energy.}
\begin{enumerate}
  \item Where is the \omterm{def:b1:systems-of-points:cm}{center of mass} $G$ of the yo-yo? Why?
  \item Show from Koenig's theorem that its \omterm{thm:b1:work-and-energy:ke}{kinetic energy} is $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item Derive $J = \tfrac12 MR^2$ by integrating over rings, and compute
        $J$.
  \item As the string unwinds without slipping from the axle, relate
        $v_G$ and $\omega$.
  \item Deduce $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ and compute the
        factor in parentheses. Comment.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The fall.}
\begin{enumerate}[resume]
  \item Forces on the yo-yo; write the center-of-mass theorem
        (vertical axis downward).
  \item Write the \omterm{def:b1:angular-momentum:L}{angular momentum} theorem about the axis through $G$
        (which forces have a moment?).
  \item Combine with question 4 to find the acceleration $a_G = g/(1 +
        R^2/2r^2)$; compute it.
  \item Compute the string tension and compare with the weight.
  \item Time to unwind the \qty{1.0}{m} of string, and the speed then;
        compare with a stone dropped from the same height.
  \item \omterm{cor:b1:point-kinematics:circular}{Angular velocity} at the bottom, in rad/s and turns per minute.
  \item Split the \omterm{thm:b1:work-and-energy:ke}{kinetic energy} at the bottom into its translational
        and rotational parts.
  \item Check the energy balance: compare $Mgh$ with the \omterm{thm:b1:work-and-energy:ke}{kinetic energy}
        at the bottom, and say why the string's tension, though large,
        does no work.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- At the bottom: the sleeper.}
When the string is fully unwound it stops the \omterm{def:b1:systems-of-points:cm}{center of mass} in a few
milliseconds; the yo-yo keeps spinning in the loop at the end of the
string (``sleeping'').
\begin{enumerate}[resume]
  \item Why does the spin survive the stop of $G$? (Which theorem, and
        which force would have been needed to stop it?)
  \item Estimate the average tension during the stop if $G$ is halted
        over \qty{5}{ms}.
  \item The loop rubs on the axle with a friction moment $\Gamma =
        \qty{1.0e-4}{N.m}$: how long does the yo-yo sleep?
  \item A sharp upward jerk on the string makes the string grip the axle
        again; the spinning yo-yo then climbs. Using energy, to what
        height could it rise if nothing were lost? Why, in practice,
        less?
  \item During the climb, is the tension larger or smaller than the
        weight? (Sign of $a_G$.)
  \item What would happen with a yo-yo whose axle radius equalled $R$
        (string wound on the rim)? Compute its fall acceleration.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The spool on the table.}
The same object lies on a rough table, axle horizontal, and its string
leaves the \emph{bottom} of the axle horizontally; one pulls with a
force $F$. It rolls without slipping.
\begin{enumerate}[resume]
  \item At the contact point $C$ with the table, what is the velocity
        of the spool's material? Deduce that the motion is, at each
        instant, a rotation about $C$.
  \item Compute the moment of $F$ about $C$ (\omterm{def:b1:angular-momentum:moment}{lever arm} $R - r$) and that
        of the weight and of the table's reaction.
  \item Using the \omterm{def:b1:angular-momentum:L}{angular momentum} theorem about $C$ with $J_C = J +
        MR^2$ (Huygens), find $a_G$ and show the spool rolls
        \emph{toward} the hand.
  \item Now the string is pulled at an angle $\beta$ above the
        horizontal. Show that the spool does not roll when the line of
        action of $F$ passes through $C$, i.e.\ $\cos\beta = r/R$, and
        rolls away from the hand for steeper pulls. Compute that angle.
  \item Compute $a_G$ for $F = \qty{0.20}{N}$ pulled horizontally, and
        the friction force needed; what limits $F$ before the spool
        slips?
  \item Summarize: the three theorems used (\omterm{def:b1:systems-of-points:cm}{center of mass}, \omterm{def:b1:angular-momentum:L}{angular
        momentum} about an axis, energy) and what each decided.
\end{enumerate}
\end{problem}
