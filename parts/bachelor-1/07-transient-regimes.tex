\chapter{Transient Regimes: First and Second Order}\label{ch:b1:transient-regimes}

Push down on the corner of a parked car and let go. A car in good
order rises once and stops; a car with worn shock absorbers bobs two or
three times before it settles. In an electronics lab the same two
behaviors appear on an oscilloscope when a capacitor discharges through
a coil: a smooth return, or a ringing that dies out. Both are the
\emph{transient} between one \omterm{def:b1:transient-regimes:firstorder}{steady state} and another, and both obey
the same small family of linear differential equations --- first order
for a single energy store, second order for two that exchange. This
chapter solves them once and for all, names their parameters (\omterm{def:b1:transient-regimes:firstorder}{time
constant}, natural frequency, \omterm{def:b1:transient-regimes:secondorder}{quality factor}), and shows how the same
numbers describe a circuit and a suspension.

\section{First-order systems}

\begin{definition}[First-order linear system]\label{def:b1:transient-regimes:firstorder}
\index{first-order system}\index{time constant}\index{transient regime}\index{steady state}
A quantity $x(t)$ obeys a \emph{first-order linear equation with
constant coefficients} when
\[
\tau\,\frac{\dd x}{\dd t} + x = x_\infty ,
\]
with $\tau > 0$ the \emph{time constant} and $x_\infty$ a constant (the
value imposed by the source). Its \emph{steady state} is $x = x_\infty$;
the \emph{transient regime} is the approach to it from the initial
value $x(0) = x_0$.
\end{definition}

\begin{theorem}[Solution]\label{thm:b1:transient-regimes:firstorder}
The unique solution with $x(0) = x_0$ is
\[
x(t) = x_\infty + (x_0 - x_\infty)\,\eu^{-t/\tau} .
\]
The gap to the \omterm{def:b1:transient-regimes:firstorder}{steady state} shrinks by the factor $\eu^{-1} \approx
0.37$ every $\tau$: $63\%$ of the way at $\tau$, $95\%$ at $3\tau$,
$99\%$ at $4.6\tau$. The tangent at the origin reaches $x_\infty$ at
$t = \tau$.
\end{theorem}

\begin{proof}
$y = x - x_\infty$ obeys $\tau y' + y = 0$, whose solutions are
$y = K\eu^{-t/\tau}$ (the mathematics course on linear differential
equations proves there are no others); $K = x_0 - x_\infty$. The slope
at $0$ is $-(x_0 - x_\infty)/\tau$, which would close the gap in time
$\tau$.
\end{proof}

\begin{proposition}[RC and RL circuits]\label{prop:b1:transient-regimes:rcrl}
\index{RC circuit}\index{RL circuit}
A capacitor $C$ charged through a \omterm{def:b1:dc-circuits:resistor}{resistor} $R$ from an ideal source $E$
obeys $RC\,\dd u_C/\dd t + u_C = E$: $\tau = RC$, $u_{C,\infty} = E$. A
coil $L$ in series with $R$ across $E$ obeys $(L/R)\,\dd i/\dd t + i =
E/R$: $\tau = L/R$, $i_\infty = E/R$. With the source removed
($E \to 0$) the same equations describe the discharge toward zero.
\end{proposition}

\begin{proof}
\omterm{thm:b1:dc-circuits:kirchhoff}{Loop law} $E = Ri + u_C$ with $i = C\,\dd u_C/\dd t$; \omterm{thm:b1:dc-circuits:kirchhoff}{loop law}
$E = Ri + L\,\dd i/\dd t$.
\end{proof}

\begin{proposition}[Continuity conditions]\label{prop:b1:transient-regimes:continuity}
\index{continuity (of $u_C$, of $i_L$)}
The \omterm{def:b1:dc-circuits:voltage}{voltage} across a capacitor and the current through an inductor are
continuous functions of time: at a switching instant $t_0$,
$u_C(t_0^+) = u_C(t_0^-)$ and $i_L(t_0^+) = i_L(t_0^-)$. Currents in
\omterm{def:b1:dc-circuits:resistor}{resistors} and \omterm{def:b1:dc-circuits:voltage}{voltages} across coils may jump.
\end{proposition}

\begin{proof}
A jump of $u_C$ would require $i = C\,\dd u_C/\dd t$ infinite, a jump
of $i_L$ would require $u = L\,\dd i/\dd t$ infinite; both are
impossible in a circuit of finite \omterm{def:b1:dc-circuits:voltage}{voltages} and currents (equivalently,
the stored energies $\tfrac12 Cu_C^2$ and $\tfrac12 Li^2$ cannot change
instantaneously without infinite power).
\end{proof}

\begin{method}[Solving a first-order transient]\label{met:b1:transient-regimes:first}
\begin{enumerate}
  \item Write the loop and \omterm{thm:b1:dc-circuits:kirchhoff}{node laws} after the switch, with the
        component laws; reduce to one equation in $u_C$ or $i_L$ and
        put it in the form $\tau x' + x = x_\infty$; read $\tau$ and
        $x_\infty$ (the latter is also the \omterm{def:b1:transient-regimes:firstorder}{steady state} found by
        replacing $C$ by an open circuit, $L$ by a wire).
  \item Find $x_0$ from the continuity of $u_C$ or $i_L$ across the
        switching instant, using the circuit \emph{before} it.
  \item Write $x = x_\infty + (x_0 - x_\infty)\eu^{-t/\tau}$; deduce
        the other quantities by differentiation and the \omterm{thm:b1:dc-circuits:kirchhoff}{loop law}.
\end{enumerate}
When the capacitor sees a network of \omterm{def:b1:dc-circuits:resistor}{resistors} and sources, replace
that network by its Thévenin equivalent $(E_{\mathrm{Th}}, R_{\mathrm{Th}})$
first (\cref{ch:b1:dc-circuits}): $\tau = R_{\mathrm{Th}}C$,
$u_\infty = E_{\mathrm{Th}}$.
\end{method}

\begin{proposition}[Energy balance of the RC charge]\label{prop:b1:transient-regimes:rcenergy}
Charging a capacitor from $0$ to $E$ through any \omterm{def:b1:dc-circuits:resistor}{resistor} $R$, the
source supplies $CE^2$, the capacitor stores $\tfrac12 CE^2$, and the
\omterm{def:b1:dc-circuits:resistor}{resistor} dissipates the other $\tfrac12 CE^2$ --- whatever $R$.
\end{proposition}

\begin{proof}
$i = (E/R)\eu^{-t/\tau}$; the source supplies $\int_0^\infty Ei\,\dd t
= E \cdot (E/R)\tau = CE^2$; the \omterm{def:b1:dc-circuits:resistor}{resistor} takes $\int_0^\infty Ri^2\,\dd t
= (E^2/R)\int_0^\infty \eu^{-2t/\tau}\dd t = (E^2/R)(\tau/2) = \tfrac12 CE^2$;
the difference is the stored $\tfrac12 CE^2$. A smaller $R$ dissipates
the same energy faster.
\end{proof}

\begin{example}[A relay coil]\label{ex:b1:transient-regimes:relay}
$L = \qty{20}{mH}$, $R = \qty{5.0}{\ohm}$, $E = \qty{10}{V}$: $\tau =
\qty{4.0}{ms}$, $i_\infty = \qty{2.0}{A}$; the contacts pull in when
$i$ reaches \qty{1.5}{A}, i.e.\ at $t = -\tau\ln(1 - 0.75) =
\qty{5.5}{ms}$. Opening the circuit, $i$ must fall from \qty{2}{A}; a
``freewheeling'' diode across the coil gives it a path through $R$
alone, $\tau = \qty{4}{ms}$, instead of a spark.
\end{example}

\section{Second-order systems}

\begin{definition}[Canonical second-order form]\label{def:b1:transient-regimes:secondorder}
\index{second-order system}\index{natural angular frequency}\index{quality factor}\index{damping ratio}
A quantity $x(t)$ is a \emph{second-order linear system} when
\[
\frac{\dd^2 x}{\dd t^2} + \frac{\omega_0}{Q}\,\frac{\dd x}{\dd t} + \omega_0^2\,x
= \omega_0^2\,x_\infty ,
\]
with $\omega_0 > 0$ the \emph{natural \omterm{def:b1:wave-propagation:signal}{angular frequency}}, $Q > 0$ the
\emph{quality factor} (\omterm{def:b1:units-dimensions:dimension}{dimensionless}), and $x_\infty$ the \omterm{def:b1:transient-regimes:firstorder}{steady state}.
One also writes $\omega_0/Q = 2\xi\omega_0$ with $\xi = 1/2Q$ the
\emph{damping ratio}.
\end{definition}

\begin{proposition}[The series RLC circuit]\label{prop:b1:transient-regimes:rlc}
\index{RLC circuit}
A capacitor $C$, a coil $L$ and a \omterm{def:b1:dc-circuits:resistor}{resistor} $R$ in series across a source
$E$: the capacitor \omterm{def:b1:dc-circuits:voltage}{voltage} obeys the canonical form with
\[
\omega_0 = \frac{1}{\sqrt{LC}}, \qquad Q = \frac{1}{R}\sqrt{\frac{L}{C}},
\qquad u_{C,\infty} = E .
\]
\end{proposition}

\begin{proof}
\omterm{thm:b1:dc-circuits:kirchhoff}{Loop law}: $E = L\,\dd i/\dd t + Ri + u_C$ with $i = C\,\dd u_C/\dd t$:
$LC\,u_C'' + RC\,u_C' + u_C = E$; divide by $LC$ and identify
$\omega_0^2 = 1/LC$, $\omega_0/Q = R/L$.
\end{proof}

\begin{proposition}[The mechanical oscillator]\label{prop:b1:transient-regimes:mechanical}
\index{mass--spring--damper}\index{electromechanical analogy}
A mass $m$ on a spring of stiffness $k$ with a viscous damper of
coefficient $\alpha$ (force $-\alpha\dot x$), displaced by $x$ from
equilibrium, obeys $m\ddot x + \alpha\dot x + kx = 0$: the canonical form
with
\[
\omega_0 = \sqrt{\frac{k}{m}}, \qquad Q = \frac{\sqrt{km}}{\alpha} .
\]
The correspondence $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$
maps every result of one system onto the other.
\end{proposition}

\begin{proof}
Newton's second law (\cref{ch:b1:newton-dynamics}) with the spring force
$-kx$ and the damping force; divide by $m$.
\end{proof}

\begin{theorem}[The three regimes]\label{thm:b1:transient-regimes:regimes}
\index{aperiodic regime}\index{critical regime}\index{pseudo-periodic regime}\index{pseudo-period}
The solutions of the homogeneous equation $x'' + (\omega_0/Q)x' +
\omega_0^2 x = 0$ are governed by the roots of $r^2 + (\omega_0/Q)r +
\omega_0^2 = 0$, of discriminant $\Delta = \omega_0^2(1/Q^2 - 4)$:
\begin{itemize}
  \item $Q < \tfrac12$, \emph{aperiodic}: two real negative roots
        $r_\pm = -\frac{\omega_0}{2Q}\big(1 \mp \sqrt{1 - 4Q^2}\big)$,
        $x = A\eu^{r_+t} + B\eu^{r_-t}$, a return without oscillation;
  \item $Q = \tfrac12$, \emph{critical}: a double root $r = -\omega_0$,
        $x = (A + Bt)\eu^{-\omega_0 t}$, the fastest return without
        overshoot;
  \item $Q > \tfrac12$, \emph{pseudo-periodic}: complex roots
        $r = -1/\tau \pm \iu\omega$ with
        \[
        \tau = \frac{2Q}{\omega_0}, \qquad
        \omega = \omega_0\sqrt{1 - \frac{1}{4Q^2}},
        \qquad
        x = \eu^{-t/\tau}\big(A\cos\omega t + B\sin\omega t\big),
        \]
        a damped oscillation of \emph{pseudo-period} $T = 2\pi/\omega$
        inside the envelope $\pm\sqrt{A^2 + B^2}\,\eu^{-t/\tau}$.
\end{itemize}
The full solution with a constant right-hand side is $x_\infty$ plus the
homogeneous solution; $A$ and $B$ follow from $x(0)$ and $x'(0)$.
\end{theorem}

\begin{proof}
Characteristic equation of a linear equation with constant
coefficients (mathematics course, second-order linear equations): the
roots are $r = -\omega_0/2Q \pm \tfrac12\sqrt\Delta$. For $Q > \tfrac12$,
$\sqrt\Delta = \iu\omega_0\sqrt{4 - 1/Q^2}$, giving the real part
$-\omega_0/2Q = -1/\tau$ and the imaginary part $\pm\omega_0\sqrt{1 - 1/4Q^2}$;
the real solutions are the combinations of $\eu^{-t/\tau}\cos\omega t$
and $\eu^{-t/\tau}\sin\omega t$.
\end{proof}

\begin{proposition}[Reading $Q$ on a pseudo-periodic trace]\label{prop:b1:transient-regimes:decrement}
\index{logarithmic decrement}
In the \omterm{thm:b1:transient-regimes:regimes}{pseudo-periodic regime}, two successive maxima are in the ratio
$x_{n+1}/x_n = \eu^{-T/\tau}$; the \emph{\omterm{prop:b1:transient-regimes:decrement}{logarithmic decrement}}
\[
\delta = \ln\frac{x_n}{x_{n+1}} = \frac{T}{\tau} = \frac{\pi}{Q\sqrt{1 - 1/4Q^2}}
\approx \frac{\pi}{Q} \quad (Q \gg 1)
\]
gives $Q$; the number of oscillations visible before the \omterm{def:b1:wave-propagation:signal}{amplitude}
falls below $5\%$ is about $Q$. For $Q \gg 1$, $\omega \approx \omega_0$
and $T \approx 2\pi/\omega_0$.
\end{proposition}

\begin{proof}
The envelope $\eu^{-t/\tau}$ multiplies by $\eu^{-T/\tau}$ per period;
$T/\tau = (2\pi/\omega)(\omega_0/2Q)$. The envelope reaches $\eu^{-3}
\approx 5\%$ at $3\tau = 6Q/\omega_0 \approx Q\,T$, i.e.\ after about
$Q$ periods.
\end{proof}

\begin{example}[An RLC ringing]\label{ex:b1:transient-regimes:rlcnum}
$L = \qty{10}{mH}$, $C = \qty{100}{nF}$, $R = \qty{100}{\ohm}$:
$\omega_0 = 1/\sqrt{10^{-9}} = \qty{3.16e4}{rad/s}$ ($f_0 =
\qty{5.03}{kHz}$), $Q = \sqrt{L/C}/R = 316/100 = 3.16$: pseudo-periodic,
$\tau = 2Q/\omega_0 = \qty{0.20}{ms}$, $\omega = 0.987\,\omega_0$,
$T = \qty{0.20}{ms}$, three visible oscillations. The critical resistance
is $R_c = 2\sqrt{L/C} = \qty{632}{\ohm}$; above it the discharge is
aperiodic.
\end{example}

\begin{method}[Solving a second-order transient]\label{met:b1:transient-regimes:second}
\begin{enumerate}
  \item Write the equation; put it in canonical form; read $\omega_0$,
        $Q$ and $x_\infty$.
  \item Decide the regime from $Q$ and write the general solution as
        $x_\infty$ plus the homogeneous solution.
  \item Find $x(0)$ and $x'(0)$ from the continuity of $u_C$ and $i_L$
        (for the RLC: $u_C(0^+) = u_C(0^-)$ and $u_C'(0^+) = i(0^-)/C$);
        solve for the two constants.
\end{enumerate}
\end{method}

\begin{example}[Initial conditions]\label{ex:b1:transient-regimes:ic}
The capacitor of the previous example is charged to $U_0$ and, at
$t = 0$, closed on $L$ and $R$ (no source): $x_\infty = 0$,
$u_C(0) = U_0$, $u_C'(0) = i(0)/C = 0$ since the coil current was zero.
Hence $A = U_0$ and $-A/\tau + B\omega = 0$, $B = U_0/(\omega\tau) =
U_0/\sqrt{4Q^2 - 1}$: $u_C = U_0\eu^{-t/\tau}[\cos\omega t + \sin(\omega
t)/\sqrt{4Q^2 - 1}]$ --- for $Q \gg 1$ simply $U_0\eu^{-t/\tau}\cos\omega_0 t$,
and the current $i = -C\,\dd u_C/\dd t$ peaks near $U_0\sqrt{C/L} =
U_0/(\omega_0 L)$.
\end{example}

\begin{remark}[Energy]\label{rem:b1:transient-regimes:energy}
The stored energy $\tfrac12 Cu_C^2 + \tfrac12 Li^2$ (or $\tfrac12 kx^2
+ \tfrac12 m\dot x^2$) decreases at the rate $Ri^2$ ($\alpha\dot x^2$):
differentiate and use the equation. Each pseudo-period the energy
shrinks by the factor $\eu^{-2T/\tau}$; with $Q = 3$ about $88\%$ of it
is lost per oscillation. A high $Q$ means a slow leak relative to the
oscillation --- a good clock, a sharp resonance
(\cref{ch:b1:sinusoidal-impedance}) --- and a low $Q$ a quick return:
the right choice for a suspension or a meter needle.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:transient-regimes:1}
An \omterm{prop:b1:transient-regimes:rcrl}{RC circuit}: $R = \qty{10}{k\ohm}$, $C = \qty{47}{nF}$, charged from
$E$. Compute $\tau$, $u_C(\tau)/E$, and the time to reach $99\%$ of $E$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:transient-regimes:2}
An \omterm{prop:b1:transient-regimes:rcrl}{RL circuit}: $L = \qty{20}{mH}$, $R = \qty{5.0}{\ohm}$, $E = \qty{10}{V}$.
Compute $\tau$, the final current, the current at $t = \tau$, and the
\omterm{def:b1:dc-circuits:voltage}{voltage} across the coil just after closing the switch.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:transient-regimes:3}
A series RLC: $L = \qty{10}{mH}$, $C = \qty{100}{nF}$, $R = \qty{100}{\ohm}$.
Compute $\omega_0$, $f_0$, $Q$; name the regime; give $\tau$ and the
pseudo-period.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:transient-regimes:4}
For the same $L$ and $C$, what resistance gives the \omterm{thm:b1:transient-regimes:regimes}{critical regime}?
What is $Q$ then? What happens for $R = \qty{2}{k\ohm}$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:transient-regimes:5}
A charged capacitor $C = \qty{1.0}{\micro F}$ discharges through an
unknown $R$; the \omterm{def:b1:dc-circuits:voltage}{voltage} halves every \qty{3.5}{ms}. Show that the
half-life is $\tau\ln 2$ and find $R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:transient-regimes:6}
Show by direct integration that charging a capacitor from $0$ to $E$
through $R$ dissipates exactly $\tfrac12 CE^2$ in $R$, independently of
$R$. What becomes of the efficiency of the transfer, and how could one
do better?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:transient-regimes:7}
On an oscilloscope, a free RLC decay shows successive maxima of
\qty{2.0}{V}, \qty{1.5}{V}, \qty{1.12}{V} at intervals of \qty{0.40}{ms}.
Deduce $Q$, $\omega_0$, and, with $C = \qty{1.0}{\micro F}$, the values
of $L$ and $R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:transient-regimes:8}
A capacitor charged to $U_0 = \qty{100}{V}$ ($C = \qty{10}{\micro F}$) is
switched onto a coil $L = \qty{1.0}{mH}$ of small resistance ($Q \gg 1$).
Write $u_C(t)$ and $i(t)$, and estimate the peak current. Where does
the energy sit at the peak?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:transient-regimes:9}
A quarter of a car: $m = \qty{400}{kg}$ on a spring $k = \qty{2.0e4}{N/m}$.
Compute $\omega_0$, $f_0$ and the critical damping coefficient. With a
damper $\alpha = \qty{2000}{N.s/m}$, give $Q$, the regime, the
pseudo-period and the decay time $\tau$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:transient-regimes:10}
A coil $L = \qty{0.50}{H}$ carries $I_0 = \qty{2.0}{A}$; the source is
disconnected and the coil is left on a bleed \omterm{def:b1:dc-circuits:resistor}{resistor} $R = \qty{100}{\ohm}$.
Give $i(t)$, $\tau$, the peak \omterm{def:b1:dc-circuits:voltage}{voltage} across the coil and the energy
dissipated. What would happen without the \omterm{def:b1:dc-circuits:resistor}{resistor}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:transient-regimes:11}
A capacitor $C = \qty{1.0}{\micro F}$ is charged from $E = \qty{12}{V}$
through $R_1 = \qty{10}{k\ohm}$ while $R_2 = \qty{20}{k\ohm}$ is connected
across it. Using a Thévenin equivalent, find the \omterm{def:b1:transient-regimes:firstorder}{time constant} and the
final \omterm{def:b1:dc-circuits:voltage}{voltage}; write $u_C(t)$ from $u_C(0) = 0$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:transient-regimes:12}
\omterm{thm:b1:transient-regimes:regimes}{Critical regime}, step from rest ($x(0) = x'(0) = 0$): show that
$x = x_\infty[1 - (1 + \omega_0 t)\eu^{-\omega_0 t}]$, and find
numerically the time to reach $95\%$ of $x_\infty$ (in units of
$1/\omega_0$). Compare with the settling time of the envelope for $Q = 5$
and for the slow root when $Q = 0.1$; conclude on why ``critical'' is the
engineer's target.
\end{exercise}

\section{Problem: Designing a suspension}

\begin{problem}[{Weekend problem --- the same differential equation on
a lab bench and under a car: how many bounces a worn shock absorber
allows, what the designer chooses instead, and how one number tells
the two apart}]
\label{pb:b1:transient-regimes:1}
Part I studies a series \omterm{prop:b1:transient-regimes:rlc}{RLC circuit}, $L = \qty{10}{mH}$, $C = \qty{100}{nF}$;
Parts II--IV a quarter-car model: a mass $m = \qty{350}{kg}$ resting on a
spring $k = \qty{22}{kN/m}$ and a shock absorber exerting $-\alpha\dot x$.
Overshoot of a step response in the \omterm{thm:b1:transient-regimes:regimes}{pseudo-periodic regime} (given):
$x_{\max}/x_\infty - 1 = \exp(-\pi\xi/\sqrt{1 - \xi^2})$, $\xi = 1/2Q$.

\medskip
\noindent\textbf{Part I --- The circuit.}
\begin{enumerate}
  \item Write the \omterm{thm:b1:dc-circuits:kirchhoff}{loop law} for the series RLC across a source $E$, and
        the differential equation for $u_C$.
  \item Put it in canonical form; express $\omega_0$ and $Q$.
  \item Compute $\omega_0$, $f_0$ and the critical resistance $R_c$.
  \item Write the characteristic equation and its discriminant; state
        the three regimes in terms of $Q$.
  \item In the pseudo-periodic case, give $\tau$ and $\omega$ and the
        form of $u_C(t)$.
  \item For $R = \qty{100}{\ohm}$ compute $Q$, $\tau$, the pseudo-period,
        and the number of visible oscillations.
  \item The capacitor, charged to $U_0$, is switched onto $L$ and $R$
        at $t = 0$. Give $u_C(0^+)$ and $\dd u_C/\dd t(0^+)$, with the
        physical reason for each.
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The quarter car.}
\begin{enumerate}[resume]
  \item Measuring $x$ from the equilibrium position, write Newton's
        second law for the body and show that weight drops out.
  \item Put the equation in canonical form; give $\omega_0$ and $Q$ in
        terms of $m$, $k$, $\alpha$, and list the electromechanical
        correspondences.
  \item Compute $\omega_0$, $f_0$ and the critical coefficient
        $\alpha_c$.
  \item The designer fits $\alpha = \qty{4000}{N.s/m}$: compute $Q$,
        $\xi$, and name the regime.
  \item Compute $\tau$ and the pseudo-period.
  \item Compute the overshoot after a step (a kerb): is the car
        comfortable?
  \item What resistance $R$ would give the circuit of Part I the same
        $Q$ as this suspension?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The worn damper.}
Oil has leaked: $\alpha = \qty{1200}{N.s/m}$.
\begin{enumerate}[resume]
  \item Compute $Q$, the pseudo-period and $\tau$.
  \item By what factor do successive maxima shrink? How many bounces
        are visible before the \omterm{def:b1:wave-propagation:signal}{amplitude} is below $5\%$?
  \item The bounce test: one pushes the corner down by $x_0 =
        \qty{5.0}{cm}$ and releases. Give the initial conditions, the
        energy stored in the spring, and say where that energy goes.
  \item Estimate the peak velocity of the body during the first
        return ($\approx x_0\omega_0$ for $Q \gg 1$) and the peak damper
        force.
  \item Why is the exactly critical damping ($Q = \tfrac12$) not the
        designer's choice, although it gives no overshoot? (Think of
        the last few millimeters of the return.)
  \item Compute the overshoot for the worn damper and compare with
        Part II.
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- Diagnosing $Q$ from a trace.}
An accelerometer on the body records the free decay.
\begin{enumerate}[resume]
  \item For the worn car, successive maxima read $1.00$, $0.25$, $0.062$
        (arbitrary units). Compute the \omterm{prop:b1:transient-regimes:decrement}{logarithmic decrement}, deduce
        $\xi$ exactly from $\delta = 2\pi\xi/\sqrt{1 - \xi^2}$, then $Q$;
        compare with Part III.
  \item For the healthy car no second maximum is visible; only the
        overshoot, $3.8\%$, can be read. Invert the overshoot formula to
        find $\xi$ and $Q$.
  \item What fraction of the mechanical energy does the worn damper
        dissipate per pseudo-period?
  \item A technician who has no car at hand builds the RLC of Part I
        with $R$ chosen so that $Q = 2.3$: what $R$, and why is the
        oscilloscope trace, plotted against $\omega_0 t$, identical in
        shape to the car's?
  \item Summarize: which \omterm{def:b1:units-dimensions:dimension}{dimensionless} number decides the shape of any
        second-order transient, what value a suspension designer aims
        for, and what a driver should conclude from ``more than one
        bounce''.
\end{enumerate}
\end{problem}
