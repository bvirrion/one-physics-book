\chapter{Mechanical Oscillators: Damping and Resonance}\label{ch:b1:oscillators-resonance}

Push a child's swing at the wrong moment and nothing much happens; push
it once per period, gently, and within a dozen swings the child is
shrieking at the top of the arc. A washing machine shudders violently
at one particular speed of its spin cycle and is quiet above it. A
wine glass sings when a wet finger circles its rim at just the right
rate, and shatters when a singer holds the same note loudly enough.
Every one of these is a \emph{resonance}: a system with a \omterm{def:b1:oscillators-resonance:harmonic}{natural
frequency}, driven near that frequency, accumulates energy cycle after
cycle until friction bleeds it off as fast as it comes in. This
chapter takes the mechanical oscillator --- free, damped, then driven
--- and derives the resonance curves whose electrical twins appeared
in \cref{ch:b1:sinusoidal-impedance}.

\section{The harmonic oscillator}

\begin{definition}[Harmonic oscillator]\label{def:b1:oscillators-resonance:harmonic}
\index{harmonic oscillator}\index{natural frequency}
A system whose displacement $x$ from a stable equilibrium obeys
\[
\ddot x + \omega_0^2\,x = 0
\]
is a \emph{harmonic oscillator} of \emph{\omterm{def:b1:transient-regimes:secondorder}{natural angular frequency}}
$\omega_0$ (period $T_0 = 2\pi/\omega_0$). The solution $x = A\cos(\omega_0t
+ \varphi)$ has an \omterm{def:b1:wave-propagation:signal}{amplitude} and phase fixed by the initial conditions
and a period independent of the \omterm{def:b1:wave-propagation:signal}{amplitude} (\emph{isochronism}).
\end{definition}

\begin{proposition}[Where harmonic oscillators come from]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item Mass $m$ on a spring $k$: $\omega_0^2 = k/m$ (\omterm{prop:b1:newton-dynamics:springstring}{Hooke's law}).
  \item \omterm{ex:b1:newton-dynamics:pendulum}{Simple pendulum} of length $\ell$, small angles: $\omega_0^2 = g/\ell$.
  \item Any particle near a stable equilibrium of a potential $E_p$:
        $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability}).
  \item LC circuit: $\omega_0^2 = 1/LC$, with $q \leftrightarrow x$,
        $L \leftrightarrow m$, $1/C \leftrightarrow k$.
\end{itemize}
The energy $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ is constant, sloshing between kinetic and potential
form twice per period.
\end{proposition}

\begin{proof}
Newton's second law with $F = -kx$; the pendulum equation
$\ddot\theta + (g/\ell)\sin\theta = 0$ with $\sin\theta \approx \theta$;
Taylor's formula at the minimum of $E_p$; the \omterm{thm:b1:dc-circuits:kirchhoff}{loop law} of the LC
circuit. Energy: differentiate and use the equation.
\end{proof}

\begin{example}[A bobbing cylinder]\label{ex:b1:oscillators-resonance:bob}
A vertical cylinder of cross-section $S$ floats with a depth $h$
immersed ($mg = \rho_wgSh$, buoyancy from \cref{ch:b1:fluid-statics}).
Pushed down by $x$, it receives an extra upward force $\rho_wgSx$:
$m\ddot x = -\rho_wgSx$, so $\omega_0^2 = \rho_wgS/m = g/h$ --- the same
as a pendulum of length $h$. A buoy immersed by \qty{1}{m} bobs with a
\qty{2}{s} period, whatever its mass.
\end{example}

\section{The damped oscillator}

\begin{proposition}[Damped oscillator]\label{prop:b1:oscillators-resonance:damped}
\index{damped oscillator}\index{quality factor!mechanical}
With a viscous friction $-\alpha\dot x$, the equation becomes
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
the canonical second-order form of \cref{ch:b1:transient-regimes}: for
$Q > \tfrac12$ a pseudo-periodic decay $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ with $\tau = 2Q/\omega_0$, $\omega = \omega_0\sqrt{1 - 1/4Q^2}$; the
energy decays as $\eu^{-2t/\tau}$ and
\[
Q = 2\pi\,\frac{\text{energy stored}}{\text{energy lost per period}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
Divide $m\ddot x + \alpha\dot x + kx = 0$ by $m$. For the energy: $E
\propto A^2\eu^{-2t/\tau}$ loses the fraction $2T/\tau = 2\pi/Q$ per
period when that is small.
\end{proof}

\begin{example}[Ringing times]\label{ex:b1:oscillators-resonance:ringing}
A tuning fork at \qty{440}{Hz} still sounds ten seconds after being
struck: its \omterm{def:b1:wave-propagation:signal}{amplitude} fell to about $5\%$ in $3\tau = \qty{10}{s}$, so
$\tau = \qty{3.3}{s}$ and $Q = \omega_0\tau/2 \approx 4600$. A car's
suspension ($Q \approx 0.7$) stops within one period; a child's swing
($Q \approx 20$) loses a third of its energy per swing and needs a push
every time.
\end{example}

\section{Forced oscillations and resonance}

\begin{theorem}[Sinusoidally driven oscillator]\label{thm:b1:oscillators-resonance:forced}
\index{forced oscillations}\index{resonance!amplitude}\index{resonance!velocity}
Driven by a force $F_0\cos\omega t$, the \omterm{prop:b1:oscillators-resonance:damped}{damped oscillator}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
settles, after a transient of duration $\sim\tau$, into the \omterm{prop:b1:sinusoidal-impedance:forced}{forced
regime} $x = X\cos(\omega t + \varphi)$ with, writing $x_r = \omega/\omega_0$
and $X_0 = F_0/k$ the static deflection,
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
The \emph{\omterm{def:b1:wave-propagation:signal}{amplitude}} $X$ peaks, for $Q > 1/\sqrt2$, at $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ with $X_{\max} \approx QX_0$; the \emph{velocity
\omterm{def:b1:wave-propagation:signal}{amplitude}} $V = \omega X$ peaks exactly at $\omega_0$, where $V_{\max} =
F_0/\alpha$, and its resonance curve
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
has the bandwidth $\Delta\omega = \omega_0/Q$. At $\omega_0$ the displacement
lags the force by $\pi/2$ and the velocity is in phase with it.
\end{theorem}

\begin{proof}
Complex method (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$, so $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$; divide numerator and denominator by
$\omega_0^2 = k/m$. Modulus and argument follow; the maximum of $X$ was
located in \cref{prop:b1:sinusoidal-impedance:overvoltage} (same
function). Velocity: $\underline V = j\omega\underline X$ gives $V = \omega X$;
writing $(1 - x_r^2)/x_r = 1/x_r - x_r$ turns the denominator into
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$, i.e.\ the RLC form. At $x_r = 1$,
$\underline X = -jQX_0$: a lag of $\pi/2$.
\end{proof}

\begin{remark}[Two resonances, one circuit]\label{rem:b1:oscillators-resonance:analogy}
\index{electromechanical analogy}
The correspondence $x \leftrightarrow q$, $\dot x \leftrightarrow i$,
$m \leftrightarrow L$, $\alpha \leftrightarrow R$, $k \leftrightarrow 1/C$,
$F \leftrightarrow e$ maps the driven oscillator onto the series RLC of
\cref{ch:b1:sinusoidal-impedance}: the velocity resonance is the current
resonance (always at $\omega_0$, bandwidth $\omega_0/Q$), the \omterm{def:b1:wave-propagation:signal}{amplitude}
resonance is the charge, i.e.\ \omterm{def:b1:dc-circuits:voltage}{capacitor-voltage}, resonance (slightly
below $\omega_0$, absent for low $Q$). Power absorbed, $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$, peaks at $\omega_0$ with the value $F_0^2/2\alpha$,
and the half-power points are the velocity bandwidth --- exactly as for
the RLC.
\end{remark}

\begin{example}[Resonant speeds]\label{ex:b1:oscillators-resonance:road}
A car's suspension has $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
On a road with corrugations every $\lambda = \qty{10}{m}$ the excitation
frequency is $v/\lambda$: resonance at $v = \lambda f_0 = \qty{13}{m/s}$,
about \qty{47}{km/h} --- the speed at which a washboard track shakes the
car hardest, and the reason a damper with $Q \approx 0.7$ is fitted:
with $Q = 5$ the body would bounce with five times the road's
\omterm{def:b1:wave-propagation:signal}{amplitude}. Above resonance ($x_r \gg 1$) the \omterm{def:b1:wave-propagation:signal}{amplitude} falls as
$X_0/x_r^2$: the body ``floats'' over fast bumps --- isolation, the other
face of resonance.
\end{example}

\section{Base excitation: isolation and the seismometer}

\begin{proposition}[Oscillator shaken by its support]\label{prop:b1:oscillators-resonance:base}
\index{base excitation}\index{vibration isolation}\index{seismometer}
Let the support of a mass--spring--damper move by $x_g(t)$ (the road
under a car, the ground under a seismometer), and let $u$ be the
displacement of the mass relative to the support and $x = x_g + u$ its
absolute displacement (from equilibrium). Then
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
and for $x_g = X_g\cos\omega t$, with $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
For $x_r \gg 1$: $U \to X_g$ (the mass stays still in space --- a
\emph{seismometer} reads the ground's displacement) and $X \to 0$ (the
mass is \emph{isolated} from the shaking); for $x_r \ll 1$: $U \approx
\ddot x_g/\omega_0^2$ (an \emph{accelerometer}) and $X \approx X_g$ (the mass
follows the support).
\end{proposition}

\begin{proof}
The spring and damper act on the relative motion: $m\ddot x = -ku -
\alpha\dot u$, and $\ddot x = \ddot x_g + \ddot u$. In complex form
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$, whence
$U/X_g$; then $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[Why machines sit on soft mounts]\label{ex:b1:oscillators-resonance:mounts}
A \qty{50}{Hz} motor on rubber mounts tuned to $f_0 = \qty{10}{Hz}$
($x_r = 5$, $Q \approx 2$): $X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$
--- nine tenths of the vibration stays out of the floor. The mounts must
be \emph{softer} than intuition suggests: stiffening them raises $f_0$
toward $50$ Hz and destroys the isolation. The same formula, read the
other way, is the car of \cref{ex:b1:oscillators-resonance:road} on a
washboard road.
\end{example}

\section{Energy: how a resonance builds up}

\begin{proposition}[Energy balance in the forced regime]\label{prop:b1:oscillators-resonance:energy}
In the steady \omterm{prop:b1:sinusoidal-impedance:forced}{forced regime} the \omterm{thm:b1:sinusoidal-impedance:power}{average power} supplied by the driving
force equals the \omterm{thm:b1:sinusoidal-impedance:power}{average power} dissipated by friction, $\tfrac12\alpha V^2$.
At resonance the stored energy $\tfrac12 mV_{\max}^2$ is $Q/2\pi$ times
the energy supplied per period: the oscillator accumulates the drive's
work over about $Q$ cycles before losses match the input, which is also
the number of cycles ($\sim\tau/T$) the transient lasts.
\end{proposition}

\begin{proof}
Average the energy theorem over a period: $\langle\dd E/\dd t\rangle = 0$
in \omterm{def:b1:transient-regimes:firstorder}{steady state}, so $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. At resonance $V_{\max} = F_0/\alpha$, stored energy
$\tfrac12 m(F_0/\alpha)^2$, supplied per period $\tfrac12\alpha(F_0/\alpha)^2T$;
ratio $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[The swing]\label{ex:b1:oscillators-resonance:swing}
A swing of length \qty{2.5}{m} ($T_0 = \qty{3.2}{s}$, $Q \approx 20$) gets
a push of \qty{50}{N} over \qty{0.5}{m} each period: \qty{25}{J} per
cycle. It grows until the loss per period, $2\pi E/Q$, equals
\qty{25}{J}: $E = \qty{80}{J}$; with a \qty{30}{kg} child, $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ gives $\theta_{\max} \approx 27^\circ$.
Pushing at any other rhythm, the work of the pushes would alternate in
sign and average to nothing.
\end{example}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:1}
A \qty{0.20}{kg} mass on a spring of stiffness \qty{80}{N/m} oscillates
with \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{5.0}{cm}. \omterm{def:b1:wave-propagation:signal}{Angular frequency}, frequency, period,
maximal speed, total energy.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:2}
Period of a \qty{1.0}{m} pendulum on Earth and on the Moon
($g = \qty{1.62}{m/s^2}$). What happens to the mass--spring period on the
Moon?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:3}
An oscillator has $f_0 = \qty{2.0}{Hz}$ and $Q = 50$. Decay time $\tau$
of the \omterm{def:b1:wave-propagation:signal}{amplitude}; time for the energy to halve; number of oscillations
before the \omterm{def:b1:wave-propagation:signal}{amplitude} is down to $5\%$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:4}
A mass on a spring ($k = \qty{100}{N/m}$, $Q = 10$) is driven by a force
of \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{N}. Static deflection; \omterm{def:b1:wave-propagation:signal}{amplitude} at resonance;
\omterm{def:b1:wave-propagation:signal}{amplitude} at $\omega = 3\omega_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:5}
A car ($f_0 = \qty{1.3}{Hz}$, $Q = 0.7$) crosses speed bumps \qty{8.0}{m}
apart. Resonant speed? At \qty{60}{km/h}, what fraction of the bump
\omterm{def:b1:wave-propagation:signal}{amplitude} does the body feel? Comment.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:6}
Explain with the energy balance why a swing must be pushed at its own
period; for $Q = 20$ and a push adding \qty{25}{J} per cycle, find the
steady-state energy, and the \omterm{def:b1:wave-propagation:signal}{amplitude} for a \qty{30}{kg} child on a
\qty{2.5}{m} swing.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:7}
Successive maxima of a free oscillation decrease by $10\%$ each period.
Find $Q$ and the bandwidth (in hertz) of its velocity resonance if
$f_0 = \qty{2.0}{Hz}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:8}
Give the phase of the displacement relative to the force well below
resonance, at resonance and well above, and the phase of the velocity
at resonance. Why is the velocity resonance the one that matters for
power?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:9}
Show that the velocity \omterm{def:b1:wave-propagation:signal}{amplitude} $V = \omega X$ is maximal exactly at
$\omega_0$, with $V_{\max} = F_0/\alpha$, and that $V/V_{\max}$ has the
form $1/\sqrt{1 + Q^2(x_r - 1/x_r)^2}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:oscillators-resonance:10}
Compute the \omterm{thm:b1:sinusoidal-impedance:power}{average power} absorbed at resonance for $F_0 = \qty{1.0}{N}$,
$\alpha = \qty{0.050}{N.s/m}$, and the stored energy if $m = \qty{0.20}{kg}$;
check that their ratio is $Q/\omega_0$ with $\omega_0 = \qty{20}{rad/s}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:oscillators-resonance:11}
A tuning fork at \qty{440}{Hz} is heard for \qty{10}{s} (\omterm{def:b1:wave-propagation:signal}{amplitude} down
to $5\%$). Estimate $\tau$ and $Q$, the bandwidth of its resonance, and
comment on why it is a good frequency standard.
\end{exercise}

\section{Problem: The seismograph}

\begin{problem}[{Weekend problem --- a mass hangs from a spring inside
a box bolted to the ground; the ground shakes: what does the pen
record, why the same instrument measures displacement at high
frequency and acceleration at low frequency, and why a long-period
seismometer cannot simply be a long spring}]
\label{pb:b1:oscillators-resonance:1}
A mass $m = \qty{1.0}{kg}$ hangs from a spring of stiffness $k$ inside
a rigid frame fixed to the ground; a damper exerts $-\alpha\dot u$, where
$u$ is the displacement of the mass relative to the frame, measured
from its equilibrium position. The ground, hence the frame, moves
vertically by $x_g(t)$. The frame of the distant stars is inertial.

\medskip
\noindent\textbf{Part I --- The equation.}
\begin{enumerate}
  \item Write the absolute position of the mass as $x = x_g + u +
        \text{const}$, list the forces on the mass (in the \omterm{thm:b1:newton-dynamics:laws}{inertial
        frame}), and show that
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item Put it in canonical form; identify $\omega_0$ and $Q$.
  \item The instrument is built with $f_0 = \qty{0.50}{Hz}$ and $Q = 0.70$:
        compute $k$ and $\alpha$.
  \item A constant ground acceleration $a_g$ (the frame tilting, say)
        gives what steady $u$? Interpret.
  \item What does the pen attached to the mass, writing on a drum fixed
        to the frame, record?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- Harmonic ground motion.}
The ground moves as $x_g = X_g\cos\omega t$; write $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item Using \omterm{def:b1:sinusoidal-impedance:complex}{complex amplitudes}, show that $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item Deduce $U/X_g$ as a function of $x_r$ and $Q$.
  \item Limit $x_r \gg 1$: show $U \to X_g$ and explain physically what
        the mass does (what is the instrument measuring?).
  \item Limit $x_r \ll 1$: show $U \approx x_r^2X_g = a_g/\omega_0^2$ with
        $a_g$ the ground's acceleration \omterm{def:b1:wave-propagation:signal}{amplitude}: what is the instrument
        now?
  \item Value at $x_r = 1$; why is $Q \approx 0.7$ a sensible choice?
  \item Sketch $U/X_g$ against $x_r$ (log scales) for $Q = 0.7$ and
        $Q = 5$.
  \item Phase of $u$ relative to $x_g$ at high frequency; interpret.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- Designing for earthquakes.}
Distant earthquakes produce surface waves at \qty{0.05}{} to \qty{0.1}{Hz}
with millimeter \omterm{def:b1:wave-propagation:signal}{amplitudes}; local tremors reach \qty{5}{} to \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item To record \qty{0.05}{Hz} waves as displacements one wants
        $f_0 \ll \qty{0.05}{Hz}$, say \qty{0.02}{Hz}. What static
        stretch would the spring have under the mass's weight? Comment.
  \item A \omterm{ex:b1:newton-dynamics:pendulum}{simple pendulum} of the same frequency: what length?
  \item Real long-period instruments use a nearly horizontal
        ``garden-gate'' pendulum, or electronic force feedback. Explain
        in one sentence what each achieves.
  \item For the \qty{0.50}{Hz} instrument, compute the static stretch:
        feasible?
  \item With it, a \qty{0.10}{Hz} wave of \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm} gives
        what $U$? In which regime is the instrument?
  \item A \qty{5.0}{Hz} local tremor of \omterm{def:b1:wave-propagation:signal}{amplitude} \qty{1.0}{mm}: $U$?
        Regime?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The accelerometer in a phone.}
A micro-machined mass on silicon springs has $f_0 = \qty{1.0}{kHz}$ and
$Q = 0.7$.
\begin{enumerate}[resume]
  \item A \qty{10}{Hz} shake of acceleration \omterm{def:b1:wave-propagation:signal}{amplitude} $g$: compute
        $U$. How is such a displacement read?
  \item Up to what frequency is the reading within $10\%$ of $a_g/\omega_0^2$?
  \item What \omterm{def:b1:transient-regimes:firstorder}{time constant} $\tau = 2Q/\omega_0$ governs its response to a
        jolt, and why does that matter for detecting a fall?
  \item The seismometer's $\tau$: value and consequence after a sudden
        ground step.
  \item Why is critical (or near-critical) damping chosen for both
        instruments, rather than a high $Q$?
  \item The phone is shaken at exactly \qty{1.0}{kHz}: by what factor is
        the reading off?
  \item Summarize: one device, two regimes, decided by which single
        comparison.
\end{enumerate}
\end{problem}
