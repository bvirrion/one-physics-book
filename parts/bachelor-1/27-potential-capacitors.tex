\chapter{Electric Potential; Conductors and Capacitors}\label{ch:b1:potential-capacitors}

A field is three numbers at every point; a potential is one. Because
the electrostatic force is \omterm{def:b1:work-and-energy:conservative}{conservative}, the whole field of a charge
distribution can be folded into a single scalar function, the
potential, from which it is recovered by differentiation --- the same
economy that turned forces into \omterm{def:b1:work-and-energy:conservative}{potential energy} in mechanics. The
potential is what a voltmeter reads, what a battery maintains, what
accelerates electrons in a tube; its level surfaces map a field at a
glance. This chapter builds it, uses it on the dipole (the first
electric model of a molecule), and then turns to the two objects every
circuit is made of: conductors, on which charge arranges itself so as
to kill the field inside, and \omterm{def:b1:potential-capacitors:capacitance}{capacitors}, which store charge and energy
in a field.

\section{The electrostatic potential}

\begin{theorem}[The electrostatic field is conservative]\label{thm:b1:potential-capacitors:potential}
\index{electrostatic potential}\index{volt}\index{circulation}
There exists a scalar function $V(M)$, the \emph{\omterm{thm:b1:potential-capacitors:potential}{electrostatic
potential}} (volt, \unit{V}), such that
\[
\vect E = -\overrightarrow{\operatorname{grad}}\,V , \qquad
V_A - V_B = \int_A^B \vect E\cdot\dd\vect l
\]
along any path from $A$ to $B$ (the \emph{circulation} of $\vect E$
from $A$ to $B$ is independent of the path, and zero round a closed
loop). For a point charge $q$ at $O$, $V(M) = \dfrac{q}{4\pi\varepsilon_0\,OM}$
with the convention $V(\infty) = 0$; for several charges the potentials
add. The \omterm{def:b1:work-and-energy:conservative}{potential energy} of a charge $q'$ at $M$ is $E_p = q'V(M)$, and
the work of the electric force from $A$ to $B$ is $q'(V_A - V_B)$.
\end{theorem}

\begin{proof}
The Coulomb force of a fixed charge $q$ on $q'$ has the form of Newton's
\omterm{def:b1:central-forces:gravitation}{gravitation} with $-Gm_1m_2$ replaced by $qq'/4\pi\varepsilon_0$; the same
computation that gave $E_p = -Gm_1m_2/r$
(\cref{prop:b1:work-and-energy:eps}) gives $E_p = qq'/4\pi\varepsilon_0r$:
dividing by $q'$ gives $V$, and $\vect E = \vect F/q' =
-\overrightarrow{\operatorname{grad}}(E_p)/q'$. Superposition carries the
result to any distribution; the circulation formula is the definition
of the gradient integrated along the path.
\end{proof}

\begin{proposition}[Reading a potential]\label{prop:b1:potential-capacitors:reading}
\index{equipotential surface}
\begin{enumerate}
  \item The \emph{\omterm{prop:b1:potential-capacitors:reading}{equipotential surfaces}} $V = \text{const}$ are
        everywhere perpendicular to the \omterm{def:b1:electrostatics-gauss:field}{field lines}, and the field
        points toward \emph{decreasing} potential.
  \item $|\vect E|$ is the rate of change of $V$ along the steepest
        direction: where equipotentials are close, the field is strong.
  \item $V$ is defined up to an additive constant; only differences
        (\omterm{def:b1:dc-circuits:voltage}{voltages}) are measurable. Ground or infinity is set to zero by
        convention.
  \item In \omterm{def:b1:point-kinematics:spherical}{spherical coordinates} with symmetry about $O$, $E_r = -\dd V/
        \dd r$; in \omterm{def:b1:point-kinematics:cylindrical}{polar coordinates} $(r, \theta)$ in a plane,
        $E_r = -\partial V/\partial r$ and $E_\theta = -\dfrac1r\,\partial V/\partial\theta$.
\end{enumerate}
\end{proposition}

\begin{proof}
On an equipotential $\dd V = -\vect E\cdot\dd\vect l = 0$ for every
displacement in the surface, so $\vect E$ is normal to it; moving along
$\vect E$, $\dd V = -E\,\dd l < 0$. The polar components follow from
$\dd V = (\partial V/\partial r)\dd r + (\partial V/\partial\theta)\dd\theta$ and
$\dd\vect l = \dd r\,\vect e_r + r\,\dd\theta\,\vect e_\theta$.
\end{proof}

\begin{method}[Computing a potential]\label{meth:b1:potential-capacitors:compute}
Either sum $\dd q/4\pi\varepsilon_0r$ over the distribution (when the sum is
tractable and the distribution is finite, so that $V(\infty) = 0$ makes
sense), or integrate $\dd V = -\vect E\cdot\dd\vect l$ along a convenient
path from a point where $V$ is known, using the field found by \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's
law}. For infinite distributions (wire, plane) only the second way
works, and the zero of $V$ must be chosen at a finite point.
\end{method}

\begin{proposition}[Potentials of the classic distributions]\label{prop:b1:potential-capacitors:classic}
\begin{enumerate}
  \item Sphere of radius $R$ with $Q$ in its volume: $V(r) = \dfrac{Q}{4\pi\varepsilon_0r}$
        outside; $V(r) = \dfrac{Q}{4\pi\varepsilon_0}\,\dfrac{3R^2 - r^2}{2R^3}$
        inside, so $V(0) = \tfrac32\,Q/4\pi\varepsilon_0R$. With $Q$ on the surface,
        $V$ is uniform inside, equal to $Q/4\pi\varepsilon_0R$.
  \item Infinite wire $\lambda$: $V(r) = -\dfrac{\lambda}{2\pi\varepsilon_0}\ln\dfrac{r}{r_0}$.
  \item \omterm{prop:b1:potential-capacitors:three}{Plane capacitor}, field $E$ from plate~1 to plate~2 a distance $d$
        apart: $V$ decreases linearly across the gap and $V_1 - V_2 = Ed$.
\end{enumerate}
\end{proposition}

\begin{proof}
Integrate $E_r = -\dd V/\dd r$ with the fields of
\cref{prop:b1:electrostatics-gauss:three}, from infinity inward for the
sphere (continuity at $R$ fixes the inside constant), from $r_0$ for the
wire, across the gap for the \omterm{def:b1:potential-capacitors:capacitance}{capacitor}.
\end{proof}

\begin{example}[The electron-volt, again]\label{ex:b1:potential-capacitors:eV}
An electron crossing a gap of \qty{100}{V} gains $eU = \qty{100}{eV} =
\qty{1.6e-17}{J}$, whatever the shape of the field between the
electrodes: the potential makes the path irrelevant
(\cref{def:b1:charged-particles:eV}). A \qty{1}{\micro C} charge at
\qty{1}{m} from another has $V = \qty{9}{kV}$ and the pair stores
\qty{9}{mJ}; the potential at the centre of a gold nucleus is
\qty{24}{MV} (\cref{exo:b1:potential-capacitors:6}).
\end{example}

\section{The electric dipole}

\begin{definition}[Electric dipole]\label{def:b1:potential-capacitors:dipole}
\index{electric dipole}\index{dipole moment}
Two opposite charges $-q$ at $N$ and $+q$ at $P$, a distance $a = NP$
apart, form an \emph{electric dipole} of \emph{dipole moment}
$\vect p = q\,\vect{NP}$ (\unit{C.m}), pointing from $-$ to $+$. The dipole
approximation is the study of its field at distances $r \gg a$. A
molecule whose centres of positive and negative charge do not coincide
(water, $p = \qty{6.2e-30}{C.m}$) is a dipole; a neutral atom in a field
becomes one (induced dipole).
\end{definition}

\begin{proposition}[Field of a dipole]\label{prop:b1:potential-capacitors:dipolefield}
At $M$ with $OM = r \gg a$ ($O$ the middle of $NP$) and $\theta$ the
angle between $\vect p$ and $\vect{OM}$,
\[
V(M) = \frac{p\cos\theta}{4\pi\varepsilon_0 r^2}, \qquad
E_r = \frac{2p\cos\theta}{4\pi\varepsilon_0r^3}, \qquad
E_\theta = \frac{p\sin\theta}{4\pi\varepsilon_0r^3} .
\]
The field of a dipole falls as $1/r^3$, faster than a charge's
$1/r^2$: from afar the two charges almost cancel, and what remains is
their separation.
\end{proposition}

\begin{proof}
$V = \dfrac{q}{4\pi\varepsilon_0}\Bigl(\dfrac1{PM} - \dfrac1{NM}\Bigr)$ with
$PM \approx r - \tfrac a2\cos\theta$ and $NM \approx r + \tfrac a2\cos\theta$
(projection of $\pm\tfrac a2$ on $\vect{OM}$); to first order in $a/r$,
$1/PM - 1/NM \approx a\cos\theta/r^2$. Then $E_r = -\partial V/\partial r$ and
$E_\theta = -(1/r)\,\partial V/\partial\theta$.
\end{proof}

\begin{proposition}[Dipole in an external field]\label{prop:b1:potential-capacitors:dipoleinfield}
\index{dipole!torque}\index{dipole!energy}
A dipole $\vect p$ in a uniform field $\vect E$ feels no net force, a
\omterm{def:b1:angular-momentum:moment}{torque} $\vect\Gamma = \vect p\wedge\vect E$ that tends to align it with
the field, and has the \omterm{def:b1:work-and-energy:conservative}{potential energy} $E_p = -\vect p\cdot\vect E$,
minimal when aligned. In a non-uniform field it is moreover pulled
toward the regions of strong field (in one dimension, $F_x = p\,\dd E/\dd x$
for an aligned dipole).
\end{proposition}

\begin{proof}
Forces $\pm q\vect E$ cancel; their moment about $O$ is $\vect{OP}\wedge
q\vect E + \vect{ON}\wedge(-q\vect E) = q\vect{NP}\wedge\vect E$. Energy:
$qV(P) - qV(N) = -q\,\vect{NP}\cdot\vect E$ for a uniform field (since $V(P)
- V(N) = -\vect E\cdot\vect{NP}$). Non-uniform, aligned along $x$: $F_x =
qE(x + a) - qE(x) \approx qa\,\dd E/\dd x$.
\end{proof}

\begin{example}[Why the comb attracts the paper]\label{ex:b1:potential-capacitors:comb}
A charged comb creates a field that falls off with distance; a scrap of
paper, neutral, is polarized by it (induced dipole along $\vect E$), and
the force $p\,\dd E/\dd x$ on an aligned dipole pulls it toward the
strong-field region: the comb. Sign of the comb's charge irrelevant;
the same mechanism holds the balloon on the wall and makes the
water stream bend. For water molecules, $pE$ in a \qty{1}{MV/m} field
is \qty{6e-24}{J}, $600$ times less than $k_BT$ at room temperature:
only a slight net alignment survives the thermal tumbling
(\cref{exo:b1:potential-capacitors:7}).
\end{example}

\section{Conductors in electrostatic equilibrium}

\begin{theorem}[Properties of a conductor in equilibrium]\label{thm:b1:potential-capacitors:conductor}
\index{conductor in equilibrium}\index{Coulomb's theorem}\index{Faraday cage}
In a conductor where the free charges are at rest:
\begin{enumerate}
  \item $\vect E = \vect 0$ inside, and the conductor is an equipotential
        volume: $V$ uniform in it and on its surface.
  \item Any net charge sits on the surface, with a surface density
        $\sigma$; the inside is neutral.
  \item Just outside, the field is normal to the surface and equals
        $\vect E = \dfrac{\sigma}{\varepsilon_0}\,\vect n$ (\emph{Coulomb's theorem}).
  \item A cavity inside a conductor, empty of charge, has $\vect E =
        \vect 0$ and uniform $V$ whatever happens outside (\emph{\omterm{thm:b1:potential-capacitors:conductor}{Faraday
        cage}}).
\end{enumerate}
\end{theorem}

\begin{proof}
(1) A field would move the free charges: equilibrium requires $\vect E =
\vect 0$, hence $\dd V = 0$ between any two inner points; the surface is
the limit. (2) \omterm{thm:b1:electrostatics-gauss:gauss}{Gauss's law} on any closed surface drawn inside the
conductor: zero flux, zero enclosed charge. (3) The surface is an
equipotential, so $\vect E$ just outside is normal to it; a pillbox
straddling the surface has flux $E\,\dd S$ through its outer face only
(inside $\vect E = \vect 0$) and charge $\sigma\,\dd S$. (4) The cavity
wall is an equipotential; a \omterm{def:b1:electrostatics-gauss:field}{field line} inside the cavity would have to
go from the wall to the wall across a potential drop --- impossible on
an equipotential; so no line, no field (admitted in this form).
\end{proof}

\begin{example}[Point effect and lightning rod]\label{ex:b1:potential-capacitors:point}
\index{point effect}
Two spheres of radii $R_1 > R_2$, far apart and joined by a wire, share
the same $V = Q_i/4\pi\varepsilon_0R_i$, so $\sigma_i = Q_i/4\pi R_i^2 = \varepsilon_0V/R_i$:
the \emph{smaller} the \omterm{def:b1:point-kinematics:frenet}{radius of curvature}, the \emph{larger} the surface
charge and the field $\sigma/\varepsilon_0 = V/R$. At a sharp point the field
exceeds the breakdown value of air (\qty{3}{MV/m}) long before it does
anywhere else: the air ionizes and charge leaks away quietly. A
lightning rod works by this \emph{\omterm{ex:b1:potential-capacitors:point}{point effect}} --- it does not attract
the stroke so much as offer it a path, and bleed the charge under a
cloud before it builds up.
\end{example}

\section{Capacitors}

\begin{definition}[Capacitance]\label{def:b1:potential-capacitors:capacitance}
\index{capacitance}\index{capacitor}\index{farad}
An isolated conductor at potential $V$ carries a charge $Q$
proportional to $V$: $Q = CV$, with $C$ its \emph{capacitance} (farad,
\unit{F}); for a sphere $C = 4\pi\varepsilon_0R$. A \emph{capacitor} is a pair
of conductors (\emph{plates}) carrying opposite charges $\pm Q$ and
facing each other so that the field is confined between them; its
capacitance is $C = Q/U$ with $U = V_1 - V_2$ the \omterm{def:b1:dc-circuits:voltage}{voltage} between the
plates. The symbol and the circuit law $i = C\,\dd u/\dd t$ were used in
\cref{ch:b1:transient-regimes}.
\end{definition}

\begin{proposition}[Three capacitors]\label{prop:b1:potential-capacitors:three}
\index{plane capacitor}\index{spherical capacitor}\index{cylindrical capacitor}
\begin{enumerate}
  \item Plane: plates of area $S$ a distance $d \ll \sqrt S$ apart:
        $C = \dfrac{\varepsilon_0S}{d}$.
  \item Spherical: concentric spheres of radii $R_1 < R_2$:
        $C = \dfrac{4\pi\varepsilon_0R_1R_2}{R_2 - R_1}$.
  \item Cylindrical (coaxial), radii $a < b$, length $L \gg b$:
        $C = \dfrac{2\pi\varepsilon_0L}{\ln(b/a)}$.
\end{enumerate}
Filling the gap with an insulator of relative permittivity
$\varepsilon_r$ multiplies $C$ by $\varepsilon_r$ (the dielectric's molecules
polarize and weaken the field, $\varepsilon_r = 2$ to $10$ for common
plastics and glass, $80$ for water).
\end{proposition}

\begin{proof}
In each case the field between the plates is that of
\cref{prop:b1:electrostatics-gauss:three}: $E = \sigma/\varepsilon_0 = Q/\varepsilon_0S$,
$U = Ed$; $E = Q/4\pi\varepsilon_0r^2$, $U = \int_{R_1}^{R_2}E\,\dd r = \dfrac{Q}{4\pi\varepsilon_0}
\Bigl(\dfrac1{R_1} - \dfrac1{R_2}\Bigr)$; $E = Q/2\pi\varepsilon_0Lr$, $U =
\dfrac{Q}{2\pi\varepsilon_0L}\ln\dfrac ba$. Divide $Q$ by $U$. The dielectric
factor is admitted.
\end{proof}

\begin{theorem}[Energy of a capacitor]\label{thm:b1:potential-capacitors:energy}
\index{energy of a capacitor}\index{energy density of the electric field}
A \omterm{def:b1:potential-capacitors:capacitance}{capacitor} charged to $Q$ under $U$ stores
\[
W = \tfrac12 CU^2 = \tfrac12 QU = \frac{Q^2}{2C} ,
\]
the work needed to carry the charge from one plate to the other
against the growing \omterm{def:b1:dc-circuits:voltage}{voltage}. In a \omterm{prop:b1:potential-capacitors:three}{plane capacitor} $W = \tfrac12\varepsilon_0E^2
\times Sd$: the energy is in the field, at a density $\tfrac12\varepsilon_0E^2$
per unit volume --- a statement that holds for every electrostatic
field.
\end{theorem}

\begin{proof}
Moving $\dd q$ from the negative to the positive plate when the charge
is $q$ costs $\dd W = (q/C)\,\dd q$; integrate from $0$ to $Q$. Then
substitute $C = \varepsilon_0S/d$ and $U = Ed$; the general density is admitted.
\end{proof}

\begin{example}[Three capacitors and their energies]\label{ex:b1:potential-capacitors:energies}
A square metre of foil pair \qty{1}{mm} apart: $C = 8.85 \times 10^{-12}/
10^{-3} = \qty{8.9}{nF}$ --- the \omterm{def:b1:potential-capacitors:capacitance}{farad} is a huge unit. A defibrillator's
\qty{150}{\micro F} at \qty{2}{kV}: \qty{300}{J}, delivered in milliseconds.
A supercapacitor of \qty{3000}{F} at \qty{2.7}{V}: \qty{11}{kJ} in a can
the size of a soda bottle --- its \omterm{def:b1:potential-capacitors:capacitance}{capacitance} comes from a molecular
gap $d$ and an enormous porous $S$. The energy density of a field at
the breakdown of air, $\tfrac12\varepsilon_0(3 \times 10^6)^2 = \qty{40}{J/m^3}$, is
why \omterm{def:b1:potential-capacitors:capacitance}{capacitors} store so little compared with batteries.
\end{example}

\begin{remark}[Capacitors in association]\label{rem:b1:potential-capacitors:association}
In parallel (same $U$) the charges add: $C = C_1 + C_2$; in series (same
$Q$) the \omterm{def:b1:dc-circuits:voltage}{voltages} add: $1/C = 1/C_1 + 1/C_2$ --- the opposite of
\omterm{def:b1:dc-circuits:resistor}{resistors}, because $C$ measures an ease (charge per volt) where $R$
measures a difficulty. These rules follow from $Q = CU$ and the
circuit laws of \cref{ch:b1:dc-circuits}.
\end{remark}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:1}
Potential at \qty{1.0}{m} from a \qty{1.0}{\micro C} charge; energy needed
to bring a second \qty{1.0}{\micro C} from infinity to that point; speed
with which it would fly off if released, if its mass is \qty{1}{g}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:2}
Plates \qty{1.0}{mm} apart under \qty{100}{V}: field; \omterm{thm:b1:work-and-energy:ke}{kinetic energy} and
speed gained by an electron crossing the gap from rest; time of
flight.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:3}
On a map of equipotentials drawn every \qty{10}{V}, the lines are
\qty{2}{mm} apart near $A$ and \qty{8}{mm} apart near $B$. Field at $A$
and at $B$; which way does the field point relative to the increasing
potentials; which way does an electron accelerate?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:potential-capacitors:4}
A Van de Graaff sphere of radius \qty{15}{cm} is charged to \qty{200}{kV}.
\omterm{def:b1:potential-capacitors:capacitance}{Capacitance}, charge, field at its surface. Maximum potential before
the air around it breaks down (\qty{3}{MV/m}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:5}
Potential on the axis of a ring of radius $R$ and charge $Q$; recover
the axial field by differentiation. Same for the uniformly charged disk
(use \cref{exo:b1:electrostatics-gauss:5}): $V(z) = \dfrac{\sigma}{2\varepsilon_0}
\bigl(\sqrt{z^2 + R^2} - z\bigr)$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:6}
Derive the potential inside a uniformly charged sphere from its field.
Potential at the centre of a gold nucleus ($Z = 79$, $R = \qty{7.0}{fm}$);
energy a proton would need to reach the centre from far away, in
\unit{MeV}; comment on Rutherford's \qty{5}{MeV} alpha particles.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:7}
Water molecule, $p = \qty{6.2e-30}{C.m}$. Field it creates at \qty{1.0}{nm}
along its axis and perpendicular to it; \omterm{def:b1:angular-momentum:moment}{torque} and energy difference
between aligned and anti-aligned positions in $E = \qty{1.0}{MV/m}$;
compare with $k_BT$ at \qty{300}{K}. Field of a sodium ion at
\qty{0.3}{nm} and the energy of the water dipole aligned in it (hydration).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:8}
A conducting sphere of radius \qty{1.0}{cm} in air: maximum charge and
potential before breakdown. Two conducting spheres, $R_1 = \qty{10}{cm}$
and $R_2 = \qty{1.0}{mm}$, joined by a thin wire and raised to
\qty{20}{kV}: surface field on each; which one discharges into the air?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:potential-capacitors:9}
\omterm{def:b1:potential-capacitors:capacitance}{Capacitances}: two \qty{1.0}{m^2} foils \qty{10}{\micro m} apart with a
polymer film $\varepsilon_r = 2.2$ (a film \omterm{def:b1:potential-capacitors:capacitance}{capacitor}); the same foils rolled
up --- does rolling change $C$? Energy at \qty{400}{V}. A \qty{150}{\micro F}
\omterm{def:b1:potential-capacitors:capacitance}{capacitor} at \qty{2.0}{kV} (defibrillator): charge, energy, mean power if
delivered in \qty{5}{ms}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:10}
\emph{Spherical and \omterm{prop:b1:potential-capacitors:three}{cylindrical capacitors}.} (a) Derive $C$ for
concentric spheres and check that for $R_2 - R_1 = d \ll R_1$ it reduces
to the plane formula, and for $R_2 \to \infty$ to the isolated sphere.
(b) Coaxial cable, $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, polyethylene
$\varepsilon_r = 2.3$: \omterm{def:b1:potential-capacitors:capacitance}{capacitance} per metre (compare with the \qty{100}{pF/m}
of standard cables). (c) Field at the inner conductor under
\qty{1}{kV}; safe?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:11}
\emph{Energy in the field.} (a) Energy of a charged conducting sphere
($Q$, $R$) from $Q^2/2C$. (b) Set it equal to the electron's rest energy
$m_ec^2 = \qty{0.511}{MeV}$ and solve for $R$: the ``classical electron
radius'' (within a factor). (c) An isolated \omterm{prop:b1:potential-capacitors:three}{plane capacitor} charged to
$Q$ has its plates pulled from $d$ to $2d$: change of stored energy, and
the force between the plates deduced from it; check against
\cref{exo:b1:electrostatics-gauss:8}. (d) Same with the \omterm{def:b1:potential-capacitors:capacitance}{capacitor} kept
at fixed $U$ by a battery: energy change, and where the difference goes.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:potential-capacitors:12}
\emph{Coulomb energy of a nucleus.} (a) Build a uniformly charged
sphere ($Q$, $R$) shell by shell and show that its electrostatic energy
is $W = \dfrac{3Q^2}{20\pi\varepsilon_0R}$. (b) Uranium-238: $Z = 92$, $R =
\qty{7.4}{fm}$: $W$ in \unit{MeV}. (c) Split it into two equal fragments
($Z/2$, $A/2$, radius $R/2^{1/3}$) far apart: Coulomb energy released.
(d) The measured energy of fission is about \qty{200}{MeV}: what opposes
the Coulomb gain?
\end{exercise}

\section{Problem: Millikan's oil drop}

\begin{problem}[{Weekend problem --- weighing the electron's charge with
a drop of oil: Stokes, the plane capacitor, and a table of numbers that
refuse to be anything but multiples of one}]
\label{pb:b1:potential-capacitors:1}
Oil drops are sprayed between two horizontal plates $d = \qty{16}{mm}$
apart, \qty{20}{cm} in diameter, and watched through a \omterm{def:b1:optical-instruments:microscope}{microscope}.
Data: oil density $\rho = \qty{900}{kg/m^3}$, air density
\qty{1.2}{kg/m^3}, air viscosity $\eta = \qty{1.8e-5}{Pa.s}$, $g =
\qty{9.8}{m/s^2}$, $\varepsilon_0 = \qty{8.85e-12}{F/m}$. A small sphere of
radius $r$ moving at $v$ in air feels the Stokes drag $6\pi\eta rv$.

\medskip
\noindent\textbf{Part I --- The falling drop.}
\begin{enumerate}
  \item Forces on a drop falling with no \omterm{def:b1:dc-circuits:voltage}{voltage} applied; show that it
        reaches a \omterm{prop:b1:newton-dynamics:terminal}{terminal velocity} $v_1 = 2\rho gr^2/9\eta$ (neglect the
        air's buoyancy --- justify).
  \item A drop falls \qty{1.0}{mm} in \qty{10}{s}: its radius and mass.
  \item \omterm{def:b1:transient-regimes:firstorder}{Time constant} of the approach to $v_1$; distance covered
        meanwhile. Is the drop ever seen accelerating?
  \item Reynolds number $\rho_{\text{air}}v_1r/\eta$ of the motion: is
        \omterm{prop:b1:newton-dynamics:drag}{Stokes's law} legitimate?
  \item Why must the drops be so small? (Compare the weight with the
        electric force on one \omterm{def:b1:electrostatics-gauss:charge}{elementary charge} in a field of
        \qty{3e5}{V/m}.)
  \item The drop is seen to jitter slightly around its mean fall: what
        is this, and why does it limit the precision?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The field.}
A \omterm{def:b1:dc-circuits:voltage}{voltage} $U$ is applied, the upper plate positive.
\begin{enumerate}[resume]
  \item Field between the plates for $U = \qty{5.0}{kV}$; why is it
        uniform, and what are the equipotentials?
  \item \omterm{def:b1:potential-capacitors:capacitance}{Capacitance} of the plates, charge on them at \qty{5.0}{kV}, and
        stored energy.
  \item The drop of question~2 is held stationary for $U_0 = \qty{1080}{V}$:
        sign and value of its charge, in coulombs and in units of $e$.
  \item With $U = \qty{5.0}{kV}$ the same drop rises at $v_2$: show that
        $q = 6\pi\eta r(v_1 + v_2)d/U$ and compute $v_2$.
  \item Why is timing a rise and a fall better than hunting for the
        balance \omterm{def:b1:dc-circuits:voltage}{voltage}?
  \item \omterm{def:b1:work-and-energy:conservative}{Potential energy} of the drop's charge between the plates at
        \qty{5}{kV}, and the energy dissipated by drag during a rise of
        \qty{1}{cm}.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The table of charges.}
Exposed to a weak source of X-rays between measurements, the same drop
changes its charge; successive measurements give (in $10^{-19}$~C):
$4.80$, $8.01$, $6.42$, $3.19$, $9.63$.
\begin{enumerate}[resume]
  \item Show that all five are close to integer multiples of one
        quantity, and give the integers.
  \item Best estimate of the \omterm{def:b1:electrostatics-gauss:charge}{elementary charge} from the five values;
        \omterm{def:b1:units-dimensions:uncertainty}{standard uncertainty} of the mean (\cref{ch:b1:units-dimensions}).
  \item Why does the charge change by whole numbers of $e$ between
        measurements?
  \item Show that, in this method, $q \propto \eta^{3/2}$: by how much does
        a $1\%$ error on the viscosity shift $e$? (Millikan's original
        value was $0.6\%$ low, for this very reason.)
  \item Quarks carry $\pm e/3$ and $\pm 2e/3$: why has no oil drop ever
        shown a third of $e$?
  \item The Faraday constant, the charge of one \omterm{def:b1:kinetic-theory:scales}{mole} of electrons, is
        $F = \qty{96485}{C/mol}$: deduce Avogadro's number.
  \item Thomson had measured $e/m_e = \qty{1.76e11}{C/kg}$ for the
        electron: deduce its mass, and compare with a hydrogen atom
        (\qty{1.67e-27}{kg}).
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The apparatus, by the potential.}
\begin{enumerate}[resume]
  \item Sketch the equipotentials and \omterm{def:b1:electrostatics-gauss:field}{field lines} between the plates,
        including the fringing near the edges; justify neglecting it.
  \item A positively charged drop at height $z$ above the lower plate
        (at \qty{0}{V}): \omterm{def:b1:work-and-energy:conservative}{potential energy} as a function of $z$; total
        \omterm{def:b1:work-and-energy:conservative}{potential energy} including gravity; condition on $U$ for the
        stationary point.
  \item Compare the energy of the drop's charge (question~12) with the
        energy stored in the \omterm{def:b1:potential-capacitors:capacitance}{capacitor}: does the drop disturb the field?
  \item The \omterm{def:b1:dc-circuits:voltage}{voltage} is reversed (lower plate positive) with
        $U = \qty{5.0}{kV}$: what does the drop of question~9 do, and at
        what speed?
  \item A dust grain of \qty{1}{\micro g} carrying $10^4e$ in the same
        field: ratio of electric force to weight; why such grains cannot
        be levitated and why Millikan chose oil mist.
  \item Sum up: what was measured (three times), what was deduced,
        with which uncertainty, and what two constants followed from it.
\end{enumerate}
\end{problem}
