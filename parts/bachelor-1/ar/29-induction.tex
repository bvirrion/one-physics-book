\chapter{التحريض الكهرمغناطيسي وتطبيقاته}\label{ch:b1:induction}

حرّك مغناطيسًا نحو وشيعة يجرِ فيها تيار؛ وأوقف
المغناطيس يتوقف التيار. وقد وجد فاراداي ذلك سنة 1831، بعد عشر
سنوات من البحث، وهو الواقعة التي يقوم عليها العالم الكهربائي:
فكل محطة توليد، وكل محوّل، وكل محرك
\omterm{prop:b1:induction:loudspeaker}{ومكبر صوت} وميكروفون، وكل موقد تحريض وشاحن
لاسلكي، يحوّل الطاقة بالقانون القائل إن تدفقًا مغناطيسيًّا
\emph{متغيّرًا} يولّد \omterm{def:b1:dc-circuits:sources}{قوة محركة كهربائية}. وينصّ هذا الفصل القانون مع
إشاراته، ويعطيه الوجهين اللذين يلبسهما (مجال متغيّر في دارة
ثابتة، ودارة تتحرك في مجال ثابت)، ويعرّف \omterm{def:b1:induction:inductance}{التحريض الذاتي} والمتبادل،
ثم يمرّ على الآلات الكهرميكانيكية ---
القضيب المنزلق، \omterm{prop:b1:induction:alternator}{والمولّد المتناوب}، والمحوّل، \omterm{prop:b1:induction:loudspeaker}{ومكبر الصوت} ---
حيث يحوّل القانون نفسه الاستطاعة الميكانيكية إلى كهربائية
والعكس، ولا شيء منه مجاني.

\section{قانون فاراداي}

\begin{definition}[التدفق المغناطيسي عبر دارة]\label{def:b1:induction:flux}
\index{تدفق مغناطيسي}\index{ويبر}
وجّه دارة مغلقة $\Gamma$ (باختيار جهة سير)؛ فيأخذ سطحها
الناظم $\vect n$ الذي تعطيه \omterm{prop:b1:magnetostatics:maps}{قاعدة اليد اليمنى}. و\emph{التدفق
المغناطيسي} عبر الدارة هو $\Phi = \sum\vect B\cdot\vect n\,\dd S$ على
أيّ سطح يحدّه $\Gamma$ (بالويبر، $\qty{1}{Wb} = \qty{1}{T.m^2}$) ---
و``أيّ'' لأن تدفق $\vect B$ عبر سطح مغلق معدوم.
وفي مجال منتظم ووشيعة مستوية من $N$ لفة ومساحتها $S$ يكون $\Phi =
NBS\cos\alpha$، حيث $\alpha$ الزاوية بين $\vect B$ و $\vect n$.
\end{definition}

\begin{theorem}[قانون فاراداي للتحريض]\label{thm:b1:induction:faraday}
\index{قانون فاراداي}\index{قوة محركة كهربائية (محرَّضة)}\index{قانون لنز}
كلما تغيّر التدفق عبر دارة كانت الدارة مقرًّا
\emph{\omterm{def:b1:dc-circuits:sources}{لقوة محركة كهربائية} محرَّضة}
\[
e = -\frac{\dd\Phi}{\dd t} ,
\]
محسوبةً في جهة توجيه الدارة، أي بوصفها \omterm{def:b1:dc-circuits:sources}{منبع توتر}
مثاليًّا قوته المحركة $e$ في تلك الجهة؛ وفي دارة مقاومتها $R$
تولّد التيار $i = e/R$. والإشارة السالبة هي \emph{\omterm{thm:b1:induction:faraday}{قانون لنز}}:
فالتيار المحرَّض يجري بحيث يعارض تغيّر التدفق الذي
ينتجه --- فمجاله هو، والقوى التي يحسّ بها، يقاومان السبب.
\end{theorem}
\admitted

\begin{remark}[وجهان لقانون واحد]\label{rem:b1:induction:twofaces}
\index{تحريض نويمان}\index{تحريض لورنتس}
يتغير التدفق إما لأن $\vect B$ يتغير مع الزمن عبر دارة
ثابتة (وهو \emph{\omterm{rem:b1:induction:twofaces}{تحريض نويمان}}: المحوّل وموقد التحريض)
وإما لأن الدارة تتحرك أو تتشوّه في مجال ثابت
(وهو \emph{\omterm{rem:b1:induction:twofaces}{تحريض لورنتس}}: المولّد \omterm{prop:b1:induction:loudspeaker}{ومكبر الصوت}). وفي الحالة الثانية
يمكن حساب القوة المحركة مباشرةً أيضًا: فحاملات عنصر ناقل
$\dd\vect l$ يتحرك بالسرعة $\vect v$ تحسّ بالقوة $q\vect v\wedge
\vect B$، وهي تكافئ مجالًا كهربائيًّا $\vect v\wedge\vect B$، فيكون
$e = \oint(\vect v\wedge\vect B)\cdot\dd\vect l$ --- وهذا يتفق دائمًا
مع $-\dd\Phi/\dd t$. ويغطي \omterm{thm:b1:induction:faraday}{قانون فاراداي} الحالتين؛ ومجلد السنة الثانية
يفسّر لماذا تعطي الآليتان صيغة واحدة.
\end{remark}

\begin{example}[رتب القدر]\label{ex:b1:induction:orders}
وشيعة من $100$ لفة ومساحتها \qty{10}{cm^2} في مجال الأرض، تُقلَب في
\qty{0.1}{s}: $\Delta\Phi = 2 \times 100 \times 10^{-3} \times 5 \times 10^{-5} = \qty{1e-5}{Wb}$
و $e \sim \qty{0.1}{mV}$ --- وهو قابل للقياس بغلفانومتر حسّاس، وبه
رُسمت خريطة مجال الأرض يومًا. والوشيعة نفسها في فجوة \qty{1}{T}
لمغناطيس، تُنزَع في \qty{0.1}{s}: \qty{1}{V}. ووشيعة محوّل من
$1000$ لفة مع \qty{1}{T} عند \qty{50}{Hz} على \qty{10}{cm^2}: $e_{\max} =
1000 \times 10^{-3} \times 314 = \qty{314}{V}$ --- أي شبكة التغذية.
\end{example}

\section{التحريض الذاتي والمتبادل}

\begin{definition}[التحريض]\label{def:b1:induction:inductance}
\index{تحريض ذاتي}\index{هنري}\index{تحريض متبادل}
تدفق مجال دارة الخاص عبر نفسها يتناسب مع
تيارها: $\Phi_{\text{الذاتي}} = Li$، حيث $L > 0$ هو \emph{تحريضها الذاتي}
(بالهنري، \unit{H}). والتدفق الذي ترسله الدارة~1 عبر الدارة~2 هو
$\Phi_{1 \to 2} = Mi_1$، وبالتناظر $\Phi_{2 \to 1} = Mi_2$، حيث $M$
\emph{تحريضهما المتبادل} (وتعيّن الجهاتُ إشارته). ويكون
التدفق الكلي للدارة~1 عندئذ $\Phi_1 = L_1i_1 + Mi_2$.
\end{definition}

\begin{proposition}[الوشيعة؛ الملف اللولبي]\label{prop:b1:induction:inductor}
\index{وشيعة}\index{طاقة مغناطيسية}\index{كثافة طاقة المجال المغناطيسي}
دارة تحريضها $L$ تحمل تيارًا متغيّرًا تكون مقرًّا
للقوة المحركة $e = -L\,\dd i/\dd t$: وفي \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح المستقبِل} يكون ذلك
التوتر $u = L\,\dd i/\dd t$ للوشيعة المستعملة منذ
\cref{ch:b1:transient-regimes}. وهي تخزّن الطاقة $W = \tfrac12Li^2$.
وملف لولبي طويل فيه $n$ لفة في المتر وطوله $\ell$ ومقطعه $S$
له $L = \mu_0n^2S\ell$؛ وطاقته المخزونة $\tfrac12\mu_0n^2I^2S\ell =
\dfrac{B^2}{2\mu_0}\,S\ell$: \omterm{prop:b1:induction:inductor}{فالطاقة المغناطيسية} تعيش في المجال بكثافة
قدرها $B^2/2\mu_0$، تمامًا كما تعيش الكهربائية بالكثافة $\varepsilon_0E^2/2$.
\end{proposition}

\begin{proof}
فاراداي مع $\Phi = Li$ و $L$ ثابتة. وأما الطاقة: فالاستطاعة التي تتلقاها
الوشيعة هي $ui = Li\,\dd i/\dd t = \dd(\tfrac12Li^2)/\dd t$. وفي الملف اللولبي:
$B = \mu_0nI$ في الداخل، والتدفق $BS$ عبر كلٍّ من اللفات $n\ell$، ومنه $\Phi =
\mu_0n^2S\ell I$. أما الكثافة العامة فمسلَّم بها.
\end{proof}

\begin{example}[مغناطيس رنين مغناطيسي]\label{ex:b1:induction:mri}
\qty{1.5}{T} على نحو متر مكعب: $B^2/2\mu_0 = 2.25/2.5 \times 10^{-6} =
\qty{0.9}{MJ/m^3}$ --- أي طاقة بطارية سيارة، مخزونة في فضاء خالٍ.
ومع تيار وشيعة قدره \qty{500}{A} يكون $L = 2W/I^2 = \qty{7}{H}$. وإذا
صار فائق الناقلية مقاومًا فجأةً (وهو \emph{الإخماد})، صار ذلك الميغاجول
حرارةً في ثوانٍ وأغلى مئات اللترات من الهيليوم
السائل: ولغرفة المغناطيس منفس لذلك.
\end{example}

\begin{proposition}[المحوّل المثالي]\label{prop:b1:induction:transformer}
\index{محوّل}
وشيعتان من $N_1$ و $N_2$ لفة ملفوفتان على قلب حديدي مغلق واحد،
بحيث يخترق التدفق $\varphi$ نفسه كل لفة، تحققان
\[
\frac{u_2}{u_1} = \frac{N_2}{N_1} , \qquad N_1i_1 + N_2i_2 = 0 \quad\text{(بلا ضياعات ولا تيار مغنطة)},
\]
ومنه $u_2i_2 = -u_1i_1$: فالاستطاعة المأخوذة عند الأولي تُقدَّم
عند الثانوي؛ وتتحول التوترات بنسبة اللفات، والتيارات
عكسًا. ولا يعمل المحوّل إلا بتيارات متغيّرة.
\end{proposition}

\begin{proof}
$u_1 = N_1\,\dd\varphi/\dd t$ و $u_2 = N_2\,\dd\varphi/\dd t$ (\omterm{thm:b1:dc-circuits:kirchhoff}{باصطلاح المستقبِل}
في كليهما، والجهتان مختارتان على امتداد التدفق المشترك): فاقسم.
وعلاقة التيارين هي مبرهنة أمبير على القلب: إذ كان يجب على تكامله الخطي
مضروبًا بالمقدار $\mu_0\mu_r$ في مجموع الأمبير-لفات أن يُحدث
التدفق؛ وفي قلب مثالي ($\mu_r \to \infty$) يقتضي ذلك أمبير-لفات
متلاشية، ومنه $N_1i_1 + N_2i_2 = 0$. ومسلَّم به بهذه الصيغة.
\end{proof}

\section{النواقل المتحركة: المولّدات والمحركات}

\begin{proposition}[القضيب المنزلق]\label{prop:b1:induction:rails}
\index{سكتا لابلاس}\index{تحويل كهرميكانيكي}
قضيب طوله $\ell$ ينزلق بالسرعة $v$ على سكتين تغلقهما
مقاومة $R$، في مجال منتظم $B$ ناظمي على المستوي. فيكون
مقرًّا للقوة المحركة $e = B\ell v$، ويولّد $i = B\ell v/R$، ويحسّ
\omterm{thm:b1:magnetostatics:laplace}{بقوة لابلاس} $F = B\ell i = B^2\ell^2v/R$ التي \emph{تعارض} حركته.
والاستطاعة الميكانيكية المبذولة لدفعه، $Fv$، تساوي \omterm{thm:b1:dc-circuits:kirchhoff}{الاستطاعة
الكهربائية} $ei = Ri^2$ المبددة: فالقضيب مولّد. أما إذا غُذّي من
منبع $E$ فهو محرك: إذ يدفعه التيار $(E - B\ell v)/R$
حتى توازن القوة المحركة المضادة $B\ell v$ المقدارَ $E$، عند سرعة اللاحمل $E/B\ell$.
\end{proposition}

\begin{proof}
التدفق $B\ell x$، ومنه $e = -B\ell\dot x$ في التوجيه الذي يكون فيه $\vect n$
على امتداد $\vect B$؛ \omterm{thm:b1:magnetostatics:laplace}{وقوة لابلاس} $i\vect\ell\wedge\vect B$ على التيار
الموجَّه هكذا تعاكس $\vect v$ (\cref{thm:b1:induction:faraday}،
لنز). واضرب المعادلة الكهربائية $e = Ri$ في $i$ والمعادلة
الميكانيكية في $v$: $Fv = B\ell iv = ei$. والاستطاعة
الكهرميكانيكية $ei$ واستطاعة لابلاس $-Fv$ متعاكستان دائمًا --- وهو قانون
التحويل.
\end{proof}

\begin{proposition}[الحلقة الدوّارة: المولّد المتناوب]\label{prop:b1:induction:alternator}
\index{مولّد متناوب}
وشيعة من $N$ لفة ومساحتها $S$ تدور بالسرعة $\omega$ حول محور
عمودي على مجال منتظم $B$ لها $\Phi = NBS\cos\omega t$ و
\[
e = NBS\omega\sin\omega t :
\]
أي قوة محركة جيبية سعتها $NBS\omega$ وتواترها تواتر الدوران.
وفي مقاومة $R$ تقدّم متوسط استطاعة $(NBS\omega)^2/2R$،
يقدّمه العزم الذي يبقيها تدور في مواجهة مزدوجة
لابلاس $-\vect m\wedge\vect B$ لتيارها هي.
\end{proposition}

\begin{proof}
اشتقّ التدفق؛ ولدينا $\langle\sin^2\rangle = \tfrac12$. والمزدوجة هي
\cref{thm:b1:magnetostatics:dipoleinfield} مع $m = Ni S$؛ ومتوسط
استطاعتها $\langle\Gamma\omega\rangle$ يساوي $\langle ei\rangle$ بالحساب
نفسه الذي جرى في القضيب.
\end{proof}

\section{مكبر الصوت: جملة مقترنة}

\begin{proposition}[الاقتران الكهرميكانيكي لوشيعة في مجال قطري]\label{prop:b1:induction:loudspeaker}
\index{مكبر صوت}\index{ميكروفون}
وشيعة طول سلكها الكلي $\ell$ تقع في مجال قطري $B$ بحيث يكون
كل عنصر من السلك عموديًّا على $\vect B$؛ وهي تتحرك على امتداد محورها
بالسرعة $v$ وتحمل $i$. فتحسّ عندئذ \omterm{thm:b1:magnetostatics:laplace}{بقوة لابلاس} المحورية
$F = B\ell i$ وتكون مقرًّا للقوة المحركة المضادة $e = -B\ell v$ (وفي \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح
المستقبِل}: توتر $B\ell v$). ومع كتلة $m$ وتعليق
صلابته $k$ وتخميد ميكانيكي $h$، وتغذية بتوتر $u$ عبر
مقاومة الوشيعة $R$ وتحريضها $L$:
\[
u = Ri + L\frac{\dd i}{\dd t} + B\ell v , \qquad
m\frac{\dd v}{\dd t} = -kx - hv + B\ell i .
\]
وبالضرب في $i$ وفي $v$ والجمع: تساوي \omterm{thm:b1:dc-circuits:kirchhoff}{الاستطاعة الكهربائية} $ui$
ضياعَ جول $Ri^2$ زائد نموّ الطاقات المخزونة
$\tfrac12Li^2 + \tfrac12mv^2 + \tfrac12kx^2$ زائد الضياع الميكانيكي
$hv^2$ (وهو يشمل الصوت): فحدّ التحويل $B\ell iv$ يتلاشى
بين المعادلتين. وإذا أُديرت إلى الوراء (بفرض الحركة وقراءة $u$) كانت
الأداة نفسها \emph{ميكروفونًا} ديناميًّا.
\end{proposition}

\begin{proof}
\omterm{thm:b1:magnetostatics:laplace}{قوة لابلاس} على كل عنصر $i\,\dd\vect l\wedge\vect B$، وكلها محورية و
بالإشارة نفسها؛ والقوة المحركة من $(\vect v\wedge\vect B)\cdot\dd\vect l$ مجموعةً
على السلك. وموازنة الطاقة هي مجموع المعادلتين
مضروبتين كما ذُكر.
\end{proof}

\begin{remark}[لماذا يخمّد الاقتران]\label{rem:b1:induction:damping}
عند التواتر المنخفض ($L\omega \ll R$) يكون $i \approx (u - B\ell v)/R$ وتكون القوة
$B\ell u/R - (B^2\ell^2/R)\,v$: فتضيف الدارة الكهربائية التخميد
$B^2\ell^2/R$ إلى الميكانيكي. \omterm{prop:b1:induction:loudspeaker}{ومكبر صوت} موصول
بمضخّم (مقاومة خرجه منخفضة) شديد التخميد --- فيتوقف مخروطه
حين تتوقف الإشارة --- والاقتران نفسه هو ما يتيح
للميكروفون أن يأخذ طاقة من الصوت (\cref{pb:b1:induction:1}).
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:induction:1}
وشيعة من $100$ لفة ومساحتها \qty{10}{cm^2} في مجال منتظم \qty{0.20}{T} بزاوية
\qty{30}{\degree} مع ناظمها. أوجد التدفق؛ ومتوسط القوة المحركة إذا قُطع المجال
في \qty{10}{ms}؛ ثم في \qty{1}{ms}؛ وجهة التيار المحرَّض
بالنسبة إلى المجال (\omterm{thm:b1:induction:faraday}{بقانون لنز}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:2}
قضيب طوله \qty{20}{cm} ينزلق بالسرعة \qty{2.0}{m/s} على سكتين تغلقهما
\qty{0.10}{\ohm}، في \qty{0.50}{T}. أوجد القوة المحركة والتيار \omterm{thm:b1:magnetostatics:laplace}{وقوة لابلاس} والقوة
التي على الدافع تطبيقها والاستطاعة الميكانيكية \omterm{thm:b1:dc-circuits:kirchhoff}{والاستطاعة الكهربائية}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:3}
ملف لولبي فيه 1000 لفة في المتر، وطوله \qty{20}{cm}، ومقطعه \qty{4.0}{cm^2}:
أوجد التحريض؛ والطاقة عند \qty{5.0}{A}؛ \omterm{def:b1:transient-regimes:firstorder}{وثابت الزمن} مع مقاومة \qty{1.0}{\ohm}؛
والمجال في الداخل وكثافة الطاقة، للتحقق من $B^2/2\mu_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:induction:4}
محوّل من \qty{230}{V} إلى \qty{12}{V}: أوجد نسبة اللفات؛ وعدد لفات الأولي
من أجل ثانوي فيه $50$ لفة؛ وتيار الأولي حين يقدّم الثانوي
\qty{2.0}{A}. ولماذا لا يعمل على بطارية سيارة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:5}
\omterm{thm:b1:induction:faraday}{قانون لنز}: أعطِ جهة التيار المحرَّض وجهة
القوة في كل حالة --- (أ) قطب شمالي يقترب من حلقة على امتداد
محورها؛ (ب) حلقة في مجال متزايد؛ (ج) حلقة تُسقَط
على مغناطيس؛ (د) مغناطيس يسقط في أنبوب نحاسي طويل. وفي (د)،
فسّر بالطاقة لماذا يسقط بسرعة صغيرة ثابتة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:6}
\omterm{prop:b1:induction:alternator}{مولّد متناوب}: $N = 200$ و $S = \qty{20}{cm^2}$ و $B = \qty{0.10}{T}$ عند \qty{50}{Hz}.
أوجد القوة المحركة القمّية والفعّالة؛ والتيار القمّي والعزم القمّي في \qty{10}{\ohm}؛ ومتوسط
\omterm{thm:b1:dc-circuits:kirchhoff}{الاستطاعة الكهربائية}؛ وقارنه بمتوسط الاستطاعة الميكانيكية، وبالعزم
اللحظي في اللحظة التي تنعدم فيها القوة المحركة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:7}
\emph{الكابح المغناطيسي.} حلقة مربعة ضلعها $a = \qty{10}{cm}$ و
مقاومتها $R = \qty{1.0}{m\ohm}$ وكتلتها \qty{1.0}{kg}، تغادر منطقة
مجال منتظم $B = \qty{1.0}{T}$ بالسرعة $v$ (وضلع واحد ما زال في
المجال). أوجد القوة المحركة والتيار وقوة الكبح؛ ومعادلة الحركة \omterm{def:b1:transient-regimes:firstorder}{وثابت
الزمن}؛ والمسافة المقطوعة قبل التوقف انطلاقًا من \qty{1.0}{m/s}؛ وأين
ذهبت \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:8}
وشيعة صغيرة ($N_2 = 50$ لفة، ومقطعها \qty{2.0}{cm^2}) تقع داخل
ملف لولبي طويل ($n_1 = \qty{2000}{m^{-1}}$)، ومحوراهما متطابقان. أوجد \omterm{def:b1:induction:inductance}{التحريض

على أيّ الوشيعتين تحمل التيار؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:induction:9}
الطاقة في مجال: $B = \qty{1.5}{T}$ على \qty{1.0}{m^3}، كما في مغناطيس
رنين مغناطيسي. أوجد الطاقة؛ والتحريض من أجل $I = \qty{500}{A}$؛ والارتفاع الذي
كانت تلك الطاقة لترفع إليه سيارة كتلتها \qty{1}{t}؛ وحجم الهيليوم السائل الذي تغليه
في إخماد (\qty{2.6}{kJ} للتر). ثم طاقة مجال الأرض
(\qty{5e-5}{T}) في غرفة حجمها \qty{50}{m^3}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:10}
\emph{السكتان محركًا.} قضيب \cref{exo:b1:induction:2} (كتلته $m$،
وبلا احتكاك) موصول بمنبع $E$ عبر المقاومة $R$.
(أ) اكتب المعادلتين الكهربائية والميكانيكية؛ وبيّن أن $m\,\dd v/\dd t = B\ell(E -
B\ell v)/R$. (ب) وأوجد \omterm{prop:b1:newton-dynamics:terminal}{السرعة الحدية} \omterm{def:b1:transient-regimes:firstorder}{وثابت الزمن}. (ج) وموازنة الاستطاعة:
استطاعة المنبع $=$ جول $+$ الميكانيكية؛ \omterm{def:b1:heat-engines:efficiency}{والمردود} بدلالة
$v$. (د) والأعداد: $E = \qty{2.0}{V}$ و $m = \qty{50}{g}$، والباقي كما في
\cref{exo:b1:induction:2}. (ه) ومدفع كهرمغناطيسي يسرّع قذيفة كتلتها \qty{10}{g}
بتيار \qty{1}{MA} على امتداد \qty{5}{m} في \qty{1}{T}: أوجد سرعة الخروج
(\omterm{def:b1:units-dimensions:oom}{برتبة القدر}).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:11}
\emph{قرص فاراداي.} قرص نحاسي نصف قطره $R$ يدور بالسرعة $\omega$
حول محوره في مجال منتظم $B$ على امتداد المحور؛ وتلمس فرشاتان المركز
والحافة. (أ) أوجد القوة المحركة بين المركز والحافة بجمع $(\vect v\wedge
\vect B)\cdot\dd\vect l$ على امتداد نصف قطر: $e = \tfrac12B\omega R^2$. (ب) والأعداد:
$R = \qty{10}{cm}$ و\qty{3000}{rpm} و\qty{1.0}{T}. (ج) ولماذا هي آلة
منخفضة التوتر عالية التيار؛ وأوجد التيار في \qty{1}{m\ohm}، و
عزم الكبح. (د) واستعد $e$ من \omterm{thm:b1:induction:faraday}{قانون فاراداي} مطبَّقًا على
الدارة فرشاة-نصف قطر-حافة-فرشاة: أيّ تدفق يتغير؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:induction:12}
\emph{موقد التحريض.} تحمل الوشيعة تحت الزجاج $i_1 =
I_1\cos\omega t$ مع $I_1 = \qty{30}{A}$ عند \qty{25}{kHz}؛ ويسلك قاع الإناء
سلوك لفة واحدة مقاومتها $R_p = \qty{10}{m\ohm}$ مقترنة \omterm{def:b1:induction:inductance}{بتحريض متبادل}
$M = \qty{1.0}{\micro H}$ (مع إهمال تحريض لفة الإناء). (أ) أوجد القوة المحركة
المحرَّضة في الإناء، بقيمتها القمّية. (ب) والتيار ومتوسط الاستطاعة المبددة
في الإناء. (ج) ولماذا تواتر مرتفع، ولماذا تظهر الحرارة في
الإناء لا في الزجاج؟ (د) ويؤثر تيار الإناء في الوشيعة:
أوجد القوة المحركة التي يُحدثها فيها (قمّيةً)، والاستطاعة الزائدة التي على المولّد
تقديمها --- وتحقّق من أنها تساوي (ب).
\end{exercise}

\section{مسألة: دينامو الدراجة ومكبر الصوت}

\begin{problem}[{مسألة نهاية الأسبوع --- محوّلان كهرميكانيكيان
مفكَّكان: دينامو ينظّم نفسه، ومكبر صوت مضخّمه
كابحه أيضًا}]
\label{pb:b1:induction:1}

\medskip
\noindent\textbf{الجزء I --- دينامو الدراجة.}
وشيعة من $N = 300$ لفة ومساحتها $S = \qty{3.0}{cm^2}$ تدور في
مجال منتظم $B = \qty{0.30}{T}$؛ ويحمل محورها بكرة نصف قطرها
$r = \qty{1.0}{cm}$ مضغوطة على الإطار، فتدور الوشيعة بالسرعة
$\omega = v/r$ من أجل سرعة دراجة $v$. ومقاومة الوشيعة $r_c = \qty{4.0}{\ohm}$،
وتحريضها $L = \qty{20}{mH}$؛ والمصباح مقاومة $R = \qty{12}{\ohm}$.
\begin{enumerate}
  \item التدفق عبر الوشيعة والقوة المحركة المحرَّضة بدلالة الزمن؛
        والسعة والتواتر عند $v = \qty{18}{km/h}$.
  \item وبإهمال $L$ أولًا: أوجد التيار الفعّال والاستطاعة في المصباح عند
        \qty{18}{km/h}؛ وهل يكون مصباح ``\qty{6}{V} و\qty{3}{W}'' مرتاحًا؟
  \item ومتوسط الاستطاعة الميكانيكية التي يقدّمها الدرّاج للدينامو، و
        القوة الموافقة على الإطار (وقارنها باحتكاك التدحرج،
        وهو نحو \qty{2}{N}).
  \item فسّر \omterm{thm:b1:induction:faraday}{بقانون لنز} لماذا يقاوم الدينامو، ولماذا لا يناقض مصباح
        يضيء ``مجانًا'' انحفاظَ الطاقة.
  \item وأدخِل $L$ الآن: أوجد \omterm{def:b1:sinusoidal-impedance:impedance}{الممانعة} العقدية للدارة؛ وبيّن أن
        سعة التيار هي $I = NBS\omega/\sqrt{(R + r_c)^2 + L^2\omega^2}$
        وأنها تتشبع عند السرعات العالية؛ وأوجد نهايتها، و
        السرعة التي تبلغ عندها $70\%$ من تلك النهاية.
  \item وبلا التحريض، ما الذي كان المصباح ليتلقاه عند
        \qty{54}{km/h} في نزول؟ ومع التحريض؟
  \item والتيار واستطاعة المصباح عند \qty{9}{km/h} وعند \qty{36}{km/h}؛ وعلّق
        على التنظيم الذاتي.
  \item والديناموات الحقيقية تُدير مغناطيسًا داخل وشيعة ثابتة. فهل يتغير
        التحليل؟ وأيّ مشكلة عملية يزيلها ذلك؟
  \item وماذا يفعل المصباح حين تتوقف الدراجة عند إشارة حمراء،
        وكيف تعالج المصابيح الحديثة ذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- \omterm{prop:b1:induction:loudspeaker}{مكبر الصوت}.}
وشيعة الصوت: طول السلك $\ell = \qty{5.0}{m}$ في مجال قطري $B =
\qty{1.0}{T}$، ومقاومتها $R = \qty{8.0}{\ohm}$، وتحريضها $L = \qty{0.50}{mH}$؛
والكتلة المتحركة (الوشيعة والمخروط) $m = \qty{10}{g}$؛ وصلابة التعليق
$k = \qty{1000}{N/m}$، والتخميد الميكانيكي $h = \qty{1.0}{N.s/m}$. والنظام
جيبي، بالسعات العقدية.
\begin{enumerate}[resume]
  \item القوة على الوشيعة من أجل تيار $i$؛ والقوة المحركة المضادة من أجل سرعة $v$
        (اشتقّهما مع الإشارتين).
  \item اكتب المعادلتين الكهربائية والميكانيكية.
  \item احذف $i$ وبيّن أن
        $\underline v = \dfrac{B\ell\,\underline u}{(R + jL\omega)(jm\omega + h + k/j\omega) + B^2\ell^2}$.
  \item وعند التواترات التي يكون فيها $L\omega \ll R$، بيّن أن الاقتران يعمل
        عمل تخميد ميكانيكي إضافي $B^2\ell^2/R$؛ وأوجد قيمته؛ وتواتر
        رنين المخروط وعامل جودته مع التخميد الكهربائي
        ومن دونه.
  \item ويقدّم المضخّم تيارًا $i$ سعته \qty{1.0}{A} عند \qty{1}{kHz}:
        أوجد سعات القوة والتسارع والانزياح والسرعة
        للمخروط (في النظام الذي تحكمه الكتلة)؛ وسعة القوة المحركة المضادة مقارنةً
        بالمقدار $Ri$.
  \item والتيار نفسه عند تواتر الرنين: أوجد سعة السرعة
        (في النظام الذي يحكمه التخميد) والقوة المحركة المضادة؛ وعلّق.
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{والاستطاعة الكهربائية} عند \qty{1}{kHz}؛ والاستطاعة الميكانيكية المبددة في
        $h$ (وتُؤخذ ضياعات الصوت والتعليق)؛ \omterm{def:b1:heat-engines:efficiency}{والمردود}.
  \item ولماذا يكون كون ضغط الصوت متناسبًا مع تسارع
        المخروط (سلّم به) هو سبب أن مخروطًا تحكمه الكتلة يعطي
        استجابة مسطّحة فوق الرنين؟
  \item ولماذا يجب أن يكون المجال قطريًّا، ولماذا يكون مكبر الحادّة صغيرًا؟
  \item ارسم طويلة \omterm{def:b1:sinusoidal-impedance:impedance}{الممانعة} $\underline u/\underline i$ التي يراها
        المضخّم من \qty{10}{Hz} إلى \qty{10}{kHz}: قيمتها عند التواتر
        المنخفض، وعند الرنين (باستعمال السرعة التي يحكمها التخميد)،
        وعند \qty{10}{kHz}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- إلى الوراء، والحساب الختامي.}
\begin{enumerate}[resume]
  \item الأداة نفسها ميكروفونًا: يحرّك صوتٌ المخروطَ بالسرعة
        $v = \qty{1.0}{mm/s}$؛ أوجد توتر الدارة المفتوحة.
  \item ومحمَّلًا بمقاومة $R_L = \qty{600}{\ohm}$: أوجد التيار، وقوة الكبح
        على المخروط؛ وبيّن أن الميكروفون يأخذ استطاعة من الصوت
        عبر تخميد فعّال $B^2\ell^2/(R + R_L)$.
  \item وتُلفّ الوشيعة على أسطوانة من الألومنيوم (وهي الحامل)، وهي
        لفة مغلقة طولها \qty{0.10}{m} ومقاومتها \qty{1.0}{m\ohm}:
        أوجد التخميد الذي تضيفه؛ ولماذا تُشقّ الحوامل.
  \item انقر مخروط \omterm{prop:b1:induction:loudspeaker}{مكبر صوت} طرفاه مفتوحان، ثم مخروط
        مكبر طرفاه مقصوران: أيّهما يرنّ أطول،
        ولماذا؟
  \item اكتب موازنة طاقة \omterm{prop:b1:induction:loudspeaker}{مكبر الصوت} على امتداد دور في النظام
        المستقر، مسمّيًا كل حدّ.
  \item لخّص: لكل أداة، القانون المستعمل والعددان
        الجديران بالحفظ.
\end{enumerate}
\end{problem}
