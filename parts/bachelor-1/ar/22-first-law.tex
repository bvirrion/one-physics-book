\section{التحولات والشغل والحرارة}

\begin{definition}[الجملة والتحول]\label{def:b1:first-law:transformation}
\index{جملة ترموديناميكية}\index{تحول}\index{تحول شبه سكوني}\index{تحول عكوس}
\emph{الجملة الترموديناميكية} هي المادة داخل حدّ مختار؛ وما
عداها هو \emph{الوسط الخارجي}. و\emph{التحول} ينقلها من حالة
توازن إلى أخرى. ويكون \emph{شبه سكوني} إذا مرّت الجملة
بمتتالية من حالات التوازن (أي ببطء يكفي لتعريف $P$ و $T$ في
كل موضع منها)، ويكون \emph{عكوسًا} إذا أدّى، فوق ذلك، عكسُ
الشروط الخارجية إلى عكس المسار (بلا احتكاك، وبلا فرق منتهٍ
في \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} أو الضغط مع الوسط الخارجي). ويُسمّى التحول
بما يبقى ثابتًا فيه: \emph{متساوي \omterm{def:b1:first-law:heat}{الحرارة}} ($T$)، \emph{متساوي
الضغط} ($P$)، \emph{متساوي الحجم} ($V$)؛ و\emph{كظوم} إن لم
تُتبادَل \omterm{def:b1:first-law:heat}{حرارة}؛ و\emph{وحيد الضغط} (\emph{وحيد \omterm{def:b1:first-law:heat}{الحرارة}}) إذا
ثبت الضغط الخارجي (أو \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} الخارجية) مهما يجرِ في
الداخل.
\end{definition}

\begin{proposition}[شغل قوى الضغط]\label{prop:b1:first-law:work}
\index{شغل!قوى الضغط}
جملة يتغير حجمها بمقدار $\dd V$ في مواجهة ضغط خارجي
$P_{\mathrm{ext}}$ تتلقى الشغل
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
وفي \omterm{def:b1:first-law:transformation}{تحول شبه سكوني} يكون $P_{\mathrm{ext}} = P$ (أي ضغط الجملة
نفسها) ويكون $W = -\int P\,\dd V$ مقابلًا للمساحة تحت المسار في
مخطط كلابيرون بإشارة سالبة: فهو موجب عند الانضغاط وسالب عند التمدد.
\end{proposition}

\begin{proof}
مكبس مساحته $S$ يدفعه الوسط الخارجي بالقوة
$P_{\mathrm{ext}}S$ فينزاح بمقدار $\dd x$ نحو الخارج: فيؤدي الوسط الخارجي
على الجملة الشغل $-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$. وأيّ
حدّ هو مجموعة من مكابس كهذه. وفي \omterm{def:b1:first-law:transformation}{التحول شبه السكوني} يكون المكبس في
توازن، فيكون $P_{\mathrm{ext}} = P$ إلى فرق متلاشٍ.
\end{proof}

\begin{definition}[الحرارة]\label{def:b1:first-law:heat}
\index{حرارة}\index{منظم حراري}
\emph{الحرارة} $Q$ طاقة تُنقل إلى الجملة بغير الشغل
العياني --- بتصادمات جزيئية عند جدار (وهو التوصيل)، أو
بجريان مائع (وهو الحمل)، أو بالإشعاع. و\emph{المنظِّم الحراري}
جسم من الكبر بحيث لا تتغير درجة حرارته
مهما تبادل من حرارة. واصطلاح الإشارة في الكتاب كله: أن $W$ و $Q$
\emph{تتلقاهما} الجملة، فهما موجبان حين يدخلان.
\end{definition}

\section{المبدأ الأول}

\begin{theorem}[المبدأ الأول للترموديناميك]\label{thm:b1:first-law:firstlaw}
\index{المبدأ الأول للترموديناميك}\index{طاقة داخلية}
\omterm{def:b1:newton-dynamics:conservation}{لجملة مغلقة} دالةُ حالة، هي \emph{طاقتها الداخلية}
$U$ (وهي امتدادية)، بحيث يكون في أيّ \omterm{def:b1:first-law:transformation}{تحول}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
حيث $E_k$ \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} العيانية (وهي معدومة لجملة ساكنة):
فالطاقة المتلقاة شغلًا وحرارةً تُخزَّن طاقةً داخلية. ويتوقف كلٌّ
من $W$ و $Q$ على المسار؛ ولا يتوقف مجموعهما عليه.
\end{theorem}
\admitted

\begin{remark}[ما الذي ينصّ عليه]\label{rem:b1:first-law:meaning}
هو انحفاظ الطاقة، مع الاعتراف \omterm{def:b1:first-law:heat}{بالحرارة} نقلًا للطاقة (تجربة
جول بالدولاب ذي الريش: فشغل مقيس يرفع دائمًا \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} الماء
بقدر ما ترفعها \omterm{def:b1:first-law:heat}{حرارة} مقيسة). وقولنا ``$U$ دالة حالة''
يعني أن $\Delta U$ تعيّنه الحالتان الابتدائية والنهائية ---
فيكون في دور $\Delta U = 0$ و $W + Q = 0$: فمحرك يقدّم شغلًا
لا بدّ أن يتلقى \omterm{def:b1:first-law:heat}{حرارة}. ومضمون $U$ المجهري هو مضمون
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[الإنثالبي؛ السعات الحرارية]\label{def:b1:first-law:enthalpy}
\index{إنثالبي}\index{سعة حرارية عند ثبات الضغط}
\emph{الإنثالبي} $H = U + PV$ دالة حالة ملائمة
للتحولات عند ثبات الضغط. و\emph{السعتان الحراريتان عند
ثبات الحجم وعند ثبات الضغط} هما
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
بوحدة \unit{J/K}؛ وللمول $C_{V,m}$ و $C_{P,m}$؛ وللكيلوغرام الحرارتان
النوعيتان $c_V$ و $c_P$.
\end{definition}

\begin{proposition}[الحرارة عند ثبات الحجم وعند ثبات الضغط]\label{prop:b1:first-law:qvqp}
من أجل جملة ساكنة لا تتبادل الشغل إلا بقوى الضغط:
\begin{itemize}
  \item \omterm{def:b1:first-law:transformation}{التحول} المتساوي الحجم: $W = 0$ و $Q_V = \Delta U$؛
  \item \omterm{def:b1:first-law:transformation}{التحول} الوحيد الضغط بين حالتين عند الضغط الخارجي
        $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
المتساوي الحجم: $\dd V = 0$. والوحيد الضغط: $W = -P_0(V_2 - V_1)$، ومنه $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ لأن $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[الغاز المثالي: $U$ و $H$ وعلاقة ماير]\label{prop:b1:first-law:gas}
\index{علاقة ماير}\index{الأس الكظوم}\index{قانونا جول}
من أجل \omterm{def:b1:kinetic-theory:model}{غاز مثالي}، تتوقف $U$ و $H$ على $T$ وحدها (وهما \omterm{prop:b1:first-law:gas}{قانونا جول}):
$\dd U = C_V\,\dd T$ و $\dd H = C_P\,\dd T$، ويكون
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
حيث $\gamma = 5/3$ للغازات أحادية الذرة، و $7/5$ لثنائية الذرة عند
درجات \omterm{def:b1:first-law:heat}{الحرارة} العادية. أما الطور المكثَّف في نموذج
\cref{prop:b1:kinetic-theory:condensed}، فيكون فيه $C_P \approx C_V = C$ و
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
لدينا $H = U + nRT$؛ ونشتقّ بالنسبة إلى $T$. وتكون $C_{V,m} = \tfrac32 R$ أو
$\tfrac52 R$ حسب \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{تحولات الغاز المثالي}

\begin{proposition}[التحولات النموذجية الأربعة]\label{prop:b1:first-law:transformations}
\index{تحول متساوي الحرارة}\index{تحول كظوم}\index{قانون لابلاس}
من أجل $n$ مولًا من \omterm{def:b1:kinetic-theory:model}{غاز مثالي}:
\begin{itemize}
  \item متساوي الحجم: $W = 0$ و $Q = \Delta U = nC_{V,m}\Delta T$؛
  \item متساوي الضغط (شبه سكوني): $W = -P\Delta V = -nR\Delta T$ و $Q = \Delta H =
        nC_{P,m}\Delta T$؛
  \item متساوي \omterm{def:b1:first-law:heat}{الحرارة} عكوس: $\Delta U = 0$ و $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$؛
  \item كظوم عكوس: $Q = 0$ و $W = \Delta U = nC_{V,m}(T_2 - T_1)$،
        وعلى امتداد المسار (وهو \emph{\omterm{prop:b1:first-law:transformations}{قانون لابلاس}})
        \[
        PV^\gamma = \text{ثابت}, \qquad TV^{\gamma - 1} = \text{ثابت}, \qquad
        T^\gamma P^{1 - \gamma} = \text{ثابت} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
المتساوي الحجم والمتساوي الضغط: من القضيتين السابقتين. والمتساوي \omterm{def:b1:first-law:heat}{الحرارة}: $W =
-\int nRT\,\dd V/V$. والكظوم: $\dd U = \delta W$، أي $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$، ومنه $\dd T/T = -(\gamma - 1)\dd V/V$ (باستعمال
$R/C_{V,m} = \gamma - 1$)؛ ونكامل فنجد $TV^{\gamma - 1}$ ثابتًا؛ وتأتي الصيغتان
الأخريان من $PV = nRT$.
\end{proof}

\begin{example}[المنفاخ والديزل]\label{ex:b1:first-law:diesel}
هواء ($\gamma = 1.4$) يُضغط كظومًا من \qty{1}{bar} و\qty{300}{K}
إلى \qty{7}{bar} (وهو منفاخ دراجة): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- أي \qty{250}{\degreeCelsius} للحظة، وهو ما يحسّه جسم المنفاخ. وفي
محرك ديزل نسبة انضغاطه $20$: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$
و $P_2 = 20^{1.4} = \qty{66}{bar}$ --- أي فوق درجة الاشتعال الذاتي
للوقود: فلا يحتاج المحرك إلى شرارة (ومسألة نهاية الأسبوع تجري
الدور كله).
\end{example}

\begin{remark}[التحولات اللاعكوسة]\label{rem:b1:first-law:irreversible}
\index{تمدد جول}\index{تحول لاعكوس}
حين لا يكون \omterm{def:b1:first-law:transformation}{التحول شبه سكوني} لا ينطبق \omterm{prop:b1:first-law:transformations}{قانون لابلاس} ولا $W =
-\int P\,\dd V$؛ ولا يبقى إلا $W = -\int P_{\mathrm{ext}}\,\dd V$ والمبدأ
الأول. وهاتان حالتان كلاسيكيتان: غاز يتمدد في الخلاء (وهو \omterm{rem:b1:first-law:irreversible}{تمدد
جول}) لا يتلقى شغلًا ولا، في جدران كظومة، حرارةً --- فيكون $\Delta U
= 0$، ويحفظ \omterm{def:b1:kinetic-theory:model}{الغاز المثالي} درجة حرارته؛ وغاز يُضغط فجأة
بضغط خارجي يقفز إلى $P_2$ يتلقى $W = -P_2\Delta V$، وهو أكبر
من الشغل العكوس إلى الضغط نفسه، فينتهي أسخن
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{المسعرية}

\begin{method}[الموازنة المسعرية]\label{met:b1:first-law:calorimetry}
\index{مسعرية}\index{حرارة كامنة}
أجسام توضع في تماسّ حراري داخل وعاء معزول عند ضغط ثابت
(هو الضغط الجوي) لا تتبادل \omterm{def:b1:first-law:heat}{الحرارة} إلا فيما بينها: فيكون المجموع
$\Delta H = \sum Q_P = 0$. ولكل جسم $\Delta H = mc\,\Delta T$ (إن لم يتغير
طوره) أو $\pm mL$ عند تغيّر حالة عند \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} تحوّله،
حيث $L$ هي \emph{\omterm{def:b1:first-law:heat}{الحرارة} الكامنة} (أي \omterm{def:b1:first-law:enthalpy}{إنثالبي} الانصهار أو التبخر
لواحدة الكتلة: ففي الماء $L_f = \qty{334}{kJ/kg}$ عند \qty{0}{\degreeCelsius}، و $L_v =
\qty{2260}{kJ/kg}$ عند \qty{100}{\degreeCelsius}). فاكتب الموازنة، وحُلّ
لإيجاد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} النهائية، وتحقّق من أن الطور المفترض لكل
جسم ينسجم معها.
\end{method}

\begin{example}[جليد في ماء]\label{ex:b1:first-law:ice}
\qty{50}{g} من الجليد عند \qty{0}{\degreeCelsius} في \qty{200}{g} من الماء عند
\qty{30}{\degreeCelsius}: يحتاج الانصهار إلى \qty{16.7}{kJ}؛ ويستطيع الماء أن
يعطي \qty{25.1}{kJ} وهو يبرد إلى \qty{0}{\degreeCelsius}، فينصهر الجليد كله
وتعطي الموازنة $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ كيلوجول
$T_f = \qty{8}{\degreeCelsius}$. ولو كان الجليد \qty{100}{g} للزم
\qty{33.4}{kJ}: فلا ينصهر إلا بعضه ويكون $T_f = \qty{0}{\degreeCelsius}$
--- ومن هنا أهمية التحقق.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
\omterm{def:b1:kinetic-theory:scales}{مول} واحد من الهواء ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) يُسخَّن من
\qty{300}{K} إلى \qty{400}{K} عند ثبات الضغط. أوجد \omterm{def:b1:first-law:heat}{الحرارة} المتلقاة
والشغل المتلقى وتغيّر \omterm{def:b1:kinetic-theory:U}{الطاقة الداخلية}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
\omterm{def:b1:kinetic-theory:scales}{مول} واحد من غاز عند \qty{300}{K} يُضغط عند ثبات \omterm{def:b1:first-law:heat}{الحرارة} وعلى
نحو عكوس من \qty{2.0}{L} إلى \qty{0.50}{L}. أوجد الشغل المتلقى
\omterm{def:b1:first-law:heat}{والحرارة} المتبادلة و $\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
هواء عند \qty{300}{K} و\qty{1}{bar} يُضغط كظومًا وعلى نحو عكوس
إلى عُشر حجمه. أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} والضغط النهائيين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
أعطِ $C_{V,m}$ و $C_{P,m}$ و $\gamma$ للأرغون وللآزوت. وفي الماء
السائل ($c = \qty{4180}{J/(kg.K)}$)، لماذا يُهمَل التمييز بين
$c_P$ و $c_V$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
\qty{200}{g} من الماء عند \qty{80}{\degreeCelsius} تُمزج مع \qty{300}{g}
عند \qty{20}{\degreeCelsius} في كأس معزول. أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} النهائية؛
وما الذي يتغير إذا كانت للكأس نفسه سعة حرارية \qty{80}{J/K} وكان
في البدء عند \qty{20}{\degreeCelsius}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
\qty{50}{g} من الجليد عند \qty{-10}{\degreeCelsius} ($c_{\text{الجليد}} = \qty{2100}{J/(kg.K)}$)
تُلقى في \qty{200}{g} من الماء عند \qty{30}{\degreeCelsius}. أوجد الحالة
\omterm{def:b1:kinetic-theory:temperature}{ودرجة الحرارة} النهائيتين. والسؤال نفسه مع \qty{150}{g} من الجليد.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
غاز يتبع دورًا مستطيلًا في مخطط كلابيرون: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ مع $V_2 > V_1$
و $P_2 > P_1$. أوجد الشغل المتلقى على امتداد الدور بدلالة المساحة؛
ثم إشارته ومعنى تلك الإشارة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
منفاخ دراجة يضغط الهواء من \qty{1}{bar} و\qty{300}{K} إلى
\qty{7}{bar} بسرعة تكفي لأن يكون \omterm{def:b1:first-law:transformation}{التحول} كظومًا. أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة}
المبلوغة؛ ولماذا يدفأ المنفاخ؛ ولماذا يهبط ضغط الإطار قليلًا بعد
انتهاء النفخ.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
\omterm{def:b1:kinetic-theory:model}{غاز مثالي} يتمدد في وعاء مفرَّغ ومعزول (وهو \omterm{rem:b1:first-law:irreversible}{تمدد
جول}). أوجد الشغل \omterm{def:b1:first-law:heat}{والحرارة} و $\Delta U$ و $\Delta T$. وهل \omterm{def:b1:first-law:transformation}{التحول
عكوس}؟ وماذا يفعل غاز حقيقي؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
غليان \qty{1.0}{kg} من الماء عند \qty{100}{\degreeCelsius} و\qty{1}{atm}
(حجم البخار \qty{1.67}{m^3}). أوجد \omterm{def:b1:first-law:heat}{الحرارة} المقدَّمة والشغل المتبادل مع
الجوّ و $\Delta U$. وأين يذهب معظم \omterm{def:b1:first-law:heat}{الحرارة} الكامنة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
\omterm{def:b1:kinetic-theory:scales}{مول} واحد من \omterm{prop:b1:kinetic-theory:Ugas}{غاز ثنائي الذرة} عند \qty{300}{K} و\qty{1}{bar}، في أسطوانة
معزولة، يُضغط برفع الضغط الخارجي فجأةً إلى
\qty{5}{bar} ثم انتظار التوازن. أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} والحجم النهائيين
(بالمبدأ الأول مع $W = -P_{\mathrm{ext}}\Delta V$)؛ وقارن بالانضغاط
الكظوم العكوس إلى \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
الصوت انضغاط سريع، ومن ثمّ كظوم: وسرعته $c = \sqrt{\gamma P/\rho}$
(وكانت القيمة المتساوية \omterm{def:b1:first-law:heat}{الحرارة} $\sqrt{P/\rho}$ خطأ نيوتن). احسب القيمتين
من أجل الهواء عند \qty{20}{\degreeCelsius} وقارنهما بالقيمة المقيسة
\qty{343}{m/s}. وبيّن أن $c = \sqrt{\gamma RT/M}$ وقارن \omterm{def:b1:kinetic-theory:kinetictemp}{بالسرعة التربيعية
المتوسطة} في \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{مسألة: من منفاخ الدراجة إلى محرك الديزل}

\begin{problem}[{مسألة نهاية الأسبوع --- منفاخ يدوي يحرق الأصابع،
وأسطوانة تشعل وقودها بنفسها، وكرة فولاذية ترتدّ على وسادة من
الهواء: المبدأ الأول في ثلاثة انضغاطات}]
\label{pb:b1:first-law:1}
الهواء: \omterm{def:b1:kinetic-theory:model}{غاز مثالي} ثنائي الذرة، $\gamma = 1.40$ و $C_{V,m} = \tfrac52 R$ و $C_{P,m} =
\tfrac72 R$ و $M = \qty{29}{g/mol}$؛ و $R = \qty{8.314}{J/(mol.K)}$. والحالة
الابتدائية في كل موضع: $P_1 = \qty{1.00}{bar}$ و $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{الجزء I --- منفاخ الدراجة.}
حجم أسطوانة المنفاخ $V_1 = \qty{212}{cm^3}$؛ ويُضغط الهواء الذي
فيها على نحو عكوس وكظوم حتى يبلغ ضغط الإطار، وهو
\qty{7.0}{bar}.
\begin{enumerate}
  \item كمية الهواء في الأسطوانة.
  \item الحجم في نهاية الانضغاط.
  \item \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} في نهاية الانضغاط.
  \item الشغل الذي يتلقاه الهواء في دفعة واحدة.
  \item على الإطار (\qty{2.0}{L}) أن ينتقل من \qty{1}{bar} إلى
        \qty{7}{bar} عند \qty{300}{K}: فكم دفعةً يلزم، وكم يبلغ الشغل
        كله (وخذ الشغل في الدفعة ثابتًا)؟
  \item ثم يبرد الهواء الساخن في الإطار إلى \qty{300}{K}: فماذا يحدث
        لضغط الإطار، وماذا يفعل الدرّاج؟
  \item قارن الشغل الكظوم في الدفعة بشغل انضغاط متساوي \omterm{def:b1:first-law:heat}{الحرارة}
        إلى \qty{7}{bar} نفسها. ولماذا يكون الشغل المتساوي \omterm{def:b1:first-law:heat}{الحرارة} أكبر
        مع أن الغاز ينتهي أبرد؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- انضغاط الديزل.}
أسطوانة حجمها $V_1 = \qty{0.50}{L}$ تضغط هواءها على نحو عكوس
وكظوم بعامل $20$.
\begin{enumerate}[resume]
  \item كمية الهواء، \omterm{def:b1:kinetic-theory:temperature}{ودرجة الحرارة} $T_2$ بعد الانضغاط.
  \item الضغط $P_2$.
  \item يشتعل الوقود من تلقائه فوق \qty{500}{K} تقريبًا: فسّر لماذا
        لا يحتاج الديزل إلى شمعة إشعال، ولماذا يجب ألّا يشتعل
        محرك البنزين ذاتيًّا (ونسبته $\approx 10$، انظر \cref{exo:b1:first-law:3}).
  \item الشغل الذي يتلقاه الهواء أثناء الانضغاط.
  \item عند \qty{3000}{rpm} تضغط أسطوانة واحدة $25$ مرة في الثانية:
        فما الاستطاعة التي يمتصّها الانضغاط (وهي تُردّ لاحقًا في
        الدور)؟
  \item الانضغاطات الحقيقية سريعة (فتكون لاعكوسة) وتفقد \omterm{def:b1:first-law:heat}{حرارة} إلى
        الجدران: فإلى أيّ جهة يزيح كلٌّ من الأثرين $T_2$؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الاحتراق والتمدد.}
عند أعلى الشوط يحترق \qty{20}{mg} من الوقود (قيمته الحرارية
\qty{43}{MJ/kg}) بينما يبدأ المكبس بالنزول، عند ثبات الضغط
$P_2$؛ ثم يتمدد الغاز على نحو عكوس وكظوم عائدًا إلى $V_1$.
وعامِل الغاز على أنه هواء في كل ذلك.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{الحرارة} التي يحرّرها الوقود.
  \item \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} $T_3$ والحجم $V_3$ في نهاية الاحتراق المتساوي
        الضغط.
  \item الشغل المتبادل في المرحلة المتساوية الضغط (انتبه للإشارة!).
  \item \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} $T_4$ في نهاية التمدد الكظوم، والشغل
        المتبادل فيه.
  \item الشغل الصافي الذي يقدّمه الغاز على امتداد الأشواط الأربعة،
        ونسبته إلى \omterm{def:b1:first-law:heat}{الحرارة} المحرَّرة (وهي المردود المثالي).
  \item قارن ذلك المردود بالمقدار $1 - T_1/T_3$، وهو الحدّ الذي
        سيقيمه \cref{ch:b1:heat-engines} لأيّ محرك يعمل
        بين درجتَي \omterm{def:b1:first-law:heat}{الحرارة} القصويين هاتين.
  \item \omterm{def:b1:first-law:heat}{الحرارة} المطروحة مع العادم، بالمبدأ الأول على امتداد الدور؛
        وتحقّق من الموازنة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- قياس $\gamma$ بكرة مرتدّة.}
كرة فولاذية كتلتها $m = \qty{16.5}{g}$ تنطبق على أنبوب زجاجي شاقولي
مقطعه $S = \qty{2.0}{cm^2}$ مغلق من أسفله بدورق حجمه
$V = \qty{10}{L}$ من الهواء عند $P = \qty{1.0}{bar}$؛ فإذا أُزيحت بمقدار $x$
عن وضع توازنها اهتزّت.
\begin{enumerate}[resume]
  \item لماذا يمكن اعتبار انضغاطات الهواء وتمدداته كظومة؟ واكتب
        العلاقة بين انزياح صغير $x$ وتغيّر الضغط $\dd P$ (\omterm{prop:b1:first-law:transformations}{قانون
        لابلاس}، بعد النشر الخطي).
  \item استنتج القوة المُعيدة على الكرة وبيّن أن حركتها
        توافقية، مع $\omega_0^2 = \gamma PS^2/mV$.
  \item احسب الدور، ثم قيمة $\gamma$ التي تُستنتج من دور
        مقيس قدره \qty{1.10}{s}.
  \item لماذا يحيد الاحتكاك بين الكرة والأنبوب، وتسرّب الهواء،
        بالنتيجة، وإلى أيّ جهة؟
  \item لخّص الانضغاطات الثلاثة: أيّ مقدار حفظه المبدأ الأول في
        كل منها، وما الذي حكمه \omterm{prop:b1:first-law:gas}{الأس الكظوم} $\gamma$.
\end{enumerate}
\end{problem}
