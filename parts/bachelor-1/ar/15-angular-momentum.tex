\chapter{العزم الحركي}\label{ch:b1:angular-momentum}

تسحب متزلجة تدور وذراعاها ممدودتان ذراعيها إليها، فتدور،
بلا دفعة من أحد، بضعف السرعة. وينفتح باب لطرف إصبع
عند المقبض ويقاوم كتفًا عند المفصلة. وتجرف الكواكب مساحات
متساوية في أزمنة متساوية --- وقد لاحظ كبلر ذلك قبل أربعة قرون، قبل
أن يستطيع أحد قول السبب بزمن طويل. والقمر، تشدّه المدود التي
يثيرها، يبتعد عن الأرض بضعة سنتيمترات في السنة بينما تطول
أيامنا طولًا لا يُحسّ. ووراء هذه الأربع جميعًا مقدار متجهي
واحد هو \omterm{def:b1:angular-momentum:L}{العزم الحركي}، وقانون واحد: مشتقه يساوي
\omterm{def:b1:angular-momentum:moment}{عزم} القوى. ويعرّف هذا الفصل الاثنين، ويبرهن على القانون من أجل
جسيم ومن أجل جملة، ويستعمله في البنادول والمدارات
والأشياء الدوّارة.

\section{عزم قوة}

\begin{definition}[العزم حول نقطة؛ وحول محور]\label{def:b1:angular-momentum:moment}
\index{عزم قوة}\index{عزم الدوران}\index{ذراع الرافعة}
إن \emph{عزم} (أو \emph{عزم دوران}) قوة حول نقطة $O$، إذا كانت
القوة $\vect F$ مؤثرة في $M$، هو المتجهة
\[
\vect{\mathcal M}_O(\vect F) = \vect{OM}\wedge\vect F \qquad (\unit{N.m}),
\]
وهي عمودية على مستوي $\vect{OM}$ و $\vect F$، ومعيارها
$OM\cdot F\sin\alpha = F\,d$، حيث $d$ --- وهو \emph{ذراع الرافعة} --- هو
بعد $O$ عن حامل $\vect F$. أما العزم حول
محور موجَّه $\Delta$ (متجهته الواحدية $\vect e_\Delta$) يمرّ بالنقطة $O$ فهو
المقدار السلّمي $\mathcal M_\Delta = \vect{\mathcal M}_O\cdot\vect e_\Delta$، وهو مستقل
عن اختيار $O$ على $\Delta$؛ ويتلاشى إذا كانت $\vect F$ موازية
للمحور $\Delta$ أو قاطعةً له. وقوتان متعاكستان على حاملين مختلفين
تشكّلان \emph{مزدوجة} عزمها $\vect F\wedge\vect{AB}$ هو نفسه
حول كل نقطة.
\end{definition}

\begin{example}[الباب]\label{ex:b1:angular-momentum:door}
دفعٌ مقداره \qty{20}{N} عمودي على باب على بعد \qty{0.80}{m} من
المفصلات عزمه $\qty{16}{N.m}$ حول محور المفصلة؛ والدفع نفسه
عند \qty{0.10}{m} عزمه \qty{2}{N.m}؛ وعلى امتداد الباب (فحامله يمرّ
بالمفصلات) عزمه معدوم. أما ردّ فعل المفصلة نفسها، وهو يقطع المحور،
فلا \omterm{def:b1:angular-momentum:moment}{عزم} له حوله: وذلك ما يجعل المحور الموضع الطبيعي لأخذ
العزوم.
\end{example}

\section{العزم الحركي لجسيم}

\begin{definition}[العزم الحركي]\label{def:b1:angular-momentum:L}
\index{عزم حركي}
إن \emph{\omterm{def:b1:angular-momentum:moment}{العزم} الحركي} حول $O$ لجسيم كتلته $m$ يقع في $M$
وسرعته $\vect v$ هو
\[
\vect L_O = \vect{OM}\wedge m\vect v \qquad (\unit{kg.m^2/s}),
\]
وحول محور $\Delta$ يمرّ بالنقطة $O$ يكون $L_\Delta = \vect L_O\cdot\vect e_\Delta$.
ومن أجل حركة في مستوٍ عمودي على $\Delta$، \omterm{def:b1:point-kinematics:cylindrical}{بالإحداثيات القطبية}
حول المحور، يكون $L_\Delta = mr^2\dot\theta$؛ ومن أجل دائرة نصف قطرها $R$،
$L_\Delta = mR^2\omega = mRv$.
\end{definition}

\begin{proof}
$\vect{OM} = r\vect e_r$، و $\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta$:
$\vect{OM}\wedge m\vect v = mr^2\dot\theta\,\vect e_r\wedge\vect e_\theta = mr^2\dot\theta\,\vect e_z$.
\end{proof}

\begin{theorem}[مبرهنة العزم الحركي]\label{thm:b1:angular-momentum:theorem}
\index{مبرهنة العزم الحركي}
في \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}، ومن أجل نقطة $O$ ثابتة في ذلك المرجع،
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_i),
\qquad
\frac{\dd L_\Delta}{\dd t} = \sum\mathcal M_\Delta(\vect F_i)
\]
من أجل محور ثابت $\Delta$. وإذا تلاشى \omterm{def:b1:angular-momentum:moment}{العزم} الكلي حول $O$
انحفظ $\vect L_O$.
\end{theorem}

\begin{proof}
$\dd(\vect{OM}\wedge m\vect v)/\dd t = \vect v\wedge m\vect v + \vect{OM}\wedge m\vect a
= \vect 0 + \vect{OM}\wedge\sum\vect F_i$، باستعمال قانون نيوتن الثاني
(مع $O$ ثابتة بحيث $\dd\vect{OM}/\dd t = \vect v$). ثم أسقِط على $\vect e_\Delta$.
\end{proof}

\begin{corollary}[القوى المركزية: الحركة المستوية وقانون المساحات]\label{cor:b1:angular-momentum:central}
\index{قوة مركزية}\index{قانون المساحات}\index{سرعة مساحية}
جسيم لا يخضع إلا إلى \emph{قوة مركزية} (أي موجَّهة دائمًا
على امتداد $\vect{OM}$، نحو المركز الثابت $O$ أو بعيدًا عنه) يبقي
$\vect L_O$ ثابتًا. وتبقى حركته في المستوي المارّ بالنقطة $O$
عموديًّا على $\vect L_O$، ويكون في ذلك المستوي $r^2\dot\theta = C$
(ثابتًا): أي إن \omterm{def:b1:point-kinematics:frame}{متجهة الموضع} $\vect{OM}$ تجرف مساحةً بمعدّل ثابت هو
\emph{\omterm{cor:b1:angular-momentum:central}{السرعة المساحية}}
\[
\frac{\dd A}{\dd t} = \frac12 r^2\dot\theta = \frac{L_O}{2m} = \frac{C}{2} .
\]
\end{corollary}

\begin{proof}
$\vect{OM}\wedge\vect F = \vect 0$ من أجل $\vect F \parallel \vect{OM}$: ومنه $\vect L_O$
ثابت. و $\vect{OM}\perp\vect L_O$ دائمًا، ومن هنا المستوي؛ و $L_O =
mr^2\dot\theta$ ثابت. والمساحة المجروفة خلال $\dd t$ هي
مثلث ضلعاه $\vect{OM}$ و $\vect v\dd t$: أي $\dd A = \tfrac12\abs{\vect{OM}
\wedge\vect v}\dd t = \tfrac12 r^2\dot\theta\,\dd t$.
\end{proof}

\begin{example}[البندول بالعزوم]\label{ex:b1:angular-momentum:pendulum}
بندول بسيط، ومحور $\Delta$ يمرّ بنقطة التعليق عموديًّا على
مستوي التأرجح: يكون $L_\Delta = m\ell^2\dot\theta$؛ والشدّ يقطع
المحور (فلا \omterm{def:b1:angular-momentum:moment}{عزم} له)، \omterm{def:b1:angular-momentum:moment}{وعزم} الوزن هو $-mg\ell\sin\theta$ (\omterm{def:b1:angular-momentum:moment}{وذراع الرافعة}
$\ell\sin\theta$، وهو مُعيد). فتعطي المبرهنة $m\ell^2\ddot\theta =
-mg\ell\sin\theta$، أي معادلة \cref{ex:b1:newton-dynamics:pendulum}
دون كتابة الشدّ أصلًا --- وهي الميزة الكبرى للعزوم:
فالقوى المارّة بالمحور تختفي.
\end{example}

\begin{example}[سحب قرص إلى الداخل]\label{ex:b1:angular-momentum:puck}
ينزلق قرص على جليد عديم الاحتكاك على دائرة، يمسكه خيط يمرّ
بثقب في المركز؛ ويُسحب الخيط حتى ينتصف نصف القطر.
والشدّ مركزي: فينحفظ $L = mr^2\omega$، ومنه يُضرب $\omega$
في $4$، والسرعة $r\omega$ في $2$، \omterm{thm:b1:work-and-energy:ke}{والطاقة الحركية} في $4$
--- والطاقة الزائدة هي شغل السحب، الذي يقاومه القرص
طوال طريقه إلى الداخل. وذراعا المتزلجة فيزياء واحدة مع كتلة
موزَّعة (\cref{exo:b1:angular-momentum:4}).
\end{example}

\section{جمل النقاط والدوران حول محور ثابت}

\begin{theorem}[العزم الحركي لجملة]\label{thm:b1:angular-momentum:system}
\index{عزم حركي!لجملة}
من أجل جملة من الجسيمات، مع $\vect L_O = \sum_i\vect{OM_i}\wedge m_i\vect v_i$
و $O$ ثابتة في \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}،
\[
\frac{\dd\vect L_O}{\dd t} = \sum\vect{\mathcal M}_O(\vect F_{\mathrm{ext}}):
\]
فلا تُحسب إلا عزوم القوى \emph{الخارجية}؛ أما القوى الداخلية،
التي تحقق قانون \omterm{thm:b1:newton-dynamics:laws}{الفعل ورد الفعل} على امتداد المستقيم الواصل بين
الجسيمين، فلا تسهم بشيء.
\end{theorem}

\begin{proof}
اجمع مبرهنات الجسيمات. ومن أجل ثنائية داخلية، $\vect{OM_i}\wedge\vect F_{j\to i}
+ \vect{OM_j}\wedge\vect F_{i\to j} = (\vect{OM_i} - \vect{OM_j})\wedge\vect F_{j\to i}
= \vect{M_jM_i}\wedge\vect F_{j\to i} = \vect 0$ لأن القوة على امتداد
$\vect{M_jM_i}$.
\end{proof}

\begin{proposition}[دوران صلب حول محور ثابت]\label{prop:b1:angular-momentum:inertia}
\index{عزم العطالة}
إذا حافظت الجسيمات على أبعاد ثابتة $r_i$ عن محور $\Delta$ ودارت
كلها حوله \omterm{cor:b1:point-kinematics:circular}{بالسرعة الزاوية} نفسها $\omega$ (أي دوران صلب)، كان
\[
L_\Delta = J_\Delta\,\omega, \qquad J_\Delta = \sum_i m_ir_i^2, \qquad
E_k = \tfrac12 J_\Delta\omega^2,
\]
حيث $J_\Delta$ هو \emph{\omterm{prop:b1:angular-momentum:inertia}{عزم العطالة}} حول $\Delta$، وتُقرأ
المبرهنة $J_\Delta\dot\omega = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})$.
\end{proposition}

\begin{proof}
يسهم كل جسيم بالمقدار $m_ir_i^2\omega$ في $L_\Delta$، وبالمقدار
$\tfrac12 m_i(r_i\omega)^2$ في $E_k$. ويوسّع \cref{ch:b1:systems-of-points}
هذا إلى الأجسام الصلبة المتصلة.
\end{proof}

\begin{example}[المتزلجة]\label{ex:b1:angular-momentum:skater}
بذراعين ممدودتين يكون لجسم المتزلجة $J \approx \qty{4}{kg.m^2}$ مضافًا إليه ذراعان
كتلة كلٍّ منهما \qty{3}{kg} على بعد \qty{0.8}{m}: أي $J_1 = 4 + 2 \times 3 \times 0.64 =
\qty{7.8}{kg.m^2}$. وبذراعين مضمومتين (على \qty{0.2}{m}): $J_2 = \qty{4.2}{kg.m^2}$.
ولا \omterm{def:b1:angular-momentum:moment}{عزم} لردّ فعل الجليد ولا للوزن حول المحور
الشاقولي: فينحفظ $J\omega$ ويرتفع معدل الدوران بالعامل $J_1/J_2 = 1.85$
--- من $2$ إلى $3.7$ دورة في الثانية --- بينما ترتفع \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}
بالعامل نفسه، وتدفعه العضلات وهي تضمّ الذراعين.
\end{example}

\begin{remark}[السكون]\label{rem:b1:angular-momentum:statics}
\index{توازن جسم صلب}
لجسم صلب ساكن $\vect L_O = \vect 0$ ثابت: فيجب أن تكون محصلة القوى
الخارجية معدومة \emph{و} أن يكون عزمها الكلي حول أيّ نقطة معدومًا.
والشرط الثاني هو قانون الرافعة في الميزان والأرجوحة الخشبية
والعتلة: فقوة صغيرة بعيدة عن نقطة الارتكاز توازن قوة كبيرة قريبة
منها.
\end{remark}

لجذع متزلجة $J = \qty{4.0}{kg.m^2}$ حول محور الدوران؛ وتُعامل كل
ذراع (\qty{3.0}{kg}) كنقطة على بعد \qty{0.80}{m}، ثم على
\qty{0.20}{m}. احسب معدّل الدوران بعد ضمّ الذراعين، انطلاقًا من
\qty{2.0}{} دورة في الثانية؛ ونسبة الطاقتين الحركيتين؛ ومن أين تأتي
الطاقة الزائدة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:5}
اشتقّ معادلة البندول $\ell\ddot\theta + g\sin\theta = 0$ من
مبرهنة \omterm{def:b1:angular-momentum:L}{العزم الحركي} حول محور التعليق، وأعطِ \omterm{def:b1:angular-momentum:moment}{عزم}
كل قوة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:6}
يتغير بعد الأرض عن الشمس من \qty{1.471e11}{m} (في الحضيض)
إلى \qty{1.521e11}{m} (في الأوج). باستعمال قانون المساحات، أوجد نسبة

أعطِ السرعتين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:7}
يدور قرص كتلته \qty{0.20}{kg} بسرعة \qty{2.0}{m/s} على خيط طوله \qty{0.50}{m}
يمرّ بثقب؛ ويُسحب الخيط حتى يصير نصف القطر \qty{0.25}{m}.
احسب السرعة \omterm{cor:b1:point-kinematics:circular}{والسرعة الزاوية} الجديدتين؛ والطاقتين الحركيتين قبل وبعد؛ والشغل
الذي يبذله السحب. ولماذا يستطيع الشدّ أن يبذل شغلًا هنا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:8}
يستند لوح منتظم كتلته \qty{20}{kg} وطوله \qty{4.0}{m} على

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:angular-momentum:9}
آلة أتوود ($m_1 = \qty{2.0}{kg}$ و $m_2 = \qty{1.0}{kg}$) معلَّقة على

والشدّين. وقارن بحالة البكرة عديمة الكتلة.
\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:11}

\begin{exercise}[$\star\star\star$]\label{exo:b1:angular-momentum:12}

\end{exercise}

\section{مسألة: لماذا يبتعد القمر}

\begin{problem}[{مسألة نهاية الأسبوع --- المدود التي يثيرها القمر على
الأرض تشدّه إلى الأمام وتشدّ الأرض إلى الخلف: كيف ينساب العزم الحركي
من أيامنا إلى مدار القمر، سنتيمترًا بعد سنتيمتر،
وإلى أين تذهب الطاقة}]
\label{pb:b1:angular-momentum:1}
المعطيات: $G = \qty{6.67e-11}{SI}$؛ وللأرض $M = \qty{5.97e24}{kg}$ و $R =
\qty{6.37e6}{m}$، ودورانها $\Omega = 2\pi/(\qty{86164}{s})$، \omterm{def:b1:angular-momentum:moment}{وعزم} عطالتها
$J = 0.33\,MR^2$؛ وللقمر $m = \qty{7.35e22}{kg}$، ويُعامل مداره
دائريًّا، $r = \qty{3.84e8}{m}$؛ ويقيس التحديد الليزري للمسافة إلى القمر
$\dd r/\dd t = \qty{3.8}{cm/yr}$؛ ويطول اليوم بمقدار \qty{2.3}{ms} في كل
قرن؛ و $1$ قرن $= \qty{3.16e9}{s}$.

\medskip
\noindent\textbf{الجزء I --- العزوم الحركية.}
\begin{enumerate}
  \item احسب \omterm{cor:b1:point-kinematics:circular}{السرعة الزاوية} المدارية $\omega$ للقمر انطلاقًا من
        قانون نيوتن (بمدار دائري)، وتحقق منها بمقارنتها بالشهر
        النجمي \qty{27.3}{d}.
  \item احسب \omterm{def:b1:angular-momentum:L}{العزم الحركي} المداري للقمر $L_{\text{مداري}} =
        mr^2\omega$.
  \item بيّن أنه من أجل مدار دائري $L_{\text{مداري}} = m\sqrt{GMr}$، ثم
        استنتج $\dd L_{\text{مداري}}/\dd r = L_{\text{مداري}}/2r$.
  \item احسب \omterm{def:b1:angular-momentum:L}{العزم الحركي} الذاتي للأرض $L_{\text{ذاتي}} =
        J\Omega$ وقارنه بالمقدار $L_{\text{مداري}}$.
  \item يدور القمر حول نفسه مرة في كل شهر: قدّر عزمه الحركي
        الذاتي (مع $J_m \approx 0.4\,mR_m^2$ و $R_m = \qty{1.74e6}{m}$)
        وبرّر إهماله.
  \item ما العزوم الخارجية المؤثرة في جملة الأرض--القمر حول
        مركز كتلتها؟ وبرهن على أن \omterm{def:b1:angular-momentum:moment}{العزم} الكلي
        $L = L_{\text{ذاتي}} + L_{\text{مداري}}$ ينحفظ بتقريب جيد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- \omterm{def:b1:angular-momentum:moment}{عزم} المدّ.}
يثير القمر انتفاخين مائيين على الأرض؛ ولأن الأرض
تدور أسرع مما يدور القمر في مداره، يجرّ الاحتكاك الانتفاخين أمام
مستقيم الأرض--القمر بزاوية صغيرة.
\begin{enumerate}[resume]
  \item ارسم تخطيطيًّا الأرض والانتفاخ السابق والقمر. فبأيّ
        اتجاه يؤثر جذب القمر للانتفاخ الأقرب،
        بالنسبة إلى دوران الأرض؟
  \item استنتج إشارة \omterm{def:b1:angular-momentum:moment}{العزم} على دوران الأرض: فهل يطول
        اليوم أم يقصر؟
  \item وبالفعل ورد الفعل، أيّ \omterm{def:b1:angular-momentum:moment}{عزم} يؤثر في حركة القمر
        المدارية؟ وهل يكبر $L_{\text{مداري}}$ أم يصغر؟
  \item انطلاقًا من السؤال 3، هل يكبر $r$ أم يصغر؟ وماذا عن سرعة
        القمر المدارية $v = \sqrt{GM/r}$؟
  \item حلّ المفارقة الظاهرية: يُدفع القمر إلى الأمام، ومع ذلك
        ينتهي به الأمر إلى حركة أبطأ.
  \item إلى أيّ حالة نهائية يميل هذا التبادل؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الأعداد.}
\begin{enumerate}[resume]
  \item انطلاقًا من طول اليوم، احسب التغير النسبي
        $\Delta\Omega/\Omega$ في كل قرن.
  \item استنتج تغير $L_{\text{ذاتي}}$ في كل قرن.
  \item استنتج تغير $L_{\text{مداري}}$ في كل قرن.
  \item باستعمال السؤال 3، حوّله إلى تغير في $r$ في كل قرن
        وفي كل سنة. وقارنه بقيمة التحديد الليزري.
  \item بأيّ نسبة، وبكم ثانية، يطول الشهر النجمي
        في كل قرن (مع $T \propto r^{3/2}$)؟
  \item احسب تغير \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} الدورانية للأرض
        في كل قرن (مع $\Delta E = L_{\text{ذاتي}}\Delta\Omega$ من الرتبة
        الأولى).
  \item احسب تغير \omterm{thm:b1:work-and-energy:em}{الطاقة الميكانيكية} المدارية للقمر،
        $E_{\text{مداري}} = -GMm/2r$، في كل قرن.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الطاقة.}
\begin{enumerate}[resume]
  \item بيّن أن الأرض تفقد طاقة أكثر مما يكسب القمر، ثم
        احسب الفرق في كل قرن والاستطاعة الموافقة.
        وإلى أين تذهب؟
  \item قارن تلك الاستطاعة باستهلاك البشرية، ونحوه
        \qty{18}{TW}، وبتدفق حرارة الأرض الداخلية، ونحوه
        \qty{47}{TW}.
  \item في الحالة النهائية من السؤال 12 يتساوى اليوم والشهر:
        $\Omega = \omega = \sqrt{GM/r^3}$. اكتب انحفاظ
        $L_{\text{ذاتي}} + L_{\text{مداري}}$، وتحقق من أن $r \approx
        \qty{5.5e8}{m}$ يحققه، وأعطِ الدور المشترك بالأيام.
  \item ولماذا لن تُبلغ تلك الحالة في الواقع أبدًا؟ (فكّر في مدود
        الشمس.)
  \item يُظهر القمر للأرض وجهًا واحدًا أصلًا. فسّر ذلك بالآلية
        نفسها، وقل لماذا حدث للقمر أولًا.
  \item لخّص في ثلاثة أسطر: ما الذي ينحفظ، وما الذي يُتبادل،
        وما الذي يتبدّد.
\end{enumerate}
\end{problem}
