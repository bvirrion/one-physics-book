\section{قوة لورنتز}

\begin{definition}[قوة لورنتز]\label{def:b1:charged-particles:lorentz}
\index{قوة لورنتز}\index{مجال كهربائي}\index{مجال مغناطيسي}
جسيم شحنته $q$ يتحرك بالسرعة $\vect v$ في مجال كهربائي
$\vect E$ ومجال مغناطيسي $\vect B$ يتلقّى القوة
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
والجزء الكهربائي على امتداد $\vect E$ (أو معاكس له، من أجل $q < 0$) ومستقل
عن الحركة؛ أما الجزء المغناطيسي فعمودي على كلٍّ من
$\vect v$ و $\vect B$، ومتناسب مع السرعة، ويتلاشى من أجل
جسيم ساكن. والوحدات: $\vect E$ بوحدة \unit{V/m} (أو \unit{N/C})، و $\vect B$
بالتسلا (\unit{T}): حيث $\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[شغل قوة لورنتز]\label{prop:b1:charged-particles:work}
لا تبذل القوة المغناطيسية شغلًا: فاستطاعتها $q(\vect v\wedge\vect B)\cdot\vect v$
معدومة، ومن ثمّ يغيّر \omterm{def:b1:charged-particles:lorentz}{مجال مغناطيسي} وحده اتجاه السرعة
لا مقدارها ولا \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}. أما القوة الكهربائية فاستطاعتها
$q\vect E\cdot\vect v$، وشغلها بين نقطتين هو $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors}): فجسيم شحنته $q$
يعبر توترًا $U$ يكسب \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. والمجال الكهرساكن يشتقّ من
الكمون، $\vect E = -\overrightarrow{\operatorname{grad}}V$، ومنه تكون $q\vect E$
محافظة مع $E_p = qV$.
\end{proof}

\begin{definition}[الإلكترون-فولط]\label{def:b1:charged-particles:eV}
\index{إلكترون-فولط}
\emph{الإلكترون-فولط} هو \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} التي تكسبها شحنة
أولية عابرةً فولطًا واحدًا: $\qty{1}{eV} = \qty{1.60e-19}{J}$. وتعيش الكيمياء
عند بضعة \unit{eV}، وأنابيب الأشعة السينية عند عشرات \unit{keV}،
والفيزياء النووية عند \unit{MeV}، وفيزياء الجسيمات عند \unit{GeV} و \unit{TeV}.
\end{definition}

\begin{remark}[لماذا تُهمل الثقالة]\label{rem:b1:charged-particles:gravity}
من أجل إلكترون في مجال \qty{1}{kV/m} --- وهي قيمة مخبرية
متواضعة --- تكون القوة الكهربائية $eE = \qty{1.6e-16}{N}$، مقابل وزن
$mg = \qty{9e-30}{N}$: أي أصغر بالعامل $10^{13}$. فيُسقط الوزن من
كل حساب في هذا الفصل دون تردد. (أما
الحدّ النسبي فأمر آخر: إذ تصحّ الصيغ أدناه من أجل
$\tfrac12 mv^2 \ll mc^2$، أي دون نحو \qty{50}{keV} من أجل الإلكترونات
و \qty{100}{MeV} من أجل البروتونات؛ انظر مجلد السنة الثالثة.)
\end{remark}

\section{المجال الكهربائي المنتظم}

\begin{proposition}[التسريع والانحراف]\label{prop:b1:charged-particles:efield}
\index{توتر التسريع}
في مجال منتظم $\vect E$ يكون التسارع $\vect a = q\vect E/m$
ثابتًا: فالحركة حركة قذيفة، مع $q\vect E/m$ محلّ
$\vect g$.
\begin{itemize}
  \item إذا أُطلق من السكون وسُرّع عبر توتر $U$، بلغ
        الجسيم السرعة $v = \sqrt{2\abs{qU}/m}$.
  \item وإذا دخل بالسرعة $v_0$ على امتداد $x$ منطقةً طولها $L$ حيث
        $\vect E = E\vect e_y$، تبع القطع المكافئ $y = (qE/2mv_0^2)x^2$
        وغادر منحرفًا بالزاوية $\alpha$ حيث $\tan\alpha =
        qEL/mv_0^2$، وكأنه أتى على خط مستقيم من
        منتصف اللوحين.
\end{itemize}
\end{proposition}

\begin{proof}
بالطاقة: $\tfrac12 mv^2 = \abs{qU}$. وللانحراف: $x = v_0t$ و $y =
\tfrac12(qE/m)t^2$؛ وعند المخرج $\dd y/\dd x = (qE/m)(x/v_0^2)$ محسوبًا
عند $L$؛ والمماس عند المخرج، $y = (qEL/mv_0^2)(x - L/2)$، يقطع
المحور عند $x = L/2$.
\end{proof}

\section{المجال المغناطيسي المنتظم}

\begin{theorem}[الحركة في مجال مغناطيسي منتظم]\label{thm:b1:charged-particles:bfield}
\index{حركة سيكلوترونية}\index{نصف قطر لارمور}\index{تواتر السيكلوترون}\index{حركة حلزونية}
جسيم شحنته $q$ وكتلته $m$ وسرعته $\vect v_0$ عمودية على
مجال منتظم $\vect B$ يصف دائرة بسرعة ثابتة $v_0$، نصف قطرها
وتواترها الزاوي
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
حيث \emph{\omterm{thm:b1:charged-particles:bfield}{تواتر السيكلوترون}} $\omega_c$ مستقل عن السرعة؛
وجهة الدوران تتوقف على إشارة $q$. وإذا كان للسرعة $\vect v_0$ أيضًا
مركّبة $v_\parallel$ على امتداد $\vect B$، بقيت تلك المركّبة دون تغيير وكان
المسار \emph{حلزونًا} نصف قطره $mv_\perp/\abs qB$ وخطوته
$2\pi mv_\parallel/\abs qB$ ملفوفًا حول خطوط المجال.
\end{theorem}

\begin{proof}
القوة $q\vect v\wedge\vect B$ عمودية على $\vect v$ وعلى $\vect B$؛
والسرعة ثابتة (فلا شغل) ولا ترى المركّبة على امتداد $\vect B$ أيّ
قوة. وفي المستوي العمودي على $\vect B$ يكون للقوة المعيار
الثابت $\abs qv_\perp B$، وهي عمودية دائمًا على السرعة: أي \omterm{cor:b1:point-kinematics:circular}{حركة دائرية}
منتظمة (\cref{thm:b1:point-kinematics:frenet}) مع $mv_\perp^2/R =
\abs qv_\perp B$. ومنه $\omega_c = v_\perp/R$؛ والدور $2\pi m/\abs qB$ مضروبًا في
$v_\parallel$ هو الخطوة.
\end{proof}

\section{الأجهزة}

\begin{proposition}[منتقي السرعات ومطياف الكتلة]\label{prop:b1:charged-particles:spectrometer}
\index{منتقي السرعات}\index{مطياف الكتلة}
في مجالين منتظمين متقاطعين $\vect E \perp \vect B$، يمرّ جسيم يتحرك
عموديًّا عليهما دون انحراف إذا وفقط إذا كان $v = E/B$، أيًّا كانت
شحنته وكتلته (\emph{\omterm{prop:b1:charged-particles:spectrometer}{منتقي السرعات}}). وفي \omterm{def:b1:charged-particles:lorentz}{مجال مغناطيسي} وحده يصف
جسيم معلوم السرعة نصف دائرة قطرها $2mv/\abs qB$:
فقياسه يعطي $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{مطياف الكتلة}}). وأيونات مسرَّعة
عبر توتر $U$ ثم مثنيّة في $B$ تحطّ عند
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
فينفصل نظيران كتلتاهما $m$ و $m + \Delta m$ بمقدار $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
تقتضي $q\vect E + q\vect v\wedge\vect B = \vect 0$ أن يكون $vB = E$ بالتوجيه
الصحيح. و $R = mv/qB$ مع $v = \sqrt{2qU/m}$؛ ثم اشتقّ
$\ln R = \tfrac12\ln m + \text{ثابت}$.
\end{proof}

\begin{example}[نظائر الكربون]\label{ex:b1:charged-particles:isotopes}
أيونات $^{12}$C و $^{13}$C أحادية الشحنة مسرَّعة عبر \qty{20}{kV}
في \qty{0.50}{T}: $R = \qty{14.1}{cm}$ و \qty{14.7}{cm}، أي على بعد \qty{5.8}{mm}
--- فبقعتان منفصلتان على الكاشف، ونسبة شدّتيهما
هي الوفرة النظيرية، وهي المادة الخام للتأريخ بالكربون المشعّ
ولكل تحليل نظيري.
\end{example}

\begin{proposition}[السيكلوترون]\label{prop:b1:charged-particles:cyclotron}
\index{سيكلوترون}
نصفا أسطوانة مجوّفان (يسمّيان ``دي'') في مجال منتظم $\vect B$، مع
توتر متناوب $U$ بينهما عند \omterm{thm:b1:charged-particles:bfield}{تواتر السيكلوترون}: فيدور
الجسيم داخل كل نصف (إذ لا مجال هناك)، ويُسرَّع بالمقدار
$\abs qU$ عند كل عبور من عبورَي الفجوة في كل دورة، ويلتفّ خارجًا
مع $R \propto v$؛ وعند نصف قطر الاستخراج $R_{\max}$ تكون طاقته
الحركية
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
مستقلةً عن التوتر، الذي لا يحدد إلا عدد الدورات.
\end{proposition}

\begin{proof}
لمّا كان $\omega_c$ لا يتوقف على السرعة، بقي توتر ثابت التواتر
متساوقًا مع الجسيم دورةً بعد دورة؛ و $v = \abs qBR/m$ عند نصف القطر
$R$.
\end{proof}

\begin{proposition}[مفعول هول]\label{prop:b1:charged-particles:hall}
\index{مفعول هول}
شريط سماكته $d$ يحمل تيارًا $I$ من حاملات شحنتها
$q$ وكثافتها $n$، في مجال $B$ عمودي على الشريط،
ينشأ عبر عرضه \emph{توتر هول}
\[
U_H = \frac{IB}{nqd} :
\]
وهو متناسب مع $B$ (فهو مقياس مغناطيسية) وعكسيًّا مع $n$ (فهو مسبار
لكثافة الحاملات وإشارتها).
\end{proposition}

\begin{proof}
تدفع القوة المغناطيسية $qv_dB$ الحاملات (وسرعة انسياقها $v_d$)
نحو حافة واحدة حتى يوازنها \omterm{def:b1:charged-particles:lorentz}{المجال الكهربائي} العرضي $E_H$:
أي $qE_H = qv_dB$، و $U_H = E_Hw$ عبر العرض $w$؛ ومع $I = nqv_dwd$ يكون
$E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:1}
حوّل \qty{1}{eV} و \qty{13.6}{eV} و \qty{1}{MeV} إلى الجول. واحسب سرعة
إلكترون وسرعة بروتون سُرّعا من السكون عبر \qty{1.0}{kV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:2}
يدخل إلكترون طاقته \qty{1.0}{keV} مجالًا مقداره \qty{1.0}{mT} عموديًّا.
احسب نصف القطر \omterm{def:b1:wave-propagation:signal}{والتواتر الزاوي} السيكلوتروني والدور.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:3}
بروتون سرعته \qty{1.0e5}{m/s} في مجال الأرض (\qty{50}{\micro T}):
احسب نصف قطر دائرته. وماذا يحدث إذا صنعت سرعته $30^\circ$ مع
المجال؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:4}
لمنتقي سرعات $E = \qty{10}{kV/m}$ و $B = \qty{20}{mT}$. فأيّ
سرعة تمرّ؟ وماذا يحدث للأيونات الأسرع، ولسرعة مماثلة بشحنة
معاكسة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:5}
راسم اهتزاز: إلكترونات عند \qty{2.0}{kV}، ولوحان طولهما \qty{4.0}{cm}
ويبعدان \qty{1.0}{cm}، وبينهما \qty{100}{V}، وشاشة تبعد \qty{30}{cm}
عن منتصف اللوحين. احسب الانحراف على الشاشة والحساسية بالمليمتر لكل فولط.
ولماذا يخفّض توتر تسريع أعلى الحساسيةَ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:6}
أيونات $^{12}$C و $^{13}$C أحادية الشحنة، مسرَّعة عبر
\qty{20}{kV}، تدخل مجالًا مقداره \qty{0.50}{T} (مع $\qty{1}{u} = \qty{1.66e-27}{kg}$).
احسب نصفَي القطر والفصل بين البقعتين بعد نصف دورة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:7}
لسيكلوترون بروتونات $B = \qty{1.5}{T}$ و $R_{\max} = \qty{0.50}{m}$.
احسب تواتر توتر النصفين، \omterm{thm:b1:work-and-energy:ke}{والطاقة الحركية} العظمى، وعدد الدورات
من أجل \qty{50}{kV} على النصفين، والزمن المقضي في الداخل. وبأيّ نسبة
ازدادت كتلة البروتون عند الخروج (مع $\gamma - 1 \approx E_k/mc^2$ و
$mc^2 = \qty{938}{MeV}$)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:8}
يدخل إلكترون سرعته \qty{1.0e6}{m/s} مجالًا مقداره \qty{2.0}{mT} بزاوية $30^\circ$
مع خطوط المجال. احسب نصف قطر حلزونه وخطوته؛ وعدد الدورات في كل
متر على امتداد المجال.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:9}
شريط نحاسي سماكته \qty{1.0}{mm} يحمل \qty{5.0}{A} في \qty{1.0}{T}
(مع $n = \qty{8.5e28}{m^{-3}}$). احسب توتر هول. ولماذا تُصنع مجسّات هول من
أنصاف النواقل ($n \sim \qty{1e22}{m^{-3}}$)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:10}
برهن انطلاقًا من قانون نيوتن على أن شحنة في أيّ \omterm{def:b1:charged-particles:lorentz}{مجال مغناطيسي} (منتظم أو
غير منتظم)، بلا \omterm{def:b1:charged-particles:lorentz}{مجال كهربائي}، تحافظ على سرعة ثابتة. ويدخل إلكترون طاقته \qty{10}{keV}
منطقةً ذات مجال قوي غير منتظم: فما الذي يمكن أن يتغير وما الذي
لا يمكن؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:11}
تنطلق شحنة $q$ من السكون في المبدأ في المجالين $\vect E = E\vect e_y$ و
$\vect B = B\vect e_z$. اكتب معادلات الحركة، وبيّن أن
الجسيم ينساق في المتوسط بالسرعة $E/B$ على امتداد $\vect e_x$
(إرشاد: ابحث عن حلّ على الصورة $\vect v = \vect u + \vect w$ مع
$\vect u$ ثابتة)، وصف المسار (وهو منحن دويري). وأعطِ الأعداد من أجل
إلكترون مع $E = \qty{1.0e4}{V/m}$ و $B = \qty{0.10}{T}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:12}
في راسم الاهتزاز في \cref{exo:b1:charged-particles:5}، استبدل
باللوحين وشيعةَ انحراف مغناطيسي تنتج \qty{1.0}{mT} على
\qty{4.0}{cm} نفسها. احسب زاوية الانحراف (بتقريب الزوايا الصغيرة: قوس نصف قطره $R$
على طول $L$)، والانحراف على الشاشة. ولماذا تستعمل أنابيب التلفاز،
بزواياها الكبيرة، الانحراف المغناطيسي بدل الكهربائي؟
\end{exercise}

\section{مسألة: بروتونات ضد ورم}

\begin{problem}[{مسألة نهاية الأسبوع --- السيكلوترون خلف جدار
مستشفى: كيف تُدار البروتونات حتى ثلث سرعة الضوء،
ولماذا تكفّ الآلة البسيطة عن العمل دون الطاقة التي يحتاجها
المعالج، وكم بروتونًا يلزم لعلاج ورم}]
\label{pb:b1:charged-particles:1}
البروتونات: $m = \qty{1.67e-27}{kg}$ و $q = e = \qty{1.60e-19}{C}$ و $mc^2 =
\qty{938}{MeV}$؛ و $c = \qty{3.00e8}{m/s}$. والسيكلوترون: $B = \qty{1.5}{T}$،
ونصف قطر الاستخراج $R_{\max} = \qty{0.50}{m}$، وسعة توتر النصفين
$U = \qty{50}{kV}$.

\medskip
\noindent\textbf{الجزء I --- دورة واحدة.}
\begin{enumerate}
  \item اكتب \omterm{def:b1:charged-particles:lorentz}{قوة لورنتز} على بروتون يتحرك بالسرعة $\vect v$
        عموديًّا على $\vect B$، وبيّن أن سرعته لا يمكن أن تتغير
        داخل نصف الأسطوانة.
  \item اشتقّ نصف قطر دائرته ودوره؛ وبيّن أن
        الدور لا يتوقف على السرعة.
  \item احسب \omterm{thm:b1:charged-particles:bfield}{تواتر السيكلوترون} $f_c$ من أجل هذه البروتونات.
  \item احسب الطاقة المكتسبة عند كل عبور للفجوة؛ ثم في كل دورة.
  \item لماذا يجب أن يتناوب توتر النصفين عند $f_c$ بالضبط؟ وماذا يحدث
        إن حاد عنه قليلًا؟
  \item تُحقن البروتونات في المركز بسرعة مهملة: فما
        نصف القطر بعد الدورة الأولى؟ وبعد $n$ دورة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- إلى الحافة.}
\begin{enumerate}[resume]
  \item احسب \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} عند الاستخراج (بالجول وبوحدة \unit{MeV})، والسرعة
        كنسبة من $c$.
  \item احسب عدد الدورات اللازم والزمن المقضي في الآلة.
  \item احسب الطول الكلي للمسار الحلزوني (أي مجموع المحيطات ---
        وقرّب المجموع بتكامل).
  \item هل تتوقف نتيجة السؤال 7 على $U$؟ وعلامَ يؤثر $U$؟
  \item يجب أن يكون الخلاء في الداخل جيدًا: قدّر المسافة التي يقطعها
        بروتون بين تصادمين مع جزيئات الغاز المتبقي إذا وجب أن يتجاوز
        متوسط المسار الحرّ طولَ الحلزون؛ ويتناسب متوسط المسار الحرّ
        مع $1/p$ وهو نحو \qty{70}{nm} عند الضغط
        الجوي. فأيّ ضغط يلزم؟
  \item احسب $\gamma - 1 \approx E_k/mc^2$ عند الاستخراج والتغير النسبي
        الناتج في الدور. وعلى مدى عدد الدورات الموجود أعلاه، بأيّ جزء
        من الدور انزلق البروتون بالنسبة إلى التوتر؟ علّق على ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- عند \qty{230}{MeV}.}
يحتاج العلاج بالبروتونات إلى نحو \qty{230}{MeV} لبلوغ أورام على عمق
\qty{30}{cm}.
\begin{enumerate}[resume]
  \item بالصيغة غير النسبية، أيّ جداء $BR$ يعطي
        \qty{230}{MeV}؟ ومع $B = \qty{2.0}{T}$، أيّ نصف قطر؟
  \item عند تلك الطاقة $\gamma - 1 = 0.245$: فماذا يحدث لدور
        السيكلوترون وهو يكسب البروتون طاقة، ولماذا يفشل
        سيكلوترون ثابت التواتر؟
  \item يوجد علاجان: تغيير تواتر التوتر أثناء
        التسريع (السيكلوترون المتزامن)، أو جعل $B$ يكبر مع نصف القطر بحيث
        يبقى $B/\gamma$ ثابتًا (السيكلوترون متساوي الزمن). فسّر
        كلًّا منهما في جملة واحدة وسمِّ عيبًا في الأول.
  \item \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة} النسبية هي $p = \gamma mv$ ويظلّ نصف القطر $R =
        p/qB$ صحيحًا: فمع $\gamma = 1.245$ و $v = 0.60c$، احسب
        $R$ من أجل $B = \qty{2.0}{T}$ وقارنه بالسؤال 13.
  \item تمرّ الحزمة المستخرَجة بين لوحين طولهما \qty{1.0}{m}
        مع $E = \qty{1.0e5}{V/m}$ لتوجيهها بزاوية صغيرة:
        قدّر زاوية الانحراف (بلا نسبية، مع $p$
        من السؤال السابق بدل $mv$). فهل التوجيه الكهربائي عملي هنا؟
  \item التوجيه نفسه بمغناطيس \qty{1.0}{T} على \qty{1.0}{m}:
        فما الزاوية؟ واستنتج لماذا تستعمل خطوط الحزم المغانط.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الجرعة.}
\begin{enumerate}[resume]
  \item يودع بروتون طاقته \qty{230}{MeV} معظم طاقته قرب نهاية
        مداه (وهي ذروة براغ). فإذا امتُصّت كلها في
        ورم كتلته \qty{0.50}{kg}، فكم بروتونًا يوصل جرعة
        \qty{2.0}{Gy} (مع $\qty{1}{Gy} = \qty{1}{J/kg}$)؟
  \item شدة تيار الحزمة \qty{1.0}{nA}: فكم بروتونًا في الثانية، وما
        مدة الجلسة؟
  \item احسب الاستطاعة التي تحملها الحزمة؛ وقارنها بمصباح.
  \item تُبطّأ البروتونات حتى التوقف داخل المريض: فما
        الطاقة المودعة لكل بروتون في السنتيمتر الأخير، تقريبًا، إذا
        ضاع فيه ربع الطاقة؟ وبالإلكترون-فولط لكل
        ميكرومتر؟
  \item يكلّف كل تأيّن في النسيج نحو \qty{30}{eV}: فكم تأيّنًا
        يسبّبه بروتون واحد على امتداد مساره كله؟
  \item لماذا تُفضَّل حزمة البروتونات على الأشعة السينية من أجل ورم عميق، في
        جملة واحدة عن مكان ذهاب الطاقة؟
  \item لخّص السلسلة من \omterm{def:b1:charged-particles:lorentz}{قوة لورنتز} إلى الجرعة: أيّ
        المعادلات حدّدت الطاقة ونصف القطر وعدد الدورات
        وزمن العلاج.
\end{enumerate}
\end{problem}
