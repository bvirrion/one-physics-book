\chapter{الدارات المستمرة: قانونا كيرشوف والمبرهنات}\label{ch:b1:dc-circuits}

أدِر المفتاح في صباح بارد: يئنّ المقلع، وتخفت أضواء لوحة القيادة
ثانيةً، ثم تسعل السيارة عائدةً إلى الحياة. فالبطارية التي كانت
تعطي \qty{12.6}{V} ثابتة قبل لحظة هبطت إلى أحد عشر بينما
يسحب المقلع مئةً وخمسين أمبيرًا --- والمصباحان الأماميان،
الموصولان على التفرع، يدفعان الثمن. وكل ما يجري في تلك الثانية يحكمه
قانونا انحفاظ وحفنة من المبرهنات عن الدارات الخطية،
وهي ما يذكره هذا الفصل ويبرهن عليه ويطبّقه: قانونا كيرشوف،
ونماذج المنابع والناقلات الأومية، والمجزّئات، وتيفنان ونورتن،
والفنّ البياني في إيجاد \omterm{met:b1:dc-circuits:loadline}{نقطة العمل}.

\section{التيار والتوتر والنظام شبه المستقر}

\begin{definition}[التيار الكهربائي]\label{def:b1:dc-circuits:current}
\index{تيار كهربائي}\index{أمبير}
\emph{التيار الكهربائي} عبر سطح (كمقطع سلك)
هو الشحنة العابرة له في واحدة الزمن، محسوبةً موجبةً في اتجاه
مختار:
\[
i = \frac{\dd q}{\dd t} \qquad (\text{بالأمبير: } \qty{1}{A} = \qty{1}{C/s}).
\]
وقلب الاتجاه يقلب إشارة $i$. وفي المعدن تكون
الحوامل إلكترونات تنساق \emph{عكس} اتجاه التيار
الاصطلاحي، بكسر من المليمتر في الثانية.
\end{definition}

\begin{definition}[الكمون والتوتر]\label{def:b1:dc-circuits:voltage}
\index{كمون كهربائي}\index{توتر}\index{مرجع أرضي}
لكل نقطة من دارة \emph{كمون كهربائي} $V$ (بالفولط)،
معرَّف إلى حدّ ثابت يثبّته اختيار \emph{مرجع أرضي} ($V = 0$).
و\emph{التوتر} بين $A$ و $B$ هو فرق الكمون
$u_{AB} = V_A - V_B$، ويُرسم سهمًا من $B$ إلى $A$. والطاقة التي
تفقدها شحنة $q$ في انتقالها من $A$ إلى $B$ هي $q\,u_{AB}$ (وهذا يربطه
بالكمون الكهرسكوني في \cref{ch:b1:potential-capacitors}).
\end{definition}

\begin{definition}[النظام شبه المستقر]\label{def:b1:dc-circuits:arqs}
\index{نظام شبه مستقر}\index{التقريب شبه المستقر}
تكون دارة حجمها $\ell$ مقادة بتواتر $f$ في
\emph{النظام شبه المستقر} حين يكون زمن انتشار $\ell/c$
الإشارات الكهربائية على امتدادها مهملًا أمام الدور $1/f$:
أي $\ell \ll c/f$. وعندئذٍ يكون التيار نفسه في كل نقطة من
سلك غير متفرع في كل لحظة، وتنطبق القوانين أدناه في كل
لحظة على التيارات والتوترات المتغيرة مع الزمن ($i$ و $u$ بالحرف الصغير)
كما تنطبق على المستقرة ($I$ و $U$).
\end{definition}

\begin{example}[كم هو ``شبه'' هذا النظام]\label{ex:b1:dc-circuits:arqs}
عند \qty{50}{Hz} يكون $c/f = \qty{6000}{km}$: فالمنزل في \omterm{def:b1:dc-circuits:arqs}{النظام
شبه المستقر}. وعند \qty{1}{MHz} يكون \qty{300}{m}: فالراديو كذلك أيضًا. وعند \qty{2}{GHz}
يكون \qty{15}{cm}: فلوحة دارة هاتف ليست كذلك، ويجب أن تُعالج مساراتها
معالجة خطوط النقل --- وهو موضوع من مواضيع مجلد السنة الثانية.
\end{example}

\begin{theorem}[قانونا كيرشوف]\label{thm:b1:dc-circuits:kirchhoff}
\index{قانونا كيرشوف}\index{قانون العقد}\index{قانون الحلقات}
في \omterm{def:b1:dc-circuits:arqs}{النظام شبه المستقر}:
\begin{enumerate}
  \item \emph{قانون العقد}: عند عقدة تلتقي فيها عدة أسلاك، يساوي مجموع
        التيارات الواردة مجموعَ التيارات الصادرة،
        $\sum_{\text{وارد}} i_k = \sum_{\text{صادر}} i_k$.
  \item \emph{قانون الحلقات}: على امتداد أيّ حلقة مغلقة، يكون مجموع التوترات
        معدومًا إذا حُسبت بالإشارة التي يعطيها اتجاه سير
        مختار: $\sum_{\text{الحلقة}} \pm u_k = 0$.
\end{enumerate}
\end{theorem}

\begin{proof}
الشحنة محفوظة، وهي في \omterm{def:b1:dc-circuits:arqs}{النظام شبه المستقر} لا تتراكم
عند عقدة: فما يصل في الثانية يغادر في الثانية. أما قانون الحلقات
فهو العبارة القائلة إن الكمون دالة للموضع: فبالدوران
حول حلقة والعودة إلى نقطة البدء يكون مجموع هبوطات
الكمون $V_A - V_A = 0$.
\end{proof}

\begin{notation}[الاصطلاحات والاستطاعة]\label{not:b1:dc-circuits:conventions}
\index{اصطلاح المستقبِل}\index{اصطلاح المولّد}\index{استطاعة كهربائية}
من أجل عنصر ذي طرفين (أي \emph{ثنائي قطب})، يرسم \emph{\omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح
المستقبِل}} سهم \omterm{def:b1:dc-circuits:voltage}{التوتر} وسهم التيار في اتجاهين
متعاكسين؛ ويرسمهما \emph{\omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح المولّد}} في الاتجاه
نفسه. وفي \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح المستقبِل} تكون الاستطاعة \emph{المستقبَلة} في
ثنائي القطب
\[
p = u\,i \qquad (\text{بالواط}),
\]
وهي موجبة \omterm{def:b1:dc-circuits:resistor}{لناقلة أومية} تسخن، وسالبة لبطارية
تعطي. وفي \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح المولّد} يكون $p = ui$ هو الاستطاعة
\emph{المقدَّمة}.
\end{notation}

\section{ثنائيات القطب ونماذجها}

\begin{definition}[المميِّزة؛ الناقلة الأومية؛ قانون أوم]\label{def:b1:dc-circuits:resistor}
\index{مميِّزة (ثنائي قطب)}\index{ناقلة أومية}\index{قانون أوم}\index{مفعول جول}
\emph{مميِّزة} ثنائي القطب هي المنحني $i(u)$ (أو $u(i)$)
الذي يفرضه. و\emph{الناقلة الأومية} لها مميِّزة خطية تمرّ بالمبدأ،
وهي \emph{قانون أوم}
\[
u = R\,i \qquad (\text{اصطلاح المستقبِل}),
\]
حيث $R$ هي \emph{مقاومتها} (بالأوم، $\qty{1}{\ohm} = \qty{1}{V/A}$) و
$G = 1/R$ هي \emph{مواصلتها} (بالسيمنس). وتستقبل $p = Ri^2 =
u^2/R \geq 0$ تتبدد حرارةً (وهو \emph{مفعول جول}). وسلك
طوله $\ell$ ومقطعه $S$ ومقاوميته $\rho$ له
$R = \rho\ell/S$.
\end{definition}

\begin{definition}[المنابع المثالية والحقيقية]\label{def:b1:dc-circuits:sources}
\index{منبع توتر}\index{منبع تيار}\index{قوة محركة كهربائية}\index{مقاومة داخلية}\index{نموذج تيفنان}\index{نموذج نورتن}
\emph{منبع \omterm{def:b1:dc-circuits:voltage}{التوتر} المثالي} يفرض $u = E$ مهما يكن التيار؛
و$E$ هي \emph{قوته المحركة الكهربائية}. و\emph{منبع التيار
المثالي} يفرض $i = I_N$ مهما يكن \omterm{def:b1:dc-circuits:voltage}{التوتر}. أما \emph{المنبع الحقيقي}
(بطارية أو مولّد أو مغذٍّ) فيُنمذج، في \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح
المولّد}، بالعلاقة
\[
u = E - r\,i \quad (\text{نموذج تيفنان: قوة محركة مثالية } E \text{ على التسلسل مع } r)
\]
أو، على نحو مكافئ، $i = I_N - u/r$ (نموذج نورتن: منبع تيار
مثالي $I_N = E/r$ على التفرع مع $r$)؛ و$r$ هي \emph{المقاومة
الداخلية}. و$E$ هي \omterm{def:b1:dc-circuits:voltage}{توتر} الدارة المفتوحة، و$I_N = E/r$ تيار
القصر.
\end{definition}

\begin{proposition}[الحصيلة الاستطاعية لمنبع حقيقي]\label{prop:b1:dc-circuits:sourcepower}
\index{النقل الأعظمي للاستطاعة}
منبع حقيقي $(E, r)$ يغذّي \omterm{def:b1:dc-circuits:resistor}{ناقلة أومية} $R$ يعطي $i = E/(R + r)$
والاستطاعة
\[
P_R = \frac{E^2 R}{(R + r)^2},
\]
وهي عظمى وتساوي $E^2/4r$ حين $R = r$ (وهو \emph{التكييف})،
بمردود $P_R/(Ei) = R/(R + r)$، لا يتجاوز $50\%$ عند العظمى.
\end{proposition}

\begin{proof}
قانون الحلقات: $E = ri + Ri$. و$P_R = Ri^2$؛ وباشتقاق $R/(R + r)^2$:
$\big[(R + r)^2 - 2R(R + r)\big]/(R + r)^4 = (r - R)/(R + r)^3$، وهو معدوم عند
$R = r$، وموجب قبله وسالب بعده. والمنبع يقدّم
$Ei = (R + r)i^2$، منها $ri^2$ تسخّن المنبع نفسه.
\end{proof}

\begin{example}[بطارية سيارة تحت الحمل]\label{ex:b1:dc-circuits:battery}
الدارة المفتوحة \qty{12.6}{V}؛ و\qty{11.4}{V} بينما يسحب المقلع
\qty{120}{A}: ومنه $r = 1.2/120 = \qty{10}{m\ohm}$ و $I_N = \qty{1260}{A}$
(ومن هنا خطر مفتاح ربط ساقط، وهو خطر بمستوى اللحام). والحمولة المكيَّفة
$R = \qty{10}{m\ohm}$ كانت ستأخذ $E^2/4r = \qty{4}{kW}$ بضع
ثوانٍ --- وهو نحو ما يحتاجه محرك المقلع.
\end{example}

\begin{remark}[ثنائيات قطب أخرى]\label{rem:b1:dc-circuits:others}
\index{ثنائي}
\emph{الثنائي} لا ينقل التيار إلا في اتجاه واحد: ونموذجه المثالي
دارة مفتوحة من أجل $u < U_0$ \omterm{def:b1:dc-circuits:voltage}{وتوتر} ثابت $u = U_0$ (نحو
\qty{0.7}{V} في السيليسيوم، ومن \qty{2}{} إلى \qty{3}{V} في LED) حين
ينقل. والمصباح ذو الشعيرة \omterm{def:b1:dc-circuits:resistor}{ناقلة أومية} تنمو $R$ فيها مع
درجة حرارتها، ومن ثمّ مع $i$ --- فمميِّزتها منحنية. وفي النظام
المستقر يكون المكثف دارةً مفتوحة ($i = C\,\dd u/\dd t = 0$) وتكون
الوشيعة دارةً قصيرة ($u = L\,\dd i/\dd t = 0$)؛ ودورهما في
الأنظمة العابرة هو \cref{ch:b1:transient-regimes}.
\end{remark}

\section{التجميعات والمجزّئات}

\begin{proposition}[الناقلات على التسلسل وعلى التفرع]\label{prop:b1:dc-circuits:seriesparallel}
\index{تسلسل (ناقلات)}\index{تفرع (ناقلات)}
الناقلات على التسلسل (بالتيار نفسه) تُجمع مقاوماتها: $R = R_1 + R_2 + \cdots$.
والناقلات على التفرع (\omterm{def:b1:dc-circuits:voltage}{بالتوتر} نفسه) تُجمع مواصلاتها:
$1/R = 1/R_1 + 1/R_2 + \cdots$؛ ومن أجل اثنتين، $R = R_1R_2/(R_1 + R_2)$،
ويُكتب $R_1 \parallel R_2$.
\end{proposition}

\begin{proof}
على التسلسل: $u = u_1 + u_2 = (R_1 + R_2)i$ (بقانون الحلقات). وعلى التفرع: $i = i_1 +
i_2 = u/R_1 + u/R_2$ (بقانون العقد).
\end{proof}

\begin{proposition}[مجزّئا التوتر والتيار]\label{prop:b1:dc-circuits:dividers}
\index{مجزّئ التوتر}\index{مجزّئ التيار}
ناقلتان على التسلسل تحت \omterm{def:b1:dc-circuits:voltage}{توتر} $u$ تقتسمانه بنسبة
مقاومتيهما:
\[
u_2 = \frac{R_2}{R_1 + R_2}\,u .
\]
وناقلتان على التفرع يغذّيهما تيار $i$ تقتسمانه بنسبة
عكسية:
\[
i_2 = \frac{R_1}{R_1 + R_2}\,i = \frac{G_2}{G_1 + G_2}\,i .
\]
\end{proposition}

\begin{proof}
على التسلسل: $i = u/(R_1 + R_2)$ و $u_2 = R_2 i$. وعلى التفرع: $u = (R_1
\parallel R_2)\,i$ و $i_2 = u/R_2$.
\end{proof}

\begin{remark}[تحميل مجزّئ]\label{rem:b1:dc-circuits:loading}
لا تصحّ صيغة المجزّئ إلا إذا لم يسحب شيء تيارًا من
النقطة الوسطى. فحمولة $R_L$ موصولة على طرفَي $R_2$ تستبدل بالمقاومة $R_2$ المجموعةَ $R_2
\parallel R_L$ وتخفض $u_2$ --- وهو ``مفعول التحميل''، والسبب
في وجوب أن تكون مقاومة الفولطمتر عالية ومقاومة المجزّئ الجهدي
منخفضة أمام ما يغذّيه
(\cref{exo:b1:dc-circuits:4}).
\end{remark}

\begin{method}[اختزال شبكة]\label{met:b1:dc-circuits:reduce}
لإيجاد تيار واحد أو \omterm{def:b1:dc-circuits:voltage}{توتر} واحد في شبكة من الناقلات الأومية ومنبع
واحد:
\begin{enumerate}
  \item ابدأ بعيدًا عن المنبع واستبدل بالمجموعات المتسلسلة
        والمتفرعة مكافئاتِها حتى تواجه المنبعَ ناقلةٌ
        واحدة؛
  \item احسب تيار المنبع؛
  \item ثم ارجع أدراجك، مجزّئًا التيارات بمجزّئات التيار
        والتوترات بمجزّئات \omterm{def:b1:dc-circuits:voltage}{التوتر}، حتى الفرع المطلوب.
\end{enumerate}
ومع عدة منابع، أو من أجل فرع تتغير جيرانه، تُستعمل
مبرهنات القسم التالي.
\end{method}

\section{مبرهنات الدارات الخطية}

\begin{definition}[ثنائي القطب الخطي والدارة الخطية]\label{def:b1:dc-circuits:linear}
\index{ثنائي قطب خطي}
يكون ثنائي القطب \emph{خطيًّا} إذا كانت مميِّزته مستقيمًا
(\omterm{def:b1:dc-circuits:resistor}{كالناقلة الأومية} والمنابع المثالية والحقيقية)؛ والدارة المؤلَّفة من ثنائيات قطب
خطية فقط \emph{دارة خطية}. ومجاهيلها تحقق جملة معادلات
خطية (قانونا كيرشوف مع المميِّزات)، وحلّها
وحيد.
\end{definition}

\begin{theorem}[تيفنان ونورتن]\label{thm:b1:dc-circuits:thevenin}
\index{مبرهنة تيفنان}\index{مبرهنة نورتن}
منظورًا إليها من طرفين $A$ و $B$، تكافئ كل \omterm{def:b1:dc-circuits:linear}{دارة خطية}
منبعًا حقيقيًّا: أي قوة محركة $E_{\mathrm{Th}}$ على التسلسل مع
مقاومة $R_{\mathrm{Th}}$ (تيفنان)، أو \omterm{def:b1:dc-circuits:sources}{منبع تيار} $I_N =
E_{\mathrm{Th}}/R_{\mathrm{Th}}$ على التفرع مع $R_{\mathrm{Th}}$
(نورتن)، حيث
\begin{itemize}
  \item $E_{\mathrm{Th}}$ هي \omterm{def:b1:dc-circuits:voltage}{توتر} الدارة المفتوحة $u_{AB}$؛
  \item $I_N$ هو تيار القصر من $A$ إلى $B$؛
  \item $R_{\mathrm{Th}}$ هي المقاومة المرئية بين $A$ و $B$
        حين يُطفأ كل منبع مستقل (فتُستبدل بمنابع \omterm{def:b1:dc-circuits:voltage}{التوتر}
        المثالية أسلاكٌ، وبمنابع التيار المثالية دارات
        مفتوحة).
\end{itemize}
\end{theorem}

\begin{proof}[برهان جزئي]
بالخطية تكون العلاقة بين \omterm{def:b1:dc-circuits:voltage}{التوتر} $u_{AB}$ والتيار
$i$ المسحوب عند الطرفين علاقةً تآلفية: $u_{AB} = a - b\,i$. وبوضع
$i = 0$ نعرف $a = E_{\mathrm{Th}}$؛ وبوضع $u_{AB} = 0$ نجد
$b = E_{\mathrm{Th}}/I_N$. وكون $b$ يساوي المقاومة المرئية عند إطفاء
المنابع يتبع من التراكب (أدناه): فالجزء من $u_{AB}$
العائد إلى التيار الخارجي وحده هو ما كانت ستنتجه شبكة سلبية
مقاومتها $R_{\mathrm{Th}}$. أما أن لكل جملة خطية علاقةً
تآلفية بين الدخل والخرج فذلك جبر المعادلات الخطية.
\end{proof}

\begin{theorem}[التراكب]\label{thm:b1:dc-circuits:superposition}
\index{مبرهنة التراكب}
في \omterm{def:b1:dc-circuits:linear}{دارة خطية} فيها عدة منابع مستقلة، يكون أيّ تيار أو
\omterm{def:b1:dc-circuits:voltage}{توتر} مجموعَ القيم التي يأخذها حين يعمل كل منبع وحده،
والبقية مطفأة.
\end{theorem}

\begin{proof}
المعادلات خطية في المجاهيل، والمنابع في طرفها
الأيمن؛ والحلّ دالة خطية لذلك الطرف
الأيمن، ومن ثمّ فهو مجموع الحلول من أجل كل منبع على حدة.
\end{proof}

\begin{proposition}[مبرهنة ميلمان]\label{prop:b1:dc-circuits:millman}
\index{مبرهنة ميلمان}
عقدة $N$ موصولة بنقاط كموناتها $V_1, \dots, V_n$ عبر
مقاومات $R_1, \dots, R_n$ (ولا شيء غير ذلك) كمونها
\[
V_N = \frac{\sum_k V_k/R_k}{\sum_k 1/R_k} = \frac{\sum_k G_k V_k}{\sum_k G_k},
\]
وهو المتوسط الموزون بالمواصلات لكمونات جيرانها.
\end{proposition}

\begin{proof}
قانون العقد عند $N$: $\sum_k (V_k - V_N)/R_k = 0$؛ ثم نحلّ بالنسبة إلى $V_N$.
\end{proof}

\begin{example}[ثلاث طرق إلى الجواب نفسه]\label{ex:b1:dc-circuits:threeways}
في الشكل، $E_1 = \qty{10}{V}$ و $R_1 = \qty{1.0}{k\ohm}$ و $E_2 =
\qty{5.0}{V}$ و $R_2 = \qty{2.0}{k\ohm}$ و $R_3 = \qty{2.0}{k\ohm}$.
\emph{بميلمان}: $V_N = (10/1 + 5/2 + 0/2)/(1 + 0.5 + 0.5) = 12.5/2 =
\qty{6.25}{V}$. و\emph{بالتراكب}: يرى $E_1$ وحده $R_2 \parallel
R_3 = \qty{1}{k\ohm}$، ومنه $u = 10 \times 1/2 = \qty{5}{V}$؛ ويرى $E_2$ وحده
$R_1 \parallel R_3 = \qty{0.667}{k\ohm}$، ومنه $u = 5 \times 0.667/2.667
= \qty{1.25}{V}$؛ والمجموع \qty{6.25}{V}. و\emph{بتيفنان} منظورًا من
$R_3$: تكون $E_{\mathrm{Th}}$ توترَ الدارة المفتوحة للمجزّئ
$(E_1, R_1, R_2, E_2)$: أي $E_2 + (E_1 - E_2)R_2/(R_1 + R_2) = 5 + 5 \times
2/3 = \qty{8.33}{V}$؛ و$R_{\mathrm{Th}} = R_1 \parallel R_2 =
\qty{0.667}{k\ohm}$؛ ومنه $u = 8.33 \times 2/2.667 = \qty{6.25}{V}$.
\end{example}

\begin{proposition}[جسر ويتستون]\label{prop:b1:dc-circuits:wheatstone}
\index{جسر ويتستون}
أربع ناقلات أومية $R_1, R_2$ (في فرع) و $R_3, R_4$ (في الآخر) تحت
منبع $E$؛ وخرج الجسر هو \omterm{def:b1:dc-circuits:voltage}{التوتر} بين النقطتين
الوسطيين:
\[
u = E\left(\frac{R_2}{R_1 + R_2} - \frac{R_4}{R_3 + R_4}\right),
\]
وهو معدوم (فيكون الجسر \emph{متوازنًا}) إذا وفقط إذا كان $R_1R_4 = R_2R_3$. ومن أجل
جسر متوازن مع اضطراب $R_4$ إلى $R_4(1 + \epsilon)$، حيث $\epsilon
\ll 1$، و $R_3 = R_4$: يكون $u \approx -E\epsilon/4$.
\end{proposition}

\begin{proof}
مجزّئا \omterm{def:b1:dc-circuits:voltage}{توتر}؛ ثم نطرح. والتوازن: $R_2(R_3 + R_4) = R_4(R_1 +
R_2)$، أي $R_2R_3 = R_1R_4$. ومع $R_3 = R_4 = R$ و $R_1 = R_2$:
$u = E[\tfrac12 - (1 + \epsilon)/(2 + \epsilon)] = E[(2 + \epsilon - 2 -
2\epsilon)/(2(2 + \epsilon))] \approx -E\epsilon/4$.
\end{proof}

\section{مسألة: الشبكة الكهربائية في سيارة}

\begin{problem}[{مسألة نهاية الأسبوع --- بطارية واحدة ومقلع ومصباحان
أماميان ومولّد متناوب وبضعة أمتار من النحاس: من يأخذ
التيار، ومن يدفع، ولماذا تخفت الأضواء حين يدور المحرك}]
\label{pb:b1:dc-circuits:1}
البطارية منبع حقيقي، قوته المحركة $E = \qty{12.6}{V}$ ومقاومته
الداخلية $r = \qty{10}{m\ohm}$. والنحاس: $\rho = \qty{1.7e-8}{\ohm.m}$.

\medskip
\noindent\textbf{الجزء I --- البطارية وحدها.}
\begin{enumerate}
  \item ارسم \omterm{def:b1:dc-circuits:sources}{نموذج تيفنان} واكتب $u(i)$ في \omterm{thm:b1:dc-circuits:kirchhoff}{اصطلاح
        المولّد}.
  \item أعطِ \omterm{def:b1:dc-circuits:voltage}{توتر} الدارة المفتوحة وتيار القصر.
        ولماذا يجب ألّا يصل مفتاح ربط ساقط بين الطرفين أبدًا؟
  \item اكتب \omterm{def:b1:dc-circuits:sources}{نموذج نورتن} للبطارية نفسها.
  \item تغذّي البطارية \omterm{def:b1:dc-circuits:resistor}{ناقلة أومية} $R$: عبّر عن التيار \omterm{def:b1:dc-circuits:voltage}{وتوتر}
        الطرفين والاستطاعة $P_R$ المقدَّمة والاستطاعة الضائعة
        في $r$.
  \item بيّن أن $P_R$ عظمى من أجل $R = r$ واحسب تلك
        القيمة العظمى. وما المردود عندئذٍ؟
  \item يقيس فنّيّ $u = \qty{12.48}{V}$ عند \qty{12}{A} و
        $u = \qty{11.40}{V}$ عند \qty{120}{A}. تحقق من أن النقطتين تقعان
        على النموذج وفسّر كيف تعطي سلسلة من نقاط كهذه $E$
        و $r$ بملاءمة مستقيمة.
  \item تخزّن البطارية \qty{60}{A.h}. فكم من الوقت تستطيع تشغيل
        المصباحين الأماميين \qty{55}{W} وحدهما (خذ \qty{12}{V})؟ وكم تبلغ
        تلك الطاقة بالجول؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الإقلاع والأضواء مشتعلة.}
المقلع \omterm{def:b1:dc-circuits:resistor}{ناقلة أومية} $R_s = \qty{80}{m\ohm}$؛ وكل مصباح أمامي
مقنَّن \qty{55}{W} عند \qty{12.0}{V} ويُعالَج معالجة \omterm{def:b1:dc-circuits:resistor}{ناقلة أومية}.
\begin{enumerate}[resume]
  \item احسب مقاومة مصباح واحد ومقاومة الاثنين على
        التفرع.
  \item المقلع وحده: التيار \omterm{def:b1:dc-circuits:voltage}{وتوتر} الطرفين والاستطاعة في المقلع،
        والاستطاعة الضائعة في البطارية.
  \item المقلع والمصباحان معًا: مقاومة الحمولة
        المكافئة والتيار الكلي \omterm{def:b1:dc-circuits:voltage}{وتوتر} الطرفين.
  \item استعمل \omterm{prop:b1:dc-circuits:dividers}{مجزّئ التيار} لإيجاد تيار المقلع
        وتيار المصباحين.
  \item احسب الاستطاعة التي يتلقاها كل مصباح الآن وقارنها
        بتقنينه: فبأيّ كسر تخفت الأضواء؟
  \item فسّر في جملتين، بقانون الحلقات، لماذا تخفت الأضواء
        بالضبط لأن المقلع يسحب تيارًا كبيرًا.
  \item وهل تنفع كبلات بطارية أغلظ الأضواءَ؟ برّر
        الآن نوعيًّا (والجزء IV يكمّم ذلك).
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- المحرك يدور: المولّد المتناوب.}
المولّد المتناوب منبع حقيقي ثانٍ، $E_a = \qty{14.2}{V}$ و
$r_a = \qty{50}{m\ohm}$، على التفرع مع البطارية؛ والمقلع
مطفأ والمصباحان مشتعلان.
\begin{enumerate}[resume]
  \item ارسم الدارة: منبعان حقيقيان وحمولة المصباحين على
        التفرع بين العقدتين نفسيهما.
  \item طبّق مبرهنة ميلمان لإيجاد \omterm{def:b1:dc-circuits:voltage}{توتر} الطرفين المشترك.
  \item استنتج تيار المولّد وتيار البطارية (بإشارته!) و
        تيار المصباحين؛ وتحقق من قانون العقد.
  \item هل تُشحن البطارية أم تُفرَّغ؟ وأيّ استطاعة تتلقى،
        وأيّ استطاعة يقدّم المولّد؟
  \item استعد \omterm{def:b1:dc-circuits:voltage}{توتر} الطرفين بالتراكب (كل منبع
        وحده والآخر مستبدَل بمقاومته الداخلية) ثم
        تحقق.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- المحسّات والأسلاك.}
\begin{enumerate}[resume]
  \item محسّ سائل التبريد مقاومة حرارية تهبط مقاومتها من
        \qty{2.0}{k\ohm} عند \qty{20}{\degreeCelsius} إلى \qty{300}{\ohm} عند
        \qty{90}{\degreeCelsius}؛ وهي الناقلة السفلى في مجزّئ
        يغذّيه \qty{5.0}{V}، والناقلة العليا $R_0$. عبّر عن
        \omterm{def:b1:dc-circuits:voltage}{توتر} الخرج، واحسبه عند درجتَي الحرارة من أجل
        $R_0 = \qty{775}{\ohm}$.
  \item بيّن أن مدى الخرج بين درجتَي الحرارة يكون
        أكبر ما يكون حين تكون $R_0$ الوسط الهندسي لقيمتَي
        المحسّ، وتحقق من القيمة أعلاه.
  \item يوضع المحسّ نفسه في \omterm{prop:b1:dc-circuits:wheatstone}{جسر ويتستون} مع ثلاث
        ناقلات \qty{775}{\ohm}: اكتب خرج الجسر وقل عند
        أيّ درجة حرارة يتوازن.
  \item كبل المقلع طوله \qty{1.5}{m} في كل اتجاه. احسب
        مقاومة ناقل واحد مقطعه \qty{16}{mm^2}، و
        هبوط \omterm{def:b1:dc-circuits:voltage}{التوتر} الكلي عند \qty{140}{A}، والاستطاعة المتبددة في
        الكبلين.
  \item والمثل مع كبل \qty{4}{mm^2}: فلماذا يكون كبل المقلع بهذه الغلاظة؟
  \item لخّص: أعطِ \omterm{def:b1:dc-circuits:voltage}{توتر} الطرفين أثناء الإقلاع، و
        كسر الاستطاعة الذي يبلغ المقلع، والقانون الوحيد الذي
        يفسّر خفوت الأضواء.
\end{enumerate}
\end{problem}
