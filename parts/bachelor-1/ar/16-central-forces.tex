\chapter{القوى المركزية: الكواكب والأقمار الصناعية}\label{ch:b1:central-forces}

يدور واحد وثلاثون قمرًا صناعيًّا حول الأرض مرتين في اليوم، كلٌّ منها ساعة
ومذياع، وهاتف يسمع أربعة منها يعرف أين هو بدقة
بضعة أمتار. ويسقط مذنّب هالي نحو الشمس ثمانيًا وثلاثين
سنة، فيدور حولها في بضعة أسابيع، ثم يصعد مبتعدًا
ثمانيًا وثلاثين سنة أخرى. ويغادر مسبار متجه إلى المريخ الأرضَ بالسرعة
التي تجعل مساره يلامس مدار المريخ ملامسةً بالكاد. وكل هذا
حركة جسم يُجذب نحو نقطة ثابتة بقوة تتناقص
كمقلوب مربع البعد، وكله ينتج
من مقدارين منحفظين --- الطاقة \omterm{def:b1:angular-momentum:L}{والعزم الحركي} --- مضافًا إليهما
فكرة جديدة واحدة هي \omterm{def:b1:central-forces:central}{الطاقة الكامنة الفعّالة}. ويشتقّ هذا الفصل
أشكال المدارات وسرعاتها، \omterm{thm:b1:central-forces:conics}{وقوانين كبلر} الثلاثة، وحساب
وضع قمر صناعي حيث يُراد.

\section{القوى المركزية المحافظة}

\begin{definition}[قوة مركزية؛ قوة مركزية محافظة]\label{def:b1:central-forces:central}
\index{قوة مركزية}\index{طاقة كامنة فعّالة}
تكون قوة \emph{مركزية} مركزها $O$ إذا كانت موجَّهة دائمًا
على امتداد $\vect{OM}$: $\vect F = F(M)\,\vect e_r$. وتكون \emph{مركزية
محافظة} إذا لم يتوقف مقدارها إلا على $r = OM$: $\vect F =
F(r)\,\vect e_r$، وهي تشتقّ عندئذ من \omterm{def:b1:work-and-energy:conservative}{طاقة كامنة} $E_p(r)$ حيث
$F(r) = -\dd E_p/\dd r$.
\end{definition}

\begin{theorem}[الحركة تحت قوة مركزية محافظة]\label{thm:b1:central-forces:reduction}
\index{قانون المساحات}
جسيم كتلته $m$ لا يخضع إلا إلى قوة \omterm{def:b1:central-forces:central}{مركزية محافظة}:
\begin{enumerate}
  \item يحفظ عزمه الحركي $\vect L_O$؛ فحركته مستوية،
        وبالإحداثيات القطبية في ذلك المستوي يكون $r^2\dot\theta = C$
        ثابتًا (وهو قانون المساحات، \cref{cor:b1:angular-momentum:central})؛
  \item ويحفظ طاقته الميكانيكية $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$، التي يمكن كتابتها
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
فالحركة القطرية إذن حركة جسيم في بُعد واحد في
\emph{\omterm{def:b1:central-forces:central}{الطاقة الكامنة الفعّالة}} $E_{p,\mathrm{eff}}$: \omterm{def:b1:work-and-energy:turning}{فنقاط الرجوع}
القطرية هي جذور $E_{p,\mathrm{eff}}(r) = E_m$، وتكون \omterm{def:b1:work-and-energy:turning}{الحركة
مرتبطة} إذا حُبست بين نقطتين منها، ودائرية إذا استقر $r$ عند
نهاية صغرى للمقدار $E_{p,\mathrm{eff}}$.
\end{theorem}

\begin{proof}
الجزء الأول هو \cref{cor:b1:angular-momentum:central}. \omterm{def:b1:work-and-energy:conservative}{والقوة
محافظة} ولا تؤثر قوة أخرى، فتنحفظ $E_m$؛ وفي
المستوي، $v^2 = \dot r^2 + r^2\dot\theta^2$ و $r^2\dot\theta^2 = C^2/r^2$.
أما الحدّ $mC^2/2r^2$، وهو \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} للحركة الزاوية، فيعمل
عمل \emph{\omterm{prop:b1:central-forces:effective}{حاجز طارد مركزي}} في المسألة القطرية، وينطبق عليه
تحليل البُعد الواحد في \cref{ch:b1:work-and-energy}.
\end{proof}

\section{جاذبية نيوتن}

\begin{definition}[الجاذبية الكونية]\label{def:b1:central-forces:gravitation}
\index{جاذبية}\index{طاقة كامنة ثقالية}
كتلة نقطية $M$ في $O$ تجذب كتلة نقطية $m$ في $M$ بالقوة
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ عند اللانهاية}),
\]
حيث $G = \qty{6.67e-11}{N.m^2/kg^2}$. وجسم متناظر كرويًّا يجذب
خارج نفسه كأنّ كتلته كلها في مركزه (وهذا مقبول؛ وهو
نتيجة لمبرهنة غاوس، \cref{ch:b1:electrostatics-gauss}).
\end{definition}

\begin{proposition}[الطاقة الكامنة الفعّالة للجاذبية]\label{prop:b1:central-forces:effective}
\index{حاجز طارد مركزي}\index{مدار دائري}
من أجل $E_p = -GMm/r$،
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
تؤول إلى $+\infty$ عند $r \to 0$ (من أجل $L \neq 0$)، وإلى $0^-$ عند اللانهاية،
ولها نهاية صغرى وحيدة عند $r_c = L^2/GMm^2$ قيمتها $-G^2M^2m^3/2L^2$.
ومن هنا: $E_m < 0$ --- \omterm{def:b1:work-and-energy:turning}{حركة مرتبطة} بين نصف قطر أصغري وآخر
أعظمي (الحضيض والأوج)؛ و $E_m \geq 0$ --- حركة غير مرتبطة، فيأتي الجسم
من اللانهاية ويعود إليها؛ و $E_m$ مساوية للنهاية الصغرى --- \omterm{prop:b1:central-forces:effective}{مدار
دائري} نصف قطره $r_c$.
\end{proposition}

\begin{proof}
النهايات بالمعاينة؛ و $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$ عند
$r_c$؛ ثم عوّض. والحالة الدائرية هي $\dot r = 0$ دائمًا، وهي ممكنة
عند النهاية الصغرى فقط.
\end{proof}

\begin{theorem}[المدارات الدائرية]\label{thm:b1:central-forces:circular}
\index{سرعة مدارية}\index{قانون كبلر الثالث}
على \omterm{prop:b1:central-forces:effective}{مدار دائري} نصف قطره $r$ حول كتلة $M$:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
وبخاصة $T^2/r^3 = 4\pi^2/GM$ هو نفسه من أجل كل المدارات الدائرية
حول المركز نفسه (وهو قانون كبلر الثالث).
\end{theorem}

\begin{proof}
تحتاج \omterm{cor:b1:point-kinematics:circular}{الحركة الدائرية} المنتظمة إلى قوة جاذبة مركزية $mv^2/r = GMm/r^2$؛
و $T = 2\pi r/v$؛ و $E_k = \tfrac12 mv^2 = GMm/2r$ و $E_p = -GMm/r$.
\end{proof}

\begin{example}[ثلاثة مدارات]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$. المدار المنخفض ($r = \qty{6770}{km}$،
أي على ارتفاع \qty{400}{km}): $v = \qty{7.7}{km/s}$ و $T = \qty{92}{min}$. \omterm{prop:b1:central-forces:escape}{والمدار الثابت
بالنسبة إلى الأرض} ($T = \qty{86164}{s}$، أي يوم نجمي واحد): $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$ و $v = \qty{3.1}{km/s}$. والقمر ($r = \qty{384000}{km}$):
$T = \qty{27.4}{d}$ --- وتسارعه الجاذب المركزي $v^2/r =
\qty{2.7e-3}{m/s^2}$ هو $g/3600$ عند ستين نصف قطر أرضي: أي التربيع
العكسي، وقد تحقق منه نيوتن نفسه.
\end{example}

\section{قوانين كبلر وشكل المدارات}

\begin{theorem}[المسارات في مجال نيوتني]\label{thm:b1:central-forces:conics}
\index{قوانين كبلر}\index{مقطع مخروطي}\index{إهليلج (مدار)}\index{اختلاف مركزي}
مسار جسم في مجال كتلة ثابتة $M$ هو \omterm{thm:b1:central-forces:conics}{مقطع مخروطي}
له $O$ في إحدى بؤرتيه: فهو إهليلج إذا كان $E_m < 0$، وقطع مكافئ إذا كان $E_m = 0$،
وقطع زائد إذا كان $E_m > 0$. ومن أجل إهليلج نصف محوره الكبير $a$
واختلافه المركزي $e$:
\begin{itemize}
  \item (كبلر الأول) يقع مركز القوة في بؤرة؛ ويقع الحضيض والأوج
        عند $r_p = a(1 - e)$ و $r_a = a(1 + e)$؛
  \item (كبلر الثاني) \omterm{cor:b1:angular-momentum:central}{السرعة المساحية} ثابتة (وهو قانون المساحات)؛
  \item (كبلر الثالث) يحقق الدور $T^2 = 4\pi^2a^3/GM$؛
  \item وتتوقف الطاقة على $a$ وحده، $E_m = -GMm/2a$، بحيث تُعطى
        السرعة على البعد $r$ بالعلاقة
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[ما هو مبرهَن وما هو مقبول]\label{rem:b1:central-forces:admitted}
بُرهن أعلاه على كبلر الثاني وعلى التصنيف الطاقي. أما أن
المدارات المرتبطة إهليلجات بالضبط لها $O$ في بؤرة، مع كبلر الثالث
و $E_m = -GMm/2a$ في الحالة العامة، فيحتاج إلى مكاملة المعادلة
القطرية (ويفعل ذلك مجلد السنة الثانية)؛ ومن أجل المدارات الدائرية ($a = r$
و $e = 0$) تؤول كل عبارة إلى \cref{thm:b1:central-forces:circular}،
وهو التحقق الممكن هنا. أما العلاقة $v^2 = GM(2/r - 1/a)$
فتنتج من $E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$ متى سُلّم بصيغة
الطاقة تلك.
\end{remark}

\begin{example}[مذنّب هالي]\label{ex:b1:central-forces:halley}
الحضيض \qty{0.586}{AU} والأوج \qty{35.1}{AU}: ومنه $a = \qty{17.8}{AU}$
و $e = 0.967$ و $T = \sqrt{17.8^3} = \qty{75}{yr}$ (بكبلر الثالث
بالوحدات الفلكية والسنوات، حيث $T^2 = a^3$ للشمس). ومع
$GM_\odot = \qty{1.33e20}{m^3/s^2}$ تكون السرعة عند الحضيض
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ وعند الأوج \qty{0.9}{km/s}
--- أي أبطأ ستين مرة، كما يقتضي قانون المساحات ($r_pv_p = r_av_a$).
\end{example}

\section{الأقمار الصناعية}

\begin{proposition}[سرعة الإفلات؛ المدار الثابت بالنسبة إلى الأرض]\label{prop:b1:central-forces:escape}
\index{سرعة الإفلات}\index{مدار ثابت بالنسبة إلى الأرض}
انطلاقًا من سطح جسم كتلته $M$ ونصف قطره $R$، تكون سرعة
الإطلاق الدنيا لبلوغ اللانهاية $v_e = \sqrt{2GM/R} = \sqrt2\,v_c$،
حيث $v_c = \sqrt{GM/R}$ هي السرعة الدائرية عند مستوى الأرض:
أي \qty{11.2}{km/s} من أجل الأرض. أما القمر الصناعي \emph{الثابت بالنسبة إلى الأرض}
فيدور في المستوي الاستوائي بالدور النجمي للأرض، عند
$r = \qty{42200}{km}$ (أي \qty{35800}{km} فوق سطح الأرض).
\end{proposition}

\begin{proof}
$E_m \geq 0$: أي $\tfrac12 mv_e^2 - GMm/R = 0$. وكبلر الثالث مع $T = \qty{86164}{s}$.
\end{proof}

\begin{proposition}[انتقال هومان]\label{prop:b1:central-forces:hohmann}
\index{انتقال هومان}
لنقل قمر صناعي من \omterm{prop:b1:central-forces:effective}{مدار دائري} نصف قطره $r_1$ إلى \omterm{prop:b1:central-forces:effective}{مدار دائري}
في المستوي نفسه نصف قطره $r_2 > r_1$ بحرقين قصيرين، أشعِل عند $r_1$
لدخول \emph{إهليلج الانتقال} الذي حضيضه $r_1$ وأوجه $r_2$
(مع $a = (r_1 + r_2)/2$)، ثم انسَبْ نصف دور، ثم أشعِل ثانيةً عند $r_2$
لجعل المدار دائريًّا:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
السرعتان على إهليلج الانتقال من $v^2 = GM(2/r - 1/a)$ عند $r_1$ و
$r_2$؛ ثم اطرح السرعتين الدائريتين؛ ونصف الدور من
\cref{thm:b1:central-forces:conics}.
\end{proof}

\begin{example}[من المدار المنخفض إلى المدار الثابت]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$ و $r_2 = \qty{42200}{km}$: ومنه $\Delta v_1 = \qty{2.4}{km/s}$
و $\Delta v_2 = \qty{1.5}{km/s}$، وانسياب \qty{5.3}{h}. والمجموع،
\qty{3.9}{km/s}، هو نصف ما كلّفه الإطلاق إلى المدار المنخفض أصلًا:
و``الميل الأخير'' إلى مدار عالٍ مكلف، ولذلك تحمل الأقمار الصناعية محركها
الخاص من أجله.
\end{example}

\begin{remark}[المراكز الدافعة]\label{rem:b1:central-forces:rutherford}
\index{تبعثر رذرفورد}
من أجل قوة دافعة تتناسب عكسًا مع مربع البعد (كشحنتين متماثلتين،
\cref{ch:b1:electrostatics-gauss}) لا يكون \omterm{def:b1:central-forces:central}{للطاقة الكامنة الفعّالة} بئر:
فكل مسار قطع زائد، إذ يقترب الجسيم فينحرف
ثم يبتعد. وإذا صُوّب جسيم $\alpha$ مباشرةً نحو نواة ذهب، نتج
بعد أقرب اقتراب من الطاقة وحدها، $E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- وهي طريقة رذرفورد في حصر حجم
النواة (\cref{exo:b1:central-forces:11}).
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:central-forces:1}
تدور محطة الفضاء الدولية على ارتفاع \qty{400}{km}
($R_\oplus = \qty{6370}{km}$ و $GM_\oplus = \qty{3.99e14}{m^3/s^2}$). احسب السرعة
والدور وعدد مرات الشروق التي يراها طاقمها في اليوم.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:central-forces:2}
أوجد نصف قطر \omterm{prop:b1:central-forces:escape}{المدار الثابت بالنسبة إلى الأرض} وارتفاعه انطلاقًا من
اليوم النجمي \qty{86164}{s}، ثم أوجد \omterm{thm:b1:central-forces:circular}{السرعة المدارية}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:central-forces:3}
احسب \omterm{prop:b1:central-forces:escape}{سرعة الإفلات} من الأرض ومن القمر ($GM = \qty{4.90e12}{m^3/s^2}$ و
$R = \qty{1740}{km}$). ولماذا لا يحتفظ القمر بغلاف جوي؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:central-forces:4}
للمريخ $a = \qty{1.524}{AU}$، وللمشتري \qty{5.20}{AU}: احسب سنتيهما بقانون
كبلر الثالث. وماذا عن جسم على \qty{30}{AU} (نبتون)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:5}
احسب الطاقة اللازمة لوضع \qty{1000}{kg} على مدار المحطة الدولية انطلاقًا من
السكون عند السطح (بإهمال دوران الأرض والهواء). وقارنها
بالطاقة الكيميائية للبنزين (\qty{46}{MJ/kg}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:6}
مذنّب هالي: الحضيض \qty{0.586}{AU} والأوج \qty{35.1}{AU}.
احسب $a$ و $e$ والدور والسرعتين عند الحضيض والأوج
($GM_\odot = \qty{1.33e20}{m^3/s^2}$ و $\qty{1}{AU} = \qty{1.496e11}{m}$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:7}
\omterm{prop:b1:central-forces:hohmann}{انتقال هومان} من مدار المحطة الدولية (\qty{6770}{km}) إلى
\omterm{prop:b1:central-forces:escape}{المدار الثابت بالنسبة إلى الأرض} (\qty{42200}{km}): احسب الحرقين ومجموعهما والمدة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:8}
بيّن أن النهاية الصغرى للمقدار $E_{p,\mathrm{eff}}$ تقع عند $r_c = L^2/GMm^2$،
وأن الاهتزازات القطرية الصغيرة حولها لها \omterm{def:b1:wave-propagation:signal}{التواتر الزاوي}
$\sqrt{GM/r_c^3}$ --- أي التواتر المداري نفسه. فماذا يقول هذا
عن \omterm{prop:b1:central-forces:effective}{مدار دائري} اضطرب قليلًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:central-forces:9}
يطفو روّاد الفضاء على متن المحطة الدولية. فسّر ذلك بالقوى؛ وما
قيمة $g$ على ارتفاعهم، ولماذا لا يناقض ذلك الطفو؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:10}
``اختبار القمر'' عند نيوتن: احسب تسارع القمر الجاذب المركزي انطلاقًا من
$r = \qty{3.84e8}{m}$ و $T = \qty{27.3}{d}$، وقارنه بالمقدار
$g(R/r)^2$. ثم استنتج.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:11}
جسيم $\alpha$ ($q = 2e$ و $E_k = \qty{5.0}{MeV}$) يُطلق مباشرةً
نحو نواة ذهب ($Z = 79$) تُعتبر ثابتة. باستعمال انحفاظ الطاقة
مع $E_p = 2Ze^2/4\pi\varepsilon_0r$ (ومع $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$)،
أوجد بعد أقرب اقتراب؛ وقارنه بنصف القطر النووي،
ونحوه \qty{7}{fm}. وما الطاقة اللازمة لملامسة النواة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:central-forces:12}
يبطّئ الغلاف الجوي المتبقي قمرًا صناعيًّا في مدار منخفض، فتتناقص
طاقته الميكانيكية. بيّن بالعلاقتين $E_m = -GMm/2r$ و
$v = \sqrt{GM/r}$ أنه يتحرك عندئذ \emph{أسرع}. وإلى أين تذهب
الطاقة، وأيّ جزء من \omterm{def:b1:work-and-energy:conservative}{الطاقة الكامنة} الضائعة يصير حرارة؟
\end{exercise}

\section{مسألة: الأقمار التي تخبرك أين أنت}

\begin{problem}[{مسألة نهاية الأسبوع --- كوكبة ملاحية على ارتفاع عشرين
ألف كيلومتر: مدارها، والوقود اللازم لبلوغها، وكم قمرًا
يرى الهاتف، ولماذا يجب أن تُضبط ساعاتها لتسير
متأخرة}]
\label{pb:b1:central-forces:1}
المعطيات: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$ و $R_\oplus = \qty{6371}{km}$،
واليوم النجمي \qty{86164}{s}، و $c = \qty{2.998e8}{m/s}$. وتدور الأقمار
حول الأرض في نصف يوم نجمي بالضبط على مدارات دائرية.

\medskip
\noindent\textbf{الجزء I --- المدار.}
\begin{enumerate}
  \item احسب نصف قطر المدار والارتفاع.
  \item احسب \omterm{thm:b1:central-forces:circular}{السرعة المدارية}.
  \item احسب تسارع القمر الصناعي وقارنه بالمقدار $g$ عند
        السطح.
  \item احسب \omterm{thm:b1:work-and-energy:em}{الطاقة الميكانيكية} لكل كيلوغرام، وجزأيها الحركي
        والكامن.
  \item لماذا يُختار نصف يوم نجمي بدل نصف يوم شمسي؟
        وما الذي يراه مراقب أرضي يتكرر؟
  \item بأيّ زاوية يُرى قرص الأرض من قمر صناعي؟
  \item قارن نصف قطر هذا المدار بنصف قطر \omterm{prop:b1:central-forces:escape}{المدار الثابت بالنسبة إلى الأرض}:
        فأيّهما أعلى، وبأيّ عامل؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الوصول إلى هناك.}
\begin{enumerate}[resume]
  \item احسب الطاقة لكل كيلوغرام للانتقال من السكون عند السطح إلى المدار
        النهائي (بإهمال الهواء ودوران الأرض).
  \item يبلغ الصاروخ أولًا مدارًا دائريًّا منخفضًا على \qty{400}{km}.
        فما السرعة هناك؟
  \item \omterm{prop:b1:central-forces:hohmann}{انتقال هومان} من ذلك المدار إلى المدار النهائي: احسب الحرقين
        ومجموعهما.
  \item احسب مدة الانتقال.
  \item قارن مجموع الحرقين \omterm{prop:b1:central-forces:escape}{بسرعة الإفلات} من المدار
        المنخفض: فكم تبعد هذه المهمة عن ``مغادرة الأرض''؟
  \item كتلة المرحلة العليا مع القمر الصناعي \qty{2000}{kg} قبل
        الانتقال؛ وسرعة عادم المحرك \qty{3.0}{km/s}، ومعادلة
        الصاروخ تعطي $\Delta v = u\ln(m_0/m_1)$. فما الوقود الدافع
        اللازم للحرقين؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- التغطية والتوقيت.}
\begin{enumerate}[resume]
  \item يُرى قمر صناعي من كل نقطة من الأرض تقع
        داخل المخروط الذي نصف زاويته $\theta$ حيث $\cos\theta = R/r$
        حول مستقيم الأرض--القمر. احسب $\theta$ و
        النسبة $(1 - \cos\theta)/2$ من سطح الأرض التي يغطيها.
  \item مع $24$ قمرًا موزَّعة بانتظام، كم قمرًا يُرى في
        المتوسط من نقطة واحدة؟ وهل الحدّ الأدنى $4$ اللازم للتحديد
        معقول؟
  \item احسب زمن انتقال الضوء من قمر في السمت؛ ثم من قمر عند
        الأفق.
  \item لتحديد المستقبِل بدقة \qty{3}{m}، بأيّ دقة يجب توقيت
        الإشارات؟
  \item ساعة المستقبِل نفسه ساعة كوارتز رخيصة، تخطئ
        بميلي ثوانٍ: فلماذا تلزم أربعة أقمار بدل ثلاثة؟
  \item ساعات الأقمار ذرية. أعطِ الاستقرار النسبي
        اللازم من أجل خطأ \qty{3}{m} بعد يوم واحد.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- لمحة عن النسبية.}
يزيح أثران ساعات الأقمار بالنسبة إلى الساعات الأرضية
(والصيغتان مقبولتان، من مجلد السنة الثالثة): فالحركة تبطّئ الساعة بالنسبة
$v^2/2c^2$؛ والارتفاع يسرّعها بالمقدار $GM(1/R - 1/r)/c^2$.
\begin{enumerate}[resume]
  \item احسب التباطؤ الناجم عن السرعة، بالميكرو ثانية في اليوم.
  \item احسب التسارع الناجم عن الارتفاع، بالميكرو ثانية في
        اليوم.
  \item احسب الأثر المحصّل، وخطأ الموضع الذي كان سيسببه في اليوم لو
        لم يُصحَّح.
  \item تُضبط ساعات الأقمار قبل الإطلاق على
        \qty{10.22999999543}{MHz} بدل \qty{10.23}{MHz}. تحقق من أن
        هذا يعوّض الأثر المحصّل.
  \item ليست المدارات كبلرية بالضبط: سمِّ ثلاثة اضطرابات
        وقل كيف تتعامل المنظومة معها.
  \item لخّص السلسلة: قانونا انحفاظ، ومدار واحد، وكوكبة
        واحدة، والتصحيح الوحيد الذي ما كان أحد ليتوقعه.
\end{enumerate}
\end{problem}
