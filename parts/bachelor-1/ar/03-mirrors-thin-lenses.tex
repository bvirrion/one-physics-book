\section{الصور وشرطا غاوس}

\begin{definition}[الجسم والصورة والاستجماع]\label{def:b1:mirrors-thin-lenses:image}
\index{صورة (بصريات)}\index{جسم (بصريات)}\index{استجماع}\index{صورة حقيقية}\index{صورة تخيلية}
يتلقى النظام البصري أشعة من نقطة $A$ (وهي \emph{النقطة الجسم}). فإذا
مرّت الأشعة الخارجة كلها بنقطة واحدة $A'$، أو بدت كلها آتيةً من نقطة
واحدة $A'$، كان النظام \emph{مستجمعًا} من أجل
$A$ وكانت $A'$ هي \emph{صورة} $A$. والصورة \emph{حقيقية} إذا تقاطعت
الأشعة الخارجة فعلًا في $A'$ (فيمكن التقاطها على شاشة)، و\emph{تخيلية}
إذا لم تتقاطع إلا امتداداتها الخلفية (فتُرى عبر النظام ولا تُسقَط).
وكذلك الجسم حقيقي إذا تباعدت الأشعة الواردة عن $A$، وتخيلي إذا تقاربت
نحو $A$ خلف مدخل النظام. ويقال عن الزوج $(A, A')$ إنه \emph{مترافق}.
\end{definition}

\begin{definition}[النظام المركزي؛ شرطا غاوس]\label{def:b1:mirrors-thin-lenses:gauss}
\index{محور بصري}\index{شرطا غاوس}\index{شعاع مجاوري}
\emph{النظام المركزي} نظام له محور تناظر دوراني هو \emph{المحور
البصري}. والأشعة التي تبقى قريبة من المحور وتصنع معه زوايا صغيرة
\emph{مجاورية}؛ ويقال عن نظام لا يُستعمل إلا بأشعة مجاورية إنه يعمل
وفق \emph{شرطَي غاوس}. وفي هذين الشرطين يكون النظام المركزي مستجمعًا
تقريبًا من أجل كل نقطة، ويكون لمستوي جسمٍ عمودي على المحور مستوي \omterm{def:b1:mirrors-thin-lenses:image}{صورة}
عمودي على المحور (\emph{الاستواء}).
\end{definition}
\admitted

\begin{remark}[لماذا الزوايا الصغيرة]\label{rem:b1:mirrors-thin-lenses:why}
\omterm{prop:b1:rays-reflection-refraction:mirror}{المرآة المستوية} هي النظام البسيط الوحيد المستجمع بدقة من أجل كل نقطة.
أما المرآة أو \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} الكروية فلا تستجمع إلا بقدر ما يكون
$\sin\theta \approx \tan\theta \approx \theta$: فهذا التقريب هو الذي
يحوّل قانون سنيل إلى علاقات خطية بين المسافات. وخارجه تصير \omterm{def:b1:mirrors-thin-lenses:image}{صورة}
النقطة لطخة (وهي \emph{الزيوغ})، ومهمة المصمّم --- الحواجز وتركيب
العدسات والسطوح غير الكروية --- أن يخفض اللطخة دون ما تستطيع العين أو
المستقبِل تمييزه.
\end{remark}

\begin{notation}[القياسات الجبرية والتكبير]\label{not:b1:mirrors-thin-lenses:signs}
\index{قياس جبري}\index{تكبير عرضي}
يُوجَّه \omterm{def:b1:mirrors-thin-lenses:gauss}{المحور البصري} في اتجاه الضوء الوارد، ويُوجَّه الاتجاه العرضي
نحو الأعلى. ومن أجل نقطتين $P$ و $Q$ على المحور يرمز
$\overline{PQ}$ إلى \omterm{rem:b1:mirrors-thin-lenses:why}{القياس الجبري} ($\overline{PQ} > 0$
إذا كانت $Q$ خلف $P$ في اتجاه الضوء)؛ ومن أجل جسم $AB$ عمودي على
المحور وصورته $A'B'$ يكون \emph{\omterm{rem:b1:mirrors-thin-lenses:why}{التكبير العرضي}}
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}} ,
\]
سالبًا من أجل \omterm{def:b1:mirrors-thin-lenses:image}{صورة} مقلوبة، وقيمته المطلقة أكبر من $1$ من أجل \omterm{def:b1:mirrors-thin-lenses:image}{صورة}
مكبَّرة.
\end{notation}

\section{المرايا الكروية}

\begin{definition}[المرآة الكروية]\label{def:b1:mirrors-thin-lenses:mirror}
\index{مرآة كروية}\index{مرآة مقعرة}\index{مرآة محدبة}\index{بؤرة}
\emph{المرآة الكروية} قلنسوة كروية عاكسة مركزها $C$
وقطبها $S$ (أي تقاطعها مع المحور). وهي \emph{مقعرة}
إذا لاقى الضوء وجهها المجوَّف ($C$ في جهة الضوء
الوارد)، و\emph{محدبة} في غير ذلك. و\emph{بؤرتها} $F$ هي \omterm{def:b1:mirrors-thin-lenses:image}{صورة} نقطة في
اللانهاية على المحور؛ وبعدها البؤري $f = \overline{SF}$.
\end{definition}

\begin{proposition}[بؤرة مرآة كروية]\label{prop:b1:mirrors-thin-lenses:focus}
وفق \omterm{def:b1:mirrors-thin-lenses:gauss}{شرطَي غاوس} تكون \omterm{def:b1:mirrors-thin-lenses:mirror}{البؤرة} منتصف $SC$:
$\overline{SF} = \overline{SC}/2$. وللمرآة المقعرة $f < 0$ (\omterm{def:b1:mirrors-thin-lenses:mirror}{بؤرة}
حقيقية أمامها)، وللمحدبة $f > 0$ (\omterm{def:b1:mirrors-thin-lenses:mirror}{بؤرة} تخيلية خلفها).
\end{proposition}

\begin{proof}
الشعاع الموازي للمحور على ارتفاع $h$ يلاقي المرآة في $I$؛ والناظم في
$I$ هو نصف القطر $CI$، ويصنع الزاوية $\alpha$ مع
المحور حيث $\sin\alpha = h/R$. وبعد الانعكاس بالزاوية $\alpha$ نفسها عن
الناظم يقطع الشعاع المحور في $F$؛ وفي المثلث $CIF$ تتساوى
الزاويتان $\alpha$ عند $C$ وعند $I$، ومنه $CF = IF$، وبالإسقاط
$CF = R/(2\cos\alpha)$. ومن أجل الأشعة المجاورية $\cos\alpha \to 1$ و
$CF = R/2$: فتقطع هذه الأشعة كلها المحور في النقطة نفسها، أي
منتصف $SC$. أما الإشارة: فللمرآة المقعرة تقع $C$ و $F$ أمامها
(قبلها)، ومنه $\overline{SC} < 0$.
\end{proof}

\begin{theorem}[علاقة الترافق لمرآة كروية]\label{thm:b1:mirrors-thin-lenses:mirrorconj}
\index{علاقة الترافق!مرآة كروية}
وفق \omterm{def:b1:mirrors-thin-lenses:gauss}{شرطَي غاوس} تحقق \omterm{def:b1:mirrors-thin-lenses:image}{صورة} $A'$ لنقطة $A$ على المحور
\[
\frac{1}{\overline{SA'}} + \frac{1}{\overline{SA}} = \frac{2}{\overline{SC}}
= \frac{1}{\overline{SF}},
\qquad
\gamma = -\frac{\overline{SA'}}{\overline{SA}} .
\]
\end{theorem}

\begin{proof}
لنأخذ \omterm{def:b1:mirrors-thin-lenses:mirror}{مرآة مقعرة} وجسمًا حقيقيًّا $A$ خلف $C$؛ فتغطي الصيغة الجبرية
عندئذٍ كل الحالات. الشعاع الصادر عن $A$ يلاقي المرآة في $I$
(على ارتفاع $h$) ويعود إلى $A'$ على المحور. ولنسمّ $u$ و $u'$ و $\omega$
زوايا $AI$ و $A'I$ و $CI$ مع المحور؛ فمجاوريًّا يكون
$u = h/AS$ و $u' = h/A'S$ و $\omega = h/CS$ (بالمسافات). ويقول قانون
الانعكاس إن الناظم $CI$ ينصّف الزاوية بين $AI$ و
$A'I$: أي $\omega - u = u' - \omega$، أي $u + u' = 2\omega$، أي
$1/AS + 1/A'S = 2/CS$؛ ومع كون $\overline{SA}$ و $\overline{SA'}$ و
$\overline{SC}$ كلها سالبة تكون هذه هي العلاقة المعروضة. أما
التكبير فالشعاع $B \to S \to B'$ المنعكس عند القطب يصنع
زاويتين متساويتين مع المحور: $A'B'/SA' = AB/SA$ بالأطوال، \omterm{def:b1:mirrors-thin-lenses:image}{والصورة}
مقلوبة، ومنه $\gamma = -\overline{SA'}/\overline{SA}$.
\end{proof}

\begin{method}[بناء صورة في مرآة]\label{met:b1:mirrors-thin-lenses:mirrorrays}
من أجل نقطة جسم $B$ خارج المحور، ارسم اثنين من الأشعة الأربعة
المميّزة؛ فتقاطعهما (أو تقاطع امتداديهما) هو $B'$:
\begin{enumerate}
  \item الموازي للمحور $\to$ ينعكس مارًّا \omterm{def:b1:mirrors-thin-lenses:mirror}{بالبؤرة} $F$؛
  \item المارّ \omterm{def:b1:mirrors-thin-lenses:mirror}{بالبؤرة} $F$ $\to$ ينعكس موازيًا للمحور؛
  \item المارّ بالمركز $C$ $\to$ ينعكس على نفسه؛
  \item المتّجه إلى القطب $S$ $\to$ ينعكس متناظرًا بالنسبة إلى المحور.
\end{enumerate}
ثم تكون $A'$ مسقط العمود النازل من $B'$ على المحور.
\end{method}

\begin{example}[مرآة الحلاقة ومرآة الرؤية الخلفية]\label{ex:b1:mirrors-thin-lenses:mirrors}
\omterm{def:b1:mirrors-thin-lenses:mirror}{مرآة مقعرة} نصف قطرها \qty{60}{cm} ($f = \qty{-30}{cm}$) ووجهٌ على بعد
\qty{20}{cm}: $1/\overline{SA'} = -1/30 + 1/20 = 1/60$، ومنه
$\overline{SA'} = \qty{+60}{cm}$ --- خلف المرآة، أي تخيلية ---
و $\gamma = -60/(-20) = +3$: أي منتصبة أكبر ثلاث مرات، وهي مرآة
الحلاقة. \omterm{def:b1:mirrors-thin-lenses:mirror}{ومرآة محدبة} نصف قطرها \qty{2.0}{m} ($f = \qty{+1.0}{m}$)
وسيارة على بعد \qty{20}{m} خلفها: $1/\overline{SA'} = 1 + 1/20$، ومنه
$\overline{SA'} = \qty{0.95}{m}$ (تخيلية خلفها) و $\gamma = +0.048$:
أي منتصبة أصغر عشرين مرة، ومن ثمّ حقل واسع --- وهي مرآة الرؤية
الخلفية، مع تحذيرها من أن الأجسام أقرب ممّا تبدو.
\end{example}

\section{العدسات الرقيقة}

\begin{definition}[العدسة الرقيقة]\label{def:b1:mirrors-thin-lenses:lens}
\index{عدسة رقيقة}\index{عدسة مجمّعة}\index{عدسة مفرّقة}\index{مركز بصري}\index{بعد بؤري}\index{تقارب}
\emph{العدسة} \omterm{def:b1:rays-reflection-refraction:index}{وسط شفاف} يحدّه سطحان كرويان (أو مستويان)؛ وهي
\emph{رقيقة} حين تكون سماكتها مهملة أمام أنصاف الأقطار وأمام المسافات
الداخلة في المسألة، فيندمج القطبان في نقطة واحدة هي \emph{المركز
البصري} $O$. والشعاع المارّ بالمركز $O$ لا ينحرف. و\emph{\omterm{def:b1:mirrors-thin-lenses:mirror}{البؤرة} \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}} $F'$
هي \omterm{def:b1:mirrors-thin-lenses:image}{صورة} نقطة في اللانهاية على المحور؛ و\emph{\omterm{def:b1:mirrors-thin-lenses:mirror}{البؤرة} الجسم} $F$ هي
النقطة التي تقع صورتها في اللانهاية؛ وكلتاهما على المسافة نفسها من
$O$: $\overline{OF'} = -\overline{OF} = f'$، وهو \emph{البعد البؤري}.
والعدسة \emph{مجمّعة} إذا كان $f' > 0$ (فتكون $F'$ حقيقية خلفها، وحوافها
أرقّ من مركزها)، و\emph{مفرّقة} إذا كان $f' < 0$. و\emph{تقاربها}
$V = 1/f'$ يقاس بوحدة \emph{الديوبتر}
($\qty{1}{\delta} = \qty{1}{m^{-1}}$).
\end{definition}

\begin{theorem}[علاقة الترافق لعدسة رقيقة]\label{thm:b1:mirrors-thin-lenses:lensconj}
\index{علاقة الترافق!عدسة رقيقة}\index{علاقة ديكارت}\index{علاقة نيوتن}
وفق \omterm{def:b1:mirrors-thin-lenses:gauss}{شرطَي غاوس} تحقق \omterm{def:b1:mirrors-thin-lenses:image}{صورة} $A'$ لنقطة $A$ على المحور علاقة
ديكارت
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'},
\qquad
\gamma = \frac{\overline{OA'}}{\overline{OA}},
\]
وتحقق كذلك، على نحو مكافئ، علاقة نيوتن
\[
\overline{FA}\cdot\overline{F'A'} = -f'^2,
\qquad
\gamma = \frac{f'}{\overline{FA}} = -\frac{\overline{F'A'}}{f'} .
\]
\end{theorem}

\begin{proof}[برهان جزئي]
الواقعة الأساسية أن \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة الرقيقة} تحرف الشعاع العابر لها على ارتفاع
$h$ بالزاوية $h/f'$ نحو المحور، مهما يكن اتجاه الشعاع
--- فكل منطقة من \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} تعمل عمل موشور رقيق (\cref{rem:b1:rays-reflection-refraction:thin})
تنمو زاوية رأسه خطيًّا مع $h$ في السطوح الكروية؛ وهذه
الخطية، الصحيحة تمامًا للأشعة المجاورية، مسلَّمة هنا ومشتقّة من قانون
سنيل عند سطحين كرويين في مجلد السنة الثانية. وبقبولها:
الشعاع الصادر عن $A$ (على المسافة $p = -\overline{OA} > 0$ لجسم
حقيقي) والواصل إلى \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} على ارتفاع $h$ يرد بميل $h/p$، ثم
ينحرف بمقدار $h/f'$ ويغادر بميل $h/p - h/f'$؛ فيقطع المحور
في $A'$ حيث $h/\overline{OA'} = h/f' - h/p$، أي $1/\overline{OA'}
- 1/\overline{OA} = 1/f'$، مستقلًّا عن $h$: فتلتقي كل الأشعة الصادرة عن $A$
في $A'$. والشعاع $B \to O \to B'$ المارّ بالمركز مستقيم، ومنه
$\overline{A'B'}/\overline{AB} = \overline{OA'}/\overline{OA}$. أما
صيغة نيوتن فتتبع بتعويض $\overline{OA} = \overline{OF} + \overline{FA}
= -f' + \overline{FA}$ و $\overline{OA'} = f' + \overline{F'A'}$.
\end{proof}

\begin{method}[بناء صورة عبر عدسة رقيقة]\label{met:b1:mirrors-thin-lenses:lensrays}
ثلاثة أشعة مميّزة صادرة عن $B$:
\begin{enumerate}
  \item المارّ بالمركز $O$: لا ينحرف؛
  \item الموازي للمحور: يخرج مارًّا \omterm{def:b1:mirrors-thin-lenses:mirror}{بالبؤرة} $F'$؛
  \item المارّ \omterm{def:b1:mirrors-thin-lenses:mirror}{بالبؤرة} $F$: يخرج موازيًا للمحور.
\end{enumerate}
ويكفي اثنان منها؛ ومن أجل \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة مفرّقة} تُستعمل الامتدادات (فالشعاع
المصوَّب نحو $F$ خلف \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} يخرج موازيًا، والشعاع الموازي يخرج كأنه
آتٍ من $F'$ أمامها).
\end{method}

\begin{proposition}[التكبير الطولي]\label{prop:b1:mirrors-thin-lenses:longitudinal}
\index{تكبير طولي}
إذا تحرك الجسم على امتداد المحور بمقدار $\dd\overline{OA}$ تحركت
\omterm{def:b1:mirrors-thin-lenses:image}{الصورة} بمقدار $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$، في
الاتجاه نفسه.
\end{proposition}

\begin{proof}
باشتقاق علاقة ديكارت عند $f'$ ثابت:
$-\dd\overline{OA'}/\overline{OA'}^2 + \dd\overline{OA}/\overline{OA}^2 = 0$،
ومنه $\dd\overline{OA'} = (\overline{OA'}/\overline{OA})^2\,\dd\overline{OA}$.
\end{proof}

\begin{remark}[عمق البؤرة]\label{rem:b1:mirrors-thin-lenses:depth}
جهاز عرض تكبيره $\gamma = -50$ له $\gamma^2 = 2500$: فشريحة تتقوّس
بمقدار \qty{0.1}{mm} تحت حرارة المصباح تقذف بصورتها
\qty{25}{cm} خارج مستوي الشاشة. وعلى العكس تحتمل آلة التصوير ($\abs\gamma
\ll 1$) انتقالات كبيرة للجسم في مقابل انزياح ضئيل \omterm{def:b1:mirrors-thin-lenses:image}{للصورة} ---
وهو عمق الميدان.
\end{remark}

\section{قياس بعد بؤري}

\begin{method}[التقويم الذاتي]\label{met:b1:mirrors-thin-lenses:autocollimation}
\index{تقويم ذاتي}
ضع \omterm{prop:b1:rays-reflection-refraction:mirror}{مرآة مستوية} خلف \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} مباشرة، وحرّك جسمًا مضاءً (صليبًا على زجاج
مصنفر) على امتداد المحور حتى تتكوّن صورته، المنعكسة راجعةً عبر \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة}،
واضحةً على الجسم نفسه، مقلوبةً وفي حجمه. ويكون الجسم عندئذٍ في المستوي
البؤري: فالأشعة تغادر \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} متوازية، وتعود من المرآة متوازية،
وتُجمَّع من جديد في المستوي البؤري. والمسافة جسم--عدسة هي $f'$.
\end{method}

\begin{proposition}[طريقة بسل]\label{prop:b1:mirrors-thin-lenses:bessel}
\index{طريقة بسل}\index{طريقة سيلبرمان}
جسم وشاشة مثبتان على بعد $D$ أحدهما من الآخر. وتكوّن \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة مجمّعة}
\omterm{def:b1:mirrors-thin-lenses:image}{صورة حقيقية} واضحة على الشاشة من أجل وضعين اثنين بالضبط
إذا كان $D > 4f'$، ومن أجل وضع واحد إذا كان $D = 4f'$، ولا من أجل أيّ
وضع إذا كان $D < 4f'$. والوضعان يفصل بينهما $d$ حيث
\[
f' = \frac{D^2 - d^2}{4D} .
\]
وعند $D = 4f'$ (سيلبرمان) يعطي الوضع الوحيد $\gamma = -1$.
\end{proposition}

\begin{proof}
لتكن $p = -\overline{OA} > 0$ المسافة جسم--عدسة؛ عندئذٍ
$\overline{OA'} = D - p$ وتعطي علاقة ديكارت $1/(D - p) + 1/p = 1/f'$،
أي $p^2 - Dp + Df' = 0$: ومميّزها $D^2 - 4Df'$ موجب إذا وفقط إذا كان
$D > 4f'$. والجذران $p_{1,2} = (D \mp \sqrt{D^2 - 4Df'})/2$ مجموعهما
$D$ (فالوضعان متناظران: إذ تتبادل مسافتا الجسم \omterm{def:b1:mirrors-thin-lenses:image}{والصورة})
والفرق بينهما $d = \sqrt{D^2 - 4Df'}$؛ ثم نحلّ بالنسبة إلى $f'$. وعند
$D = 4f'$ يكون $p = 2f' = \overline{OA'}$ و $\gamma = -1$.
\end{proof}

\section{عدستان رقيقتان}

\begin{proposition}[عدستان متلامستان؛ الثنائية اللابؤرية]\label{prop:b1:mirrors-thin-lenses:doublet}
\index{نظام لا بؤري}\index{تكبير زاوي}
تتصرف عدستان رقيقتان متلامستان تصرّف \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة رقيقة} واحدة تقاربها
$V = V_1 + V_2$. وعدستان موضوعتان بحيث $F_1' = F_2$ تكوّنان نظامًا
\emph{لا بؤري}: فالحزمة المتوازية تخرج متوازية، والحزمة
المائلة بالزاوية $\alpha$ تخرج مائلة بالزاوية $\alpha' = G\alpha$ حيث
\emph{التكبير الزاوي} $G = -f_1'/f_2'$.
\end{proposition}

\begin{proof}
\omterm{def:b1:mirrors-thin-lenses:image}{الصورة} $A_1$ التي تعطيها $L_1$ هي الجسم بالنسبة إلى $L_2$. وعند
التلامس $O_1 = O_2 = O$: فيكون $1/\overline{OA_1} - 1/\overline{OA} = V_1$ و
$1/\overline{OA'} - 1/\overline{OA_1} = V_2$؛ ونجمع. أما اللابؤرية: فحزمة
متوازية بالزاوية $\alpha$ تتجمّع في المستوي البؤري المشترك على ارتفاع
$h = f_1'\alpha$ عن المحور (بالشعاع المارّ بالمركز $O_1$)؛ وهذه النقطة، منظورًا
إليها من $L_2$، تقع في مستويها البؤري الجسم، فتخرج الحزمة متوازية،
بزاوية الشعاع الآتي من النقطة عبر $O_2$: أي $\alpha' = -h/f_2'$.
\end{proof}

\begin{example}[مقراب في سطر واحد]\label{ex:b1:mirrors-thin-lenses:afocal}
$f_1' = \qty{1000}{mm}$ و $f_2' = \qty{25}{mm}$: ومنه $G = -40$؛ فالقمر،
وعرضه نصف درجة، يصير قرصًا في $20^\circ$ --- وهو المقراب
الفلكي في الفصل التالي. ومع عينية مفرّقة،
$f_2' = \qty{-25}{mm}$ و $G = +40$: أي منتصبًا، وهو منظار غاليليو.
\end{example}

\section{مسألة: جهاز عرض الشرائح}

\begin{problem}[{مسألة نهاية الأسبوع --- عدسة واحدة، تُقاس على طاولة

عدسة العرض $L_1$ \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة رقيقة} مجمّعة بعدها البؤري $f_1'$ مجهول.

\noindent\textbf{الجزء I --- \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} على طاولة التجارب.}
\begin{enumerate}
  \item اذكر علاقة ديكارت والتكبير من أجل \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة
        رقيقة}، مع اصطلاحات الإشارة.
  \item الجسم والشاشة مثبتان على بعد $D$. بيّن أن وجود
        \omterm{def:b1:mirrors-thin-lenses:image}{صورة حقيقية} على الشاشة يقتضي $D \geq 4f_1'$.
  \item بيّن أنه حين يكون $D > 4f_1'$ يكون وضعا \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} اللذان يعطيان
        \omterm{def:b1:mirrors-thin-lenses:image}{صورة} واضحة متناظرين بالنسبة إلى منتصف ما بين
        الجسم والشاشة، وأن $f_1' = (D^2 - d^2)/4D$ حيث $d$
        هو الفصل بينهما.
  \item مع $D = \qty{80.0}{cm}$ و $d = \qty{40.0}{cm}$، احسب
        $f_1'$.
  \item توضع \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} في أسطوانة سميكة يتعذّر تحديد مركزها البصري
        بدقة. فلماذا لا يؤثر هذا في الطريقة؟
  \item تحمل المسافتان $u(D) = \qty{0.2}{cm}$ و
        $u(d) = \qty{0.5}{cm}$. احسب $u(f_1')$ واكتب النتيجة.
  \item أعطِ التكبير عند كلٍّ من الوضعين. وأيّهما
        يوافق جهاز عرض؟
  \item ما أصغر $D$ توجد عنده \omterm{def:b1:mirrors-thin-lenses:image}{صورة} واضحة، وما
        التكبير عندئذٍ (سيلبرمان)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- رمي \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}.}
يجب أن يكون عرض \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} على الشاشة \qty{1.80}{m}. خذ
$f_1' = \qty{15.0}{cm}$.
\begin{enumerate}[resume]
  \item ما التكبير المطلوب (بإشارته)؟
  \item احسب المسافة عدسة--شاشة والمسافة شريحة--عدسة.
  \item لماذا تُدخل الشريحة مقلوبة ومعكوسة يمينًا ويسارًا؟
  \item تُنقل الشاشة إلى \qty{10.0}{m}. فبأيّ مقدار، وفي أيّ
        اتجاه، يجب تحريك \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} بالنسبة إلى الشريحة لإعادة الضبط؟
        وما عرض \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} الآن؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- تسامح الضبط والضوء.}
\begin{enumerate}[resume]
  \item بيّن أن انتقالًا صغيرًا $\dd\overline{OA}$ للشريحة
        يزيح \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} بمقدار $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$.
  \item يتقوّس الفيلم بمقدار \qty{0.10}{mm} تحت حرارة المصباح. فبأيّ
        مقدار يتحرك مستوي \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} (والشاشة على \qty{7.65}{m})؟
        علّق على ذلك.
  \item فتحة \omterm{def:b1:mirrors-thin-lenses:lens}{العدسة} قطرها \qty{30}{mm}. فإذا وقعت الشاشة
        على \qty{20}{cm} خارج مستوي \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}، فما قطر
        لطخة الضبابية لنقطة من الشريحة؟ وهل تلاحظها العين من \qty{5}{m}
        (وتمييزها الزاوي نحو $\qty{3e-4}{rad}$)؟
  \item تتلقى الشريحة \qty{2.0}{W} من الضوء، يصل كله إلى
        الشاشة. احسب الإشعاع (الاستطاعة على واحدة المساحة) على
        الشريحة وعلى الشاشة، وبيّن أن نسبتهما هي $\gamma^2$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- محوّل الرمي البعيد.}
في قاعة أكبر تبعد الشاشة \qty{12.0}{m}. وتُثبَّت \omterm{def:b1:mirrors-thin-lenses:lens}{عدسة مفرّقة}
$L_2$ بعدها البؤري $f_2' = \qty{-30.0}{cm}$ على \qty{8.0}{cm} خلف
$L_1$؛ والشاشة على \qty{12.0}{m} من $L_2$.
\begin{enumerate}[resume]
  \item كي تكون \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} النهائية حقيقية وعلى الشاشة، أين (بالنسبة إلى
        $L_2$) يجب أن تقع \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} الوسيطة التي تعطيها $L_1$؟ وهل هي
        جسم حقيقي أم تخيلي بالنسبة إلى $L_2$؟
  \item استنتج موضع الشريحة $\overline{O_1A}$.
  \item احسب $\gamma_1$ و $\gamma_2$ والتكبير الكلي.
  \item أعطِ عرض \omterm{def:b1:mirrors-thin-lenses:image}{الصورة}، وقارنه بما تعطيه $L_1$ وحدها
        على الشاشة نفسها (على \qty{12.08}{m} من $L_1$).
  \item من أجل جسم في اللانهاية، بيّن أن النظام يكافئ
        عدسة واحدة بعدها البؤري $f_{\mathrm{eq}}$ حيث
        \[
        \frac{1}{f_{\mathrm{eq}}} = \frac{1}{f_1'} + \frac{1}{f_2'} - \frac{e}{f_1' f_2'},
        \]
        مع $e = \overline{O_1O_2}$، بمعنى أن حزمة مائلة بالزاوية
        $\alpha$ تتجمّع على ارتفاع $f_{\mathrm{eq}}\,\alpha$ عن
        المحور. احسب $f_{\mathrm{eq}}$.
  \item تحقق من أن نسبة عرضَي \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} (مع المحوّل ودونه)
        قريبة من $f_1'/f_{\mathrm{eq}}$. ولماذا هي قريبة فقط؟
  \item لماذا يجب أن تقع \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} الوسيطة على أقلّ من $\abs{f_2'}$
        خلف $L_2$ كي تكون \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} النهائية حقيقية؟
  \item مع المصباح نفسه، بأيّ عامل يتغير إشعاع الشاشة
        عند تركيب المحوّل (وعلى بعد الشاشة نفسه)؟
  \item لخّص: بأيّ عامل يضرب المحوّل \omterm{def:b1:mirrors-thin-lenses:lens}{البعد البؤري}، وماذا
        يفعل بحجم \omterm{def:b1:mirrors-thin-lenses:image}{الصورة} وسطوعها عند رميٍ معطى؟
\end{enumerate}
\end{problem}
