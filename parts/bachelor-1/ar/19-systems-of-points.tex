\chapter{جمل النقاط؛ مدخل إلى الأجسام الصلبة}\label{ch:b1:systems-of-points}

تغادر غاطسة اللوح، فتتكوّر وتشقلب مرتين ثم تستقيم؛
وخلال ذلك كله ترسم نقطة واحدة من جسمها --- مركز كتلتها ---
القطعَ المكافئ البسيط نفسه الذي يرسمه حجر. ويستغرق يويو يُسقط من
اليد ثلاثة أضعاف الزمن ليبلغ طرف خيطه مقارنةً بحجر،
ويصل دائرًا بألفَي دورة في الدقيقة. وتسبق كرة
مصمتة كرةً مجوّفة على أيّ منحدر، أيًّا كانت أبعادهما
أو كتلتاهما. وميكانيك النقطة الواحدة، الذي حُمل عبر الفصول
السابقة، يمتدّ إلى جمل من نقاط كثيرة --- وإلى الأجسام الصلبة
في الحياة اليومية --- عبر حفنة من المبرهنات: \omterm{def:b1:systems-of-points:cm}{فمركز الكتلة}
يتحرك كنقطة، وتنقسم الطاقة \omterm{def:b1:angular-momentum:L}{والعزم الحركي} إلى جزء \omterm{def:b1:systems-of-points:cm}{لمركز الكتلة}
وجزء ``حوله''، والجسم الدائر حول محور ثابت
جملة ذات درجة حرية واحدة يحكمها عزم عطالته.

\section{مركز الكتلة وكمية الحركة}

\begin{definition}[مركز الكتلة؛ كمية حركة جملة]\label{def:b1:systems-of-points:cm}
\index{مركز الكتلة}\index{المرجَّح}
من أجل جسيمات كتلها $m_i$ في النقاط $M_i$، وكتلتها الكلية $M = \sum m_i$، يُعرَّف
\emph{مركز الكتلة} (أو المرجَّح) $G$ بالعلاقة
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{أي}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
من أجل أيّ مبدأ $O$. و\emph{كمية حركة} الجملة هي $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[مبرهنة مركز الكتلة]\label{thm:b1:systems-of-points:cmtheorem}
\index{مبرهنة مركز الكتلة}\index{انحفاظ كمية الحركة}
في \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}،
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
أي إن \omterm{def:b1:systems-of-points:cm}{مركز الكتلة} يتحرك كجسيم نقطي كتلته $M$ يخضع
لمحصلة القوى \emph{الخارجية} وحدها. وفي \omterm{def:b1:newton-dynamics:conservation}{جملة مغلقة} بلا
قوة خارجية، تنحفظ $\vect P$ ويتحرك $G$ حركة منتظمة.
\end{theorem}

\begin{proof}
اجمع قانون نيوتن الثاني على الجسيمات؛ فتتلاشى القوى الداخلية
في ثنائيات \omterm{thm:b1:newton-dynamics:laws}{الفعل ورد الفعل}؛ و $\sum m_i\vect a_i = M\vect a_G$ باشتقاق
تعريف $G$ مرتين.
\end{proof}

\begin{example}[الغاطسة، والارتداد، والتصادم]\label{ex:b1:systems-of-points:cm}
\index{تصادم}
يتبع $G$ الغاطسة قطعًا مكافئًا تحت وزنها وحده، مهما
التوت --- فعضلاتها قوى داخلية. وبندقية كتلتها \qty{4.0}{kg}
تطلق رصاصة كتلتها \qty{10}{g} بسرعة \qty{800}{m/s}: تكون $\vect P = \vect 0$ قبل
وبعد، فترتدّ البندقية بسرعة \qty{2.0}{m/s}. وسيارتان، إحداهما \qty{1000}{kg}
عند \qty{20}{m/s} والأخرى \qty{1500}{kg} ساكنة، تلتحمان: $v =
20000/2500 = \qty{8.0}{m/s}$ --- فتنحفظ \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة}، بينما ذهب $60\%$ من
\omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} إلى معدن متجعّد وحرارة. \omterm{def:b1:newton-dynamics:momentum}{وكمية الحركة}
تنحفظ في كل تصادم؛ أما \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} فلا تنحفظ إلا في التصادمات \emph{المرنة}
(\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{مبرهنتا كونيغ}

\begin{definition}[المرجع المرجَّحي]\label{def:b1:systems-of-points:barycentric}
\index{مرجع مرجَّحي}
\emph{المرجع المرجَّحي} $\mathcal R^*$ مبدؤه $G$ وهو
في انسحاب بالنسبة إلى \omterm{thm:b1:newton-dynamics:laws}{المرجع العطالي} $\mathcal R$ (فمحاوره
تحافظ على اتجاهات ثابتة). وتُوسم المقادير المقيسة فيه بنجمة: $\vect v_i^*
= \vect v_i - \vect v_G$، مع $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[مبرهنتا كونيغ]\label{thm:b1:systems-of-points:koenig}
\index{مبرهنتا كونيغ}
ينقسم \omterm{def:b1:angular-momentum:L}{العزم الحركي} حول نقطة $O$ \omterm{thm:b1:work-and-energy:ke}{والطاقة الحركية} لجملة
إلى إسهام مركز كتلتها، حاملًا الكتلة
كلها، والإسهامات في \omterm{def:b1:systems-of-points:barycentric}{المرجع المرجَّحي}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
حيث $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ و $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ لا يتوقفان على مبدأ المرجع.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ و $\vect v_i = \vect v_G + \vect v_i^*$؛
انشر $\sum\vect{OM_i}\wedge m_i\vect v_i$ و $\sum\tfrac12 m_iv_i^2$: فالحدود
المتقاطعة تتضمن $\sum m_i\vect{GM_i} = \vect 0$ أو $\sum m_i\vect v_i^* = \vect 0$
وتتلاشى.
\end{proof}

\begin{example}[عجلة تتدحرج]\label{ex:b1:systems-of-points:rolling}
\index{تدحرج دون انزلاق}
عجلة كتلتها $M$ ونصف قطرها $R$ وعزم عطالتها $J$ حول محورها،
تتدحرج دون انزلاق بالسرعة $v$: تكون نقطة التماسّ ساكنة، ومنه
$\omega = v/R$ و $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- أي ثلاثة أرباع $Mv^2$ من أجل قرص منتظم ($J = \tfrac12 MR^2$)،
و $Mv^2$ من أجل طوق. والحدّ الثاني هو سبب صنع عجلات الدراجة
خفيفةً: فدورانها يكلّف طاقة لا تكلّفها كتلة الهيكل.
\end{example}

\section{مسألة الجسمين}

\begin{proposition}[الكتلة المختزلة]\label{prop:b1:systems-of-points:twobody}
\index{مسألة الجسمين}\index{كتلة مختزلة}
جسيمان لا يتآثران إلا أحدهما مع الآخر (بالقوة $\vect F$ على $m_2$
من $m_1$، و $-\vect F$ على $m_1$): يتحرك $G$ حركة منتظمة، ويحقق الموضع
النسبي $\vect r = \vect{M_1M_2}$ العلاقة
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
وهي معادلة جسيم وهمي واحد \emph{كتلته المختزلة} $\mu$
يتحرك تحت $\vect F$ حول مركز ثابت. وفي $\mathcal R^*$ يكون $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ و $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. وفي $\mathcal R^*$، $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ و
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$؛ ثم عوّض في $E_k^*$ و $\vect L^*$.
\end{proof}

\begin{example}[حين يتحرك المركز هو أيضًا]\label{ex:b1:systems-of-points:reduced}
في \cref{ch:b1:central-forces} كانت الشمس ثابتة: وهو دقيق في
الحالة الحدية $m \ll M$ فقط؛ وفي العام يُقرأ قانون كبلر الثالث $T^2 = 4\pi^2a^3/G(M +
m)$. ومن أجل جزيء كلوريد الهدروجين المهتزّ في \cref{pb:b1:work-and-energy:1}،
تكون الكتلة التي تهتزّ في البئر $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$، وهي ترفع التواتر بنسبة $1.4\%$؛ ومن أجل ذرة
الهدروجين تختلف $\mu$ للإلكترون عن $m_e$ بجزء من $1836$ ---
وهو قابل للقياس في الطيف.
\end{example}

\section{جسم صلب يدور حول محور ثابت}

\begin{definition}[الجسم الصلب؛ الدوران حول محور ثابت]\label{def:b1:systems-of-points:solid}
\index{جسم صلب}\index{دوران حول محور ثابت}
\emph{الجسم الصلب} جملة تحافظ نقاطها على أبعاد متبادلة
ثابتة. وإذا دار حول محور $\Delta$ ثابت في المرجع، وصف كل
نقطة دائرة مركزها على $\Delta$ \omterm{cor:b1:point-kinematics:circular}{بالسرعة الزاوية} نفسها
$\omega = \dot\theta$؛ ولنقطة تبعد $r_i$ عن المحور السرعة
$r_i\omega$.
\end{definition}

\begin{proposition}[عزم العطالة؛ العزم الحركي؛ الطاقة الحركية]\label{prop:b1:systems-of-points:inertia}
\index{عزم العطالة!للأجسام الصلبة}\index{مبرهنة هويغنس}
مع \emph{\omterm{prop:b1:angular-momentum:inertia}{عزم العطالة}} $J_\Delta = \sum m_ir_i^2$ (وهو تكامل
$\int r^2\,\dd m$ من أجل جسم متصل):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
والقيم المعيارية (بالكتلة $M$): حلقة رفيعة أو أسطوانة مجوّفة حول محورها،
$MR^2$؛ وقرص منتظم أو أسطوانة مصمتة، $\tfrac12 MR^2$؛ وقضيب رفيع
طوله $L$ حول مركزه، $\tfrac{1}{12}ML^2$، وحول أحد طرفيه،
$\tfrac13 ML^2$؛ وكرة مصمتة حول قطر، $\tfrac25 MR^2$.
و\emph{مبرهنة هويغنس}: حول محور $\Delta$ مواز لمحور
$\Delta_G$ يمرّ بالنقطة $G$ على بعد $d$، يكون $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
أما $L_\Delta$ و $E_k$ فمن \cref{prop:b1:angular-momentum:inertia}.
وللقضيب حول طرفه نجد أن
$\int_0^L x^2(M/L)\,\dd x = ML^2/3$؛ وحول مركزه نجد أن
$\int_{-L/2}^{L/2} x^2(M/L)\dd x = ML^2/12$. أما القرص: فحلقة نصف قطرها $r$
وعرضها $\dd r$ وكتلتها
$(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$؛ أما الحلقة
والكرة فمقبولتان (إذ تحتاج الكرة إلى تكامل مضاعف). ولهويغنس: مع
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2$، أي $r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$، يُجمع الحدّ المتقاطع إلى صفر بتعريف $G$.
\end{proof}

\begin{theorem}[ديناميك جسم صلب حول محور ثابت]\label{thm:b1:systems-of-points:rotation}
\index{مفصل}\index{استطاعة عزم}
من أجل \omterm{def:b1:systems-of-points:solid}{جسم صلب} يدور حول محور ثابت $\Delta$ في \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}،
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
أي إن عزوم القوى الخارجية حول المحور تقود التسارع
الزاوي، وعزم $\mathcal M_\Delta$ يقدّم الاستطاعة
$\mathcal M_\Delta\omega$. أما \emph{المفصل} المثالي (وهو محمل بلا احتكاك) فلا يؤثر
بأيّ عزم حول محوره، فلا يسرّع الدوران ولا
يكبحه.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} مسقطة على $\Delta$، مع $L_\Delta
= J_\Delta\dot\theta$ و $J_\Delta$ ثابت من أجل \omterm{def:b1:systems-of-points:solid}{جسم صلب}؛ ثم اضرب في
$\dot\theta$ من أجل الصيغة الطاقية (فالقوى الداخلية \omterm{def:b1:systems-of-points:solid}{لجسم صلب} لا تبذل
شغلًا: إذ لا تتغير الأبعاد).
\end{proof}

\begin{example}[البندول المركّب؛ بندول الفتل]\label{ex:b1:systems-of-points:pendulums}
\index{بندول مركّب}\index{بندول الفتل}
\omterm{def:b1:systems-of-points:solid}{جسم صلب} كتلته $M$ يدور حول محور أفقي على بعد $d$
من $G$: عزم الوزن هو $-Mgd\sin\theta$، ومنه $J_\Delta\ddot\theta =
-Mgd\sin\theta$، وللتأرجحات الصغيرة الدور
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
وهو دور بندول بسيط طوله $J_\Delta/Md$ --- ومن أجل قضيب منتظم
مفصَّل عند أحد طرفيه، $2L/3$. وقرص معلّق بسلك ثابت فتله
$C$ (وعزمه المعيد $-C\theta$): $J\ddot\theta = -C\theta$ و $T = 2\pi\sqrt{J/C}$
--- وهو ميزان الفتل الذي قاس $G$ وقانون كولون.
\end{example}

\begin{proposition}[التدحرج دون انزلاق على منحدر]\label{prop:b1:systems-of-points:rollingincline}
\omterm{def:b1:systems-of-points:solid}{جسم صلب} دوراني ($J = kMR^2$ حول محوره) يُطلق على
منحدر زاويته $\alpha$ يتدحرج دون انزلاق بالتسارع
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
أي $\tfrac57 g\sin\alpha$ من أجل كرة مصمتة، و $\tfrac23 g\sin\alpha$ من أجل قرص،
و $\tfrac12 g\sin\alpha$ من أجل طوق --- مستقلًّا عن الكتلة ونصف القطر.
\end{proposition}

\begin{proof}
بالطاقة: $Mgh = \tfrac12 Mv^2(1 + k)$ على امتداد المنحدر (فالاحتكاك السكوني
عند نقطة التماسّ الساكنة لا يبذل شغلًا)، ومنه $v^2 = 2ah$ مع $a$ كما
ذُكر؛ أو بقانون نيوتن من أجل $G$ ومبرهنة العزوم حول $G$ مع
الاحتكاك $f$: $Ma = Mg\sin\alpha - f$ و $kMR^2\dot\omega = fR$ و $a = R\dot\omega$.
\end{proof}

\begin{remark}[الجمل القابلة للتشوّه]\label{rem:b1:systems-of-points:deformable}
\index{جملة قابلة للتشوّه}
من أجل جملة غير صلبة، تُقرأ مبرهنة \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: فالقوى الداخلية تبذل
شغلًا (كالعضلات والنوابض والاحتكاك داخل الجملة). فالمتزلجة التي تضمّ
ذراعيها، والغاطسة التي تتكوّر، والدراج الواقف على الدوّاستين، يكسبون كلهم
\omterm{thm:b1:work-and-energy:ke}{طاقة حركية} بلا شغل خارجي --- فمبرهنة \omterm{def:b1:systems-of-points:cm}{مركز الكتلة}
تبقى صحيحة، والطاقة تأتي من الداخل.
\end{remark}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:9}

\begin{exercise}[$\star\star\star$]\label{exo:b1:systems-of-points:12}

بنوى خفيفة؟
\end{exercise}

\section{مسألة: اليويو}

\begin{problem}[{مسألة نهاية الأسبوع --- قرص على محور رفيع في طرف
خيط: لماذا يسقط ببطء شديد، وبأيّ سرعة يدور في الأسفل،
ولماذا ``ينام'' هناك، وبأيّ اتجاه تتدحرج بكرة حين تسحب
خيطها}]
\label{pb:b1:systems-of-points:1}
اليويو قرص منتظم كتلته $M = \qty{50}{g}$ ونصف قطره $R =
\qty{3.0}{cm}$ له محور رفيع نصف قطره $r = \qty{0.40}{cm}$ يُلفّ حوله
خيط طوله $h = \qty{1.0}{m}$؛ ويُمسك طرف الخيط
العلوي ثابتًا. ولدينا $g = \qty{9.81}{m/s^2}$ و $J = \tfrac12 MR^2$ حول
المحور.

\medskip
\noindent\textbf{الجزء I --- التحريك والطاقة.}
\begin{enumerate}
  \item أين يقع مركز كتلة اليويو $G$؟ ولماذا؟
  \item بيّن بمبرهنة كونيغ أن طاقته الحركية هي $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item اشتقّ $J = \tfrac12 MR^2$ بالمكاملة على الحلقات، واحسب
        $J$.
  \item وبينما ينحلّ الخيط دون انزلاق عن المحور، اربط بين
        $v_G$ و $\omega$.
  \item استنتج $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ واحسب
        العامل بين القوسين. علّق على ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- السقوط.}
\begin{enumerate}[resume]
  \item عدّد القوى المؤثرة في اليويو؛ واكتب مبرهنة \omterm{def:b1:systems-of-points:cm}{مركز الكتلة}
        (على محور شاقولي نحو الأسفل).
  \item اكتب مبرهنة \omterm{def:b1:angular-momentum:L}{العزم الحركي} حول المحور المارّ بالنقطة $G$
        (فأيّ القوى لها عزم؟).
  \item اجمع مع السؤال 4 لإيجاد التسارع $a_G = g/(1 +
        R^2/2r^2)$؛ ثم احسبه.
  \item احسب شدّ الخيط وقارنه بالوزن.
  \item احسب زمن انحلال \qty{1.0}{m} من الخيط، والسرعة عندئذ؛
        وقارنها بحجر يُسقط من الارتفاع نفسه.
  \item احسب \omterm{cor:b1:point-kinematics:circular}{السرعة الزاوية} في الأسفل، بوحدة \unit{rad/s} وبالدورات في الدقيقة.
  \item اقسم \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية} في الأسفل إلى جزء انسحابي
        وجزء دوراني.
  \item تحقق من موازنة الطاقة: قارن $Mgh$ \omterm{thm:b1:work-and-energy:ke}{بالطاقة الحركية}
        في الأسفل، وقل لماذا لا يبذل شدّ الخيط، على كبره، أيّ شغل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- في الأسفل: النائم.}
حين ينحلّ الخيط كله يوقف \omterm{def:b1:systems-of-points:cm}{مركز الكتلة} في بضعة
ميلي ثوانٍ؛ ويظلّ اليويو دائرًا في العروة عند طرف
الخيط (فهو ``ينام'').
\begin{enumerate}[resume]
  \item لماذا يبقى الدوران بعد توقف $G$؟ (أيّ مبرهنة، وأيّ
        قوة كانت ستلزم لإيقافه؟)
  \item قدّر الشدّ المتوسط أثناء التوقف إذا أُوقف $G$
        على مدى \qty{5}{ms}.
  \item تحتكّ العروة بالمحور بعزم احتكاك $\Gamma =
        \qty{1.0e-4}{N.m}$: فكم يدوم نوم اليويو؟
  \item نتّة حادّة إلى أعلى على الخيط تجعله يقبض على المحور
        من جديد؛ فيصعد اليويو الدائر. فباستعمال الطاقة، إلى أيّ
        ارتفاع كان يمكن أن يرتفع لو لم يضع شيء؟ ولماذا يكون أقلّ
        عمليًّا؟
  \item أثناء الصعود، هل الشدّ أكبر من الوزن أم أصغر؟ (بحسب
        إشارة $a_G$.)
  \item ماذا كان سيحدث ليويو نصف قطر محوره يساوي $R$
        (بخيط ملفوف على الحافة)؟ احسب تسارع سقوطه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- البكرة على الطاولة.}
يستلقي الجسم نفسه على طاولة خشنة، ومحوره أفقي، ويغادر خيطه
\emph{أسفل} المحور أفقيًّا؛ ويُسحب بقوة
$F$. وهو يتدحرج دون انزلاق.
\begin{enumerate}[resume]
  \item عند نقطة التماسّ $C$ مع الطاولة، ما سرعة
        مادة البكرة؟ واستنتج أن الحركة، في كل
        لحظة، دوران حول $C$.
  \item احسب عزم $F$ حول $C$ (\omterm{def:b1:angular-momentum:moment}{وذراع الرافعة} $R - r$) وعزم
        الوزن وعزم ردّ فعل الطاولة.
  \item باستعمال مبرهنة \omterm{def:b1:angular-momentum:L}{العزم الحركي} حول $C$ مع $J_C = J +
        MR^2$ (بهويغنس)، أوجد $a_G$ وبيّن أن البكرة تتدحرج
        \emph{نحو} اليد.
  \item ثم يُسحب الخيط بزاوية $\beta$ فوق
        الأفقي. بيّن أن البكرة لا تتدحرج حين يمرّ حامل
        القوة $F$ بالنقطة $C$، أي حين $\cos\beta = r/R$، وأنها
        تتدحرج بعيدًا عن اليد من أجل سحب أشدّ ميلًا. واحسب تلك الزاوية.
  \item احسب $a_G$ من أجل $F = \qty{0.20}{N}$ مسحوبة أفقيًّا، وقوة
        الاحتكاك اللازمة؛ وما الذي يحدّ $F$ قبل أن تنزلق البكرة؟
  \item لخّص: المبرهنات الثلاث المستعملة (\omterm{def:b1:systems-of-points:cm}{مركز الكتلة}، \omterm{def:b1:angular-momentum:L}{والعزم
        الحركي} حول محور، والطاقة) وما الذي قرّرته كلٌّ منها.
\end{enumerate}
\end{problem}
