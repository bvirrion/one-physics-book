\chapter{المبدأ الثاني: الإنتروبيا}\label{ch:b1:second-law-entropy}

ألقِ ملعقة ساخنة في كوب من الماء البارد: تبرد الملعقة، ويدفأ
الماء، وينتهيان عند \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} واحدة. وكان المبدأ الأول ليجيز
العكس تمامًا --- أن يزداد الماء بردًا بينما تزداد الملعقة
سخونة، والطاقة محفوظة إلى الجول --- ومع ذلك لا يقع أبدًا. وشريط
يُدار إلى الوراء يُعرَف من فوره؛ فالجزيئات تخضع لقوانين لا سهم
لها، وللعالم سهم. \omterm{thm:b1:second-law-entropy:secondlaw}{والمبدأ الثاني للترموديناميك} يسمّي المقدار
الذي يميّز الجهتين إحداهما من الأخرى: وهو الإنتروبيا، التي
لا تستطيع \omterm{def:b1:newton-dynamics:conservation}{جملة مغلقة} إلا أن تُنشئها، ولا أن تُفنيها أبدًا. وينصّ
هذا الفصل المبدأ، ويحسب الإنتروبيا لجمل هذا المجلد، ويبيّن
كيف تقيس اللاعكوسية، ويعطيها في آخره معناها بلغة العدّ.

\section{المبدأ الثاني}

\begin{theorem}[المبدأ الثاني للترموديناميك]\label{thm:b1:second-law-entropy:secondlaw}
\index{المبدأ الثاني للترموديناميك}\index{إنتروبيا}\index{إنتروبيا!مُنشأة}\index{إنتروبيا!متبادَلة}
لكل \omterm{def:b1:newton-dynamics:conservation}{جملة مغلقة} دالةُ حالة امتدادية، هي
\emph{الإنتروبيا} $S$ (بوحدة \unit{J/K})، بحيث يكون في أيّ تحول
\[
\Delta S = S_{\mathrm{exch}} + S_{\mathrm{created}}, \qquad
S_{\mathrm{exch}} = \int\frac{\delta Q}{T_{\mathrm{ext}}}, \qquad
S_{\mathrm{created}} \geq 0 ,
\]
حيث $\delta Q$ الحرارة المتلقاة من الوسط الخارجي عند \omterm{def:b1:kinetic-theory:temperature}{درجة
الحرارة} $T_{\mathrm{ext}}$ في الموضع الذي تدخل منه، وتكون
$S_{\mathrm{created}} = 0$ إذا كان التحول عكوسًا وفقط إذا كان
كذلك. ولا تستطيع جملة معزولة ($\delta Q = 0$) إلا أن ترى إنتروبيتها
تنمو، ويكون توازنها حالةَ الإنتروبيا العظمى.
\end{theorem}
\admitted

\begin{remark}[قراءة المبدأ]\label{rem:b1:second-law-entropy:reading}
خلافًا للطاقة، لا تنحفظ الإنتروبيا: بل \emph{تُتبادَل} مع الحرارة
(ومعها وحدها --- إذ لا يحمل الشغل إنتروبيا) و\emph{تُنشأ} في كل
سيرورة لاعكوسة: الاحتكاك، وجريان الحرارة عبر فرق في \omterm{def:b1:kinetic-theory:temperature}{درجة
الحرارة}، واندفاع غاز في الخلاء، والمزج. والحدّ المُنشأ هو
مقياس اللاعكوسية، ولا يكون التحول ممكنًا إلا إذا كان
غير سالب. و``العكوس'' مثال أعلى --- شبه سكوني، بلا احتكاك، وبحرارة
تُتبادَل عبر فروق متلاشية في \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} --- تقترب منه السيرورات
الحقيقية ولا تبلغه أبدًا.
\end{remark}

\begin{proposition}[المتطابقة الأساسية؛ إنتروبيا الغاز المثالي والأطوار المكثَّفة]\label{prop:b1:second-law-entropy:identity}
\index{المتطابقة الأساسية للترموديناميك}\index{إنتروبيا!غاز مثالي}
من أجل \omterm{def:b1:newton-dynamics:conservation}{جملة مغلقة} ثابتة التركيب، وعلى امتداد أيّ مسار عكوس، يكون
$\delta Q_{\mathrm{rev}} = T\,\dd S$ و $\delta W_{\mathrm{rev}} = -P\,\dd V$، ومنه
\[
\dd U = T\,\dd S - P\,\dd V ,
\]
وهي علاقة بين دوال حالة تصحّ من أجل أيّ تغيّر تفاضلي.
ومن ثمّ يكون، من أجل $n$ مولًا من \omterm{def:b1:kinetic-theory:model}{غاز مثالي}،
\[
S(T, V) = nC_{V,m}\ln T + nR\ln V + \text{ثابت}, \qquad
S(T, P) = nC_{P,m}\ln T - nR\ln P + \text{ثابت},
\]
ويكون من أجل طور مكثَّف سعته الحرارية $C$: $S(T) = C\ln T + \text{ثابت}$.
\end{proposition}

\begin{proof}
في \omterm{def:b1:first-law:transformation}{التحول العكوس} يعطي $T_{\mathrm{ext}} = T$ و $S_{\mathrm{created}} = 0$ أن $\dd S =
\delta Q/T$؛ ويعطي المبدأ الأول عندئذ المتطابقة، التي لم تعد
تشير إلى مسار. \omterm{def:b1:kinetic-theory:model}{والغاز المثالي}: $\dd S = (\dd U + P\,\dd V)/T = nC_{V,m}\dd T/T
+ nR\,\dd V/V$؛ فنكامل؛ ونستعمل $PV = nRT$ للصيغة الأخرى. والطور
المكثَّف: $\dd V = 0$، ومنه $\dd S = C\,\dd T/T$.
\end{proof}

\begin{example}[قراءة الصيغتين]\label{ex:b1:second-law-entropy:formulas}
\omterm{prop:b1:first-law:transformations}{التحول الكظوم} العكوس \emph{متساوي الإنتروبيا}: فثبات $S(T, V)$
يعطي $T^{C_{V,m}}V^R$ ثابتًا، أي $TV^{\gamma - 1}$ ثابتًا ---
وهو \omterm{prop:b1:first-law:transformations}{قانون لابلاس} من جديد. ومضاعفة حجم \omterm{def:b1:kinetic-theory:scales}{مول} من غاز عند ثبات
\omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} ترفع إنتروبيته بمقدار $R\ln2 = \qty{5.8}{J/K}$، سواء
جرى ذلك على نحو عكوس (فتأتي الإنتروبيا حرارةً من الحوض) أم \omterm{rem:b1:first-law:irreversible}{بتمدد
جول} في الخلاء (فلا حرارة: وتكون \qty{5.8}{J/K} نفسها عندئذ
\emph{مُنشأة}). فدالة الحالة لا يعنيها كيف بلغت حالتها؛
أما الموازنة فيعنيها ذلك.
\end{example}

\section{موازنات الإنتروبيا}

\begin{proposition}[الحرارة المتبادلة مع منظِّم حراري]\label{prop:b1:second-law-entropy:thermostat}
جملة تتلقى الحرارة $Q$ من منظِّم حراري عند $T_0$ (مهما يجرِ
في داخلها) تكون لها $S_{\mathrm{exch}} = Q/T_0$، وتتغير إنتروبيا
المنظِّم نفسه بمقدار $-Q/T_0$: فالمنظِّم الحراري يتبادل الإنتروبيا على نحو عكوس.
\end{proposition}

\begin{proof}
الحدّ $T_{\mathrm{ext}} = T_0$ ثابت؛ وتتغير حالة المنظِّم الحراري
الداخلية تغيّرًا تفاضليًّا، فيكون ما يُنشئه مهمَلًا.
\end{proof}

\begin{example}[جسم ساخن في بحيرة باردة]\label{ex:b1:second-law-entropy:quench}
كتلة من الحديد ($m = \qty{1.0}{kg}$ و $c = \qty{450}{J/(kg.K)}$) عند $T_1 =
\qty{373}{K}$ تُلقى في بحيرة عند $T_0 = \qty{293}{K}$: فتنتهي عند $T_0$،
مع $\Delta S_{\text{الكتلة}} = mc\ln(T_0/T_1) = \qty{-109}{J/K}$ و $Q =
mc(T_0 - T_1) = \qty{-36}{kJ}$ تُعطى للبحيرة، ومنه $S_{\mathrm{exch}} =
Q/T_0 = \qty{-123}{J/K}$ و $S_{\mathrm{created}} = -109 + 123 = \qty{+14}{J/K}$.
فقد هبطت إنتروبيا الكتلة، وارتفعت إنتروبيا البحيرة بأكثر منها: فالسيرورة
لاعكوسة، والعدد يقيس بكم.
\end{example}

\begin{proposition}[نتيجتان]\label{prop:b1:second-law-entropy:clausiuskelvin}
\index{نصّ كلاوزيوس}\index{نصّ كلفن}
\begin{enumerate}
  \item \emph{(كلاوزيوس)} لا تجري الحرارة تلقائيًّا من جسم بارد
        إلى جسم أسخن منه: فمقدار $Q > 0$ ينتقل من $T_h$ إلى
        $T_c < T_h$ يُنشئ $S = Q(1/T_c - 1/T_h) > 0$؛ والعكس كان
        ليُفني إنتروبيا.
  \item \emph{(كلفن)} لا تستطيع آلة دورية أن تنتج شغلًا من منظِّم
        حراري واحد: ففي دور يفرض $\Delta S = 0 = Q/T_0 + S_{\mathrm{created}}$
        أن $Q \leq 0$، ومنه $W = -Q \geq 0$ --- فلا تستطيع الآلة إلا أن
        \emph{تتلقى} شغلًا وتطرح حرارة.
\end{enumerate}
\end{proposition}

\begin{proof}
كلتاهما هي الموازنة مطبَّقة على الجملة ``المنظِّمان الحراريان'' أو
``الآلة على امتداد دور''، مع $S$ دالةَ حالة بحيث يكون $\Delta S
= 0$ على امتداد دور.
\end{proof}

\begin{definition}[مخطط الإنتروبيا]\label{def:b1:second-law-entropy:TS}
\index{مخطط الإنتروبيا}
في المستوي $(S, T)$ يكون \omterm{def:b1:first-law:transformation}{التحول العكوس} منحنيًا، وتكون
الحرارة المتلقاة هي المساحة تحته: $Q_{\mathrm{rev}} = \int T\,\dd S$.
والمتساوي الحرارة العكوس أفقي، والمتساوي الإنتروبيا (أي الكظوم العكوس)
شاقولي؛ والدور العكوس يحصر مساحة تساوي الحرارة المتلقاة،
ومن ثمّ تساوي، بالمبدأ الأول، الشغل المقدَّم
(\cref{ch:b1:heat-engines}).
\end{definition}

\begin{method}[إقامة موازنة إنتروبيا]\label{met:b1:second-law-entropy:method}
\begin{enumerate}
  \item احسب $\Delta S$ للجملة من دوال الحالة
        (\cref{prop:b1:second-law-entropy:identity}): من الحالتين
        الابتدائية والنهائية لا غير.
  \item احسب $S_{\mathrm{exch}}$ من الحرارة المتلقاة فعلًا ومن
        \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} المنبع: أي $\sum Q_i/T_i$ للمنظِّمات الحرارية،
        و $\int\delta Q/T_{\mathrm{ext}}$ فيما عدا ذلك.
  \item استنتج $S_{\mathrm{created}} = \Delta S - S_{\mathrm{exch}}$؛ وعليها أن
        تكون $\geq 0$ --- وإلا كان التحول مستحيلًا (أو كان في الحساب
        خطأ)؛ وتكون $= 0$ في \omterm{def:b1:first-law:transformation}{التحول العكوس}.
\end{enumerate}
\end{method}

\begin{example}[مزج الساخن بالبارد]\label{ex:b1:second-law-entropy:mixing}
كتلتان متساويتان $m$ من الماء عند $T_1$ و $T_2$ تُمزجان في إناء عازل فتنتهيان عند
$(T_1 + T_2)/2$ (بالمبدأ الأول)؛ ويكون $\Delta S = mc[\ln\frac{T_f}{T_1} + \ln\frac{T_f}{T_2}]
= mc\ln\frac{(T_1 + T_2)^2}{4T_1T_2} \geq 0$ (لأن المتوسط الحسابي يفوق المتوسط
الهندسي)، بلا أيّ تبادل: فكلها مُنشأة. ومن أجل \qty{1}{kg} عند \qty{300}{K} و
\qty{400}{K}: تبلغ \qty{86}{J/K}. وتبيّن مسألة نهاية الأسبوع أن مزجهما
\emph{على نحو عكوس} --- عبر محرك مثالي --- كان ليتيح
استعادة شغل والانتهاء عند $\sqrt{T_1T_2}$.
\end{example}

\section{ما الذي تعدّه الإنتروبيا}

\begin{proposition}[صيغة بولتزمان]\label{prop:b1:second-law-entropy:boltzmann}
\index{صيغة بولتزمان}\index{حالة مجهرية}
يمكن أن تتحقق حالة عيانية (بمعلومية $U$ و $V$ و $N$) بعدد
$\Omega$ من التشكيلات المجهرية (\emph{\omterm{prop:b1:second-law-entropy:boltzmann}{الحالات المجهرية}}). وإنتروبيتها هي
\[
S = k_B\ln\Omega .
\]
تتطور جملة معزولة نحو الحالة العيانية التي لها، وبفارق هائل،
أكبر عدد من \omterm{prop:b1:second-law-entropy:boltzmann}{الحالات المجهرية}: فنموّ الإنتروبيا هو السير نحو ما
هو راجح رجحانًا ساحقًا.
\end{proposition}
\admitted

\begin{example}[لماذا لا يعود الغاز أبدًا]\label{ex:b1:second-law-entropy:count}
$N$ جزيئًا حرّةً في أن تكون في أيّ من نصفَي صندوق: $2^N$ ترتيبًا،
كلها متساوية الرجحان؛ وواحد منها فقط يضعها كلها في اليسار. ودعُ
غازًا محبوسًا في نصف واحد يتمدد يضرب $\Omega$ بالعامل $2^N$، ومنه $\Delta S =
Nk_B\ln2 = nR\ln2$ --- وهي بالضبط نتيجة \omterm{rem:b1:first-law:irreversible}{تمدد جول} أعلاه، ولها الآن
معنًى: أن للجزيئات $2^N$ ضعفًا من الطرائق في أن تكون. واحتمال
عودتها التلقائية هو $2^{-N}$: أي واحد من ألف لعشر
جزيئات، وواحد من $10^{30}$ لمئة، وللمول عدد فيه
$10^{23}$ صفرًا. فالمبدأ الثاني ليس منعًا؛ بل هو احتمال
من الجور بحيث لم يُضبط مرة واحدة وهو يخيب.
\end{example}

القمة نتوءًا عرضه النسبي $10^{-11}$: فالغاز المنتظم ليس قانونًا، بل
هو الشيء الوحيد الذي يمكن أن يُرى.}
\end{omfigure}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:1}
تغيّر الإنتروبيا في \qty{1.0}{kg} من الجليد ينصهر عند \qty{0}{\degreeCelsius}
($L_f = \qty{334}{kJ/kg}$)، وفي \qty{1.0}{kg} من الماء يغلي عند
\qty{100}{\degreeCelsius} ($L_v = \qty{2260}{kJ/kg}$). ولماذا يكون الثاني
أكبر بكثير؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:2}
تغيّر الإنتروبيا في \qty{1.0}{kg} من الماء يُسخَّن من \qty{20}{} إلى
\qty{80}{\degreeCelsius}. وهل يتوقف على الطريقة التي جرى بها التسخين؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:3}
\omterm{def:b1:kinetic-theory:scales}{مول} واحد من \omterm{def:b1:kinetic-theory:model}{غاز مثالي} يضاعف حجمه عند ثبات \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة}.
أوجد $\Delta S$؛ والإنتروبيا المتبادَلة والمُنشأة إن كان التمدد عكوسًا؛
ثم إن كان \omterm{rem:b1:first-law:irreversible}{تمدد جول} في الخلاء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:4}
بيّن انطلاقًا من $S(T, V)$ أن تحولًا كظومًا عكوسًا \omterm{def:b1:kinetic-theory:model}{لغاز
مثالي} يحقق $TV^{\gamma - 1} = \text{ثابت}$. وما تغيّر الإنتروبيا في
انضغاط كظوم \emph{مفاجئ} (\cref{exo:b1:first-law:11})؟
احسبه بأعداد ذلك التمرين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:5}
كتلة من الحديد وزنها \qty{1.0}{kg} ($c = \qty{450}{J/(kg.K)}$) عند \qty{100}{\degreeCelsius}
تُلقى في بحيرة عند \qty{20}{\degreeCelsius}. أوجد تغيّر الإنتروبيا في
الكتلة، وفي البحيرة، وفي الكون.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:6}
كتلتان من الماء وزن كلٍّ منهما \qty{1.0}{kg} عند \qty{300}{K} و\qty{400}{K} تُمزجان
كظومًا. أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} النهائية والإنتروبيا المُنشأة. وبيّن عمومًا أن
الإنتروبيا المُنشأة $mc\ln[(T_1 + T_2)^2/4T_1T_2]$ غير سالبة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:7}
تسرّب حراري قدره \qty{1.0}{kJ} يعبر جدارًا من غرفة عند \qty{300}{K} إلى
الخارج عند \qty{280}{K}. أوجد الإنتروبيا المُنشأة. ثم الحرارة نفسها من فرن
عند \qty{1000}{K} إلى الغرفة: قارن، وعلّق على الموضع الذي يكون فيه
``الضياع'' أكبر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:8}
برهن على نصّ كلفن انطلاقًا من موازنة إنتروبيا آلة تجري
دورًا في تماسّ مع منظِّم حراري واحد. ولماذا لا يستخرج محرك سفينة
الحرارة من البحر ببساطة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:9}
\omterm{def:b1:kinetic-theory:scales}{مول} من الآزوت \omterm{def:b1:kinetic-theory:scales}{ومول} من الأكسجين، عند $T$ و $P$ نفسيهما،
يفصل بينهما جدار في صندوق حجمه الكلي $2V$، يُتركان ليمتزجا. أوجد تغيّر
الإنتروبيا (وعامِل كل غاز على أنه يتمدد في الصندوق كله). وهل هي مُنشأة
أم متبادَلة؟ وماذا لو كان في الجانبين آزوت؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:10}
مصباح استطاعته \qty{100}{W} يشتغل ساعةً؛ وتنتهي طاقته كلها حرارةً في
غرفة عند \qty{293}{K}. أوجد الإنتروبيا المُنشأة. وأين أُنشئت
فيزيائيًّا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:11}
جسم سعته الحرارية $C$ عند $T_1$ يُبرَّد إلى $T_0 < T_1$: (أ) بتماسّ
مباشر مع منظِّم حراري عند $T_0$، (ب) على مرحلتين عبر منظِّم
حراري وسيط عند $\sqrt{T_1T_0}$، (ج) عبر متصلة من
المنظِّمات الحرارية. أوجد الإنتروبيا المُنشأة في كل حالة؛ واستنتج ما الذي
يقتضيه ``التبريد العكوس''.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:12}
احتمال أن تجلس $N$ جزيئًا كلها في النصف الأيسر من صندوق، من أجل
$N = 10$ ثم $100$ ثم $10^{22}$ (وعبّر عن الأخير بقوة للعشرة). وبيّن أن
$\Delta S = k_B\ln(2^N)$ يساوي $nR\ln2$ وعلّق على كلمتَي
``لاعكوس'' و``مستحيل''.
\end{exercise}

\section{مسألة: كوب القهوة وسهم الزمن}

\begin{problem}[{مسألة نهاية الأسبوع --- كوب يبرد، وحوض يُمزج،
وثلاجة وصندوق من الجزيئات: أربع موازنات إنتروبيا، وكم كانت
الإنتروبيا المُنشأة لتساوي شغلًا}]
\label{pb:b1:second-law-entropy:1}
الماء: $c = \qty{4180}{J/(kg.K)}$؛ والغرفة (وهي منظِّم حراري): $T_0 = \qty{293}{K}$؛
و $R = \qty{8.314}{J/(mol.K)}$ و $k_B = \qty{1.38e-23}{J/K}$.

\medskip
\noindent\textbf{الجزء I --- الكوب يبرد.}
كوب فيه \qty{0.25}{kg} من القهوة عند $T_1 = \qty{363}{K}$ يبرد إلى \omterm{def:b1:kinetic-theory:temperature}{درجة
حرارة} الغرفة.
\begin{enumerate}
  \item الحرارة المعطاة للغرفة، وتغيّر إنتروبيا الغرفة.
  \item تغيّر إنتروبيا القهوة.
  \item الإنتروبيا المُنشأة؛ وتحقّق من إشارتها.
  \item يُشرَب الكوب حين يبلغ \qty{330}{K}: فما الإنتروبيا المُنشأة حتى
        تلك اللحظة؟
  \item بيّن أنه من أجل أيّ $T_1 > T_0$ يكون $S_{\mathrm{created}} = mc[T_1/T_0 - 1 -
        \ln(T_1/T_0)] \geq 0$ (بدراسة $f(x) = x - 1 - \ln x$).
  \item تبرد القهوة أسرع إذا حُرّكت أو نُفخ عليها: فهل يغيّر ذلك
        الإنتروبيا المُنشأة؟ ولماذا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- المزج، رديئًا وجيدًا.}
كتلتان من الماء وزن كلٍّ منهما \qty{1.0}{kg} عند $T_1 = \qty{400}{K}$ و $T_2 =
\qty{300}{K}$، في إناء عازل.
\begin{enumerate}[resume]
  \item المزج المباشر: أوجد \omterm{def:b1:kinetic-theory:temperature}{درجة الحرارة} النهائية $T_f$ والإنتروبيا المُنشأة.
  \item وبدلًا من ذلك، يأخذ محرك مثالي الحرارة من الماء الساخن، ويطرح
        حرارة إلى البارد ويقدّم شغلًا، حتى يصير الاثنان عند \omterm{def:b1:kinetic-theory:temperature}{درجة
        الحرارة} $T_f'$ نفسها؛ والسيرورة كلها عكوسة. اكتب موازنة
        الإنتروبيا وبيّن أن $T_f' = \sqrt{T_1T_2}$.
  \item احسب $T_f'$ والشغل المقدَّم (بالمبدأ الأول على الكتلتين
        المائيتين).
  \item تحقّق عدديًّا من موازنة إنتروبيا المخطط العكوس: أن الإنتروبيا
        التي يفقدها الماء الساخن تساوي التي يكسبها البارد.
  \item قارن الشغل بالمقدار $T_0 \times S_{\mathrm{created}}$ من السؤال 6
        (وخذ الغرفة مرجعًا \omterm{def:b1:kinetic-theory:temperature}{لدرجة الحرارة} هنا، $T_0 \approx T_2$):
        فما الذي تقيسه الإنتروبيا المُنشأة في المزج المتهاون؟
  \item ولماذا لا يستعيد أيّ مطبخ ذلك الشغل؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الثلاجة.}
ثلاجة تُبقي داخلها عند $T_c = \qty{278}{K}$ في غرفة عند
$T_0$؛ وتتسرب إليها الحرارة بمعدل \qty{40}{W}، تضخّها الآلة إلى الخارج.
\begin{enumerate}[resume]
  \item على امتداد ثانية واحدة، اكتب موازنة إنتروبيا الآلة (وهي
        جهاز دوري): الإنتروبيا المتلقاة من الداخل البارد، والمعطاة
        للغرفة، والمُنشأة في الداخل.
  \item استنتج أصغر شغل في الثانية (أي \omterm{thm:b1:dc-circuits:kirchhoff}{الاستطاعة الكهربائية}) لازم،
        ومعامل الأداء الأعظمي الموافق
        $Q_c/W$.
  \item عبّر عن ذلك الحدّ بدلالة $T_c$ و $T_0$ وحدهما، وقدّره
        من أجل مجمّد عند $T_c = \qty{255}{K}$.
  \item ثلاجة حقيقية بهذا الحجم تسحب نحو \qty{15}{W}: فما الإنتروبيا
        المُنشأة في الثانية؟
  \item وإذا تُرك بابها مفتوحًا دخلت الحرارة بمعدل \qty{200}{W}:
        فما أصغر استطاعة عندئذ، ولماذا تدفئ ثلاجة مفتوحة غرفةً؟
  \item وماذا يقول المبدأ الثاني عن ثلاجة لا تحتاج إلى كهرباء
        البتة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- العدّ.}
صندوق يقسمه جدار نصفين متساويين؛ و $N$ جزيئًا من \omterm{def:b1:kinetic-theory:model}{غاز
مثالي} في اليسار. ثم يُرفع الجدار.
\begin{enumerate}[resume]
  \item عدد \omterm{prop:b1:second-law-entropy:boltzmann}{الحالات المجهرية} قبل ذلك وبعده (فلكل جزيء أن يكون في
        أيّ من النصفين)، و $\Delta S$ \omterm{prop:b1:second-law-entropy:boltzmann}{بصيغة بولتزمان}.
  \item تحقّق مقابل $\Delta S = nR\ln2$ من الصيغة الترموديناميكية
        \omterm{rem:b1:first-law:irreversible}{لتمدد جول}.
  \item من أجل $N = 50$، ما احتمال أن نجد الجزيئات كلها من جديد في
        اليسار عند لحظة عشوائية؟ وللمول، ما رتبة قدر ذلك
        الاحتمال بقوة للعشرة؟
  \item كم سؤالًا من نوع نعم/لا (أي كم بتًّا) يعيّن أيّ نصف يشغله كلٌّ من
        الجزيئات $N$؟ وقارن بالمقدار $\Delta S/(k_B\ln2)$.
  \item الإنتروبيا المتبادَلة والمُنشأة في \omterm{rem:b1:first-law:irreversible}{تمدد جول}؛ ولماذا هو المثال
        المدرسي على اللاعكوسية؟
  \item ثم يُضغط الغاز نفسه عائدًا على نحو عكوس ومتساوي الحرارة إلى
        النصف الأيسر: أوجد الشغل المتلقى، والحرارة المعطاة للغرفة،
        والإنتروبيا المتبادَلة؛ وما الذي يجب تقديمه لنقض التمدد الحرّ؟
  \item لخّص في ثلاثة أسطر: ما الذي يحفظه المبدأ الثاني (لا شيء)،
        وما الذي يمنعه، وما الذي تكلّفه الإنتروبيا المُنشأة.
\end{enumerate}
\end{problem}
