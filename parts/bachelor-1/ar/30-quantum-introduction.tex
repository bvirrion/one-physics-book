\chapter{مدخل إلى الفيزياء الكمومية}\label{ch:b1:quantum-introduction}

سلّط ضوءًا فوق بنفسجي على معدن نظيف تطر الإلكترونات
من فورها، بطاقة تتوقف على لون الضوء ولا تتوقف
البتّة على شدته؛ وسلّط ضوءًا أحمر فلا يخرج شيء مهما
يشتدّ المصباح. وأرسل إلكترونات واحدًا واحدًا عبر شقّين فتصل
نقاطًا، كلٌّ منها في موضع واحد --- وتتراكم النقاط، بالآلاف،
في هُدُب موجة. ولا تنسجم أيّ من الواقعتين مع فيزياء
الفصول التسعة والعشرين الأولى. ويقدّم هذا الفصل الأخير
الأفكار التي تنسجم معهما: الضوء يأتي كمّات، هي الفوتونات؛ وللمادة
طول موجة؛ وما ينتشر سعة مربّعُها احتمال؛ والموضع
\omterm{def:b1:newton-dynamics:momentum}{وكمية الحركة} لا يكونان حادّين معًا؛ والجسيم
المحصور لا يمكن أن تكون له إلا طاقات معينة --- ولهذا يكون للذرات
أحجام، وللأطياف خطوط، وتتوهج حبة من نصف ناقل قطرها بضعة
نانومترات بلون يعيّنه قطرها.

\section{الفوتون}

\begin{theorem}[علاقتا بلانك-أينشتاين]\label{thm:b1:quantum-introduction:photon}
\index{فوتون}\index{ثابت بلانك}
ضوء تواتره $\nu$ وطول موجته $\lambda$ لا يتبادل الطاقة مع
المادة إلا كمّات، هي \emph{الفوتونات}، طاقتها \omterm{def:b1:newton-dynamics:momentum}{وكمية حركتها}
\[
E = h\nu = \frac{hc}{\lambda} , \qquad p = \frac{h}{\lambda} = \frac{E}{c} ,
\]
حيث \emph{\omterm{thm:b1:quantum-introduction:photon}{ثابت بلانك}} $h = \qty{6.626e-34}{J.s}$ (و $\hbar = h/2\pi =
\qty{1.055e-34}{J.s}$). ومن المفيد: $hc = \qty{1240}{eV.nm}$ --- فيحمل فوتون
\qty{620}{nm} المقدارَ \qty{2}{eV}.
\end{theorem}
\admitted

\begin{proposition}[الأثر الكهرضوئي]\label{prop:b1:quantum-introduction:photoelectric}
\index{أثر كهرضوئي}\index{شغل الخروج}\index{كمون الإيقاف}
ضوء يسقط على معدن \emph{شغل خروجه} $W$ (وهي الطاقة
التي تربط أقلّ إلكتروناته ارتباطًا، وهي بضعة \omterm{def:b1:charged-particles:eV}{إلكترون-فولط}) ينتزع
إلكترونات طاقتها الحركية العظمى
\[
E_{k,\max} = h\nu - W ,
\]
ومن ثمّ لا تخرج إلا فوق \emph{العتبة} $\nu_0 = W/h$، فورًا، بطاقة
لا تتوقف على الشدة؛ وتعيّن الشدة \emph{عدد}
الإلكترونات في الثانية. و\emph{\omterm{prop:b1:quantum-introduction:photoelectric}{كمون الإيقاف}}
$V_s$ الذي يلغي التيار بالضبط يقيس $E_{k,\max} = eV_s$، ويكون
منحني $V_s$ بدلالة $\nu$ مستقيمًا ميله $h/e$ --- وهكذا
قاس ميليكان $h$.
\end{proposition}

\begin{proof}
يمتصّ كلَّ فوتون إلكترونٌ واحد، يحتفظ بالمقدار $h\nu - W$ على
أحسن تقدير؛ أما موجة فكانت لتقدّم الطاقة باستمرار إلى كل إلكترون،
فتتزايد الطاقة مع الشدة، ويحتاج ضوء ضعيف إلى ساعات
ليجمع $W$ على ذرة واحدة (\cref{pb:b1:quantum-introduction:1}) ---
ولا يُرصَد شيء من ذلك. وتفسير أينشتاين (1905) مسلَّم به
قانونًا.
\end{proof}

\begin{example}[رتب القدر]\label{ex:b1:quantum-introduction:orders}
فوتون مرئي: \qty{2}{eV}؛ وفوتون أشعة سينية طوله \qty{0.1}{nm}: \qty{12}{keV}؛
وفوتون راديوي عند \qty{100}{MHz}: \qty{4e-7}{eV}. وليزر أحمر استطاعته \qty{1}{mW}
يصدر $10^{-3}/(3.1 \times 10^{-19}) = 3 \times 10^{15}$ فوتون في الثانية؛
وهوائي راديوي يلتقط \qty{1}{pW} عند \qty{100}{MHz} يتلقى $10^{13}$
في الثانية --- وهي من الكثرة والصغر بحيث يكون وصف الموجة دقيقًا
لكل غرض؛ في حين تستطيع العين أن تسجّل وميضًا من
مئة فوتون. وتظهر طبيعة الضوء الكمومية حين تكون الفوتونات
قليلة أو ذات طاقة: في الخلايا الكهرضوئية، وكواشف الأشعة السينية، وحبيبات
صورة خافتة.
\end{example}

\begin{remark}[تبعثر كومبتون]\label{rem:b1:quantum-introduction:compton}
\index{تبعثر كومبتون}
فوتون يرتدّ عن إلكترون حرّ ينقل كمية حركة كما تفعل
كرة بلياردو: فيتزايد طول موجته بمقدار $\Delta\lambda = \frac{h}{m_ec}(1 -
\cos\theta)$، مع $h/m_ec = \qty{2.43}{pm}$ (وهو مسلَّم به --- إذ هو تصادم
نسبي، في السنة الثانية). وهو غير مرئي في الضوء ($\Delta\lambda/\lambda \sim 10^{-5}$)،
لكنه أثر بنسبة $20\%$ في الأشعة السينية، وكان التجربة التي أقنعت
المتشككين (1923) بأن الفوتون يحمل \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة} $h/\lambda$.
\end{remark}

\section{موجات المادة}

\begin{theorem}[دي بروي]\label{thm:b1:quantum-introduction:debroglie}
\index{طول موجة دي بروي}\index{موجة مادة}
جسيم كمية حركته $p$ ينتشر موجةً طول موجتها
\[
\lambda = \frac{h}{p} ,
\]
ويُظهر تبعًا لذلك انعراجًا وتداخلًا: إلكترونات على
بلورة (دافيسون وجيرمر، 1927)، وعبر شقّين (يونسون، 1961)،
ونترونات وذرات وجزيئات من ستين ذرة كربون. ومن أجل إلكترون
مسرَّع بالتوتر $U$: $\lambda = h/\sqrt{2m_eeU} = \qty{1.23}{nm}/\sqrt{U/\unit{V}}$.
\end{theorem}
\admitted

\begin{example}[لماذا لم نرَها قط]\label{ex:b1:quantum-introduction:wavelengths}
إلكترون طاقته \qty{100}{eV}: $\lambda = \qty{0.12}{nm}$، أي تباعد
الذرات في بلورة --- ومن هنا الانعراج، \omterm{def:b1:optical-instruments:microscope}{والمجهر}
الإلكتروني. ونترون حراري (\qty{0.025}{eV}): \qty{0.18}{nm}. وكرة
كتلتها \qty{1}{g} بسرعة \qty{1}{m/s}: $\lambda = \qty{7e-31}{m}$، أي أصغر $10^{20}$ مرة
من نواة: فلا شقّ ولا بلورة يستطيع أن يعرّجها أبدًا. و
طبيعة المادة الموجية شاملة؛ وطول موجتها هو ما يجعلها
مرئية عند الخفيف والبطيء وحدهما.
\end{example}

\begin{proposition}[الجسيمات المفردة تبني الهُدُب]\label{prop:b1:quantum-introduction:buildup}
\index{ازدواجية الموجة والجسيم}
إلكترونات تُرسَل واحدًا في كل مرة عبر شقّين (تونومورا، 1989) يصنع كلٌّ منها
نقطة واحدة على الشاشة، في موضع لا يمكن التنبؤ به؛
وتتراكم النقاط في شكل تداخل الشقّين لموجة
طول موجتها $h/p$. وإغلاق أحد الشقّين يمحو الهُدُب --- فليس
النمط صنيع إلكترونات يتداخل بعضها مع بعض، بل صنيع
موجة كل إلكترون تمرّ عبر الشقّين معًا. وكشف الشقّ الذي
مرّ منه الجسيم يمحو الهُدُب أيضًا.
\end{proposition}
\admitted

\begin{definition}[الدالة الموجية]\label{def:b1:quantum-introduction:wavefunction}
\index{دالة موجية}\index{كثافة الاحتمال}
تُوصَف حالة جسيم بما يسمّى \emph{الدالة الموجية}
$\psi(x, t)$ (عقدية)، بحيث يكون $|\psi(x, t)|^2\,\dd x$
احتمال إيجاد الجسيم بين $x$ و $x + \dd x$ حين
يُنظَر: فيكون $|\psi|^2$ \emph{كثافة احتمال}، ويكون $\int|\psi|^2\dd x =
1$. وتتجمع السعات (وهو التراكب): فإذا أدّى مسلكان إلى النقطة
نفسها كان $\psi = \psi_1 + \psi_2$ و $|\psi|^2 = |\psi_1|^2 + |\psi_2|^2 +
2\,\mathrm{Re}(\psi_1^\ast\psi_2)$ --- والحدّ الأخير هو التداخل.
ويعطي القياس نتيجةً من النتائج الممكنة، عشوائيًّا وفق
الاحتمالات التي تمليها السعات، ويترك الجسيم في
الحالة التي وُجد فيها.
\end{definition}

\begin{remark}[ما الذي يتموّج]\label{rem:b1:quantum-introduction:whatwaves}
$\psi$ ليست مجالًا فيزيائيًّا مثل $\vect E$: فهي لا تُقاس، ولا يُقاس إلا
مربّعها، وهي تصف أفضل معرفة ممكنة لدينا بموضع
إيجاد الجسيم. فالإلكترون الواحد ليس ممتدًّا على
الشاشة؛ بل يُكشَف كاملًا عند نقطة واحدة؛ والممتدّ هو
الاحتمال. وهذا هو مضمون النظرية، ولم يخفق قط
في اختبار.
\end{remark}

\begin{theorem}[مبدأ الارتياب لهايزنبرغ]\label{thm:b1:quantum-introduction:heisenberg}
\index{مبدأ الارتياب}
لا يمكن أن يكون موضع جسيم وكمية حركته على المحور نفسه
معرَّفَين تعريفًا حادًّا معًا: بل يحقق انحرافاهما المعياريان في أيّ حالة
\[
\Delta x\,\Delta p_x \geq \frac{\hbar}{2} .
\]
ومن ثمّ يكون لجسيم محصور في منطقة قياسها $\ell$ تشتت في \omterm{def:b1:newton-dynamics:momentum}{كمية
الحركة} لا يقلّ عن $\hbar/2\ell$، ومنه \omterm{thm:b1:work-and-energy:ke}{طاقة حركية} من رتبة
$\hbar^2/2m\ell^2$ لا يزيلها أيّ تبريد: فالحصر يكلّف طاقة.
\end{theorem}
\admitted

\begin{example}[حجم ذرة الهيدروجين]\label{ex:b1:quantum-introduction:hsize}
\index{نصف قطر بور}
إلكترون على بعد $r$ من البروتون له $p \gtrsim \hbar/r$ وطاقة
$E(r) \approx \dfrac{\hbar^2}{2m_er^2} - \dfrac{e^2}{4\pi\varepsilon_0r}$: فالحدّ الأول
ينفجر إذا صغرت $r$، والثاني إذا كبرت. ويتحقق $\dd E/\dd r = 0$ عند
\[
a_0 = \frac{4\pi\varepsilon_0\hbar^2}{m_ee^2} = \qty{53}{pm} , \qquad
E(a_0) = -\frac{m_ee^4}{2(4\pi\varepsilon_0)^2\hbar^2} = \qty{-13.6}{eV} :
\]
وهما \emph{\omterm{ex:b1:quantum-introduction:hsize}{نصف قطر بور}} وطاقة تأيّن الهيدروجين، إلى
آخر رقم. ولم تستطع الفيزياء الكلاسيكية أن تفسّر لماذا لا يلولب الإلكترون
ساقطًا في النواة (إذ يشعّ) ولا لماذا تتطابق كل ذرات
عنصر واحد؛ ويجيب \omterm{thm:b1:quantum-introduction:heisenberg}{مبدأ الارتياب} عن السؤالين في سطر
واحد: فالذرة أصغر ما يمكن أن تكون من دون أن تفوق \omterm{thm:b1:work-and-energy:ke}{الطاقة الحركية}
طاقةَ الارتباط.
\end{example}

\section{تكميم الطاقة}

\begin{proposition}[جسيم في علبة]\label{prop:b1:quantum-introduction:box}
\index{بئر لانهائي}\index{تكميم}\index{حالة أساسية}
جسيم كتلته $m$ محصور بين جدارين لا يُخترقان يفترقان
بالمسافة $L$ لا يقبل، مثل وتر ملدة
(\cref{ch:b1:wave-propagation})، إلا \omterm{thm:b1:wave-propagation:standing}{الموجات المستقرة} $\lambda_n = 2L/n$،
ومن ثمّ إلا كميات الحركة $p_n = nh/2L$ والطاقات
\[
E_n = \frac{n^2h^2}{8mL^2} , \qquad n = 1, 2, 3, \ldots
\]
بدوال موجية $\psi_n(x) \propto \sin(n\pi x/L)$. فالطاقة
\emph{مكمّاة}؛ وأدناها، وهي \emph{\omterm{prop:b1:quantum-introduction:box}{الحالة الأساسية}} $E_1 > 0$، هي
طاقة الحصر في \omterm{thm:b1:quantum-introduction:heisenberg}{مبدأ الارتياب} مضبوطةً؛ ويقفز
الجسيم بين المستويات بامتصاص فوتون طاقته $h\nu = E_p - E_n$ أو
بإصداره.
\end{proposition}

\begin{proof}
يجب أن تنعدم $\psi$ عند الجدارين (إذ لا يمكن للجسيم أن يكون هناك)، فتنطبق
على $L$ أنصافُ أطوال موجة بعدد صحيح --- وهو شرط الوتر؛
ويعطي دي بروي $p$، ويكون $E = p^2/2m$. أما كون $\psi$ يحقق
معادلة موجية بهذه الشروط الحدّية فمسلَّم به (وهي
معادلة شرودنغر، في السنة الثانية).
\end{proof}

\begin{proposition}[ذرة الهيدروجين]\label{prop:b1:quantum-introduction:hydrogen}
\index{ذرة الهيدروجين}\index{مستويات الطاقة}\index{خطوط طيفية}
لإلكترون \omterm{prop:b1:quantum-introduction:hydrogen}{ذرة الهيدروجين} الطاقات المنفصلة
\[
E_n = -\frac{\qty{13.6}{eV}}{n^2} , \qquad n = 1, 2, 3, \ldots ,
\]
حيث $n = 1$ \omterm{prop:b1:quantum-introduction:box}{الحالة الأساسية} و $E \geq 0$ الذرة المؤيَّنة. ويتألف
طيفها من خطوط عند $h\nu = E_p - E_n$: فسلسلة ليمان نزولًا
إلى $n = 1$ (فوق بنفسجية، \qty{122}{nm} من أجل $2 \to 1$)، وسلسلة بالمر
نزولًا إلى $n = 2$ (مرئية: \qty{656}{nm} و\qty{486}{nm} و\qty{434}{nm}، \ldots)،
وهكذا. ويحصل نموذج بور (1913) على هذه الطاقات بتكميم
\omterm{def:b1:angular-momentum:L}{العزم الحركي} \omterm{prop:b1:central-forces:effective}{لمدار دائري}، $m_evr = n\hbar$
(\cref{exo:b1:quantum-introduction:9})؛ وتستبدل النظرية التامّة بالمدارات
دوالَّ موجية لكنها تُبقي الطاقات.
\end{proposition}
\admitted

\begin{example}[ألوان من الحجم: النقطة الكمومية]\label{ex:b1:quantum-introduction:dot}
\index{نقطة كمومية}
بلورة نانوية من نصف ناقل قطرها بضعة نانومترات تحصر
إلكتروناتها كعلبة: فكلما صغرت البلورة كبرت
طاقة الحصر المضافة إلى فجوة نصف الناقل، وازداد ازرقاق
الضوء الذي تصدره. ونقاط سيلينيد الكادميوم نصف قطرها \qty{2}{nm} تتوهج زرقاء،
و\qty{3}{nm} صفراء، و\qty{4}{nm} حمراء --- مادة واحدة، وألوان تُختار
بمسطرة (\cref{pb:b1:quantum-introduction:1}). وتستعملها شاشات التلفاز.
\end{example}

\begin{remark}[ما تضيفه بقية النظرية]\label{rem:b1:quantum-introduction:rest}
\index{أثر نفقي}\index{ليزر}
تجعل معادلة شرودنغر العلبةَ والذرةَ والجزيءَ
مضبوطةً؛ وتتنبأ بأن جسيمًا يستطيع عبور حاجز لا يملك من
الطاقة ما يتسلقه (وهو \emph{\omterm{rem:b1:quantum-introduction:rest}{الأثر النفقي}}، وراء النشاط الإشعاعي و
\omterm{def:b1:optical-instruments:microscope}{المجهر} النفقي الماسح)؛ وبأن الجسيمات المتطابقة إما اجتماعية
(بوزونات --- ومن هنا \emph{الليزر}، حيث تتكدس الفوتونات في نمط
واحد) وإما حصرية (فرميونات --- ومن هنا الجدول الدوري و
صلابة المادة)؛ وبأن سبين الإلكترون \omterm{def:b1:magnetostatics:moment}{عزم مغناطيسي}
لا نظير كلاسيكيًّا له. وكل ذلك في السنة الثانية وما بعدها؛
وكله قائم على الأفكار القليلة في هذا الفصل.
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:1}
أوجد طاقة فوتون من: راديو عند \qty{100}{MHz}؛ وضوء عند \qty{500}{nm}؛
وأشعة سينية عند \qty{0.10}{nm}؛ وأشعة غاما طاقتها \qty{1}{MeV} (وأعطِ
طول موجتها). ثم عدد الفوتونات في الثانية من ليزر استطاعته \qty{1.0}{mW} عند
\qty{633}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:2}
الصوديوم، $W = \qty{2.3}{eV}$: أوجد طول موجة العتبة؛ \omterm{thm:b1:work-and-energy:ke}{والطاقة الحركية} العظمى
\omterm{prop:b1:quantum-introduction:photoelectric}{وكمون الإيقاف} من أجل ضوء \qty{400}{nm}؛ وما الذي يتغير إذا تضاعفت
الشدة؟ وماذا مع ضوء \qty{600}{nm}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:3}
\omterm{thm:b1:quantum-introduction:debroglie}{طول موجة دي بروي} لإلكترون طاقته \qty{100}{eV}، ثم \qty{1.0}{keV}؛
ولنترون طاقته \qty{0.025}{eV}؛ ولكرة كتلتها \qty{1}{g} بسرعة \qty{1}{m/s}.
وأيّها يمكن أن تعرّجه بلورة تباعدها \qty{0.2}{nm}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:quantum-introduction:4}
إلكترون في علبة قياسها \qty{1.0}{nm}: أوجد $E_1$ و $E_2$، وطول موجة
الفوتون المنبعث في الانتقال $2 \to 1$. ثم بروتون في علبة قياسها \qty{5}{fm}
(أي نواة): أوجد $E_1$ بوحدة \unit{MeV}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:5}
كمونات إيقاف مقيسة من أجل معدن: \qty{0.40}{V} عند
\qty{6.0e14}{Hz}، و\qty{1.23}{V} عند \qty{8.0e14}{Hz}، و\qty{2.05}{V} عند
\qty{1.0e15}{Hz}. استنتج $h$ (مع معلومية $e$) \omterm{prop:b1:quantum-introduction:photoelectric}{وشغل الخروج}؛
وطول موجة العتبة؛ وأيّ معدن قد يكون (السيزيوم \qty{2.1}{eV}،
والبوتاسيوم \qty{2.3}{eV}، والزنك \qty{4.3}{eV})؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:6}
كومبتون: فوتون أشعة سينية طوله \qty{10}{pm} يتبعثر بزاوية \qty{90}{\degree} عن
إلكترون. أوجد \omterm{prop:b1:wave-propagation:sinusoidal}{طول الموجة} الجديد؛ وطاقتَي الفوتون قبل وبعد؛ والطاقة
المعطاة للإلكترون. ثم الإزاحة النسبية من أجل ضوء \qty{500}{nm} --- ولماذا
احتاج كومبتون إلى الأشعة السينية.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:7}
\emph{تجربة يونسون.} إلكترونات طاقتها \qty{50}{keV} (استعمل $p =
\sqrt{2m_eE}$)، وشقّان يفترقان \qty{2.0}{\micro m}، والشاشة على
\qty{35}{cm}. أوجد \omterm{prop:b1:wave-propagation:sinusoidal}{طول الموجة}؛ وتباعد الهُدُب؛ ولماذا لزمت عدسة إلكترونية
مكبّرة. وفي صيغة تونومورا يعبر الجهاز إلكترون واحد في كل مرة:
فماذا تُظهر الشاشة بعد 10 إلكترونات، وبعد $10^5$،
وما الذي يعلّمه ذلك؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:8}
أوجد أصغر \omterm{thm:b1:work-and-energy:ke}{طاقة حركية}، \omterm{thm:b1:quantum-introduction:heisenberg}{بمبدأ الارتياب}، من أجل: إلكترون
محصور في \qty{0.1}{nm} (أي ذرة)؛ وإلكترون محصور في
\qty{5}{fm} (أي نواة --- واستعمل $E \approx pc$ إن كانت النتيجة
نسبية، واستنتج هل تستطيع النوى أن تحوي إلكترونات)؛ وحبة
غبار كتلتها \qty{1}{\micro g} محدَّدة إلى \qty{1}{\micro m} (وأعطِ $\Delta v$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:quantum-introduction:9}
\emph{نموذج بور.} إلكترون على \omterm{prop:b1:central-forces:effective}{مدار دائري} نصف قطره $r$ حول
بروتون: (أ) أوجد السرعة والطاقة بدلالة $r$ (بديناميك كولوم،
\cref{ch:b1:central-forces})؛ (ب) وافرض $m_evr = n\hbar$ واستنتج
$r_n = n^2a_0$ و $E_n = E_1/n^2$ مع $a_0$ و $E_1$ كما في
\cref{ex:b1:quantum-introduction:hsize}؛ (ج) وأوجد $v_1$ و $v_1/c$؛ (د) و
طولَي موجتَي ليمان $\alpha$ (أي $2 \to 1$) و H$\alpha$ (أي $3 \to 2$)؛
(ه) وبيّن أن المدار $n$ يسع بالضبط $n$ من أطوال موجة دي بروي.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:10}
\emph{الجمل الشبيهة بالهيدروجين.} (أ) من أجل نواة شحنتها $Ze$ بيّن أن
$E_n = -Z^2 \times \qty{13.6}{eV}/n^2$ و $r_n = n^2a_0/Z$: وأوجد طاقة تأيّن
He$^+$ وطاقة تأيّن آخر إلكترون في اليورانيوم ($Z = 92$؛ وعلّق).
(ب) والهيدروجين الميوني: كتلة الميون $207$ ضعف كتلة الإلكترون:
أوجد نصف القطر وطاقة الأساس؛ وطول موجة خطّه ليمان $\alpha$؛ ولماذا
يُستعمل لقياس حجم البروتون (نصف قطره \qty{0.84}{fm}).
(ج) والبوزيترونيوم (إلكترون وبوزيترون، بكتلتين متساويتين): بيّن أن
\omterm{prop:b1:systems-of-points:twobody}{الكتلة المختزلة} تنصّف الطاقات: وأوجد خطّه ليمان $\alpha$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:11}
\emph{\omterm{ex:b1:quantum-introduction:dot}{النقاط الكمومية}.} في بلورة نانوية كروية نصف قطرها $R$ تكون
طاقة حصر جسيم كتلته الفعّالة $m^\ast$ هي
$h^2/8m^\ast R^2$ (مسلَّم به). وفي سيلينيد الكادميوم: الفجوة $E_g = \qty{1.74}{eV}$ و $m_e^\ast =
0.13\,m_e$ و $m_h^\ast = 0.45\,m_e$ (كما أن ``الثقب'' الذي يتركه الإلكترون
المثار محصور هو أيضًا). (أ) أوجد طاقة الفوتون المنبعث $E_g + E_e + E_h$
وطول موجته من أجل $R = \qty{2.0}{nm}$ و\qty{3.0}{nm} و\qty{4.0}{nm}؛ والألوان.
(ب) ونصف القطر الموافق لانبعاث عند \qty{520}{nm} (أخضر). (ج) ولماذا لا يصدر
سيلينيد الكادميوم الكتلي إلا عند \qty{713}{nm}؟ (د) وتشتت قدره \qty{10}{\percent} في نصف القطر في
دفعة: أوجد تشتت \omterm{prop:b1:wave-propagation:sinusoidal}{طول الموجة} عند \qty{3}{nm}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:quantum-introduction:12}
\emph{طاقة النقطة الصفرية.} (أ) من أجل \omterm{def:b1:oscillators-resonance:harmonic}{هزّاز توافقي} $E = p^2/2m +
\tfrac12m\omega^2x^2$، خذ $\Delta x\,\Delta p = \hbar/2$ وصغّر $E \approx
\Delta p^2/2m + \tfrac12m\omega^2\Delta x^2$ بالنسبة إلى $\Delta x$: وبيّن أن $E_{\min} = \hbar\omega/2$
(وهي طاقة الأساس المضبوطة). (ب) ويهتزّ جزيء H$_2$ عند $\omega =
\qty{8.3e14}{rad/s}$: أوجد طاقة النقطة الصفرية \omterm{def:b1:charged-particles:eV}{بالإلكترون-فولط}، وأدنى
\omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} يكون عندها اهتزازه ``حراريًّا'' ($k_BT \sim \hbar\omega$).
(ج) والهيليوم السائل لا يتجمد أبدًا عند الضغط الجوي: قدّر
طاقة النقطة الصفرية لذرة هيليوم محدَّدة إلى $\Delta x = \qty{0.05}{nm}$
(وهي جزء من المسافة بين الذرات) في بلورة، وقارنها
بطاقة ارتباطها في الصلب، وهي نحو \qty{1}{meV}. (د) وقارن صيغة العلبة $E_1 = h^2/8mL^2$
بحدّ الارتياب من أجل $\Delta x = L/2$: أيّهما أكبر، ولماذا يجب أن يكون كذلك؟
\end{exercise}

\section{مسألة: طيف الهيدروجين ونقطة كمومية}

\begin{problem}[{مسألة نهاية الأسبوع --- قياس $h$ بضوء و
فولطمتر، وبناء ذرة الهيدروجين من ثلاثة ثوابت، و
اختيار لون بلورة بحجمها}]
\label{pb:b1:quantum-introduction:1}
الثوابت: $h = \qty{6.626e-34}{J.s}$ و $\hbar = \qty{1.055e-34}{J.s}$ و $c =
\qty{3.00e8}{m/s}$ و $e = \qty{1.602e-19}{C}$ و $m_e = \qty{9.11e-31}{kg}$
و $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$ و $hc = \qty{1240}{eV.nm}$.

\medskip
\noindent\textbf{الجزء I --- قياس \omterm{thm:b1:quantum-introduction:photon}{ثابت بلانك}.}
يُضاء مهبط ضوئي من البوتاسيوم بضوء وحيد اللون؛ ويُقاس \omterm{prop:b1:quantum-introduction:photoelectric}{كمون
الإيقاف}: \qty{0.30}{V} عند \qty{6.0e14}{Hz}، و\qty{0.69}{V} عند
\qty{7.0e14}{Hz}، و\qty{1.13}{V} عند \qty{8.0e14}{Hz}، و\qty{1.50}{V} عند
\qty{9.0e14}{Hz}، و\qty{2.36}{V} عند \qty{1.1e15}{Hz}.
\begin{enumerate}
  \item صِف الخلية وفسّر لماذا يستطيع توتر عكسي أن يوقف
        التيار، وما الذي يقيسه $eV_s$.
  \item اكتب علاقة أينشتاين وقل ما الذي يعطيه ميل
        منحني $V_s(\nu)$ وتقاطعه.
  \item استنتج $h$ من المعطيات (بموافقة خطية، أو بالنقطتين
        الطرفيتين)؛ وقارنه بالقيمة المعتمدة.
  \item شغل خروج البوتاسيوم \omterm{def:b1:charged-particles:eV}{بالإلكترون-فولط}؛ وتواتر العتبة وطولها
        الموجي؛ وهل يستطيع مؤشر ليزر أحمر انتزاع إلكترونات؟
  \item وتتضاعف شدة المصباح: فماذا يحدث للكمون $V_s$ وللتيار؟
        ولماذا لا ينسجم ذلك مع صورة موجية خالصة؟
  \item التقدير الكلاسيكي: ضوء شدته \qty{1}{\micro W/m^2} يسقط على
        المعدن؛ وتعرض ذرة مساحة نحو \qty{1e-20}{m^2}: فكم من الوقت
        كان يلزم لتجميع \qty{2.2}{eV} على ذرة واحدة؟ وما الذي
        يُرصَد بدل ذلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- \omterm{prop:b1:quantum-introduction:hydrogen}{ذرة الهيدروجين} عند بور.}
\begin{enumerate}[resume]
  \item إلكترون على \omterm{prop:b1:central-forces:effective}{مدار دائري} نصف قطره $r$ حول بروتون
        ثابت: أوجد السرعة $v(r)$ \omterm{thm:b1:work-and-energy:em}{والطاقة الميكانيكية} $E(r)$.
  \item شرط بور $m_evr = n\hbar$: بيّن أن $r_n = n^2a_0$ وأعطِ
        $a_0 = 4\pi\varepsilon_0\hbar^2/m_ee^2$ عدديًّا.
  \item $E_n = E_1/n^2$ مع $E_1 = -m_ee^4/[2(4\pi\varepsilon_0)^2\hbar^2]$؛
        وأوجد قيمتها العددية \omterm{def:b1:charged-particles:eV}{بالإلكترون-فولط}.
  \item السرعة على المدار الأول والنسبة $v_1/c$ (وهي
        ثابت البنية الدقيقة $\alpha \approx 1/137$).
  \item أطوال موجات الخطوط $2 \to 1$ (ليمان $\alpha$) و $3 \to 2$ و
        $4 \to 2$ (بالمر H$\alpha$ و H$\beta$): وأيّها مرئي؟
  \item بيّن أن $1/\lambda = R_H(1/n^2 - 1/p^2)$ واحسب ثابت
        ريدبرغ $R_H$؛ وأقصر طول موجة في بالمر.
  \item طاقة تأيّن الهيدروجين من \omterm{prop:b1:quantum-introduction:box}{الحالة الأساسية}؛ وطول موجة
        الفوتون الذي يؤيّنه بالضبط.
  \item دي بروي: بيّن أن محيط المدار $n$ يساوي $n$ من
        أطوال الموجة --- فشرط بور شرط \omterm{thm:b1:wave-propagation:standing}{موجة مستقرة}.
  \item شيئان يخطئ فيهما النموذج، وشيء يصيبه بالضبط،
        وما الذي يحلّ محلّ المدار في النظرية التامّة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- \omterm{ex:b1:quantum-introduction:dot}{النقطة الكمومية}.}
بلورة نانوية من سيلينيد الكادميوم نصف قطرها $R$ تحصر إلكترونًا مثارًا
(كتلته الفعّالة $m_e^\ast = 0.13\,m_e$) والثقب الذي يتركه
($m_h^\ast = 0.45\,m_e$)؛ وطاقة حصر كلٍّ منهما في كرة
نصف قطرها $R$ هي $h^2/8m^\ast R^2$ (مسلَّم به)؛ وفجوة سيلينيد الكادميوم الكتلي
$E_g = \qty{1.74}{eV}$.
\begin{enumerate}[resume]
  \item اشتقّ الطاقات $E_n = n^2h^2/8mL^2$ لجسيم في علبة
        وحيدة البعد من شرط \omterm{thm:b1:wave-propagation:standing}{الموجة المستقرة}، ثم
        علّق على المماثلة مع صيغة الكرة.
  \item طاقتا حصر الإلكترون والثقب من أجل
        $R = \qty{2.0}{nm}$.
  \item طاقة الفوتون المنبعث وطول موجته من أجل $R = \qty{2.0}{nm}$
        و\qty{3.0}{nm} و\qty{4.0}{nm}؛ والألوان.
  \item فسّر في جملة واحدة لماذا تكون النقاط الأصغر أكثر زرقة.
  \item تحقّق من \omterm{def:b1:units-dimensions:oom}{رتبة القدر} \omterm{thm:b1:quantum-introduction:heisenberg}{بمبدأ الارتياب}:
        $\Delta p \sim \hbar/R$ للإلكترون عند $R = \qty{2}{nm}$، \omterm{thm:b1:work-and-energy:ke}{والطاقة
        الحركية} التي يقتضيها ذلك.
  \item نقطة تصدر \qty{1.0}{nW} عند \qty{577}{nm}: أوجد عدد الفوتونات في الثانية.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- عدّ الكمّات.}
\begin{enumerate}[resume]
  \item عدد الفوتونات في الثانية من ليزر استطاعته \qty{1.0}{mW} عند \qty{633}{nm}.
  \item تكشف العين المتكيّفة مع الظلام وميضًا من نحو $100$ فوتون عند
        \qty{500}{nm} تصل في غضون \qty{0.1}{s}: أوجد الاستطاعة الموافقة.
  \item هوائي إذاعي يتلقى \qty{1.0}{pW} عند \qty{100}{MHz}: أوجد عدد الفوتونات في
        الثانية؛ وهل من أمل في كشفها واحدًا واحدًا؟
  \item لخّص: انطلاقًا من $h$ و $e$ و $m_e$ وحدها، أيّ ثلاثة مقادير من
        العالم الذري أنتجتها هذه المسألة، وبأيّ
        رتب؟
\end{enumerate}
\end{problem}
