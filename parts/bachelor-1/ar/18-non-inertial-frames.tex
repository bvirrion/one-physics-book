\section{تغيير المرجع: التحريك}

\begin{definition}[الحركة النسبية؛ مرجع في انسحاب، ومرجع في دوران]\label{def:b1:non-inertial-frames:frames}
\index{حركة نسبية}\index{مرجع في انسحاب}\index{مرجع في دوران}
ليكن $\mathcal R$ مرجعًا (مبدؤه $O$) و $\mathcal R'$ مرجعًا آخر (مبدؤه
$O'$) يتحرك بالنسبة إليه. ويكون $\mathcal R'$ \emph{في انسحاب} إذا
حافظت محاوره على اتجاهات ثابتة في $\mathcal R$ (فلكل نقطة من $\mathcal R'$
سرعة $O'$)؛ ويكون \emph{في دوران حول محور ثابت}
$\Delta$ إذا كان $\Delta$ ثابتًا في المرجعين ودار $\mathcal R'$ حوله
\omterm{cor:b1:point-kinematics:circular}{بالسرعة الزاوية} $\Omega(t)$. أما سرعة نقطة $M$ وتسارعها
\emph{النسبيان} إلى $\mathcal R'$، أي $\vect v'$ و $\vect a'$، فهما ما
يقيسه مراقب ساكن في $\mathcal R'$.
\end{definition}

\begin{theorem}[تركيب السرعات والتسارعات]\label{thm:b1:non-inertial-frames:composition}
\index{سرعة الجرّ}\index{تسارع الجرّ}\index{تسارع كوريوليس}
\begin{itemize}
  \item إذا كان $\mathcal R'$ في انسحاب بالسرعة $\vect v_{O'}$ والتسارع
        $\vect a_{O'}$:
        \[
        \vect v = \vect v' + \vect v_{O'}, \qquad \vect a = \vect a' + \vect a_{O'} .
        \]
  \item وإذا كان $\mathcal R'$ يدور حول المحور الثابت $\Delta$ بالسرعة $\Omega$ (وبالمتجهة
        $\vect\Omega = \Omega\vect e_z$ على امتداد المحور)، مع $H$ مسقط
        النقطة $M$ على المحور:
        \[
        \vect v = \vect v' + \vect\Omega\wedge\vect{HM}, \qquad
        \vect a = \vect a' + \underbrace{\big(-\Omega^2\vect{HM} + \dot\Omega\,\vect e_z\wedge\vect{HM}\big)}_{\vect a_e}
        + \underbrace{2\vect\Omega\wedge\vect v'}_{\vect a_C} .
        \]
\end{itemize}
حيث $\vect a_e$ هو \emph{\omterm{thm:b1:non-inertial-frames:composition}{تسارع الجرّ}} (أي تسارع
نقطة $\mathcal R'$ المنطبقة على $M$: وهو جاذب مركزي نحو المحور
في الدوران المنتظم)، و $\vect a_C = 2\vect\Omega\wedge\vect v'$ هو
\emph{\omterm{thm:b1:non-inertial-frames:composition}{تسارع كوريوليس}}، ولا يوجد إلا حين تتحرك $M$ بالنسبة إلى
المرجع الدوّار.
\end{theorem}

\begin{proof}
للانسحاب: $\vect{OM} = \vect{OO'} + \vect{O'M}$ مع مركّبات
$\vect{O'M}$ في الأساس المشترك؛ ثم اشتقّ مرتين. وللدوران: استعمل
\omterm{def:b1:point-kinematics:cylindrical}{الإحداثيات الأسطوانية} حول $\Delta$ في $\mathcal R$، أي $(r, \theta, z)$،
وفي $\mathcal R'$، أي $(r, \theta', z)$ مع $\theta = \theta' + \int\Omega\,\dd t$.
ثم طبّق \cref{thm:b1:point-kinematics:cylindrical} في $\mathcal R$ مع
$\dot\theta = \dot\theta' + \Omega$:
$\vect v = \dot r\vect e_r + r(\dot\theta' + \Omega)\vect e_\theta + \dot z\vect e_z
= \vect v' + r\Omega\vect e_\theta$، و $r\Omega\vect e_\theta = \vect\Omega\wedge\vect{HM}$.
وللتسارع، $\ddot\theta = \ddot\theta' + \dot\Omega$:
$\vect a = [\ddot r - r(\dot\theta' + \Omega)^2]\vect e_r + [r(\ddot\theta' + \dot\Omega)
+ 2\dot r(\dot\theta' + \Omega)]\vect e_\theta + \ddot z\vect e_z$؛ وبالنشر
والتجميع، $\vect a = \vect a' - r\Omega^2\vect e_r + r\dot\Omega\vect e_\theta
+ 2\Omega(-r\dot\theta'\vect e_r + \dot r\vect e_\theta)$، والقوس الأخير هو
$2\vect\Omega\wedge\vect v'$ لأن $\vect e_z\wedge\vect e_r = \vect e_\theta$ و
$\vect e_z\wedge\vect e_\theta = -\vect e_r$.
\end{proof}

\begin{example}[المشي على قرص دوّار]\label{ex:b1:non-inertial-frames:turntable}
حسب \cref{exo:b1:point-kinematics:12} تسارع شخص
يمشي قطريًّا بالسرعة $u$ على قرص يدور بالسرعة $\Omega$: أي $-ut\Omega^2\vect e_r
+ 2u\Omega\vect e_\theta$. وباللغة الجديدة: التسارع النسبي معدوم،
\omterm{thm:b1:non-inertial-frames:composition}{وتسارع الجرّ} $-r\Omega^2\vect e_r$، \omterm{thm:b1:non-inertial-frames:composition}{وتسارع كوريوليس} $2\vect\Omega\wedge u\vect e_r = 2u\Omega\vect e_\theta$.
\end{example}

\section{الديناميك في مرجع غير عطالي: القوى العطالية}

\begin{theorem}[قانون نيوتن في مرجع غير عطالي]\label{thm:b1:non-inertial-frames:dynamics}
\index{قوة عطالية}\index{قوة طاردة مركزية}\index{قوة كوريوليس}
في مرجع $\mathcal R'$ غير عطالي، تحقق حركة جسيم
العلاقة
\[
m\vect a' = \sum\vect F + \vect F_e + \vect F_C, \qquad
\vect F_e = -m\vect a_e, \qquad \vect F_C = -m\vect a_C = -2m\vect\Omega\wedge\vect v' :
\]
أي إن قانون نيوتن الثاني يصحّ شريطة أن يُضاف إلى القوى الحقيقية
\emph{قوة الجرّ} (وهي، من أجل مرجع يدور دورانًا منتظمًا،
\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{القوة الطاردة المركزية}} $+m\Omega^2\vect{HM}$، الموجَّهة بعيدًا عن
المحور؛ ومن أجل \omterm{def:b1:non-inertial-frames:frames}{مرجع في انسحاب}، $-m\vect a_{O'}$) و\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{قوة
كوريوليس}}. وهذه \emph{القوى العطالية} ليست تآثرات: فلا جسم
يؤثر بها، ولا ردّ فعل لها، وتتلاشى في \omterm{thm:b1:newton-dynamics:laws}{مرجع
عطالي}.
\end{theorem}

\begin{proof}
$m\vect a = \sum\vect F$ في \omterm{thm:b1:newton-dynamics:laws}{المرجع العطالي} $\mathcal R$؛ ثم عوّض
$\vect a = \vect a' + \vect a_e + \vect a_C$ وانقل الحدّين إلى الطرف
الآخر.
\end{proof}

\begin{example}[الراكب المائل، والشاقول]\label{ex:b1:non-inertial-frames:bus}
في حافلة تكبح بالتسارع $a = \qty{3.0}{m/s^2}$، يشعر مقبض معلّق، في
مرجع الحافلة، بالثقالة مضافًا إليها \omterm{thm:b1:non-inertial-frames:dynamics}{قوة عطالية} أمامية $ma$: فيتدلّى
على امتداد \emph{الثقالة الظاهرية} $\vect g' = \vect g - \vect a_{O'}$، مائلًا
إلى الأمام بالزاوية $\arctan(a/g) = 17^\circ$. والراكب الذي يريد الوقوف
ساكنًا عليه أن يميل نحو ذلك الشاقول الظاهري. وفي مصعد يسقط
سقوطًا حرًّا يكون $\vect g' = \vect 0$: أي انعدام الوزن، منظورًا من الداخل.
\end{example}

\begin{example}[الدلو الدوّار]\label{ex:b1:non-inertial-frames:bucket}
الماء في دلو يدور بالسرعة $\Omega$ حول محوره الشاقولي ساكن
في المرجع الدوّار تحت الثقالة \omterm{thm:b1:non-inertial-frames:dynamics}{والقوة الطاردة المركزية}: فسطحه
الحرّ عمودي على الثقالة الظاهرية $\vect g' = -g\vect e_z +
\Omega^2r\vect e_r$ (إذ يكون سطح مائع ساكن ناظميًّا على المجال،
\cref{ch:b1:fluid-statics}). وميله $\dd z/\dd r = \Omega^2r/g$، ومنه
$z = \Omega^2r^2/2g$: أي مجسم قطع مكافئ. وعند \qty{10}{rad/s} ترتفع حافة
دلو نصف قطره \qty{10}{cm} بمقدار \qty{5}{cm} فوق المركز --- وهو مبدأ
مقراب المرآة السائلة، الذي يأخذ زئبقه الدوّار تمامًا
الشكل المكافئ الذي تحتاجه المرآة.
\end{example}

\begin{method}[العمل في مرجع غير عطالي]\label{met:b1:non-inertial-frames:method}
اختر المرجع الذي يكون فيه السؤال أبسط (الحافلة أو
القرص الدوّار أو الأرض)؛ وعدّد القوى الحقيقية؛ وأضف $-m\vect a_e$، وإذا
تحرك الجسم في المرجع، أضف $-2m\vect\Omega\wedge\vect v'$؛ ثم تابع كما في
\cref{met:b1:newton-dynamics:method}. وتحقق: لا تظهر القوى العطالية أبدًا
في موازنة الطاقة على أنها ``شغل يبذله أحد'' --- \omterm{thm:b1:non-inertial-frames:dynamics}{فقوة
كوريوليس}، وهي عمودية على $\vect v'$، لا تبذل شغلًا؛ \omterm{thm:b1:non-inertial-frames:dynamics}{والقوة الطاردة المركزية}
تشتقّ من \omterm{def:b1:work-and-energy:conservative}{الطاقة الكامنة} $-\tfrac12 m\Omega^2r^2$ في مرجع يدور
دورانًا منتظمًا.
\end{method}

\section{المرجع الأرضي}

\begin{definition}[المرجعان المركزي الأرضي والأرضي]\label{def:b1:non-inertial-frames:earth}
\index{مرجع مركزي أرضي}\index{مرجع أرضي}
\emph{المرجع المركزي الأرضي} مبدؤه مركز الأرض
ومحاوره موجَّهة نحو نجوم ثابتة؛ وهو عطالي بتقريب ممتاز
من أجل حركات تدوم يومًا أو أقل (إذ إن تسارعه حول
الشمس لا يتعدى \qty{6e-3}{m/s^2}، وهو منتظم تقريبًا على امتداد الأرض).
أما \emph{المرجع الأرضي}، المرتبط بالأرض، فيدور بالنسبة
إليه حول المحور القطبي بالسرعة
\[
\Omega = \frac{2\pi}{\qty{86164}{s}} = \qty{7.29e-5}{rad/s} .
\]
\end{definition}

\begin{proposition}[الوزن، وتغيّر $g$]\label{prop:b1:non-inertial-frames:weight}
\index{وزن!ودوران الأرض}
في \omterm{def:b1:non-inertial-frames:earth}{المرجع الأرضي} يشعر جسم ساكن بالجذب
الثقالي $m\vect{\mathcal G}$ وبالقوة الطاردة المركزية $m\Omega^2\vect{HM}$؛ ومجموعهما
هو ما يقيسه ميزان وما يبيّنه خيط الشاقول: أي \emph{الوزن}
$m\vect g = m\vect{\mathcal G} + m\Omega^2\vect{HM}$. والحدّ الطارد المركزي أكبر ما يكون
عند خط الاستواء، $\Omega^2R = \qty{0.034}{m/s^2}$، أي $0.35\%$ من $g$؛ وهو
يخفّض $g$ من القطبين إلى الاستواء ويميل الشاقول بعيدًا
عن مركز الأرض بما يبلغ $0.1^\circ$ عند خطوط العرض الوسطى.
\end{proposition}

\begin{proof}
طبّق \cref{thm:b1:non-inertial-frames:dynamics} مع $\vect v' = \vect 0$
(فلا \omterm{thm:b1:non-inertial-frames:dynamics}{قوة كوريوليس})؛ و $\Omega^2R = (7.29\times10^{-5})^2 \times 6.37\times10^6$.
وتنتج زاوية الانحراف من مثلث $m\vect{\mathcal G}$ \omterm{thm:b1:non-inertial-frames:dynamics}{والقوة
الطاردة المركزية} (\cref{exo:b1:non-inertial-frames:6}).
\end{proof}

\begin{proposition}[قوة كوريوليس على الأرض]\label{prop:b1:non-inertial-frames:coriolis}
عند خط العرض $\lambda$، يشعر جسم يتحرك أفقيًّا بالسرعة $v'$
\omterm{thm:b1:non-inertial-frames:dynamics}{بقوة كوريوليس} أفقية مقدارها
\[
F_C = 2m\Omega v'\sin\lambda ,
\]
موجَّهة إلى \emph{يمين} الحركة في نصف الكرة الشمالي و
إلى يسارها في الجنوبي، أيًّا كانت جهة السير؛ أما
الجسم الساقط شاقوليًّا فينحرف \emph{شرقًا} بالمقدار $2m\Omega v'\cos\lambda$.
\end{proposition}

\begin{proof}
حلّل $\vect\Omega = \Omega(\cos\lambda\,\vect e_N + \sin\lambda\,\vect e_z)$
(على الشمال والشاقول المحلي). ومن أجل $\vect v'$ أفقية، يكون للمقدار $-2m\vect\Omega\wedge\vect v'$
جزء أفقي $-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$ --- مُدار
$90^\circ$ في اتجاه عقارب الساعة عن $\vect v'$ منظورًا من فوق، من أجل $\lambda > 0$ ---
وجزء شاقولي (مهمل أمام الثقالة). ومن أجل الحالة الشاقولية $\vect v' =
-v'\vect e_z$: $-2m\Omega\cos\lambda\,\vect e_N\wedge(-v'\vect e_z) = 2m\Omega v'\cos\lambda\,
\vect e_E$، أي شرقًا.
\end{proof}

\begin{example}[قوى صغيرة ونتائج كبيرة]\label{ex:b1:non-inertial-frames:coriolisnum}
قطار عند \qty{100}{km/h} على خط العرض $45^\circ$: $a_C = 2 \times 7.29\times
10^{-5} \times 27.8 \times 0.71 = \qty{2.9e-3}{m/s^2}$، أي ثلاثة أجزاء من ألف من
$g$ --- فتتآكل السكة اليمنى في خط أحادي الاتجاه أسرع قليلًا. والهواء
المتدفق بسرعة \qty{20}{m/s} نحو مركز منخفض الضغط يُدار يمينًا طوال
طريقه إلى الداخل فينتهي دائرًا حوله عكس عقارب الساعة: فكل إعصار في
نصف الكرة الشمالي يدور هكذا، وكل إعصار جنوبي بالعكس. وحجر
يُسقط في بئر عمقها \qty{100}{m} عند $45^\circ$ يحطّ على بعد \qty{1.6}{cm} شرق
خيط الشاقول (\cref{exo:b1:non-inertial-frames:7}).
\end{example}

\begin{proposition}[بندول فوكو]\label{prop:b1:non-inertial-frames:foucault}
\index{بندول فوكو}
بندول يتأرجح بسعة صغيرة عند خط العرض $\lambda$ يحفظ
مستوي اهتزازه ثابتًا بالنسبة إلى دوران الشاقول المحلي في \omterm{def:b1:non-inertial-frames:earth}{المرجع
المركزي الأرضي} وحده: فيبدو في \omterm{def:b1:non-inertial-frames:earth}{المرجع الأرضي} أن المستوي يدور
\emph{في اتجاه عقارب الساعة} (منظورًا من فوق، في نصف الكرة الشمالي) بالمعدّل الزاوي
$\Omega\sin\lambda$، أي دورة كاملة في
\[
T_F = \frac{\qty{23.93}{h}}{\sin\lambda} .
\]
\end{proposition}

\begin{proof}[برهان جزئي]
عند القطب ($\lambda = 90^\circ$) يكون البرهان هندسيًّا: فلا تؤثر أيّ قوة
أفقية في الثقّالة إلا الشدّ، ومركّبته الأفقية
على امتداد التأرجح، ومن ثمّ يكون مستوي التأرجح ثابتًا في \omterm{def:b1:non-inertial-frames:earth}{المرجع المركزي الأرضي}
بينما تدور الأرض تحته بالسرعة $\Omega$؛ فيبدو من الأرض دائرًا
بالسرعة $-\Omega$. وعند خط العرض $\lambda$ تكون \omterm{thm:b1:non-inertial-frames:dynamics}{قوة كوريوليس} الأفقية
$-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$: وهي بالضبط ما ينتجه مرجع يدور بالسرعة
$\Omega\sin\lambda$ حول الشاقول المحلي. وفي مرجع يدور
مع المستوي بالسرعة $-\Omega\sin\lambda$ تختفي تلك القوة من الرتبة الأولى
ويصير البندول هزّازًا مستويًا عاديًّا؛ ومن هنا يبادر المستوي
بالسرعة $\Omega\sin\lambda$ في \omterm{def:b1:non-inertial-frames:earth}{المرجع الأرضي}. أما الحساب
الكامل، مع الحدود المهملة من الرتبة الثانية، فهو مسألة نهاية
الأسبوع.
\end{proof}

\begin{remark}[المدّ والجزر في سطرين]\label{rem:b1:non-inertial-frames:tides}
\index{المدّ والجزر}
في مرجع مركز الأرض، وهو يسقط سقوطًا حرًّا نحو القمر،
لا يكون جذب القمر منتظمًا: فهو أقوى قليلًا على الجانب القريب،
وأضعف على البعيد. والفرق، وهو من رتبة $2GM_{\text{القمر}}R/d^3
\approx \qty{1e-6}{m/s^2}$، يمطّ المحيطات إلى انتفاخين --- أي مدّين
في اليوم --- وإسهام الشمس، وهو $46\%$ من إسهام القمر، يصنع
المدّ الربيعي والمدّ الأدنى بحسب اصطفاف الاثنين أو تقاطعهما
(\cref{exo:b1:non-inertial-frames:12}).
\end{remark}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:1}
يتدلّى شاقول في سيارة تتسارع بالمقدار \qty{2.5}{m/s^2}: احسب الزاوية مع
الشاقول، والشدّ بوحدات $mg$. وماذا لو كبحت السيارة نفسها
بالمعدّل نفسه؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:2}
يجلس طفل على بعد \qty{3.0}{m} من محور مرجوحة دوّارة تدور دورة
كل \qty{4.0}{s}. احسب التسارع الطارد المركزي في المرجع الدوّار، كنسبة
من $g$؛ وأيّ قوة حقيقية توفّره؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:3}
احسب \omterm{cor:b1:point-kinematics:circular}{السرعة الزاوية} للأرض $\Omega$، والتسارع الطارد المركزي
عند خط الاستواء، ونسبته من $g$. وعند أيّ دور دوران تطفو
الأجسام عند خط الاستواء مبتعدةً؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:4}
يسير قطار كتلته \qty{400}{t} بسرعة \qty{100}{km/h} على خط العرض $45^\circ$.
احسب \omterm{thm:b1:non-inertial-frames:composition}{تسارع كوريوليس} والقوة الجانبية على السكتين؛ وأيّ سكة
تُدفع عند التوجه شمالًا؟ وعند التوجه شرقًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:5}
يدور دلو نصف قطره \qty{10}{cm} بالسرعة \qty{2.0}{rad/s}، ثم
\qty{10}{rad/s}. احسب ارتفاع الماء عند الحافة بالنسبة إلى المركز.
ولماذا يكون حوض زئبق دوّار مرآة مقراب جيدة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:6}
بيّن أن خيط الشاقول عند خط العرض $\lambda$ ينحرف عن مركز
الأرض بنحو $\Omega^2R\sin\lambda\cos\lambda/g$ راديان، واحسبه
عند $45^\circ$ بدقائق القوس. وبكم يكون $g$ أصغر عند خط الاستواء
منه عند القطب بفعل الدوران وحده؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:7}
يسقط حجر من السكون في بئر عمقها \qty{100}{m} عند خط العرض $45^\circ$.
عامِل \omterm{thm:b1:non-inertial-frames:composition}{تسارع كوريوليس} $2\Omega v\cos\lambda$ (نحو الشرق) على أنه اضطراب
صغير مع $v = gt$؛ وكامِل مرتين لإيجاد الانحراف الشرقي
عند القاع. ولماذا نحو الشرق؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:8}
احسب دور مبادرة \omterm{prop:b1:non-inertial-frames:foucault}{بندول فوكو} في باريس ($\lambda = 48.9^\circ$)،
وعند القطب، وعند خط الاستواء. وبأيّ زاوية يدور المستوي
في ساعة واحدة في باريس؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:9}
يتدفق الهواء بسرعة \qty{20}{m/s} نحو مركز منخفض الضغط عند $45^\circ$
شمالًا. احسب \omterm{thm:b1:non-inertial-frames:composition}{تسارع كوريوليس}؛ وبأيّ جهة ينتهي الهواء
دائرًا حول المركز؟ ولماذا لا تخضع الدوّامات الصغيرة (كمصرف أو إعصار قمعي)
لهذه القاعدة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:10}
يطفو بالون هليوم مربوط بخيط داخل سيارة. فحين
تتسارع السيارة إلى الأمام، بأيّ جهة يميل البالون؟ فسّر ذلك بالثقالة
الظاهرية وبالدافعة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:11}
محطة فضائية على شكل حلقة نصف قطرها \qty{100}{m} تُدار لتعطي $1g$ من
الثقالة الاصطناعية عند الحافة. فما دور الدوران؟ ويمشي رائد فضاء على
الحافة بسرعة \qty{1.5}{m/s}: احسب \omterm{thm:b1:non-inertial-frames:composition}{تسارع كوريوليس}، وأثره في
وزنه الظاهري حين يمشي مع الدوران وضده. ويُسقط
كرة من \qty{2.0}{m}: فعلى أيّ بعد من قدميه تحطّ تقريبًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:12}
بيّن أن الفرق بين تسارع \omterm{def:b1:central-forces:gravitation}{جاذبية} القمر
عند مركز الأرض وعند نقطة السطح الأقرب إلى القمر هو
نحو $2GM_Mr/d^3$ (حيث $r$ نصف قطر الأرض و $d$ بعد الأرض عن القمر)؛
واحسبه، ثم احسب نظيره من أجل الشمس؛ واستنتج نسبة المدّ الشمسي إلى
المدّ القمري.
\end{exercise}

\section{مسألة: بندول فوكو}

\begin{problem}[{مسألة نهاية الأسبوع --- ثقّالة كتلتها ثمانية وعشرون كيلوغرامًا على سلك
طوله سبعة وستون مترًا تحت قبة: لماذا يدور مستوي تأرجحها، وبأيّ سرعة، وما الذي
تثبته، وكم صغيرة هي القوة التي تفعل ذلك}]
\label{pb:b1:non-inertial-frames:1}
سنة 1851 تأرجح بندول طوله $\ell = \qty{67}{m}$ وكتلته $m = \qty{28}{kg}$
تحت قبة البانثيون في باريس، عند خط العرض $\lambda =
48.85^\circ$، بسعة نحو $x_0 = \qty{3.0}{m}$. ولدينا $\Omega =
\qty{7.29e-5}{rad/s}$ و $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{الجزء I --- البندول في ذاته.}
\begin{enumerate}
  \item احسب دور \omterm{prop:b1:work-and-energy:stability}{الاهتزازات الصغيرة}؛ وتحقق من أن السعة صغيرة
        (بحساب $x_0/\ell$).
  \item احسب السرعة العظمى للثقّالة، وارتفاعها الأعظمي فوق
        أخفض نقطة.
  \item احسب الشدّ عند أخفض نقطة، بوحدات $mg$.
  \item قدّر سحب الهواء على الثقّالة عند السرعة العظمى (بالقانون
        التربيعي، مع $C_xS \approx \qty{0.05}{m^2}$ و $\rho = \qty{1.2}{kg/m^3}$) ثم
        قارنه بالقوة المعيدة العظمى $m\omega_0^2x_0$.
  \item لماذا سلك طويل وثقّالة ثقيلة؟ (سببان، أحدهما يخصّ
        الدور والآخر التخميد.)
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- عند القطب.}
تخيّل البندول نفسه عند القطب الشمالي.
\begin{enumerate}[resume]
  \item في \omterm{def:b1:non-inertial-frames:earth}{المرجع المركزي الأرضي}، عدّد القوى المؤثرة في الثقّالة وبيّن
        أنه لا شيء يستطيع إدارة مستوي تأرجحها.
  \item استنتج ما يراه مراقب واقف على الجليد أن المستوي يفعله،
        وفي كم من الزمن يعود إلى اتجاهه الابتدائي.
  \item صف مسار الثقّالة على الأرض (وهو وردة): فكم بتلة
        في اليوم الواحد؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- عند خط العرض $\lambda$: حدّ كوريوليس.}
استعمل محاور محلية: $x$ نحو الشرق و $y$ نحو الشمال و $z$ نحو الأعلى؛ \omterm{prop:b1:work-and-energy:stability}{والاهتزازات صغيرة}، فتكون
الحركة أفقية تقريبًا، $\vect v' \approx (\dot x, \dot y, 0)$؛ والقوة
المعيدة هي $-m\omega_0^2(x, y, 0)$ مع $\omega_0 = \sqrt{g/\ell}$.
\begin{enumerate}[resume]
  \item اكتب $\vect\Omega$ في هذه المحاور.
  \item احسب $-2m\vect\Omega\wedge\vect v'$ واحتفظ بمركّبتيه
        الأفقيتين؛ وبيّن أنهما لا تتضمنان إلا $\Omega\sin\lambda$.
  \item اكتب معادلتَي الحركة من أجل $x$ و $y$؛ وضع
        $\omega_F = \Omega\sin\lambda$ وقارن $\omega_F$ بالمقدار $\omega_0$.
  \item أدخِل $u = x + jy$ وبيّن أن $\ddot u + 2j\omega_F\dot u +
        \omega_0^2u = 0$.
  \item جرّب $u = \eu^{j\alpha t}$: أوجد قيمتَي $\alpha$، و
        بيّن أنه من أجل $\omega_F \ll \omega_0$ تكون $\alpha \approx \pm\omega_0
        - \omega_F$.
  \item استنتج أن $u \approx \eu^{-j\omega_Ft}(A\eu^{j\omega_0t} + B\eu^{-j\omega_0t})$:
        وفسّر العامل $\eu^{-j\omega_Ft}$ --- فبأيّ جهة، وبأيّ
        معدّل، يدور مستوي التأرجح؟
  \item احسب دور المبادرة في باريس والزاوية المقطوعة في
        ساعة واحدة.
  \item ولماذا لا يدور المستوي عند خط الاستواء؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- كم صغيرة هي القوة.}
\begin{enumerate}[resume]
  \item احسب \omterm{thm:b1:non-inertial-frames:dynamics}{قوة كوريوليس} العظمى على الثقّالة وقارنها
        بالوزن وبالقوة المعيدة العظمى.
  \item بكم تنزاح نقطة نهاية التأرجح جانبيًّا خلال
        نصف دور؟ (فالمستوي يدور بالمقدار $\omega_FT/2$.)
  \item سنة 1851 حملت الثقّالة مِسنًّا يقلب أوتادًا صغيرة
        موضوعة في حلقة عند السعة: فبعد كم من الزمن تقريبًا يسقط
        الوتد التالي؟
  \item أُطلق البندول بحرق خيط كان يمسك الثقّالة
        جانبًا. فلماذا لا يُدفع باليد؟
  \item يوقف سحب الهواء بندولًا بعد بضع ساعات. وتستعمل بنادول المتاحف
        الحديثة مغناطيسًا كهربائيًّا صغيرًا في الأسفل لإعادة
        الطاقة دون مسّ مستوي التأرجح: فلماذا يجب أن تكون تلك الركلة
        قطرية تمامًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء V --- ما الذي يثبته، وما الذي يفعله أيضًا.}
\begin{enumerate}[resume]
  \item أقنع عرض فوكو الجمهور بدوران
        الأرض. فما الذي يقيسه معدّل المبادرة بالضبط، وهل
        تستطيع التجربة تحديد خط العرض؟
  \item عند أيّ خط عرض يدور المستوي دورة في \qty{48}{h} بالضبط؟
  \item الجيروسكوب (وهو دولاب سريع الدوران في محاور معلّقة) يحفظ محوره
        ثابتًا في \omterm{def:b1:non-inertial-frames:earth}{المرجع المركزي الأرضي}. فسّر في جملة واحدة لماذا يكشف هو أيضًا
        الدوران، وسمِّ تطبيقًا له.
  \item لخّص: \omterm{thm:b1:non-inertial-frames:dynamics}{القوة العطالية} المسؤولة الوحيدة، وتوقفها على
        خط العرض، والدور الذي تعطيه في باريس.
\end{enumerate}
\end{problem}
