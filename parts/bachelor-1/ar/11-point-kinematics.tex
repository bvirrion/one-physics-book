\chapter{تحريك النقطة المادية}\label{ch:b1:point-kinematics}

عربة الأفعوانية تدخل الحلقة عند \qty{90}{km/h}، فتبطئ
وهي تصعد، وتتعلّق لحظةً في القمة، ثم تندفع خارجةً من جديد؛
ويشعر راكبوها بأنهم مسحوقون في مقاعدهم عند الأسفل وبأنهم بلا وزن
تقريبًا عند القمة. ووصف تلك الرحلة --- أين العربة، وكم
تسرع، وكيف تدور سرعتها وتتغير --- هو التحريك،
وهو يحتاج إلى أكثر من $x$ و $y$ و $z$ لطريق مستقيم: أي إلى \omterm{def:b1:point-kinematics:cylindrical}{الإحداثيات
القطبية} لكل ما يدور، وإلى \omterm{def:b1:point-kinematics:frame}{مرجع} يسافر مع العربة
نفسها. ويقيم هذا الفصل تلك الأدوات، ويبرهن على صيغ
\omterm{def:b1:point-kinematics:velocity}{السرعة} \omterm{def:b1:point-kinematics:velocity}{والتسارع} في كل منها، ويطبّقها على
الحركات التي سيستدعيها كل فصل تال: الدائرية والحلزونية
والتوافقية والحركة على أيّ منحنٍ.

\section{الموضع والسرعة والتسارع}

\begin{definition}[المرجع والموضع والمسار]\label{def:b1:point-kinematics:frame}
\index{مرجع}\index{متجهة الموضع}\index{مسار}\index{نقطة مادية}
\emph{المرجع} جسم صلب يُؤخذ ثابتًا (كالمختبر
أو الأرض أو قطار)، مع ساعة. و\emph{النقطة
المادية} $M$ --- وهي جسم لا يهمّ حجمه في الحركة المدروسة
--- تُحدَّد في كل لحظة بواسطة \emph{متجهة موضعها}
$\vect{OM}(t)$ من مبدأ $O$ مرتبط بالمرجع؛ والمنحني
الذي ترسمه $M$ هو \emph{مسارها}. والحركة دائمًا حركة
\emph{بالنسبة إلى مرجع}: فالراكب نفسه ساكن في القطار
ومتحرك في المحطة.
\end{definition}

\begin{definition}[السرعة والتسارع]\label{def:b1:point-kinematics:velocity}
\index{سرعة}\index{تسارع}
\emph{سرعة} النقطة $M$ و\emph{تسارعها} في \omterm{def:b1:point-kinematics:frame}{المرجع} هما
\[
\vect v = \frac{\dd\vect{OM}}{\dd t}, \qquad
\vect a = \frac{\dd\vect v}{\dd t} = \frac{\dd^2\vect{OM}}{\dd t^2},
\]
وهما مشتقّا دالتين متجهيتين: ويُحصَّل مشتقّ متجهة
باشتقاق مركّباتها في أساس \emph{ثابت في
\omterm{def:b1:point-kinematics:frame}{المرجع}}. والسرعة مماسّة \omterm{def:b1:point-kinematics:frame}{للمسار}
وتشير في اتجاه الحركة؛ ومعيارها $v$ هو \emph{قيمة السرعة}. والوحدات:
\unit{m/s} و \unit{m/s^2}.
\end{definition}

\begin{proposition}[الإحداثيات الديكارتية]\label{prop:b1:point-kinematics:cartesian}
\index{إحداثيات ديكارتية}
مع أساس متعامد ممنظم ثابت $(\vect e_x, \vect e_y, \vect e_z)$ و
$\vect{OM} = x\,\vect e_x + y\,\vect e_y + z\,\vect e_z$، يكون
\[
\vect v = \dot x\,\vect e_x + \dot y\,\vect e_y + \dot z\,\vect e_z, \qquad
\vect a = \ddot x\,\vect e_x + \ddot y\,\vect e_y + \ddot z\,\vect e_z,
\]
حيث ترمز النقطة إلى $\dd/\dd t$.
\end{proposition}

\begin{proof}
متجهات الأساس ثابتة؛ ولا تتغير إلا المركّبات.
\end{proof}

\begin{example}[قذيفة]\label{ex:b1:point-kinematics:projectile}
كرة تُقذف من $O$ بقيمة \omterm{def:b1:point-kinematics:velocity}{سرعة} $v_0$ وزاوية $\alpha$ فوق
الأفق، وإحداثياتها (من الديناميك في الفصل التالي) هي
$x = v_0\cos\alpha\,t$ و $z = v_0\sin\alpha\,t - \tfrac12 gt^2$. ومنه
$\vect v = v_0\cos\alpha\,\vect e_x + (v_0\sin\alpha - gt)\vect e_z$ و
$\vect a = -g\vect e_z$: \omterm{def:b1:point-kinematics:velocity}{فالتسارع} ثابت بينما تدور
\omterm{def:b1:point-kinematics:velocity}{السرعة}؛ وعند القمة ($\dot z = 0$ و $t = v_0\sin\alpha/g$) تكون
\omterm{def:b1:point-kinematics:velocity}{السرعة} أفقية \omterm{def:b1:point-kinematics:velocity}{والتسارع} عموديًّا عليها.
وبحذف $t$: $z = x\tan\alpha - gx^2/(2v_0^2\cos^2\alpha)$، أي
قطع مكافئ.
\end{example}

\section{الإحداثيات الأسطوانية}

\begin{definition}[الإحداثيات الأسطوانية وأساسها]\label{def:b1:point-kinematics:cylindrical}
\index{إحداثيات أسطوانية}\index{إحداثيات قطبية}\index{أساس محلي}
تُحدَّد النقطة $M$ بالمقدار $r \geq 0$ (وهو بعدها عن المحور $z$)، وبالزاوية
$\theta$ التي يصنعها مسقطها على المستوي $xy$ مع $\vect e_x$،
وبالإحداثي $z$: أي $\vect{OM} = r\,\vect e_r + z\,\vect e_z$، حيث
\[
\vect e_r = \cos\theta\,\vect e_x + \sin\theta\,\vect e_y, \qquad
\vect e_\theta = -\sin\theta\,\vect e_x + \cos\theta\,\vect e_y,
\]
وتشكّل هذه مع $\vect e_z$ \emph{الأساس المحلي} $(\vect e_r, \vect e_\theta,
\vect e_z)$، وهو متعامد ممنظم ومباشر، و\emph{يتحرك مع $M$}. وفي
المستوي ($z$ ثابت) تكون هذه هي \emph{الإحداثيات القطبية} $(r, \theta)$.
\end{definition}

\begin{lemma}[مشتقّات الأساس المحلي]\label{lem:b1:point-kinematics:basisderiv}
\[
\frac{\dd\vect e_r}{\dd t} = \dot\theta\,\vect e_\theta, \qquad
\frac{\dd\vect e_\theta}{\dd t} = -\dot\theta\,\vect e_r, \qquad
\frac{\dd\vect e_z}{\dd t} = \vect 0 .
\]
\end{lemma}

\begin{proof}
باشتقاق المركّبات: $\dd\vect e_r/\dd t = \dot\theta(-\sin\theta\,
\vect e_x + \cos\theta\,\vect e_y) = \dot\theta\vect e_\theta$، و
كذلك من أجل $\vect e_\theta$. فمشتقّ متجهة واحدة دائرة
عمودي عليها ومعياره هو المعدل الزاوي.
\end{proof}

\begin{theorem}[السرعة والتسارع بالإحداثيات الأسطوانية]\label{thm:b1:point-kinematics:cylindrical}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + \dot z\,\vect e_z,
\qquad
\vect a = \big(\ddot r - r\dot\theta^2\big)\vect e_r
+ \big(r\ddot\theta + 2\dot r\dot\theta\big)\vect e_\theta + \ddot z\,\vect e_z .
\]
\end{theorem}

\begin{proof}
نشتقّ $\vect{OM} = r\vect e_r + z\vect e_z$ بمساعدة المبرهنة المساعدة:
$\vect v = \dot r\vect e_r + r\dot\theta\vect e_\theta + \dot z\vect e_z$.
ثم نشتقّ من جديد: $\ddot r\vect e_r + \dot r\dot\theta\vect e_\theta +
\dot r\dot\theta\vect e_\theta + r\ddot\theta\vect e_\theta - r\dot\theta^2
\vect e_r + \ddot z\vect e_z$؛ ثم نجمع الأمثال.
\end{proof}

\begin{corollary}[الحركة الدائرية]\label{cor:b1:point-kinematics:circular}
\index{حركة دائرية}\index{تسارع مركزي}\index{سرعة زاوية}
على دائرة نصف قطرها $R$ (حيث $r = R$ و $z = 0$)، ومع \emph{\omterm{cor:b1:point-kinematics:circular}{السرعة
الزاوية}} $\omega = \dot\theta$:
\[
\vect v = R\omega\,\vect e_\theta, \qquad
\vect a = -R\omega^2\,\vect e_r + R\dot\omega\,\vect e_\theta
= -\frac{v^2}{R}\,\vect e_r + \frac{\dd v}{\dd t}\,\vect e_\theta .
\]
ومن أجل \omterm{cor:b1:point-kinematics:circular}{الحركة الدائرية} \emph{المنتظمة} ($\omega$ ثابت) يكون \omterm{def:b1:point-kinematics:velocity}{التسارع}
\emph{مركزيًّا} خالصًا، معياره $v^2/R = R\omega^2$، مع أن
قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} ثابتة.
\end{corollary}

\begin{proof}
نضع $\dot r = \ddot r = 0$ في المبرهنة؛ ويكون $v = R\omega$.
\end{proof}

\begin{example}[دائرتان]\label{ex:b1:point-kinematics:circles}
سيارة تدور في مستديرة نصف قطرها \qty{20}{m} \omterm{def:b1:point-kinematics:velocity}{بسرعة} \qty{36}{km/h}
(أي \qty{10}{m/s}): فيكون $a = v^2/R = \qty{5.0}{m/s^2}$، أي نصف $g$، موجَّهًا
نحو المركز --- وهو ما يجب أن توفّره الإطارات. والقمر
($R = \qty{3.84e8}{m}$ و $T = \qty{27.3}{d}$): $\omega = 2\pi/T =
\qty{2.66e-6}{rad/s}$ و $a = R\omega^2 = \qty{2.7e-3}{m/s^2}$، أي جزء من
$3600$ من $g$ عند سطح الأرض، على بعد ستين نصف قطر أرضي --- وهي
المقارنة التي أجراها نيوتن (\cref{ch:b1:central-forces}).
\end{example}

\section{الإحداثيات الكروية}

\begin{definition}[الإحداثيات الكروية]\label{def:b1:point-kinematics:spherical}
\index{إحداثيات كروية}
تُحدَّد النقطة $M$ ببعدها $r = OM$، وبالزاوية \emph{القطبية}
$\theta \in [0, \pi]$ بين $\vect{OM}$ و $\vect e_z$، وبزاوية \emph{السمت}
$\varphi$ لمسقطها على المستوي $xy$:
أي $\vect{OM} = r\,\vect e_r$ مع
\[
\vect e_r = \sin\theta\cos\varphi\,\vect e_x + \sin\theta\sin\varphi\,\vect e_y + \cos\theta\,\vect e_z ;
\]
وتُتمّ $\vect e_\theta$ (المماسّة لخط الزوال، نحو $\theta$ المتزايدة)
و $\vect e_\varphi$ (المماسّة لدائرة العرض) الأساسَ المحلي
المتعامد الممنظم. وعلى الأرض تكون $\theta$ هي $90^\circ$ ناقص
خط العرض، و $\varphi$ هي خط الطول.
\end{definition}

\begin{proposition}[السرعة بالإحداثيات الكروية]\label{prop:b1:point-kinematics:sphericalv}
\[
\vect v = \dot r\,\vect e_r + r\dot\theta\,\vect e_\theta + r\sin\theta\,\dot\varphi\,\vect e_\varphi .
\]
\end{proposition}

\begin{proof}
$\dd\vect e_r/\dd t = \dot\theta\,\vect e_\theta + \sin\theta\,\dot\varphi\,
\vect e_\varphi$: فالحدّ الأول هو الدوران في مستوي خط الزوال
(كما في الحالة القطبية)، والثاني هو الدوران حول $\vect e_z$ بمعدل
$\dot\varphi$ لمتجهة بعدها عن المحور
$\sin\theta$. ثم $\vect v = \dd(r\vect e_r)/\dd t$. أما \omterm{def:b1:point-kinematics:velocity}{التسارع} فلا
حاجة إليه هذه السنة.
\end{proof}

\section{أساس فريني}

\begin{definition}[الفاصلة المنحنية وأساس فريني]\label{def:b1:point-kinematics:frenet}
\index{فاصلة منحنية}\index{أساس فريني}\index{نصف قطر التقوّس}
على امتداد \omterm{def:b1:point-kinematics:frame}{مسار}، تكون \emph{الفاصلة المنحنية} $s(t)$ هي طول
القوس من نقطة مرجعية، محسوبًا جبريًّا؛ و $v = \dot s$. وعند
$M$ يشير المماسّ الواحدي $\vect T = \dd\vect{OM}/\dd s$ نحو
$s$ المتزايدة؛ ويُعرَّف \emph{نصف قطر التقوّس} $\rho > 0$ والناظم
الواحدي $\vect N$، المشير نحو الجهة المقعرة (أي نحو مركز
التقوّس)، بالعلاقة
\[
\frac{\dd\vect T}{\dd s} = \frac{1}{\rho}\,\vect N .
\]
والزوج $(\vect T, \vect N)$ هو \emph{أساس فريني}؛ ويكون $\rho = \infty$ على
مستقيم، و $\rho = R$ على دائرة نصف قطرها $R$.
\end{definition}

\begin{theorem}[التسارع في أساس فريني]\label{thm:b1:point-kinematics:frenet}
\index{تسارع مماسي}\index{تسارع ناظمي}
\[
\vect v = v\,\vect T, \qquad
\vect a = \frac{\dd v}{\dd t}\,\vect T + \frac{v^2}{\rho}\,\vect N .
\]
المركّبة \emph{المماسية} $\dd v/\dd t$ تقيس \omterm{def:b1:point-kinematics:velocity}{سرعة} تغيّر
قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة}؛ والمركّبة \emph{الناظمية} $v^2/\rho$، وهي دائمًا نحو
مركز التقوّس، تقيس \omterm{def:b1:point-kinematics:velocity}{سرعة} دوران الاتجاه. وتكون الحركة
منتظمة إذا وفقط إذا كان $\vect a \perp \vect v$؛ وتكون مستقيمة إذا وفقط إذا كان
$\vect a \parallel \vect v$.
\end{theorem}

\begin{proof}
$\vect v = (\dd\vect{OM}/\dd s)(\dd s/\dd t) = v\vect T$. ومنه $\vect a =
\dot v\vect T + v\,\dd\vect T/\dd t = \dot v\vect T + v(\dd\vect T/\dd s)\dot s
= \dot v\vect T + (v^2/\rho)\vect N$. أما كون $\dd\vect T/\dd s \perp \vect T$
فيتبع من اشتقاق $\vect T\cdot\vect T = 1$.
\end{proof}

\begin{example}[الكبح في منعطف]\label{ex:b1:point-kinematics:bend}
سيارة عند \qty{90}{km/h} (أي \qty{25}{m/s}) في منعطف نصف قطره \qty{100}{m}
تكبح بمقدار \qty{2.0}{m/s^2}: فيكون $a_T = \qty{-2.0}{m/s^2}$ و $a_N = 625/100 =
\qty{6.25}{m/s^2}$ و $a = \sqrt{4 + 39} = \qty{6.6}{m/s^2}$. فالجزء الناظمي
هو المهيمن: فالمنعطف قبل كل شيء دوران، وتماسك الإطارات
(في الفصل التالي) يحدّ $\sqrt{a_T^2 + a_N^2}$، ولهذا يُكبح
\emph{قبل} المنعطف.
\end{example}

\section{الحركات المعيارية}

\begin{proposition}[الحركة المستقيمة المنتظمة التسارع]\label{prop:b1:point-kinematics:uarm}
\index{حركة منتظمة التسارع}
على محور، $\ddot x = a$ ثابت، انطلاقًا من $x_0$ و $v_0$:
$v = v_0 + at$ و $x = x_0 + v_0t + \tfrac12 at^2$ و
$v^2 - v_0^2 = 2a(x - x_0)$.
\end{proposition}

\begin{proof}
نكامل مرتين؛ ثم نحذف $t$ بين العلاقتين الأوليين.
\end{proof}

\begin{proposition}[الحركة الحلزونية]\label{prop:b1:point-kinematics:helix}
\index{حركة حلزونية}
$r = R$ و $\theta = \omega t$ و $z = v_zt$ (والثوابت $R$ و $\omega$ و $v_z$):
تكون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} $v = \sqrt{R^2\omega^2 + v_z^2}$ ثابتة، ويكون \omterm{def:b1:point-kinematics:velocity}{التسارع}
$\vect a = -R\omega^2\vect e_r$ مركزيًّا ثابت المعيار، ويكون
\omterm{def:b1:point-kinematics:frenet}{نصف قطر التقوّس} $\rho = v^2/a = R(1 + v_z^2/R^2\omega^2) > R$.
\end{proposition}

\begin{proof}
\cref{thm:b1:point-kinematics:cylindrical} مع $\dot r = 0$ و
$\ddot\theta = 0$ و $\ddot z = 0$؛ والحركة منتظمة، فيكون $\vect a$ ناظميًّا
خالصًا و $v^2/\rho = R\omega^2$.
\end{proof}

\begin{proposition}[الحركة التوافقية]\label{prop:b1:point-kinematics:harmonic}
\index{حركة توافقية}
$x = A\cos(\omega t + \varphi)$: ومنه $\dot x = -A\omega\sin(\omega t + \varphi)$ و
$\ddot x = -\omega^2 x$. فتتقدّم \omterm{def:b1:point-kinematics:velocity}{السرعة} على الموضع بربع
دور ويكون \omterm{def:b1:point-kinematics:velocity}{التسارع} معاكسًا للموضع؛ و $v_{\max} =
A\omega$ و $a_{\max} = A\omega^2$. وهذه هي مسقط \omterm{cor:b1:point-kinematics:circular}{حركة دائرية} منتظمة
نصف قطرها $A$ وسرعتها الزاوية $\omega$ على قطر.
\end{proposition}

\begin{proof}
نشتقّ؛ وللمسقط نأخذ $x = A\cos\theta$ مع $\theta =
\omega t + \varphi$.
\end{proof}

\begin{method}[اختيار الإحداثيات]\label{met:b1:point-kinematics:choose}
\begin{itemize}
  \item حركة مستقيمة أو مكافئية \omterm{def:b1:point-kinematics:velocity}{بتسارع} ثابت:
        الديكارتية.
  \item حركة حول محور أو نقطة (كالدوائر واللوالب والمدارات
        وأقراص الدوران): الأسطوانية أو القطبية؛ والأساس يتحرك، فتُستعمل
        \cref{thm:b1:point-kinematics:cylindrical}، لا الاشتقاق
        الساذج للمركّبات.
  \item منحنٍ معطى (كمضمار أو منعطف أو حلقة) حين يُعرف قانون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة}:
        \omterm{def:b1:point-kinematics:frenet}{أساس فريني}، وهو يفصل ``كم \omterm{def:b1:point-kinematics:velocity}{السرعة}'' عن ``أيّ اتجاه''.
\end{itemize}
ومهما يكن الاختيار، تبقى $\vect v$ و $\vect a$ متجهتين تتوقفان على
\omterm{def:b1:point-kinematics:frame}{المرجع}، ولا تكتمل النتيجة إلا بتسمية \omterm{def:b1:point-kinematics:frame}{المرجع}.
\end{method}

\begin{exercise}[$\star\star\star$]\label{exo:b1:point-kinematics:11}

\end{exercise}

\section{مسألة: حلقة الأفعوانية}

\begin{problem}[{مسألة نهاية الأسبوع --- عربة تدخل حلقة شاقولية عند
تسعين كيلومترًا في الساعة: أين تبطئ، وإلى أين يشير تسارعها،
وكم من $g$ يشعر الراكبون، ولماذا لا تكون الحلقات الحديثة
دوائر}]
\label{pb:b1:point-kinematics:1}
عربة، تُعامل نقطةً $M$، تسير على حلقة دائرية شاقولية
نصف قطرها $R = \qty{10}{m}$ ومركزها $C$. ويُعطى موضعها
بالزاوية $\theta$ بين $\vect{CM}$ والشاقول النازل (فتكون $\theta =
0$ عند الأسفل و $\pi$ عند القمة). ويُهمَل الاحتكاك، و
قانون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} (المشتقّ في \cref{ch:b1:work-and-energy}) هو
\[
v^2 = v_0^2 - 2gR(1 - \cos\theta),
\]
مع $v_0 = \qty{25}{m/s}$ قيمةَ \omterm{def:b1:point-kinematics:velocity}{السرعة} عند الأسفل؛ و $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{الجزء I --- التحريك على الدائرة.}
\begin{enumerate}
  \item \omterm{def:b1:point-kinematics:cylindrical}{بالإحداثيات القطبية} المتمركزة عند $C$ (مع $\vect e_r$ من $C$
        إلى $M$)، اكتب $\vect v$ و $\vect a$ للحركة على الدائرة.
  \item عيّن متجهتَي فريني $\vect T$ و $\vect N$ بدلالة
        $\vect e_r$ و $\vect e_\theta$، \omterm{def:b1:point-kinematics:frenet}{ونصف قطر التقوّس}.
  \item احسب قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} عند القمة، وعند النقطتين الجانبيتين
        $\theta = \pi/2$ و $3\pi/2$.
  \item اشتقّ قانون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} بالنسبة إلى الزمن لتبيّن أن
        \omterm{thm:b1:point-kinematics:frenet}{التسارع المماسي} هو $a_T = -g\sin\theta$. وفسّر
        إشارته في الصعود وفي النزول.
  \item عبّر عن \omterm{thm:b1:point-kinematics:frenet}{التسارع الناظمي} $a_N(\theta)$ واحسبه عند
        الأسفل والجانبين والقمة.
  \item أعطِ معيار $\vect a$ عند النقاط الأربع والزاوية التي
        يصنعها مع الشاقول عند القمة.
  \item عند أيّ نقاط تكون $\vect a$ عمودية على $\vect v$؟ وموازية لها؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- المعدلات الزاوية والتوقيت.}
\begin{enumerate}[resume]
  \item عبّر عن $\dot\theta$ بدلالة $\theta$ واحسبها
        عند الأسفل والقمة.
  \item بيّن أن $\ddot\theta = -(g/R)\sin\theta$ (وهي معادلة
        نواس) وعلّق على ذلك.
  \item للمقارنة، احسب دور التذبذبات الصغيرة لنواس
        بسيط طوله $R$ (وقد أعطت \cref{ex:b1:units-dimensions:pendulum}
        صيغته؛ مع $k = 2\pi$).
  \item قدّر زمن الدوران حول الحلقة، مستعملًا متوسط
        قيم \omterm{def:b1:point-kinematics:velocity}{السرعة} عند النقاط المرجعية الأربع.
  \item أصغر قيمة \omterm{def:b1:point-kinematics:velocity}{للسرعة} عند القمة تُبقي العربة على المضمار
        (في الفصل التالي) هي $\sqrt{gR}$. فهل تتحقق؟ وأيّ أصغر
        $v_0$ يلزم؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- ما يشعر به الراكبون.}
الوزن الظاهري لواحدة الكتلة الذي يشعر به راكب هو $\vect a - \vect g$
(وهو مسلَّم هنا، ومشتقّ في \cref{ch:b1:newton-dynamics})؛ ومعياره
بوحدات $g$ هو ``حمولة $g$''.
\begin{enumerate}[resume]
  \item احسب حمولة $g$ عند الأسفل.
  \item احسبها عند القمة، وقل إلى أيّ جهة يُضغط الراكب
        (المقعد أم المقيِّد).
  \item احسبها عند النقطتين الجانبيتين.
  \item يتحمّل الراكبون نحو $4g$ برهةً. فهل تنجح هذه الحلقة؟ وأين
        المشكلة؟
  \item بيّن أنه من أجل حلقة دائرية يُدخل إليها \omterm{def:b1:point-kinematics:velocity}{بسرعة} تكفي بالضبط
        لتجاوز القمة (بوزن ظاهري معدوم هناك)، تكون حمولة $g$ عند الأسفل
        $6g$ حتمًا، مهما تكن $R$.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الحلقة الكلوثويدية.}
الحلقات الحديثة على شكل ``دمعة'': \omterm{def:b1:point-kinematics:frenet}{فنصف قطر التقوّس} كبير عند
الأسفل وصغير عند القمة.
\begin{enumerate}[resume]
  \item باستعمال صورة فريني \omterm{def:b1:point-kinematics:velocity}{للتسارع} $\vect a$، فسّر لماذا يغيّر تغيير $\rho$
        على امتداد المضمار المقدارَ $a_N$ دون تغيير قانون قيمة \omterm{def:b1:point-kinematics:velocity}{السرعة}
        (الذي يتوقف على الارتفاع وحده).
  \item اختر $\rho_{\text{الأسفل}}$ بحيث تكون حمولة $g$ عند
        الأسفل $3g$ مع $v_0 = \qty{25}{m/s}$.
  \item عند القمة يبقى الارتفاع $2R = \qty{20}{m}$ فوق
        الأسفل. اختر $\rho_{\text{القمة}}$ بحيث تكون حمولة $g$ هناك
        $1.5g$.
  \item للكلوثويد تقوّس يتناسب مع طول القوس،
        $1/\rho = s/A^2$. فعلى امتداد مدخل كهذا، وقيمة \omterm{def:b1:point-kinematics:velocity}{السرعة} شبه
        ثابتة، كيف ينمو $a_N$ مع الزمن، ولماذا يكون ذلك أرفق
        بالعنق من دائرة (حيث يقفز $a_N$ عند المدخل)؟
  \item ارسم شكل الحلقة الناتج رسمًا نوعيًّا ولخّص في
        جملتين لماذا حلّ محلّ الدائرة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء V --- منظورًا من الأرض.}
آلة تصوير على مستوى الأرض، على بعد \qty{50}{m} من مركز الحلقة في
مستويها، تتعقّب العربة.
\begin{enumerate}[resume]
  \item حين تكون العربة عند القمة متحركةً أفقيًّا \omterm{def:b1:point-kinematics:velocity}{بسرعة}
        \qty{15}{m/s}، بأيّ معدل زاوي (بالراديان في الثانية) يجب أن تدور \omterm{def:b1:optical-instruments:camera}{آلة التصوير}؟
        (\omterm{def:b1:point-kinematics:cylindrical}{بالإحداثيات القطبية} من \omterm{def:b1:optical-instruments:camera}{آلة التصوير}.)
  \item لماذا تكون آلة تصوير تدور بمعدل ثابت متعقّبًا رديئًا
        \omterm{cor:b1:point-kinematics:circular}{لحركة دائرية} منتظمة؟
