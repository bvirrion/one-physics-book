\chapter{قوانين نيوتن في الديناميك}\label{ch:b1:newton-dynamics}

يقفز مظلّي من الطائرة، فيكفّ خلال بضع عشرة ثانية عن
التسارع: إذ يدفعه الهواء عند مئتَي كيلومتر في الساعة \omterm{def:b1:newton-dynamics:momentum}{بقوة} تساوي تمامًا
جذب الأرض له. وسيارة تدخل منعطفًا بسرعة مفرطة فتخون
الإطارات؛ وشاحنة على طريق جبلية تزحف في تُرس منخفض،
يمسكها الاحتكاك نفسه الذي يتيح لها التوقف على المستوي. وقد وصف التحريكُ
الحركاتِ؛ أما الديناميك فيفسّرها، انطلاقًا من قانون واحد يربط
تسارع جسم بالقوى المؤثرة فيه. ويذكر هذا الفصل
\omterm{thm:b1:newton-dynamics:laws}{قوانين نيوتن} الثلاثة، ويجرد القوى التي تُلاقى في
مستوى هذا المجلد --- الوزن والنوابض والخيوط والتماسّ \omterm{prop:b1:newton-dynamics:friction}{والاحتكاك
الجاف} والمائعي --- ويبيّن كيف يُنتقل من رسم للقوى
إلى معادلة تفاضلية وحلّها.

\section{الكتلة وكمية الحركة والقوانين الثلاثة}

\begin{definition}[الكتلة وكمية الحركة والقوة]\label{def:b1:newton-dynamics:momentum}
\index{كتلة}\index{كمية الحركة}\index{قوة}
\omterm{def:b1:point-kinematics:frame}{للنقطة المادية} \emph{كتلة عطالية} $m > 0$ (بالكيلوغرام)، وهي
مقياس لمقاومتها لتغيّرات السرعة، ولا تتوقف على
المرجع. و\emph{كمية حركتها} في مرجع هي
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
و\emph{القوة} $\vect F$ (بالنيوتن) هي تأثير جسم آخر في
النقطة؛ والقوى تُجمع جمع المتجهات، و\emph{المحصلة} هي مجموعها
$\sum\vect F$.
\end{definition}

\begin{theorem}[قوانين نيوتن]\label{thm:b1:newton-dynamics:laws}
\index{قوانين نيوتن}\index{مرجع عطالي}\index{مبدأ العطالة}\index{الفعل ورد الفعل}
\begin{enumerate}
  \item \emph{مبدأ العطالة.} توجد مراجع تُسمّى
        \emph{عطالية} (أو غاليلية) تتحرك فيها نقطة لا تخضع
        لأيّ \omterm{def:b1:newton-dynamics:momentum}{قوة}، أو تخضع لقوى محصلتها معدومة، على خطّ
        مستقيم بسرعة ثابتة. وكل مرجع في انسحاب مستقيم منتظم
        بالنسبة إلى \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي} هو \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}.
  \item \emph{قانون \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة}.} في \omterm{thm:b1:newton-dynamics:laws}{مرجع عطالي}،
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{أي}\qquad
        m\,\vect a = \sum\vect F \ \text{ من أجل كتلة ثابتة } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{الفعل ورد الفعل}.} إذا أثّر الجسم $A$ \omterm{def:b1:newton-dynamics:momentum}{بالقوة} $\vect F_{A\to B}$
        في الجسم $B$، أثّر $B$ \omterm{def:b1:newton-dynamics:momentum}{بالقوة} $\vect F_{B\to A} = -\vect F_{A\to B}$
        في $A$، على امتداد المستقيم الواصل بينهما.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[ماذا تقول القوانين]\label{rem:b1:newton-dynamics:meaning}
القانون الأول ليس حالة خاصة من الثاني: فهو \emph{يعرّف}
المراجع التي يصحّ فيها الثاني. وعمليًّا يكون مختبر على
الأرض عطاليًّا بدقة ممتازة في تجارب تدوم دقائق
(و\cref{ch:b1:non-inertial-frames} تكمّم الخطأ)؛ ومرجع
مرتبط بالشمس والنجوم البعيدة أفضل منه بعد. والقانون
الثاني معادلة متجهية: أي ثلاث معادلات سلّمية، واحدة لكل محور، و
اختيار المحاور (\cref{ch:b1:point-kinematics}) هو القرار الأول
في كل مسألة. والقانون الثالث يصحّ في كل لحظة من أجل
قوى التماسّ والجاذبية في هذا المجلد.
\end{remark}

\begin{definition}[الجملة المغلقة؛ انحفاظ كمية الحركة]\label{def:b1:newton-dynamics:conservation}
\index{جملة مغلقة}\index{انحفاظ كمية الحركة}
\emph{الجملة المغلقة} جملة لا تتبادل مادة مع الخارج. ومن أجل
جملة من النقاط المادية تتلاشى القوى الداخلية مثنى مثنى (بالقانون الثالث)،
وتحقق \omterm{def:b1:newton-dynamics:momentum}{كمية الحركة} الكلية $\vect P = \sum m_i\vect v_i$ العلاقة
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$: فتنحفظ حين
تنعدم القوى الخارجية (و\cref{ch:b1:systems-of-points} تطوّر
هذا).
\end{definition}

\section{القوى المعتادة}

\begin{proposition}[الوزن والجاذبية]\label{prop:b1:newton-dynamics:weight}
\index{وزن}\index{مجال الجاذبية}
قرب سطح الأرض يُجذب جسم كتلته $m$ \omterm{def:b1:newton-dynamics:momentum}{بقوة} \emph{وزنه}
$\vect P = m\vect g$، حيث $\vect g$ هو \omterm{prop:b1:newton-dynamics:weight}{مجال الجاذبية}
المحلي، $g \approx \qty{9.81}{m/s^2}$، موجَّهًا إلى أسفل (نحو
المركز تقريبًا). وأعمّ من ذلك: \omterm{def:b1:newton-dynamics:momentum}{كتلة} $M$ على بعد $r$ تجذب
$m$ \omterm{def:b1:newton-dynamics:momentum}{بالقوة} $\vect F = -GMm/r^2\,\vect e_r$ (\cref{ch:b1:central-forces})؛
وعند سطح كرة نصف قطرها $R$ يكون $g = GM/R^2$. ولاحظ الوحدة:
إن \unit{N/kg} و\unit{m/s^2} هما الشيء نفسه، ولهذا تسقط الأجسام كلها
سقوطًا واحدًا.
\end{proposition}

\begin{proof}
قانون نيوتن في الجاذبية مسلَّم ويُستكشف في
\cref{ch:b1:central-forces}؛ و $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[النابض والخيط]\label{prop:b1:newton-dynamics:springstring}
\index{قانون هوك}\index{نابض}\index{شدّ}
نابض مثالي صلابته $k$ وطوله الطبيعي $\ell_0$، ممدود
إلى الطول $\ell$ على امتداد المتجهة الواحدية $\vect u$ المشيرة من طرفه
الثابت إلى الجسم، يؤثر في الجسم بما يسمّى \emph{\omterm{def:b1:newton-dynamics:momentum}{القوة} المرجِعة}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{قانون هوك}),
\]
نحو طوله الطبيعي. أما الخيط المثالي (عديم \omterm{def:b1:newton-dynamics:momentum}{الكتلة} وغير قابل للاستطالة)
فيؤثر \omterm{def:b1:newton-dynamics:momentum}{بقوة} \emph{شدّ} $\vect T$ على امتداده، ساحبًا، بالمقدار
نفسه عند طرفيه وعلى امتداده كله (حتى فوق بكرة عديمة \omterm{def:b1:newton-dynamics:momentum}{الكتلة}
عديمة الاحتكاك).
\end{proposition}

\begin{proof}
قانون هوك نموذج تجريبي صالح للتشوّهات الصغيرة. أما
الخيط: فعنصر عديم \omterm{def:b1:newton-dynamics:momentum}{الكتلة} يحقق $0 = \vect T_1 + \vect T_2$ بقانون
نيوتن الثاني، ومنه يُنقل الشدّ دون تغيّر.
\end{proof}

\begin{proposition}[قوى التماسّ: رد الفعل الناظمي والاحتكاك الجاف]\label{prop:b1:newton-dynamics:friction}
\index{رد الفعل الناظمي}\index{احتكاك جاف}\index{قوانين كولوم للاحتكاك}\index{معامل الاحتكاك}
يؤثر سطح صلب في جسم يلامسه \omterm{def:b1:newton-dynamics:momentum}{بقوة} \emph{رد فعل ناظمي}
$\vect N$ عمودي على السطح ودافع ($N \geq 0$: فالسطح
لا يستطيع أن يسحب)، وبقوة \emph{احتكاك} مماسية $\vect T$.
وقوانين كولوم: إن لم ينزلق الجسم كان $\abs{\vect T} \leq \mu_s N$
(فالاحتكاك \emph{السكوني} يتكيّف مع ما يلزم، إلى ذلك الحدّ)؛
وإن انزلق بسرعة $\vect v_g$ بالنسبة إلى السطح كان
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (وهو الاحتكاك \emph{الحركي}،
المعاكس للانزلاق)، مع معاملين $\mu_d \leq \mu_s$ من رتبة
$0.1$ إلى $1$ حسب المواد، لا يتوقفان على مساحة التماسّ
ولا على السرعة.
\end{proposition}
\admitted

\begin{proposition}[الاحتكاك المائعي]\label{prop:b1:newton-dynamics:drag}
\index{احتكاك مائعي}\index{قوة السحب}\index{قانون ستوكس}
جسم يتحرك بسرعة $\vect v$ في مائع ساكن يشعر \omterm{def:b1:newton-dynamics:momentum}{بقوة} \emph{سحب}
معاكسةً للمتجهة $\vect v$: فعند السرعات الصغيرة (جسم صغير في مائع
لزج) تكون خطية، $\vect F = -\alpha\vect v$ --- ومن أجل كرة نصف
قطرها $r$ في مائع لزوجته $\eta$ يكون $\alpha = 6\pi\eta r$
(ستوكس)؛ وعند السرعات الكبيرة (أجسام كبيرة سريعة في الهواء أو الماء) تكون
تربيعية،
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
حيث $\rho$ كثافة المائع، و $S$ مقطع الجسم المواجه
للجريان، و $C_x$ \emph{معامل سحب} عديم البعد (يساوي $0.5$ لكرة،
و$0.3$ لسيارة، و$1$ لقرص مستوٍ). وعدد رينولدز
$\rho vL/\eta$ (\cref{rem:b1:units-dimensions:pi}) هو الحَكَم: فخطية
دون $1$، وتربيعية فوق $1000$ تقريبًا.
\end{proposition}
\admitted

\begin{remark}[قوى أخرى]\label{rem:b1:newton-dynamics:others}
دافعة أرخميدس محصلةُ قوى ضغط، وتُشتقّ في
\cref{ch:b1:fluid-statics}؛ أما القوتان الكهربائية والمغناطيسية في
شحنة فموضوع \cref{ch:b1:charged-particles}. وكل قوى
هذا الفصل \emph{نماذج} لتفاعلات تماسّ أو تفاعلات
مجال، صالحة ضمن حدود معلنة --- فنابض هوك ينكسر،
واحتكاك كولوم يفشل عند الضغوط العالية، وسحب ستوكس يفشل عند
السرعات العالية. وجزء من كل حلّ هو التحقق من انطباق النموذج المستعمل.
\end{remark}

\section{حلّ مسألة في الديناميك}

\begin{method}[من الوضع إلى المعادلة]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{الجملة}: سمِّ الجسم المعامَل نقطةً.
  \item \emph{المرجع}: سمِّه وتحقق من أنه عطالي (أو عامله
        على أنه كذلك بالدقة المطلوبة).
  \item \emph{القوى}: أحصِ كل جسم يلامس الجملة أو يجذبها،
        وارسم كل \omterm{def:b1:newton-dynamics:momentum}{قوة} على مخطط (وهو مخطط القوى)؛
        ولا يدخل شيء غير ذلك.
  \item \emph{الإحداثيات}: اختر محاور ملائمة للحركة
        (\cref{met:b1:point-kinematics:choose}) وعبّر عن $\vect a$.
  \item \emph{اسقط} العلاقة $m\vect a = \sum\vect F$ على المحاور؛ واستعمل
        القيود (فالجسم على سطح يبقى عليه: $a_\perp = 0$؛ والخيط
        الثابت الطول: قيمتا سرعة متساويتان عند طرفيه).
  \item \emph{حلّ} ثم \emph{تحقق}: من الأبعاد والحالات الحدية و
        الإشارات ($N \geq 0$، و $\abs T \leq \mu_sN$ عند السكون).
\end{enumerate}
\end{method}

\begin{example}[كتلة على مستوٍ مائل]\label{ex:b1:newton-dynamics:incline}
\omterm{def:b1:newton-dynamics:momentum}{كتلة} $m$ على مستوٍ مائل بزاوية $\alpha$، بمعاملين $\mu_s$ و
$\mu_d$. والمحاور: $x$ نازلًا على المنحدر، و $y$ على امتداد الناظم. والوزن
$(mg\sin\alpha, -mg\cos\alpha)$، ورد الفعل $(0, N)$، والاحتكاك $(-T, 0)$
إن انزلقت نازلةً. وعلى $y$: $N = mg\cos\alpha$. وعند السكون على $x$:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$، أي $\tan\alpha
\leq \mu_s$: \omterm{def:b1:newton-dynamics:momentum}{فالكتلة} تصمد حتى الزاوية $\arctan\mu_s$ --- وهو
قياس للمعامل $\mu_s$ لا يحتاج إلى ميزان. وعند الانزلاق: $ma = mg\sin\alpha -
\mu_dmg\cos\alpha$، ومنه $a = g(\sin\alpha - \mu_d\cos\alpha)$، مستقلًّا
عن $m$: أي \qty{3.2}{m/s^2} من أجل $\alpha = 30^\circ$ و $\mu_d = 0.2$.
\end{example}

\begin{example}[النواس البسيط]\label{ex:b1:newton-dynamics:pendulum}
\index{نواس بسيط}
\omterm{def:b1:newton-dynamics:momentum}{كتلة} $m$ على خيط طوله $\ell$، بزاوية $\theta$ عن الشاقول
النازل. وبالإحداثيات القطبية عند نقطة التعليق: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$؛ والقوى:
الوزن $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$ و
الشدّ $-T\vect e_r$. وعلى $\vect e_\theta$: $m\ell\ddot\theta =
-mg\sin\theta$، أي
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
وهي من أجل الزوايا الصغيرة ($\sin\theta \approx \theta$) المعادلةُ التوافقية
مع $\omega_0 = \sqrt{g/\ell}$: فالدور $T_0 = 2\pi\sqrt{\ell/g}$،
وهو القيمة $k = 2\pi$ التي لم تستطع \cref{ex:b1:units-dimensions:pendulum}
تقديمها. وعلى $\vect e_r$: $T = mg\cos\theta + m\ell\dot\theta^2$ --- فالخيط
يشدّ أشدّ ما يشدّ عند الأسفل حيث تكون السرعة أكبر ما تكون.
\end{example}

\begin{proposition}[السقوط مع السحب؛ السرعة الحدية]\label{prop:b1:newton-dynamics:terminal}
\index{سرعة حدية}
جسم كتلته $m$ يسقط من السكون تحت وزنه \omterm{def:b1:newton-dynamics:momentum}{وقوة} سحب
يبلغ \emph{\omterm{prop:b1:newton-dynamics:terminal}{سرعة حدية}} تتوازن عندها القوتان. ففي السحب
الخطي: $m\dot v = mg - \alpha v$ و $v = v_\infty(1 - \eu^{-t/\tau})$ مع
$v_\infty = mg/\alpha$ و $\tau = m/\alpha$. وفي السحب التربيعي: $m\dot v = mg -
\tfrac12\rho C_xSv^2$ و $v = v_\infty\tanh(t/\tau)$ مع
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
الخطي: معادلة من الرتبة الأولى، $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). والتربيعي: مع
$u = v/v_\infty$ يكون $\dot u = (1 - u^2)/\tau$؛ وبفصل المتغيرات،
$\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$ (فالكسر
الناطق $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$ يتكامل إلى
$\tfrac12\ln\frac{1 + u}{1 - u}$)، ومنه $u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[سرعتان حديتان]\label{ex:b1:newton-dynamics:terminal}
مظلّي كتلته $m = \qty{80}{kg}$، منبطحًا، $C_xS \approx \qty{0.8}{m^2}$
في الهواء ($\rho = \qty{1.2}{kg/m^3}$): فيكون $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$، ويُبلَغ (إلى $90\%$) في نحو $1.5\tau =
\qty{6}{s}$. وقطرة ضباب نصف قطرها $r = \qty{10}{\micro m}$، في نظام ستوكس
(حيث $\eta_{\text{الهواء}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- ولهذا يعلق الضباب في الهواء.
\end{example}

\section{الحركة على دائرة: قوى القيد}

\begin{proposition}[الشرط المركزي]\label{prop:b1:newton-dynamics:centripetal}
\index{قوة مركزية}
النقطة المتحركة على دائرة نصف قطرها $R$ بقيمة سرعة $v$ يجب أن تتلقى
من محصلة القوى مركّبةً قدرها $mv^2/R$ موجَّهة نحو
المركز. وأيًّا كان ما يوفّرها --- شدّ خيط أو احتكاك أو
رد فعل طريق مائل أو الجاذبية --- فهو الذي يثبّت قيمة السرعة العظمى أو
الصغرى التي تكون الحركة عندها ممكنة.
\end{proposition}

\begin{proof}
قانون نيوتن الثاني مع تسارع فريني: فالمركّبة الناظمية
للمحصلة $\sum\vect F$ تساوي $mv^2/\rho$.
\end{proof}

\begin{example}[ثلاث دوائر]\label{ex:b1:newton-dynamics:circles}
\emph{المنعطف المستوي}: يجب أن يوفّر الاحتكاك $mv^2/R \leq \mu_sN = \mu_smg$،
ومنه $v \leq \sqrt{\mu_sgR}$: أي \qty{20}{m/s} من أجل $R = \qty{50}{m}$ و
$\mu_s = 0.8$. و\emph{المنعطف المائل} بزاوية $\beta$ بلا احتكاك: يكون
رد الفعل $\vect N$ ناظميًّا على الطريق؛ فشاقوليًّا $N\cos\beta = mg$، و
أفقيًّا $N\sin\beta = mv^2/R$، ومنه $v^2 = gR\tan\beta$ --- وهي قيمة السرعة
الوحيدة التي يُؤخذ عندها المنعطف بلا أيّ احتكاك. و\emph{الحلقة}: عند
القمة، مع $\vect N$ مشيرًا إلى أسفل نحو المركز، $N + mg =
mv^2/R$، ويقتضي $N \geq 0$ أن $v \geq \sqrt{gR}$: ودون ذلك تغادر العربة
المضمار (وهي \cref{pb:b1:point-kinematics:1} من جديد، مفسَّرةً الآن).
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
يقف شخص كتلته \qty{70}{kg} على ميزان في مصعد. فماذا
يقرأ الميزان (بالنيوتن وبوحدة ``الكيلوغرام'') حين يتسارع المصعد
إلى أعلى بمقدار \qty{2.0}{m/s^2}، وإلى أسفل بمقدار \qty{2.0}{m/s^2}،
وحين يتحرك بسرعة ثابتة، وحين يسقط سقوطًا حرًّا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
\omterm{def:b1:newton-dynamics:momentum}{كتلة} $m_1 = \qty{2.0}{kg}$ على طاولة عديمة الاحتكاك مربوطة بخيط
فوق بكرة عديمة الاحتكاك \omterm{def:b1:newton-dynamics:momentum}{بكتلة} معلَّقة $m_2 = \qty{1.0}{kg}$.
أوجد التسارع والشدّ.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
\omterm{def:b1:newton-dynamics:momentum}{كتلة} تنزلق على مستوٍ مائل بزاوية $30^\circ$ مع $\mu_d = 0.20$. احسب

الإجابة على كتلتها؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
\omterm{def:b1:newton-dynamics:momentum}{كتلة} \qty{0.50}{kg} معلَّقة بنابض صلابته \qty{200}{N/m}.

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}

\section{مسألة: المظلّي}

\begin{problem}[{مسألة نهاية الأسبوع --- من باب الطائرة إلى
الأرض: سقوط حرّ، وهواء يدفع، ومظلّة تُفتح بسرعة مفرطة،
والهبوط --- قانون حركة واحد وقانونا سحب}]
\label{pb:b1:newton-dynamics:1}
مظلّي بعتاده كتلته $m = \qty{80}{kg}$. والهواء:
$\rho = \qty{1.2}{kg/m^3}$ و $\eta = \qty{1.8e-5}{Pa.s}$. وفي السقوط
منبطحًا، $C_xS = \qty{0.80}{m^2}$؛ وتحت المظلّة المفتوحة، $C_xS =
\qty{25}{m^2}$. و $g = \qty{9.81}{m/s^2}$؛ ويُحسب $z$ نازلًا من
نقطة القفز.

\medskip
\noindent\textbf{الجزء I --- بلا هواء.}
\begin{enumerate}
  \item اكتب قانون نيوتن الثاني وأعطِ $v(t)$ و $z(t)$؛ واحسبهما
        معًا عند $t = \qty{10}{s}$.
  \item احسب الوزن.
  \item يشعر المظلّي بأنه ``بلا وزن'' في هذه المرحلة مع أن
        الجاذبية تعمل بكاملها. فسّر ذلك بالقوى (ما الذي ينقص
        مقارنةً بالوقوف على الأرض).
  \item هل مرجع الأرض عطالي بما فيه الكفاية لهذه المسألة؟ وسمِّ ما
        يُهمَل.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- السقوط عبر الهواء.}
\begin{enumerate}[resume]
  \item اكتب \omterm{prop:b1:newton-dynamics:drag}{قوة السحب} في السقوط منبطحًا بدلالة $v$.
  \item اكتب معادلة الحركة من أجل $v(t)$.
  \item أوجد \omterm{prop:b1:newton-dynamics:terminal}{السرعة الحدية} $v_\infty$ (بوحدتَي \unit{m/s} و
        \unit{km/h}).
  \item مع $u = v/v_\infty$ و $\tau = v_\infty/g$، بيّن أن
        $\dd u/\dd t = (1 - u^2)/\tau$؛ واحسب $\tau$.
  \item حلّ بفصل المتغيرات وبيّن أن
        $v = v_\infty\tanh(t/\tau)$.
  \item بعد كم من الزمن يكون $v = 0.9v_\infty$؟ و $0.99v_\infty$؟
  \item بيّن أن $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ واحسب
        المسافة المقطوعة عند $t = \qty{10}{s}$؛ وقارنها بالجزء I.
  \item قدّر عدد رينولدز عند $v_\infty$ مع $L = \qty{0.5}{m}$
        وبرّر القانون التربيعي.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- المظلّة تُفتح.}
\begin{enumerate}[resume]
  \item احسب \omterm{prop:b1:newton-dynamics:terminal}{السرعة الحدية} الجديدة تحت المظلّة المفتوحة.
  \item لو فُتحت المظلّة لحظيًّا عند $v = \qty{40}{m/s}$، فما
        قيمة التباطؤ بوحدة \unit{m/s^2} وبوحدة $g$؟ ولماذا يكون
        هذا غير مقبول؟
  \item تُفتح المظلّات الحقيقية على مدى نحو \qty{3}{s}. قدّر متوسط
        التباطؤ ومتوسط \omterm{def:b1:newton-dynamics:momentum}{قوة} الحزام على
        المظلّي.
  \item حالما تنفتح المظلّة كاملةً تتناقص السرعة نحو
        $v_\infty$ الجديدة: فما \omterm{def:b1:transient-regimes:firstorder}{ثابت الزمن} الجديد $\tau' = v_\infty'/g$،
        وماذا يعني بالنسبة إلى بلوغ النزول المستقر؟
  \item في النزول المستقر، ما \omterm{def:b1:newton-dynamics:momentum}{قوة} الحزام؟ برّر بالقانون
        الثاني.
  \item فسّر لماذا تتغير \omterm{prop:b1:newton-dynamics:terminal}{السرعة الحدية} بالمقدار $1/\sqrt{C_xS}$: فبأيّ
        عامل ترفعها مظلّة نصف المساحة؟
  \item قافز أثقل (\qty{100}{kg}) تحت المظلّة نفسها: فما \omterm{prop:b1:newton-dynamics:terminal}{السرعة
        الحدية}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الهبوط.}
\begin{enumerate}[resume]
  \item قيمة سرعة الهبوط هي $v_\infty$ للمظلّة. فمن أيّ ارتفاع
        كان يجب القفز، بلا هواء، لبلوغ الأرض بتلك
        السرعة؟
  \item الهبوط بساقين متيبستين يوقف الجسم على مدى \qty{0.5}{m}؛ والتدحرج
        على مدى \qty{1.5}{m}. قدّر متوسط التباطؤ (بوحدة $g$) و
        متوسط \omterm{def:b1:newton-dynamics:momentum}{القوة} على الساقين في كل حالة، واستنتج.
  \item قبيل ملامسة الأرض يسحب المظلّي المقبضين معًا،
        فيزيد ذلك سحب المظلّة برهةً. فسّر نوعيًّا
        أثر ذلك في السرعة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء V --- الأشياء الصغيرة تسقط سقوطًا مختلفًا.}
\begin{enumerate}[resume]
  \item قطرة مطر نصف قطرها \qty{1.0}{mm} (مع $C_x = 0.5$): احسب
        سرعتها الحدية بالقانون التربيعي، وتحقق من عدد
        رينولدز.
  \item قطرة ضباب نصف قطرها \qty{10}{\micro m}: استعمل قانون ستوكس
        من أجل $v_\infty$ وتحقق من أن عدد رينولدز صغير
        فعلًا.
  \item لخّص: معادلة حركة واحدة، ونظاما سحب، و
        الأعداد الثلاثة (بوحدة \unit{m/s}) التي تقرّر هل يسقط الشيء
        سقوط حجر أم سقوط ورقة أم لا يسقط البتة.
\end{enumerate}
\end{problem}
