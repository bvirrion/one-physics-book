\section{نموذج البصريات الهندسية}

\begin{definition}[الشعاع الضوئي؛ الوسط المتجانس]\label{def:b1:rays-reflection-refraction:ray}
\index{شعاع ضوئي}\index{بصريات هندسية}\index{وسط متجانس}
في نموذج \emph{البصريات الهندسية} ينتشر الضوء على امتداد
\emph{أشعة ضوئية}: وهي منحنيات موجَّهة تكون مستقيمات في وسط
\emph{متجانس} (له الخصائص نفسها في كل نقطة)، وتحمل الطاقة بعضها مستقلًّا
عن بعض. والحزمة رزمة من الأشعة؛ والمنبع النقطي يبعث أشعة في كل اتجاه.
\end{definition}

\begin{definition}[قرينة الانكسار]\label{def:b1:rays-reflection-refraction:index}
\index{قرينة الانكسار}\index{وسط شفاف}
الوسط \emph{الشفاف} يمرّر الضوء؛ وينتشر الضوء فيه بسرعة $v < c$.
و\emph{قرينة انكساره} هي
\[
n = \frac{c}{v} \geq 1 .
\]
الخلاء: $n = 1$؛ والهواء: $1.0003$؛ والماء: $1.33$؛ والزجاج العادي: من
$1.5$ إلى $1.6$؛ والألماس: $2.42$. والموجة ذات التواتر $\nu$ تحتفظ
بتواترها عند دخولها وسطًا، فينكمش طول موجتها من $\lambda_0 =
c/\nu$ في الخلاء إلى $\lambda = \lambda_0/n$.
\end{definition}

\begin{remark}[التبدد]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{تبدد (بصري)}\index{قانون كوشي}
تتوقف القرينة في المادة توقفًا طفيفًا على طول الموجة --- أي إن الوسط
\emph{مبدِّد}. وللزجاج في المجال المرئي يلائم قانون كوشي التجريبي
$n(\lambda) = A + B/\lambda^2$ ملاءمةً جيدة: فقيمة $n$ للأزرق أكبر منها
للأحمر بنحو $0.01$ إلى $0.02$. وعلى هذا الفرق يعيش الموشور \omterm{ex:b1:rays-reflection-refraction:rainbow}{وقوس قزح} في
آخر هذا الفصل.
\end{remark}

\begin{remark}[حدود النموذج]\label{rem:b1:rays-reflection-refraction:limits}
الضوء موجة، والموجة العابرة لفتحة عرضها $a$ تنتشر بزاوية من رتبة
$\lambda/a$ (الحيود، \cref{ch:b1:wave-propagation}). وتبقى الأشعة
وصفًا جيدًا ما دامت كل فتحة وكل عدسة وكل حاجز أكبر بكثير من طول الموجة
(من \qty{0.4}{} إلى \qty{0.8}{\micro m} للضوء المرئي): فثقب قطره
\qty{1}{mm} ينشر حزمة بمقدار $10^{-3}$ راديان، وهو مهمل على طاولة
تجارب؛ أما ثقب قطره \qty{1}{\micro m} فينشرها على نصف الفضاء كله، ولا
تبقى صورة الأشعة قائمة. \omterm{def:b1:rays-reflection-refraction:fiber}{وقلب} \omterm{def:b1:rays-reflection-refraction:fiber}{الليف البصري} أحادي النمط، وقطره
\qty{9}{\micro m}، هو أصلًا خارج النموذج.
\end{remark}

\begin{proposition}[رجوع الضوء]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{مبدأ الرجوع العكسي}
لا يتوقف مسار الشعاع على الجهة التي يقطعه فيها الضوء: فإذا ذهب الضوء
من $A$ إلى $B$ على مسار، سلك الضوء الآتي من $B$ المسار نفسه إلى $A$.
\end{proposition}
\admitted

\section{الانعكاس والانكسار}

\begin{definition}[الورود]\label{def:b1:rays-reflection-refraction:incidence}
\index{مستوي الورود}\index{زاوية الورود}\index{سطح كاسر}
الشعاع الذي يلاقي السطح الفاصل بين وسطين (أي \emph{السطح الكاسر}) في
نقطة $I$ يصنع مع \emph{الناظم} إلى السطح في $I$
\emph{زاوية الورود} $i_1$. أما \emph{مستوي الورود} فيحتوي الشعاع
الوارد والناظم.
\end{definition}

\begin{theorem}[قانونا الانعكاس والانكسار (سنيل--ديكارت)]\label{thm:b1:rays-reflection-refraction:snell}
\index{انعكاس!قانون}\index{انكسار!قانون}\index{قانون سنيل}
عند السطح الفاصل بين وسطين قرينتاهما $n_1$ (جهة الورود) و
$n_2$:
\begin{enumerate}
  \item يقع الشعاعان المنعكس والمنكسر في \omterm{def:b1:rays-reflection-refraction:incidence}{مستوي الورود}؛
  \item يصنع الشعاع المنعكس مع الناظم الزاوية $i_1' = i_1$،
        في الجهة الأخرى من الناظم؛
  \item يصنع الشعاع المنكسر مع الناظم زاوية $i_2$ تحقق
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[من أين يأتي القانونان]\label{rem:b1:rays-reflection-refraction:fermat}
\index{مبدأ فيرما}
يتبع القانونان معًا من \emph{مبدأ فيرما}: بين نقطتين يسلك الضوء المسار
الأقصر زمنَ قطع. فالانعكاس بزاويتين متساويتين هو أقصر مسار منكسر يلامس
المرآة؛ وقانون سنيل هو المسار الذي يقايض طريقًا أطول في الوسط السريع
بطريق أقصر في الوسط البطيء
(\cref{exo:b1:rays-reflection-refraction:12}). وفي مجلد السنة الثانية
يُشتقّ القانونان من الطبيعة الموجية للضوء عند سطح فاصل.
\end{remark}

\begin{proposition}[إلى أين ينحني الشعاع]\label{prop:b1:rays-reflection-refraction:bending}
عند الدخول إلى وسط أشدّ انكسارًا ($n_2 > n_1$) ينحني الشعاع
\emph{نحو} الناظم ($i_2 < i_1$)؛ وعند الدخول إلى وسط أقلّ انكسارًا
ينحني \emph{مبتعدًا} عنه ($i_2 > i_1$). أما الشعاع المنطبق على الناظم
فيعبر دون انحراف.
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$ مع تزايد $\sin$ على
$[0, \pi/2]$.
\end{proof}

\begin{example}[العمق الظاهري]\label{ex:b1:rays-reflection-refraction:depth}
\index{عمق ظاهري}
قطعة نقدية على عمق $h$ تحت الماء ($n = 1.33$)، تُرى من فوقها تقريبًا.
الشعاع الخارج منها والواصل إلى السطح بزاوية صغيرة $i_2$ (في الماء)
يغادر بالزاوية $i_1$ حيث $\sin i_1 = n\sin i_2$؛ وللزوايا الصغيرة
$i_1 \approx n\,i_2$. وتمدّ العين الشعاع الخارج إلى الوراء: فيبدو
آتيًا من عمق $h'$ حيث $\tan i_1 \approx x/h'$ و
$\tan i_2 \approx x/h$ لنقطة السطح نفسها الواقعة على بعد $x$ من
الشاقول، ومنه $h' = h\,i_2/i_1 = h/n$. فمسبح عمقه \qty{2.0}{m} يبدو
بعمق \qty{1.5}{m}؛ وسمكة ترى الصياد أعلى ممّا هو بمقدار $1.33$ مرة.
\end{example}

\begin{theorem}[الانعكاس الكلي]\label{thm:b1:rays-reflection-refraction:tir}
\index{انعكاس كلي}\index{زاوية حدية}
الضوء الذاهب من وسط قرينته $n_1$ نحو وسط أقلّ انكسارًا
($n_2 < n_1$) لا ينكسر إلا إذا كان $i_1 \leq i_c$، حيث تحقق
\emph{\omterm{thm:b1:rays-reflection-refraction:tir}{الزاوية الحدية}}
\[
\sin i_c = \frac{n_2}{n_1} .
\]
ومن أجل $i_1 > i_c$ لا يوجد شعاع منكسر: فيعكس السطح الضوء كله ---
وهو \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{الانعكاس الكلي}}.
\end{theorem}

\begin{proof}
يطلب قانون سنيل أن يكون $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$؛
وهذا يتجاوز $1$ حالما يصير $\sin i_1 > n_2/n_1$، فلا توجد عندئذٍ زاوية
$i_2$. وعند $i_1 = i_c$ يمسّ الشعاع المنكسر السطح
($i_2 = 90^\circ$). أما أن الطاقة تنعكس \emph{كلها} حينئذٍ فعبارة عن
الشدّات لا تعالجها \omterm{def:b1:rays-reflection-refraction:ray}{البصريات الهندسية}؛ وتؤكدها المعالجة الموجية في
مجلد السنة الثانية.
\end{proof}

\section{الليف البصري}

\begin{definition}[الليف ذو القرينة المتدرّجة القفزية]\label{def:b1:rays-reflection-refraction:fiber}
\index{ليف بصري}\index{قلب}\index{غلاف}\index{فتحة عددية}
\emph{الليف البصري ذو القفزة القرينية} \emph{قلبٌ} زجاجي أسطواني
قرينته $n_1$ يكسوه \emph{غلاف} قرينته $n_2$ أدنى قليلًا.
والضوء الداخل إلى القلب داخل مخروط نصف زاويته $\theta_a$ حول
المحور يُوجَّه بانعكاسات كلية متتالية على سطح القلب--الغلاف.
و\emph{الفتحة العددية} هي $\mathrm{NA} = \sin\theta_a$.
\end{definition}

\begin{proposition}[مخروط القبول والتبدد بين الأنماط]\label{prop:b1:rays-reflection-refraction:fiber}
\index{زاوية القبول}\index{تبدد بين الأنماط}
من أجل ليف وجهُ دخوله في الهواء،
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
وعلى طول $L$ يصل الشعاع الموجَّه المحاذي للمحور والشعاع الموجَّه الأشدّ
ميلًا بفارق زمني
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
بينهما (\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{التبدد بين الأنماط}}).
\end{proposition}

\begin{proof}
الشعاع الداخل بزاوية $\theta$ إلى المحور ينكسر إلى $\theta'$
حيث $\sin\theta = n_1\sin\theta'$، ويلاقي \omterm{def:b1:rays-reflection-refraction:fiber}{الغلاف} بورود
$i = 90^\circ - \theta'$. والتوجيه يقتضي $\sin i \geq n_2/n_1$، أي
$\cos\theta' \geq n_2/n_1$، أي $\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$؛ ثم نضرب في $n_1$. والشعاع المحوري يقطع $L$
بسرعة $c/n_1$، فيكون $t_0 = Ln_1/c$. أما الشعاع الأشدّ ميلًا فيتعرّج
بزاوية $\theta'_{\max}$ حيث $\cos\theta'_{\max} = n_2/n_1$، ومساره
$L/\cos\theta'_{\max} = L n_1/n_2$، ومنه $t_1 = L n_1^2/(n_2 c)$ ثم
$\Delta t = t_1 - t_0$.
\end{proof}

\begin{example}[أعداد ليف اتصالات]\label{ex:b1:rays-reflection-refraction:fibernum}
$n_1 = 1.480$ و $n_2 = 1.465$: فتكون $\mathrm{NA} = \sqrt{2.190 - 2.146} =
0.21$ و $\theta_a = 12^\circ$؛ و $i_c = 81.8^\circ$، فتبقى الأشعة
الموجَّهة داخل $8.2^\circ$ من المحور. والاتساع بين الأنماط: $\Delta t/L =
(1.48/c)(1.48/1.465 - 1) = \qty{51}{ns/km}$. وعلى \qty{2}{km} تتسع
النبضة بمقدار \qty{0.1}{\micro s}، وهو ما يحدّ معدّل البتّات قرب
\qty{5}{Mbit/s}: وتتابع مسألة نهاية الأسبوع هذا القيد إلى علاجه، أي
الليف أحادي النمط.
\end{example}

\section{المرآة المستوية}

\begin{proposition}[الصورة في مرآة مستوية]\label{prop:b1:rays-reflection-refraction:mirror}
\index{مرآة مستوية}\index{صورة تخيلية}\index{استجماع}
تبدو كل الأشعة الصادرة عن نقطة $A$ والمنعكسة على \omterm{prop:b1:rays-reflection-refraction:mirror}{مرآة مستوية} آتيةً من
النقطة $A'$ المناظرة للنقطة $A$ بالنسبة إلى مستوي المرآة: فالنقطة $A'$
هي \emph{صورة} $A$، وهي \emph{تخيلية} (لا يمرّ الضوء بها)، والمرآة
\emph{مستجمعة بدقة}: أي إن لكل نقطة صورةً نقطية تامة.
\end{proposition}

\begin{proof}
الشعاع الصادر عن $A$ والملاقي للمرآة في $I$ بورود $i$ يغادر بالزاوية
$i$ في الجهة الأخرى من الناظم. والمستقيم المنعكس والناظم في $I$ يصنعان
الزاوية $i$، وكذلك يفعل المستقيم $IA$؛ فبالتناظر بالنسبة إلى مستوي
المرآة يكون المستقيم المنعكس نظير المستقيم $IA$، ومنه يمرّ بالنقطة
المناظرة $A'$ --- ومهما تكن $I$.
\end{proof}

\begin{proposition}[إدارة مرآة]\label{prop:b1:rays-reflection-refraction:rotation}
إدارة \omterm{prop:b1:rays-reflection-refraction:mirror}{مرآة مستوية} بزاوية $\alpha$ حول محور في مستويها، عمودي على \omterm{def:b1:rays-reflection-refraction:incidence}{مستوي
الورود}، تدير الشعاع المنعكس بزاوية
$2\alpha$.
\end{proposition}

\begin{proof}
يدور الناظم بالزاوية $\alpha$، فتتغير \omterm{def:b1:rays-reflection-refraction:incidence}{زاوية الورود} من $i$ إلى
$i + \alpha$؛ والشعاع المنعكس، وهو عند $i + \alpha$ في الجهة الأخرى من
الناظم الجديد، قد تحرك بمقدار $\alpha$ (الناظم) $+ \alpha$ (الزاوية) $=
2\alpha$ عن اتجاهه القديم.
\end{proof}

\begin{example}[رافعة الغلفانومتر]\label{ex:b1:rays-reflection-refraction:galvanometer}
مرآة صغيرة ملصقة بوشيعة غلفانومتر، يضيئها مصباح فتلقي بقعة على سُلّم
على بعد \qty{2}{m}: فدوران الوشيعة بمقدار
$\qty{1}{mrad}$ يحرّك البقعة بمقدار $2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ --- إنها رافعة بصرية تكبّر دون
احتكاك، وهي مبدأ الغلفانومتر ذي المرآة والماسح الليزري.
\end{example}

\section{الموشور}

\begin{proposition}[صيغ الموشور]\label{prop:b1:rays-reflection-refraction:prism}
\index{موشور}\index{انحراف}\index{انحراف أصغري}
الشعاع العابر لموشور زاوية رأسه $A$ وقرينته $n$ في الهواء، في المستوي
العمودي على الحرف، بالزوايا $i$ (الدخول) و $r$ و $r'$
(في الداخل) و $i'$ (الخروج) مقيسةً من النواظم، يحقق
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
وينحرف بمقدار $D = i + i' - A$. ويكون الانحراف أصغريًّا حين يكون
المسار متناظرًا، $i = i'$ و $r = r' = A/2$؛ وعندئذٍ
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
قانون سنيل عند كل وجه؛ و $r + r' = A$ لأن الناظمين يصنعان
الزاوية $A$ (في الرباعي المكوَّن من الرأس ونقطتَي الورود وتقاطع
النواظم). والانحراف الكلي
$(i - r) + (i' - r') = i + i' - A$. وبرجوع الضوء يكون الانحراف الدالة
نفسها من $i$ ومن $i'$؛ ولو وقع الأصغر عند $i_0 \neq i_0'$ لوقع أيضًا
عند الورود المعكوس $i_0'$، فكان هناك أصغران، وهو ما يمنعه رتابة
$D(i)$ على كل جانب من أصغرها: فالأصغر عند $i = i'$.
وعندئذٍ $r = r' = A/2$ و $D_{\min} = 2i - A$؛ وبتعويض $i = (A +
D_{\min})/2$ في $\sin i = n\sin(A/2)$ نحصل على الصيغة.
\end{proof}

\begin{example}[قياس قرينة]\label{ex:b1:rays-reflection-refraction:prismnum}
موشور زاويته $60^\circ$ يُدار حتى يصير انحراف حزمة صفراء أصغريًّا:
$D_{\min} = 38.9^\circ$. عندئذٍ $n = \sin 49.45^\circ/\sin 30^\circ
= 0.760/0.500 = 1.52$. \omterm{prop:b1:rays-reflection-refraction:prism}{والانحراف الأصغري} هو الطريقة المعيارية لقياس
قرينة زجاج --- فالأصغر الحادّ، الذي يُوجد بإدارة الموشور جيئةً
وذهابًا، غير حساس لسوء ضبط صغير.
\end{example}

\begin{remark}[الموشورات الرقيقة والمطيافات]\label{rem:b1:rays-reflection-refraction:thin}
من أجل $A$ صغيرة وورود صغير تساوي الجيوبُ الزوايا فيكون
$D \approx (n - 1)A$ مستقلًّا عن $i$: أي إن الموشور الرقيق يحرف كل
شعاع بالزاوية نفسها --- فإسفين من الزجاج ``موشور'' يحرف الضوء، وستتبين
العدسة رصًّا لأسافين كهذه. ولأن $n$ يتوقف على $\lambda$ يتوقف الانحراف
عليه أيضًا: فالموشور ينشر الضوء الأبيض طيفًا (والأزرق أشدّ انحرافًا)،
وهذا \omterm{def:b1:rays-reflection-refraction:fiber}{قلب} المطياف الموشوري.
\end{remark}

\section{قوس قزح}

\begin{example}[قوس قزح عند ديكارت]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{قوس قزح}
شعاع من ضوء الشمس يدخل قطرة مطر كروية بورود $i$، فينكسر
($\sin i = n\sin r$)، وينعكس مرة واحدة على السطح الخلفي، ويخرج بعد
انكسار ثانٍ. وكل انكسار يحرفه بمقدار $i - r$، والانعكاس الداخلي
بمقدار $\pi - 2r$: فالانحراف الكلي $D(i) = \pi + 2i - 4r$. وتملأ
الأشعة القطرة عند كل $i$؛ ويرى الراصد أشدّ الضوء حيث يغادر عدد كبير
من الأشعة في الاتجاه نفسه تقريبًا، أي حيث يكون $D$ ساكنًا:
$\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$. وباشتقاق قانون سنيل نجد
$\cos i = n\cos r\,\dd r/\dd i$، فيصير الشرط $2\cos i = n\cos r$؛
وبالتربيع واستعمال
$n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$ نجد
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
وللماء، حيث $n = 1.333$: $i = 59.4^\circ$ و $r = 40.2^\circ$ و
$D_{\min} = 137.9^\circ$. ويعود الضوء نحو جهة الشمس
بزاوية $180^\circ - 137.9^\circ = 42^\circ$ عن الاتجاه المقابل للشمس:
\omterm{ex:b1:rays-reflection-refraction:rainbow}{فقوس قزح} مخروط نصف زاويته $42^\circ$ حول ظلّ رأس
الراصد. ويقع الأحمر ($n = 1.331$) عند $42.4^\circ$، والبنفسجي
($n = 1.343$) عند $40.6^\circ$ --- فالأحمر خارجًا والبنفسجي داخلًا.
\end{example}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}

ساكنًا تمامًا حين يكون $\sin i_1/v_1 = \sin i_2/v_2$ --- أي قانون سنيل.
\end{exercise}

\section{مسألة: وصلة الليف البصري}

\begin{problem}[{مسألة نهاية الأسبوع --- خيط زجاجي تحت البحر: إلى أيّ
مدى وبأيّ سرعة تستطيع نبضة ضوء أن تحمل بتًّا قبل أن تطمسها الهندسة
والزجاج والتبدد}]
\label{pb:b1:rays-reflection-refraction:1}
تستعمل وصلةٌ ليفًا ذا قفزة قرينية، قلبه $n_1 = 1.480$ وغلافه
$n_2 = 1.465$، ونصف قطر قلبه $a = \qty{25}{\micro m}$، عند طول الموجة
$\lambda_0 = \qty{1550}{nm}$. المعطيات: $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{الجزء I --- الضوء في الزجاج.}
\begin{enumerate}
  \item عرّف القرينة واحسب سرعة الضوء في \omterm{def:b1:rays-reflection-refraction:fiber}{القلب}.
  \item كم تستغرق نبضة لتقطع \qty{50}{km} على امتداد
        المحور؟
  \item احسب تواتر الضوء وطول موجته داخل
        \omterm{def:b1:rays-reflection-refraction:fiber}{القلب}.
  \item لماذا يُكسى \omterm{def:b1:rays-reflection-refraction:fiber}{القلب} بزجاج أدنى قرينة بدلًا من تركه
        عاريًا في الهواء، وهو ما يعطي قفزة قرينية أكبر؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- التوجيه.}
\begin{enumerate}[resume]
  \item احسب \omterm{thm:b1:rays-reflection-refraction:tir}{الزاوية الحدية} عند سطح القلب--الغلاف.
  \item استنتج أكبر زاوية يمكن أن يصنعها شعاع موجَّه مع المحور.
  \item يدخل شعاع من الهواء عبر الوجه المستوي بزاوية $\theta$ إلى
        المحور. بيّن أنه يكون موجَّهًا إذا كان $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$، واحسب \omterm{prop:b1:rays-reflection-refraction:fiber}{زاوية القبول}
        $\theta_a$.
  \item أعطِ \omterm{def:b1:rays-reflection-refraction:fiber}{الفتحة العددية}. ولماذا تجعل القفزة القرينية الأكبر
        تغذية الليف بالضوء أسهل؟
  \item يُثنى الليف على دائرة نصف قطرها $R_b$. لنأخذ
        شعاعًا يسير موازيًا للمحور على امتداد الحافة الداخلية
        \omterm{def:b1:rays-reflection-refraction:fiber}{للقلب}: بيّن أنه يلاقي الجدار الخارجي بورود $i$ حيث
        $\sin i = R_b/(R_b + 2a)$، وأوجد أصغر $R_b$ يبقى عنده
        موجَّهًا.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- \omterm{prop:b1:rays-reflection-refraction:fiber}{التبدد بين الأنماط}.}
\begin{enumerate}[resume]
  \item عبّر عن زمنَي قطع الشعاع المحوري والشعاع الموجَّه الأشدّ
        ميلًا على طول $L$.
  \item استنتج الاتساع $\Delta t$ لكل كيلومتر، وقيمته على
        $L = \qty{2}{km}$.
  \item البتّ نبضة قصيرة؛ ويجب أن تبقى نبضتان متتاليتان
        متباعدتين بمقدار $2\Delta t$ على الأقل كي يُميَّز بينهما. قدّر
        معدّل البتّات الأعظمي على \qty{2}{km}.
  \item بيّن أن جداء (معدّل البتّات) $\times$ (الطول) ثابت
        لهذا الليف، واحسبه بوحدة \unit{Mbit.s^{-1}.km}.
  \item في الليف ذي القرينة المتدرّجة تتناقص القرينة تناقصًا سلسًا من
        المحور إلى الخارج. فسّر نوعيًّا لماذا يقلّل هذا الاتساع
        بين الأنماط.
  \item قطر \omterm{def:b1:rays-reflection-refraction:fiber}{قلب} الليف أحادي النمط نحو \qty{9}{\micro m}.
        قارنه بالمقدار $\lambda_0$ وفسّر لماذا لا يستطيع نموذج الأشعة
        وصفه؛ وما المكسب؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الوهن والتبدد اللوني.}
يوهن الليف الاستطاعة بمعامل $\alpha = \qty{0.20}{dB/km}$ عند
\qty{1550}{nm}: $P(L) = P_0\,10^{-\alpha L/10}$. ويطلق المرسل
$P_0 = \qty{1.0}{mW}$؛ ويحتاج المستقبل إلى
$P_{\min} = \qty{1.0}{\micro W}$ على الأقل. ولأن $n$ يتوقف على $\lambda$
ينتشر طولا موجة بسرعتين مختلفتين قليلًا: فمنبع عرضه الطيفي
$\Delta\lambda$ يعطي اتساعًا للنبضة $\Delta t_c = D\,\Delta\lambda\,L$
حيث $D = \qty{17}{ps/(nm.km)}$.
\begin{enumerate}[resume]
  \item ما الجزء الباقي من الاستطاعة بعد \qty{100}{km}؟
  \item أوجد الطول الأعظمي قبل الحاجة إلى مضخّم.
  \item تُوضع المضخّمات كل \qty{100}{km}: فما الهامش، بوحدة dB
        وكنسبة استطاعات، الذي يتركه ذلك للوصلات واللحامات؟
  \item احسب $\Delta t_c$ على \qty{100}{km} من أجل منبع LED
        ($\Delta\lambda = \qty{40}{nm}$) ومن أجل ليزر
        ($\Delta\lambda = \qty{0.10}{nm}$)؛ واستنتج معدّل البتّات
        الأعظمي لكلٍّ منهما على ذلك الطول (بالمعيار نفسه أعلاه).
  \item عند \qty{850}{nm} يوهن السيليس \qty{2.0}{dB/km}. فما
        المسافة التي تسمح بها ميزانية الاستطاعة نفسها هناك؟ ولماذا
        \qty{1550}{nm} هي طول موجة الاتصالات؟
  \item يوهن الكبل النحاسي المحوري نحو \qty{30}{dB/km} عند
        التواترات اللازمة لمعدّل \qty{1}{Gbit/s}. قارن تباعد
        المضخّمات بتباعدها في الليف.
\end{enumerate}

\medskip
\noindent\textbf{الجزء V --- الوصلة كاملة.}
\begin{enumerate}[resume]
  \item من أجل الليف ذي القفزة القرينية على \qty{2}{km}، أيّ أثر يحدّ
        معدّل البتّات: الاتساع بين الأنماط أم الاتساع اللوني (منبع
        ليزري) أم الوهن؟ أعطِ معدّل البتّات الحادّ.
  \item من أجل ليف أحادي النمط على \qty{100}{km} مع المنبع
        الليزري: أيّ أثر يحدّ، وعند أيّ معدّل بتّات؟
  \item تحمل الوصلة أحادية النمط $80$ ليزرًا بأطوال موجة
        متباعدة بمقدار \qty{0.8}{nm} (تعديد بأطوال الموجة)، كلٌّ عند معدّل
        البتّات في السؤال السابق. فما السعة الكلية؟
  \item كوّن جداء معدّل البتّات في الطول للوصلة ذات القفزة القرينية
        ولقناة أحادية النمط واحدة؛ وبأيّ عامل يفوز الليف
        أحادي النمط، وأيّ حقيقة فيزيائية واحدة مسؤولة عن ذلك؟
\end{enumerate}
\end{problem}
