\section{الضغط في مائع ساكن}

\begin{proposition}[تماثل مناحي الضغط]\label{prop:b1:fluid-statics:isotropy}
\index{ضغط!في مائع}
في مائع ساكن، القوة التي يؤثر بها المائع في سطح صغير
$\dd S$ عند نقطة $M$ هي $\dd\vect F = P(M)\,\dd S\,\vect n$، ناظمية على
السطح وموجَّهة نحوه، بضغط $P(M)$ يتوقف على النقطة
و\emph{لا} يتوقف على توجّه السطح.
\end{proposition}

\begin{proof}[برهان جزئي]
لا ينقل مائع ساكن أيّ قوة مماسية (وإلا لجرى): فالقوة ناظمية.
ولننظر إلى موشور مثلثي صغير جدًّا من المائع عند $M$
لوجوهه توجّهات مختلفة: فهو في توازن تحت تأثير قوى الضغط
على وجوهه وتحت تأثير ثقله؛ والثقل يتناسب مع الحجم (القوة
الثالثة للبعد)، وقوى الضغط مع المساحات (القوة الثانية)،
فإذا صغر الموشور كفايةً لم تبق إلا قوى الضغط، ويقتضي
توازنها في كل اتجاه الضغط $P$ نفسه على كل وجه.
\end{proof}

\begin{theorem}[العلاقة الأساسية لسكون الموائع]\label{thm:b1:fluid-statics:fundamental}
\index{علاقة هيدروستاتيكية}
في مائع ساكن في ثقالة منتظمة $\vect g = -g\vect e_z$ (المحور $z$
نحو الأعلى)، لا يتوقف الضغط إلا على $z$ ويكون
\[
\frac{\dd P}{\dd z} = -\rho g ,
\]
حيث $\rho$ كثافة المائع عند ذلك المستوى. فالضغط يتزايد
نزولًا؛ والأسطح متساوية الضغط (\emph{الأسطح المتساوية الضغط})
أفقية، وكذلك السطح الحر لسائل.
\end{theorem}

\begin{proof}
لنأخذ شريحة أفقية من المائع مساحتها $S$ بين $z$ و $z + \dd z$:
تدفعها من الأسفل $P(z)S$ نحو الأعلى، ومن الأعلى $P(z + \dd z)S$
نحو الأسفل، ويجذبها ثقلها $\rho gS\,\dd z$. وعند السكون:
$P(z)S - P(z + \dd z)S - \rho gS\,\dd z = 0$. وليس على شريحة أفقية
من الثقالة أيّ مركّبة أفقية، فالضغط $P$ منتظم أفقيًّا.
\end{proof}

\begin{corollary}[مائع غير قابل للانضغاط]\label{cor:b1:fluid-statics:incompressible}
\index{ضغط هيدروستاتيكي}
في سائل كثافته $\rho$ منتظمة، يكون الضغط على عمق $h$ تحت
نقطة قيمته فيها $P_0$ هو
\[
P = P_0 + \rho gh .
\]
ففي الماء: \qty{1}{bar} إضافي كل \qty{10}{m}. ونقطتان من سائل
متصل عند المستوى نفسه لهما الضغط نفسه؛ وتقف السطوح الحرة
في الأواني المستطرقة عند مستوى واحد؛ وينتقل ضغط يُطبَّق عند
أيّ موضع من سائل محبوس، دون نقصان، إلى كل نقطة منه
(مبدأ باسكال).
\end{corollary}

\begin{proof}
نكامل $\dd P/\dd z = -\rho g$ مع $\rho$ ثابتة؛ وما بقي هو أفقية
الأسطح المتساوية الضغط ووحدانية $P$ عند مستوى معطى.
\end{proof}

\begin{example}[مقاييس الضغط والمكابس]\label{ex:b1:fluid-statics:manometer}
\index{مقياس ضغط}\index{مكبس هيدروليكي}
أنبوب على شكل U فيه زئبق ($\rho = \qty{13600}{kg/m^3}$) يفترق
عمودا السائل فيه بمقدار $h = \qty{760}{mm}$ يقيس فرق ضغط
$\rho gh = \qty{1.013e5}{Pa}$ --- أي الجو، الذي يحمل عمودًا من الماء
ارتفاعه \qty{10.3}{m}. وفي \omterm{ex:b1:fluid-statics:manometer}{مكبس هيدروليكي}: قوة $F_1$ على مكبس
مساحته $S_1$ ترفع الضغط في المائع كله بمقدار $F_1/S_1$؛ فتعطي
على المكبس الكبير $S_2$ القوة $F_2 = F_1S_2/S_1$ --- أي خمسين ضعفًا
مقابل خمسين ضعفًا من المساحة، بثمن مسافة أقصر خمسين مرة
(فالطاقة محفوظة: $F_1d_1 = F_2d_2$).
\end{example}

\section{الجو}

\begin{proposition}[الجو المتساوي الحرارة]\label{prop:b1:fluid-statics:atmosphere}
\index{صيغة بارومترية}\index{ارتفاع مميّز}
من أجل \omterm{def:b1:kinetic-theory:model}{غاز مثالي} كتلته المولية $M$ عند \omterm{def:b1:kinetic-theory:temperature}{درجة حرارة} $T$ منتظمة
في ثقالة منتظمة، يكون
\[
P(z) = P_0\,\eu^{-z/H}, \qquad H = \frac{RT}{Mg} ,
\]
وهو \emph{\omterm{prop:b1:fluid-statics:atmosphere}{الارتفاع المميّز}}: نحو \qty{8}{km} للهواء؛ وينتصف الضغط
كل $H\ln2 \approx \qty{5.5}{km}$.
\end{proposition}

\begin{proof}
لدينا $\rho = PM/RT$ (معادلة الحالة)، ومنه $\dd P/\dd z = -(Mg/RT)P$: وهي
معادلة خطية من المرتبة الأولى حلّها $P = P_0\eu^{-Mgz/RT}$.
\end{proof}

\begin{example}[هواء رقيق]\label{ex:b1:fluid-statics:altitude}
مع $T = \qty{260}{K}$ (وهو متوسط على الطبقات الدنيا من الجو) يكون $H =
\qty{7.6}{km}$: فيكون $P = 0.52P_0$ عند \qty{5000}{m}،
و $0.31P_0$ عند قمة إفرست --- وهو قريب من القيمتين المقيستين
$0.54$ و $0.33$؛ والجو الحقيقي يبرد مع الارتفاع، وهو ما يأخذه
مجلد السنة الثانية في الحسبان. ويبيّن النموذج أيضًا لماذا يختلف
البحر: فالماء أكثف ثمانمئة مرة ويكاد لا ينضغط، فيتزايد ضغطه
خطيًّا، بارًا كل عشرة أمتار، بلا \omterm{prop:b1:fluid-statics:atmosphere}{ارتفاع مميّز}.
\end{example}

\section{القوى المؤثرة في الجدران}

\begin{proposition}[القوة المؤثرة في جدار شاقولي]\label{prop:b1:fluid-statics:wall}
\index{مركز الضغط}
سائل عمقه $h$ يستند إلى جدار شاقولي مستطيل عرضه $L$ (والضغط
الجوي يؤثر في الوجهين فيتلاشى) يؤثر فيه بالقوة الأفقية
\[
F = \tfrac12\rho gh^2L ,
\]
وهي تكافئ قوة وحيدة مطبَّقة عند \emph{\omterm{prop:b1:fluid-statics:wall}{مركز الضغط}}، على العمق
$2h/3$ --- فالمحصلة تؤثر في الأسفل، حيث الضغط أعظمي.
\end{proposition}

\begin{proof}
على العمق $x$ يكون الضغط المقياسي $\rho gx$؛ وعلى الشريط الذي
ارتفاعه $\dd x$ تكون القوة $\rho gxL\,\dd x$؛ ونكامل من $0$ إلى $h$. وعزمها
بالنسبة إلى مستقيم السطح هو $\int_0^h\rho gx^2L\,\dd x = \tfrac13\rho gh^3L$؛
وبقسمته على $F$ نجد \omterm{def:b1:angular-momentum:moment}{ذراع الرافعة} $2h/3$.
\end{proof}

\section{مبرهنة أرخميدس}

\begin{theorem}[أرخميدس]\label{thm:b1:fluid-statics:archimedes}
\index{مبرهنة أرخميدس}\index{دافعة}\index{مركز الدفع}
جسم مغمور في مائع ساكن يتلقى من قوى الضغط محصلةً --- هي
\emph{الدافعة} --- مساوية لثقل المائع الذي يزيحه ومعاكسة له،
\[
\vect\Pi = -\rho_{\text{المائع}}V_{\text{المغمور}}\,\vect g ,
\]
ومطبَّقة عند مركز كتلة المائع المُزاح (وهو \emph{\omterm{thm:b1:fluid-statics:archimedes}{مركز
الدفع}}).
\end{theorem}

\begin{proof}
لا تتوقف قوى الضغط على سطح الجسم إلا على شكل ذلك السطح
وعلى المائع من حوله. ولنستبدل بالجسم، في الذهن، مائعًا ساكنًا
يملأ الحجم نفسه: فهذا المائع في توازن تحت تأثير ثقله وقوى
الضغط نفسها، ومن ثمّ توازن قوى الضغط ثقله --- أي أن مجموعها
$-\rho V\vect g$، مطبَّقًا عند مركز كتلته. ثم نرفع التجربة الذهنية؛
وقوى الضغط لم تتغير.
\end{proof}

\begin{corollary}[الطفو]\label{cor:b1:fluid-statics:floating}
\index{طفو}
جسم كثافته $\rho_b$ يطفو في مائع كثافته $\rho_f > \rho_b$ وقد غُمر
منه الجزء $\rho_b/\rho_f$ من حجمه؛ فإذا كان $\rho_b >
\rho_f$ غاص، بثقل ظاهري $(\rho_b - \rho_f)Vg$.
\end{corollary}

\begin{proof}
الثقل $\rho_bVg$ يساوي الدافعة $\rho_fV_{\text{المغمور}}g$.
\end{proof}

\begin{example}[ثلاث دافعات]\label{ex:b1:fluid-statics:buoy}
مقياس الكثافة أنبوب مثقَّل يغوص حتى يزيح ثقله هو: فكلما
ازدادت كثافة السائل قلّ غوصه --- وهو مقياس كثافة يُقرأ على
ساقه. ومنطاد حجمه \qty{2800}{m^3} من الهواء الساخن عند
\qty{373}{K} يزيح \qty{3.4}{t} من الهواء البارد ويزن \qty{2.6}{t} من
الهواء الساخن: فتكون الحمولة \qty{0.7}{t} (\cref{pb:b1:kinetic-theory:1}).
وسفينة حمولتها \qty{100000}{t} تزيح \qty{97600}{m^3} من ماء البحر؛
وفي الماء العذب ($\qty{1000}{kg/m^3}$) عليها أن تزيح $2.5\%$ أكثر
فتغطس أعمق --- ومن هنا علامات الحمولة على بدنها.
\end{example}

\begin{remark}[الاستقرار]\label{rem:b1:fluid-statics:stability}
\index{مركز الاستقرار}
جسم طافٍ يميل بزاوية صغيرة يعود إلى وضعه إذا أنتجت الدافعة،
وقد صارت مطبَّقة عند المركز المزاح للحجم المغمور الجديد،
عزمًا مُعيدًا --- وهذا يقع حين يعلو \emph{\omterm{rem:b1:fluid-statics:stability}{مركز الاستقرار}} (وهو
النقطة التي يلقى فيها حامل الدافعة محور الجسم) النقطةَ $G$.
والصابورة في أسفل البدن تخفض $G$ وتثبّت السفينة؛ وينقلب
القارب الذي ثقله عالٍ.
\end{remark}

ارتفاع عمود زئبق يوازن \qty{1.013e5}{Pa}؛ ثم عمود ماء.
ولماذا لا ترفع مضخة شفط الماء أكثر من \qty{10}{m} تقريبًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fluid-statics:3}
رافعة هيدروليكية: قوة \qty{100}{N} على مكبس مساحته
\qty{2.0}{cm^2}، والحمولة على مكبس مساحته \qty{200}{cm^2}؛ ثم مسافة
كل منهما إذا قطع المكبس الصغير \qty{20}{cm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:fluid-statics:4}
أنبوب على شكل U فيه ماء؛ يُصبّ زيت ($\rho = \qty{850}{kg/m^3}$) في
أحد ذراعيه فيؤلف عمودًا ارتفاعه \qty{12}{cm}. ما فرق الارتفاع
بين السطحين الحرّين؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fluid-statics:5}
\omterm{prop:b1:fluid-statics:atmosphere}{الارتفاع المميّز} للجو عند \qty{273}{K}؛ والضغط عند \qty{5000}{m}
ثم عند \qty{8848}{m}؛ والارتفاع الذي عنده $P = P_0/2$. وقارن مع
القيم المذكورة في \cref{ch:b1:kinetic-theory}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fluid-statics:6}
سدّ مستطيل عرضه \qty{100}{m} يحجز \qty{20}{m} من الماء. أوجد القوة
الكلية، وعمق \omterm{prop:b1:fluid-statics:wall}{مركز الضغط}، وعزم الماء بالنسبة إلى قاعدة
السدّ. ولماذا يُقوَّس السدّ نحو الماء؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fluid-statics:7}
ما نسبة جبل جليدي تحت السطح (الجليد \qty{917}{kg/m^3}، وماء
البحر \qty{1025}{kg/m^3})؟ وجذع كثافته \qty{600}{kg/m^3} في الماء
العذب؟ وكم شخصًا كتلة الواحد \qty{75}{kg} يحمل طوف من
\qty{2.0}{m^3} من ذلك الخشب قبل أن يغمر الماء سطحه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fluid-statics:8}
مقياس كثافة كتلته \qty{20}{g} ومقطع ساقه \qty{0.50}{cm^2} يطفو في
الماء وقد برز من ساقه \qty{4.0}{cm}. فكم يبرز منها في محلول ملحي
كثافته \qty{1100}{kg/m^3}؟ وما حساسيته، بالسنتيمتر لواحدة
الكثافة النسبية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:fluid-statics:9}
غواصة حجمها \qty{3000}{m^3} كتلتها \qty{2800}{t} وخزانات موازنتها
فارغة. ما النسبة البارزة منها عند السطح في ماء البحر؟ وما
كتلة الماء التي عليها أن تأخذها لتصير في طفو محايد؟ وماذا لو
دخلت بعد ذلك مصبًّا من الماء العذب؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fluid-statics:10}
منطاد طفل فيه \qty{10}{L} من الهيليوم ($M = \qty{4}{g/mol}$) عند
\qty{1}{bar} و \qty{293}{K}؛ ويزن المطاط \qty{3.0}{g}. أوجد الحمولة
الصافية وكتلة الخيط الذي يستطيع حمله.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fluid-statics:11}
برهن على أن \omterm{prop:b1:fluid-statics:wall}{مركز الضغط} على جدار شاقولي مستطيل يقع عند
$2h/3$ بحساب عزم قوى الضغط بالنسبة إلى مستقيم السطح. وأين
يقع من أجل جدار غُمر بعضه فقط، من العمق $h_1$ إلى العمق
$h_2$؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:fluid-statics:12}
مقياس كثافة أسطواني (كتلته $m$ ومقطعه $S$) يطفو في سائل
كثافته $\rho$ فيُدفع نحو الأسفل بمقدار $x$ ثم يُترك. بيّن أنه
يهتزّ اهتزازًا توافقيًّا وأعطِ الدور؛ ثم احسبه من أجل $m =
\qty{20}{g}$ و $S = \qty{0.50}{cm^2}$ في الماء. (وهو
\cref{ex:b1:oscillators-resonance:bob} من جديد.)
\end{exercise}

\section{مسألة: الغواصة}

\begin{problem}[{مسألة نهاية الأسبوع --- بدن فولاذي على عمق ثلاثمئة
متر: الضغط على صفائحه، والماء الذي عليه أن يبتلعه لينزل
ويقذفه لينهض، والغوّاص الذي يغادره، والناقلة التي تمرّ
فوقه}]
\label{pb:b1:fluid-statics:1}
ماء البحر: $\rho = \qty{1025}{kg/m^3}$؛ والماء العذب \qty{1000}{kg/m^3}؛
و $P_0 = \qty{1.013e5}{Pa}$؛ و $g = \qty{9.81}{m/s^2}$. ويحصر بدن الغواصة
الحجم $V = \qty{3000}{m^3}$؛ وكتلتها بخزانات موازنة فارغة
$m_0 = \qty{2800}{t}$.

\medskip
\noindent\textbf{الجزء I --- الضغط على البدن.}
\begin{enumerate}
  \item الضغط المطلق على عمق \qty{300}{m}، \omterm{def:b1:kinetic-theory:pressure}{بالباسكال} وبالبار.
  \item القوة على فتحة دائرية قطرها \qty{0.80}{m} على ذلك العمق؛
        وقارنها بثقل شاحنة.
  \item القوة على متر مربع واحد من البدن عند العارضة السفلى إذا
        كان ارتفاع البدن \qty{10}{m} وأعلاه على عمق \qty{300}{m}: بكم
        تختلف عن القوة في الأعلى؟
  \item الماء قابل للانضغاط قليلًا ($\chi_T = \qty{4.6e-10}{Pa^{-1}}$):
        بأيّ نسبة يكون ماء البحر أكثف على عمق \qty{300}{m} منه عند
        السطح؟ وهل لذلك أثر في الضغط المحسوب أعلاه؟
  \item يتنفس الطاقم في الداخل هواء عند \qty{1}{atm}: فلماذا يجب أن
        يكون البدن أسطوانة سميكة، ولماذا تقاوم الأسطوانة خيرًا
        من صندوق مسطّح؟
  \item قشرة أسطوانية رقيقة نصف قطرها $R$ وسماكة جدارها $e$ تحت
        فرط ضغط خارجي $\Delta P$ تحمل في جدارها إجهاد انضغاط
        مقداره $\Delta P\,R/e$. احسبه من أجل $R = \qty{5.0}{m}$ و
        $e = \qty{3.0}{cm}$ على عمق \qty{300}{m}، وقارنه بإجهاد خضوع
        الفولاذ، وهو نحو \qty{600}{MPa}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الغوص والطفو.}
\begin{enumerate}[resume]
  \item عند السطح وخزاناتها فارغة، ما حجم البدن البارز؟
  \item ما كتلة ماء البحر التي على خزانات الموازنة أن تأخذها كي
        تحوم الغواصة في طفو محايد؟
  \item وقد صارت في طفو محايد على عمق \qty{300}{m}، ينضغط البدن
        بنسبة $0.1\%$ تحت أثر الضغط. احسب القوة الصافية الناتجة
        وقل هل التوازن مستقر بالنسبة إلى العمق.
  \item ينتقل المركب في طفو محايد من البحر إلى مصبّ من الماء
        العذب. ما القوة الصافية؟ وإلى أين يمضي، وما الذي على
        الطاقم فعله؟
  \item يُطلَق طوربيد كتلته \qty{20}{t}. ما الذي على جملة الموازنة
        فعله في الحال، وبأيّ مقدار؟
  \item لماذا تستعمل الغواصة هواءً مضغوطًا لتفريغ خزاناتها،
        ولماذا يكون ضغط ذلك الهواء هو الحدّ الذي يقيّد العمق الذي
        تستطيع تفريغها عنده؟
  \item قدّر حجم الهواء عند \qty{1}{atm} اللازم لتفريغ
        \qty{275}{m^3} من الخزانات على عمق \qty{300}{m} (قانون بويل،
        مع $PV$ ثابت).
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الغوّاصة.}
تغادر غوّاصة السطح وفي رئتيها \qty{6.0}{L} من الهواء.
\begin{enumerate}[resume]
  \item الضغط المطلق على عمق \qty{10}{m} ثم \qty{30}{m}.
  \item إذا نزلت حابسةً نفَسها، فما حجم رئتيها على عمق
        \qty{30}{m}؟
  \item غوّاصة بأسطوانة تتنفس هواءً عند الضغط المحيط على عمق
        \qty{30}{m} ثم تصعد حابسةً نفَسها: ما الحجم الذي كانت
        رئتاها ستحتاجان إليه عند السطح، وما العبرة؟
  \item أنبوب تنفس طوله \qty{1.0}{m}: ما فرق الضغط بين الماء على
        صدر الغوّاصة والهواء في رئتيها؟ وما القوة على صدر
        مساحته \qty{0.05}{m^2}؟ وهل تستطيع الشهيق؟
  \item تحوي أسطوانة الهواء \qty{12}{L} عند \qty{200}{bar}: فكم لترًا
        من الهواء تعطي على عمق \qty{30}{m}، ولكم من الوقت باستهلاك
        \qty{20}{L} في الدقيقة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الناقلة فوقه.}
\begin{enumerate}[resume]
  \item ناقلة حمولتها \qty{100000}{t} تطفو في ماء البحر: ما حجم
        الماء المُزاح؟
  \item مساحة خط الماء عندها \qty{8000}{m^2}: بكم ترتفع أو تغطس
        حين تدخل الماء العذب؟
  \item وقد حُمّلت \qty{10000}{t} من النفط: ما تغيّر الغاطس؟
  \item للناقلة الفارغة مركز كتلة عالٍ؛ ولذلك تحمل ماء البحر
        صابورةً حين تكون فارغة. لماذا، بدلالة \omterm{rem:b1:fluid-statics:stability}{مركز الاستقرار}؟
  \item بدن الناقلة عند عارضتها السفلى (بغاطس \qty{20}{m}): ما
        الضغط والقوة على المتر المربع، مقارنةً بالغواصة على عمق
        \qty{300}{m}؟
  \item تمرّ الناقلة فوق الغواصة تمامًا. فهل تشعر الغواصة
        بحمولتها البالغة \qty{100000}{t}؟ علّل بمجال الضغط.
  \item لخّص: ما العلاقة الوحيدة التي عيّنت كل عدد في هذه
        المسألة، وما المبرهنة التي تلتها؟
\end{enumerate}
\end{problem}
