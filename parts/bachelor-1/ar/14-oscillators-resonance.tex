\chapter{الهزّازات الميكانيكية: التخميد والرنين}\label{ch:b1:oscillators-resonance}

ادفع أرجوحة طفل في اللحظة الخطأ فلا يحدث شيء يُذكر؛ وادفعها
مرة في كل دور، برفق، فيصرخ الطفل بعد عشرات
التأرجحات من أعلى القوس. وترتجف غسّالة ارتجافًا عنيفًا
عند سرعة بعينها من دورة العصر، ثم تهدأ فوقها. وتغنّي
كأس نبيذ حين يدور إصبع مبلول على حافتها بالمعدّل
الصحيح تمامًا، وتتحطم حين يمسك مغنٍّ النغمة نفسها بصوت عالٍ كفايةً.
وكل واحدة من هذه \emph{رنين}: أي جملة لها \omterm{def:b1:oscillators-resonance:harmonic}{تواتر
خاص}، تُقاد قرب ذلك التواتر، فتراكم الطاقة دورةً بعد
دورة حتى ينزفها الاحتكاك بالسرعة التي تدخل بها. ويأخذ هذا
الفصل الهزّاز الميكانيكي --- حرًّا ثم مخمَّدًا ثم مقادًا ---
ويشتقّ منحنيات الرنين التي ظهرت توائمها الكهربائية
في \cref{ch:b1:sinusoidal-impedance}.

\section{الهزّاز التوافقي}

\begin{definition}[الهزّاز التوافقي]\label{def:b1:oscillators-resonance:harmonic}
\index{هزّاز توافقي}\index{تواتر خاص}
جملة يحقق فيها الانتقال $x$ عن توازن مستقر
\[
\ddot x + \omega_0^2\,x = 0
\]
هي \emph{هزّاز توافقي} \emph{تواتره الزاوي الخاص}
$\omega_0$ (ودوره $T_0 = 2\pi/\omega_0$). وللحلّ $x = A\cos(\omega_0t
+ \varphi)$ سعة وطور تحدّدهما الشروط الابتدائية،
ودور مستقل عن السعة (وهو \emph{تساوي الأدوار}).
\end{definition}

\begin{proposition}[من أين تأتي الهزّازات التوافقية]\label{prop:b1:oscillators-resonance:sources}
\begin{itemize}
  \item كتلة $m$ على نابض $k$: $\omega_0^2 = k/m$ (بقانون هوك).
  \item بندول بسيط طوله $\ell$، عند الزوايا الصغيرة: $\omega_0^2 = g/\ell$.
  \item أيّ جسيم بجوار توازن مستقر \omterm{def:b1:work-and-energy:conservative}{لطاقة كامنة} $E_p$:
        $\omega_0^2 = E_p''(x_0)/m$ (\cref{prop:b1:work-and-energy:stability}).
  \item دارة LC: $\omega_0^2 = 1/LC$، مع $q \leftrightarrow x$
        و $L \leftrightarrow m$ و $1/C \leftrightarrow k$.
\end{itemize}
والطاقة $E = \tfrac12 m\dot x^2 + \tfrac12 m\omega_0^2x^2 = \tfrac12
m\omega_0^2A^2$ ثابتة، تتقاذف بين الصورة الحركية والصورة الكامنة
مرتين في كل دور.
\end{proposition}

\begin{proof}
قانون نيوتن الثاني مع $F = -kx$؛ ومعادلة البندول
$\ddot\theta + (g/\ell)\sin\theta = 0$ مع $\sin\theta \approx \theta$؛
وصيغة تايلور عند النهاية الصغرى للطاقة $E_p$؛ وقانون العروة في دارة
LC. وللطاقة: اشتقّ واستعمل المعادلة.
\end{proof}

\begin{example}[أسطوانة تتأرجح طفوًا]\label{ex:b1:oscillators-resonance:bob}
تطفو أسطوانة شاقولية مقطعها $S$ وقد غُمر منها عمق $h$
(إذ $mg = \rho_wgSh$، ودافعة أرخميدس من \cref{ch:b1:fluid-statics}).
فإذا دُفعت إلى أسفل بالمقدار $x$ تلقّت قوة إضافية نحو الأعلى $\rho_wgSx$:
أي $m\ddot x = -\rho_wgSx$، ومنه $\omega_0^2 = \rho_wgS/m = g/h$ --- وهو نفسه
ما لبندول طوله $h$. وعوّامة مغمورة بعمق \qty{1}{m} تتأرجح بدور
\qty{2}{s}، أيًّا كانت كتلتها.
\end{example}

\section{الهزّاز المخمَّد}

\begin{proposition}[الهزّاز المخمَّد]\label{prop:b1:oscillators-resonance:damped}
\index{هزّاز مخمَّد}\index{عامل الجودة!ميكانيكي}
مع احتكاك لزج $-\alpha\dot x$ تصير المعادلة
\[
\ddot x + \frac{\omega_0}{Q}\dot x + \omega_0^2x = 0, \qquad
Q = \frac{\sqrt{km}}{\alpha} = \frac{m\omega_0}{\alpha},
\]
وهي الصيغة القانونية من الرتبة الثانية في \cref{ch:b1:transient-regimes}: فمن أجل
$Q > \tfrac12$ يكون التخميد شبه دوري $x = A\eu^{-t/\tau}\cos(\omega t +
\varphi)$ مع $\tau = 2Q/\omega_0$ و $\omega = \omega_0\sqrt{1 - 1/4Q^2}$؛ وتتناقص
الطاقة وفق $\eu^{-2t/\tau}$، ولدينا
\[
Q = 2\pi\,\frac{\text{الطاقة المخزونة}}{\text{الطاقة المتبددة في كل دور}}
\quad (Q \gg 1) .
\]
\end{proposition}

\begin{proof}
اقسم $m\ddot x + \alpha\dot x + kx = 0$ على $m$. وللطاقة: تفقد $E
\propto A^2\eu^{-2t/\tau}$ الجزءَ $2T/\tau = 2\pi/Q$ في كل
دور حين يكون ذلك صغيرًا.
\end{proof}

\begin{example}[كم يدوم الرنّ]\label{ex:b1:oscillators-resonance:ringing}
تظلّ شوكة رنانة عند \qty{440}{Hz} تُسمع بعد عشر ثوانٍ من
قرعها: إذ هبطت سعتها إلى نحو $5\%$ في $3\tau = \qty{10}{s}$، ومنه
$\tau = \qty{3.3}{s}$ و $Q = \omega_0\tau/2 \approx 4600$. أما تعليق
سيارة ($Q \approx 0.7$) فيتوقف خلال دور واحد؛ وأرجوحة طفل
($Q \approx 20$) تفقد ثلث طاقتها في كل تأرجحة وتحتاج إلى دفعة
في كل مرة.
\end{example}

\section{الاهتزازات القسرية والرنين}

\begin{theorem}[هزّاز مقاد جيبيًّا]\label{thm:b1:oscillators-resonance:forced}
\index{اهتزازات قسرية}\index{رنين!السعة}\index{رنين!السرعة}
إذا قادته قوة $F_0\cos\omega t$ فإن \omterm{prop:b1:oscillators-resonance:damped}{الهزّاز المخمَّد}
\[
m\ddot x + \alpha\dot x + kx = F_0\cos\omega t
\]
يستقر، بعد \omterm{def:b1:transient-regimes:firstorder}{نظام عابر} مدته $\sim\tau$، على \omterm{prop:b1:sinusoidal-impedance:forced}{النظام
القسري} $x = X\cos(\omega t + \varphi)$ حيث، بكتابة $x_r = \omega/\omega_0$
و $X_0 = F_0/k$ الانحرافَ السكوني،
\[
X = \frac{X_0}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}}, \qquad
\tan\varphi = -\frac{x_r/Q}{1 - x_r^2}, \quad \varphi \in [-\pi, 0] .
\]
تبلغ \emph{السعة} $X$ ذروتها، من أجل $Q > 1/\sqrt2$، عند $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$ حيث $X_{\max} \approx QX_0$؛ أما \emph{سعة
السرعة} $V = \omega X$ فتبلغ ذروتها تمامًا عند $\omega_0$، حيث $V_{\max} =
F_0/\alpha$، ومنحني رنينها
\[
\frac{V}{V_{\max}} = \frac{1}{\sqrt{1 + Q^2(x_r - 1/x_r)^2}}
\]
عرض نطاقه $\Delta\omega = \omega_0/Q$. وعند $\omega_0$ يتأخر الانتقال
عن القوة بمقدار $\pi/2$، وتكون السرعة على الطور معها.
\end{theorem}

\begin{proof}
بالطريقة العقدية (\cref{ch:b1:sinusoidal-impedance}): $\underline X(-\omega^2
+ j\omega\omega_0/Q + \omega_0^2) = F_0/m$، ومنه $\underline X = (F_0/m)/(\omega_0^2
- \omega^2 + j\omega\omega_0/Q)$؛ ثم اقسم البسط والمقام على
$\omega_0^2 = k/m$. وينتج المقياس والعمدة؛ وقد حُدّدت النهاية العظمى للسعة $X$
في \cref{prop:b1:sinusoidal-impedance:overvoltage} (فالدالة
نفسها). وللسرعة: $\underline V = j\omega\underline X$ يعطي $V = \omega X$؛
وبكتابة $(1 - x_r^2)/x_r = 1/x_r - x_r$ يصير المقام
$\sqrt{1/Q^2 + (1/x_r - x_r)^2}$، أي صيغة RLC. وعند $x_r = 1$ يكون
$\underline X = -jQX_0$: أي تأخر بمقدار $\pi/2$.
\end{proof}

\begin{remark}[رنينان في دارة واحدة]\label{rem:b1:oscillators-resonance:analogy}
\index{تناظر كهرميكانيكي}
إن التقابل $x \leftrightarrow q$ و $\dot x \leftrightarrow i$
و $m \leftrightarrow L$ و $\alpha \leftrightarrow R$ و $k \leftrightarrow 1/C$
و $F \leftrightarrow e$ يحمل الهزّاز المقاد على \omterm{prop:b1:transient-regimes:rlc}{دارة RLC} على التسلسل في
\cref{ch:b1:sinusoidal-impedance}: فرنين السرعة هو رنين
التيار (وهو دائمًا عند $\omega_0$ وعرض نطاقه $\omega_0/Q$)، ورنين السعة
هو رنين الشحنة، أي رنين توتر المكثفة (وهو دون $\omega_0$
قليلًا، وغائب من أجل $Q$ الصغير). أما الاستطاعة الممتصة $\langle F\dot x\rangle
= \tfrac12 F_0V\cos\varphi_v$ فتبلغ ذروتها عند $\omega_0$ بالقيمة $F_0^2/2\alpha$،
ونقطتا نصف الاستطاعة هما عرض نطاق السرعة --- تمامًا كما في
\omterm{prop:b1:transient-regimes:rlc}{دارة RLC}.
\end{remark}

\begin{example}[سرعات الرنين]\label{ex:b1:oscillators-resonance:road}
لتعليق سيارة $f_0 \approx \qty{1.3}{Hz}$ (\cref{pb:b1:transient-regimes:1}).
وعلى طريق فيها تموّجات كل $\lambda = \qty{10}{m}$ يكون تواتر
الإثارة $v/\lambda$: فالرنين عند $v = \lambda f_0 = \qty{13}{m/s}$،
أي نحو \qty{47}{km/h} --- وهي السرعة التي تهزّ عندها الطريق المضلّعة
السيارةَ أشدّ ما تهزّ، وهي سبب تركيب مخمّد $Q \approx 0.7$:
فمع $Q = 5$ كان الهيكل سيتأرجح بخمسة أضعاف
سعة الطريق. وفوق الرنين ($x_r \gg 1$) تتناقص السعة وفق
$X_0/x_r^2$: فيبدو الهيكل ``طافيًا'' فوق المطبّات السريعة --- أي العزل،
وهو الوجه الآخر للرنين.
\end{example}

\section{إثارة القاعدة: العزل ومقياس الزلازل}

\begin{proposition}[هزّاز تهزّه دعامته]\label{prop:b1:oscillators-resonance:base}
\index{إثارة القاعدة}\index{عزل الاهتزازات}\index{مقياس الزلازل}
لتتحرك دعامة جملة كتلة--نابض--مخمّد بالمقدار $x_g(t)$ (كالطريق
تحت سيارة، أو الأرض تحت مقياس زلازل)، وليكن $u$ انتقال
الكتلة بالنسبة إلى الدعامة، و $x = x_g + u$ انتقالها
المطلق (عن التوازن). عندئذ
\[
m\ddot u + \alpha\dot u + ku = -m\ddot x_g ,
\]
ومن أجل $x_g = X_g\cos\omega t$، مع $x_r = \omega/\omega_0$:
\[
\frac{U}{X_g} = \frac{x_r^2}{\sqrt{(1 - x_r^2)^2 + x_r^2/Q^2}},
\qquad
\frac{X}{X_g} = \sqrt{\frac{1 + x_r^2/Q^2}{(1 - x_r^2)^2 + x_r^2/Q^2}} .
\]
من أجل $x_r \gg 1$: $U \to X_g$ (فالكتلة تبقى ساكنة في الفضاء --- و\emph{\omterm{prop:b1:oscillators-resonance:base}{مقياس
الزلازل}} يقرأ انتقال الأرض) و $X \to 0$ (فالكتلة
\emph{معزولة} عن الهزّ)؛ ومن أجل $x_r \ll 1$: $U \approx
\ddot x_g/\omega_0^2$ (أي \emph{مقياس تسارع}) و $X \approx X_g$ (فالكتلة
تتبع الدعامة).
\end{proposition}

\begin{proof}
يؤثر النابض والمخمّد في الحركة النسبية: $m\ddot x = -ku -
\alpha\dot u$، ولدينا $\ddot x = \ddot x_g + \ddot u$. وبالصيغة العقدية
$\underline U(\omega_0^2 - \omega^2 + j\omega\omega_0/Q) = \omega^2X_g$، ومنه
$U/X_g$؛ ثم $\underline X = \underline U + X_g = X_g(1 + jx_r/Q)/(1 -
x_r^2 + jx_r/Q)$.
\end{proof}

\begin{example}[لماذا توضع الآلات على قواعد ليّنة]\label{ex:b1:oscillators-resonance:mounts}
محرّك عند \qty{50}{Hz} على قواعد مطاطية موالفة على $f_0 = \qty{10}{Hz}$
(أي $x_r = 5$ و $Q \approx 2$): $X/X_g = \sqrt{(1 + 6.25)/(576 + 6.25)} = 0.11$
--- أي إن تسعة أعشار الاهتزاز تبقى خارج الأرضية. ويجب أن تكون القواعد
\emph{ألين} مما يوحي به الحدس: فتقسيتها ترفع $f_0$
نحو $50$ هرتز وتقضي على العزل. والصيغة نفسها، مقروءةً
بالاتجاه الآخر، هي سيارة \cref{ex:b1:oscillators-resonance:road} على
طريق مضلّعة.
\end{example}

\section{الطاقة: كيف يُبنى الرنين}

\begin{proposition}[موازنة الطاقة في النظام القسري]\label{prop:b1:oscillators-resonance:energy}
في \omterm{prop:b1:sinusoidal-impedance:forced}{النظام القسري} الدائم تساوي \omterm{thm:b1:sinusoidal-impedance:power}{الاستطاعة المتوسطة} التي تقدّمها قوة
الإقادة الاستطاعةَ المتوسطة التي يبدّدها الاحتكاك، $\tfrac12\alpha V^2$.
وعند الرنين تكون الطاقة المخزونة $\tfrac12 mV_{\max}^2$ مساويةً $Q/2\pi$ مرة
الطاقةَ المقدَّمة في كل دور: فالهزّاز يراكم شغل الإقادة
على نحو $Q$ دورًا قبل أن تساوي الضياعاتُ الوارد، وهو أيضًا
عدد الدورات ($\sim\tau/T$) التي يدوم فيها \omterm{def:b1:transient-regimes:firstorder}{النظام العابر}.
\end{proposition}

\begin{proof}
توسّط مبرهنة الطاقة على دور: $\langle\dd E/\dd t\rangle = 0$
في \omterm{def:b1:transient-regimes:firstorder}{النظام الدائم}، ومنه $\langle F\dot x\rangle = \langle\alpha\dot x^2\rangle =
\tfrac12\alpha V^2$. وعند الرنين $V_{\max} = F_0/\alpha$، والطاقة المخزونة
$\tfrac12 m(F_0/\alpha)^2$، والمقدَّمة في كل دور $\tfrac12\alpha(F_0/\alpha)^2T$؛
والنسبة $m/(\alpha T) = Q/2\pi$.
\end{proof}

\begin{example}[الأرجوحة]\label{ex:b1:oscillators-resonance:swing}
أرجوحة طولها \qty{2.5}{m} (أي $T_0 = \qty{3.2}{s}$ و $Q \approx 20$) تتلقّى
دفعة مقدارها \qty{50}{N} على \qty{0.5}{m} في كل دور: أي \qty{25}{J} في
كل دورة. فتنمو حتى يساوي الضياعُ في كل دور، $2\pi E/Q$،
\qty{25}{J}: ومنه $E = \qty{80}{J}$؛ ومع طفل كتلته \qty{30}{kg} يعطي $mg\ell(1 -
\cos\theta_{\max}) = \qty{80}{J}$ القيمةَ $\theta_{\max} \approx 27^\circ$.
ولو دُفعت بأيّ إيقاع آخر لتناوب شغل الدفعات في
الإشارة وتوسّط إلى الصفر.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:1}
كتلة \qty{0.20}{kg} على نابض صلابته \qty{80}{N/m} تهتزّ
بسعة \qty{5.0}{cm}. احسب \omterm{def:b1:wave-propagation:signal}{التواتر الزاوي} والتواتر والدور
والسرعة العظمى والطاقة الكلية.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:2}
احسب دور بندول طوله \qty{1.0}{m} على الأرض وعلى القمر
($g = \qty{1.62}{m/s^2}$). وماذا يحدث لدور جملة الكتلة--النابض على
القمر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:3}
لهزّاز $f_0 = \qty{2.0}{Hz}$ و $Q = 50$. احسب زمن تخميد السعة $\tau$؛
وزمن انتصاف الطاقة؛ وعدد الاهتزازات
قبل أن تهبط السعة إلى $5\%$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:oscillators-resonance:4}
كتلة على نابض ($k = \qty{100}{N/m}$ و $Q = 10$) تُقاد بقوة
سعتها \qty{1.0}{N}. احسب الانحراف السكوني؛ والسعة عند الرنين؛
والسعة عند $\omega = 3\omega_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:5}
سيارة ($f_0 = \qty{1.3}{Hz}$ و $Q = 0.7$) تعبر مطبّات تبعد \qty{8.0}{m}
بعضها عن بعض. فما سرعة الرنين؟ وعند \qty{60}{km/h}، أيّ جزء من سعة
المطبّ يشعر به الهيكل؟ علّق على ذلك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:6}
فسّر بموازنة الطاقة لماذا يجب دفع الأرجوحة عند دورها
الخاص؛ ومن أجل $Q = 20$ ودفعة تضيف \qty{25}{J} في كل دورة، أوجد
طاقة \omterm{def:b1:transient-regimes:firstorder}{النظام الدائم}، والسعة من أجل طفل كتلته \qty{30}{kg} على
أرجوحة طولها \qty{2.5}{m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:7}
تتناقص القيم العظمى المتتالية لاهتزاز حرّ بنسبة $10\%$ في كل دور.
أوجد $Q$ وعرض نطاق رنين سرعته (بالهرتز) إذا كان
$f_0 = \qty{2.0}{Hz}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:8}
أعطِ طور الانتقال بالنسبة إلى القوة دون الرنين بكثير،
وعند الرنين، وفوقه بكثير، ثم طور السرعة
عند الرنين. ولماذا يكون رنين السرعة هو المهمّ من أجل
الاستطاعة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:oscillators-resonance:9}
بيّن أن سعة السرعة $V = \omega X$ عظمى تمامًا عند
$\omega_0$، حيث $V_{\max} = F_0/\alpha$، وأن $V/V_{\max}$ يأخذ
الصيغة $1/\sqrt{1 + Q^2(x_r - 1/x_r)^2}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:oscillators-resonance:10}
احسب \omterm{thm:b1:sinusoidal-impedance:power}{الاستطاعة المتوسطة} الممتصة عند الرنين من أجل $F_0 = \qty{1.0}{N}$ و
$\alpha = \qty{0.050}{N.s/m}$، والطاقة المخزونة إذا كان $m = \qty{0.20}{kg}$؛
وتحقق من أن نسبتهما $Q/\omega_0$ مع $\omega_0 = \qty{20}{rad/s}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:oscillators-resonance:11}
تُسمع شوكة رنانة عند \qty{440}{Hz} مدة \qty{10}{s} (حتى تهبط السعة
إلى $5\%$). قدّر $\tau$ و $Q$ وعرض نطاق رنينها، وعلّق
على سبب كونها مرجعًا جيدًا للتواتر.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:oscillators-resonance:12}
اشتقّ تواتر تأرجح أسطوانة طافية
(\cref{ex:b1:oscillators-resonance:bob}) انطلاقًا من قانون نيوتن وقوة
أرخميدس $\rho_wgS(h + x)$؛ وبيّن أن $\omega_0^2 = g/h$؛ واحسب
الدور من أجل $h = \qty{10}{cm}$. ولماذا تكون النتيجة مستقلة عن
الكتلة وعن السائل؟
\end{exercise}

\section{مسألة: مرسمة الزلازل}

\begin{problem}[{مسألة نهاية الأسبوع --- كتلة معلّقة بنابض داخل
صندوق مثبّت في الأرض، والأرض تهتزّ: فماذا يسجّل القلم،
ولماذا تقيس الآلة نفسها الانتقال عند التواترات العالية
والتسارع عند التواترات المنخفضة، ولماذا لا يمكن أن يكون مقياس
الزلازل الطويل الدور مجرّد نابض طويل}]
\label{pb:b1:oscillators-resonance:1}
كتلة $m = \qty{1.0}{kg}$ معلّقة بنابض صلابته $k$ داخل
هيكل صلب مثبّت في الأرض؛ ويؤثر مخمّد بالقوة $-\alpha\dot u$، حيث
$u$ انتقال الكتلة بالنسبة إلى الهيكل، مقيسًا
عن موضع توازنها. وتتحرك الأرض، ومعها الهيكل،
شاقوليًّا بالمقدار $x_g(t)$. ومرجع النجوم البعيدة عطالي.

\medskip
\noindent\textbf{الجزء I --- المعادلة.}
\begin{enumerate}
  \item اكتب الموضع المطلق للكتلة على الصورة $x = x_g + u +
        \text{ثابت}$، وعدّد القوى المؤثرة في الكتلة (في \omterm{thm:b1:newton-dynamics:laws}{المرجع
        العطالي})، وبيّن أن
        \[
        m\ddot u + \alpha\dot u + ku = -m\ddot x_g .
        \]
  \item ضعها في الصيغة القانونية؛ وحدّد $\omega_0$ و $Q$.
  \item بُنيت الآلة على $f_0 = \qty{0.50}{Hz}$ و $Q = 0.70$:
        احسب $k$ و $\alpha$.
  \item تسارع أرضي ثابت $a_g$ (كأن يميل الهيكل) يعطي
        أيّ قيمة دائمة $u$؟ فسّر ذلك.
  \item ماذا يسجّل القلم المثبّت في الكتلة، وهو يكتب على أسطوانة
        مثبّتة في الهيكل؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- حركة أرضية توافقية.}
تتحرك الأرض وفق $x_g = X_g\cos\omega t$؛ ولنكتب $x_r = \omega/\omega_0$.
\begin{enumerate}[resume]
  \item باستعمال السعات العقدية، بيّن أن $\underline U = X_g\,x_r^2/
        (1 - x_r^2 + jx_r/Q)$.
  \item استنتج $U/X_g$ بدلالة $x_r$ و $Q$.
  \item في الحالة الحدية $x_r \gg 1$: بيّن أن $U \to X_g$ وفسّر فيزيائيًّا ما
        تفعله الكتلة (فماذا تقيس الآلة؟).
  \item في الحالة الحدية $x_r \ll 1$: بيّن أن $U \approx x_r^2X_g = a_g/\omega_0^2$ حيث
        $a_g$ سعة تسارع الأرض: فما الآلة
        الآن؟
  \item ما القيمة عند $x_r = 1$؟ ولماذا يكون $Q \approx 0.7$ اختيارًا معقولًا؟
  \item ارسم تخطيطيًّا $U/X_g$ بدلالة $x_r$ (بسلّمين لوغاريتميين) من أجل $Q = 0.7$ و
        $Q = 5$.
  \item ما طور $u$ بالنسبة إلى $x_g$ عند التواترات العالية؟ فسّر ذلك.
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- التصميم من أجل الزلازل.}
تُنتج الزلازل البعيدة موجات سطحية من \qty{0.05}{} إلى \qty{0.1}{Hz}
بسعات مليمترية؛ وتبلغ الهزّات المحلية من \qty{5}{} إلى \qty{10}{Hz}.
\begin{enumerate}[resume]
  \item لتسجيل موجات \qty{0.05}{Hz} انتقالاتٍ نريد
        $f_0 \ll \qty{0.05}{Hz}$، فلتكن \qty{0.02}{Hz}. فما الاستطالة
        السكونية للنابض تحت وزن الكتلة؟ علّق على ذلك.
  \item بندول بسيط له التواتر نفسه: فما طوله؟
  \item تستعمل الآلات الطويلة الدور الحقيقية بندولًا شبه أفقي
        على هيئة ``بوابة حديقة''، أو تغذيةً راجعة إلكترونية بالقوة. فسّر
        في جملة واحدة ما يحققه كل منهما.
  \item من أجل آلة \qty{0.50}{Hz}، احسب الاستطالة السكونية:
        أهي ممكنة؟
  \item بهذه الآلة، تعطي موجة \qty{0.10}{Hz} سعتها \qty{1.0}{mm}
        أيّ $U$؟ وفي أيّ نظام تكون الآلة؟
  \item هزّة محلية عند \qty{5.0}{Hz} سعتها \qty{1.0}{mm}: فما $U$؟
        وما النظام؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- مقياس التسارع في الهاتف.}
لكتلة مصنوعة مجهريًّا على نوابض من السيليسيوم $f_0 = \qty{1.0}{kHz}$ و
$Q = 0.7$.
\begin{enumerate}[resume]
  \item هزّ عند \qty{10}{Hz} سعة تسارعه $g$: احسب
        $U$. وكيف يُقرأ انتقال كهذا؟
  \item إلى أيّ تواتر تبقى القراءة ضمن $10\%$ من $a_g/\omega_0^2$؟
  \item أيّ ثابت زمن $\tau = 2Q/\omega_0$ يحكم استجابتها
        لهزّة، ولماذا يهمّ ذلك في كشف السقوط؟
  \item \omterm{def:b1:transient-regimes:firstorder}{ثابت الزمن} $\tau$ \omterm{prop:b1:oscillators-resonance:base}{لمقياس الزلازل}: قيمته ونتيجته بعد
        قفزة أرضية مفاجئة.
  \item لماذا يُختار التخميد الحرج (أو القريب من الحرج) للآلتين
        كلتيهما، بدل $Q$ كبير؟
  \item يُهزّ الهاتف عند \qty{1.0}{kHz} تمامًا: فبأيّ عامل تخطئ
        القراءة؟
  \item لخّص: جهاز واحد، ونظامان، تقرّرهما مقارنة واحدة
        بعينها.
\end{enumerate}
\end{problem}
