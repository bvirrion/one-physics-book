\chapter{Perubahan Fase Zat Murni}\label{ch:b1:phase-changes}

Air adalah satu-satunya zat yang pernah dilihat semua orang dalam ketiga
wujudnya, dan ia adalah fluida kerja sebagian besar pembangkit listrik dunia,
\omterm{def:b1:heat-engines:vapourcycle}{refrigeran} mesin es yang pertama, dan mesin penggerak cuaca.
Es mencair pada suhu yang tak bergeser sementara kalornya terus mengalir
masuk; sebuah cerek mendidih pada \qty{100}{\degreeCelsius} di tepi laut dan pada
\qty{70}{\degreeCelsius} di Everest; sebotol air yang sangat dingin dan tersegel
membeku menjadi padat begitu diketuk. Bab ini menggambar peta \omterm{def:b1:phase-changes:phase}{fase} sebuah
\omterm{def:b1:phase-changes:phase}{zat murni} --- diagram $(T, P)$ dan $(v, P)$ --- lalu memberikan
tiga perkakas yang menyertainya: aturan tuas (berapa banyak tiap \omterm{def:b1:phase-changes:phase}{fasenya}),
\omterm{def:b1:phase-changes:phase}{kalor laten} (berapa harga sebuah perubahan, dalam energi dan dalam entropi) dan
\omterm{thm:b1:phase-changes:clapeyron}{hubungan Clausius--Clapeyron} (bagaimana kemiringan garis batasnya). Ia berakhir
di tempat petanya gagal: \omterm{def:b1:phase-changes:metastable}{keadaan metastabil}, yang dari sanalah cuaca
dan kejutan panci prestonya datang.

\section{Fase dan diagram $(T, P)$}

\begin{definition}[Zat murni, fase, perubahan fase]\label{def:b1:phase-changes:phase}
\index{fase}\index{perubahan fase}\index{zat murni}\index{kalor laten}
Yang disebut \emph{zat murni} adalah benda dari satu jenis kimia saja. Sebuah
\emph{fase} adalah bagiannya yang homogen dalam sifat fisisnya:
padat, cair, gas (\emph{uap} bila ia berdampingan dengan
cairannya). Sebuah \emph{perubahan fase} (peleburan/pembekuan, penguapan/
pengembunan, sublimasi/deposisi) adalah peralihan dari satu fase ke
fase yang lain; pada tekanan tetap ia terjadi pada suhu yang tetap, dengan kalor
yang dipertukarkan tetapi tanpa perubahan suhu: itulah \emph{kalor laten}.
\end{definition}

\begin{proposition}[Diagram kesetimbangan]\label{prop:b1:phase-changes:PTdiagram}
\index{diagram fase}\index{titik tripel}\index{titik kritis}\index{tekanan uap jenuh}
Pada bidang $(T, P)$ tiap \omterm{def:b1:phase-changes:phase}{fasenya} menempati sebuah daerah; dua \omterm{def:b1:phase-changes:phase}{fase} berdampingan
hanya di sepanjang sebuah kurva (sublimasi, peleburan, penguapan), dan ketiganya hanya
di satu titik, yaitu \emph{\omterm{prop:b1:phase-changes:PTdiagram}{titik tripel}}. Kurva penguapannya
$P = P_s(T)$ --- yaitu \emph{\omterm{prop:b1:phase-changes:PTdiagram}{tekanan uap jenuh}} --- berakhir di
\emph{\omterm{prop:b1:phase-changes:PTdiagram}{titik kritis}} $(T_c, P_c)$, yang di luarnya cairan dan gas tak lagi
terbedakan. Untuk air: \omterm{prop:b1:phase-changes:PTdiagram}{titik tripelnya} \qty{273.16}{K}, \qty{611}{Pa};
titik kritisnya \qty{647}{K}, \qty{221}{bar}. Kurva peleburan air
condong ke kiri (es kurang padat daripada air); bagi hampir setiap
zat lain ia condong ke kanan.
\end{proposition}
\admitted

\begin{remark}[Pendidihan, penguapan, dan panci presto]\label{rem:b1:phase-changes:boiling}
Sebuah cairan \emph{menguap} pada suhu berapa pun dari permukaannya, selama
tekanan parsial uap di atasnya di bawah $P_s(T)$; ia
\emph{mendidih} ketika $P_s(T)$ mencapai tekanan sekitarnya, sehingga gelembung
uap dapat tumbuh di dalamnya. Air mendidih pada \qty{100}{\degreeCelsius}
karena $P_s(\qty{100}{\degreeCelsius}) = \qty{1.013}{bar}$; pada
\qty{0.33}{bar} di Everest ia mendidih dekat \qty{70}{\degreeCelsius}, di dalam
panci presto pada \qty{1.8}{bar} dekat \qty{117}{\degreeCelsius}. Suhu
sebuah cairan yang mendidih terpasak oleh tekanannya --- kenyataan yang
dipakai untuk mendefinisikan skala Celsius selama dua abad.
\end{remark}

\section{Diagram $(v, P)$ dan aturan tuas}

\begin{proposition}[Isoterm Andrews]\label{prop:b1:phase-changes:andrews}
\index{isoterm Andrews}\index{kurva jenuh}
Pada bidang Clapeyron $(v, P)$ volume jenisnya, sebuah isoterm
$T < T_c$ mempunyai tiga bagian: sebuah cabang yang curam (cairan, hampir
tak termampatkan), sebuah \emph{plato} mendatar pada $P = P_s(T)$ tempat cairan
dan uapnya berdampingan, dan sebuah cabang mirip hiperbola (uap). Ujung
platonya adalah volume jenis $v_l(T)$ dan $v_v(T)$ cairan dan uap
jenuhnya; tempat kedudukannya adalah \emph{\omterm{prop:b1:phase-changes:andrews}{kurva jenuh}} (yaitu
kubahnya), yang puncaknya \omterm{prop:b1:phase-changes:PTdiagram}{titik kritis}, tempat isoterm $T_c$
mempunyai belokan mendatar. Di atas $T_c$ tak ada plato.
\end{proposition}
\admitted

\begin{theorem}[Aturan tuas]\label{thm:b1:phase-changes:lever}
\index{aturan tuas}\index{fraksi uap}
Massa $m$ bervolume jenis $v$ pada platonya pada suhu $T$ terbelah menjadi
fraksi massa uap $x = m_v/m$ (yaitu \emph{\omterm{thm:b1:phase-changes:lever}{fraksi uap}}) dan fraksi cairan
$1 - x$ dengan
\[
v = (1 - x)v_l + x\,v_v , \qquad\text{yakni}\qquad
x = \frac{v - v_l}{v_v - v_l} = \frac{LM}{LV} ,
\]
yaitu nisbah ruas garis yang terpotong pada platonya oleh titik keadaan $M$.
Aturan yang sama berlaku bagi setiap besaran ekstensif tiap satuan massa ($u$,
$h$, $s$): $h = (1 - x)h_l + xh_v$.
\end{theorem}

\begin{proof}
Volumenya berjumlah: $V = m_lv_l + m_vv_v$; bagilah dengan $m$. Energi, \omterm{def:b1:first-law:enthalpy}{entalpi} dan
entropi juga ekstensif.
\end{proof}

\begin{example}[Sebuah cerek dan sebuah ketel]\label{ex:b1:phase-changes:lever}
Air pada \qty{100}{\degreeCelsius}, \qty{1}{atm}: $v_l = \qty{1.04e-3}{m^3/kg}$,
$v_v = \qty{1.67}{m^3/kg}$ --- nisbahnya $1600$. Sebuah bejana tertutup \qty{1.0}{L}
yang memuat \qty{0.50}{kg} air pada suhu itu mempunyai $v =
\qty{2.0e-3}{m^3/kg}$ dan $x = (2.0 - 1.04) \times 10^{-3}/1.67 = 5.8 \times 10^{-4}$:
\qty{0.29}{g} uap yang menempati separuh bejananya. Melipatduakan massa
uapnya di dalam bejana tersegel nyaris tak menggeser titik keadaannya; tekanannya
ditetapkan oleh suhunya saja.
\end{example}

\section{Energetika sebuah perubahan fase}

\begin{proposition}[Kalor laten, entalpi dan entropi peralihan]\label{prop:b1:phase-changes:latent}
\index{entalpi peralihan}\index{entropi peralihan}
Pada suhu $T$ dan tekanan $P_s(T)$ tempat kedua \omterm{def:b1:phase-changes:phase}{fasenya} berdampingan,
peralihan satu satuan massa dari \omterm{def:b1:phase-changes:phase}{fase} 1 ke \omterm{def:b1:phase-changes:phase}{fase} 2 bersifat reversibel,
isotermal dan isobarik, dan
\[
L_{12}(T) = h_2 - h_1 , \qquad s_2 - s_1 = \frac{L_{12}}{T} , \qquad
u_2 - u_1 = L_{12} - P_s(v_2 - v_1) .
\]
$L_{12} > 0$ bila peralihannya menuju \omterm{def:b1:phase-changes:phase}{fase} yang kurang teratur (peleburan,
penguapan, sublimasi), dan $L_{21} = -L_{12}$. Untuk air pada \qty{1}{atm}:
$L_f = \qty{334}{kJ/kg}$ pada \qty{0}{\degreeCelsius}, $L_v = \qty{2257}{kJ/kg}$ pada
\qty{100}{\degreeCelsius}.
\end{proposition}

\begin{proof}
Pada tekanan tetap kalor yang diterimanya adalah $\Delta H$
(\cref{prop:b1:first-law:qvqp}); karena reversibel dan isotermal, ia
$T\Delta S$ (\cref{prop:b1:second-law-entropy:identity}); dan $U = H - PV$.
\end{proof}

\begin{example}[Ke mana kalor penguapannya pergi]\label{ex:b1:phase-changes:whereheat}
Untuk air pada \qty{100}{\degreeCelsius}: $P_s(v_v - v_l) = 1.013 \times 10^5 \times
1.67 = \qty{169}{kJ/kg}$, sehingga $u_v - u_l = 2257 - 169 = \qty{2088}{kJ/kg}$:
$93\%$ \omterm{def:b1:phase-changes:phase}{kalor latennya} dipakai untuk merenggutkan molekulnya, $7\%$
untuk mendorong mundur atmosfernya. Entropi penguapannya,
$2257/373 = \qty{6.05}{kJ/(kg.K)}$, jauh melampaui entropi peleburannya, $334/273 =
\qty{1.22}{kJ/(kg.K)}$: peleburan melonggarkan molekulnya, pendidihan membebaskannya.
\end{example}

\begin{method}[Kalorimetri dengan perubahan fase]\label{meth:b1:phase-changes:calorimetry}
Di dalam bejana terisolasi pada tekanan tetap, $\sum\Delta H_i = 0$. Terkalah
keadaan akhirnya (\omterm{def:b1:phase-changes:phase}{fase} mana yang hadir, pada suhu berapa);
tuliskan $\Delta H$ tiap benda sebagai kalor sensibel $mc\,\Delta T$ ditambah \omterm{def:b1:phase-changes:phase}{kalor
laten} $\pm mL$ bagi perubahan yang dialaminya; selesaikan; lalu \emph{periksalah terkaan itu}:
sebuah benda yang berakhir di atas titik didihnya atau di bawah titik
leburnya, atau massa sebuah \omterm{def:b1:phase-changes:phase}{fase} yang negatif, berarti terkaannya keliru --- keadaan
akhirnya lalu berupa campuran pada suhu peralihannya, dan yang
tak diketahui menjadi massa yang berubah (\cref{ex:b1:first-law:ice}).
\end{method}

\begin{example}[Uap membakar]\label{ex:b1:phase-changes:steam}
\qty{10}{g} uap pada \qty{100}{\degreeCelsius} yang mengembun di kulit memberikan
$0.010 \times 2257 = \qty{22.6}{kJ}$ sebelum ia mendingin satu derajat pun ---
sebanyak yang diberikan \qty{10}{g} air mendidih ketika mendingin sebesar
\qty{540}{K}. Itulah sebabnya uap jauh lebih parah melepuhkan daripada air pada
suhu yang sama, dan sebabnya radiator uap menuntut aliran massa yang begitu kecil.
\end{example}

\section{Hubungan Clausius--Clapeyron}

\begin{theorem}[Clausius--Clapeyron]\label{thm:b1:phase-changes:clapeyron}
\index{hubungan Clausius--Clapeyron}
Di sepanjang kurva pendampingan \omterm{def:b1:phase-changes:phase}{fase} 1 dan 2,
\[
\frac{\dd P_s}{\dd T} = \frac{L_{12}(T)}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
Bawalah satu satuan massa mengelilingi \omterm{def:b1:heat-engines:carnotcycle}{siklus Carnot} yang sangat kecil di dalam kubahnya:
peralihan penuh $1 \to 2$ menyusuri platonya pada $T + \dd T$, dengan tekanan
$P_s + \dd P_s$ (kalor yang diterima $L_{12}$ sampai orde pertama); sebuah langkah
adiabatik yang sangat kecil turun ke $T$; peralihan baliknya menyusuri platonya pada
$T$, $P_s$; sebuah langkah adiabatik kembali. Siklusnya reversibel, diterm
di antara $T + \dd T$ dan $T$, sehingga \omterm{def:b1:heat-engines:efficiency}{efisiensinya} $\dd T/T$
(\cref{thm:b1:heat-engines:carnot}); usahanya, yaitu luas persegi panjang
yang tipis itu, adalah $\dd P_s\,(v_2 - v_1)$. Maka $\dd P_s\,(v_2 - v_1)/L_{12} =
\dd T/T$.
\end{proof}

\begin{corollary}[Tekanan uap jenuh sebuah cairan]\label{cor:b1:phase-changes:vapourpressure}
Bila uapnya \omterm{def:b1:kinetic-theory:model}{gas ideal} dan $v_v \gg v_l$, sehingga $v_v - v_l \approx
RT/(MP_s)$, dan bila $L_v$ berubah sedikit saja,
\[
\frac{\dd P_s}{P_s} = \frac{ML_v}{R}\,\frac{\dd T}{T^2} , \qquad
\ln\frac{P_s(T)}{P_s(T_0)} = \frac{ML_v}{R}\Bigl(\frac{1}{T_0} - \frac{1}{T}\Bigr) :
\]
tekanan uapnya naik kira-kira secara eksponensial terhadap suhu. Untuk
air dekat \qty{100}{\degreeCelsius}, $ML_v/R = 0.018 \times 2.257 \times 10^6/
8.314 = \qty{4890}{K}$ dan $\dd P_s/\dd T = 2.257 \times 10^6/(373 \times 1.67) =
\qty{3.6}{kPa/K}$: tiap \omterm{def:b1:kinetic-theory:temperature}{kelvin} suhu didih tambahan berharga
\qty{36}{mbar} tekanan tambahan.
\end{corollary}

\begin{proof}
Substitusikan $v_v - v_l \approx RT/(MP_s)$ ke dalam teoremanya lalu pisahkan
variabelnya; integralkan dengan $L_v$ tetap.
\end{proof}

\section{Keadaan metastabil}

\begin{definition}[Lewat dingin, lewat panas]\label{def:b1:phase-changes:metastable}
\index{lewat dingin}\index{lewat panas}\index{keadaan metastabil}\index{nukleasi}
Sebuah \omterm{def:b1:phase-changes:phase}{fase} dapat bertahan di luar batas kesetimbangannya: air cair
yang didinginkan di bawah \qty{0}{\degreeCelsius} tanpa membeku (\emph{lewat dingin}),
yang dipanaskan di atas titik didihnya tanpa mendidih (\emph{lewat panas}),
uap yang dimampatkan melewati $P_s$ tanpa mengembun (lewat jenuh).
\emph{Keadaan metastabil} semacam itu menuntut ketiadaan tapak \emph{nukleasi}
--- debu, goresan, gas terlarut, kristal es --- yang di atasnya
\omterm{def:b1:phase-changes:phase}{fase} barunya dapat bermula; sebuah gangguan menjungkalkannya dengan tiba-tiba dan tak reversibel
ke kesetimbangan.
\end{definition}

\begin{example}[Botol yang lewat dingin]\label{ex:b1:phase-changes:supercooled}
Air yang tenang pada \qty{-8}{\degreeCelsius} di dalam botol yang bersih, lalu diketuk: kristal
es menjalar cepat menembusnya dan suhunya melonjak ke
\qty{0}{\degreeCelsius}. Prosesnya adiabatik dan cepat, sehingga kalor
sensibelnya $c\,\Delta T$ membekukan pecahan $x = c\,\Delta T/L_f = 4.18 \times 8/334 =
0.10$ airnya; entropi yang tercipta, $c\ln(273/265) - xL_f/273 =
124 - 122 = \qty{+2}{J/(kg.K)}$, positif sebagaimana seharusnya. Awan
penuh dengan tetesan yang \omterm{def:b1:phase-changes:metastable}{lewat dingin} sampai \qty{-40}{\degreeCelsius}; pembekuan es
pada pesawat dan penyemaian awan sama-sama memanfaatkannya.
\end{example}

\begin{remark}[Batu didih dan kamar gelembung]\label{rem:b1:phase-changes:nucleation}
Sebuah gelembung berjari-jari $r$ di dalam cairan bertekanan $P$ memuat uap pada
$P + 2\sigma/r$ (dengan tegangan permukaan $\sigma$, volume Tahun ke-2); ia hanya dapat
tumbuh bila $P_s(T) > P + 2\sigma/r$, sehingga cairan yang bersih pada titik didihnya
tak mempunyai gelembung yang cukup kecil untuk bermula dan menjadi \omterm{def:b1:phase-changes:metastable}{lewat panas} beberapa \omterm{def:b1:kinetic-theory:temperature}{kelvin}
sebelum meletup --- yaitu ``letupan'' yang dicegah batu didih dengan
menyediakan rongga yang penuh gas terperangkap. Kamar gelembung fisika
partikel bekerja pada ketakmantapan yang sama secara terbalik: cairan yang \omterm{def:b1:phase-changes:metastable}{lewat panas}
mendidih lebih dulu menyusuri jejak sebuah partikel pengion.
\end{remark}

\begin{example}[Kelembapan dan embun]\label{ex:b1:phase-changes:dew}
\index{kelembapan nisbi}\index{titik embun}
Udara memuat uap air pada tekanan parsial $P_v \leq P_s(T)$; yang disebut
\emph{\omterm{ex:b1:phase-changes:dew}{kelembapan nisbi}} adalah $P_v/P_s(T)$, dan \emph{\omterm{ex:b1:phase-changes:dew}{titik embun}} adalah
suhu ketika $P_s$ turun sampai $P_v$. Udara pada \qty{25}{\degreeCelsius}
($P_s = \qty{3.17}{kPa}$) dengan kelembapan $60\%$ mempunyai $P_v = \qty{1.9}{kPa}$,
$\rho_v = P_vM/(RT) = \qty{14}{g/m^3}$, dan mengembun pada permukaan mana pun di bawah
kira-kira \qty{17}{\degreeCelsius} --- rumput di pagi hari, botol yang dingin, cermin
kamar mandi. Ketika terangkat, udara yang sama mendingin secara adiabatik sebesar
\qty{9.8}{K/km} (\cref{pb:b1:phase-changes:1}) lalu mencapai \omterm{ex:b1:phase-changes:dew}{titik embunnya} kira-kira
sekilometer di atas: di sanalah dasar rata awan kumulus berada.
\end{example}

\section{Soal: Air dalam empat adegan}

\begin{problem}[{Soal akhir pekan --- panci presto, awan, bejana
tersegel dan botol yang lewat dingin: satu zat, empat pemakaian
diagram fasenya}]
\label{pb:b1:phase-changes:1}
Data untuk air: $L_v = \qty{2257}{kJ/kg}$ pada \qty{100}{\degreeCelsius} dan
kira-kira \qty{2.2}{MJ/kg} pada \qty{117}{\degreeCelsius}, \qty{2.45}{MJ/kg} dekat
\qty{20}{\degreeCelsius}; $L_f = \qty{334}{kJ/kg}$; $c = \qty{4.18}{kJ/(kg.K)}$
(cairan); $M = \qty{18}{g/mol}$; $ML_v/R = \qty{4890}{K}$ dekat
\qty{100}{\degreeCelsius}. Tekanan jenuhnya: \qty{0.61}{kPa}
(\qty{0}{\degreeCelsius}), \qty{1.23} (\qty{10}), \qty{1.70} (\qty{15}),
\qty{2.34} (\qty{20}), \qty{3.17} (\qty{25}), \qty{4.24}{kPa}
(\qty{30}{\degreeCelsius}), \qty{101.3}{kPa} (\qty{100}{\degreeCelsius}).
Volume jenis jenuhnya: $v_l = \qty{1.0e-3}{m^3/kg}$ (air dingin),
\qty{1.9e-3}{m^3/kg} pada \qty{360}{\degreeCelsius}; $v_v = \qty{0.0217}{m^3/kg}$
pada \qty{300}{\degreeCelsius}, $0.0108$ pada \qty{340}{\degreeCelsius}, $0.0088$
pada \qty{350}{\degreeCelsius}; titik kritisnya \qty{374}{\degreeCelsius},
\qty{221}{bar}, $v_c = \qty{3.1e-3}{m^3/kg}$. Udara: $M = \qty{29}{g/mol}$, $c_p =
\qty{1.0}{kJ/(kg.K)}$, $\gamma = 1.4$; $g = \qty{9.8}{m/s^2}$; $R =
\qty{8.314}{J/(mol.K)}$.

\medskip
\noindent\textbf{Bagian I --- Panci prestonya.}
\begin{enumerate}
  \item Mengapa air di dalam panci terbuka di permukaan laut tak dapat dipanaskan di atas
        \qty{100}{\degreeCelsius}, sekuat apa pun apinya?
  \item Berangkat dari \omterm{thm:b1:phase-changes:clapeyron}{hubungan Clausius--Clapeyron}, turunkan
        $\ln[P_s(T)/P_s(T_0)] = (ML_v/R)(1/T_0 - 1/T)$, dengan menyebutkan
        hampirannya.
  \item Berapa suhu di dalam panci yang katupnya membuka pada \qty{1.8}{bar}
        mutlak.
  \item Katupnya adalah sebuah pemberat yang duduk di atas lubang \qty{20}{mm^2};
        dengan tekanan atmosfer \qty{1.0}{bar}: berapa massa pemberatnya.
  \item \qty{2.0}{L} air dibawa dari \qty{20}{\degreeCelsius} ke suhu
        kerjanya oleh sebuah tungku \qty{2.0}{kW} (abaikan pancinya):
        berapa kalornya dan berapa waktunya.
  \item Setelah sampai di sana, tungkunya dibiarkan pada \qty{2.0}{kW}: berapa massa uap
        yang meninggalkan katupnya tiap menit. Apa yang harus dilakukan juru masaknya?
  \item Laju reaksi masaknya kira-kira berlipat dua tiap \qty{10}{K}:
        sebesar faktor berapa pancinya memendekkan resep \qty{30}{min} di
        permukaan laut? Dan berapa lama resep itu akan memakan waktu di Everest, tempat
        air mendidih pada \qty{70}{\degreeCelsius}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Awannya.}
\begin{enumerate}[resume]
  \item Definisikan \omterm{ex:b1:phase-changes:dew}{kelembapan nisbi}. Udara pada \qty{25}{\degreeCelsius} dengan $60\%$:
        berapa tekanan parsial uapnya, dan berapa massa uap tiap meter kubik
        (\omterm{def:b1:kinetic-theory:model}{gas ideal}).
  \item Berapa \omterm{ex:b1:phase-changes:dew}{titik embun} udara itu (interpolasikan tabelnya). Jelaskan embun
        di rumput pada pagi hari dan tetesan pada botol yang dingin.
  \item Sebongkah udara kering naik tanpa bertukar kalor. Dari
        hukum adiabatik reversibel dan \omterm{thm:b1:fluid-statics:fundamental}{hubungan hidrostatiknya}
        $\dd P/\dd z = -\rho g$, tunjukkan bahwa $\dd T/\dd z = -g/c_p$ lalu
        hitunglah.
  \item \omterm{ex:b1:phase-changes:dew}{Titik embun} bongkahan yang naik itu sendiri turun kira-kira
        \qty{2}{K/km} ketika tekanannya turun. Pada ketinggian berapa bongkahan
        pertanyaan~8 menjadi jenuh --- yaitu dasar awannya.
  \item Di atas dasarnya, uap yang mengembun melepaskan \omterm{def:b1:phase-changes:phase}{kalor latennya}
        ke dalam bongkahannya. Sebesar apa pengembunan \qty{1}{g}
        air tiap kilogram udara menghangatkan udara itu? Apa pengaruhnya pada
        laju pendinginan bongkahannya ketika ia terus naik, dan mengapa itu
        berarti bagi badai guruh?
  \item Sebuah awan kumulus \qty{1}{km^3} memuat \qty{1}{g} air cair tiap meter
        kubik: berapa massa airnya, dan berapa energi yang dilepaskan
        pengembunannya, dalam joule dan dalam \unit{GW h}.
  \item Sebuah cermin kamar mandi pada \qty{18}{\degreeCelsius}, dengan udara pada
        \qty{25}{\degreeCelsius} dan $80\%$: apakah ia berembun? Pada kelembapan berapa
        ia akan berhenti berembun?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Bejana yang tersegel.}
Sebuah bejana baja tegar \qty{1.0}{L} diisi massa air $m$
lalu disegel tanpa udara, kemudian dipanaskan perlahan.
\begin{enumerate}[resume]
  \item $m = \qty{0.50}{kg}$ pada \qty{20}{\degreeCelsius}: berapa tekanan di dalamnya,
        volume jenisnya, massa uapnya (pakailah \omterm{def:b1:kinetic-theory:model}{gas ideal} untuk $v_v$),
        dan pecahan volume yang terisi cairan.
  \item Ketika dipanaskan: apakah aras cairannya naik atau turun? Pada suhu kira-kira
        berapa cairannya memenuhi bejananya, dan apa yang terjadi pada
        tekanannya setelah itu? (Pakailah $v_l$ pada \qty{360}{\degreeCelsius}.)
  \item $m = \qty{0.10}{kg}$: apakah aras cairannya naik atau turun ketika
        dipanaskan? Pada suhu kira-kira berapa tetes terakhirnya
        menguap (pakailah tabel $v_v$)?
  \item $m = \qty{0.31}{kg}$: lukiskan apa yang dilihat seorang pengamat pada
        meniskusnya ketika suhunya mendekati \qty{374}{\degreeCelsius}.
  \item Sketsalah ketiga \omterm{def:b1:point-kinematics:frame}{lintasan} pemanasannya pada diagram $(v, P)$ beserta
        kubah jenuhnya, lalu katakan di sisi mana titik kritisnya masing-masing
        lewat.
  \item Bejana \qty{1.0}{L} yang sama dengan \qty{0.50}{kg} air kini
        disegel \emph{beserta} udara yang dimuatnya pada \qty{20}{\degreeCelsius},
        \qty{1.0}{bar}, lalu dipanaskan sampai \qty{100}{\degreeCelsius}: berapa tekanan
        total di dalamnya (Dalton), dan berapa gaya pada tutup seluas \qty{100}{cm^2}.
\end{enumerate}

\noindent\textbf{Bagian IV --- Botol yang \omterm{def:b1:phase-changes:metastable}{lewat dingin}.}
\begin{enumerate}[resume]
  \item Sebotol air yang bersih dan tenang telah mendingin sampai \qty{-8}{\degreeCelsius}
        tanpa membeku. Keadaan ini disebut apa, mengapa ia mungkin,
        dan apa yang terjadi ketika botolnya diketuk? Mengapa suhunya
        lalu berhenti tepat pada \qty{0}{\degreeCelsius}?
  \item Berapa pecahan airnya yang membeku (prosesnya cepat, sehingga
        adiabatik).
  \item Berapa entropi yang tercipta tiap kilogram; periksalah tandanya.
  \item Kejutan yang sebaliknya: secangkir air yang dipanaskan sampai \qty{105}{\degreeCelsius}
        di dalam oven mikrogelombang tanpa mendidih, lalu diganggu. Berapa pecahan
        airnya yang berubah seketika menjadi uap, berapa volume uap yang dihasilkan tiap
        liter ($v_v = \qty{1.67}{m^3/kg}$), dan mengapa cangkirnya meletup.
  \item \omterm{thm:b1:heat-engines:lostwork}{Usaha yang hilang}: seliter air yang \omterm{def:b1:phase-changes:metastable}{lewat dingin} itu dapat saja menggerakkan sebuah mesin
        reversibel bersama lingkungan \qty{0}{\degreeCelsius}-nya; dengan mengambil
        $T_0 = \qty{273}{K}$, usaha yang dibuang dengan membiarkannya membeku karena
        sebuah ketukan adalah $T_0S_{\mathrm{cipta}}$. Hitunglah tiap liter lalu
        bandingkan dengan mengangkat botolnya setinggi satu meter. Simpulkan: apa yang
        disimpan sebuah \omterm{def:b1:phase-changes:metastable}{keadaan metastabil}?
\end{enumerate}
\end{problem}
