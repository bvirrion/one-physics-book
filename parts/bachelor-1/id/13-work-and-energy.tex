\chapter{Usaha, Energi, dan Energi Potensial}\label{ch:b1:work-and-energy}

Seorang pesepeda berhenti mengayuh di puncak bukit lalu tiba di dasarnya
dengan laju yang bergantung pada turunannya, bukan pada bentuk jalannya.
Seorang peloncat bungee jatuh enam puluh meter lalu berhenti, sesaat,
tepat di tempat tarikan talinya telah memakan seluruh laju yang
diberikan jatuhnya. Sebuah atom hidrogen yang terikat pada atom klor
bergetar di dalam sumur yang kedalamannya adalah energi yang harus
dibayar untuk mencerai-beraikan molekulnya. Pada tiap kasus, satu angka
dipertukarkan antara berbagai bentuknya --- gerak, ketinggian, regangan,
ikatan --- dan ketika dijumlahkan, ia tidak berubah. Bab ini membangun
pembukuan itu dari hukum kedua Newton: \omterm{def:b1:work-and-energy:work}{usaha} dan daya, teorema \omterm{thm:b1:work-and-energy:ke}{energi
kinetik}, \omterm{def:b1:work-and-energy:conservative}{energi potensial} bagi gaya yang mengizinkannya, dan \omterm{thm:b1:work-and-energy:em}{kekekalan
energi mekanik} yang mengubah banyak soal dinamika menjadi aljabar satu
baris serta menjelaskan, lewat bentuk sebuah \omterm{def:b1:work-and-energy:turning}{sumur potensial}, kedudukan
mana yang mantap.

\section{Daya dan usaha}

\begin{definition}[Daya dan usaha sebuah gaya]\label{def:b1:work-and-energy:work}
\index{daya (sebuah gaya)}\index{usaha}
\emph{Daya} sebuah gaya $\vect F$ yang bekerja pada sebuah titik yang
bergerak dengan kecepatan $\vect v$ adalah
\[
\mathcal P = \vect F\cdot\vect v \qquad (\text{watt}),
\]
dan \emph{usaha}nya antara saat $t_1$ dan $t_2$ (kedudukan $A$ dan $B$)
adalah daya yang terhimpun,
\[
W_{A\to B} = \int_{t_1}^{t_2}\vect F\cdot\vect v\,\dd t
= \int_A^B \vect F\cdot\dd\vect\ell \qquad (\text{joule}),
\]
yaitu jumlah \emph{usaha elementer} $\delta W = \vect F\cdot\dd\vect\ell$
sepanjang perpindahan elementer $\dd\vect\ell = \vect v\,\dd t$ pada
\omterm{def:b1:point-kinematics:frame}{lintasannya}. Usahanya positif bila gayanya mendorong searah geraknya
(\emph{memacu}), negatif bila melawannya (\emph{menghambat}), dan nol
bila tegak lurus.
\end{definition}

\begin{proposition}[Usaha yang sering ditemui]\label{prop:b1:work-and-energy:works}
\begin{itemize}
  \item Gaya tetap: $W_{A\to B} = \vect F\cdot\vect{AB}$, apa pun
        \omterm{def:b1:point-kinematics:frame}{lintasannya}. Berat: $W = mg(z_A - z_B)$, dengan $z$ dihitung ke atas.
  \item Pegas berkekakuan $k$, dari pertambahan panjang $x_A$ ke $x_B$:
        $W = \tfrac12 k(x_A^2 - x_B^2)$.
  \item \omterm{prop:b1:newton-dynamics:friction}{Reaksi normal} permukaan tetap, tegangan tali pada benda yang
        bergerak tegak lurus terhadapnya: $W = 0$ (gaya $\perp$ perpindahan).
  \item Gesekan kinetis dan \omterm{prop:b1:newton-dynamics:drag}{gaya hambat}: $W < 0$ selalu (gayanya melawan
        gerak relatifnya), dan bergantung pada panjang \omterm{def:b1:point-kinematics:frame}{lintasannya}.
\end{itemize}
\end{proposition}

\begin{proof}
$\vect F$ tetap: $\int\vect F\cdot\dd\vect\ell = \vect F\cdot\int\dd\vect\ell
= \vect F\cdot\vect{AB}$; dengan $\vect F = -mg\vect e_z$, $\vect F\cdot\vect{AB}
= -mg(z_B - z_A)$. Pegas: $\vect F\cdot\dd\vect\ell = -kx\,\dd x$, dan
$\int_{x_A}^{x_B}-kx\,\dd x = \tfrac12 k(x_A^2 - x_B^2)$. Gesekan: $\vect T
= -\mu N\vect v/v$, sehingga $\vect T\cdot\vect v = -\mu Nv < 0$.
\end{proof}

\section{Teorema energi kinetik}

\begin{theorem}[Teorema energi kinetik dan teorema daya]\label{thm:b1:work-and-energy:ke}
\index{energi kinetik}\index{teorema energi kinetik}
Dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial}, \emph{\omterm{thm:b1:work-and-energy:ke}{energi kinetik}} $E_k = \tfrac12 mv^2$
sebuah \omterm{def:b1:point-kinematics:frame}{partikel titik} menuruti
\[
\frac{\dd E_k}{\dd t} = \sum\mathcal P_i = \Big(\sum\vect F_i\Big)\cdot\vect v,
\qquad
E_k(B) - E_k(A) = \sum_i W_{i,A\to B}:
\]
perubahan \omterm{thm:b1:work-and-energy:ke}{energi kinetik} antara dua kedudukan sama dengan \omterm{def:b1:work-and-energy:work}{usaha} total
semua gaya yang diterapkan di sepanjang jalannya.
\end{theorem}

\begin{proof}
Kalikan titik hukum kedua Newton dengan $\vect v$: $m\vect a\cdot\vect v =
\tfrac{\dd}{\dd t}(\tfrac12 m\vect v\cdot\vect v) = \dd E_k/\dd t$, sedangkan
$(\sum\vect F_i)\cdot\vect v = \sum\mathcal P_i$. Integralkan terhadap waktu.
\end{proof}

\begin{example}[Jarak pengereman]\label{ex:b1:work-and-energy:braking}
Sebuah mobil \qty{1000}{kg} pada \qty{100}{km/h} ($E_k = \qty{386}{kJ}$)
yang direm oleh gaya tetap \qty{7.0}{kN} berhenti setelah
$d = E_k/F = \qty{55}{m}$: jaraknya tumbuh sebagai \emph{kuadrat}
lajunya. Karena gaya rem maksimumnya adalah $\mu_smg$
(\cref{ch:b1:newton-dynamics}), jarak berhenti terpendeknya adalah
$v^2/2\mu_sg$ berapa pun massa mobilnya --- dan ia berlipat dua di jalan
basah tempat $\mu_s$ berkurang separuh.
\end{example}

\section{Gaya konservatif dan energi potensial}

\begin{definition}[Gaya konservatif; energi potensial]\label{def:b1:work-and-energy:conservative}
\index{gaya konservatif}\index{energi potensial}
Sebuah gaya disebut \emph{konservatif} bila \omterm{def:b1:work-and-energy:work}{usahanya} antara dua titik
tidak bergantung pada \omterm{def:b1:point-kinematics:frame}{lintasan} yang diikuti. Setara dengan itu, ada
sebuah fungsi kedudukan, yaitu \emph{energi potensial} $E_p$, sedemikian
sehingga
\[
W_{A\to B} = E_p(A) - E_p(B) = -\Delta E_p ,
\qquad \delta W = -\dd E_p ,
\]
dengan $E_p$ terdefinisi sampai pada sebuah tetapan tambahan (pilihan
titik asalnya).
\end{definition}

\begin{proposition}[Gaya dari energi potensial]\label{prop:b1:work-and-energy:gradient}
\index{gradien}
\omterm{def:b1:work-and-energy:conservative}{Gaya konservatif} diturunkan dari \omterm{def:b1:work-and-energy:conservative}{energi potensialnya}:
\[
\vect F = -\overrightarrow{\operatorname{grad}}\,E_p
= -\frac{\partial E_p}{\partial x}\,\vect e_x
- \frac{\partial E_p}{\partial y}\,\vect e_y
- \frac{\partial E_p}{\partial z}\,\vect e_z ,
\]
dan dalam satu dimensi $F_x = -\dd E_p/\dd x$: gayanya menunjuk ke arah
\omterm{def:b1:work-and-energy:conservative}{energi potensial} yang menurun, ``menuruni bukit'' pada grafik $E_p$.
\end{proposition}

\begin{proof}
$\delta W = F_x\dd x + F_y\dd y + F_z\dd z = -\dd E_p = -(\partial_xE_p\,\dd x
+ \partial_yE_p\,\dd y + \partial_zE_p\,\dd z)$ untuk setiap perpindahan:
kenali koefisiennya (diferensial sebuah fungsi beberapa variabel, dari
kuliah matematika).
\end{proof}

\begin{proposition}[Energi potensial yang lazim]\label{prop:b1:work-and-energy:eps}
\index{energi potensial!gravitasi}\index{energi potensial!elastik}
\begin{itemize}
  \item Berat: $E_p = mgz + \text{tetapan}$ ($z$ ke atas).
  \item Pegas: $E_p = \tfrac12 k(\ell - \ell_0)^2 + \text{tetapan}$.
  \item Gravitasi Newton oleh massa $M$: $E_p = -GMm/r + \text{tetapan}$,
        dengan tetapannya biasanya dipilih agar $E_p \to 0$ di tak hingga.
  \item Gaya elektrostatik pada muatan $q$ dalam potensial $V$:
        $E_p = qV$ (\cref{ch:b1:potential-capacitors}).
\end{itemize}
Gesekan kinetis dan \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} fluida \emph{bukan} \omterm{def:b1:work-and-energy:conservative}{gaya konservatif}
(\omterm{def:b1:work-and-energy:work}{usahanya} bergantung pada \omterm{def:b1:point-kinematics:frame}{lintasannya}, dan selalu negatif).
\end{proposition}

\begin{proof}
Masing-masing dibaca dari \omterm{def:b1:work-and-energy:work}{usaha} \cref{prop:b1:work-and-energy:works}:
$W = -\Delta E_p$. Untuk gravitasinya, $\vect F = -GMm/r^2\,\vect e_r$ dan
$\delta W = -GMm\,\dd r/r^2 = -\dd(-GMm/r)$.
\end{proof}

\section{Energi mekanik}

\begin{theorem}[Energi mekanik]\label{thm:b1:work-and-energy:em}
\index{energi mekanik}\index{kekekalan energi mekanik}
Misalkan $E_m = E_k + E_p$ adalah \emph{\omterm{thm:b1:work-and-energy:em}{energi mekanik}}, dengan $E_p$
sebagai jumlah \omterm{def:b1:work-and-energy:conservative}{energi potensial} \omterm{def:b1:work-and-energy:conservative}{gaya konservatifnya}. Maka
\[
\Delta E_m = W_{\text{nk}} ,
\]
yaitu \omterm{def:b1:work-and-energy:work}{usaha} total gaya yang tak \omterm{def:b1:work-and-energy:conservative}{konservatif}. Jika gaya itu tak berusaha
(tanpa gesekan, atau gaya kendalanya tegak lurus geraknya), $E_m$
bersifat \emph{kekal}: geraknya \emph{\omterm{def:b1:work-and-energy:conservative}{konservatif}}.
\end{theorem}

\begin{proof}
$\Delta E_k = W_{\text{k}} + W_{\text{nk}} = -\Delta E_p + W_{\text{nk}}$.
\end{proof}

\begin{method}[Memakai energi]\label{met:b1:work-and-energy:method}
Ketika sebuah soal menanyakan \emph{laju pada suatu kedudukan} (bukan
waktu, bukan gaya), cobalah energi lebih dahulu:
\begin{enumerate}
  \item daftarkan gayanya; periksa mana yang berusaha dan mana yang
        \omterm{def:b1:work-and-energy:conservative}{konservatif};
  \item tuliskan $E_m$ pada kedua kedudukannya, dengan titik asal $E_p$
        yang jelas;
  \item samakan keduanya (dengan menambahkan $W_{\text{nk}}$ bila ada
        gesekan) lalu selesaikan untuk laju atau kedudukan yang dicari.
\end{enumerate}
Energi tak pernah memberi waktunya atau gaya kendalanya; untuk keduanya,
kembalilah ke \omterm{thm:b1:newton-dynamics:laws}{hukum Newton} --- sering dengan laju yang baru saja
diberikan energinya.
\end{method}

\begin{example}[Hukum kelajuan loopnya]\label{ex:b1:work-and-energy:loop}
Pada loop tanpa gesekan \cref{pb:b1:point-kinematics:1}, reaksi
\omterm{def:b1:point-kinematics:frame}{lintasannya} tegak lurus geraknya dan tidak berusaha: $E_m$ kekal.
Dengan $z = R(1 - \cos\theta)$ di atas dasarnya, $\tfrac12 mv^2 + mgR(1 -
\cos\theta) = \tfrac12 mv_0^2$: itulah hukum kelajuan $v^2 = v_0^2 - 2gR(1 -
\cos\theta)$ yang dipakai di sana, kini diturunkan dalam satu baris.
Laju masuk minimum $\sqrt{5gR}$ lalu menyusul dari syarat puncaknya
$v^2 \geq gR$ pada \cref{ex:b1:newton-dynamics:circles}.
\end{example}

\begin{example}[Energi yang hilang oleh gesekan]\label{ex:b1:work-and-energy:friction}
Sebuah balok tergelincir menuruni lereng bersudut $\alpha$ dan
bertinggi $h$ dengan gesekan kinetis $\mu_d$: $W_{\text{nk}} = -\mu_dmg\cos\alpha \times
h/\sin\alpha = -\mu_dmgh\cot\alpha$, sehingga $\tfrac12 mv^2 = mgh(1 - \mu_d\cot\alpha)$.
Untuk $h = \qty{2.0}{m}$, $\alpha = 30^\circ$, $\mu_d = 0.20$: $v =
\qty{5.1}{m/s}$ alih-alih $\qty{6.3}{m/s}$; sepertiga \omterm{def:b1:work-and-energy:conservative}{energi
potensialnya} menjadi kalor.
\end{example}

\section{Kesetimbangan, kemantapan, dan potret fase}

\begin{proposition}[Kesetimbangan dan kemantapan dalam satu dimensi]\label{prop:b1:work-and-energy:stability}
\index{kesetimbangan!mantap, goyah}\index{osilasi kecil}
Sebuah partikel yang bergerak pada sebuah sumbu di bawah \omterm{def:b1:work-and-energy:conservative}{gaya
konservatif} berenergi potensial $E_p(x)$:
\begin{itemize}
  \item berada dalam \emph{kesetimbangan} di $x_0$ jika dan hanya jika
        $F(x_0) = -E_p'(x_0) = 0$: yaitu pada ekstremum $E_p$;
  \item kesetimbangannya \emph{mantap} bila $E_p$ mempunyai minimum
        lokal di sana ($E_p''(x_0) > 0$): perpindahan kecil menghasilkan
        gaya pemulih; dan \emph{goyah} pada sebuah maksimum;
  \item di dekat kesetimbangan yang mantap, gerak kecilnya bersifat
        harmonik, dengan
        \[
        \omega_0^2 = \frac{E_p''(x_0)}{m} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Rumus Taylor (kuliah matematika): $E_p(x) \approx E_p(x_0) +
\tfrac12 E_p''(x_0)(x - x_0)^2$, sehingga $F \approx -E_p''(x_0)(x - x_0)$ dan
$m\ddot x = -E_p''(x_0)(x - x_0)$ --- sebuah pegas berkekakuan $E_p''(x_0)$.
Untuk $E_p'' < 0$ ``kekakuan''nya negatif: kepergian yang eksponensial.
\end{proof}

\begin{definition}[Titik balik; gerak terikat dan gerak bebas]\label{def:b1:work-and-energy:turning}
\index{titik balik}\index{gerak terikat}
\index{sumur potensial}\index{penghalang potensial}
Untuk gerak satu dimensi yang \omterm{def:b1:work-and-energy:conservative}{konservatif} berenergi $E_m$, partikelnya
hanya dapat berada di tempat $E_p(x) \leq E_m$ (karena $E_k \geq 0$);
titik tempat $E_p = E_m$ disebut \emph{titik balik}, tempat kecepatannya
lenyap lalu berbalik. Ketika terperangkap di antara dua titik balik
dalam sebuah \emph{sumur potensial}, geraknya \emph{terikat} dan
berkala; ketika ia dapat mencapai tak hingga, geraknya \emph{bebas};
sedangkan \emph{penghalang potensial} yang lebih tinggi daripada $E_m$
tak dapat dilintasi.
\end{definition}

\begin{definition}[Potret fase]\label{def:b1:work-and-energy:phase}
\index{potret fase}
\emph{Potret fase} sebuah gerak satu dimensi adalah keluarga kurva yang
dijejakkan titik $(x, \dot x)$ pada \emph{bidang fase}. Untuk sistem
\omterm{def:b1:work-and-energy:conservative}{konservatif}, tiap kurvanya adalah garis aras $\tfrac12 m\dot x^2 +
E_p(x) = E_m$: kurva tertutup di sekitar kesetimbangan mantap
(osilasi), kurva terbuka bagi gerak bebas, dan, lewat kesetimbangan
goyah, \emph{separatriks} yang membagi keduanya. Kurvanya ditelusuri
searah jarum jam ($\dot x > 0$ berarti $x$ menaik) dan tak pernah
berpotongan.
\end{definition}

\begin{example}[Bentang bandulnya]\label{ex:b1:work-and-energy:pendulum}
$E_p = -mg\ell\cos\theta$: minimum di $\theta = 0$, $E_p'' = mg\ell$,
sehingga $\omega_0^2 = g\ell/(m\ell^2) = g/\ell$ --- hasil \omterm{prop:b1:work-and-energy:stability}{osilasi kecil}
\cref{ex:b1:newton-dynamics:pendulum} yang dibaca dari kelengkungan
sumurnya; maksimum di $\theta = \pi$, yaitu bandul terbalik, yang goyah.
Energi $E_m < mg\ell$: osilasi antara $\pm\theta_{\max}$; $E_m > mg\ell$:
ia melewati puncaknya lalu berputar; $E_m = mg\ell$: separatriksnya,
yaitu pendekatan yang sangat lambat menuju puncaknya.
\end{example}

\section{Soal: Sumur energi sebuah ikatan kimia}

\begin{problem}[{Soal akhir pekan --- dua atom, satu kurva: energi
potensial sebuah ikatan yang dibaca sebagai sebuah bentang --- lantainya,
dindingnya, getaran kecil di dasarnya, dan energi yang dituntut untuk
memanjatnya keluar}]
\label{pb:b1:work-and-energy:1}
Interaksi kedua atom sebuah molekul dwiatom dimodelkan oleh \omterm{def:b1:work-and-energy:conservative}{energi
potensial}
\[
E_p(r) = \epsilon\left[\left(\frac{\sigma}{r}\right)^{12} - 2\left(\frac{\sigma}{r}\right)^{6}\right],
\]
dengan $r$ jarak antara kedua intinya. Untuk hidrogen klorida ambil
$\epsilon = \qty{4.4}{eV}$, $\sigma = \qty{0.127}{nm}$, dan biarkan atom
hidrogen yang ringan ($m = \qty{1.67e-27}{kg}$) bergerak sementara klor
yang berat tinggal diam. $\qty{1}{eV} = \qty{1.60e-19}{J}$;
$k_B = \qty{1.38e-23}{J/K}$; $N_A = \num{6.02e23}$.

\medskip
\noindent\textbf{Bagian I --- Bentangnya.}
\begin{enumerate}
  \item Berikan limit $E_p$ ketika $r \to 0$ dan $r \to \infty$, lalu
        sketsalah kurvanya.
  \item Tunjukkan bahwa $E_p$ mempunyai satu minimum tunggal, pada
        $r = \sigma$, yang bernilai $-\epsilon$.
  \item Dengan $u = (\sigma/r)^6$, tunjukkan bahwa $E_p = \epsilon(u^2 - 2u)$:
        sebuah parabola dalam $u$. Perolehlah kembali minimumnya darinya.
  \item Nyatakan gayanya $F(r)$ (positif bila menolak) lalu berikan
        tandanya pada tiap sisi $\sigma$.
  \item Hitunglah $E_p$ dan $F$ pada $r = 1.2\sigma$ (dalam eV dan dalam nN).
  \item Berapa energi disosiasi molekulnya, dalam eV dan dalam
        \unit{kJ/mol}? (Energi ikat HCl yang terukur: \qty{431}{kJ/mol}.)
  \item Periksalah kriteria kemantapannya pada $r = \sigma$: hitunglah
        $E_p''(\sigma)$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Getaran kecil.}
\begin{enumerate}[resume]
  \item Tuliskan uraian Taylor orde kedua $E_p$ di sekitar $\sigma$ lalu
        kenali sebuah kekakuan efektif $k$.
  \item Hitunglah $k$ (dalam \unit{N/m}).
  \item Simpulkan \omterm{def:b1:wave-propagation:signal}{frekuensi sudut}, frekuensi dan panjang gelombang
        getarannya, serta \omterm{prop:b1:wave-propagation:sinusoidal}{bilangan gelombangnya} $1/\lambda$ dalam \unit{cm^{-1}}.
  \item Getaran HCl yang terukur berada pada \qty{2990}{cm^{-1}}. Berikan
        komentar tentang kecocokannya dan tentang apa yang sudah tepat
        pada modelnya.
  \item Pada \qty{300}{K} energi getaran rata-ratanya berorde $k_BT$.
        Taksirlah \omterm{def:b1:wave-propagation:signal}{amplitudo} getarannya lalu bandingkan dengan $\sigma$.
  \item Simpulkan laju maksimum atom hidrogennya dalam getaran itu.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Osilasi besar dan jalan keluarnya.}
\begin{enumerate}[resume]
  \item Gambarkan \omterm{def:b1:work-and-energy:phase}{potret fasenya}: energi mana yang memberi kurva
        tertutup, mana yang memberi kurva terbuka, dan apa yang
        memisahkan keduanya?
  \item Untuk $E_m = -\epsilon/2$, carilah kedua \omterm{def:b1:work-and-energy:turning}{titik baliknya} (dalam
        satuan $\sigma$).
  \item Tunjukkan bahwa titik tengahnya terletak di luar $\sigma$, lalu
        jelaskan mengapa pemanasan (menaikkan energi getarannya) membuat
        panjang ikatan rata-ratanya tumbuh --- asal-usul pemuaian.
  \item Apakah periode osilasi besarnya lebih panjang atau lebih pendek
        daripada $2\pi/\omega_0$? Berikan alasan dari bentuk dindingnya.
  \item Dari $E_m = -\epsilon/2$, berapa energi yang harus dipasok untuk
        mencerai-beraikan molekulnya? Jika diberikan sebagai \omterm{thm:b1:work-and-energy:ke}{energi
        kinetik} atom hidrogennya pada $r = \sigma$, berapa lajunya?
  \item Dua atom yang saling mendekat dari kejauhan dengan energi total
        positif tak dapat membentuk molekul terikat dengan sendirinya.
        Mengapa? Apa yang diperlukan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- \omterm{def:b1:work-and-energy:work}{Usaha}, gaya, dan suhu.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:b1:work-and-energy:work}{usaha} gaya ikatannya ketika $r$ berjalan dari
        $2\sigma$ ke $\sigma$. Apakah gayanya memacu atau menghambat di sana?
  \item Hitunglah \omterm{def:b1:work-and-energy:work}{usaha} yang harus dilakukan agen luar untuk meregangkan
        ikatannya dari $\sigma$ ke $1.5\sigma$ dengan laju yang menuju nol.
  \item Hitunglah $F$ pada $r = 0.9\sigma$ lalu bandingkan dengan
        $r = 1.2\sigma$: apa yang diberitahukan ketaksetangkupannya
        tentang menekan lawan meregangkan?
  \item Pada suhu berapa $k_BT$ akan sama dengan $\epsilon$? Mengapa
        molekul tercerai-berai pada suhu yang jauh lebih rendah dalam
        praktiknya?
  \item Model yang lebih baik (Morse) mengubah bentuk dindingnya tetapi
        tetap mempunyai minimum pada panjang ikatannya. Hasil mana di
        atas yang bertahan tanpa berubah metodenya, dan angka mana yang
        akan berubah?
  \item Ringkaslah ketiga cara metode energi dipakai: untuk mencari
        kesetimbangan, untuk mencari frekuensi getarannya, dan untuk
        mencari apa yang diperlukan agar lepas.
\end{enumerate}
\end{problem}
