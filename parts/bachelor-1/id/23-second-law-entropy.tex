\chapter{Hukum Kedua: Entropi}\label{ch:b1:second-law-entropy}

Jatuhkan sendok panas ke dalam secangkir air dingin: sendoknya mendingin,
airnya menghangat, dan keduanya berakhir pada satu suhu. Hukum pertama akan
mengizinkan kebalikannya sama saja --- airnya makin dingin sementara
sendoknya makin panas, dengan energi yang kekal sampai ke joule --- namun itu
tak pernah terjadi. Film yang diputar mundur langsung dikenali; molekulnya
menuruti hukum yang tak berarah, sedangkan dunianya berarah. \omterm{thm:b1:second-law-entropy:secondlaw}{Hukum kedua
termodinamika} menamai besaran yang membedakan kedua arah itu:
entropi, yang hanya dapat diciptakan sebuah \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup}, tak pernah
dimusnahkan. Bab ini menyatakan hukumnya, menghitung entropi bagi sistem
dalam volume ini, memperlihatkan bagaimana ia mengukur ketakreversibelan, dan
memberinya, pada akhirnya, maknanya dalam hal pencacahan.

\section{Hukum kedua}

\begin{theorem}[Hukum kedua termodinamika]\label{thm:b1:second-law-entropy:secondlaw}
\index{hukum kedua termodinamika}\index{entropi}\index{entropi!yang tercipta}\index{entropi!yang dipertukarkan}
Setiap \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup} memiliki fungsi keadaan yang ekstensif, yaitu
\emph{entropi} $S$ (\unit{J/K}), sehingga dalam sebarang transformasi
\[
\Delta S = S_{\mathrm{tukar}} + S_{\mathrm{cipta}}, \qquad
S_{\mathrm{tukar}} = \int\frac{\delta Q}{T_{\mathrm{ext}}}, \qquad
S_{\mathrm{cipta}} \geq 0 ,
\]
dengan $\delta Q$ adalah kalor yang diterima dari lingkungannya pada
suhu $T_{\mathrm{ext}}$ tempat ia masuk, dan
$S_{\mathrm{cipta}} = 0$ jika dan hanya jika transformasinya
reversibel. Sistem terkucil ($\delta Q = 0$) hanya dapat melihat entropinya
tumbuh, dan kesetimbangannya adalah keadaan berentropi maksimum.
\end{theorem}
\admitted

\begin{remark}[Membaca hukumnya]\label{rem:b1:second-law-entropy:reading}
Tak seperti energi, entropi tidak kekal: ia \emph{dipertukarkan} bersama kalor
(dan hanya bersama kalor --- usaha tak membawa entropi) dan \emph{diciptakan} oleh
setiap proses tak reversibel: gesekan, kalor yang mengalir menyeberangi selisih
suhu, gas yang menyerbu masuk ke hampa, percampuran. Suku ciptaannya adalah
ukuran ketakreversibelan, dan sebuah transformasi hanya mungkin bila
suku itu tak negatif. ``Reversibel'' adalah idealisasi --- kuasistatik,
tanpa gesekan, kalor yang dipertukarkan menyeberangi selisih suhu yang lenyap --- yang
didekati proses nyata tetapi tak pernah dicapainya.
\end{remark}

\begin{proposition}[Identitas dasar; entropi gas ideal dan fase terkondensasi]\label{prop:b1:second-law-entropy:identity}
\index{identitas dasar termodinamika}\index{entropi!gas ideal}
Bagi \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup} yang komposisinya tetap, sepanjang sebarang \omterm{def:b1:point-kinematics:frame}{lintasan} reversibel,
$\delta Q_{\mathrm{rev}} = T\,\dd S$ dan $\delta W_{\mathrm{rev}} = -P\,\dd V$, sehingga
\[
\dd U = T\,\dd S - P\,\dd V ,
\]
yaitu hubungan antar fungsi keadaan yang berlaku bagi sebarang perubahan
yang sangat kecil. Akibatnya, bagi $n$ \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{def:b1:kinetic-theory:model}{gas ideal},
\[
S(T, V) = nC_{V,m}\ln T + nR\ln V + \text{tetap}, \qquad
S(T, P) = nC_{P,m}\ln T - nR\ln P + \text{tetap},
\]
dan bagi \omterm{prop:b1:kinetic-theory:condensed}{fase terkondensasi} berkapasitas kalor $C$, $S(T) = C\ln T + \text{tetap}$.
\end{proposition}

\begin{proof}
Reversibel: $T_{\mathrm{ext}} = T$ dan $S_{\mathrm{cipta}} = 0$ memberikan $\dd S =
\delta Q/T$; hukum pertamanya lalu memberikan identitasnya, yang tak lagi
mengacu pada sebuah \omterm{def:b1:point-kinematics:frame}{lintasan}. \omterm{def:b1:kinetic-theory:model}{Gas ideal}: $\dd S = (\dd U + P\,\dd V)/T = nC_{V,m}\dd T/T
+ nR\,\dd V/V$; integralkan; pakailah $PV = nRT$ untuk bentuk yang lain. \omterm{prop:b1:kinetic-theory:condensed}{Fase
terkondensasi}: $\dd V = 0$, $\dd S = C\,\dd T/T$.
\end{proof}

\begin{example}[Membaca rumusnya]\label{ex:b1:second-law-entropy:formulas}
\omterm{prop:b1:first-law:transformations}{Transformasi adiabatik} reversibel bersifat \emph{isentropik}: $S(T, V)$
yang tetap memberikan $T^{C_{V,m}}V^R$ tetap, yakni $TV^{\gamma - 1}$ tetap ---
\omterm{prop:b1:first-law:transformations}{hukum Laplace} sekali lagi. Melipatduakan volume satu \omterm{def:b1:kinetic-theory:scales}{mol} gas pada suhu
tetap menaikkan entropinya sebesar $R\ln2 = \qty{5.8}{J/K}$, baik dilakukan
secara reversibel (entropinya masuk sebagai kalor dari rendamannya) maupun melalui \omterm{rem:b1:first-law:irreversible}{ekspansi
Joule} ke dalam hampa (tanpa kalor: \qty{5.8}{J/K} yang sama lalu
\emph{diciptakan}). Fungsi keadaannya tak peduli bagaimana ia sampai ke sana;
neracanya peduli.
\end{example}

\section{Neraca entropi}

\begin{proposition}[Kalor yang dipertukarkan dengan termostat]\label{prop:b1:second-law-entropy:thermostat}
Sebuah sistem yang menerima kalor $Q$ dari termostat pada $T_0$ (apa pun yang
terjadi di dalamnya) mempunyai $S_{\mathrm{tukar}} = Q/T_0$, dan entropi termostatnya
sendiri berubah sebesar $-Q/T_0$: sebuah termostat mempertukarkan entropi secara reversibel.
\end{proposition}

\begin{proof}
$T_{\mathrm{ext}} = T_0$ tetap; keadaan dalam termostatnya
berubah secara sangat kecil, sehingga ciptaannya sendiri dapat diabaikan.
\end{proof}

\begin{example}[Benda panas di dalam danau yang dingin]\label{ex:b1:second-law-entropy:quench}
Sebuah balok besi ($m = \qty{1.0}{kg}$, $c = \qty{450}{J/(kg.K)}$) pada $T_1 =
\qty{373}{K}$ yang dijatuhkan ke dalam danau bersuhu $T_0 = \qty{293}{K}$: ia berakhir pada $T_0$,
dengan $\Delta S_{\text{balok}} = mc\ln(T_0/T_1) = \qty{-109}{J/K}$ dan $Q =
mc(T_0 - T_1) = \qty{-36}{kJ}$ diberikan kepada danaunya, sehingga $S_{\mathrm{tukar}} =
Q/T_0 = \qty{-123}{J/K}$ dan $S_{\mathrm{cipta}} = -109 + 123 = \qty{+14}{J/K}$.
Entropi baloknya turun, entropi danaunya naik lebih besar: prosesnya
tak reversibel, dan angkanya mengukur seberapa besar.
\end{example}

\begin{proposition}[Dua akibat]\label{prop:b1:second-law-entropy:clausiuskelvin}
\index{pernyataan Clausius}\index{pernyataan Kelvin}
\begin{enumerate}
  \item \emph{(Clausius)} Kalor tak mengalir secara spontan dari benda yang
        dingin ke benda yang lebih panas: kalor $Q > 0$ yang berpindah dari $T_h$ ke
        $T_c < T_h$ menciptakan $S = Q(1/T_c - 1/T_h) > 0$; kebalikannya akan
        memusnahkan entropi.
  \item \emph{(Kelvin)} Tak ada mesin bersiklus yang dapat menghasilkan usaha dari satu
        termostat saja: sepanjang satu siklus $\Delta S = 0 = Q/T_0 + S_{\mathrm{cipta}}$
        memaksa $Q \leq 0$, sehingga $W = -Q \geq 0$ --- mesinnya hanya dapat
        \emph{menerima} usaha dan membuang kalor.
\end{enumerate}
\end{proposition}

\begin{proof}
Keduanya adalah neracanya yang diterapkan pada sistem ``kedua termostatnya'' atau
``mesinnya sepanjang satu siklus'', dengan $S$ sebuah fungsi keadaan sehingga $\Delta S
= 0$ sepanjang satu siklus.
\end{proof}

\begin{definition}[Diagram entropi]\label{def:b1:second-law-entropy:TS}
\index{diagram entropi}
Pada bidang $(S, T)$ sebuah \omterm{def:b1:first-law:transformation}{transformasi reversibel} berupa kurva, dan
kalor yang diterima adalah luas di bawahnya: $Q_{\mathrm{rev}} = \int T\,\dd S$. Sebuah
isoterm reversibel mendatar, sebuah isentropik (adiabatik reversibel)
tegak; sebuah siklus reversibel melingkupi luas yang sama dengan kalor yang diterimanya
dan karenanya, menurut hukum pertama, dengan usaha yang disampaikannya
(\cref{ch:b1:heat-engines}).
\end{definition}

\begin{method}[Membuat neraca entropi]\label{met:b1:second-law-entropy:method}
\begin{enumerate}
  \item Hitunglah $\Delta S$ sistemnya dari fungsi keadaannya
        (\cref{prop:b1:second-law-entropy:identity}): keadaan awal dan
        akhirnya saja.
  \item Hitunglah $S_{\mathrm{tukar}}$ dari kalor yang benar-benar diterima dan
        suhu sumbernya: $\sum Q_i/T_i$ bagi termostat,
        $\int\delta Q/T_{\mathrm{ext}}$ selain itu.
  \item Simpulkan $S_{\mathrm{cipta}} = \Delta S - S_{\mathrm{tukar}}$; ia harus
        $\geq 0$ --- bila tidak, transformasinya mustahil (atau ada kekeliruan
        perhitungan); $= 0$ bagi yang reversibel.
\end{enumerate}
\end{method}

\begin{example}[Mencampur yang panas dan yang dingin]\label{ex:b1:second-law-entropy:mixing}
Dua massa air yang sama $m$ pada $T_1$ dan $T_2$ yang dicampur dalam termos berakhir pada
$(T_1 + T_2)/2$ (hukum pertama); $\Delta S = mc[\ln\frac{T_f}{T_1} + \ln\frac{T_f}{T_2}]
= mc\ln\frac{(T_1 + T_2)^2}{4T_1T_2} \geq 0$ (rata-rata hitung di atas rata-rata
ukur), tanpa pertukaran: semuanya tercipta. Untuk \qty{1}{kg} pada \qty{300}{K} dan
\qty{400}{K}: \qty{86}{J/K}. Soal akhir pekannya memperlihatkan bahwa dengan mencampurnya
secara \emph{reversibel} --- melalui sebuah mesin ideal --- kita justru akan
memperoleh usaha dan berakhir pada $\sqrt{T_1T_2}$.
\end{example}

\section{Apa yang dicacah entropi}

\begin{proposition}[Rumus Boltzmann]\label{prop:b1:second-law-entropy:boltzmann}
\index{rumus Boltzmann}\index{keadaan mikro}
Sebuah keadaan makroskopik (dengan $U$, $V$, $N$ tertentu) dapat diwujudkan oleh sejumlah
$\Omega$ konfigurasi mikroskopik (\emph{\omterm{prop:b1:second-law-entropy:boltzmann}{keadaan mikro}}). Entropinya adalah
\[
S = k_B\ln\Omega .
\]
Sistem terkucil berkembang menuju keadaan makroskopik yang mempunyai, jauh
melebihi yang lain, \omterm{prop:b1:second-law-entropy:boltzmann}{keadaan mikro} terbanyak: pertumbuhan entropi adalah perjalanan menuju
apa yang secara luar biasa paling mungkin.
\end{proposition}
\admitted

\begin{example}[Mengapa gasnya tak pernah kembali]\label{ex:b1:second-law-entropy:count}
$N$ molekul yang bebas berada di separuh mana pun sebuah kotak: $2^N$ susunan,
semuanya sama mungkin; hanya satu yang menempatkan semuanya di sebelah kiri. Membiarkan gas
yang terkurung di satu separuhnya memuai melipatgandakan $\Omega$ dengan $2^N$, sehingga $\Delta S =
Nk_B\ln2 = nR\ln2$ --- tepat hasil \omterm{rem:b1:first-law:irreversible}{ekspansi Joule} di atas, kini dengan
sebuah makna: molekulnya mempunyai $2^N$ kali lebih banyak cara untuk berada. Peluang
kembalinya secara spontan adalah $2^{-N}$: satu dalam seribu bagi sepuluh
molekul, satu dalam $10^{30}$ bagi seratus, dan bagi satu \omterm{def:b1:kinetic-theory:scales}{mol} sebuah bilangan dengan
$10^{23}$ angka nol. Hukum kedua bukanlah larangan; ia adalah peluang
yang begitu berat sebelah sehingga ia tak pernah sekali pun tertangkap gagal.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:1}
Perubahan entropi \qty{1.0}{kg} es yang mencair pada \qty{0}{\degreeCelsius}
($L_f = \qty{334}{kJ/kg}$), dan \qty{1.0}{kg} air yang mendidih pada
\qty{100}{\degreeCelsius} ($L_v = \qty{2260}{kJ/kg}$). Mengapa yang kedua jauh
lebih besar?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:2}
Perubahan entropi \qty{1.0}{kg} air yang dipanaskan dari \qty{20}{} ke
\qty{80}{\degreeCelsius}. Apakah ia bergantung pada cara pemanasannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:3}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{def:b1:kinetic-theory:model}{gas ideal} melipatduakan volumenya pada suhu tetap.
Berapa $\Delta S$; entropi yang dipertukarkan dan yang tercipta bila ekspansinya reversibel;
lalu bila ia berupa \omterm{rem:b1:first-law:irreversible}{ekspansi Joule} ke dalam hampa.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:second-law-entropy:4}
Tunjukkan dari $S(T, V)$ bahwa \omterm{prop:b1:first-law:transformations}{transformasi adiabatik} reversibel sebuah
\omterm{def:b1:kinetic-theory:model}{gas ideal} menuruti $TV^{\gamma - 1} =$ tetap. Berapa perubahan entropi dalam
pemampatan adiabatik yang \emph{tiba-tiba} (\cref{exo:b1:first-law:11})?
Hitunglah untuk angka-angka latihan itu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:5}
Sebuah balok besi \qty{1.0}{kg} ($c = \qty{450}{J/(kg.K)}$) pada \qty{100}{\degreeCelsius}
dijatuhkan ke dalam danau bersuhu \qty{20}{\degreeCelsius}. Perubahan entropi
baloknya, danaunya, dan semestanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:6}
Dua massa air \qty{1.0}{kg} pada \qty{300}{K} dan \qty{400}{K} dicampur
secara adiabatik. Suhu akhirnya, entropi yang tercipta. Tunjukkan secara umum bahwa
entropi yang tercipta $mc\ln[(T_1 + T_2)^2/4T_1T_2]$ tak negatif.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:7}
Kebocoran kalor \qty{1.0}{kJ} menyeberangi sebuah dinding dari ruangan bersuhu \qty{300}{K} ke
luar yang bersuhu \qty{280}{K}. Entropi yang tercipta. Kalor yang sama dari tungku
\qty{1000}{K} ke ruangannya: bandingkan, lalu berilah komentar tentang di mana ``kerugiannya''
paling besar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:8}
Buktikan \omterm{prop:b1:second-law-entropy:clausiuskelvin}{pernyataan Kelvin} dari neraca entropi sebuah mesin yang menjalankan
satu siklus dalam sentuhan dengan satu termostat. Mengapa mesin sebuah kapal tak
sekadar mengambil kalor dari lautnya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:second-law-entropy:9}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} nitrogen dan satu \omterm{def:b1:kinetic-theory:scales}{mol} oksigen, pada $T$ dan $P$ yang sama,
dipisahkan sebuah dinding di dalam kotak bervolume total $2V$, dibiarkan bercampur. Perubahan
entropinya (perlakukan tiap gas sebagai memuai ke seluruh kotaknya). Apakah ia tercipta
atau dipertukarkan? Bagaimana bila kedua sisinya berisi nitrogen?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:10}
Sebuah bohlam \qty{100}{W} menyala selama satu jam; seluruh energinya berakhir sebagai kalor di

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:11}

termostat antara pada $\sqrt{T_1T_0}$, (c) melalui kontinum

\begin{exercise}[$\star\star\star$]\label{exo:b1:second-law-entropy:12}

\section{Soal: Secangkir kopi dan anak panah waktu}

\begin{problem}[{Soal akhir pekan --- secangkir kopi yang mendingin, sebuah rendaman
yang tercampur, sebuah lemari es dan sekotak molekul: empat neraca entropi, dan
berapa nilai entropi yang tercipta itu seandainya diubah menjadi usaha}]
\label{pb:b1:second-law-entropy:1}
Air: $c = \qty{4180}{J/(kg.K)}$; ruangan (sebuah termostat): $T_0 = \qty{293}{K}$;
$R = \qty{8.314}{J/(mol.K)}$, $k_B = \qty{1.38e-23}{J/K}$.

\medskip
\noindent\textbf{Bagian I --- Cangkir yang mendingin.}
Secangkir kopi \qty{0.25}{kg} pada $T_1 = \qty{363}{K}$ mendingin sampai suhu
ruangannya.
\begin{enumerate}
  \item Kalor yang diberikan kepada ruangannya, dan perubahan entropi ruangannya.
  \item Perubahan entropi kopinya.
  \item Entropi yang tercipta; periksalah tandanya.
  \item Kopinya diminum ketika mencapai \qty{330}{K}: entropi yang tercipta sampai
        saat itu.
  \item Tunjukkan bahwa bagi sebarang $T_1 > T_0$, $S_{\mathrm{cipta}} = mc[T_1/T_0 - 1 -
        \ln(T_1/T_0)] \geq 0$ (selidikilah $f(x) = x - 1 - \ln x$).
  \item Kopinya mendingin lebih cepat bila diaduk atau ditiup: apakah itu mengubah
        entropi yang tercipta? Mengapa ya atau mengapa tidak?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Mencampur, dengan buruk dan dengan baik.}
Dua massa air \qty{1.0}{kg} pada $T_1 = \qty{400}{K}$ dan $T_2 =
\qty{300}{K}$, di dalam sebuah termos.
\begin{enumerate}[resume]
  \item Pencampuran langsung: suhu akhirnya $T_f$ dan entropi yang tercipta.
  \item Sebagai gantinya, sebuah mesin ideal mengambil kalor dari air panasnya, membuang
        kalor ke air dinginnya lalu menyampaikan usaha, sampai keduanya bersuhu
        sama $T_f'$; seluruh prosesnya reversibel. Tuliskan
        neraca entropinya lalu tunjukkan bahwa $T_f' = \sqrt{T_1T_2}$.
  \item Hitunglah $T_f'$ dan usaha yang disampaikannya (hukum pertama pada kedua
        airnya).
  \item Periksalah neraca entropi skema reversibelnya secara numerik:
        entropi yang hilang dari air panasnya sama dengan yang diperoleh air dinginnya.
  \item Bandingkan usahanya dengan $T_0 \times S_{\mathrm{cipta}}$ pertanyaan 6
        (ambillah ruangannya sebagai suhu acuan di sini, $T_0 \approx T_2$):
        apa yang diukur entropi yang tercipta pada pencampuran yang serampangan itu?
  \item Mengapa tak ada dapur yang memperoleh kembali usaha itu?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Lemari esnya.}
Sebuah lemari es menjaga bagian dalamnya pada $T_c = \qty{278}{K}$ di dalam ruangan bersuhu
$T_0$; kalor bocor masuk ke dalamnya sebesar \qty{40}{W}, dan mesinnya memompanya keluar kembali.
\begin{enumerate}[resume]
  \item Sepanjang satu sekon, tuliskan neraca entropi mesinnya (sebuah
        peranti bersiklus): entropi yang diterima dari bagian dalamnya yang dingin, yang diberikan ke
        ruangannya, dan yang tercipta di dalamnya.
  \item Simpulkan usaha minimum tiap sekon (yaitu \omterm{thm:b1:dc-circuits:kirchhoff}{daya listriknya}) yang diperlukan,
        dan koefisien kinerja maksimum yang bersesuaian
        $Q_c/W$.
  \item Nyatakan batas itu hanya dengan $T_c$ dan $T_0$, lalu hitunglah
        bagi sebuah pembeku pada $T_c = \qty{255}{K}$.
  \item Sebuah lemari es nyata seukuran ini menyedot kira-kira \qty{15}{W}: entropi
        yang tercipta tiap sekon.
  \item Bila dibiarkan terbuka, pintu lemari esnya membiarkan kalor masuk sebesar \qty{200}{W}:
        berapa daya minimumnya kini, dan mengapa lemari es yang terbuka menghangatkan ruangan.
  \item Apa yang akan dikatakan hukum kedua tentang sebuah lemari es yang sama sekali
        tak memerlukan listrik?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Mencacah.}
Sebuah kotak dibagi menjadi dua separuh yang sama oleh sebuah dinding; $N$ molekul
\omterm{def:b1:kinetic-theory:model}{gas ideal} berada di sebelah kiri. Dindingnya lalu disingkirkan.
\begin{enumerate}[resume]
  \item Banyaknya \omterm{prop:b1:second-law-entropy:boltzmann}{keadaan mikro} sebelum dan sesudahnya (tiap molekul boleh berada di
        separuh mana pun), dan $\Delta S$ menurut rumus Boltzmann.
  \item Periksalah terhadap $\Delta S = nR\ln2$ dari rumus termodinamika
        bagi \omterm{rem:b1:first-law:irreversible}{ekspansi Joule}.
  \item Bagi $N = 50$, berapa peluang menemukan semua molekulnya kembali di sebelah
        kiri pada saat yang acak; bagi satu \omterm{def:b1:kinetic-theory:scales}{mol}, berapa \omterm{def:b1:units-dimensions:oom}{orde besar}
        peluang itu sebagai pangkat sepuluh.
  \item Berapa banyak pertanyaan ya/tidak (bit) yang menetapkan separuh mana yang ditempati tiap
        molekul dari $N$ molekulnya? Bandingkan dengan $\Delta S/(k_B\ln2)$.
  \item Entropi yang dipertukarkan dan yang tercipta dalam \omterm{rem:b1:first-law:irreversible}{ekspansi Joule}; mengapa ia
        menjadi contoh baku ketakreversibelan?
  \item Gas yang sama dimampatkan kembali secara reversibel dan isotermal ke
        separuh kirinya: usaha yang diterimanya, kalor yang diberikan ke ruangannya, entropi yang
        dipertukarkan; apa yang harus dipasok untuk membatalkan ekspansi bebasnya?
  \item Ringkaslah dalam tiga baris: apa yang dikekalkan hukum kedua
        (tak ada), apa yang dilarangnya, dan berapa harga entropi yang tercipta.
\end{enumerate}
\end{problem}
