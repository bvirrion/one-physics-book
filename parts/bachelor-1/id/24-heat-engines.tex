\chapter{Mesin Kalor dan Mesin Termodinamika}\label{ch:b1:heat-engines}

Mesin mobil, pembangkit listrik, \omterm{def:b1:heat-engines:machine}{lemari es} dan \omterm{def:b1:heat-engines:machine}{pompa kalor} adalah
benda yang sama yang dilihat dari empat sisi: sebuah fluida yang dibawa mengelilingi
sebuah siklus, bergantian bersentuhan dengan sumber panas dan sumber dingin,
bertukar kalor dengan keduanya dan bertukar usaha dengan sebuah poros. Hukum
pertama membukukan energinya; hukum kedua, melalui neraca entropi sepanjang satu
siklus, menetapkan batas yang tak dapat dikalahkan kepandaian apa pun ---
teorema Carnot, yaitu ketaksamaan yang paling berpengaruh dalam keteknikan. Bab
ini menurunkan batas itu, menghitung siklus idealnya dan siklus nyata yang
mendekatinya (Otto, Diesel, Stirling), membalikkan mesinnya untuk membuat dingin
dan untuk memanaskan rumah, serta memperlihatkan bagaimana entropi yang tercipta
pada mesin nyata dibayar, joule demi joule, dengan \omterm{thm:b1:heat-engines:lostwork}{usaha yang hilang}.

\section{Mesin bersiklus dan kedua hukumnya}

\begin{definition}[Mesin termodinamika]\label{def:b1:heat-engines:machine}
\index{mesin kalor}\index{lemari es}\index{pompa kalor}\index{termostat}
Yang disebut \emph{mesin termodinamika} adalah \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup} (fluida kerjanya)
yang dibawa mengelilingi sebuah \emph{siklus}, sehingga ia kembali secara berkala ke keadaan
yang sama, sambil bertukar usaha $W$ dengan luarnya dan kalor $Q_i$ dengan
sumber-sumber bersuhu $T_i$ --- yaitu \emph{termostat}, yang suhunya
tak diubah oleh pertukaran itu. Sebuah mesin \emph{diterm} memakai dua
termostat, sumber panas pada $T_h$ dan sumber dingin pada $T_c < T_h$.
Ia disebut \emph{mesin} bila ia menyampaikan usaha ($W < 0$), sebuah
\emph{lemari es} bila ia digerakkan ($W > 0$) untuk mengambil kalor dari sumber
dinginnya, dan sebuah \emph{pompa kalor} bila ia digerakkan untuk memberi kalor kepada sumber panasnya.
Semua besarannya dihitung sepanjang satu siklus, secara aljabar, sebagai yang diterima
fluidanya.
\end{definition}

\begin{theorem}[Kedua hukumnya sepanjang satu siklus; ketaksamaan Clausius]\label{thm:b1:heat-engines:clausius}
\index{ketaksamaan Clausius}
Sepanjang satu siklus sebuah mesin diterm,
\[
W + Q_h + Q_c = 0 , \qquad
\frac{Q_h}{T_h} + \frac{Q_c}{T_c} = -S_{\mathrm{cipta}} \leq 0 ,
\]
dengan kesamaan pada hubungan yang kedua jika dan hanya jika siklusnya
reversibel. Secara lebih umum, bagi berapa pun banyaknya \omterm{def:b1:heat-engines:machine}{termostat},
$\sum_i Q_i/T_i \leq 0$ (\emph{ketaksamaan Clausius}).
\end{theorem}

\begin{proof}
$U$ dan $S$ adalah fungsi keadaan, sehingga $\Delta U = 0$ dan $\Delta S = 0$ sepanjang
satu siklus. Hukum pertamanya memberikan $0 = W + Q_h + Q_c$; neraca entropinya
(\cref{thm:b1:second-law-entropy:secondlaw}) memberikan $0 = S_{\mathrm{tukar}} +
S_{\mathrm{cipta}}$ dengan $S_{\mathrm{tukar}} = \sum Q_i/T_i$ karena tiap
\omterm{def:b1:heat-engines:machine}{termostatnya} bertukar pada suhunya sendiri yang tetap
(\cref{prop:b1:second-law-entropy:thermostat}).
\end{proof}

\begin{corollary}[Kelvin sekali lagi]\label{cor:b1:heat-engines:kelvin}
Sebuah mesin \emph{monoterm} ($Q_c = 0$) mempunyai $Q_h \leq 0$, sehingga $W \geq 0$:
tak ada siklus yang dapat mengubah kalor satu sumber saja menjadi usaha. Sebuah kapal
tak dapat berlayar dengan mendinginkan lautnya --- sebuah mesin memerlukan sumber dingin
untuk membuang kalor, dan hukum kedua mengatakan berapa banyak yang harus dibuangnya.
\end{corollary}

\section{Teorema Carnot}

\begin{definition}[Efisiensi dan koefisien kinerja]\label{def:b1:heat-engines:efficiency}
\index{efisiensi!mesin kalor}\index{koefisien kinerja}
Yang disebut \emph{efisiensi} sebuah mesin adalah usaha yang disampaikannya tiap satuan kalor
yang diambil dari sumber panasnya, $\eta = -W/Q_h$; yang disebut
\emph{koefisien kinerja} (COP) sebuah \omterm{def:b1:heat-engines:machine}{lemari es} adalah dingin yang
dihasilkannya tiap satuan usaha, $e_f = Q_c/W$; dan bagi \omterm{def:b1:heat-engines:machine}{pompa kalor} adalah kalor yang
disampaikannya tiap satuan usaha, $e_p = -Q_h/W$. Masing-masing adalah nisbah ``apa yang
dikehendaki'' terhadap ``apa yang dibayar''; sebuah efisiensi paling besar~1, sebuah COP dapat
melebihi~1.
\end{definition}

\begin{theorem}[Carnot]\label{thm:b1:heat-engines:carnot}
\index{teorema Carnot}\index{efisiensi Carnot}
Di antara \omterm{def:b1:heat-engines:machine}{termostat} $T_h > T_c$,
\[
\eta \leq \eta_C = 1 - \frac{T_c}{T_h}, \qquad
e_f \leq \frac{T_c}{T_h - T_c}, \qquad
e_p \leq \frac{T_h}{T_h - T_c} = 1 + \frac{T_c}{T_h - T_c},
\]
dengan kesamaan jika dan hanya jika siklusnya reversibel. Batasnya hanya bergantung
pada suhunya --- bukan pada fluidanya, bukan pada mekanismenya.
\end{theorem}

\begin{proof}
Bagi sebuah mesin ($Q_h > 0$): $\eta = -W/Q_h = (Q_h + Q_c)/Q_h = 1 + Q_c/Q_h$,
dan ketaksamaan Clausius memberikan $Q_c/Q_h \leq -T_c/T_h$. Bagi sebuah
\omterm{def:b1:heat-engines:machine}{lemari es} ($Q_c > 0$, $W > 0$): $W = -Q_h - Q_c$ dengan $-Q_h \geq Q_c T_h/T_c$,
sehingga $W \geq Q_c(T_h/T_c - 1)$ dan $e_f = Q_c/W \leq T_c/(T_h - T_c)$.
Bagi \omterm{def:b1:heat-engines:machine}{pompa kalornya}, $e_p = -Q_h/W = (W + Q_c)/W = 1 + e_f$. Kesamaannya adalah
kasus reversibel $S_{\mathrm{cipta}} = 0$ di seluruhnya.
\end{proof}

\begin{example}[Tiga angka]\label{ex:b1:heat-engines:numbers}
Sebuah turbin uap di antara $T_h = \qty{800}{K}$ dan sebuah sungai pada
$T_c = \qty{300}{K}$: $\eta_C = 0.625$; pembangkit nyata mencapai kira-kira $0.40$.
Sebuah \omterm{def:b1:heat-engines:machine}{lemari es} rumah tangga di antara $\qty{275}{K}$ di dalamnya dan sebuah dapur pada
$\qty{298}{K}$: $e_f \leq 275/23 = 12$; mesin nyata memberikan $2$ sampai $4$. Sebuah
\omterm{def:b1:heat-engines:machine}{pompa kalor} yang menghangatkan rumah pada $\qty{293}{K}$ dari udara pada $\qty{273}{K}$:
$e_p \leq 293/20 = 14.7$; pompa nyata memberikan $3$ sampai $5$ --- masih tiga sampai
lima kali yang akan diberikan listrik yang sama di dalam sebuah \omterm{def:b1:dc-circuits:resistor}{resistor}
(\cref{pb:b1:heat-engines:1}).
\end{example}

\begin{remark}[Mengapa efisiensinya tak dapat bernilai~1]\label{rem:b1:heat-engines:why}
Sebuah mesin menerima entropi $Q_h/T_h$ bersama kalor yang diambilnya; usaha
tak membawa apa pun keluar; satu-satunya pintu keluarnya adalah kalor yang dibuang ke sumber dinginnya,
yang membawa $|Q_c|/T_c$. Karena entropi tak dapat diciptakan negatif, sekurang-kurangnya
$Q_h T_c/T_h$ kalor harus dibuang, dan usahanya adalah sisanya.
Makin rendah sumber dinginnya dan makin tinggi sumber panasnya, makin baik --- dari sinilah
menara pendinginnya, dan perburuan sudu turbin yang makin panas.
\end{remark}

\section{Siklus Carnot}

\begin{definition}[Siklus Carnot]\label{def:b1:heat-engines:carnotcycle}
\index{siklus Carnot}
Yang disebut \emph{siklus Carnot} adalah siklus diterm yang reversibel: dua isoterm
reversibel, pada $T_h$ (dalam sentuhan dengan sumber panasnya) dan pada $T_c$ (dengan
sumber dinginnya), yang dihubungkan oleh dua kaki adiabatik reversibel (isentropik,
tanpa sumber yang bersentuhan). Ditempuh searah jarum jam pada bidang $(V, P)$ ia sebuah
mesin; berlawanan arah jarum jam, sebuah \omterm{def:b1:heat-engines:machine}{lemari es} atau \omterm{def:b1:heat-engines:machine}{pompa kalor}. Pada \omterm{def:b1:second-law-entropy:TS}{diagram
entropinya} ia berupa persegi panjang.
\end{definition}

\begin{proposition}[Siklus Carnot sebuah gas ideal]\label{prop:b1:heat-engines:carnotgas}
Bagi $n$ \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{def:b1:kinetic-theory:model}{gas ideal} yang dibawa mengelilingi $A \to B$ (isoterm $T_h$,
ekspansi), $B \to C$ (adiabat), $C \to D$ (isoterm $T_c$,
pemampatan), $D \to A$ (adiabat):
\[
Q_h = nRT_h\ln\frac{V_B}{V_A}, \qquad
Q_c = -nRT_c\ln\frac{V_B}{V_A}, \qquad
\eta = 1 - \frac{T_c}{T_h} ,
\]
yaitu nilai Carnot --- sebagaimana seharusnya bagi sebarang siklus diterm yang reversibel.
\end{proposition}

\begin{proof}
Pada sebuah isoterm $\Delta U = 0$, sehingga $Q = -W = nRT\ln(V_{\text{akhir}}/
V_{\text{awal}})$. Pada adiabatnya $TV^{\gamma-1}$ tetap:
$T_hV_B^{\gamma-1} = T_cV_C^{\gamma-1}$ dan $T_hV_A^{\gamma-1} =
T_cV_D^{\gamma-1}$, sehingga $V_C/V_D = V_B/V_A$ dan $Q_c = nRT_c\ln(V_D/V_C) =
-nRT_c\ln(V_B/V_A)$. Lalu $\eta = 1 + Q_c/Q_h = 1 - T_c/T_h$.
\end{proof}

\begin{remark}[Siklus Carnot adalah teorema, bukan mesin]\label{rem:b1:heat-engines:notengine}
Isoterm reversibel menuntut perpindahan kalor yang sangat lambat menyeberangi
selisih suhu yang lenyap: sebuah mesin Carnot menyampaikan usaha maksimalnya
pada daya nol. Mesin nyata menukar \omterm{def:b1:heat-engines:efficiency}{efisiensi} dengan daya dengan menerima
selisih suhu yang berhingga pada penukar kalornya
(\cref{exo:b1:heat-engines:10}); nilai Carnot tetap menjadi langit-langit
yang menjadi ukuran setiap rancangan.
\end{remark}

\section{Siklus mesin yang nyata}

\begin{proposition}[Siklus Otto]\label{prop:b1:heat-engines:otto}
\index{siklus Otto}\index{nisbah kompresi}
Mesin bensin yang diidealkan membawa udaranya ($\gamma$) mengelilingi: $1 \to 2$
pemampatan adiabatik dari $V_1$ ke $V_2$ (\emph{\omterm{prop:b1:heat-engines:otto}{nisbah kompresi}}
$r = V_1/V_2$); $2 \to 3$ pemanasan isokorik (yaitu pembakarannya); $3 \to 4$
ekspansi adiabatik kembali ke $V_1$; $4 \to 1$ pendinginan isokorik (yaitu
gas buangnya). \omterm{def:b1:heat-engines:efficiency}{Efisiensinya} adalah
\[
\eta_{\text{Otto}} = 1 - \frac{1}{r^{\gamma-1}} .
\]
\end{proposition}

\begin{proof}
Kalor dipertukarkan hanya pada isokornya: $Q_h = nC_{V,m}(T_3 - T_2)$ dan
$Q_c = nC_{V,m}(T_1 - T_4)$. Pada adiabatnya $TV^{\gamma-1}$ tetap:
$T_2 = T_1r^{\gamma-1}$ dan $T_3 = T_4r^{\gamma-1}$, sehingga $T_3 - T_2 =
(T_4 - T_1)r^{\gamma-1}$ dan
$\eta = 1 + Q_c/Q_h = 1 - (T_4 - T_1)/(T_3 - T_2) = 1 - r^{1-\gamma}$.
\end{proof}

\begin{example}[Angka-angka untuk mesin bensin]\label{ex:b1:heat-engines:ottonumbers}
$r = 9$, $\gamma = 1.4$: $9^{0.4} = 2.41$ dan $\eta = 0.58$. Mesin yang nyata
memberikan kira-kira $0.30$: pembakarannya tidak seketika, gasnya tidak
ideal pada \qty{2500}{K}, kalornya bocor ke dindingnya, gesekan dan pemompaan
mengambil bagiannya. Menaikkan $r$ membantu --- sampai campuran udara--bahan bakarnya
menyala sendiri di akhir pemampatannya (ketukan); mesin Diesel
mengubah cacat itu menjadi asasnya.
\end{example}

\begin{proposition}[Siklus Diesel]\label{prop:b1:heat-engines:diesel}
\index{siklus Diesel}
Gantilah pemanasan isokorik \omterm{prop:b1:heat-engines:otto}{siklus Otto} dengan pemanasan isobarik, dari
$V_2$ ke $V_3 = \rho V_2$ (\emph{nisbah pancung} $\rho > 1$): bahan bakar yang disuntikkan
ke dalam udara yang sudah panas oleh \omterm{prop:b1:heat-engines:otto}{nisbah kompresi} $r$ sebesar $15$ sampai $22$ terbakar
begitu ia tiba. Maka
\[
\eta_{\text{Diesel}} = 1 - \frac{1}{r^{\gamma-1}}\,\frac{\rho^{\gamma} - 1}{\gamma(\rho - 1)} ,
\]
yang lebih kecil daripada nilai Otto untuk $r$ yang sama tetapi dicapai dengan $r$ yang jauh
lebih besar, sehingga dalam praktiknya lebih besar ($0.40$ sampai $0.45$ bagi mesin besar).
\end{proposition}

\begin{proof}
$Q_h = nC_{P,m}(T_3 - T_2)$, $Q_c = nC_{V,m}(T_1 - T_4)$, sehingga $\eta = 1 -
(T_4 - T_1)/[\gamma(T_3 - T_2)]$. Dengan $T_2 = T_1r^{\gamma-1}$, $T_3 =
\rho T_2$ dan, pada adiabat $3 \to 4$ ($V_4 = V_1 = rV_2$), $T_4 =
T_3(V_3/V_4)^{\gamma-1} = \rho T_1 r^{\gamma-1}(\rho/r)^{\gamma-1} = T_1\rho^{\gamma}$:
$\eta = 1 - (\rho^\gamma - 1)/[\gamma r^{\gamma-1}(\rho - 1)]$.
\end{proof}

\begin{proposition}[Siklus Stirling dan regeneratornya]\label{prop:b1:heat-engines:stirling}
\index{siklus Stirling}\index{regenerator}
Dua isoterm ($T_c$, $T_h$) yang dihubungkan oleh dua isokor. Kalor pada
kaki isokoriknya, $\pm nC_{V,m}(T_h - T_c)$, sama besar dan berlawanan; bila ia
disimpan pada kaki pendinginannya di dalam sebuah \emph{regenerator} (sebuah massa berpori yang
dialiri gasnya) lalu dikembalikan pada kaki pemanasannya, hanya isotermnya
yang bertukar dengan sumbernya, dan $\eta = 1 - T_c/T_h$: mesin Stirling
adalah mesin Carnot dalam segala hal kecuali bentuk siklusnya. Tanpa
regeneratornya kalor isokoriknya harus datang dari sumber panasnya dan
$\eta = \dfrac{nR(T_h - T_c)\ln(V_1/V_2)}{nRT_h\ln(V_1/V_2) + nC_{V,m}(T_h - T_c)}$,
jauh lebih kecil (\cref{exo:b1:heat-engines:8}).
\end{proposition}

\begin{proof}
Pada isotermnya $Q_h = nRT_h\ln(V_1/V_2)$ dan $Q_c = -nRT_c\ln(V_1/V_2)$, sehingga
$-W = Q_h + Q_c + 0$ begitu kalor isokoriknya saling meniadakan; bagilah dengan kalor
yang benar-benar diambil dari sumber panasnya pada tiap kasusnya.
\end{proof}

\begin{remark}[Turbin uap dan turbin gas]\label{rem:b1:heat-engines:turbines}
Pembangkit listrik tak memakai gas di dalam tabung melainkan air yang dididihkan,
diekspansikan melalui sebuah turbin, diembunkan lalu dipompa kembali (siklus
Rankine, dengan dua perubahan wujud --- \cref{ch:b1:phase-changes}), atau udara
yang dimampatkan, dipanaskan oleh pembakaran lalu diekspansikan dalam sebuah turbin gas (siklus
Brayton, dua adiabat dan dua isobar). Analisisnya menuntut
neraca energi sebuah fluida yang mengalir, yang diberikan dalam volume Tahun ke-2;
batas Carnot dan logika bab ini berlaku tanpa perubahan.
\end{remark}

\section{Lemari es dan pompa kalor}

\begin{definition}[Siklus kompresi uap]\label{def:b1:heat-engines:vapourcycle}
\index{siklus kompresi uap}\index{refrigeran}
Hampir setiap \omterm{def:b1:heat-engines:machine}{lemari es}, pembeku, penyejuk udara dan \omterm{def:b1:heat-engines:machine}{pompa kalor} menjalankan
sebuah \emph{refrigeran} mengelilingi empat komponen: sebuah \emph{kompresor}
(menerima usahanya, menaikkan tekanan uapnya), sebuah \emph{kondensor}
(uap panasnya mencair pada tekanan tinggi, melepaskan
$|Q_h|$ ke sisi panasnya), sebuah \emph{katup ekspansi} (cairannya jatuh ke
tekanan rendah tanpa usaha maupun kalor, dan sebagian menguap seketika) dan sebuah
\emph{evaporator} (sisanya mendidih pada tekanan rendah, menyerap $Q_c$ dari
sisi dinginnya). Kalornya dibawa sebagai kalor laten penguapan, dan
kedua tekanannya menetapkan kedua suhu didihnya
(\cref{ch:b1:phase-changes}).
\end{definition}

\begin{proposition}[Harga selisih suhunya]\label{prop:b1:heat-engines:gaps}
Sebuah mesin reversibel yang bekerja di antara suhu dalamnya sendiri
$T_{\mathrm{ev}} < T_c$ dan $T_{\mathrm{cd}} > T_h$ mempunyai $e_p = T_{\mathrm{cd}}/(T_{\mathrm{cd}} - T_{\mathrm{ev}})$,

sebuah \omterm{def:b1:heat-engines:machine}{pompa kalor} karenanya turun ketika suhu luarnya
turun dan ketika suhu yang dituntut rangkaian pemanasnya naik.
\end{proposition}

\begin{proof}
Terapkan \cref{thm:b1:heat-engines:carnot} pada fluidanya, yang \omterm{def:b1:heat-engines:machine}{termostatnya}
sesungguhnya adalah suhu penukar kalornya; $T_{\mathrm{cd}} - T_{\mathrm{ev}} >
T_h - T_c$ menurunkan nisbahnya.
\end{proof}

\section{Entropi dan usaha yang hilang}

\begin{theorem}[Usaha yang hilang]\label{thm:b1:heat-engines:lostwork}
\index{usaha yang hilang}
Sebuah mesin diterm yang menciptakan entropi $S_{\mathrm{cipta}}$ tiap siklus
menyampaikan
\[
-W = Q_h\Bigl(1 - \frac{T_c}{T_h}\Bigr) - T_c\,S_{\mathrm{cipta}} ,
\]
dan sebuah \omterm{def:b1:heat-engines:machine}{pompa kalor} diterm yang menyampaikan $|Q_h|$ menghabiskan
$W = |Q_h|(1 - T_c/T_h) + T_c\,S_{\mathrm{cipta}}$: setiap joule tiap \omterm{def:b1:kinetic-theory:temperature}{kelvin}
yang tercipta berharga $T_c$ joule usaha, dengan suhu sumber dinginnya
menjadi kurs tukar antara entropi dan usaha.
\end{theorem}

\begin{proof}
Dari \cref{thm:b1:heat-engines:clausius}, $Q_c = -T_c(Q_h/T_h + S_{\mathrm{cipta}})$;
sisipkan ke dalam $-W = Q_h + Q_c$. Bagi pompanya, tuliskan kedua hubungan yang sama
dengan $Q_h < 0$.
\end{proof}

\begin{method}[Mengaudit sebuah mesin]\label{meth:b1:heat-engines:audit}
Ukurlah (atau hitunglah) kalor yang dipertukarkan dengan tiap sumbernya dan usahanya;
periksalah hukum pertamanya; hitunglah $S_{\mathrm{cipta}} = -\sum Q_i/T_i$ tiap siklus
atau tiap sekon; daya yang hilang adalah $T_c\dot S_{\mathrm{cipta}}$; carilah letak
penciptaannya (selisih suhu yang berhingga pada penukar kalornya, gesekan, pencekikan,
pemampatan yang tak kuasistatik) dengan menerapkan neraca entropinya pada tiap
komponen secara bergantian. Komponen yang tak menciptakan entropi tak layak
diperbaiki.
\end{method}

\begin{example}[Audit sebuah pembangkit listrik]\label{ex:b1:heat-engines:audit}
Sumber panas \qty{800}{K}, sungai \qty{300}{K}, $\qty{1.0}{GW}$ listrik
pada $\eta = 0.40$: $\dot Q_h = \qty{2.5}{GW}$, $\dot Q_c = \qty{1.5}{GW}$, sehingga
$\dot S_{\mathrm{cipta}} = 1.5/300 - 2.5/800 = \qty{1.9}{MW/K}$ dan daya
yang hilang adalah $300 \times 1.9 = \qty{0.56}{GW}$: tepat selisih antara
keluaran Carnot $0.625 \times 2.5 = \qty{1.56}{GW}$ dan \qty{1.0}{GW} yang nyata.
Sungainya menghangat oleh \qty{1.5}{GW} yang harus dibawanya pergi; pada musim panas, ketika
ia hangat dan dangkal, pembangkitnya menahan diri.
\end{example}

\label{pb:b1:heat-engines:1}
Sebuah rumah kehilangan kalor dengan laju $K(T_{\mathrm{dalam}} - T_{\mathrm{luar}})$ dengan
$K = \qty{250}{W/K}$; ia dijaga pada $T_{\mathrm{dalam}} = \qty{20}{\degreeCelsius}$.
Suhu luar rancangannya $\qty{-5}{\degreeCelsius}$; musim pemanasannya:
$200$ hari dengan suhu luar rata-rata \qty{7}{\degreeCelsius}.
Harga: listrik 0.25~\texteuro{} tiap \unit{kW h}, gas 0.10~\texteuro{} tiap
\unit{kW h}; $\qty{1}{kW h} = \qty{3.6}{MJ}$.

\medskip
\noindent\textbf{Bagian I --- Rumahnya dan cara-cara lama.}
\begin{enumerate}
  \item Daya pemanasan yang diperlukan pada suhu rancangannya.
  \item Energi yang diperlukan sepanjang musimnya, dalam joule dan dalam \unit{kW h}.
  \item Biaya semusim dengan \omterm{def:b1:dc-circuits:resistor}{resistor} listrik.
  \item Biayanya dengan ketel gas berefisiensi $0.90$.
  \item Listriknya datang dari pembangkit termal berefisiensi
        $0.40$. Berapa energi bahan bakar yang dihabiskan tiap musim oleh \omterm{def:b1:dc-circuits:resistor}{resistornya} dan oleh
        ketelnya; berilah komentar tentang ``pemanasan listrik itu boros secara
        termodinamika''.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:b1:heat-engines:machine}{Pompa kalor} yang ideal.}
\begin{enumerate}[resume]
  \item Definisikan $e_p$ lalu buktikan, dari kedua hukumnya sepanjang satu siklus, bahwa
        $e_p \leq T_{\mathrm{dalam}}/(T_{\mathrm{dalam}} - T_{\mathrm{luar}})$.
  \item Berapa $e_p$ idealnya pada suhu rancangannya, dan \omterm{thm:b1:dc-circuits:kirchhoff}{daya listrik}
        yang diperlukan ketika itu.
  \item Berapa $e_p$ idealnya pada suhu rata-rata musimnya, dan (dengan mengambil nilai
        itu bagi seluruh musimnya) berapa listrik semusimnya.
  \item Jelaskan secara fisis mengapa $e_p$ turun ketika suhu luarnya
        turun, dan mengapa itu perilaku terburuk yang mungkin bagi sebuah alat
        pemanas.
  \item Pada titik rancangannya pompa idealnya menyedot \qty{0.53}{kW} dan
        menyampaikan \qty{6.25}{kW}: dari mana datangnya \qty{5.7}{kW} yang lain,
        dan mengapa mendinginkan udara luar yang dingin untuk memanaskan rumah yang hangat
        tak melanggar hukum kedua?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Mesin yang nyata.}
Pompanya menjalankan \omterm{def:b1:heat-engines:vapourcycle}{siklus kompresi uap}. \omterm{def:b1:heat-engines:vapourcycle}{Refrigerannya} menguap pada
$T_{\mathrm{ev}} = T_{\mathrm{luar}} - \qty{8}{K}$ dan mengembun pada
$T_{\mathrm{cd}} = \qty{313}{K}$ (air rangkaian pemanas lantainya mengalir pada
\qty{35}{\degreeCelsius}); mesinnya mencapai $0.55$ dari nilai Carnot
yang dihitung di antara $T_{\mathrm{ev}}$ dan $T_{\mathrm{cd}}$. Kalor laten
\omterm{def:b1:heat-engines:vapourcycle}{refrigerannya}: $L = \qty{200}{kJ/kg}$.
\begin{enumerate}[resume]
  \item Sebutkan keempat komponen siklusnya, katakan pada komponen mana fluidanya
        menerima $Q_c$, memberikan $|Q_h|$ dan menerima $W$, serta mengapa
        evaporatornya harus lebih dingin daripada udara luarnya dan
        kondensornya lebih panas daripada rumahnya.
  \item Pada suhu rancangannya: berapa $T_{\mathrm{ev}}$, dan berapa $e_p$ Carnot
        di antara $T_{\mathrm{ev}}$ dan $T_{\mathrm{cd}}$.
  \item Berapa $e_p$ nyatanya, \omterm{thm:b1:dc-circuits:kirchhoff}{daya listrik} yang disedotnya, dan kalor yang diambil dari
        udara luarnya, pada suhu rancangannya.
  \item Laju aliran massa \omterm{def:b1:heat-engines:vapourcycle}{refrigeran} melalui evaporatornya.
  \item Pertanyaan yang sama (berapa $e_p$ nyatanya, berapa \omterm{thm:b1:dc-circuits:kirchhoff}{daya listriknya}) pada suhu
        rata-rata \qty{7}{\degreeCelsius}, tempat rumahnya kehilangan
        \qty{3.25}{kW}.
  \item Berikan $e_p$ sebagai fungsi $T_{\mathrm{luar}}$; berapa listrik semusim
        dan biayanya, dengan mengambil $e_p$ pada suhu rata-ratanya bagi seluruh musim.
        Bandingkan dengan Bagian~I.
  \item Berapa energi bahan bakar yang dihabiskan di pembangkitnya bagi musim \omterm{def:b1:heat-engines:machine}{pompa
        kalornya}; bandingkan dengan \omterm{def:b1:dc-circuits:resistor}{resistornya} dan ketelnya.
  \item Sebuah gelombang dingin pada \qty{-15}{\degreeCelsius}: berapa tuntutan kalornya, berapa $e_p$ nyatanya,
        dan berapa \omterm{thm:b1:dc-circuits:kirchhoff}{daya listriknya} seandainya pompanya dapat menyampaikannya. Kemampuan kompresornya
        sesungguhnya turun bersama $T_{\mathrm{ev}}$ dan pompanya hanya dapat
        menyampaikan \qty{3.8}{kW} di sana: berapa daya \omterm{def:b1:dc-circuits:resistor}{resistor} cadangan yang diperlukan, dan
        apa makna ``menakar ukuran sebuah \omterm{def:b1:heat-engines:machine}{pompa kalor}''.
  \item Seandainya rumahnya mempunyai radiator lama yang menuntut air pada
        \qty{60}{\degreeCelsius} ($T_{\mathrm{cd}} = \qty{338}{K}$): berapa $e_p$ nyatanya
        dan \omterm{thm:b1:dc-circuits:kirchhoff}{daya listriknya} pada suhu rancangannya; simpulkan tentang
        pilihan pemancar kalornya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Ke mana listriknya pergi.}
\begin{enumerate}[resume]
  \item Tunjukkan bahwa bagi \omterm{def:b1:heat-engines:machine}{pompa kalor} yang menyampaikan $|Q_h|$ ke rumahnya pada
        $T_{\mathrm{dalam}}$ dari udara luar pada $T_{\mathrm{luar}}$,
        $W = |Q_h|(1 - T_{\mathrm{luar}}/T_{\mathrm{dalam}}) + T_{\mathrm{luar}}S_{\mathrm{cipta}}$.
  \item Pada titik rancangannya, berapa entropi yang tercipta tiap sekon oleh perpindahan
        kalor menyeberangi kondensornya (dari \qty{313}{K} ke rumahnya) dan
        menyeberangi evaporatornya (dari udara luarnya ke \qty{260}{K}).
  \item Berapa \omterm{thm:b1:dc-circuits:kirchhoff}{daya listrik} yang dihabiskan kedua selisih suhu itu; berapa pecahannya dari
        \qty{1.92}{kW} yang disedot.
  \item Berapa entropi total yang tercipta tiap sekon oleh mesin nyatanya (bandingkan
        \qty{1.92}{kW}-nya dengan \qty{0.53}{kW} yang ideal); berapa bagian
        penukar kalornya; sebutkan sumber yang lain.
  \item Separuhkan kedua selisihnya (\qty{4}{K} dan \qty{10}{K}): berapa entropi yang tercipta
        pada penukar kalornya dan berapa \omterm{thm:b1:dc-circuits:kirchhoff}{daya listrik} yang dihemat; apa yang harus dibayar
        perancangnya untuk menyeparuhkan sebuah selisih?
  \item Ringkaslah dalam tiga angka: berapa kilowatt-jam kalor tiap
        kilowatt-jam listrik sepanjang musimnya; biaya semusimnya
        dibandingkan dengan gas; dan, dengan listrik sebesar \qty{60}{g} CO$_2$ tiap
        \unit{kW h} dan gas sebesar \qty{200}{g}, berapa karbon yang dipancarkan tiap
        jalurnya.
\end{enumerate}
\end{problem}
