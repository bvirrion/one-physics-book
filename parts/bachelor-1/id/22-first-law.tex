\chapter{Hukum Pertama Termodinamika}\label{ch:b1:first-law}

Pompalah ban sepeda dan tabung pompanya menjadi terlalu panas untuk dipegang.
Sebuah mesin diesel tak berbusi: ia memampatkan udaranya sampai dua puluh kali
lebih kecil, dan udaranya menjadi cukup panas untuk menyalakan bahan bakarnya sendiri. Sebuah
tabung gas mampat berselimut embun beku ketika ia mengosong. Pada tiap kasusnya
energinya berganti rupa --- usaha menjadi \omterm{def:b1:first-law:heat}{kalor}, \omterm{def:b1:first-law:heat}{kalor} menjadi usaha, keduanya
menjadi \omterm{def:b1:kinetic-theory:U}{energi dalam} --- dan \omterm{thm:b1:first-law:firstlaw}{hukum pertama termodinamika} adalah
pembukuan yang membuat neracanya seimbang. Bab ini mendefinisikan usaha
dan \omterm{def:b1:first-law:heat}{kalor} bagi sebuah \omterm{def:b1:first-law:transformation}{sistem termodinamika}, menyatakan hukum pertamanya, memperkenalkan
\omterm{def:b1:kinetic-theory:U}{energi dalam} dan \omterm{def:b1:first-law:enthalpy}{entalpi} sebagai besaran yang dikekalkannya, lalu menerapkannya
pada \omterm{def:b1:first-law:transformation}{transformasi} \omterm{def:b1:kinetic-theory:model}{gas ideal} dan pada kalorimetri.

\section{Transformasi, usaha, dan kalor}

\begin{definition}[Sistem, transformasi]\label{def:b1:first-law:transformation}
\index{sistem termodinamika}\index{transformasi}\index{transformasi kuasistatik}\index{transformasi reversibel}
Yang disebut \emph{sistem termodinamika} adalah materi di dalam batas yang dipilih;
selebihnya adalah \emph{lingkungannya}. Sebuah \emph{transformasi} membawanya
dari satu keadaan setimbang ke keadaan setimbang yang lain. Ia disebut \emph{kuasistatik} bila
sistemnya melewati rentetan keadaan setimbang (cukup lambat
agar $P$, $T$ terdefinisi di seluruhnya), dan \emph{reversibel} bila, di
samping itu, membalikkan syarat luarnya membalikkan \omterm{def:b1:point-kinematics:frame}{lintasannya} (tanpa
gesekan, tanpa selisih suhu atau tekanan yang berhingga dengan lingkungannya).
Namanya menurut apa yang tetap: \emph{isotermal} ($T$), \emph{isobarik}
($P$), \emph{isokorik} ($V$); \emph{adiabatik} bila tak ada \omterm{def:b1:first-law:heat}{kalor} yang dipertukarkan;
\emph{monobarik} (\emph{monotermal}) bila tekanan (suhu) luarnya
tetap, apa pun yang terjadi di dalamnya.
\end{definition}

\begin{proposition}[Usaha gaya tekanan]\label{prop:b1:first-law:work}
\index{usaha!gaya tekanan}
Sebuah sistem yang volumenya berubah sebesar $\dd V$ melawan tekanan luar
$P_{\mathrm{ext}}$ menerima usaha
\[
\delta W = -P_{\mathrm{ext}}\,\dd V, \qquad
W = -\int_{V_1}^{V_2}P_{\mathrm{ext}}\,\dd V .
\]
Bagi \omterm{def:b1:first-law:transformation}{transformasi kuasistatik} $P_{\mathrm{ext}} = P$ (tekanan sistemnya
sendiri) dan $W = -\int P\,\dd V$ adalah minus luas di bawah \omterm{def:b1:point-kinematics:frame}{lintasannya} pada
diagram Clapeyron: positif ketika dimampatkan, negatif ketika memuai.
\end{proposition}

\begin{proof}
Sebuah torak seluas $S$ yang didorong lingkungannya dengan gaya
$P_{\mathrm{ext}}S$ bergerak sejauh $\dd x$ ke luar: lingkungannya melakukan usaha
$-P_{\mathrm{ext}}S\,\dd x = -P_{\mathrm{ext}}\,\dd V$ pada sistemnya. Sebarang
batas adalah kumpulan torak semacam itu. Kuasistatik: toraknya
setimbang, $P_{\mathrm{ext}} = P$ sampai selisih yang lenyap.
\end{proof}

\begin{definition}[Kalor]\label{def:b1:first-law:heat}
\index{kalor}\index{termostat}
Yang disebut \emph{kalor} $Q$ adalah energi yang dipindahkan ke sistemnya selain melalui
usaha makroskopik --- melalui tumbukan molekul pada sebuah dinding (konduksi),
melalui peredaran sebuah fluida (konveksi), melalui radiasi. Sebuah
\emph{termostat} adalah benda yang begitu besar sehingga suhunya tak berubah
berapa pun kalor yang dipertukarkannya. Perjanjian tanda di seluruh buku: $W$ dan $Q$
\emph{diterima} oleh sistemnya, positif ketika keduanya masuk.
\end{definition}

\section{Hukum pertama}

\begin{theorem}[Hukum pertama termodinamika]\label{thm:b1:first-law:firstlaw}
\index{hukum pertama termodinamika}\index{energi dalam}
Sebuah \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup} memiliki fungsi keadaan, yaitu \emph{\omterm{def:b1:kinetic-theory:U}{energi dalamnya}}
$U$ (ekstensif), sehingga bagi sebarang \omterm{def:b1:first-law:transformation}{transformasi}
\[
\Delta U + \Delta E_{k} = W + Q ,
\]
dengan $E_k$ \omterm{thm:b1:work-and-energy:ke}{energi kinetik} makroskopiknya (nol bagi sistem yang diam):
energi yang diterima sebagai usaha dan \omterm{def:b1:first-law:heat}{kalor} tersimpan sebagai \omterm{def:b1:kinetic-theory:U}{energi dalam}. $W$
dan $Q$ masing-masing bergantung pada \omterm{def:b1:point-kinematics:frame}{lintasannya}; jumlahnya tidak.
\end{theorem}
\admitted

\begin{remark}[Apa yang dikatakannya]\label{rem:b1:first-law:meaning}
Ia adalah kekekalan energi, dengan \omterm{def:b1:first-law:heat}{kalor} diakui sebagai perpindahan
energi (percobaan roda dayung Joule: usaha yang terukur selalu menaikkan
suhu air sebanyak \omterm{def:b1:first-law:heat}{kalor} yang terukur). ``$U$ adalah fungsi
keadaan'' berarti $\Delta U$ ditetapkan oleh keadaan awal dan akhirnya ---
sehingga sebuah siklus mempunyai $\Delta U = 0$ dan $W + Q = 0$: mesin yang menyampaikan
usaha harus menerima \omterm{def:b1:first-law:heat}{kalor}. Isi mikroskopik $U$ adalah isi
\cref{def:b1:kinetic-theory:U}.
\end{remark}

\begin{definition}[Entalpi; kapasitas kalor]\label{def:b1:first-law:enthalpy}
\index{entalpi}\index{kapasitas kalor pada tekanan tetap}
Yang disebut \emph{entalpi} $H = U + PV$ adalah fungsi keadaan yang cocok bagi
\omterm{def:b1:first-law:transformation}{transformasi} pada tekanan tetap. Yang disebut \emph{\omterm{prop:b1:kinetic-theory:Ugas}{kapasitas kalor pada
volume tetap} dan pada tekanan tetap} adalah
\[
C_V = \left(\frac{\partial U}{\partial T}\right)_V, \qquad
C_P = \left(\frac{\partial H}{\partial T}\right)_P ,
\]
dalam \unit{J/K}; tiap \omterm{def:b1:kinetic-theory:scales}{mol} $C_{V,m}$, $C_{P,m}$; tiap kilogram \omterm{def:b1:first-law:heat}{kalor}
jenisnya $c_V$, $c_P$.
\end{definition}

\begin{proposition}[Kalor pada volume tetap dan pada tekanan tetap]\label{prop:b1:first-law:qvqp}
Bagi sistem yang diam yang bertukar usaha hanya melalui gaya tekanan:
\begin{itemize}
  \item \omterm{def:b1:first-law:transformation}{transformasi} isokorik: $W = 0$ dan $Q_V = \Delta U$;
  \item \omterm{def:b1:first-law:transformation}{transformasi} monobarik di antara dua keadaan pada tekanan luar
        $P_0$: $Q_P = \Delta H$.
\end{itemize}
\end{proposition}

\begin{proof}
Isokorik: $\dd V = 0$. Monobarik: $W = -P_0(V_2 - V_1)$, sehingga $Q = \Delta U +
P_0\Delta V = \Delta(U + PV)$ karena $P_1 = P_2 = P_0$.
\end{proof}

\begin{proposition}[Gas ideal: $U$, $H$, hubungan Mayer]\label{prop:b1:first-law:gas}
\index{hubungan Mayer}\index{indeks adiabatik}\index{hukum Joule}
Bagi \omterm{def:b1:kinetic-theory:model}{gas ideal}, $U$ dan $H$ hanya bergantung pada $T$ (\omterm{prop:b1:kinetic-theory:Ugas}{hukum Joule}):
$\dd U = C_V\,\dd T$, $\dd H = C_P\,\dd T$, dan
\[
C_{P,m} - C_{V,m} = R, \qquad \gamma = \frac{C_P}{C_V}, \qquad
C_{V,m} = \frac{R}{\gamma - 1}, \quad C_{P,m} = \frac{\gamma R}{\gamma - 1} ;
\]
$\gamma = 5/3$ bagi \omterm{prop:b1:kinetic-theory:Ugas}{gas monoatomik}, $7/5$ bagi \omterm{prop:b1:kinetic-theory:Ugas}{gas diatomik} pada suhu
biasa. Bagi \omterm{prop:b1:kinetic-theory:condensed}{fase terkondensasi} dalam model
\cref{prop:b1:kinetic-theory:condensed}, $C_P \approx C_V = C$ dan
$\dd U \approx \dd H = C\,\dd T$.
\end{proposition}

\begin{proof}
$H = U + nRT$; turunkan terhadap $T$. $C_{V,m} = \tfrac32 R$ atau
$\tfrac52 R$ dari \cref{prop:b1:kinetic-theory:Ugas}.
\end{proof}

\section{Transformasi gas ideal}

\begin{proposition}[Keempat transformasi baku]\label{prop:b1:first-law:transformations}
\index{transformasi isotermal}\index{transformasi adiabatik}\index{hukum Laplace}
Bagi $n$ \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{def:b1:kinetic-theory:model}{gas ideal}:
\begin{itemize}
  \item isokorik: $W = 0$, $Q = \Delta U = nC_{V,m}\Delta T$;
  \item isobarik (kuasistatik): $W = -P\Delta V = -nR\Delta T$, $Q = \Delta H =
        nC_{P,m}\Delta T$;
  \item isotermal reversibel: $\Delta U = 0$, $W = -Q = -nRT\ln(V_2/V_1) =
        nRT\ln(P_2/P_1)$;
  \item adiabatik reversibel: $Q = 0$, $W = \Delta U = nC_{V,m}(T_2 - T_1)$,
        dan sepanjang \omterm{def:b1:point-kinematics:frame}{lintasannya} berlaku \emph{\omterm{prop:b1:first-law:transformations}{hukum Laplace}}
        \[
        PV^\gamma = \text{tetap}, \qquad TV^{\gamma - 1} = \text{tetap}, \qquad
        T^\gamma P^{1 - \gamma} = \text{tetap} .
        \]
\end{itemize}
\end{proposition}

\begin{proof}
Isokorik dan isobarik: proposisi sebelumnya. Isotermal: $W =
-\int nRT\,\dd V/V$. Adiabatik: $\dd U = \delta W$, yakni $nC_{V,m}\dd T =
-P\,\dd V = -nRT\,\dd V/V$, sehingga $\dd T/T = -(\gamma - 1)\dd V/V$ (memakai
$R/C_{V,m} = \gamma - 1$); integralkan: $TV^{\gamma - 1}$ tetap; bentuk yang
lain dari $PV = nRT$.
\end{proof}

\begin{example}[Pompa dan diesel]\label{ex:b1:first-law:diesel}
Udara ($\gamma = 1.4$) yang dimampatkan secara adiabatik dari \qty{1}{bar}, \qty{300}{K}
ke \qty{7}{bar} (sebuah pompa sepeda): $T_2 = 300 \times 7^{0.286} = \qty{523}{K}$
--- \qty{250}{\degreeCelsius} untuk sesaat, dan itulah yang dirasakan tabungnya. Nisbah
kompresi sebuah diesel $20$: $T_2 = 300 \times 20^{0.4} = \qty{994}{K}$,
$P_2 = 20^{1.4} = \qty{66}{bar}$ --- di atas suhu nyala sendiri
bahan bakarnya: mesinnya tak memerlukan busi (soal akhir pekannya menjalankan
seluruh siklusnya).
\end{example}

\begin{remark}[Transformasi tak reversibel]\label{rem:b1:first-law:irreversible}
\index{ekspansi Joule}\index{transformasi tak reversibel}
Ketika \omterm{def:b1:first-law:transformation}{transformasinya} tidak kuasistatik, \omterm{prop:b1:first-law:transformations}{hukum Laplace} dan $W =
-\int P\,\dd V$ tak berlaku; yang berlaku hanya $W = -\int P_{\mathrm{ext}}\,\dd V$ dan
hukum pertamanya. Dua kasus klasik: gas yang memuai ke dalam hampa (\omterm{rem:b1:first-law:irreversible}{ekspansi
Joule}) tak menerima usaha dan, dalam dinding adiabatik, tak menerima \omterm{def:b1:first-law:heat}{kalor} --- $\Delta U
= 0$, dan \omterm{def:b1:kinetic-theory:model}{gas ideal} menjaga suhunya; gas yang tiba-tiba dimampatkan
oleh tekanan luar yang melonjak ke $P_2$ menerima $W = -P_2\Delta V$, lebih besar
daripada usaha reversibel ke tekanan yang sama, dan berakhir lebih panas
(\cref{exo:b1:first-law:11}).
\end{remark}

\section{Kalorimetri}

\begin{method}[Neraca kalorimetrik]\label{met:b1:first-law:calorimetry}
\index{kalorimetri}\index{kalor laten}
Benda-benda yang disentuhkan secara termal di dalam bejana terisolasi pada tekanan
tetap (atmosfer) bertukar \omterm{def:b1:first-law:heat}{kalor} hanya di antara mereka sendiri: totalnya
$\Delta H = \sum Q_P = 0$. Bagi tiap benda $\Delta H = mc\,\Delta T$ (tanpa perubahan
wujud) atau $\pm mL$ bagi perubahan wujud pada suhu peralihannya,
dengan $L$ adalah \emph{\omterm{def:b1:first-law:heat}{kalor} laten}nya (\omterm{def:b1:first-law:enthalpy}{entalpi} peleburan atau penguapan tiap
satuan massa: air, $L_f = \qty{334}{kJ/kg}$ pada \qty{0}{\degreeCelsius}, $L_v =
\qty{2260}{kJ/kg}$ pada \qty{100}{\degreeCelsius}). Tuliskan neracanya, selesaikan
untuk suhu akhirnya, lalu periksa bahwa wujud yang diandaikan bagi tiap benda
konsisten dengannya.
\end{method}

\begin{example}[Es di dalam air]\label{ex:b1:first-law:ice}
\qty{50}{g} es pada \qty{0}{\degreeCelsius} ke dalam \qty{200}{g} air pada
\qty{30}{\degreeCelsius}: peleburannya menuntut \qty{16.7}{kJ}; airnya dapat memberikan
\qty{25.1}{kJ} ketika mendingin ke \qty{0}{\degreeCelsius}, sehingga seluruh esnya mencair
dan neraca $0.250 \times 4180 \times (T_f - 0) = 25.1 - 16.7$ kJ memberikan
$T_f = \qty{8}{\degreeCelsius}$. Seandainya esnya \qty{100}{g},
\qty{33.4}{kJ} akan dituntut: hanya sebagian yang mencair dan $T_f = \qty{0}{\degreeCelsius}$
--- pemeriksaan itu berarti.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:first-law:1}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} udara ($C_{P,m} = \qty{29.1}{J/(mol.K)}$) dipanaskan dari
\qty{300}{K} ke \qty{400}{K} pada tekanan tetap. \omterm{def:b1:first-law:heat}{Kalor} yang diterimanya, usaha
yang diterimanya, perubahan \omterm{def:b1:kinetic-theory:U}{energi dalamnya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:2}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} gas pada \qty{300}{K} dimampatkan secara isotermal dan
reversibel dari \qty{2.0}{L} ke \qty{0.50}{L}. Usaha yang diterimanya, \omterm{def:b1:first-law:heat}{kalor}
yang dipertukarkan, $\Delta U$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:3}
Udara pada \qty{300}{K}, \qty{1}{bar} dimampatkan secara adiabatik dan
reversibel sampai sepersepuluh volumenya. Suhu dan tekanan akhirnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:first-law:4}
Berikan $C_{V,m}$, $C_{P,m}$ dan $\gamma$ untuk argon dan untuk nitrogen. Untuk
air cair ($c = \qty{4180}{J/(kg.K)}$), mengapa pembedaan antara
$c_P$ dan $c_V$ ditiadakan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:5}
\qty{200}{g} air pada \qty{80}{\degreeCelsius} dicampur dengan \qty{300}{g}
pada \qty{20}{\degreeCelsius} di dalam cangkir terisolasi. Suhu akhirnya; apa
yang berubah bila cangkirnya sendiri berkapasitas \omterm{def:b1:first-law:heat}{kalor} \qty{80}{J/K} dan bermula
pada \qty{20}{\degreeCelsius}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:6}
\qty{50}{g} es pada \qty{-10}{\degreeCelsius} ($c_{\text{es}} = \qty{2100}{J/(kg.K)}$)
dijatuhkan ke dalam \qty{200}{g} air pada \qty{30}{\degreeCelsius}. Wujud dan suhu
akhirnya. Pertanyaan yang sama dengan \qty{150}{g} es.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:7}
Sebuah gas menuruti siklus persegi panjang pada diagram Clapeyron: $(V_1, P_1)
\to (V_2, P_1) \to (V_2, P_2) \to (V_1, P_2) \to (V_1, P_1)$ dengan $V_2 > V_1$,
$P_2 > P_1$. Usaha yang diterimanya sepanjang siklusnya, sebagai fungsi luasnya;
tandanya, dan makna tanda itu.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:8}
Sebuah pompa sepeda memampatkan udara dari \qty{1}{bar}, \qty{300}{K} ke
\qty{7}{bar} cukup cepat sehingga adiabatik. Suhu yang dicapainya; mengapa
pompanya menghangat; dan mengapa tekanan bannya turun sedikit setelah
pemompaannya berhenti.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:first-law:9}
Sebuah \omterm{def:b1:kinetic-theory:model}{gas ideal} memuai ke dalam bejana hampa yang terisolasi (\omterm{rem:b1:first-law:irreversible}{ekspansi
Joule}). Usaha, \omterm{def:b1:first-law:heat}{kalor}, $\Delta U$, $\Delta T$. Apakah \omterm{def:b1:first-law:transformation}{transformasinya}
reversibel? Apa yang akan dilakukan gas nyata?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:10}
Mendidihkan \qty{1.0}{kg} air pada \qty{100}{\degreeCelsius} dan \qty{1}{atm}
(volume uapnya \qty{1.67}{m^3}). \omterm{def:b1:first-law:heat}{Kalor} yang diberikan, usaha yang dipertukarkan dengan
atmosfernya, $\Delta U$. Ke mana sebagian besar \omterm{def:b1:first-law:heat}{kalor} latennya pergi?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:11}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} \omterm{prop:b1:kinetic-theory:Ugas}{gas diatomik} pada \qty{300}{K}, \qty{1}{bar}, di dalam tabung
terisolasi, dimampatkan dengan tiba-tiba menetapkan tekanan luarnya ke
\qty{5}{bar} lalu menunggu kesetimbangannya. Suhu dan volume akhirnya
(hukum pertama dengan $W = -P_{\mathrm{ext}}\Delta V$); bandingkan dengan pemampatan
adiabatik reversibel ke \qty{5}{bar}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:first-law:12}
Bunyi adalah pemampatan yang cepat, sehingga adiabatik: lajunya $c = \sqrt{\gamma P/\rho}$
(nilai isotermal $\sqrt{P/\rho}$ adalah kekeliruan Newton). Hitunglah keduanya
untuk udara pada \qty{20}{\degreeCelsius} lalu bandingkan dengan hasil ukur
\qty{343}{m/s}. Tunjukkan bahwa $c = \sqrt{\gamma RT/M}$ lalu bandingkan dengan \omterm{def:b1:kinetic-theory:kinetictemp}{kelajuan
rms} \cref{ch:b1:kinetic-theory}.
\end{exercise}

\section{Soal: Dari pompa sepeda ke mesin diesel}

\begin{problem}[{Soal akhir pekan --- sebuah pompa tangan yang membakar jari,
sebuah tabung yang menyalakan bahan bakarnya sendiri, dan sebuah bola baja yang memantul di atas
bantalan udara: hukum pertama dalam tiga pemampatan}]
\label{pb:b1:first-law:1}
Udara: \omterm{prop:b1:kinetic-theory:Ugas}{gas diatomik} ideal, $\gamma = 1.40$, $C_{V,m} = \tfrac52 R$, $C_{P,m} =
\tfrac72 R$, $M = \qty{29}{g/mol}$; $R = \qty{8.314}{J/(mol.K)}$. Keadaan
awal di mana-mana: $P_1 = \qty{1.00}{bar}$, $T_1 = \qty{300}{K}$.

\medskip
\noindent\textbf{Bagian I --- Pompa sepeda.}
Tabung pompanya bervolume $V_1 = \qty{212}{cm^3}$; udara di dalamnya
dimampatkan secara reversibel dan adiabatik sampai mencapai tekanan
bannya, \qty{7.0}{bar}.
\begin{enumerate}
  \item Jumlah zat udara di dalam tabungnya.
  \item Volumenya pada akhir pemampatannya.
  \item Suhunya pada akhir pemampatannya.
  \item Usaha yang diterima udaranya selama satu langkah.
  \item Bannya (\qty{2.0}{L}) harus dibawa dari \qty{1}{bar} ke
        \qty{7}{bar} pada \qty{300}{K}: berapa langkah, dan berapa usaha
        seluruhnya (ambillah usaha tiap langkahnya tetap)?
  \item Udara panasnya lalu mendingin di dalam bannya sampai \qty{300}{K}: apa yang terjadi
        pada tekanan bannya, dan apa yang dilakukan pesepedanya?
  \item Bandingkan usaha adiabatik tiap langkahnya dengan usaha sebuah
        pemampatan isotermal ke \qty{7}{bar} yang sama. Mengapa usaha
        isotermalnya lebih besar padahal gasnya berakhir lebih dingin?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Pemampatan dieselnya.}
Sebuah tabung $V_1 = \qty{0.50}{L}$ memampatkan udaranya secara reversibel dan
adiabatik dengan faktor $20$.
\begin{enumerate}[resume]
  \item Jumlah zat udaranya, dan suhu $T_2$ setelah pemampatannya.
  \item Tekanan $P_2$.
  \item Bahan bakarnya menyala sendiri di atas kira-kira \qty{500}{K}: jelaskan mengapa
        diesel tak memerlukan busi, dan mengapa mesin bensin (nisbah
        $\approx 10$, lihat \cref{exo:b1:first-law:3}) tak boleh menyala sendiri.
  \item Usaha yang diterima udaranya selama pemampatannya.
  \item Pada \qty{3000}{rpm} satu tabung memampatkan $25$ kali tiap sekon:
        daya yang diserap pemampatannya (ia dikembalikan kemudian dalam
        siklusnya).
  \item Pemampatan yang sebenarnya berlangsung cepat (ketakreversibelan) dan kehilangan \omterm{def:b1:first-law:heat}{kalor} ke
        dindingnya: ke arah mana masing-masing pengaruh itu menggeser $T_2$?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Pembakaran dan ekspansi.}
Di puncak langkahnya \qty{20}{mg} bahan bakar (nilai \omterm{def:b1:first-law:heat}{kalor}
\qty{43}{MJ/kg}) terbakar sementara toraknya mulai turun, pada tekanan tetap
$P_2$; lalu gasnya memuai secara reversibel dan adiabatik kembali ke $V_1$.
Perlakukan gasnya sebagai udara di seluruhnya.
\begin{enumerate}[resume]
  \item \omterm{def:b1:first-law:heat}{Kalor} yang dilepaskan bahan bakarnya.
  \item Suhu $T_3$ dan volume $V_3$ pada akhir pembakaran
        isobariknya.
  \item Usaha yang dipertukarkan selama tahap isobariknya (perhatikan tandanya!).
  \item Suhu $T_4$ pada akhir ekspansi adiabatiknya, dan
        usaha yang dipertukarkan selama itu.
  \item Usaha total yang disampaikan gasnya sepanjang keempat langkahnya, dan
        nisbahnya terhadap \omterm{def:b1:first-law:heat}{kalor} yang dilepaskan (efisiensi idealnya).
  \item Bandingkan efisiensi itu dengan $1 - T_1/T_3$, yaitu batas yang akan
        ditegakkan \cref{ch:b1:heat-engines} bagi sebarang mesin yang bekerja
        di antara kedua suhu ekstrem ini.
  \item \omterm{def:b1:first-law:heat}{Kalor} yang dibuang bersama gas buangnya, menurut hukum pertama sepanjang siklusnya;
        periksalah neracanya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Mengukur $\gamma$ dengan bola yang berosilasi.}
Sebuah bola baja bermassa $m = \qty{16.5}{g}$ pas di dalam tabung kaca tegak
berpenampang $S = \qty{2.0}{cm^2}$ yang tertutup di bawahnya oleh labu
bervolume $V = \qty{10}{L}$ berisi udara pada $P = \qty{1.0}{bar}$; bila digeser sejauh $x$
dari kedudukan setimbangnya, ia berosilasi.
\begin{enumerate}[resume]
  \item Mengapa pemampatan dan ekspansi udaranya boleh dianggap
        adiabatik? Tuliskan hubungan antara simpangan kecil $x$
        dan perubahan tekanan $\dd P$ (\omterm{prop:b1:first-law:transformations}{hukum Laplace}, yang dilinearkan).
  \item Simpulkan gaya pemulih pada bolanya lalu tunjukkan bahwa geraknya
        harmonik, dengan $\omega_0^2 = \gamma PS^2/mV$.
  \item Hitunglah periodenya, dan nilai $\gamma$ yang akan disimpulkan
        dari periode terukur \qty{1.10}{s}.
  \item Mengapa gesekan antara bola dan tabungnya, dan kebocoran udaranya, membuat
        hasilnya berat sebelah, dan ke arah mana?
  \item Ringkaslah ketiga pemampatannya: besaran mana yang dikekalkan hukum
        pertama pada masing-masing, dan apa yang diatur eksponen adiabatik
        $\gamma$.
\end{enumerate}
\end{problem}
