\chapter{Teori Kinetik dan Gas Ideal}\label{ch:b1:kinetic-theory}

Udara di dalam sebuah ruang kelas seberat seorang dewasa, dan memuat lebih
banyak molekul daripada butir pasir di Bumi, masing-masing terbang dengan
laju peluru senapan dan bertumbukan sepuluh miliar kali tiap sekon dengan
tetangganya. Tak seorang pun dapat mengikuti satu di antaranya; tak seorang
pun perlu. Segenggam rata-rata --- \omterm{def:b1:kinetic-theory:pressure}{tekanan}, \omterm{def:b1:kinetic-theory:temperature}{suhu}, volume, jumlah zat ---
melukiskan udaranya secara lengkap bagi setiap keperluan keteknikan, dan
gambaran sederhana tentang molekul yang memantul dari dinding menjelaskan
dari mana rata-rata itu datang dan bagaimana semuanya berhubungan. Bab ini
membuka bagian termodinamika volume ini dengan memandang materi dari kedua
ujungnya: kekacauan mikroskopik dan ketenangan makroskopik, serta teori
kinetik yang menghubungkan keduanya pada kasus \omterm{def:b1:kinetic-theory:model}{gas ideal}.

\section{Dua skala pemerian}

\begin{definition}[Skala mikroskopik dan makroskopik]\label{def:b1:kinetic-theory:scales}
\index{skala mikroskopik}\index{skala makroskopik}\index{tetapan Avogadro}\index{mol}
Pada skala \emph{mikroskopik} materi tersusun atas molekul: $\qty{1}{mol}$
memuat $N_A = \qty{6.02e23}{}$ di antaranya, masing-masing berukuran \qty{0.3}{nm};
dalam gas pada keadaan biasa jaraknya \qty{3}{nm}, bergerak dengan
ratusan meter tiap sekon dan bertumbukan tiap \qty{70}{nm} atau sekitarnya.
Pada skala \emph{makroskopik} sebuah sistem dilukiskan oleh beberapa
\emph{\omterm{def:b1:kinetic-theory:equilibrium}{variabel keadaan}} --- \omterm{def:b1:kinetic-theory:pressure}{tekanan} $P$, volume $V$, \omterm{def:b1:kinetic-theory:temperature}{suhu} $T$,
jumlah zat $n$ --- yang merupakan rata-rata atas molekul yang jumlahnya
luar biasa besar. Sebuah variabel disebut \emph{ekstensif} bila ia berlipat dua ketika sistemnya
dilipatduakan ($V$, $n$, massa, energi), \emph{intensif} bila tidak
($P$, $T$, massa jenis).
\end{definition}

\begin{definition}[Kesetimbangan termodinamika]\label{def:b1:kinetic-theory:equilibrium}
\index{kesetimbangan termodinamika}\index{variabel keadaan}
Sebuah sistem berada dalam \emph{kesetimbangan termodinamika} bila variabel keadaannya
seragam dan tetap terhadap waktu serta tak ada aliran materi atau
energi makroskopik yang melintasinya. Barulah ketika itu $P$ dan $T$ terdefinisi bagi sistemnya
sebagai keseluruhan; \emph{persamaan keadaan} menghubungkan keduanya dengan $V$ dan $n$.
\end{definition}

\begin{definition}[Tekanan]\label{def:b1:kinetic-theory:pressure}
\index{tekanan}\index{pascal}\index{bar}
Fluida yang diam mendorong setiap unsur permukaan $\dd S$ sebuah dinding dengan
gaya $\dd\vect F = P\,\dd S\,\vect n$ yang tegak lurus padanya, menuju dindingnya; $P$ adalah
\emph{tekanan}, dalam pascal ($\qty{1}{Pa} = \qty{1}{N/m^2}$); $\qty{1}{bar}
= \qty{1e5}{Pa}$, $\qty{1}{atm} = \qty{1.013e5}{Pa}$. Pada sebuah titik ia tak
bergantung pada arah hadap permukaannya (\cref{ch:b1:fluid-statics}).
\end{definition}

\begin{definition}[Suhu]\label{def:b1:kinetic-theory:temperature}
\index{suhu}\index{kelvin}\index{hukum ke-nol}
Dua sistem yang bersentuhan melalui dinding yang meneruskan energi lama-kelamaan
mencapai keadaan bersama: keduanya lalu bersuhu sama, itulah \emph{suhu}, dan
dua benda yang masing-masing setimbang dengan benda ketiga juga setimbang satu sama
lain (\emph{hukum ke-nol}). Skala kelvin ditetapkan oleh
$k_B = \qty{1.380649e-23}{J/K}$ secara eksak; Celsius: $\theta = T - 273.15$. Teori
kinetik di bawah memberi $T$ makna mikroskopiknya.
\end{definition}

\section{Model kinetik gas ideal}

\begin{definition}[Gas ideal, model mikroskopik]\label{def:b1:kinetic-theory:model}
\index{gas ideal}\index{gas sempurna}
Yang disebut \emph{gas ideal} (gas sempurna) adalah kumpulan $N$ molekul,
yang diperlakukan sebagai titik bermassa $m$, bergerak bebas dan acak --- tanpa
interaksi selain tumbukan singkat --- di dalam wadah bervolume $V$.
Pada kesetimbangan distribusi kecepatannya isotropik dan tunak;
$n^* = N/V$ adalah konsentrasi molekulnya dan $\langle v^2\rangle$ kelajuan kuadrat
rata-ratanya.
\end{definition}

\begin{theorem}[Tekanan kinetik]\label{thm:b1:kinetic-theory:pressure}
\index{tekanan kinetik}
\omterm{def:b1:kinetic-theory:pressure}{Tekanan} yang dikerahkan gasnya pada dindingnya adalah
\[
P = \tfrac13\,n^*m\langle v^2\rangle = \tfrac23\,n^*\langle\epsilon_k\rangle ,
\]
dengan $\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle$ adalah \omterm{thm:b1:work-and-energy:ke}{energi
kinetik} rata-rata sebuah molekul.
\end{theorem}

\begin{proof}[Bukti sebagian]
Ambil model sederhana yang di dalamnya semua molekulnya berkelajuan $u$ dan
bergerak menyusuri ketiga sumbunya, seperenam ke tiap arah. Sebuah molekul
yang menghantam dinding $x = 0$ tegak lurus memantul balik secara lenting dan menyampaikan
momentum $2mu$ kepadanya. Dalam waktu $\dd t$, molekul yang bergerak menuju
dindingnya dan mencapai luas $S$ adalah molekul di dalam lempeng setebal $u\,\dd t$,
yaitu seperenam dari $n^*Su\,\dd t$ molekul di sana. Gaya $=$
momentum tiap waktu $= (n^*Su\,\dd t/6)(2mu)/\dd t = \tfrac13 n^*mu^2S$, sehingga
$P = \tfrac13 n^*mu^2$. Merata-ratakan atas distribusi kelajuan dan arah
yang sebenarnya menggantikan $u^2$ dengan $\langle v^2\rangle$ (keisotropikannya memberikan
$\langle v_x^2\rangle = \tfrac13\langle v^2\rangle$); volume Tahun ke-2 mengerjakannya
secara lengkap.
\end{proof}

\begin{definition}[Suhu kinetik; persamaan keadaan]\label{def:b1:kinetic-theory:kinetictemp}
\index{suhu kinetik}\index{persamaan keadaan!gas ideal}\index{tetapan gas}\index{kelajuan rms}
\omterm{def:b1:kinetic-theory:temperature}{Suhu} sebuah \omterm{def:b1:kinetic-theory:model}{gas ideal} \emph{didefinisikan} oleh \omterm{thm:b1:work-and-energy:ke}{energi kinetik} rata-rata
molekulnya,
\[
\langle\epsilon_k\rangle = \tfrac12 m\langle v^2\rangle = \tfrac32 k_BT ,
\]
sehingga, digabungkan dengan \cref{thm:b1:kinetic-theory:pressure},
\[
PV = Nk_BT = nRT, \qquad R = N_Ak_B = \qty{8.314}{J/(mol.K)} ,
\]
itulah \emph{persamaan keadaan \omterm{def:b1:kinetic-theory:model}{gas ideal}}. Yang disebut \emph{kelajuan rms}
(akar rata-rata kuadrat) adalah $u = \sqrt{\langle v^2\rangle} = \sqrt{3k_BT/m} = \sqrt{3RT/M}$,
dengan $M$ massa molarnya.
\end{definition}

\begin{example}[Angka-angka untuk udara]\label{ex:b1:kinetic-theory:air}
Nitrogen pada \qty{300}{K}: $u = \sqrt{3 \times 8.314 \times 300/0.028} =
\qty{517}{m/s}$; hidrogen, $\qty{1930}{m/s}$; makin berat makin lambat, sebagai
$1/\sqrt M$. Pada \qty{1}{\omterm{def:b1:kinetic-theory:pressure}{bar}} dan \qty{300}{K}, $n^* = P/k_BT = \qty{2.4e25}{m^{-3}}$:
\qty{24}{L} tiap \omterm{def:b1:kinetic-theory:scales}{mol}, \qty{2.7e22}{} molekul dalam satu liter. Ruang kelas
\qty{200}{m^3}: \qty{8300}{\omterm{def:b1:kinetic-theory:scales}{mol}}, \qty{240}{kg} udara.
\end{example}

\begin{remark}[Mengapa modelnya berhasil, dan kapan ia gagal]\label{rem:b1:kinetic-theory:validity}
\index{lintasan bebas rata-rata}
Di antara dua tumbukan sebuah molekul udara terbang sejauh \emph{\omterm{rem:b1:kinetic-theory:validity}{lintasan bebas rata-rata}}
$\lambda \approx 1/(\sqrt2\pi d^2n^*) \approx \qty{70}{nm}$ --- dua ratus
kali ukurannya: gasnya sebagian besar kosong, interaksinya jarang, dan
$PV = nRT$ berlaku dengan ketelitian lebih baik daripada satu persen pada \omterm{def:b1:kinetic-theory:pressure}{tekanan} biasa. Pada \omterm{def:b1:kinetic-theory:pressure}{tekanan}
tinggi atau \omterm{def:b1:kinetic-theory:temperature}{suhu} rendah volume molekulnya sendiri dan tarikannya
berperan: persamaan van der Waals $(P + an^2/V^2)(V - nb) = nRT$
mengoreksi keduanya, dan di bawah \omterm{def:b1:kinetic-theory:temperature}{suhu} kritis gasnya mencair
(\cref{ch:b1:phase-changes}).
\end{remark}

\section{Energi dalam gas ideal}

\begin{definition}[Energi dalam]\label{def:b1:kinetic-theory:U}
\index{energi dalam}
Yang disebut \emph{energi dalam} $U$ sebuah sistem adalah jumlah \omterm{thm:b1:work-and-energy:ke}{energi
kinetik} molekulnya dalam kerangka tempat sistemnya diam ditambah
\omterm{def:b1:work-and-energy:conservative}{energi potensial} timbal baliknya; ia adalah fungsi keadaan yang ekstensif.
\end{definition}

\begin{proposition}[Energi dalam gas ideal]\label{prop:b1:kinetic-theory:Ugas}
\index{gas monoatomik}\index{gas diatomik}\index{kapasitas kalor pada volume tetap}\index{hukum Joule}
Bagi \omterm{def:b1:kinetic-theory:model}{gas ideal} $U$ hanya bergantung pada $T$ (\omterm{prop:b1:kinetic-theory:Ugas}{hukum Joule} pertama):
\begin{gather*}
U = \tfrac32 nRT \quad (\text{monoatomik: He, Ne, Ar}), \\
U = \tfrac52 nRT \quad (\text{diatomik pada suhu biasa: N}_2, \text{O}_2, \text{H}_2),
\end{gather*}
sehingga kapasitas kalor molar pada volume tetap $C_{V,m} = \dd U_m/\dd T$
berturut-turut bernilai $\tfrac32 R = \qty{12.5}{J/(mol.K)}$ dan
$\tfrac52 R = \qty{20.8}{J/(mol.K)}$.
\end{proposition}

\begin{proof}[Bukti sebagian]
Tak ada \omterm{def:b1:work-and-energy:conservative}{energi potensial} timbal balik menurut hipotesisnya, sehingga $U = N\langle\epsilon_k\rangle
= \tfrac32 Nk_BT$ bagi molekul titik. Molekul diatomik juga berputar:
masing-masing dari kedua derajat kebebasan rotasinya mengemban, pada kesetimbangan,
$\tfrac12 k_BT$ yang sama seperti tiap derajat kebebasan translasinya (teorema
\emph{ekipartisi}, diterima tanpa bukti; volume Tahun ke-3 menurunkannya dan menjelaskan mengapa
getarannya baru ikut serta di atas seribu \omterm{def:b1:kinetic-theory:temperature}{kelvin}).
\end{proof}

\begin{example}[Sekelas udara]\label{ex:b1:kinetic-theory:room}
Udara \qty{8300}{\omterm{def:b1:kinetic-theory:scales}{mol}} di ruang kelas pada \qty{300}{K}: $U = \tfrac52 \times
8300 \times 8.314 \times 300 = \qty{5.2e7}{J}$ --- energi satu setengah liter
bensin, yang tersimpan dalam gerak molekul. Menghangatkan ruangannya sebesar $1$ K
menuntut $\tfrac52 nR = \qty{170}{kJ}$ (pada volume tetap).
\end{example}

\section{Fase terkondensasi}

\begin{proposition}[Model fase tak termampatkan dan tak termuaikan]\label{prop:b1:kinetic-theory:condensed}
\index{fase terkondensasi}\index{pemuaian}\index{kompresibilitas}
Zat cair dan zat padat dimodelkan, pada tingkat ini, sebagai fase yang
volumenya tetap --- tak termampatkan maupun termuaikan --- yang \omterm{def:b1:kinetic-theory:U}{energi dalamnya}
hanya bergantung pada $T$: $\dd U = C\,\dd T$ dengan $C = mc$, dengan $c$ kalor
jenisnya (\qty{4180}{J/(kg.K)} untuk air, \qty{450}{J/(kg.K)} untuk
baja). Simpangan nyatanya yang kecil diukur oleh koefisien pemuaian
$\alpha = (1/V)(\partial V/\partial T)_P$ ($\sim\qty{1e-5}{K^{-1}}$
untuk logam, \qty{2e-4}{K^{-1}} untuk zat cair) dan
kompresibilitas isotermal $\chi_T = -(1/V)(\partial V/\partial P)_T$ ($\sim\qty{5e-10}{Pa^{-1}}$
untuk air, berbanding $1/P \approx \qty{1e-5}{Pa^{-1}}$ untuk gas).
\end{proposition}
\admitted

\begin{example}[Seberapa tak termampatkan]\label{ex:b1:kinetic-theory:water}
Air di dasar Palung Mariana (\qty{1100}{\omterm{def:b1:kinetic-theory:pressure}{bar}}) termampatkan
sebesar $\chi_T\Delta P \approx 5\times10^{-10} \times 1.1\times10^8 = 5\%$; sebatang
rel baja \qty{30}{m} yang dihangatkan sebesar $50$ K memanjang sebesar $\alpha L\Delta T =
\qty{18}{mm}$ --- itulah alasan sambungan muai. Dibandingkan gas, yang
volumenya menjadi separuh di bawah \omterm{def:b1:kinetic-theory:pressure}{tekanan} yang dilipatduakan, model volume tetap ini
baik sampai beberapa persen pada kebanyakan keadaan dalam volume ini.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:1}
Jumlah molekul dalam \qty{1.0}{L} gas pada \qty{0}{\degreeCelsius} dan
\qty{1}{atm}; jarak rata-rata antar tetangga; bandingkan dengan ukuran
molekulnya \qty{0.3}{nm}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:2}
\omterm{def:b1:kinetic-theory:kinetictemp}{Kelajuan rms} H$_2$, N$_2$, O$_2$ dan CO$_2$ pada \qty{300}{K},
dan N$_2$ pada \qty{77}{K} (nitrogen yang mendidih).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:3}
Sebuah ban mobil bervolume \qty{30}{L} dipompa sampai \omterm{def:b1:kinetic-theory:pressure}{tekanan} tolok
\qty{2.5}{\omterm{def:b1:kinetic-theory:pressure}{bar}} pada \qty{20}{\degreeCelsius}. Jumlah zat dan massa udara di dalamnya;
\omterm{def:b1:kinetic-theory:pressure}{tekanan} mutlaknya setelah perjalanan yang memanaskannya sampai \qty{50}{\degreeCelsius}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:kinetic-theory:4}
Taksirlah \omterm{rem:b1:kinetic-theory:validity}{lintasan bebas rata-rata} molekul udara pada \qty{1}{\omterm{def:b1:kinetic-theory:pressure}{bar}}, \qty{300}{K}
($d = \qty{0.37}{nm}$), dan frekuensi tumbukan satu molekul. Pada
\omterm{def:b1:kinetic-theory:pressure}{tekanan} berapa \omterm{rem:b1:kinetic-theory:validity}{lintasan bebas rata-ratanya} mencapai \qty{1}{m} (sebuah ``hampa'')?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:5}
Massa jenis udara pada \qty{1.013}{\omterm{def:b1:kinetic-theory:pressure}{bar}} dan \qty{20}{\degreeCelsius}
($M = \qty{29}{g/mol}$), lalu pada \qty{100}{\degreeCelsius}. Daya angkat sebuah
balon udara panas \qty{2800}{m^3} (selisih kedua berat udaranya).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:6}
Udara terdiri atas $21\%$ O$_2$ menurut jumlah molekulnya. \omterm{def:b1:kinetic-theory:pressure}{Tekanan} parsial oksigen di permukaan laut;
pada \qty{5000}{m} tempat $P = \qty{0.54}{atm}$; di puncak Everest
(\qty{0.33}{atm}). Mengapa tubuh kepayahan di sana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:7}
Satu \omterm{def:b1:kinetic-theory:scales}{mol} CO$_2$ pada \qty{300}{K} dalam \qty{1.0}{L}: \omterm{def:b1:kinetic-theory:pressure}{tekanannya} menurut
hukum \omterm{def:b1:kinetic-theory:model}{gas ideal}, lalu menurut van der Waals ($a = \qty{0.364}{Pa.m^6/mol^2}$,
$b = \qty{4.27e-5}{m^3/mol}$). Koreksi mana yang berkuasa di sini?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:8}
\omterm{def:b1:kinetic-theory:U}{Energi dalam} udara di sebuah ruangan \qty{50}{m^3} pada \qty{1}{\omterm{def:b1:kinetic-theory:pressure}{bar}},
\qty{300}{K}; energi untuk menghangatkannya sebesar \qty{5}{K} pada volume tetap; waktunya
dengan pemanas \qty{2}{kW} (tanpa kerugian).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:kinetic-theory:9}
Sebatang rel baja \qty{30}{m} ($\alpha = \qty{1.2e-5}{K^{-1}}$) di antara
\qty{-10}{\degreeCelsius} dan \qty{40}{\degreeCelsius}: perubahan panjangnya.
Sebuah cincin kuningan ($\alpha = \qty{1.9e-5}{K^{-1}}$) bergaris tengah dalam
\qty{49.95}{mm} harus dapat disarungkan pada poros \qty{50.00}{mm}: sebesar apa ia
harus dipanaskan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:10}
Energi tiap molekul pada \qty{300}{K} ($\tfrac32 k_BT$) dalam joule dan eV;
kapasitas kalor molar $C_{V,m}$ argon dan nitrogen; mengapa kapasitas kalor
nitrogen lebih besar padahal molekulnya lebih ringan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:11}
Ulangi penurunan \omterm{thm:b1:kinetic-theory:pressure}{tekanan kinetiknya} pada model enam arah,
lalu tunjukkan bahwa bagi distribusi kecepatan yang isotropik $\langle v_x^2\rangle
= \tfrac13\langle v^2\rangle$ dan bahwa rumus \omterm{def:b1:kinetic-theory:pressure}{tekanannya} tak berubah.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:kinetic-theory:12}
\omterm{def:b1:kinetic-theory:kinetictemp}{Kelajuan rms} H$_2$ dan O$_2$ pada \qty{300}{K} dibandingkan dengan
\omterm{prop:b1:central-forces:escape}{laju lepas} Bumi, dan \omterm{prop:b1:central-forces:escape}{laju lepas} Bulan (\qty{2.4}{km/s}).
Dengan mengetahui bahwa pecahan molekul yang lebih cepat daripada $3u$ kecil tetapi tak
nol, jelaskan mengapa Bumi telah kehilangan hidrogennya dan Bulan seluruh
atmosfernya.
\end{exercise}

\section{Soal: Udara di dalam ruangan}

\begin{problem}[{Soal akhir pekan --- seruang kelas udara: menghitung
molekulnya, menimbangnya, mengukur waktu terbangnya, dan memulihkan dari
model yang sama daya angkat sebuah balon, tipisnya udara gunung, dan
laju bunyi}]
\label{pb:b1:kinetic-theory:1}
Sebuah ruang kelas berukuran $10 \times 8 \times \qty{2.5}{m}$; udaranya berada pada
$P = \qty{1.013e5}{Pa}$, $T = \qty{293}{K}$, massa molar $M = \qty{29.0}{g/mol}$,
$79\%$ N$_2$ dan $21\%$ O$_2$ menurut jumlah molekulnya. $R = \qty{8.314}{J/(mol.K)}$,
$k_B = \qty{1.38e-23}{J/K}$, $N_A = \num{6.02e23}$, $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Bagian I --- Menghitung dan menimbang.}
\begin{enumerate}
  \item Jumlah zat udara (\omterm{def:b1:kinetic-theory:scales}{mol}) dan jumlah molekul di dalam ruangannya.
  \item Massa udaranya; bandingkan dengan seorang manusia.
  \item Konsentrasi molekulnya $n^*$ dan jarak rata-rata antar molekul
        ($n^{*-1/3}$); bandingkan dengan garis tengah molekulnya \qty{0.37}{nm}.
  \item \omterm{def:b1:kinetic-theory:kinetictemp}{Kelajuan rms} N$_2$ dan O$_2$; mengapa gas yang lebih berat
        bergerak lebih lambat pada \omterm{def:b1:kinetic-theory:temperature}{suhu} yang sama?
  \item \omterm{thm:b1:work-and-energy:ke}{Energi kinetik} rata-rata satu molekul; \omterm{thm:b1:work-and-energy:ke}{energi kinetik} translasi
        total udara di ruangan itu; bandingkan dengan \omterm{thm:b1:work-and-energy:ke}{energi kinetik} sebuah
        mobil pada \qty{100}{km/h}.
  \item \omterm{rem:b1:kinetic-theory:validity}{Lintasan bebas rata-rata} $\lambda \approx 1/(\sqrt2\pi d^2n^*)$ dan waktu
        rata-rata antara dua tumbukan sebuah molekul.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- \omterm{def:b1:kinetic-theory:pressure}{Tekanan} dari hantaman.}
\begin{enumerate}[resume]
  \item Dengan memakai model enam arah dengan $u = \qty{500}{m/s}$, hitunglah
        jumlah hantaman tiap sekon pada \qty{1}{cm^2} dindingnya.
  \item Hitunglah momentum yang disampaikan tiap sekon, lalu periksalah bahwa ia
        menghasilkan kembali \omterm{def:b1:kinetic-theory:pressure}{tekanan} atmosfernya.
  \item Gaya total udaranya pada satu dinding seluas \qty{8}{m} kali \qty{2.5}{m};
        mengapa dindingnya tak bergerak?
  \item Ruangannya dihangatkan sampai \qty{303}{K} dengan jendela tertutup dan
        tersegel: berapa \omterm{def:b1:kinetic-theory:pressure}{tekanan} barunya, dan berapa gaya total pada dindingnya kini.
  \item Penghangatan yang sama dengan jendela terbuka: apa yang pergi, dan berapa banyak?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Gaya apung.}
\begin{enumerate}[resume]
  \item Massa jenis udara di ruangan itu, dan massa jenis udara pada \qty{373}{K} pada
        \omterm{def:b1:kinetic-theory:pressure}{tekanan} yang sama.
  \item Sebuah balon udara panas bervolume \qty{2800}{m^3} diisi udara pada
        \qty{373}{K}: berat udara panasnya, berat udara dingin yang
        didesaknya, dan daya angkat yang tersedia (Archimedes,
        \cref{ch:b1:fluid-statics}).
  \item Selubung, keranjang, pembakar dan gasnya berbobot \qty{300}{kg}: berapa banyak
        penumpang \qty{75}{kg} yang dapat diangkatnya?
  \item Sampai \omterm{def:b1:kinetic-theory:temperature}{suhu} berapa udaranya harus dipanaskan untuk melipatduakan daya angkatnya,
        dan mengapa itu tidak dilakukan (selubungnya nilon)?
  \item Sebuah balon helium bervolume sama pada \qty{293}{K}
        ($M = \qty{4}{g/mol}$): berapa daya angkatnya? Mengapa balon wisata tetap
        memakai udara panas?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Udara tipis dan laju bunyi.}
\begin{enumerate}[resume]
  \item \omterm{def:b1:kinetic-theory:pressure}{Tekanan} parsial oksigen di dalam ruangan itu.
  \item Di puncak Everest $P = \qty{0.33}{atm}$ dan $T \approx \qty{250}{K}$:
        konsentrasi molekul oksigennya dibandingkan dengan yang di ruangan itu.
  \item \omterm{def:b1:kinetic-theory:pressure}{Tekanan} atmosfer turun terhadap ketinggian kira-kira sebagai
        $P(z) = P_0\eu^{-z/H}$ dengan $H = RT/Mg$ (\cref{ch:b1:fluid-statics}):
        hitunglah $H$ untuk \qty{260}{K} dan ketinggian tempat $P = P_0/2$.
  \item Laju bunyi dalam \omterm{def:b1:kinetic-theory:model}{gas ideal} adalah $c_s = \sqrt{\gamma RT/M}$
        dengan $\gamma = 1.4$ untuk udara (\cref{ch:b1:first-law}): hitunglah pada
        \qty{293}{K} lalu bandingkan dengan kelajuan rms-nya; tafsirkan
        kedekatannya.
  \item Bagaimana $c_s$ berubah terhadap $T$? Laju bunyi di ruangan itu pada
        \qty{303}{K}, dan dalam helium pada \qty{293}{K} ($\gamma = 5/3$): mengapa
        suara helium terdengar tinggi?
  \item Di puncak Everest, apakah bunyi lebih cepat atau lebih lambat daripada di
        ruangan itu? Sebesar apa?
  \item Udara di ruangan itu diganti argon pada $P$ dan $T$ yang sama:
        apa yang berubah pada jumlahnya, massanya, kelajuannya, dan \omterm{def:b1:kinetic-theory:U}{energi
        dalamnya}?
  \item \omterm{def:b1:kinetic-theory:U}{Energi dalam} udara di ruangan itu ($U = \tfrac52 nRT$) dan
        energi untuk menghangatkannya sebesar \qty{10}{K} pada volume tetap.
  \item Ringkaslah dalam tiga baris apa yang diberikan model kinetiknya: besaran
        makroskopik mana yang dijelaskannya, dan dari satu rata-rata
        mikroskopik yang mana.
\end{enumerate}
\end{problem}
