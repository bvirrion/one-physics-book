\chapter{Gaya Sentral: Planet dan Satelit}\label{ch:b1:central-forces}

Tiga puluh satu satelit mengelilingi Bumi dua kali sehari, masing-masing
sebuah jam sekaligus sebuah radio, dan sebuah telepon yang mendengar
empat di antaranya tahu di mana ia berada sampai beberapa meter. Komet
Halley jatuh menuju Matahari selama tiga puluh delapan tahun, mengayun
mengitarinya dalam beberapa pekan, lalu memanjat menjauh lagi selama
tiga puluh delapan tahun berikutnya. Sebuah wahana yang menuju Mars
meninggalkan Bumi tepat dengan laju yang membuat \omterm{def:b1:point-kinematics:frame}{lintasannya} sekadar
mencium orbit Mars. Semuanya adalah gerak sebuah benda yang tertarik ke
sebuah titik tetap oleh gaya yang jatuh sebagai kebalikan kuadrat
jaraknya, dan semuanya menyusul dari dua besaran kekal --- energi dan
\omterm{def:b1:angular-momentum:L}{momentum sudut} --- ditambah satu gagasan baru, yaitu potensial efektif.
Bab ini menurunkan bentuk dan \omterm{thm:b1:central-forces:circular}{laju orbitnya}, ketiga \omterm{thm:b1:central-forces:conics}{hukum Kepler}, serta
aritmetika menempatkan sebuah satelit di tempat yang dikehendaki.

\section{Gaya sentral yang konservatif}

\begin{definition}[Gaya sentral; gaya sentral konservatif]\label{def:b1:central-forces:central}
\index{gaya sentral}\index{energi potensial efektif}
Sebuah gaya disebut \emph{sentral} berpusat $O$ bila ia selalu terarah
menyusuri $\vect{OM}$: $\vect F = F(M)\,\vect e_r$. Ia disebut
\emph{sentral \omterm{def:b1:work-and-energy:conservative}{konservatif}} bila besarnya hanya bergantung pada
$r = OM$: $\vect F =
F(r)\,\vect e_r$, yang diturunkan dari \omterm{def:b1:work-and-energy:conservative}{energi potensial} $E_p(r)$ dengan
$F(r) = -\dd E_p/\dd r$.
\end{definition}

\begin{theorem}[Gerak di bawah gaya sentral konservatif]\label{thm:b1:central-forces:reduction}
\index{hukum luas sapuan}
Sebuah partikel bermassa $m$ yang hanya dikenai gaya \omterm{def:b1:central-forces:central}{sentral konservatif}:
\begin{enumerate}
  \item menjaga \omterm{def:b1:angular-momentum:L}{momentum sudutnya} $\vect L_O$; geraknya bidang dan, dalam
        \omterm{def:b1:point-kinematics:cylindrical}{koordinat polar} pada bidang itu, $r^2\dot\theta = C$ bersifat
        tetap (\omterm{cor:b1:angular-momentum:central}{hukum luas sapuan},
        \cref{cor:b1:angular-momentum:central});
  \item menjaga \omterm{thm:b1:work-and-energy:em}{energi mekaniknya} $E_m = \tfrac12 m(\dot r^2 + r^2\dot\theta^2)
        + E_p(r)$, yang dapat ditulis
        \[
        E_m = \tfrac12 m\dot r^2 + E_{p,\mathrm{eff}}(r), \qquad
        E_{p,\mathrm{eff}}(r) = E_p(r) + \frac{mC^2}{2r^2} = E_p(r) + \frac{L^2}{2mr^2} .
        \]
\end{enumerate}
Gerak radialnya dengan demikian adalah gerak sebuah partikel satu
dimensi dalam \emph{potensial efektif} $E_{p,\mathrm{eff}}$: \omterm{def:b1:work-and-energy:turning}{titik balik}
radialnya adalah akar $E_{p,\mathrm{eff}}(r) = E_m$, geraknya terikat
bila ia terperangkap di antara dua di antaranya, dan melingkar bila $r$
duduk pada sebuah minimum $E_{p,\mathrm{eff}}$.
\end{theorem}

\begin{proof}
Bagian 1 adalah \cref{cor:b1:angular-momentum:central}. Gayanya
\omterm{def:b1:work-and-energy:conservative}{konservatif} dan tak ada gaya lain yang bekerja, sehingga $E_m$ kekal; di
bidangnya, $v^2 = \dot r^2 + r^2\dot\theta^2$ dan $r^2\dot\theta^2 = C^2/r^2$.
Suku $mC^2/2r^2$, yaitu \omterm{thm:b1:work-and-energy:ke}{energi kinetik} gerak sudutnya, berlaku sebagai
\emph{\omterm{prop:b1:central-forces:effective}{penghalang sentrifugal}} yang menolak dalam soal radialnya, dan
kepadanya berlaku analisis satu dimensi \cref{ch:b1:work-and-energy}.
\end{proof}

\section{Gravitasi Newton}

\begin{definition}[Gravitasi semesta]\label{def:b1:central-forces:gravitation}
\index{gravitasi}\index{energi potensial gravitasi}
Massa titik $M$ di $O$ menarik massa titik $m$ di $M$ dengan
\[
\vect F = -\frac{GMm}{r^2}\,\vect e_r, \qquad E_p = -\frac{GMm}{r} \quad (E_p \to 0 \text{ di tak hingga}),
\]
$G = \qty{6.67e-11}{N.m^2/kg^2}$. Benda yang setangkup bola menarik di
luar dirinya seolah seluruh massanya berada di pusatnya (diterima saja;
akibat teorema Gauss, \cref{ch:b1:electrostatics-gauss}).
\end{definition}

\begin{proposition}[Potensial efektif gravitasi]\label{prop:b1:central-forces:effective}
\index{penghalang sentrifugal}\index{orbit lingkaran}
Untuk $E_p = -GMm/r$,
\[
E_{p,\mathrm{eff}}(r) = -\frac{GMm}{r} + \frac{L^2}{2mr^2}
\]
menuju $+\infty$ ketika $r \to 0$ (untuk $L \neq 0$), menuju $0^-$ di
tak hingga, dan mempunyai satu minimum pada $r_c = L^2/GMm^2$ yang
bernilai $-G^2M^2m^3/2L^2$. Maka: $E_m < 0$ --- \omterm{def:b1:work-and-energy:turning}{gerak terikat} antara
jari-jari minimum dan maksimum (perigee dan apogee); $E_m \geq 0$ ---
tak terikat, bendanya datang dari dan kembali ke tak hingga; $E_m$ sama
dengan minimumnya --- \omterm{prop:b1:central-forces:effective}{orbit lingkaran} berjari-jari $r_c$.
\end{proposition}

\begin{proof}
Limitnya terlihat langsung; $E_{p,\mathrm{eff}}' = GMm/r^2 - L^2/mr^3 = 0$
pada $r_c$; substitusikan. Kasus melingkarnya adalah $\dot r = 0$
selamanya, yang hanya mungkin pada minimumnya.
\end{proof}

\begin{theorem}[Orbit lingkaran]\label{thm:b1:central-forces:circular}
\index{laju orbit}\index{hukum ketiga Kepler}
Pada \omterm{prop:b1:central-forces:effective}{orbit lingkaran} berjari-jari $r$ mengelilingi massa $M$:
\[
v = \sqrt{\frac{GM}{r}}, \qquad
T = 2\pi\sqrt{\frac{r^3}{GM}}, \qquad
E_m = -\frac{GMm}{2r} = -E_k = \tfrac12 E_p .
\]
Khususnya $T^2/r^3 = 4\pi^2/GM$ sama bagi semua \omterm{prop:b1:central-forces:effective}{orbit lingkaran}
mengelilingi pusat yang sama (\omterm{thm:b1:central-forces:circular}{hukum ketiga Kepler}).
\end{theorem}

\begin{proof}
\omterm{cor:b1:point-kinematics:circular}{Gerak melingkar} beraturan menuntut \omterm{prop:b1:newton-dynamics:centripetal}{gaya sentripetal} $mv^2/r = GMm/r^2$;
$T = 2\pi r/v$; $E_k = \tfrac12 mv^2 = GMm/2r$ dan $E_p = -GMm/r$.
\end{proof}

\begin{example}[Tiga orbit]\label{ex:b1:central-forces:orbits}
$GM_\oplus = \qty{3.99e14}{m^3/s^2}$. Orbit rendah ($r = \qty{6770}{km}$,
\qty{400}{km} di atas): $v = \qty{7.7}{km/s}$, $T = \qty{92}{min}$.
Geostasioner ($T = \qty{86164}{s}$, satu hari sideris): $r = (GMT^2/4\pi^2)^{1/3} =
\qty{42200}{km}$, $v = \qty{3.1}{km/s}$. Bulan ($r = \qty{384000}{km}$):
$T = \qty{27.4}{d}$ --- dan \omterm{cor:b1:point-kinematics:circular}{percepatan sentripetalnya} $v^2/r =
\qty{2.7e-3}{m/s^2}$ adalah $g/3600$ pada enam puluh jari-jari Bumi:
kebalikan kuadrat, yang diperiksa Newton sendiri.
\end{example}

\section{Hukum Kepler dan bentuk orbit}

\begin{theorem}[Lintasan dalam medan Newton]\label{thm:b1:central-forces:conics}
\index{hukum Kepler}\index{irisan kerucut}\index{elips (orbit)}\index{eksentrisitas}
\omterm{def:b1:point-kinematics:frame}{Lintasan} sebuah benda dalam medan massa tetap $M$ adalah \omterm{thm:b1:central-forces:conics}{irisan kerucut}
dengan $O$ pada salah satu fokusnya: elips bila $E_m < 0$, parabola bila
$E_m = 0$, hiperbola bila $E_m > 0$. Untuk elips bersetengah sumbu
panjang $a$ dan bereksentrisitas $e$:
\begin{itemize}
  \item (Kepler I) pusat gayanya berada di sebuah fokus; perigee dan
        apogeenya berada pada $r_p = a(1 - e)$ dan $r_a = a(1 + e)$;
  \item (Kepler II) \omterm{cor:b1:angular-momentum:central}{kecepatan arealnya} tetap (\omterm{cor:b1:angular-momentum:central}{hukum luas sapuan});
  \item (Kepler III) periodenya menuruti $T^2 = 4\pi^2a^3/GM$;
  \item energinya hanya bergantung pada $a$, $E_m = -GMm/2a$, sehingga
        laju pada jarak $r$ diberikan oleh
        \[
        v^2 = GM\left(\frac{2}{r} - \frac{1}{a}\right) .
        \]
\end{itemize}
\end{theorem}
\admitted

\begin{remark}[Apa yang dibuktikan dan apa yang diterima saja]\label{rem:b1:central-forces:admitted}
Kepler II dan penggolongan energinya telah dibuktikan di atas. Bahwa
orbit terikatnya tepat berupa elips dengan $O$ pada sebuah fokus,
beserta Kepler III dan $E_m = -GMm/2a$ secara umum, menuntut
pengintegralan persamaan radialnya (volume Tahun ke-2 melakukannya);
untuk \omterm{prop:b1:central-forces:effective}{orbit lingkaran} ($a = r$, $e = 0$) setiap pernyataannya menyusut
menjadi \cref{thm:b1:central-forces:circular}, dan itulah pemeriksaan
yang dapat kita lakukan di sini. Hubungan $v^2 = GM(2/r - 1/a)$ menyusul
dari $E_m = \tfrac12 mv^2 - GMm/r = -GMm/2a$ begitu rumus energi itu diterima.
\end{remark}

\begin{example}[Komet Halley]\label{ex:b1:central-forces:halley}
Perihelion \qty{0.586}{AU}, aphelion \qty{35.1}{AU}: $a = \qty{17.8}{AU}$,
$e = 0.967$, $T = \sqrt{17.8^3} = \qty{75}{tahun}$ (Kepler III dalam
satuan astronomi dan tahun, tempat $T^2 = a^3$ bagi Matahari). Dengan
$GM_\odot = \qty{1.33e20}{m^3/s^2}$, lajunya di perihelion adalah
$\sqrt{GM(2/r_p - 1/a)} = \qty{54}{km/s}$ dan di aphelion \qty{0.9}{km/s}
--- enam puluh kali lebih lambat, sebagaimana dituntut \omterm{cor:b1:angular-momentum:central}{hukum luas
sapuannya} ($r_pv_p = r_av_a$).
\end{example}

\section{Satelit}

\begin{proposition}[Laju lepas; orbit geostasioner]\label{prop:b1:central-forces:escape}
\index{laju lepas}\index{orbit geostasioner}
Dari permukaan sebuah benda bermassa $M$ dan berjari-jari $R$, laju
lontar minimum untuk mencapai tak hingga adalah
$v_e = \sqrt{2GM/R} = \sqrt2\,v_c$, dengan $v_c = \sqrt{GM/R}$ sebagai
laju lingkaran di permukaan tanahnya: \qty{11.2}{km/s} bagi Bumi.
Satelit \emph{geostasioner} mengorbit pada bidang khatulistiwa dengan
periode sideris Bumi, pada $r = \qty{42200}{km}$ (\qty{35800}{km} di
atas tanah).
\end{proposition}

\begin{proof}
$E_m \geq 0$: $\tfrac12 mv_e^2 - GMm/R = 0$. Kepler III dengan $T = \qty{86164}{s}$.
\end{proof}

\begin{proposition}[Alih Hohmann]\label{prop:b1:central-forces:hohmann}
\index{alih Hohmann}
Untuk memindahkan sebuah satelit dari \omterm{prop:b1:central-forces:effective}{orbit lingkaran} berjari-jari $r_1$
ke \omterm{prop:b1:central-forces:effective}{orbit lingkaran} sebidang berjari-jari $r_2 > r_1$ dengan dua
penyalaan singkat, nyalakan mesinnya di $r_1$ untuk memasuki \emph{elips
alih} yang berperigee $r_1$ dan berapogee $r_2$ ($a = (r_1 + r_2)/2$),
meluncur bebaslah setengah periode, lalu nyalakan lagi di $r_2$ untuk
membundarkannya:
\[
\Delta v_1 = \sqrt{\frac{GM}{r_1}}\left(\sqrt{\frac{2r_2}{r_1 + r_2}} - 1\right),
\qquad
\Delta v_2 = \sqrt{\frac{GM}{r_2}}\left(1 - \sqrt{\frac{2r_1}{r_1 + r_2}}\right),
\qquad
t = \pi\sqrt{\frac{(r_1 + r_2)^3}{8GM}} .
\]
\end{proposition}

\begin{proof}
Laju pada elips alihnya dari $v^2 = GM(2/r - 1/a)$ di $r_1$ dan $r_2$;
kurangkan laju lingkarannya; dan setengah periode
\cref{thm:b1:central-forces:conics}.
\end{proof}

\begin{example}[Dari orbit rendah ke geostasioner]\label{ex:b1:central-forces:leogeo}
$r_1 = \qty{6770}{km}$, $r_2 = \qty{42200}{km}$: $\Delta v_1 = \qty{2.4}{km/s}$,
$\Delta v_2 = \qty{1.5}{km/s}$, meluncur bebas \qty{5.3}{h}. Totalnya,
\qty{3.9}{km/s}, adalah separuh dari yang sudah dibayar peluncuran ke
orbit rendah: ``mil terakhir'' menuju orbit tinggi itu mahal, dan
satelitnya membawa mesinnya sendiri untuk itu.
\end{example}

\begin{remark}[Pusat yang menolak]\label{rem:b1:central-forces:rutherford}
\index{hamburan Rutherford}
Untuk gaya kebalikan kuadrat yang menolak (dua muatan sejenis,
\cref{ch:b1:electrostatics-gauss}) potensial efektifnya tak bersumur:
setiap \omterm{def:b1:point-kinematics:frame}{lintasannya} berupa hiperbola, partikelnya mendekat, dibelokkan
lalu pergi. Ketika sebuah partikel $\alpha$ ditembakkan lurus ke sebuah
inti emas, jarak pendekatan terdekatnya menyusul dari energinya semata,
$E_k =
2Ze^2/4\pi\varepsilon_0 r_{\min}$ --- cara Rutherford membatasi ukuran
intinya (\cref{exo:b1:central-forces:11}).
\end{remark}

\section{Soal: Satelit yang memberitahu di mana Anda berada}

\begin{problem}[{Soal akhir pekan --- sebuah konstelasi navigasi dua
puluh ribu kilometer di atas: orbitnya, bahan bakar untuk sampai ke
sana, berapa satelit yang dapat dilihat sebuah telepon, dan mengapa
jamnya harus disuruh berjalan lambat}]
\label{pb:b1:central-forces:1}
Data: $GM_\oplus = \qty{3.986e14}{m^3/s^2}$, $R_\oplus = \qty{6371}{km}$,
hari sideris \qty{86164}{s}, $c = \qty{2.998e8}{m/s}$. Satelitnya
mengelilingi Bumi tepat dalam setengah hari sideris pada \omterm{prop:b1:central-forces:effective}{orbit lingkaran}.

\medskip
\noindent\textbf{Bagian I --- Orbitnya.}
\begin{enumerate}
  \item Hitunglah jari-jari orbitnya dan ketinggiannya.
  \item Hitunglah \omterm{thm:b1:central-forces:circular}{laju orbitnya}.
  \item Hitunglah percepatan satelitnya lalu bandingkan dengan $g$ di
        permukaannya.
  \item Hitunglah \omterm{thm:b1:work-and-energy:em}{energi mekaniknya} tiap kilogram, beserta bagian
        kinetik dan bagian potensialnya.
  \item Mengapa dipilih setengah hari sideris dan bukan setengah hari
        surya? Apa yang dilihat berulang oleh pengamat di tanah?
  \item Dilihat dari sebuah satelit, cakram Bumi membentang berapa derajat?
  \item Bandingkan jari-jari orbitnya dengan jari-jari \omterm{prop:b1:central-forces:escape}{orbit
        geostasioner}; mana yang lebih tinggi, dan dengan faktor berapa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Sampai ke sana.}
\begin{enumerate}[resume]
  \item Berapa energi tiap kilogram untuk berjalan dari keadaan diam di
        permukaan ke orbit akhirnya (abaikan udaranya dan putaran Bumi).
  \item Peluncurnya lebih dulu mencapai \omterm{prop:b1:central-forces:effective}{orbit lingkaran} rendah pada
        \qty{400}{km}. Berapa lajunya di sana?
  \item \omterm{prop:b1:central-forces:hohmann}{Alih Hohmann} dari orbit itu ke orbit akhirnya: berapa kedua
        penyalaannya dan jumlahnya?
  \item Berapa durasi alihnya?
  \item Bandingkan jumlah penyalaannya dengan \omterm{prop:b1:central-forces:escape}{laju lepas} dari orbit
        rendah: seberapa jauh misi ini dari ``meninggalkan Bumi''?
  \item Tingkat atas beserta satelitnya bermassa \qty{2000}{kg} sebelum
        alihnya; laju semburan mesinnya \qty{3.0}{km/s} dan persamaan
        roket memberi $\Delta v = u\ln(m_0/m_1)$. Berapa propelan yang
        diperlukan bagi kedua penyalaannya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Cakupan dan pewaktuan.}
\begin{enumerate}[resume]
  \item Sebuah satelit terlihat dari setiap titik Bumi yang terletak di
        dalam kerucut bersetengah sudut $\theta$ dengan $\cos\theta = R/r$
        terhadap garis Bumi--satelit. Hitunglah $\theta$ dan bagian
        $(1 - \cos\theta)/2$ permukaan Bumi yang dicakupnya.
  \item Dengan $24$ satelit yang tersebar merata, berapa yang terlihat
        rata-rata dari satu titik? Apakah minimum $4$ yang diperlukan
        bagi sebuah penentuan posisi masuk akal?
  \item Berapa waktu tempuh cahaya dari sebuah satelit di zenit; di
        cakrawala?
  \item Untuk menentukan letak penerimanya sampai \qty{3}{m}, sampai
        ketelitian berapa \omterm{def:b1:wave-propagation:signal}{sinyalnya} harus diwaktukan?
  \item Jam penerimanya sendiri berupa kuarsa murah yang meleset
        beberapa milisekon: mengapa diperlukan empat satelit dan bukan tiga?
  \item Jam satelitnya bersifat atom. Berikan kemantapan nisbi yang
        diperlukan agar galatnya \qty{3}{m} setelah satu hari.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kerelatifan, sekilas.}
Dua pengaruh menggeser jam satelitnya relatif terhadap jam di tanah
(rumusnya diterima saja, dari volume Tahun ke-3): gerak memperlambat
sebuah jam sebesar bagian $v^2/2c^2$; ketinggian mempercepatnya sebesar
$GM(1/R - 1/r)/c^2$.
\begin{enumerate}[resume]
  \item Hitunglah perlambatan akibat lajunya, dalam mikrosekon tiap hari.
  \item Hitunglah percepatan akibat ketinggiannya, dalam mikrosekon tiap
        hari.
  \item Berapa pengaruh bersihnya, dan berapa galat posisi yang akan
        ditimbulkannya tiap hari jika tak dikoreksi?
  \item Jam satelitnya disetel sebelum diluncurkan ke
        \qty{10.22999999543}{MHz} alih-alih \qty{10.23}{MHz}. Periksalah
        bahwa hal ini mengimbangi pengaruh bersihnya.
  \item Orbitnya tidak tepat menuruti Kepler: sebutkan tiga usikan lalu
        katakan bagaimana sistemnya mengatasinya.
  \item Ringkaslah rangkaiannya: dua hukum kekekalan, satu orbit, satu
        konstelasi, dan satu koreksi yang takkan diduga siapa pun.
\end{enumerate}
\end{problem}
