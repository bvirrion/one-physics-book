\chapter{Tapis dan Fungsi Alih}\label{ch:b1:filters-transfer-functions}

Di dalam kotak sebuah pengeras suara, sebuah kumparan dan sebuah
kapasitor memutuskan, tanpa satu pun transistor, bahwa nada rendah pergi
ke kerucut besar dan nada tinggi ke kubah kecil. Tombol ``AC'' sebuah
osiloskop menyingkirkan simpangan tetap yang menghanyut perlahan dan
menyisakan geliat di atasnya. Sebuah catu daya mengubah \omterm{def:b1:dc-circuits:voltage}{tegangan}
jaringan yang telah disearahkan dan masih berdenyut menjadi \omterm{def:b1:dc-circuits:voltage}{tegangan}
yang mantap dengan satu kapasitor. Ketiganya adalah \emph{\omterm{def:b1:filters-transfer-functions:H}{tapis}}:
\omterm{def:b1:dc-circuits:linear}{rangkaian linear} yang memperlakukan tiap frekuensi secara berbeda.
Karena setiap \omterm{def:b1:wave-propagation:signal}{sinyal} berkala merupakan jumlah sinusoid, mengetahui apa
yang dilakukan sebuah rangkaian terhadap tiap sinusoid --- \omterm{def:b1:filters-transfer-functions:H}{fungsi
alihnya} --- berarti mengetahui apa yang dilakukannya terhadap segalanya.
Bab ini mendefinisikan \omterm{def:b1:filters-transfer-functions:H}{fungsi alih}, melukis \omterm{def:b1:filters-transfer-functions:bode}{diagram Bode-nya} untuk
rangkaian orde pertama dan orde kedua bab sebelumnya, lalu memakainya
untuk meramalkan bagaimana gelombang persegi keluar dari sebuah \omterm{def:b1:filters-transfer-functions:H}{tapis}.

\section{Fungsi alih dan diagram Bode}

\begin{definition}[Fungsi alih]\label{def:b1:filters-transfer-functions:H}
\index{fungsi alih}\index{penguatan}\index{tapis}
\omterm{def:b1:dc-circuits:linear}{Rangkaian linear} dengan \omterm{def:b1:dc-circuits:voltage}{tegangan} masukan $\underline U_e$ dan \omterm{def:b1:dc-circuits:voltage}{tegangan}
keluaran $\underline U_s$ (yang diambil tanpa apa pun tersambung pada
keluarannya) mempunyai \emph{fungsi alih}
\[
\underline H(j\omega) = \frac{\underline U_s}{\underline U_e},
\qquad G(\omega) = \abs{\underline H}, \qquad \varphi(\omega) = \arg\underline H,
\]
yaitu \emph{penguatan} $G$ (nisbah \omterm{def:b1:wave-propagation:signal}{amplitudo}) dan pergeseran fase
$\varphi$ keluarannya relatif terhadap masukannya. \emph{Tapis} adalah
rangkaian semacam itu yang dipakai justru karena kebergantungannya pada
frekuensi: sinusoid beramplitudo $E$ pada $\omega$ keluar sebagai
$G(\omega)E\cos(\omega t + \varphi(\omega))$.
\end{definition}

\begin{definition}[Desibel; diagram Bode]\label{def:b1:filters-transfer-functions:bode}
\index{desibel}\index{diagram Bode}\index{frekuensi potong}\index{pita lolos}
\omterm{def:b1:filters-transfer-functions:H}{Penguatan} dalam \emph{desibel} adalah $G_{\mathrm{dB}} = 20\log_{10}G$:
faktor $10$ adalah \qty{20}{dB}, faktor $2$ adalah \qty{6}{dB},
$1/\sqrt2$ adalah \qty{-3}{dB}. \emph{Diagram Bode} menggambarkan
$G_{\mathrm{dB}}$ dan $\varphi$ terhadap $\log_{10}\omega$ (atau $f$).
\emph{Frekuensi potong}nya adalah tempat $G$ jatuh ke $G_{\max}/\sqrt2$
(\qty{-3}{dB} dari maksimumnya); \emph{pita lolos}nya adalah rentang
frekuensi yang berada dalam \qty{3}{dB} dari maksimumnya. Satu dekade
adalah faktor $10$ pada frekuensinya, satu oktaf adalah faktor $2$.
\end{definition}

\begin{method}[Sifat sebuah tapis pada pandangan pertama]\label{met:b1:filters-transfer-functions:nature}
Sebelum menghitung, lihatlah limitnya: pada frekuensi yang sangat rendah
kapasitor adalah rangkaian terbuka dan kumparan adalah kawat; pada
frekuensi yang sangat tinggi sebaliknya. Gantilah masing-masing lalu
bacalah apakah keluarannya sama dengan masukannya (lolos) atau nol
(terhalang). Meloloskan yang rendah, menghalangi yang tinggi: lolos
rendah; sebaliknya: lolos tinggi; menghalangi keduanya: lolos pita;
meloloskan keduanya: tolak pita. Lalu hitunglah $\underline H$ dengan
\omterm{prop:b1:dc-circuits:dividers}{pembagi tegangan}.
\end{method}

\section{Tapis orde pertama}

\begin{proposition}[Lolos rendah orde pertama]\label{prop:b1:filters-transfer-functions:lowpass1}
\index{tapis lolos rendah}\index{pengintegral}
Pembagi RC dengan keluaran pada $C$ (atau pembagi RL dengan keluaran
pada $R$) mempunyai
\[
\underline H = \frac{1}{1 + j\omega/\omega_c}, \qquad
\omega_c = \frac{1}{RC} \ \Big(\text{atau } \frac{R}{L}\Big),
\qquad G = \frac{1}{\sqrt{1 + (\omega/\omega_c)^2}}, \qquad
\varphi = -\arctan\frac{\omega}{\omega_c} .
\]
Asimtot Bode: $G_{\mathrm{dB}} = 0$ untuk $\omega \ll \omega_c$, sebuah
garis lurus yang jatuh \qty{20}{dB} tiap dekade untuk
$\omega \gg \omega_c$, yang bertemu pada $\omega_c$ tempat kurva
sejatinya berada di \qty{-3}{dB} dan $\varphi = -45^\circ$. Untuk
$\omega \gg \omega_c$, $\underline H \approx
\omega_c/(j\omega)$: keluarannya sebanding dengan \emph{integral}
masukannya --- tapisnya menjadi pengintegral di sana.
\end{proposition}

\begin{proof}
\omterm{prop:b1:dc-circuits:dividers}{Pembagi tegangan}:
\[
\underline H = \frac{1/jC\omega}{R + 1/jC\omega} = \frac{1}{1 + jRC\omega} .
\]
Jauh di atas \omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya}, $1 + j\omega/\omega_c \approx j\omega/\omega_c$,
dan membagi \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} dengan $j\omega$ berarti mengintegralkan
(\cref{prop:b1:sinusoidal-impedance:rules}): $u_s \approx \omega_c\int u_e\,\dd t$.
$20\log(\omega_c/\omega)$ kehilangan \qty{20}{dB} tiap $\omega$ berlipat sepuluh.
\end{proof}

\begin{proposition}[Lolos tinggi orde pertama]\label{prop:b1:filters-transfer-functions:highpass1}
\index{tapis lolos tinggi}\index{diferensiator}
Pembagi CR dengan keluaran pada $R$ (atau LR dengan keluaran pada $L$)
mempunyai
\[
\underline H = \frac{j\omega/\omega_c}{1 + j\omega/\omega_c},
\qquad G = \frac{\omega/\omega_c}{\sqrt{1 + (\omega/\omega_c)^2}},
\qquad \varphi = \frac{\pi}{2} - \arctan\frac{\omega}{\omega_c},
\]
yang naik \qty{20}{dB} tiap dekade di bawah $\omega_c$, mendatar di
atasnya, dan bernilai \qty{-3}{dB} serta $+45^\circ$ pada $\omega_c$.
Untuk $\omega \ll \omega_c$, $\underline H
\approx j\omega/\omega_c$: keluarannya sebanding dengan
\emph{turunan} masukannya.
\end{proposition}

\begin{proof}
$\underline H = R/(R + 1/jC\omega) = jRC\omega/(1 + jRC\omega)$;
mengalikan dengan $j\omega$ berarti menurunkan.
\end{proof}

Masukan AC sebuah osiloskop adalah \omterm{prop:b1:filters-transfer-functions:highpass1}{tapis lolos tinggi} dengan
$f_c \approx \qty{10}{Hz}$: \omterm{def:b1:wave-propagation:signal}{sinyal} \qty{1}{kHz} yang menumpang pada

harmonik rendah \omterm{def:b1:wave-propagation:signal}{sinyalnya} bergeser fase
(\cref{exo:b1:filters-transfer-functions:12}).
\end{example}

\section{Tapis orde kedua}

\begin{proposition}[RLC seri sebagai tiga tapis]\label{prop:b1:filters-transfer-functions:rlc}
\index{tapis lolos pita}\index{tapis orde kedua}
Dengan $x = \omega/\omega_0$, $\omega_0 = 1/\sqrt{LC}$ dan $Q =
\sqrt{L/C}/R$, RLC seri yang diberi $\underline U_e$ memberikan, menurut
tempat keluarannya diambil:
\begin{align*}
\text{pada } C:&\quad \underline H = \frac{1}{1 - x^2 + jx/Q}, \\
\text{pada } R:&\quad \underline H = \frac{jx/Q}{1 - x^2 + jx/Q} = \frac{1}{1 + jQ(x - 1/x)}, \\
\text{pada } L:&\quad \underline H = \frac{-x^2}{1 - x^2 + jx/Q}:
\end{align*}
sebuah lolos rendah, sebuah lolos pita dan sebuah lolos tinggi orde
kedua. Jauh dari $\omega_0$, lolos rendah dan lolos tingginya jatuh
\qty{40}{dB} tiap dekade, sedangkan lolos pitanya \qty{20}{dB} tiap
dekade pada tiap sisinya; lolos pitanya bermaksimum $1$ pada $\omega_0$
dan berlebar pita $\Delta\omega =
\omega_0/Q$ menurut \cref{thm:b1:sinusoidal-impedance:resonance}.
\end{proposition}

\begin{proof}
\omterm{prop:b1:dc-circuits:dividers}{Pembagi tegangan} dengan \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} total $R + j(L\omega - 1/C\omega)$;
kalikan pembilang dan penyebutnya dengan $jC\omega$ lalu pakai
$LC\omega_0^2 = 1$, $RC\omega_0 = 1/Q$. Lolos pitanya adalah resonansi
arus bab sebelumnya yang dibaca sebagai \omterm{def:b1:dc-circuits:voltage}{tegangan} pada $R$.
\end{proof}

\begin{proposition}[Lolos rendah orde kedua: peranan $Q$]\label{prop:b1:filters-transfer-functions:lowpass2}
\index{resonansi!sebuah tapis}
Untuk lolos rendahnya, $G = 1/\sqrt{(1 - x^2)^2 + x^2/Q^2}$: untuk
$Q > 1/\sqrt2$ \omterm{def:b1:filters-transfer-functions:H}{penguatannya} naik di atas $1$ di dekat $\omega_0$
(resonansi, puncaknya $\approx Q$ untuk $Q$ yang besar); untuk
$Q = 1/\sqrt2$ ia ``sedatar mungkin'', $G = 1/\sqrt{1 + x^4}$, tepat
\qty{-3}{dB} pada $\omega_0$; untuk $Q$ yang lebih kecil ia melandai
lebih awal. Fasenya berjalan dari $0$ ke $-\pi$, melalui $-\pi/2$ pada $\omega_0$.
\end{proposition}

\begin{proof}
$(1 - x^2)^2 + x^2/Q^2 = 1 + x^4 + x^2(1/Q^2 - 2)$: suku tengahnya
lenyap untuk $Q^2 = \tfrac12$, negatif (puncak) di atasnya, positif
(landai) di bawahnya; perilakunya di dekat $x = 1$ telah dianalisis
dalam \cref{prop:b1:sinusoidal-impedance:overvoltage}.
\end{proof}

\begin{example}[Sebuah crossover]\label{ex:b1:filters-transfer-functions:crossover}
Woofer dan tweeter sebuah pengeras suara, masing-masing \qty{8}{\ohm},
berbagi penguat lewat sebuah lolos rendah (kumparan seri) dan sebuah
lolos tinggi (kapasitor seri) yang dipotong pada \qty{2}{kHz}:
$L = R/\omega_c = \qty{0.64}{mH}$, $C = 1/R\omega_c = \qty{10}{\micro F}$.
Pada \qty{2}{kHz} tiap penggeraknya memperoleh \qty{-3}{dB}, yaitu
separuh dayanya, dan $G_L^2 + G_H^2 = 1$ pada setiap frekuensi: dayanya
dibagi, bukan hilang. Soal akhir pekan merancang versi yang lebih curam.
\end{example}

\section{Menapis sinyal berkala}

\begin{theorem}[Penguraian Fourier]\label{thm:b1:filters-transfer-functions:fourier}
\index{deret Fourier}\index{harmonik}\index{spektrum!sinyal berkala}
\omterm{def:b1:wave-propagation:signal}{Sinyal} berkala berperiode $T = 2\pi/\omega$ (yang cukup teratur)
merupakan jumlah sebuah tetapan dan sinusoid pada kelipatan $\omega$:
\[
s(t) = c_0 + \sum_{n = 1}^{\infty} c_n\cos(n\omega t + \varphi_n),
\]
dengan $c_0 = \langle s\rangle$ sebagai nilai rata-ratanya; suku $n = 1$
adalah \emph{nada dasar}nya, sisanya \emph{harmonik}nya, dan \omterm{def:b1:wave-propagation:signal}{amplitudo}
$c_n$ membentuk \emph{spektrum}nya. Gelombang persegi beramplitudo
$\pm E$ mempunyai $c_n = 4E/(n\pi)$ untuk $n$ ganjil dan $0$ untuk $n$
genap; gelombang segitiga beramplitudo $\pm E$ mempunyai
$c_n = 8E/(n\pi)^2$ untuk $n$ ganjil.
\end{theorem}
\admitted

\begin{remark}[Apa yang perlu diambil darinya]\label{rem:b1:filters-transfer-functions:fourier}
Teoremanya dan perhitungan $c_n$ termasuk kuliah matematika Tahun ke-2.
Di sini ia sebuah alat: sudut yang tajam menuntut banyak harmonik
(harmonik gelombang persegi hanya turun sebagai $1/n$), \omterm{def:b1:wave-propagation:signal}{sinyal} yang
mulus menuntut sedikit (harmonik gelombang segitiga turun sebagai
$1/n^2$). Kandungan harmonik sebuah \omterm{def:b1:wave-propagation:signal}{sinyal} adalah bahan warna bunyi
sebuah alat musik, dan itulah yang dikerjai sebuah \omterm{def:b1:filters-transfer-functions:H}{tapis}.
\end{remark}

\begin{proposition}[Penapisan linear sinyal berkala]\label{prop:b1:filters-transfer-functions:filtering}
Lewat \omterm{def:b1:filters-transfer-functions:H}{tapis} berfungsi alih $\underline H$, \omterm{def:b1:wave-propagation:signal}{sinyal} pada teorema itu menjadi
\[
s_{\text{keluar}}(t) = G(0)\,c_0 + \sum_{n \geq 1} G(n\omega)\,c_n\cos\!\big(n\omega t + \varphi_n + \varphi(n\omega)\big):
\]
tiap harmoniknya diskalakan oleh \omterm{def:b1:filters-transfer-functions:H}{penguatan} dan digeser oleh fase pada
frekuensinya \emph{sendiri}, lalu potongannya dijumlahkan. Maka:
\begin{itemize}
  \item lolos rendah dengan $\omega_c \ll \omega$ pada dasarnya hanya
        menyimpan $c_0$: ia \emph{merata-ratakan} (memuluskan,
        menyingkirkan riak), dan mengubah gelombang persegi menjadi
        segitiga kecil (pengintegralan);
  \item lolos tinggi dengan $\omega_c \ll \omega$ menyingkirkan $c_0$
        lalu meloloskan sisanya (gandengan AC); dengan
        $\omega_c \gg \omega$ ia menurunkan (paku pada tepi gelombang
        persegi);
  \item lolos pita yang dilaraskan ke $n\omega$ dengan $Q \gg 1$
        menyarikan harmonik ke-$n$ saja: sebuah sinusoid dari sebuah
        gelombang persegi.
\end{itemize}
\end{proposition}

\begin{proof}
Superposisi untuk \omterm{def:b1:dc-circuits:linear}{rangkaian linear} (\cref{thm:b1:dc-circuits:superposition})
yang diterapkan pada jumlah masukan sinusoidalnya, masing-masing
diperlakukan menurut \cref{def:b1:filters-transfer-functions:H}.
\end{proof}

\begin{example}[Angka-angkanya]\label{ex:b1:filters-transfer-functions:numbers}
Gelombang persegi \qty{1}{kHz} beramplitudo \qty{5}{V}: harmoniknya
\qty{6.37}{V} pada \qty{1}{kHz}, \qty{2.12}{V} pada \qty{3}{kHz},
\qty{1.27}{V} pada \qty{5}{kHz}. Lewat lolos rendah RC yang dipotong
pada \qty{10}{Hz} ($\tau = \qty{16}{ms}$) keluarannya adalah segitiga
beramplitudo $ET/4\tau \approx \qty{80}{mV}$ di sekitar nol; lewat
lolos pita yang dilaraskan ke \qty{3}{kHz} dengan $Q = 20$, sebuah
sinusoid \qty{2.1}{V} pada \qty{3}{kHz} dengan hanya beberapa persen
tetangganya (\cref{exo:b1:filters-transfer-functions:8}).
\end{example}

\section{Menderetkan tapis}

\begin{proposition}[Deretan dan pembebanan]\label{prop:b1:filters-transfer-functions:cascade}
\index{deretan (tapis)}\index{pembebanan (tapis)}
Dua \omterm{def:b1:filters-transfer-functions:H}{tapis} yang berderet mempunyai $\underline H = \underline H_1\underline H_2$
--- \omterm{def:b1:filters-transfer-functions:H}{penguatannya} berkalian, \omterm{def:b1:filters-transfer-functions:bode}{desibel} dan fasenya berjumlahan ---
\emph{asalkan yang kedua tidak membebani yang pertama}, yaitu menarik
arus yang dapat diabaikan dari keluarannya. Bila tidak, hubungan
pembaginya harus ditulis ulang untuk jaringan gabungannya. Sebuah
pengikut penguat operasional
(\cref{ch:b1:operational-amplifier}) di antara tingkatnya membuat
kaidah perkalian itu eksak.
\end{proposition}

\begin{proof}
$\underline H_1$ didefinisikan dengan keluaran terbuka; jika \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi}
masukan tingkat berikutnya jauh lebih besar daripada \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} keluaran
tingkat pertamanya (hambatan Thévenin-nya), \omterm{def:b1:dc-circuits:voltage}{tegangan} keluarannya tak
berubah dan tingkat kedua melihat tepat $\underline U_{s1}$ sebagai masukannya.
\end{proof}

\begin{example}[Dua tingkat RC]\label{ex:b1:filters-transfer-functions:tworc}
Dua lolos rendah RC yang serupa dan bersambungan \emph{tidak} memberi
$1/(1 + j\omega/\omega_c)^2$: $R$ tingkat kedua membebani yang pertama.
Hasil eksaknya adalah $\underline H = 1/[1 - (\omega/\omega_c)^2 +
3j\omega/\omega_c]$, yaitu \qty{-9.5}{dB} pada $\omega_c$ alih-alih
\qty{-6}{dB} yang dijanjikan kaidah perkaliannya, walaupun keduanya
pada akhirnya mencapai kemiringan \qty{-40}{dB/dekade} yang sama.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:filters-transfer-functions:1}
Ubahlah menjadi \omterm{def:b1:filters-transfer-functions:bode}{desibel}: \omterm{def:b1:filters-transfer-functions:H}{penguatan} $0.5$, $10$, $1/\sqrt2$, $100$.
Ubahlah menjadi nisbah: \qty{-40}{dB}, \qty{+6}{dB}. Dua \omterm{def:b1:filters-transfer-functions:H}{tapis} yang
berderet (tanpa pembebanan) memberi \qty{-6}{dB} dan \qty{-20}{dB} pada
suatu frekuensi: berapa \omterm{def:b1:filters-transfer-functions:H}{penguatan} keseluruhannya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:filters-transfer-functions:2}
Lolos rendah RC, $R = \qty{10}{k\ohm}$, $C = \qty{16}{nF}$: berapa
\omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya}, dan \omterm{def:b1:filters-transfer-functions:H}{penguatannya} dalam \omterm{def:b1:filters-transfer-functions:bode}{desibel} pada \qty{100}{Hz},
\qty{1}{kHz}, \qty{10}{kHz}, \qty{100}{kHz}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:filters-transfer-functions:3}
Tanpa menghitung, berikan sifatnya (lolos rendah, lolos tinggi, lolos
pita) untuk: (a) $R$ terangkai seri, $C$ melintasi keluarannya; (b) $C$
terangkai seri, $R$ melintasi keluarannya; (c) $L$ terangkai seri, $C$
melintasi keluarannya; (d) $L$ dan $C$ terangkai seri, $R$ melintasi
keluarannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:filters-transfer-functions:4}
Gelombang persegi \qty{1.0}{kHz} beramplitudo \qty{5.0}{V}: berikan
frekuensi dan \omterm{def:b1:wave-propagation:signal}{amplitudo} tiga harmonik tak nol pertamanya. Mengapa ia
tak mempunyai harmonik genap?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:filters-transfer-functions:5}
Lolos tinggi CR yang dipotong pada \qty{10}{kHz} menerima gelombang

ia berlaku sebagai diferensiator lalu berikan bentuk dan \omterm{def:b1:wave-propagation:signal}{amplitudo} keluarannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:filters-transfer-functions:6}
Lolos rendah RC yang dipotong pada \qty{10}{Hz} menerima gelombang
persegi \qty{1.0}{kHz} beramplitudo $\pm\qty{1.0}{V}$. Tunjukkan bahwa
keluarannya berupa segitiga lalu hitunglah \omterm{def:b1:wave-propagation:signal}{amplitudonya}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:filters-transfer-functions:7}
Lolos rendah RLC seri dengan $L = \qty{10}{mH}$, $C = \qty{100}{nF}$:
berikan $f_0$; carilah $R$ untuk tanggapan sedatar mungkin; berikan
\omterm{def:b1:filters-transfer-functions:H}{penguatannya} dalam \omterm{def:b1:filters-transfer-functions:bode}{desibel} pada $f_0$ dan pada $10f_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:filters-transfer-functions:8}
Sebuah lolos pita ($Q = 20$) yang dilaraskan ke \qty{3.0}{kHz} menerima

\cref{exo:b1:filters-transfer-functions:4}. Hitunglah \omterm{def:b1:filters-transfer-functions:H}{penguatannya} pada

gambarkan keluarannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:filters-transfer-functions:9}
Dua lolos rendah RC yang serupa ($R = \qty{1.0}{k\ohm}$, $C = \qty{159}{nF}$)

pertama) dan \omterm{def:b1:filters-transfer-functions:H}{penguatannya} pada $\omega_c = 1/RC$; bandingkan dengan

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:filters-transfer-functions:10}

\begin{exercise}[$\star\star\star$]\label{exo:b1:filters-transfer-functions:11}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:filters-transfer-functions:12}
Gandengan AC sebuah osiloskop adalah lolos tinggi dengan $C = \qty{0.10}{\micro F}$

\end{exercise}

\section{Soal: Crossover pengeras suara}

\begin{problem}[{Soal akhir pekan --- sebuah woofer, sebuah tweeter,
sebuah kumparan dan sebuah kapasitor: bagaimana crossover pasif membelah
musiknya, mengapa yang paling sederhana membocorkan nada rendah ke
tweeter, dan apa yang dilakukan kumparan suara yang sesungguhnya
terhadap rancangannya}]
\label{pb:b1:filters-transfer-functions:1}
Woofer dan tweeternya mula-mula dimodelkan sebagai hambatan murni
$R = \qty{8.0}{\ohm}$. Penguatnya adalah \omterm{def:b1:dc-circuits:sources}{sumber tegangan} ideal
$\underline U_e$. Frekuensi silangnya $f_c = \qty{2.0}{kHz}$.

\medskip
\noindent\textbf{Bagian I --- Crossover orde pertama.}
Woofernya diberi daya lewat kumparan seri $L$, dan tweeternya lewat
kapasitor seri $C$.
\begin{enumerate}
  \item Definisikan \omterm{def:b1:filters-transfer-functions:H}{fungsi alih} tiap cabangnya (\omterm{def:b1:dc-circuits:voltage}{tegangan} pada
        penggeraknya dibagi $\underline U_e$) dan \omterm{def:b1:filters-transfer-functions:H}{penguatannya} dalam \omterm{def:b1:filters-transfer-functions:bode}{desibel}.
  \item Tunjukkan bahwa cabang woofernya adalah lolos rendah orde
        pertama lalu berikan \omterm{def:b1:filters-transfer-functions:bode}{frekuensi potongnya} $\omega_c$ dalam $R$ dan $L$.
  \item Hitunglah $L$ untuk $f_c = \qty{2.0}{kHz}$.
  \item Tunjukkan bahwa cabang tweeternya adalah lolos tinggi orde
        pertama; berikan $\omega_c$ lalu hitunglah $C$.
  \item Berikan \omterm{def:b1:filters-transfer-functions:H}{penguatan} tiap penggeraknya dalam \omterm{def:b1:filters-transfer-functions:bode}{desibel} pada $f_c$,
        pada \qty{200}{Hz} dan pada \qty{20}{kHz}.
  \item Tunjukkan bahwa $G_L^2 + G_H^2 = 1$ pada setiap frekuensi. Apa
        artinya bagi daya yang ditarik dari penguatnya?
  \item Berikan fase tiap cabangnya pada $f_c$; seberapa besar \omterm{def:b1:dc-circuits:voltage}{tegangan}
        kedua penggeraknya berbeda fase di sana?
  \item Berapa kemiringan kedua tanggapannya, dalam dB tiap oktaf, jauh
        dari $f_c$?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Crossover orde kedua.}
Woofernya kini diberi lewat kumparan seri $L$ dengan kapasitor $C$ yang
terangkai paralel dengan penggeraknya.
\begin{enumerate}[resume]
  \item Tunjukkan bahwa $\underline H = 1/(1 - LC\omega^2 + jL\omega/R)$.
  \item Kenali $\omega_0$ dan $Q$ dalam $L$, $C$, $R$.
  \item Untuk $Q = 1/\sqrt2$ tunjukkan bahwa $G = 1/\sqrt{1 + (\omega/\omega_0)^4}$
        dan bahwa \omterm{def:b1:filters-transfer-functions:H}{penguatannya} \qty{-3}{dB} pada $\omega_0$.
  \item Hitunglah $L$ dan $C$ untuk $f_0 = \qty{2.0}{kHz}$ dan $Q = 1/\sqrt2$.
  \item Hitunglah \omterm{def:b1:filters-transfer-functions:H}{penguatannya} dalam \omterm{def:b1:filters-transfer-functions:bode}{desibel} pada \qty{4}{kHz} dan
        \qty{8}{kHz}; simpulkan kemiringannya dalam dB tiap oktaf.
  \item Cabang tweeternya menukar peran $L$ dan $C$ ($C$ seri, $L$
        paralel): tuliskan \omterm{def:b1:filters-transfer-functions:H}{fungsi alihnya} tanpa perhitungan baru.
  \item Hitunglah fase tiap cabangnya pada $f_0$. Apa yang harus
        dilakukan terhadap pengawatan tweeternya, dan mengapa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Musik lewat crossover-nya.}
\begin{enumerate}[resume]
  \item Nada sebuah selo pada \qty{220}{Hz} membawa harmonik sampai
        harmonik ke-20. Harmonik mana yang sebagian besar pergi ke
        tweeternya, dengan crossover orde pertama?
  \item Apa yang terjadi pada harmonik ke-9?
  \item Gelombang persegi \qty{1.0}{kHz} dipakai sebagai \omterm{def:b1:wave-propagation:signal}{sinyal} uji.
        Dengan crossover orde pertama, berikan \omterm{def:b1:filters-transfer-functions:H}{penguatan} tiap cabangnya
        untuk nada dasarnya serta harmonik ketiga dan kelimanya.
  \item Penguatnya secara tak sengaja menimbulkan simpangan searah
        \qty{1}{V}. Penggerak mana yang terlindungi, dan oleh apa?
  \item Dengan crossover orde pertama, berapa bagian \omterm{def:b1:dc-circuits:voltage}{tegangan} dan berapa
        bagian daya sebuah nada \qty{500}{Hz} yang mencapai tweeternya?
        Sama untuk yang orde kedua. Mengapa hal ini penting bagi
        keselamatan tweeternya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Kumparan suara yang sesungguhnya.}
Woofernya sesungguhnya berupa $R = \qty{8.0}{\ohm}$ yang terangkai seri
dengan induktansi kumparan suara $L_{vc} = \qty{0.50}{mH}$.
\begin{enumerate}[resume]
  \item Hitunglah \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} woofernya (modulusnya) pada \qty{2}{kHz} dan
        pada \qty{8}{kHz}.
  \item Hitunglah kembali \omterm{def:b1:filters-transfer-functions:H}{penguatan} woofer orde pertamanya pada
        \qty{2}{kHz} dan \qty{8}{kHz} dengan \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} ini, lalu
        bandingkan dengan \qty{-3}{dB} dan \qty{-12.3}{dB} yang ideal.
        Apa yang terjadi pada landaiannya?
  \item Sebuah ``jaringan Zobel'' --- $R_z = R$ yang terangkai seri
        dengan $C_z =
        L_{vc}/R^2$ --- dipasang paralel dengan woofernya.
        Tunjukkan bahwa gabungannya berlaku sebagai hambatan murni $R$
        pada semua frekuensi, lalu hitunglah $C_z$.
  \item Penguatnya menyampaikan \qty{50}{W} ke \qty{8}{\ohm}: berapa
        \omterm{def:b1:dc-circuits:voltage}{tegangan} rmsnya, dan berapa daya yang diterima tiap penggeraknya
        pada $f_c$ (crossover orde pertama)? Pada \qty{500}{Hz}, berapa
        yang mencapai tweeternya?
  \item Ringkaslah tukar-tambahnya: kemiringan, fase, jumlah komponen,
        perlindungan tweeternya, dan perlunya jaringan Zobel.
\end{enumerate}
\end{problem}
