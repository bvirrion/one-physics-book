\chapter{Keadaan Tunak Sinusoidal dan Impedansi}\label{ch:b1:sinusoidal-impedance}

Putarlah tombol sebuah radio tua dan, dari seratus stasiun yang
semuanya mencapai antenanya sekaligus, satu saja yang tembus: sebuah
kumparan dan sebuah kapasitor variabel telah dilaraskan sehingga tepat
satu frekuensi membuat keduanya beresonansi. Jaringan listrik yang
berdengung di setiap dinding adalah sinusoid \qty{50}{Hz}; pabrik yang
menjalankan motor besar dari jaringan itu membayar denda bila arusnya
terlalu jauh tertinggal di belakang \omterm{def:b1:dc-circuits:voltage}{tegangannya}. Sinusoid adalah bahasa alami
\omterm{def:b1:dc-circuits:linear}{rangkaian linear} --- ia keluar sebagaimana ia masuk, hanya terskalakan
dan tergeser --- dan bab ini memberinya aljabar: \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks},
\omterm{def:b1:sinusoidal-impedance:impedance}{impedansi}, resonansi \omterm{prop:b1:transient-regimes:rlc}{rangkaian RLC}, dan daya yang sungguh-sungguh
disampaikan arus sinusoidal.

\section{Keadaan tunak sinusoidal}

\begin{proposition}[Rezim sinusoidal paksa]\label{prop:b1:sinusoidal-impedance:forced}
\index{keadaan tunak sinusoidal}\index{rezim paksa}
Dalam \omterm{def:b1:dc-circuits:linear}{rangkaian linear} yang digerakkan sumber sinusoidal berfrekuensi
sudut $\omega$, setiap arus dan \omterm{def:b1:dc-circuits:voltage}{tegangan}, begitu peralihannya padam,
merupakan sinusoid berfrekuensi sudut $\omega$ yang sama, masing-masing
dengan \omterm{def:b1:wave-propagation:signal}{amplitudo} dan fasenya sendiri: $x(t) = X\cos(\omega t + \varphi)$.
\end{proposition}

\begin{proof}
Persamaan rangkaiannya adalah persamaan diferensial linear berkoefisien
tetap; penyelesaian umumnya adalah satu penyelesaian khusus ditambah
penyelesaian persamaan homogennya, yang merupakan peralihan
\cref{ch:b1:transient-regimes} dan meluruh (setiap rangkaian nyata
mempunyai sedikit hambatan). Penyelesaian khusus yang sinusoidal ada
karena menurunkan sinusoid berfrekuensi $\omega$ menghasilkan sinusoid
berfrekuensi sama: persamaannya dapat dipenuhi suku demi suku ---
metode kompleks di bawah ini membangunnya secara gamblang.
\end{proof}

\begin{definition}[Amplitudo kompleks]\label{def:b1:sinusoidal-impedance:complex}
\index{amplitudo kompleks}\index{fasor}
Kepada sinusoid $x(t) = X\cos(\omega t + \varphi)$ orang mengaitkan
\omterm{def:b1:wave-propagation:signal}{sinyal} kompleks $\underline{x}(t) = X\eu^{j(\omega t + \varphi)}$, sehingga
$x = \Rea(\underline{x})$, dan \emph{\omterm{def:b1:wave-propagation:signal}{amplitudo} kompleks}
\[
\underline{X} = X\eu^{j\varphi}, \qquad \underline{x}(t) = \underline{X}\,\eu^{j\omega t},
\]
yang membawa \omterm{def:b1:wave-propagation:signal}{amplitudo} $X = \abs{\underline{X}}$ dan fase
$\varphi = \arg\underline{X}$. Di sini $j$ menyatakan satuan imajiner,
$j^2 = -1$, sedangkan huruf $i$ disediakan bagi arus. Digambar sebagai
vektor sepanjang $X$ bersudut $\varphi$, $\underline{X}$ adalah sebuah
\emph{fasor} (vektor Fresnel).
\end{definition}

\begin{proposition}[Kaidah metode kompleks]\label{prop:b1:sinusoidal-impedance:rules}
\begin{enumerate}
  \item Operasi linear berkomutasi dengan pengambilan bagian real:
        \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} sebuah jumlah adalah jumlah \omterm{def:b1:wave-propagation:signal}{amplitudo}
        kompleksnya.
  \item Menurunkan terhadap waktu mengalikan \omterm{def:b1:wave-propagation:signal}{amplitudo} kompleksnya
        dengan $j\omega$; mengintegralkan membaginya dengan $j\omega$.
  \item \omterm{thm:b1:dc-circuits:kirchhoff}{Hukum Kirchhoff}, karena linear, berlaku bagi \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks}.
\end{enumerate}
\end{proposition}

\begin{proof}
$\Rea(\underline a + \underline b) = \Rea\underline a + \Rea\underline b$;
$\dd(\underline X\eu^{j\omega t})/\dd t = j\omega\underline X\eu^{j\omega t}$,
dan $\dd x/\dd t = \Rea(\dd\underline x/\dd t)$ karena penurunan bersifat
linear-real. Jumlah arus di sebuah simpul dan jumlah \omterm{def:b1:dc-circuits:voltage}{tegangan}
mengelilingi sebuah loop bersifat linear.
\end{proof}

\begin{example}[Menjumlahkan dua sinusoid]\label{ex:b1:sinusoidal-impedance:add}
$3\cos\omega t + 4\sin\omega t$: \omterm{def:b1:wave-propagation:signal}{amplitudo} kompleksnya $3$ dan
$4\eu^{-j\pi/2} = -4j$ (karena $\sin\omega t = \cos(\omega t - \pi/2)$);
jumlahnya $3 - 4j$ bermodulus $5$ dan berargumen $-0.927\ \unit{rad}$:
$5\cos(\omega t - 0.927)$. Tanpa perlu satu pun identitas trigonometri.
\end{example}

\section{Impedansi}

\begin{definition}[Impedansi, admitansi]\label{def:b1:sinusoidal-impedance:impedance}
\index{impedansi}\index{admitansi}\index{reaktansi}
Untuk \omterm{def:b1:dc-circuits:linear}{dipol linear} dalam \omterm{prop:b1:sinusoidal-impedance:forced}{keadaan tunak sinusoidal} (\omterm{thm:b1:dc-circuits:kirchhoff}{kesepakatan
penerima}), \emph{impedansi}nya adalah nisbah \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks}
\[
\underline Z = \frac{\underline U}{\underline I} = Z\eu^{j\varphi}
\quad (\text{ohm}),
\]
yang modulusnya $Z = U/I$ menghubungkan \omterm{def:b1:wave-propagation:signal}{amplitudonya} dan yang argumennya
$\varphi = \varphi_u - \varphi_i$ merupakan keunggulan fase \omterm{def:b1:dc-circuits:voltage}{tegangan}
terhadap arusnya. Kebalikannya $\underline Y = 1/\underline Z$ adalah
\emph{admitansi} (siemens). Bagian real $\underline Z$ adalah
hambatannya, bagian imajinernya adalah \emph{reaktansi}nya.
\end{definition}

\begin{proposition}[Impedansi R, L, C]\label{prop:b1:sinusoidal-impedance:rlc}
\[
\underline Z_R = R, \qquad \underline Z_L = jL\omega, \qquad
\underline Z_C = \frac{1}{jC\omega} = -\frac{j}{C\omega} .
\]
\omterm{def:b1:dc-circuits:resistor}{Resistor} menjaga \omterm{def:b1:dc-circuits:voltage}{tegangan} dan arusnya sefase; pada kumparan \omterm{def:b1:dc-circuits:voltage}{tegangannya}
mendahului arusnya sebesar $\pi/2$; pada kapasitor ia tertinggal
sebesar $\pi/2$. Pada frekuensi rendah kumparan cenderung menjadi kawat
dan kapasitor menjadi rangkaian terbuka; pada frekuensi tinggi sebaliknya.
\end{proposition}

\begin{proof}
$u = Ri$; $u = L\,\dd i/\dd t \to \underline U = jL\omega\,\underline I$;
$i = C\,\dd u/\dd t \to \underline I = jC\omega\,\underline U$. Argumen
$1$, $j$ dan $-j$ adalah $0$, $\pi/2$, $-\pi/2$.
\end{proof}

\begin{proposition}[Perangkaian; pembagi; teorema]\label{prop:b1:sinusoidal-impedance:assoc}
\omterm{def:b1:sinusoidal-impedance:impedance}{Impedansi} yang terangkai seri saling menjumlah; \omterm{def:b1:sinusoidal-impedance:impedance}{admitansi} yang terangkai
paralel saling menjumlah; \omterm{prop:b1:dc-circuits:dividers}{pembagi tegangan} dan \omterm{prop:b1:dc-circuits:dividers}{pembagi arus}, teorema
Thévenin dan Norton, superposisi serta teorema Millman
\cref{ch:b1:dc-circuits} semuanya berlaku dengan \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} dan
\omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} menggantikan \omterm{def:b1:dc-circuits:voltage}{tegangan}, arus dan hambatan yang real.
\end{proposition}

\begin{proof}
Setiap bukti dalam \cref{ch:b1:dc-circuits} hanya memakai \omterm{thm:b1:dc-circuits:kirchhoff}{hukum
Kirchhoff} dan kelinearan $u = Ri$, yang keduanya berlaku dalam bentuk kompleks.
\end{proof}

\begin{example}[Pembagi RC]\label{ex:b1:sinusoidal-impedance:rcdivider}
\omterm{def:b1:dc-circuits:resistor}{Resistor} $R$ dan kapasitor $C$ yang terangkai seri melintasi
$\underline E$, dengan keluarannya diambil pada $C$:
$\underline U_C = \underline E\,\dfrac{1/jC\omega}
{R + 1/jC\omega} = \dfrac{\underline E}{1 + jRC\omega}$. \omterm{def:b1:wave-propagation:signal}{Amplitudonya}
$E/\sqrt{1 + (RC\omega)^2}$, fasenya $-\arctan(RC\omega)$: teruskan
seluruhnya pada frekuensi rendah, $1/\sqrt2$ dan $-45^\circ$ pada
$\omega = 1/RC$, lalu jatuh sebagai $1/\omega$ di atasnya --- sebuah
tapis lolos rendah, pokok bahasan \cref{ch:b1:filters-transfer-functions}.
\end{example}

\section{Rangkaian RLC seri: resonansi}

\begin{theorem}[Resonansi arus]\label{thm:b1:sinusoidal-impedance:resonance}
\index{resonansi}\index{resonansi!arus}\index{lebar pita}
\omterm{def:b1:dc-circuits:resistor}{Resistor} $R$, kumparan $L$ dan kapasitor $C$ yang terangkai seri
melintasi $e(t) = E\cos\omega t$ membawa arus beramplitudo kompleks
\[
\underline I = \frac{E}{R + j\left(L\omega - \dfrac{1}{C\omega}\right)},
\qquad
I = \frac{E}{\sqrt{R^2 + \left(L\omega - \dfrac{1}{C\omega}\right)^2}},
\qquad
\varphi_i = -\arctan\frac{L\omega - 1/C\omega}{R} .
\]
\omterm{def:b1:wave-propagation:signal}{Amplitudonya} maksimum, $I_{\max} = E/R$, dengan arus dan \omterm{def:b1:dc-circuits:voltage}{tegangan}
sefase, pada \emph{resonansi} $\omega_0 = 1/\sqrt{LC}$. Dengan
$Q = L\omega_0/R = 1/(RC\omega_0)$ dan $x = \omega/\omega_0$,
\[
\frac{I}{I_{\max}} = \frac{1}{\sqrt{1 + Q^2\left(x - \dfrac1x\right)^2}} ,
\]
dan \emph{\omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pita}}nya --- selang $\omega$ tempat
$I \geq I_{\max}/\sqrt2$ --- adalah
\[
\Delta\omega = \omega_2 - \omega_1 = \frac{\omega_0}{Q} = \frac{R}{L} .
\]
\end{theorem}

\begin{proof}
\omterm{def:b1:sinusoidal-impedance:impedance}{Impedansi} seri saling menjumlah: $\underline Z = R + jL\omega + 1/jC\omega$, dan
$\underline I = \underline E/\underline Z$. Modulusnya terbesar ketika
\omterm{def:b1:sinusoidal-impedance:impedance}{reaktansi} $L\omega - 1/C\omega$ lenyap, yaitu pada $\omega_0$.
Faktorkan $L\omega - 1/C\omega = L\omega_0(x - 1/x)$ lalu bagi dengan $R$:
$L\omega_0/R = Q$. Pada tepi pitanya, $Q(x - 1/x) = \pm 1$, yaitu
$x^2 \mp x/Q - 1 = 0$, yang akar positifnya $x_{1,2} = \mp\frac{1}{2Q}
+ \sqrt{1 + \frac{1}{4Q^2}}$; selisihnya adalah $1/Q$.
\end{proof}

\begin{proposition}[Penguatan tegangan]\label{prop:b1:sinusoidal-impedance:overvoltage}
\index{resonansi!tegangan}\index{tegangan lebih}
Pada resonansi, \omterm{def:b1:dc-circuits:voltage}{tegangan} pada kumparan dan pada kapasitor beramplitudo
sama $QE$ dan berlawanan fase: untuk $Q \gg 1$ kapasitornya membawa
\omterm{def:b1:dc-circuits:voltage}{tegangan} $Q$ kali lebih besar daripada \omterm{def:b1:dc-circuits:voltage}{tegangan} sumbernya. \omterm{def:b1:wave-propagation:signal}{Amplitudo}
$U_C(\omega)$ sendiri memuncak, untuk $Q > 1/\sqrt2$, pada $\omega_r =
\omega_0\sqrt{1 - 1/2Q^2}$, sedikit di bawah $\omega_0$, dengan
$U_{C,\max} = QE/\sqrt{1 - 1/4Q^2}$.
\end{proposition}

\begin{proof}
Pada $\omega_0$, $\underline U_C = \underline I/jC\omega_0 = -jQE$ dan
$\underline U_L = jL\omega_0\underline I = jQE$ karena $I = E/R$. Secara
umum $U_C = I/C\omega = E/\sqrt{(1 - x^2)^2 + x^2/Q^2}$ (kalikan
pembilang dan penyebutnya dengan $1/C\omega_0$); kuadrat penyebutnya
adalah $(1 - x^2)^2 + x^2/Q^2$, yaitu kuadrat dalam $x^2$ yang minimum pada $x^2 =
1 - 1/2Q^2$ ketika nilai itu positif, dengan minimum $1/Q^2 - 1/4Q^4$.
\end{proof}

\begin{example}[Melaras sebuah radio]\label{ex:b1:sinusoidal-impedance:radio}
Kumparan $L = \qty{200}{\micro H}$ dan kapasitor variabel: $C =
\qty{127}{pF}$ melaraskan $f_0 = \qty{1.00}{MHz}$. Dengan hambatan kawat
kumparannya $R = \qty{12.6}{\ohm}$, $L\omega_0 = \qty{1257}{\ohm}$ dan
$Q = 100$: \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} $\Delta f = f_0/Q = \qty{10}{kHz}$ --- selebar
sebuah kanal siaran modulasi \omterm{def:b1:wave-propagation:signal}{amplitudo} --- dan \qty{10}{\micro V} yang diimbaskan pemancar
jauh menjadi $QE = \qty{1}{mV}$ pada kapasitornya, cuma-cuma. Soal akhir
pekan membangun penerima ini.
\end{example}

\begin{remark}[Mengapa $Q$ lagi]\label{rem:b1:sinusoidal-impedance:Q}
$Q$ bab ini adalah $Q$ \cref{ch:b1:transient-regimes}: rangkaian yang
berdering kira-kira $Q$ osilasi ketika dipukul adalah rangkaian yang
beresonansi pada pita selebar $\omega_0/Q$ ketika digerakkan. Keduanya
menyatakan satu nisbah: $2\pi$ kali energi tersimpan dibagi energi yang
dilesapkan tiap periode (\cref{exo:b1:sinusoidal-impedance:12}).
Resonansi yang tajam dan dering yang panjang adalah sifat yang sama,
begitu pula tanggapan yang lambat: tapis berlebar pita $\Delta\omega$
memerlukan waktu berorde $1/\Delta\omega$ untuk mapan.
\end{remark}

\section{Daya dalam rezim sinusoidal}

\begin{definition}[Nilai rms]\label{def:b1:sinusoidal-impedance:rms}
\index{nilai rms}\index{nilai efektif}
Nilai \emph{akar rata-rata kuadrat} (rms, atau efektif) sebuah \omterm{def:b1:wave-propagation:signal}{sinyal}
berkala $x(t)$ adalah $X_{\mathrm{rms}} = \sqrt{\langle x^2\rangle}$,
yaitu akar kuadrat rata-rata waktu $x^2$ sepanjang satu periode. Untuk
sinusoid beramplitudo $X$, $X_{\mathrm{rms}} = X/\sqrt2$. Angka
\qty{230}{V} jaringan listrik adalah nilai rms; \omterm{def:b1:wave-propagation:signal}{amplitudonya} \qty{325}{V}.
\end{definition}

\begin{theorem}[Daya rata-rata]\label{thm:b1:sinusoidal-impedance:power}
\index{daya rata-rata}\index{faktor daya}\index{daya reaktif}
Dipol dengan $u = U\cos\omega t$ dan $i = I\cos(\omega t - \varphi)$
(\omterm{thm:b1:dc-circuits:kirchhoff}{kesepakatan penerima}) menerima daya sesaat $p = ui$, yang rata-ratanya
\[
P = \tfrac12\,UI\cos\varphi = U_{\mathrm{rms}}I_{\mathrm{rms}}\cos\varphi
= \tfrac12\Rea\!\big(\underline U\,\underline I^{\,*}\big) = \tfrac12\Rea(\underline Z)\,I^2,
\]
dengan $\cos\varphi$ sebagai \emph{\omterm{thm:b1:sinusoidal-impedance:power}{faktor daya}}. \omterm{def:b1:dc-circuits:resistor}{Resistor} menerima
$\tfrac12 RI^2 = RI_{\mathrm{rms}}^2$; kumparan atau kapasitor ideal
($\varphi = \pm\pi/2$) tidak menerima \omterm{thm:b1:sinusoidal-impedance:power}{daya rata-rata} --- ia menyimpan
energi selama setengah periode lalu mengembalikannya pada setengah
periode berikutnya.
\end{theorem}

\begin{proof}
$p = UI\cos\omega t\cos(\omega t - \varphi) = \tfrac12 UI[\cos\varphi +
\cos(2\omega t - \varphi)]$; suku keduanya berrata-rata nol. Dengan
$\underline U = U$ dan $\underline I = I\eu^{-j\varphi}$,
$\underline U\,\underline I^* = UI\eu^{j\varphi}$, yang bagian realnya
$UI\cos\varphi$; dan $\underline U = \underline Z\,\underline I$ memberi
$\underline U\,\underline I^* = \underline Z I^2$.
\end{proof}

\begin{example}[Perbaikan faktor daya]\label{ex:b1:sinusoidal-impedance:pf}
Motor sebuah bengkel pada \qty{230}{V} rms menarik \qty{2.0}{kW} dengan
$\cos\varphi = 0.70$ (induktif): $I_{\mathrm{rms}} = 2000/(230 \times
0.70) = \qty{12.4}{A}$, terhadap \qty{8.7}{A} seandainya \omterm{thm:b1:sinusoidal-impedance:power}{faktor dayanya}
$1$ --- arus tambahannya memanaskan kabel perusahaan listrik tanpa
guna, karena itulah dendanya. Kapasitor yang dipasang paralel memasok
ayunan energi seperempat periode kumparannya secara setempat:
\emph{\omterm{thm:b1:sinusoidal-impedance:power}{daya reaktif}} $P\tan\varphi =
\qty{2.0}{kvar}$ menuntut $C = P\tan\varphi/(\omega U_{\mathrm{rms}}^2)
\approx \qty{120}{\micro F}$ pada \qty{50}{Hz}, dan arus salurannya turun
ke \qty{8.7}{A}.
\end{example}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:1}
Berikan \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} $u = 5\cos(\omega t + \pi/3)$ (V), dalam
bentuk kutub dan bentuk Kartesius. Dengan memakai \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks},
tuliskan $3\cos\omega t
+ 4\sin\omega t$ sebagai satu kosinus.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:2}
Hitunglah \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} $R = \qty{100}{\ohm}$, $L = \qty{0.10}{H}$ dan
$C = \qty{10}{\micro F}$ pada \qty{50}{Hz} dan pada \qty{1.0}{kHz}.
Berikan modulus dan fase \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} $R$ dan $L$ yang terangkai seri pada
\qty{50}{Hz}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:3}
Jaringan listrik \qty{230}{V} rms: berapa \omterm{def:b1:wave-propagation:signal}{amplitudonya}? Sebuah pemanas
\qty{2.0}{kW}: arus rms dan arus puncaknya. Tunjukkan bahwa \omterm{thm:b1:sinusoidal-impedance:power}{daya
rata-rata} pada \omterm{def:b1:dc-circuits:resistor}{resistor} adalah $RI_{\mathrm{rms}}^2$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:sinusoidal-impedance:4}
RLC seri: $L = \qty{10}{mH}$, $C = \qty{100}{nF}$, $R = \qty{100}{\ohm}$,
$E = \qty{1.0}{V}$. Berikan $f_0$, $Q$, \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} dalam hertz dan
\omterm{def:b1:wave-propagation:signal}{amplitudo} arus pada resonansi.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:sinusoidal-impedance:5}
Pembagi RC, keluarannya pada $C$: $R = \qty{1.0}{k\ohm}$, $C = \qty{159}{nF}$.
Pada frekuensi berapa keluarannya $1/\sqrt2$ masukannya? Berikan nisbah
\omterm{def:b1:wave-propagation:signal}{amplitudo} dan fase keluarannya pada \qty{1.0}{kHz} dan pada \qty{10}{kHz}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:sinusoidal-impedance:6}
Pada suatu frekuensi sebuah RLC seri mempunyai $R = \qty{100}{\ohm}$, $L\omega =
\qty{250}{\ohm}$, $1/C\omega = \qty{100}{\ohm}$, dan $E = \qty{1.0}{V}$.
Hitunglah $\underline Z$, \omterm{def:b1:wave-propagation:signal}{amplitudo} dan fase arusnya, serta ketiga
\omterm{def:b1:wave-propagation:signal}{amplitudo} \omterm{def:b1:dc-circuits:voltage}{tegangannya}; lukislah diagram \omterm{def:b1:sinusoidal-impedance:complex}{fasornya}. Apakah rangkaiannya
berada di atas atau di bawah resonansi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:sinusoidal-impedance:7}
Radio: $L = \qty{200}{\micro H}$, $C$ yang dapat disetel dari \qty{50}{}
sampai \qty{500}{pF}. Carilah rentang frekuensinya, kapasitansi untuk
\qty{1.00}{MHz}, dan, dengan $Q = 100$, \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} serta pelemahan
(nisbah \omterm{def:b1:wave-propagation:signal}{amplitudo}) sebuah stasiun yang berjarak \qty{9}{kHz}. Ggl
antena \qty{10}{\micro V}: berapa \omterm{def:b1:dc-circuits:voltage}{tegangan} pada $C$ saat resonansi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:sinusoidal-impedance:8}
Turunkan kedua frekuensi setengah daya RLC seri lalu tunjukkan bahwa
selisihnya adalah $\omega_0/Q$ dan hasil kalinya $\omega_0^2$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:sinusoidal-impedance:9}
Sebuah motor pada \qty{230}{V} rms, \qty{50}{Hz}, menyerap \qty{2.0}{kW}
dengan $\cos\varphi = 0.70$. Hitunglah arus rmsnya, \omterm{thm:b1:sinusoidal-impedance:power}{daya reaktifnya},
kapasitansi paralel yang membawa \omterm{thm:b1:sinusoidal-impedance:power}{faktor dayanya} ke $1$, dan arus barunya.
Saluran catunya berhambatan \qty{0.50}{\ohm}: berapa rugi Joule sebelum
dan sesudahnya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:sinusoidal-impedance:10}
RLC paralel yang diberi \omterm{def:b1:dc-circuits:sources}{sumber arus}: tuliskan \omterm{def:b1:sinusoidal-impedance:impedance}{admitansinya}, tunjukkan
bahwa \omterm{def:b1:dc-circuits:voltage}{tegangannya} maksimum pada $\omega_0 = 1/\sqrt{LC}$ dan sama dengan
$RI$ di sana, serta bahwa \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} $\Delta\omega = 1/RC$, yaitu $Q =
R\sqrt{C/L}$. Angkanya: $R = \qty{10}{k\ohm}$, $L = \qty{1.0}{mH}$,
$C = \qty{1.0}{nF}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:sinusoidal-impedance:11}
Tunjukkan bahwa \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitor sebuah RLC seri adalah $U_C = E/\sqrt{(1 -
x^2)^2 + x^2/Q^2}$, bahwa ia memuncak pada $x_r = \sqrt{1 - 1/2Q^2}$ jika
$Q > 1/\sqrt2$, lalu hitunglah $x_r$ dan $U_{C,\max}/E$ untuk $Q = 3.16$.
Apa yang terjadi untuk $Q < 1/\sqrt2$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:sinusoidal-impedance:12}
Pada resonansi sebuah RLC seri, tunjukkan bahwa energi tersimpan
totalnya $\tfrac12 Li^2 + \tfrac12 Cu_C^2$ bersifat tetap dan sama dengan
$\tfrac12 LI^2$, hitunglah energi yang dilesapkan di $R$ tiap periode,
lalu buktikan bahwa
$2\pi\,(\text{tersimpan})/(\text{dilesapkan tiap periode}) = Q$.
\end{exercise}

\section{Soal: Penerima radio resonan}

\begin{problem}[{Soal akhir pekan --- sebuah kumparan, sebuah kapasitor
variabel dan sebuah antena: bagaimana resonansi seri memilih satu
stasiun dari seratus, mengalikan sinyalnya yang lemah seratus kali, dan
mengapa ia tak dapat dibuat selektif sesuka hati}]
\label{pb:b1:sinusoidal-impedance:1}
Antenanya dimodelkan sebagai ggl sinusoidal ideal $e(t) = E\cos\omega t$
yang terangkai seri dengan kumparan $L = \qty{200}{\micro H}$, yang
kawatnya berhambatan $R = \qty{12.6}{\ohm}$, dan kapasitor variabel $C$.
Penerimanya membaca \omterm{def:b1:dc-circuits:voltage}{tegangan} pada $C$. Kemudian (Bagian IV) hambatan
antenanya sendiri $R_a = \qty{50}{\ohm}$ ikut diperhitungkan.

\medskip
\noindent\textbf{Bagian I --- Arus resonan.}
\begin{enumerate}
  \item Tuliskan \omterm{def:b1:sinusoidal-impedance:impedance}{impedansi} kompleks rangkaian serinya.
  \item Berikan \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} arusnya, lalu \omterm{def:b1:wave-propagation:signal}{amplitudonya}
        $I(\omega)$ dan fasenya $\varphi_i(\omega)$ relatif terhadap $e$.
  \item Tunjukkan bahwa $I$ maksimum pada $\omega_0 = 1/\sqrt{LC}$;
        berikan $I_{\max}$ dan fasenya ketika itu.
  \item Kapasitansi berapa yang melaraskan $f_0 = \qty{1.000}{MHz}$?
        Hitunglah $L\omega_0$.
  \item Definisikan $Q = L\omega_0/R$ lalu hitunglah.
  \item Tunjukkan bahwa $I/I_{\max} = 1/\sqrt{1 + Q^2(x - 1/x)^2}$ dengan
        $x = \omega/\omega_0$.
  \item Nyatakan tanda $\varphi_i$ di bawah dan di atas resonansi lalu
        tafsirkan (perilaku kapasitif atau induktif).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Selektivitas.}
\begin{enumerate}[resume]
  \item Tuliskan syarat yang mendefinisikan kedua frekuensi tempat
        $I = I_{\max}/\sqrt2$.
  \item Selesaikan lalu tunjukkan bahwa $\Delta\omega = \omega_2 - \omega_1 =
        \omega_0/Q$.
  \item Hitunglah \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pita} $\Delta f$ penerima ini.
  \item Stasiun kedua memancar pada \qty{1.010}{MHz}, stasiun ketiga
        pada \qty{1.020}{MHz}: hitunglah nisbah \omterm{def:b1:wave-propagation:signal}{amplitudo} $I/I_{\max}$
        untuk masing-masing, dan nisbah daya yang bersesuaian dalam desibel.
  \item Stasiun modulasi \omterm{def:b1:wave-propagation:signal}{amplitudo} berjarak \qty{9}{} sampai \qty{10}{kHz}. Apakah
        penerimanya cukup selektif? Apa yang akan memperbaikinya?
  \item Kapasitornya dapat disetel antara \qty{50}{} dan \qty{500}{pF}:
        pita frekuensi berapa yang dicakup penerimanya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Penguatan \omterm{def:b1:dc-circuits:voltage}{tegangan}.}
\begin{enumerate}[resume]
  \item Nyatakan \omterm{def:b1:sinusoidal-impedance:complex}{amplitudo kompleks} \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitornya lalu tunjukkan
        bahwa pada resonansi \omterm{def:b1:wave-propagation:signal}{amplitudonya} adalah $QE$.
  \item Ggl antenanya $E = \qty{10}{\micro V}$: hitunglah $I_{\max}$ dan
        \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitornya pada resonansi. Berikan komentar.
  \item Tunjukkan bahwa \omterm{def:b1:dc-circuits:voltage}{tegangan} kumparannya beramplitudo sama dan
        berlawanan fase dengan \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitornya pada resonansi: apa
        yang ``dilihat'' sumbernya?
  \item Hitunglah energi yang tersimpan dalam rangkaiannya pada
        resonansi dan energi yang dilesapkan tiap periode; periksalah
        bahwa nisbahnya dikali $2\pi$ adalah $Q$.
  \item Dari \cref{ch:b1:transient-regimes}, berapa \omterm{def:b1:transient-regimes:firstorder}{tetapan waktu}
        $\tau = 2Q/\omega_0$ rangkaian ini? Mengapa $Q$ yang sangat
        tinggi pada akhirnya mencacatkan suara dan musik (\omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pita}
        audio: sekitar \qty{5}{kHz} untuk modulasi \omterm{def:b1:wave-propagation:signal}{amplitudo})?
  \item Jika hambatan kawat kumparannya berlipat dua (kawat lebih
        tipis), bagaimana $Q$, \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} dan \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitornya berubah?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Hambatan antenanya, dan dayanya.}
\begin{enumerate}[resume]
  \item Dengan $R_a = \qty{50}{\ohm}$ terangkai seri, hitunglah kembali
        $Q$, \omterm{thm:b1:sinusoidal-impedance:resonance}{lebar pitanya} dan \omterm{def:b1:dc-circuits:voltage}{tegangan} kapasitornya untuk
        $E = \qty{10}{\micro V}$. Simpulkan.
  \item Hitunglah \omterm{thm:b1:sinusoidal-impedance:power}{daya rata-rata} yang diberikan ggl antenanya pada
        resonansi dan bagian yang dilesapkan di $R_a$.
  \item Teorema daya maksimum akan menuntut hambatan beban yang sama
        dengan $R_a$: mengapa sasaran itu bertentangan dengan
        selektivitasnya?
  \item Penerima sungguhan menyambungkan antenanya ke sebuah sadapan
        pada kumparannya, atau lewat kapasitor gandeng yang kecil.
        Jelaskan secara kualitatif apa yang dicapainya.
  \item Dengan memakai $P = \tfrac12 UI\cos\varphi$, hitunglah \omterm{thm:b1:sinusoidal-impedance:power}{faktor
        dayanya} pada kedua frekuensi setengah daya lalu jelaskan nama
        ``setengah daya'' itu.
  \item Ringkaslah rancangannya: apa yang dibeli $Q$, apa harganya, dan
        nilai apa yang dipilih untuk penerima \qty{1}{MHz} dengan kanal
        \qty{10}{kHz}.
\end{enumerate}
\end{problem}
