\chapter{Hukum Dinamika Newton}\label{ch:b1:newton-dynamics}

Seorang penerjun payung melangkah keluar dari pesawatnya dan, dalam
belasan sekon, berhenti mengalami percepatan: pada dua ratus kilometer
tiap jam udara mendorong balik sekuat tarikan Bumi. Sebuah mobil
mengambil tikungan terlalu cepat dan bannya menyerah; sebuah truk di
jalan gunung merangkak pada gigi rendah, ditahan oleh gesekan yang sama
yang memungkinkannya berhenti di jalan datar. Kinematika menggambarkan
gerak; dinamika menjelaskannya, dari satu hukum yang menghubungkan
percepatan sebuah benda dengan \omterm{def:b1:newton-dynamics:momentum}{gaya} yang bekerja padanya. Bab ini
menyatakan ketiga \omterm{thm:b1:newton-dynamics:laws}{hukum Newton}, mendaftar \omterm{def:b1:newton-dynamics:momentum}{gaya} yang ditemui pada tingkat
volume ini --- berat, pegas, tali, sentuhan, \omterm{prop:b1:newton-dynamics:friction}{gesekan kering} dan fluida
--- lalu memperlihatkan cara berjalan dari gambar \omterm{def:b1:newton-dynamics:momentum}{gayanya} ke sebuah
persamaan diferensial dan penyelesaiannya.

\section{Massa, momentum, dan ketiga hukumnya}

\begin{definition}[Massa, momentum, gaya]\label{def:b1:newton-dynamics:momentum}
\index{massa}\index{momentum}\index{gaya}
Sebuah \omterm{def:b1:point-kinematics:frame}{partikel titik} mempunyai \emph{massa inersia} $m > 0$ (kilogram),
yaitu ukuran perlawanannya terhadap perubahan kecepatan, yang bebas dari
kerangkanya. \emph{Momentum}nya dalam sebuah kerangka adalah
\[
\vect p = m\,\vect v \qquad (\unit{kg.m/s}).
\]
Sebuah \emph{gaya} $\vect F$ (newton) adalah tindakan benda lain
terhadap partikelnya; gaya berjumlah sebagai vektor, dan
\emph{resultan}nya adalah jumlahnya $\sum\vect F$.
\end{definition}

\begin{theorem}[Hukum Newton]\label{thm:b1:newton-dynamics:laws}
\index{hukum Newton}\index{kerangka inersial}\index{asas kelembaman}\index{aksi dan reaksi}
\begin{enumerate}
  \item \emph{Asas kelembaman.} Ada kerangka, yang disebut
        \emph{inersial} (atau Galilei), yang di dalamnya partikel yang
        tak dikenai \omterm{def:b1:newton-dynamics:momentum}{gaya}, atau dikenai \omterm{def:b1:newton-dynamics:momentum}{gaya} yang resultannya nol,
        bergerak lurus dengan kecepatan tetap. Sebarang kerangka yang
        bertranslasi lurus beraturan terhadap \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial} juga
        bersifat inersial.
  \item \emph{Hukum \omterm{def:b1:newton-dynamics:momentum}{momentum}.} Dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial},
        \[
        \frac{\dd\vect p}{\dd t} = \sum\vect F, \qquad \text{yaitu}\qquad
        m\,\vect a = \sum\vect F \ \text{ untuk massa tetap } m .
        \]
  \item \emph{\omterm{thm:b1:newton-dynamics:laws}{Aksi dan reaksi}.} Jika benda $A$ mengerahkan
        $\vect F_{A\to B}$ pada benda $B$, maka $B$ mengerahkan
        $\vect F_{B\to A} = -\vect F_{A\to B}$ pada $A$, menyusuri garis
        yang menghubungkan keduanya.
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Apa yang dikatakan hukum-hukum itu]\label{rem:b1:newton-dynamics:meaning}
Hukum pertamanya bukan kasus khusus hukum kedua: ia
\emph{mendefinisikan} kerangka yang di dalamnya hukum kedua berlaku.
Dalam praktik, sebuah laboratorium di Bumi bersifat inersial dengan
ketelitian yang sangat baik bagi percobaan yang berlangsung beberapa
menit (\cref{ch:b1:non-inertial-frames} mengukur galatnya); kerangka
yang melekat pada Matahari dan bintang jauh lebih baik lagi. Hukum
keduanya adalah persamaan vektor: tiga persamaan skalar, satu tiap
sumbu, dan pemilihan sumbunya (\cref{ch:b1:point-kinematics}) adalah
keputusan pertama setiap soal. Hukum ketiganya berlaku pada tiap saat
bagi \omterm{def:b1:newton-dynamics:momentum}{gaya} sentuh dan \omterm{def:b1:newton-dynamics:momentum}{gaya} gravitasi volume ini.
\end{remark}

\begin{definition}[Sistem tertutup; kekekalan momentum]\label{def:b1:newton-dynamics:conservation}
\index{sistem tertutup}\index{kekekalan momentum}
\emph{Sistem tertutup} tidak bertukar materi dengan luarnya. Untuk
sistem partikel, \omterm{def:b1:newton-dynamics:momentum}{gaya} dalamnya saling meniadakan berpasangan (hukum
ketiga), dan \omterm{def:b1:newton-dynamics:momentum}{momentum} totalnya $\vect P = \sum m_i\vect v_i$ menuruti
$\dd\vect P/\dd t = \sum\vect F_{\mathrm{ext}}$: ia kekal ketika \omterm{def:b1:newton-dynamics:momentum}{gaya}
luarnya lenyap (\cref{ch:b1:systems-of-points} mengembangkannya).
\end{definition}

\section{Gaya-gaya yang lazim}

\begin{proposition}[Berat dan gravitasi]\label{prop:b1:newton-dynamics:weight}
\index{berat}\index{medan gravitasi}
Di dekat permukaan Bumi, benda bermassa $m$ ditarik oleh \emph{berat}nya
$\vect P = m\vect g$, dengan $\vect g$ sebagai \omterm{prop:b1:newton-dynamics:weight}{medan gravitasi} setempat,
$g \approx \qty{9.81}{m/s^2}$, yang terarah ke bawah (ke pusatnya,
nyaris tepat). Secara lebih umum, \omterm{def:b1:newton-dynamics:momentum}{massa} $M$ pada jarak $r$ menarik $m$
dengan $\vect F = -GMm/r^2\,\vect e_r$ (\cref{ch:b1:central-forces});
di permukaan sebuah bola berjari-jari $R$, $g = GM/R^2$. Perhatikan
satuannya: \unit{N/kg} dan \unit{m/s^2} adalah sama, dan itulah sebabnya
semua benda jatuh dengan cara yang sama.
\end{proposition}

\begin{proof}
Hukum gravitasi Newton diterima saja di sini dan digali dalam
\cref{ch:b1:central-forces}; $g = GM/R^2 = 6.67\times10^{-11} \times
5.97\times10^{24}/(6.37\times10^6)^2 = \qty{9.8}{m/s^2}$.
\end{proof}

\begin{proposition}[Pegas; tali]\label{prop:b1:newton-dynamics:springstring}
\index{hukum Hooke}\index{pegas}\index{tegangan tali}
Sebuah pegas ideal berkekakuan $k$ dan berpanjang alami $\ell_0$, yang
teregang sampai panjang $\ell$ menyusuri vektor satuan $\vect u$ yang
menunjuk dari ujung tetapnya ke bendanya, mengerahkan pada bendanya
\emph{\omterm{def:b1:newton-dynamics:momentum}{gaya} pemulih}
\[
\vect F = -k(\ell - \ell_0)\,\vect u \qquad (\text{hukum Hooke}),
\]
ke arah panjang alaminya. Tali ideal (tak bermassa, tak dapat
memanjang) mengerahkan \emph{tegangan} $\vect T$ menyusuri dirinya
sendiri, yang menarik, dengan besar yang sama pada kedua ujungnya dan di
sepanjangnya (bahkan melalui katrol tak bermassa tanpa gesekan).
\end{proposition}

\begin{proof}
\omterm{prop:b1:newton-dynamics:springstring}{Hukum Hooke} adalah model empiris yang berlaku untuk perubahan bentuk
yang kecil. Talinya: elemen tak bermassa menuruti
$0 = \vect T_1 + \vect T_2$ dari hukum kedua Newton, sehingga
tegangannya diteruskan tanpa berubah.
\end{proof}

\begin{proposition}[Gaya sentuh: reaksi normal dan gesekan kering]\label{prop:b1:newton-dynamics:friction}
\index{reaksi normal}\index{gesekan kering}
\index{hukum gesekan Coulomb}\index{koefisien gesek}
Permukaan padat mengerahkan pada benda yang menyentuhnya sebuah
\emph{\omterm{prop:b1:newton-dynamics:friction}{reaksi normal}} $\vect N$, yang tegak lurus permukaannya dan
menolak ($N \geq 0$: permukaan tak dapat menarik), dan sebuah \emph{\omterm{def:b1:newton-dynamics:momentum}{gaya}
gesek} tangensial $\vect T$. Hukum Coulomb: jika bendanya tidak
tergelincir, $\abs{\vect T} \leq \mu_s N$ (gesekan \emph{statis}
menyesuaikan diri dengan apa yang diperlukan, sampai batas itu); jika ia
tergelincir dengan kecepatan $\vect v_g$ relatif terhadap permukaannya,
$\vect T = -\mu_d N\,\vect v_g/\abs{\vect v_g}$ (gesekan \emph{kinetis},
yang melawan gelincirannya), dengan $\mu_d \leq \mu_s$ berupa koefisien
berorde $0.1$ sampai $1$ bergantung pada bahannya, bebas dari luas
sentuhnya dan dari kelajuannya.
\end{proposition}
\admitted

\begin{proposition}[Gesekan fluida]\label{prop:b1:newton-dynamics:drag}
\index{gesekan fluida}\index{gaya hambat}\index{hukum Stokes}
Benda yang bergerak dengan kecepatan $\vect v$ menembus fluida yang diam
merasakan \emph{\omterm{prop:b1:newton-dynamics:drag}{gaya hambat}} yang berlawanan dengan $\vect v$: pada laju
rendah (benda kecil, fluida kental) ia linear,
$\vect F = -\alpha\vect v$ --- untuk bola berjari-jari $r$ dalam fluida
berkekentalan $\eta$, $\alpha = 6\pi\eta r$ (Stokes); pada laju tinggi
(benda besar dan cepat di udara atau air) ia kuadratik,
\[
\vect F = -\tfrac12\,\rho\,C_xS\,v\,\vect v ,
\]
dengan $\rho$ \omterm{def:b1:newton-dynamics:momentum}{massa} jenis fluidanya, $S$ penampang lintang bendanya yang
menghadap alirannya dan $C_x$ sebuah \emph{koefisien hambat} \omterm{def:b1:units-dimensions:dimension}{tak
berdimensi} ($0.5$ untuk bola, $0.3$ untuk mobil, $1$ untuk cakram
datar). Bilangan Reynolds $\rho vL/\eta$
(\cref{rem:b1:units-dimensions:pi}) yang memutuskan: linear di bawah
$1$, kuadratik di atas kira-kira $1000$.
\end{proposition}
\admitted

\begin{remark}[Gaya lainnya]\label{rem:b1:newton-dynamics:others}
\omterm{def:b1:newton-dynamics:momentum}{Gaya} apung (Archimedes) adalah resultan \omterm{def:b1:newton-dynamics:momentum}{gaya} tekanan, yang diturunkan
dalam \cref{ch:b1:fluid-statics}; \omterm{def:b1:newton-dynamics:momentum}{gaya} listrik dan \omterm{def:b1:newton-dynamics:momentum}{gaya} magnetik pada
sebuah muatan adalah pokok bahasan \cref{ch:b1:charged-particles}.
Semua \omterm{def:b1:newton-dynamics:momentum}{gaya} bab ini adalah \emph{model} interaksi sentuh atau medan, yang
berlaku dalam batas yang dinyatakan --- pegas Hooke patah, gesekan
Coulomb gagal pada tekanan tinggi, \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} Stokes gagal pada laju
tinggi. Bagian dari setiap penyelesaian adalah memeriksa bahwa model
yang dipakai memang berlaku.
\end{remark}

\section{Menyelesaikan sebuah soal dinamika}

\begin{method}[Dari situasinya ke persamaannya]\label{met:b1:newton-dynamics:method}
\begin{enumerate}
  \item \emph{Sistem}: sebutkan benda yang diperlakukan sebagai titik.
  \item \emph{Kerangka}: sebutkan kerangkanya lalu periksa bahwa ia
        inersial (atau perlakukan demikian sampai ketelitian yang
        diminta).
  \item \emph{\omterm{def:b1:newton-dynamics:momentum}{Gaya}}: daftarkan setiap benda yang menyentuh atau menarik
        sistemnya, lalu lukiskan tiap \omterm{def:b1:newton-dynamics:momentum}{gayanya} pada sebuah gambar
        (diagram benda bebas); tak ada yang lain yang masuk.
  \item \emph{Koordinat}: pilihlah sumbu yang cocok dengan geraknya
        (\cref{met:b1:point-kinematics:choose}) lalu nyatakan $\vect a$.
  \item \emph{Proyeksikan} $m\vect a = \sum\vect F$ pada sumbunya;
        pakailah kendalanya (benda di atas permukaan tetap di atasnya:
        $a_\perp = 0$; tali berpanjang tetap: kelajuannya sama pada
        kedua ujungnya).
  \item \emph{Selesaikan} lalu \emph{periksalah}: dimensinya, kasus
        batasnya, tandanya ($N \geq 0$, $\abs T \leq \mu_sN$ bila diam).
\end{enumerate}
\end{method}

\begin{example}[Balok di bidang miring]\label{ex:b1:newton-dynamics:incline}
Balok bermassa $m$ pada bidang yang miring sebesar $\alpha$, dengan
koefisien $\mu_s$, $\mu_d$. Sumbunya: $x$ menuruni lerengnya, $y$
menyusuri normalnya. Berat $(mg\sin\alpha, -mg\cos\alpha)$, reaksi
$(0, N)$, gesekan $(-T, 0)$ bila ia tergelincir turun. Menyusuri $y$:
$N = mg\cos\alpha$. Ketika diam, menyusuri $x$:
$T = mg\sin\alpha \leq \mu_sN = \mu_smg\cos\alpha$, yaitu $\tan\alpha
\leq \mu_s$: baloknya bertahan sampai sudut $\arctan\mu_s$ --- sebuah
pengukuran $\mu_s$ yang tak memerlukan neraca. Ketika tergelincir:
$ma = mg\sin\alpha -
\mu_dmg\cos\alpha$, sehingga $a = g(\sin\alpha - \mu_d\cos\alpha)$, bebas
dari $m$: \qty{3.2}{m/s^2} untuk $\alpha = 30^\circ$, $\mu_d = 0.2$.
\end{example}

\begin{example}[Bandul sederhana]\label{ex:b1:newton-dynamics:pendulum}
\index{bandul sederhana}
\omterm{def:b1:newton-dynamics:momentum}{Massa} $m$ pada tali sepanjang $\ell$, bersudut $\theta$ dari garis tegak
ke bawah. \omterm{def:b1:point-kinematics:cylindrical}{Koordinat polar} di titik gantungnya: $\vect a =
-\ell\dot\theta^2\,\vect e_r + \ell\ddot\theta\,\vect e_\theta$;
\omterm{def:b1:newton-dynamics:momentum}{gayanya}: berat $mg(\cos\theta\,\vect e_r - \sin\theta\,\vect e_\theta)$
dan tegangan $-T\vect e_r$. Menyusuri $\vect e_\theta$: $m\ell\ddot\theta =
-mg\sin\theta$,
\[
\ddot\theta + \frac{g}{\ell}\sin\theta = 0 ,
\]
yang untuk sudut kecil ($\sin\theta \approx \theta$) menjadi persamaan
harmonik dengan $\omega_0 = \sqrt{g/\ell}$: periodenya
$T_0 = 2\pi\sqrt{\ell/g}$, yaitu $k = 2\pi$ yang tak dapat dipasok
\cref{ex:b1:units-dimensions:pendulum}. Menyusuri $\vect e_r$:
$T = mg\cos\theta + m\ell\dot\theta^2$ --- talinya menarik paling kuat
di dasarnya, tempat kelajuannya terbesar.
\end{example}

\begin{proposition}[Jatuh dengan gaya hambat; laju terminal]\label{prop:b1:newton-dynamics:terminal}
\index{laju terminal}
Benda bermassa $m$ yang jatuh dari keadaan diam di bawah beratnya dan
sebuah \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} mencapai \emph{\omterm{prop:b1:newton-dynamics:terminal}{laju terminal}} tempat keduanya
mengimbangi. Hambat linear: $m\dot v = mg - \alpha v$,
$v = v_\infty(1 - \eu^{-t/\tau})$ dengan
$v_\infty = mg/\alpha$, $\tau = m/\alpha$. Hambat kuadratik: $m\dot v = mg -
\tfrac12\rho C_xSv^2$, $v = v_\infty\tanh(t/\tau)$ dengan
\[
v_\infty = \sqrt{\frac{2mg}{\rho C_xS}}, \qquad \tau = \frac{v_\infty}{g} .
\]
\end{proposition}

\begin{proof}
Linear: persamaan orde pertama, $\tau\dot v + v = v_\infty$
(\cref{thm:b1:transient-regimes:firstorder}). Kuadratik: dengan
$u = v/v_\infty$, $\dot u = (1 - u^2)/\tau$; dengan memisahkan
variabelnya, $\int\dd u/(1 - u^2) = \operatorname{artanh}u = t/\tau$
(pecahan rasional $1/(1 - u^2) = \tfrac12[1/(1 - u) + 1/(1 + u)]$
terintegral menjadi $\tfrac12\ln\frac{1 + u}{1 - u}$), sehingga
$u = \tanh(t/\tau)$.
\end{proof}

\begin{example}[Dua laju terminal]\label{ex:b1:newton-dynamics:terminal}
Seorang penerjun payung, $m = \qty{80}{kg}$, tengkurap, $C_xS \approx \qty{0.8}{m^2}$
di udara ($\rho = \qty{1.2}{kg/m^3}$): $v_\infty = \sqrt{2 \times 785/(1.2
\times 0.8)} = \qty{40}{m/s}$, yang dicapai (sampai $90\%$) dalam kira-kira $1.5\tau =
\qty{6}{s}$. Sebuah tetes kabut, $r = \qty{10}{\micro m}$, dalam rezim
Stokes ($\eta_{\text{udara}} = \qty{1.8e-5}{Pa.s}$): $v_\infty = mg/6\pi\eta r =
\qty{1.2}{cm/s}$ --- dan itulah sebabnya kabut menggantung di udara.
\end{example}

\section{Gerak pada lingkaran: gaya kendala}

\begin{proposition}[Tuntutan sentripetal]\label{prop:b1:newton-dynamics:centripetal}
\index{gaya sentripetal}
Partikel yang bergerak pada lingkaran berjari-jari $R$ dengan laju $v$
harus menerima dari resultan \omterm{def:b1:newton-dynamics:momentum}{gayanya} sebuah komponen $mv^2/R$ yang
terarah ke pusatnya. Apa pun yang memasoknya --- tegangan sebuah tali,
gesekan, reaksi jalan yang miring, gravitasi --- menetapkan laju
maksimum atau laju minimum yang membuat geraknya mungkin.
\end{proposition}

\begin{proof}
Hukum kedua Newton dengan percepatan Frenet: komponen normal
$\sum\vect F$ sama dengan $mv^2/\rho$.
\end{proof}

\begin{example}[Tiga lingkaran]\label{ex:b1:newton-dynamics:circles}
\emph{Tikungan datar}: gesekannya harus memasok $mv^2/R \leq \mu_sN = \mu_smg$,
sehingga $v \leq \sqrt{\mu_sgR}$: \qty{20}{m/s} untuk $R = \qty{50}{m}$,
$\mu_s = 0.8$. \emph{Tikungan miring} bersudut $\beta$, tanpa gesekan:
reaksi $\vect N$ tegak lurus jalannya; secara tegak $N\cos\beta = mg$,
secara mendatar $N\sin\beta = mv^2/R$, sehingga $v^2 = gR\tan\beta$ ---
satu-satunya laju yang membuat tikungan itu dapat diambil tanpa gesekan
sama sekali. \emph{Loop}: di puncaknya, dengan $\vect N$ menunjuk ke
bawah ke arah pusatnya, $N + mg =
mv^2/R$, dan $N \geq 0$ menuntut $v \geq \sqrt{gR}$: di bawah itu,
keretanya meninggalkan \omterm{def:b1:point-kinematics:frame}{lintasannya} (\cref{pb:b1:point-kinematics:1}
lagi, kini dijelaskan).
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:1}
Seseorang bermassa \qty{70}{kg} berdiri di atas timbangan di dalam
sebuah lift. Berapa yang terbaca pada timbangannya (dalam newton dan
dalam ``kilogram'') ketika liftnya mempercepat ke atas pada
\qty{2.0}{m/s^2}, ke bawah pada \qty{2.0}{m/s^2}, bergerak dengan
kecepatan tetap, dan jatuh bebas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:2}
Sebuah balok $m_1 = \qty{2.0}{kg}$ di atas meja tanpa gesekan diikat
dengan tali lewat katrol tanpa gesekan ke \omterm{def:b1:newton-dynamics:momentum}{massa} gantung
$m_2 = \qty{1.0}{kg}$. Carilah percepatan dan tegangannya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:3}
Sebuah balok tergelincir menuruni bidang miring $30^\circ$ dengan
$\mu_d = 0.20$. Hitunglah percepatannya dan kelajuannya setelah
\qty{5.0}{m} dari keadaan diam. Apakah jawabannya bergantung pada \omterm{def:b1:newton-dynamics:momentum}{massanya}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:newton-dynamics:4}
Sebuah \omterm{def:b1:newton-dynamics:momentum}{massa} \qty{0.50}{kg} tergantung pada pegas berkekakuan
\qty{200}{N/m}. Carilah pertambahan panjang statisnya; lalu \omterm{def:b1:wave-propagation:signal}{frekuensi
sudut} dan periode osilasi tegaknya di sekitar kesetimbangannya
(tunjukkan bahwa persamaan geraknya harmonik).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:5}
Sebuah peti duduk di atas landasan yang dimiringkan; ia mulai
tergelincir pada $22^\circ$. Carilah $\mu_s$. Sebuah mobil di jalan
datar dengan $\mu_s = 0.80$: berapa laju maksimumnya di tikungan
berjari-jari \qty{50}{m}, dan di jalan basah dengan $\mu_s = 0.40$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:6}
\omterm{prop:b1:newton-dynamics:terminal}{Laju terminal}: seorang penerjun payung ($m = \qty{80}{kg}$, $C_xS = \qty{0.70}{m^2}$,
$\rho = \qty{1.2}{kg/m^3}$); sebuah tetes kabut berjari-jari \qty{10}{\micro m}
di udara ($\eta = \qty{1.8e-5}{Pa.s}$), memakai \omterm{prop:b1:newton-dynamics:drag}{hukum Stokes}. Periksalah
dengan bilangan Reynolds bahwa tiap rezimnya memang yang tepat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:7}
Sebuah bandul sepanjang $\ell$ dilepaskan dari keadaan diam pada
$\theta_0 = 60^\circ$. Diketahui bahwa kelajuannya di dasarnya adalah
$\sqrt{2g\ell(1 - \cos\theta_0)}$ (energi, bab berikutnya), carilah
tegangan talinya di dasarnya dalam satuan $mg$, dan di titik pelepasannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:8}
Sebuah tikungan berjari-jari \qty{200}{m} dimiringkan sebesar $15^\circ$.
Pada laju berapa ia dapat diambil tanpa gesekan sama sekali? Apa yang
harus dilakukan gesekannya pada laju yang lebih rendah dan yang lebih tinggi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:newton-dynamics:9}
Bandul kerucut: sebuah \omterm{def:b1:newton-dynamics:momentum}{massa} pada tali sepanjang $\ell = \qty{1.0}{m}$
menggambarkan lingkaran mendatar, dengan talinya membentuk $30^\circ$
terhadap garis tegak. Carilah \omterm{cor:b1:point-kinematics:circular}{kecepatan sudutnya}, periodenya dan
tegangannya (dalam satuan $mg$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:10}
Sebuah mobil \qty{1000}{kg} mencapai \qty{100}{km/h} dari keadaan diam
dalam \qty{8.0}{s} dengan percepatan tetap. Carilah \omterm{def:b1:newton-dynamics:momentum}{gaya} resultannya,
dan \omterm{prop:b1:newton-dynamics:friction}{koefisien gesek} minimum antara ban penggeraknya dan jalannya jika
roda penggeraknya memikul seluruh beratnya. Apa yang membatasi mobil
sport di jalan basah?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:11}
Sebuah benda jatuh dari keadaan diam dengan \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} linear,
$\tau = m/\alpha = \qty{0.50}{s}$. Tuliskan $v(t)$ lalu turunkan jarak
jatuhnya $z(t)$; hitunglah $v_\infty$, dan $z$ setelah \qty{2.0}{s};
bandingkan dengan jatuh bebas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:newton-dynamics:12}
Sebuah kereta bermassa $m$ berjalan di sisi dalam loop melingkar tegak
berjari-jari $R$ (\cref{pb:b1:point-kinematics:1}). Nyatakan \omterm{prop:b1:newton-dynamics:friction}{reaksi
normal} \omterm{def:b1:point-kinematics:frame}{lintasannya} sebagai fungsi sudut $\theta$ dari dasarnya dan
kelajuannya $v(\theta)$; simpulkan laju minimum di puncaknya; dengan
memakai $v^2 = v_0^2 - 2gR(1 - \cos\theta)$, carilah laju masuk minimum
dan reaksinya di dasarnya pada kasus itu.
\end{exercise}

\section{Soal: Sang penerjun payung}

\begin{problem}[{Soal akhir pekan --- dari pintu pesawat sampai ke
tanah: jatuh bebas, udara yang mendorong balik, payung yang terbuka
terlalu cepat, dan pendaratannya --- satu hukum gerak, dua hukum hambat}]
\label{pb:b1:newton-dynamics:1}
Seorang penerjun payung beserta perlengkapannya bermassa $m = \qty{80}{kg}$.
Udara: $\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$. Ketika
jatuh tengkurap, $C_xS = \qty{0.80}{m^2}$; di bawah payung terbuka, $C_xS =
\qty{25}{m^2}$. $g = \qty{9.81}{m/s^2}$; $z$ dihitung ke bawah
dari titik terjunnya.

\medskip
\noindent\textbf{Bagian I --- Tanpa udara.}
\begin{enumerate}
  \item Tuliskan hukum kedua Newton lalu berikan $v(t)$ dan $z(t)$;
        hitunglah keduanya pada $t = \qty{10}{s}$.
  \item Hitunglah beratnya.
  \item Penerjunnya merasa ``tanpa berat'' selama tahap ini walaupun
        gravitasi bekerja sepenuhnya. Jelaskan dengan \omterm{def:b1:newton-dynamics:momentum}{gayanya} (apa yang
        hilang dibandingkan dengan berdiri di atas tanah).
  \item Apakah kerangka Bumi cukup inersial bagi soal ini? Sebutkan apa
        yang diabaikan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Jatuh menembus udara.}
\begin{enumerate}[resume]
  \item Tuliskan \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} untuk jatuh tengkurap sebagai fungsi $v$.
  \item Tuliskan persamaan gerak untuk $v(t)$.
  \item Carilah \omterm{prop:b1:newton-dynamics:terminal}{laju terminalnya} $v_\infty$ (dalam \unit{m/s} dan
        \unit{km/h}).
  \item Dengan $u = v/v_\infty$ dan $\tau = v_\infty/g$, tunjukkan bahwa
        $\dd u/\dd t = (1 - u^2)/\tau$; lalu hitunglah $\tau$.
  \item Selesaikan dengan pemisahan variabel lalu tunjukkan bahwa
        $v = v_\infty\tanh(t/\tau)$.
  \item Setelah berapa lama $v = 0.9v_\infty$? $0.99v_\infty$?
  \item Tunjukkan bahwa $z(t) = v_\infty\tau\ln\cosh(t/\tau)$ lalu
        hitunglah jarak jatuhnya pada $t = \qty{10}{s}$; bandingkan
        dengan Bagian I.
  \item Taksirlah bilangan Reynolds pada $v_\infty$ dengan $L = \qty{0.5}{m}$
        lalu benarkan hukum kuadratiknya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Payungnya terbuka.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{prop:b1:newton-dynamics:terminal}{laju terminal} yang baru di bawah payung yang terbuka.
  \item Jika payungnya terbuka seketika pada $v = \qty{40}{m/s}$, berapa
        perlambatannya, dalam \unit{m/s^2} dan dalam $g$? Mengapa hal
        ini tak dapat diterima?
  \item Payung sungguhan terbuka dalam sekitar \qty{3}{s}. Taksirlah
        perlambatan rata-ratanya dan \omterm{def:b1:newton-dynamics:momentum}{gaya} rata-rata tali pengikatnya
        pada penerjunnya.
  \item Begitu payungnya terbuka penuh, kelajuannya meluruh menuju
        $v_\infty$ yang baru: berapa \omterm{def:b1:transient-regimes:firstorder}{tetapan waktunya} yang baru
        $\tau' = v_\infty'/g$, dan apa artinya bagi pendekatan ke
        penurunan yang tunak?
  \item Dalam penurunan yang tunak, berapa \omterm{def:b1:newton-dynamics:momentum}{gaya} tali pengikatnya?
        Berikan alasan dengan hukum kedua.
  \item Jelaskan mengapa \omterm{prop:b1:newton-dynamics:terminal}{laju terminalnya} berubah sebagai
        $1/\sqrt{C_xS}$: dengan faktor berapa payung yang luasnya
        separuh akan menaikkannya?
  \item Seorang penerjun yang lebih berat (\qty{100}{kg}) di bawah
        payung yang sama: berapa \omterm{prop:b1:newton-dynamics:terminal}{laju terminalnya}?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pendaratan.}
\begin{enumerate}[resume]
  \item Laju mendaratnya adalah $v_\infty$ payungnya. Dari ketinggian
        berapa orang harus melompat, tanpa udara, untuk membentur tanah
        dengan laju itu?
  \item Pendaratan berkaki kaku menghentikan tubuhnya sepanjang
        \qty{0.5}{m}; berguling sepanjang \qty{1.5}{m}. Taksirlah
        perlambatan rata-ratanya (dalam $g$) dan \omterm{def:b1:newton-dynamics:momentum}{gaya} rata-rata pada
        kakinya pada tiap kasus, lalu simpulkan.
  \item Tepat sebelum menyentuh tanah, penerjunnya menarik kedua tali
        kendalinya, yang sesaat memperbesar \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} payungnya.
        Jelaskan secara kualitatif pengaruhnya pada kelajuannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian V --- Benda kecil jatuh secara berbeda.}
\begin{enumerate}[resume]
  \item Sebuah tetes hujan berjari-jari \qty{1.0}{mm} ($C_x = 0.5$):
        hitunglah \omterm{prop:b1:newton-dynamics:terminal}{laju terminalnya} dengan hukum kuadratiknya, lalu
        periksalah bilangan Reynoldsnya.
  \item Sebuah tetes kabut berjari-jari \qty{10}{\micro m}: pakailah
        \omterm{prop:b1:newton-dynamics:drag}{hukum Stokes} untuk $v_\infty$ lalu periksalah bahwa bilangan
        Reynoldsnya memang kecil.
  \item Ringkaslah: satu persamaan gerak, dua rezim hambat, dan ketiga
        angka (\unit{m/s}) yang memutuskan apakah sesuatu jatuh seperti
        batu, seperti daun, atau tidak jatuh sama sekali.
\end{enumerate}
\end{problem}
