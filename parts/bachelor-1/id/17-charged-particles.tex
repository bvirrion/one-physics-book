\chapter{Partikel Bermuatan dalam Medan E dan B}\label{ch:b1:charged-particles}

Di dalam osiloskop yang menggambar jejak \cref{ch:b1:transient-regimes},
seberkas elektron dikemudikan melintasi layar oleh \omterm{def:b1:dc-circuits:voltage}{tegangan} di antara
dua keping; di dalam magnet sebuah \omterm{prop:b1:charged-particles:spectrometer}{spektrometer massa}, ion dua isotop
terpisah beberapa milimeter karena yang satu delapan persen lebih berat
daripada yang lain; di dalam siklotron sebuah rumah sakit, proton
berpilin keluar melewati dua ratus putaran sebelum pergi dengan
sepertiga laju cahaya untuk membakar sebuah tumor. Ketiganya adalah
dinamika yang sama: sebuah muatan dalam medan seragam, listrik atau
magnet, dan \omterm{thm:b1:newton-dynamics:laws}{hukum Newton}. Bab ini menuntaskannya --- parabola dalam
\omterm{def:b1:charged-particles:lorentz}{medan listrik}, lingkaran dan heliks dalam \omterm{def:b1:charged-particles:lorentz}{medan magnet} --- lalu
mengubahnya menjadi alat yang memilah, mempercepat dan mengukur
partikel bermuatan.

\section{Gaya Lorentz}

\begin{definition}[Gaya Lorentz]\label{def:b1:charged-particles:lorentz}
\index{gaya Lorentz}\index{medan listrik}\index{medan magnet}
Sebuah partikel bermuatan $q$ yang bergerak dengan kecepatan $\vect v$ dalam
medan listrik $\vect E$ dan medan magnet $\vect B$ menerima gaya
\[
\vect F = q\big(\vect E + \vect v\wedge\vect B\big) .
\]
Bagian listriknya searah $\vect E$ (atau berlawanan, untuk $q < 0$) dan
tak bergantung pada geraknya; bagian magnetnya tegak lurus baik pada
$\vect v$ maupun pada $\vect B$, sebanding dengan kelajuannya, dan lenyap bagi
partikel yang diam. Satuan: $\vect E$ dalam \unit{V/m} (atau \unit{N/C}), $\vect B$
dalam tesla (\unit{T}): $\qty{1}{T} = \qty{1}{N/(A.m)}$.
\end{definition}

\begin{proposition}[Usaha gaya Lorentz]\label{prop:b1:charged-particles:work}
Gaya magnet tidak melakukan usaha: dayanya $q(\vect v\wedge\vect B)\cdot\vect v$
bernilai nol, sehingga \omterm{def:b1:charged-particles:lorentz}{medan magnet} saja mengubah arah kecepatannya
tetapi tak pernah kelajuan maupun \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya}. Gaya listriknya
berdaya $q\vect E\cdot\vect v$, dan di antara dua titik usahanya $q(V_A - V_B) =
qU_{AB}$ (\cref{ch:b1:potential-capacitors}): partikel bermuatan $q$
yang menyeberangi \omterm{def:b1:dc-circuits:voltage}{tegangan} $U$ memperoleh \omterm{thm:b1:work-and-energy:ke}{energi kinetik} $\abs{qU}$.
\end{proposition}

\begin{proof}
$\vect v\wedge\vect B \perp \vect v$. Medan elektrostatiknya diturunkan dari
potensialnya, $\vect E = -\overrightarrow{\operatorname{grad}}V$, sehingga $q\vect E$
bersifat \omterm{def:b1:work-and-energy:conservative}{konservatif} dengan $E_p = qV$.
\end{proof}

\begin{definition}[Elektronvolt]\label{def:b1:charged-particles:eV}
\index{elektronvolt}
Yang disebut \emph{elektronvolt} adalah \omterm{thm:b1:work-and-energy:ke}{energi kinetik} yang diperoleh
sebuah muatan elementer ketika menyeberangi satu volt: $\qty{1}{eV} = \qty{1.60e-19}{J}$.
Kimia berlangsung pada beberapa eV, tabung sinar-X pada puluhan keV,
fisika inti pada MeV, fisika partikel pada GeV dan TeV.
\end{definition}

\begin{remark}[Mengapa gravitasi diabaikan]\label{rem:b1:charged-particles:gravity}
Bagi sebuah elektron dalam medan \qty{1}{kV/m} --- nilai laboratorium
yang sederhana --- gaya listriknya $eE = \qty{1.6e-16}{N}$, berbanding berat
$mg = \qty{9e-30}{N}$: $10^{13}$ kali lebih kecil. Berat dibuang dari
setiap perhitungan bab ini tanpa perlu berpikir dua kali. (Batas
relativistiknya soal lain: rumus-rumus di bawah berlaku untuk
$\tfrac12 mv^2 \ll mc^2$, yakni di bawah kira-kira \qty{50}{keV} bagi elektron
dan \qty{100}{MeV} bagi proton; lihat volume Tahun ke-3.)
\end{remark}

\section{Medan listrik seragam}

\begin{proposition}[Percepatan dan pembelokan]\label{prop:b1:charged-particles:efield}
\index{tegangan pemercepat}
Dalam medan seragam $\vect E$ percepatan $\vect a = q\vect E/m$ bersifat
tetap: geraknya adalah gerak sebuah proyektil, dengan $q\vect E/m$ menggantikan
$\vect g$.
\begin{itemize}
  \item Dilepaskan dari keadaan diam dan dipercepat melalui \omterm{def:b1:dc-circuits:voltage}{tegangan} $U$,
        partikelnya mencapai $v = \sqrt{2\abs{qU}/m}$.
  \item Masuk dengan kelajuan $v_0$ menyusuri $x$ ke daerah sepanjang $L$ tempat
        $\vect E = E\vect e_y$, ia menuruti parabola $y = (qE/2mv_0^2)x^2$
        dan keluar terbelokkan sebesar sudut $\alpha$ dengan $\tan\alpha =
        qEL/mv_0^2$, seolah-olah datang menyusuri garis lurus dari
        tengah kepingnya.
\end{itemize}
\end{proposition}

\begin{proof}
Energi: $\tfrac12 mv^2 = \abs{qU}$. Pembelokan: $x = v_0t$, $y =
\tfrac12(qE/m)t^2$; di pintu keluarnya, $\dd y/\dd x = (qE/m)(x/v_0^2)$ dihitung
di $L$; garis singgung di pintu keluar, $y = (qEL/mv_0^2)(x - L/2)$, memotong
sumbunya di $x = L/2$.
\end{proof}

\section{Medan magnet seragam}

\begin{theorem}[Gerak dalam medan magnet seragam]\label{thm:b1:charged-particles:bfield}
\index{gerak siklotron}\index{jari-jari Larmor}\index{frekuensi siklotron}\index{gerak heliks}
Sebuah partikel bermuatan $q$, bermassa $m$, berkecepatan $\vect v_0$ tegak lurus pada
$\vect B$ yang seragam, melukiskan lingkaran dengan kelajuan tetap $v_0$, berjari-jari
dan berfrekuensi sudut
\[
R = \frac{mv_0}{\abs q B}, \qquad \omega_c = \frac{\abs q B}{m},
\]
dengan \emph{\omterm{thm:b1:charged-particles:bfield}{frekuensi siklotron}} $\omega_c$ yang tak bergantung pada kelajuannya;
arah putarannya bergantung pada tanda $q$. Bila $\vect v_0$ juga mempunyai
komponen $v_\parallel$ menyusuri $\vect B$, komponen itu tak berubah dan
\omterm{def:b1:point-kinematics:frame}{lintasannya} berupa \emph{heliks} berjari-jari $mv_\perp/\abs qB$ dan berlangkah
$2\pi mv_\parallel/\abs qB$ yang membelit garis-garis medannya.
\end{theorem}

\begin{proof}
Gaya $q\vect v\wedge\vect B$ tegak lurus pada $\vect v$ dan pada $\vect B$;
kelajuannya tetap (tak ada usaha) dan komponen menyusuri $\vect B$ tak
merasakan gaya. Pada bidang yang tegak lurus $\vect B$ gaya itu bernorma
tetap $\abs qv_\perp B$, selalu tegak lurus pada kecepatannya: \omterm{cor:b1:point-kinematics:circular}{gerak
melingkar} beraturan (\cref{thm:b1:point-kinematics:frenet}) dengan $mv_\perp^2/R =
\abs qv_\perp B$. Maka $\omega_c = v_\perp/R$; periode $2\pi m/\abs qB$ dikali
$v_\parallel$ adalah langkahnya.
\end{proof}

\begin{example}[Orde besar]\label{ex:b1:charged-particles:orders}
Sebuah elektron \qty{1}{keV} ($v = \qty{1.9e7}{m/s}$) dalam \qty{1}{mT}:
$R = \qty{11}{cm}$, $\omega_c = \qty{1.8e8}{rad/s}$, satu putaran dalam

menuju kutub --- aurora.
\end{example}

\section{Instrumen}

\begin{proposition}[Selektor kecepatan dan spektrometer massa]\label{prop:b1:charged-particles:spectrometer}
\index{selektor kecepatan}\index{spektrometer massa}
Dalam medan seragam bersilang $\vect E \perp \vect B$, sebuah partikel yang
bergerak tegak lurus pada keduanya lewat tanpa dibelokkan jika dan hanya jika $v = E/B$,
berapa pun muatan dan massanya (\emph{\omterm{prop:b1:charged-particles:spectrometer}{selektor kecepatan}}). Dalam \omterm{def:b1:charged-particles:lorentz}{medan magnet} saja,
partikel yang kelajuannya diketahui melukiskan setengah lingkaran bergaris tengah $2mv/\abs qB$:
mengukurnya memberikan $m/q$ (\emph{\omterm{prop:b1:charged-particles:spectrometer}{spektrometer massa}}). Ion yang dipercepat
melalui \omterm{def:b1:dc-circuits:voltage}{tegangan} $U$ lalu dibelokkan dalam $B$ mendarat pada
\[
R = \frac{1}{B}\sqrt{\frac{2mU}{\abs q}} :
\]
dua isotop bermassa $m$ dan $m + \Delta m$ terpisah sejauh $\Delta R \approx
R\,\Delta m/2m$.
\end{proposition}

\begin{proof}
$q\vect E + q\vect v\wedge\vect B = \vect 0$ menuntut $vB = E$ dengan arah
yang tepat. $R = mv/qB$ dengan $v = \sqrt{2qU/m}$; diferensialkan
$\ln R = \tfrac12\ln m + \text{tetapan}$.
\end{proof}

\begin{example}[Isotop karbon]\label{ex:b1:charged-particles:isotopes}
Ion $^{12}$C dan $^{13}$C bermuatan tunggal yang dipercepat melalui \qty{20}{kV}
dalam \qty{0.50}{T}: $R = \qty{14.1}{cm}$ dan \qty{14.7}{cm}, terpisah \qty{5.8}{mm}
--- dua bintik terpisah pada detektornya, dan nisbah intensitasnya adalah
kelimpahan isotopnya, yaitu bahan mentah penanggalan radiokarbon dan
setiap analisis isotop.
\end{example}

\begin{proposition}[Siklotron]\label{prop:b1:charged-particles:cyclotron}
\index{siklotron}
Dua setengah tabung berongga (``dee'') dalam medan seragam $\vect B$, dengan
\omterm{def:b1:dc-circuits:voltage}{tegangan} bolak-balik $U$ di antaranya pada \omterm{thm:b1:charged-particles:bfield}{frekuensi siklotron}: partikelnya
berputar di dalam tiap dee (tak ada medan di sana), dipercepat sebesar
$\abs qU$ pada masing-masing dari dua penyeberangan celah tiap putaran, dan
berpilin keluar dengan $R \propto v$; pada jari-jari ekstraksi $R_{\max}$
\omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} adalah
\[
E_k = \frac{q^2B^2R_{\max}^2}{2m},
\]
tak bergantung pada \omterm{def:b1:dc-circuits:voltage}{tegangannya}, yang hanya menetapkan jumlah putarannya.
\end{proposition}

\begin{proof}
Karena $\omega_c$ tak bergantung pada kelajuannya, \omterm{def:b1:dc-circuits:voltage}{tegangan} yang frekuensinya
tetap terus selaras dengan partikelnya putaran demi putaran; $v = \abs qBR/m$ pada
jari-jari $R$.
\end{proof}

\begin{example}[Sebuah siklotron rumah sakit]\label{ex:b1:charged-particles:medical}
Proton, $B = \qty{1.5}{T}$, $R_{\max} = \qty{0.50}{m}$: $E_k =
(1.6\times10^{-19} \times 1.5 \times 0.5)^2/(2 \times 1.67\times10^{-27}) =
\qty{4.3e-12}{J} = \qty{27}{MeV}$, pada $f_c = \omega_c/2\pi = \qty{23}{MHz}$.
Dengan \qty{50}{kV} pada dee-nya, \qty{100}{keV} tiap putaran: $270$ putaran dalam
\qty{12}{\micro s}. Di atas beberapa puluh MeV pertambahan massa
relativistiknya (volume Tahun ke-3) membuat protonnya keluar dari langkah;
mesin terapi pada \qty{230}{MeV} memodulasi frekuensi atau medannya untuk
mengimbanginya --- itulah pokok soal akhir pekannya.
\end{example}

\begin{proposition}[Efek Hall]\label{prop:b1:charged-particles:hall}
\index{efek Hall}
Sebuah pita setebal $d$ yang mengalirkan arus $I$ berupa pembawa bermuatan
$q$ dan berkonsentrasi $n$, dalam medan $B$ yang tegak lurus pada pitanya,
membangkitkan pada lebarnya \emph{\omterm{def:b1:dc-circuits:voltage}{tegangan} Hall}
\[
U_H = \frac{IB}{nqd} :
\]
sebanding dengan $B$ (sebuah magnetometer) dan berbanding terbalik dengan $n$
(sebuah penduga konsentrasi dan tanda pembawanya).
\end{proposition}

\begin{proof}
Gaya magnet $qv_dB$ pada pembawanya (laju hanyut $v_d$) mendorongnya
ke satu tepi sampai \omterm{def:b1:charged-particles:lorentz}{medan listrik} melintang $E_H$ mengimbanginya:
$qE_H = qv_dB$, dan $U_H = E_Hw$ sepanjang lebar $w$; dengan $I = nqv_dwd$,
$E_Hw = v_dBw = IB/(nqd)$.
\end{proof}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:3}

\begin{exercise}[$\star$]\label{exo:b1:charged-particles:4}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:5}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:6}
Ion $^{12}$C dan $^{13}$C bermuatan tunggal, dipercepat melalui
\qty{20}{kV}, masuk ke dalam medan \qty{0.50}{T} ($\qty{1}{u} = \qty{1.66e-27}{kg}$).
Jari-jari dan pemisahan kedua bintiknya setelah setengah putaran.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:7}
Sebuah siklotron proton mempunyai $B = \qty{1.5}{T}$ dan $R_{\max} = \qty{0.50}{m}$.
Frekuensi \omterm{def:b1:dc-circuits:voltage}{tegangan} dee-nya, \omterm{thm:b1:work-and-energy:ke}{energi kinetik} maksimalnya, jumlah putaran
untuk \qty{50}{kV} pada dee-nya, waktu yang dihabiskan di dalamnya. Sebesar
pecahan berapa massa protonnya bertambah di pintu keluar ($\gamma - 1 \approx E_k/mc^2$,
$mc^2 = \qty{938}{MeV}$)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:8}
Sebuah elektron pada \qty{1.0e6}{m/s} masuk ke dalam medan \qty{2.0}{mT} pada $30^\circ$
terhadap garis medannya. Jari-jari dan langkah heliksnya; jumlah putaran
tiap meter menyusuri medannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:charged-particles:9}
Sebuah bilah tembaga setebal \qty{1.0}{mm} mengalirkan \qty{5.0}{A} dalam \qty{1.0}{T}
($n = \qty{8.5e28}{m^{-3}}$). \omterm{def:b1:dc-circuits:voltage}{Tegangan} Hallnya. Mengapa penginderaan Hall dibuat dari
semikonduktor ($n \sim \qty{1e22}{m^{-3}}$)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:10}
Buktikan dari \omterm{thm:b1:newton-dynamics:laws}{hukum Newton} bahwa sebuah muatan dalam \omterm{def:b1:charged-particles:lorentz}{medan magnet} apa pun
(seragam atau tidak), tanpa \omterm{def:b1:charged-particles:lorentz}{medan listrik}, mempertahankan kelajuan yang tetap.
Sebuah elektron \qty{10}{keV} masuk ke daerah bermedan kuat dan tak seragam:
apa yang dapat dan tak dapat berubah?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:11}
Sebuah muatan $q$ mulai dari keadaan diam di titik asal dalam $\vect E = E\vect e_y$ dan
$\vect B = B\vect e_z$. Tuliskan persamaan geraknya, tunjukkan bahwa
partikelnya menghanyut secara rata-rata dengan kecepatan $E/B$ menyusuri $\vect e_x$
(petunjuk: carilah penyelesaian berbentuk $\vect v = \vect u + \vect w$ dengan
$\vect u$ tetap), dan lukiskan \omterm{def:b1:point-kinematics:frame}{lintasannya} (sebuah sikloida). Angkanya untuk
sebuah elektron dengan $E = \qty{1.0e4}{V/m}$, $B = \qty{0.10}{T}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:charged-particles:12}
Pada osiloskop \cref{exo:b1:charged-particles:5}, gantilah kepingnya
dengan kumparan pembelok magnetik yang menghasilkan \qty{1.0}{mT} sepanjang
\qty{4.0}{cm} yang sama. Sudut pembelokannya (sudut kecil: busur berjari-jari $R$
sepanjang $L$), dan pembelokan pada layarnya. Mengapa tabung televisi,
dengan sudutnya yang besar, memakai pembelokan magnetik dan bukan listrik?
\end{exercise}

\section{Soal: Proton melawan sebuah tumor}

\begin{problem}[{Soal akhir pekan --- siklotron di balik dinding sebuah
rumah sakit: bagaimana proton dipusingkan sampai sepertiga laju cahaya,
mengapa mesin sederhananya berhenti bekerja sebelum mencapai energi yang
dituntut seorang terapis, dan berapa banyak proton yang diperlukan untuk
mengobati sebuah tumor}]
\label{pb:b1:charged-particles:1}
Proton: $m = \qty{1.67e-27}{kg}$, $q = e = \qty{1.60e-19}{C}$, $mc^2 =
\qty{938}{MeV}$; $c = \qty{3.00e8}{m/s}$. Siklotron: $B = \qty{1.5}{T}$,
jari-jari ekstraksi $R_{\max} = \qty{0.50}{m}$, \omterm{def:b1:wave-propagation:signal}{amplitudo} \omterm{def:b1:dc-circuits:voltage}{tegangan} dee
$U = \qty{50}{kV}$.

\medskip
\noindent\textbf{Bagian I --- Satu putaran.}
\begin{enumerate}
  \item Tuliskan \omterm{def:b1:charged-particles:lorentz}{gaya Lorentz} pada sebuah proton yang bergerak dengan $\vect v$
        tegak lurus pada $\vect B$, dan tunjukkan bahwa kelajuannya tak dapat
        berubah di dalam sebuah dee.
  \item Turunkan jari-jari lingkarannya dan periodenya; tunjukkan bahwa
        periodenya tak bergantung pada kelajuannya.
  \item Hitunglah \omterm{thm:b1:charged-particles:bfield}{frekuensi siklotron} $f_c$ bagi proton-proton ini.
  \item Energi yang diperoleh pada tiap penyeberangan celah; tiap putaran.
  \item Mengapa \omterm{def:b1:dc-circuits:voltage}{tegangan} dee-nya harus berbolak-balik tepat pada $f_c$? Apa
        yang terjadi bila ia meleset sedikit?
  \item Protonnya disuntikkan di pusat dengan kelajuan yang dapat diabaikan:
        jari-jari setelah putaran pertama? Setelah $n$ putaran?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Menuju tepi.}
\begin{enumerate}[resume]
  \item \omterm{thm:b1:work-and-energy:ke}{Energi kinetik} pada ekstraksinya (dalam J dan MeV), dan kelajuannya,
        sebagai pecahan $c$.
  \item Jumlah putaran yang diperlukan dan waktu yang dihabiskan di dalam mesinnya.
  \item Panjang total \omterm{def:b1:point-kinematics:frame}{lintasan} pilinnya (jumlah kelilingnya ---
        hampirilah jumlahnya dengan sebuah integral).
  \item Apakah hasil pertanyaan 7 bergantung pada $U$? Pada apa $U$
        bekerja?
  \item Hampa di dalamnya harus baik: taksirlah jarak yang ditempuh sebuah
        proton di antara dua tumbukan dengan molekul gas sisa bila \omterm{def:b1:point-kinematics:frame}{lintasan}
        bebas rata-ratanya harus melampaui panjang pilinnya; \omterm{def:b1:point-kinematics:frame}{lintasan} bebas
        rata-rata berskala sebagai $1/p$ dan kira-kira \qty{70}{nm} pada tekanan
        atmosfer. Tekanan berapa yang diperlukan?
  \item Hitunglah $\gamma - 1 \approx E_k/mc^2$ pada ekstraksinya dan
        perubahan nisbi periodenya yang menyusul. Sepanjang jumlah putaran
        yang ditemukan di atas, sebesar pecahan periode berapa protonnya
        tergelincir terhadap \omterm{def:b1:dc-circuits:voltage}{tegangannya}? Berilah komentar.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- 230 MeV.}
Terapi proton menuntut kira-kira \qty{230}{MeV} untuk mencapai tumor
sedalam \qty{30}{cm}.
\begin{enumerate}[resume]
  \item Dengan rumus tak relativistiknya, hasil kali $BR$ berapa yang akan
        memberikan \qty{230}{MeV}? Dengan $B = \qty{2.0}{T}$, berapa jari-jarinya?
  \item Pada energi itu $\gamma - 1 = 0.245$: apa yang terjadi pada periode
        siklotronnya ketika protonnya memperoleh energi, dan mengapa
        siklotron berfrekuensi tetap gagal?
  \item Ada dua penawarnya: mengubah-ubah frekuensi \omterm{def:b1:dc-circuits:voltage}{tegangannya} selama
        pemercepatan (sinkrosiklotron), atau membuat $B$ tumbuh terhadap jari-jari sehingga
        $B/\gamma$ tinggal tetap (siklotron isokron). Jelaskan
        masing-masing dalam satu kalimat dan sebutkan satu kelemahan yang pertama.
  \item Momentum relativistiknya adalah $p = \gamma mv$ dan jari-jarinya $R =
        p/qB$ masih berlaku: dengan $\gamma = 1.245$ dan $v = 0.60c$, hitunglah
        $R$ untuk $B = \qty{2.0}{T}$ dan bandingkan dengan pertanyaan 13.
  \item Berkas yang diekstraksi lewat di antara dua keping sepanjang \qty{1.0}{m}
        dengan $E = \qty{1.0e5}{V/m}$ agar dikemudikan sebesar sudut kecil:
        taksirlah sudut pembelokannya (secara tak relativistik, dengan $p$
        dari pertanyaan sebelumnya sebagai pengganti $mv$). Apakah pengemudian
        listrik praktis di sini?
  \item Pengemudian yang sama dengan magnet \qty{1.0}{T} sepanjang \qty{1.0}{m}:
        sudutnya? Simpulkan mengapa saluran berkas memakai magnet.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Dosisnya.}
\begin{enumerate}[resume]
  \item Sebuah proton \qty{230}{MeV} menyimpan sebagian besar energinya di dekat
        ujung jangkauannya (puncak Bragg). Bila semuanya diserap dalam sebuah
        tumor bermassa \qty{0.50}{kg}, berapa banyak proton yang menyampaikan dosis
        \qty{2.0}{Gy} ($\qty{1}{Gy} = \qty{1}{J/kg}$)?
  \item Arus berkasnya \qty{1.0}{nA}: berapa proton tiap sekon, dan
        lamanya satu sesi.
  \item Daya yang dibawa berkasnya; bandingkan dengan sebuah bohlam.
  \item Protonnya diperlambat sampai berhenti di dalam tubuh pasiennya: berapa
        kira-kira energi yang disimpan tiap proton pada sentimeter terakhirnya,
        bila seperempat energinya hilang di sana? Dalam \omterm{def:b1:charged-particles:eV}{elektronvolt} tiap
        mikrometer?
  \item Tiap ionisasi dalam jaringan menuntut kira-kira \qty{30}{eV}: berapa
        banyak ionisasi yang ditimbulkan satu proton sepanjang seluruh \omterm{def:b1:point-kinematics:frame}{lintasannya}?
  \item Mengapa berkas proton lebih disukai daripada sinar-X bagi tumor yang
        dalam, dalam satu kalimat tentang ke mana energinya pergi?
  \item Ringkaslah rantainya dari \omterm{def:b1:charged-particles:lorentz}{gaya Lorentz} sampai dosisnya: persamaan mana
        yang menetapkan energinya, jari-jarinya, jumlah putarannya, dan
        waktu pengobatannya.
\end{enumerate}
\end{problem}
