\section{Model optika geometri}

\begin{definition}[Sinar cahaya; medium homogen]\label{def:b1:rays-reflection-refraction:ray}
\index{sinar cahaya}\index{optika geometri}\index{medium homogen}
Dalam model \emph{optika geometri}, cahaya merambat sepanjang
\emph{sinar cahaya}: kurva berarah yang berupa garis lurus dalam medium
\emph{homogen} (sifatnya sama di setiap titik) dan yang membawa energi
saling bebas satu sama lain. Berkas adalah seikat sinar; sumber titik
memancarkan sinar ke segala arah.
\end{definition}

\begin{definition}[Indeks bias]\label{def:b1:rays-reflection-refraction:index}
\index{indeks bias}\index{medium bening}
Medium \emph{bening} meneruskan cahaya; di dalamnya cahaya merambat
dengan laju $v < c$. \emph{Indeks bias}nya adalah
\[
n = \frac{c}{v} \geq 1 .
\]
Hampa: $n = 1$; udara: $1.0003$; air: $1.33$; kaca biasa: $1.5$
sampai $1.6$; intan: $2.42$. Gelombang berfrekuensi $\nu$ tetap
memegang frekuensinya ketika memasuki medium, sehingga panjang
gelombangnya menyusut dari $\lambda_0 =
c/\nu$ di hampa menjadi $\lambda = \lambda_0/n$.
\end{definition}

\begin{remark}[Dispersi]\label{rem:b1:rays-reflection-refraction:dispersion}
\index{dispersi (optik)}\index{hukum Cauchy}
Di dalam materi indeksnya sedikit bergantung pada panjang gelombang ---
mediumnya \emph{dispersif}. Untuk kaca di daerah tampak, hukum empiris
Cauchy $n(\lambda) = A + B/\lambda^2$ cocok dengan baik: $n$ lebih besar
untuk biru daripada untuk merah, selisihnya sekitar $0.01$ sampai
$0.02$. Prisma dan pelangi, di penghujung bab ini, hidup dari selisih itu.
\end{remark}

\begin{remark}[Batas model]\label{rem:b1:rays-reflection-refraction:limits}
Cahaya adalah gelombang, dan gelombang yang melewati celah selebar $a$
melebar dengan sudut berorde $\lambda/a$ (difraksi,
\cref{ch:b1:wave-propagation}). Sinar merupakan gambaran yang baik
selama setiap celah, lensa dan penghalang jauh lebih besar daripada
panjang gelombangnya (\qty{0.4}{} sampai \qty{0.8}{\micro m} untuk
cahaya tampak): lubang \qty{1}{mm} melebarkan berkas sebesar $10^{-3}$
radian, dapat diabaikan di atas meja kerja; lubang \qty{1}{\micro m}
melebarkannya ke seluruh setengah ruang, dan tak ada gambaran sinar
yang bertahan. \omterm{def:b1:rays-reflection-refraction:fiber}{Inti} \qty{9}{\micro m} sebuah serat telekomunikasi moda
tunggal sudah berada di luar jangkauan model ini.
\end{remark}

\begin{proposition}[Jalan balik cahaya]\label{prop:b1:rays-reflection-refraction:reversibility}
\index{asas jalan balik cahaya}
Lintasan sebuah sinar tidak bergantung pada arah rambat cahaya yang
menempuhnya: jika cahaya berjalan dari $A$ ke $B$ menyusuri sebuah
lintasan, cahaya dari $B$ menempuh lintasan yang sama menuju $A$.
\end{proposition}
\admitted

\section{Pemantulan dan pembiasan}

\begin{definition}[Datangnya sinar]\label{def:b1:rays-reflection-refraction:incidence}
\index{bidang datang}\index{sudut datang}\index{permukaan bias}
Sinar yang menemui permukaan pemisah dua medium (sebuah
\emph{permukaan bias}) di titik $I$ membentuk terhadap \emph{garis
normal} permukaan di $I$ sebuah \emph{sudut datang} $i_1$. \emph{Bidang
datang} memuat sinar datang dan garis normalnya.
\end{definition}

\begin{theorem}[Hukum pemantulan dan pembiasan (Snell--Descartes)]\label{thm:b1:rays-reflection-refraction:snell}
\index{pemantulan!hukum}\index{pembiasan!hukum}\index{hukum Snell}
Pada permukaan antara medium berindeks $n_1$ (sisi datang) dan
$n_2$:
\begin{enumerate}
  \item sinar pantul dan sinar bias terletak di \omterm{def:b1:rays-reflection-refraction:incidence}{bidang datang};
  \item sinar pantul membentuk terhadap garis normal sudut $i_1' = i_1$,
        di sisi seberang garis normal;
  \item sinar bias membentuk terhadap garis normal sudut $i_2$ sedemikian
        sehingga
        \[
        n_1 \sin i_1 = n_2 \sin i_2 .
        \]
\end{enumerate}
\end{theorem}
\admitted

\begin{remark}[Dari mana hukum-hukum itu datang]\label{rem:b1:rays-reflection-refraction:fermat}
\index{asas Fermat}
Kedua hukum itu menyusul dari \emph{asas Fermat}: di antara dua titik,
cahaya menempuh lintasan dengan waktu tempuh terkecil. Pemantulan dengan
sudut sama adalah lintasan patah terpendek yang menyentuh cermin; \omterm{thm:b1:rays-reflection-refraction:snell}{hukum
Snell} adalah lintasan yang menukar rute lebih panjang di medium cepat
dengan rute lebih pendek di medium lambat
(\cref{exo:b1:rays-reflection-refraction:12}). Dalam volume Tahun ke-2
kedua hukum itu diturunkan dari sifat gelombang cahaya di sebuah antarmuka.
\end{remark}

\begin{proposition}[Ke mana sinarnya membelok]\label{prop:b1:rays-reflection-refraction:bending}
Ketika memasuki medium yang lebih membiaskan ($n_2 > n_1$), sebuah sinar
membelok \emph{mendekati} garis normal ($i_2 < i_1$); ketika memasuki
medium yang kurang membiaskan, ia membelok \emph{menjauhinya}
($i_2 > i_1$). Sinar yang berimpit dengan garis normal lewat tanpa
dibelokkan.
\end{proposition}

\begin{proof}
$\sin i_2 = (n_1/n_2)\sin i_1$ dengan $\sin$ yang naik pada
$[0, \pi/2]$.
\end{proof}

\begin{example}[Kedalaman semu]\label{ex:b1:rays-reflection-refraction:depth}
\index{kedalaman semu}
Sekeping uang logam terletak pada kedalaman $h$ di dalam air
($n = 1.33$), dilihat hampir tegak lurus dari atas. Sinar dari uang itu
mencapai permukaan dengan sudut kecil $i_2$ (di dalam air) lalu keluar
pada $i_1$ dengan $\sin i_1 = n\sin i_2$; untuk sudut kecil
$i_1 \approx n\,i_2$. Mata memperpanjang sinar yang keluar itu ke
belakang: uangnya seolah datang dari kedalaman $h'$ dengan
$\tan i_1 \approx x/h'$ dan $\tan i_2 \approx x/h$ untuk titik permukaan
yang sama pada jarak $x$ dari garis tegak, sehingga
$h' = h\,i_2/i_1 = h/n$. Kolam sedalam \qty{2.0}{m} tampak sedalam
\qty{1.5}{m}; ikan melihat pemancing $1.33$ kali lebih tinggi daripada
tingginya yang sebenarnya.
\end{example}

\begin{theorem}[Pemantulan sempurna]\label{thm:b1:rays-reflection-refraction:tir}
\index{pemantulan sempurna}\index{sudut kritis}
Cahaya yang berjalan dari medium berindeks $n_1$ menuju medium yang
kurang membiaskan ($n_2 < n_1$) hanya dibiaskan jika $i_1 \leq i_c$,
dengan \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{sudut kritis}} yang memenuhi
\[
\sin i_c = \frac{n_2}{n_1} .
\]
Untuk $i_1 > i_c$ tidak ada sinar bias: permukaannya memantulkan seluruh
cahaya --- \emph{\omterm{thm:b1:rays-reflection-refraction:tir}{pemantulan sempurna}}.
\end{theorem}

\begin{proof}
\omterm{thm:b1:rays-reflection-refraction:snell}{Hukum Snell} menuntut $\sin i_2 = (n_1/n_2)\sin i_1 > \sin i_1$; nilai ini
melampaui $1$ begitu $\sin i_1 > n_2/n_1$, dan setelah itu tak ada sudut
$i_2$ yang ada. Pada $i_1 = i_c$ sinar biasnya menyusur permukaan
($i_2 = 90^\circ$). Bahwa \emph{seluruh} energinya lalu dipantulkan
adalah pernyataan tentang intensitas, yang tak ditangani \omterm{def:b1:rays-reflection-refraction:ray}{optika
geometri}; pembahasan gelombang dalam volume Tahun ke-2 membenarkannya.
\end{proof}

\begin{example}[Intan dan perenang]\label{ex:b1:rays-reflection-refraction:tir}
Kaca--udara: $\sin i_c = 1/1.5$, $i_c = 41.8^\circ$, yang menjelaskan

\end{example}

\section{Serat optik}

\begin{definition}[Serat indeks tangga]\label{def:b1:rays-reflection-refraction:fiber}
\index{serat optik}\index{inti}\index{selubung}\index{apertur numerik}
\emph{Serat optik indeks tangga} adalah \emph{inti} kaca berbentuk
tabung berindeks $n_1$ yang dibungkus \emph{selubung} berindeks $n_2$
yang sedikit lebih rendah. Cahaya yang memasuki intinya di dalam kerucut
bersetengah sudut $\theta_a$ di sekitar sumbunya dituntun oleh
\omterm{thm:b1:rays-reflection-refraction:tir}{pemantulan sempurna} berturut-turut pada permukaan inti--selubung.
\emph{Apertur numerik}nya adalah $\mathrm{NA} = \sin\theta_a$.
\end{definition}

\begin{proposition}[Kerucut penerimaan dan pemuluran antarmoda]\label{prop:b1:rays-reflection-refraction:fiber}
\index{sudut penerimaan}\index{dispersi antarmoda}
Untuk serat yang muka masuknya berada di udara,
\[
\sin\theta_a = \sqrt{n_1^2 - n_2^2} .
\]
Sepanjang panjang $L$, sinar terpandu yang menyusuri sumbu dan sinar
terpandu yang paling curam tiba dengan selisih waktu
\[
\Delta t = \frac{L\,n_1}{c}\left(\frac{n_1}{n_2} - 1\right)
\]
di antara keduanya (\emph{\omterm{prop:b1:rays-reflection-refraction:fiber}{dispersi antarmoda}}).
\end{proposition}

\begin{proof}
Sinar yang masuk dengan sudut $\theta$ terhadap sumbu dibiaskan menjadi
$\theta'$ dengan $\sin\theta = n_1\sin\theta'$ lalu menemui \omterm{def:b1:rays-reflection-refraction:fiber}{selubungnya}
dengan \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang} $i = 90^\circ - \theta'$. Penuntunan menuntut
$\sin i \geq n_2/n_1$, yaitu $\cos\theta' \geq n_2/n_1$, yaitu $\sin\theta' \leq
\sqrt{1 - n_2^2/n_1^2}$; kalikan dengan $n_1$. Sinar sumbunya menempuh
$L$ dengan laju $c/n_1$: $t_0 = Ln_1/c$. Sinar paling curam berkelok
dengan sudut $\theta'_{\max}$ dengan $\cos\theta'_{\max} = n_2/n_1$;
lintasannya $L/\cos\theta'_{\max} = L n_1/n_2$, sehingga
$t_1 = L n_1^2/(n_2 c)$, dan $\Delta t = t_1 - t_0$.
\end{proof}

\begin{example}[Angka-angka sebuah serat telekomunikasi]\label{ex:b1:rays-reflection-refraction:fibernum}
$n_1 = 1.480$, $n_2 = 1.465$: $\mathrm{NA} = \sqrt{2.190 - 2.146} =

antarmoda: $\Delta t/L =
(1.48/c)(1.48/1.465 - 1) = \qty{51}{ns/km}$. Sepanjang \qty{2}{km}
sebuah pulsa melebar sebesar \qty{0.1}{\micro s}, yang membatasi laju
bitnya di sekitar \qty{5}{Mbit/s}: soal akhir pekan mengikuti
keterbatasan ini sampai ke obatnya, yaitu serat moda tunggal.
\end{example}

\section{Cermin datar}

\begin{proposition}[Bayangan pada cermin datar]\label{prop:b1:rays-reflection-refraction:mirror}
\index{cermin datar}\index{bayangan maya}\index{stigmatisme}
Semua sinar dari sebuah titik $A$ yang dipantulkan \omterm{prop:b1:rays-reflection-refraction:mirror}{cermin datar} seolah
datang dari titik $A'$ yang setangkup terhadap $A$ relatif terhadap
bidang cerminnya: $A'$ adalah \emph{bayangan} $A$. Bayangan itu
\emph{maya} (tak ada cahaya yang melaluinya) dan cerminnya
\emph{stigmatik secara ketat}: setiap titik mempunyai bayangan titik
yang eksak.
\end{proposition}

\begin{proof}
Sinar dari $A$ yang mengenai cermin di $I$ dengan \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang} $i$
keluar pada $i$ di sisi seberang garis normal. Garis pantulnya dan garis
normal di $I$ membentuk sudut $i$, sebagaimana garis $IA$; oleh
kesetangkupan terhadap bidang cermin, garis pantulnya adalah bayangan
cermin garis $IA$, sehingga ia melalui titik setangkup $A'$ --- untuk
setiap $I$.
\end{proof}

\begin{proposition}[Memutar sebuah cermin]\label{prop:b1:rays-reflection-refraction:rotation}
Memutar \omterm{prop:b1:rays-reflection-refraction:mirror}{cermin datar} sebesar sudut $\alpha$ terhadap sumbu yang terletak
pada bidangnya dan tegak lurus \omterm{def:b1:rays-reflection-refraction:incidence}{bidang datang} memutar sinar pantulnya
sebesar $2\alpha$.
\end{proposition}

\begin{proof}
Garis normalnya berputar sebesar $\alpha$, sehingga \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datangnya}
berubah dari $i$ menjadi $i + \alpha$; sinar pantulnya, yang berada pada
$i + \alpha$ di sisi seberang garis normal yang baru, telah bergeser
sebesar $\alpha$ (normal) $+ \alpha$ (sudut) $=
2\alpha$ dari arahnya yang lama.
\end{proof}

\begin{example}[Tuas sang galvanometer]\label{ex:b1:rays-reflection-refraction:galvanometer}
Cermin mungil yang dilekatkan pada kumparan sebuah galvanometer, disinari
lampu dan melemparkan bintik cahaya pada skala sejauh \qty{2}{m}:
perputaran kumparan sebesar $\qty{1}{mrad}$ menggeser bintiknya sebesar
$2 \times 10^{-3} \times
\qty{2}{m} = \qty{4}{mm}$ --- sebuah tuas optik yang memperbesar tanpa
gesekan, asas galvanometer cermin dan pemindai laser.
\end{example}

\section{Prisma}

\begin{proposition}[Rumus prisma]\label{prop:b1:rays-reflection-refraction:prism}
\index{prisma}\index{deviasi}\index{deviasi minimum}
Sinar yang melintasi prisma bersudut puncak $A$ dan berindeks $n$ di
udara, pada bidang yang tegak lurus rusuknya, dengan sudut $i$ (masuk),
$r$, $r'$ (di dalam) dan $i'$ (keluar) yang diukur dari garis-garis
normalnya, memenuhi
\[
\sin i = n \sin r, \qquad r + r' = A, \qquad \sin i' = n \sin r',
\]
dan dideviasikan sebesar $D = i + i' - A$. Deviasinya minimum ketika
lintasannya setangkup, $i = i'$, $r = r' = A/2$; maka
\[
n = \frac{\sin\frac{A + D_{\min}}{2}}{\sin\frac{A}{2}} .
\]
\end{proposition}

\begin{proof}
\omterm{thm:b1:rays-reflection-refraction:snell}{Hukum Snell} pada tiap muka; $r + r' = A$ karena kedua garis normalnya
membentuk sudut $A$ (segi empat yang dibentuk puncaknya, kedua titik
datang dan perpotongan garis-garis normalnya). Deviasi totalnya
$(i - r) + (i' - r') = i + i' - A$. Oleh jalan balik cahaya, deviasinya
adalah fungsi yang sama dari $i$ maupun dari $i'$; jika minimumnya
terjadi pada $i_0 \neq i_0'$ ia juga akan terjadi pada \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang}
terbalik $i_0'$, sehingga ada dua minimum, yang dilarang oleh sifat
monoton $D(i)$ pada tiap sisi minimumnya: minimumnya berada di $i = i'$.
Maka $r = r' = A/2$ dan $D_{\min} = 2i - A$; menyubstitusikan $i = (A +
D_{\min})/2$ ke $\sin i = n\sin(A/2)$ memberi rumusnya.
\end{proof}

\begin{example}[Mengukur sebuah indeks]\label{ex:b1:rays-reflection-refraction:prismnum}
Prisma $60^\circ$ diputar sampai deviasi berkas kuning menjadi terkecil:
$D_{\min} = 38.9^\circ$. Maka $n = \sin 49.45^\circ/\sin 30^\circ
= 0.760/0.500 = 1.52$. \omterm{prop:b1:rays-reflection-refraction:prism}{Deviasi minimum} adalah cara baku mengukur indeks
sebuah kaca --- minimumnya yang tajam, ditemukan dengan memutar
prismanya maju mundur, tidak peka terhadap salah setel yang kecil.
\end{example}

\begin{remark}[Prisma tipis dan spektroskop]\label{rem:b1:rays-reflection-refraction:thin}
Untuk $A$ kecil dan \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang} kecil, sinusnya sama dengan sudutnya
dan $D \approx (n - 1)A$, bebas dari $i$: prisma tipis mendeviasikan
setiap sinar dengan sudut yang sama --- sebuah baji kaca adalah
``prisma'' yang membelokkan cahaya, dan lensa kelak ternyata merupakan
tumpukan baji semacam itu. Karena $n$ bergantung pada $\lambda$,
deviasinya pun bergantung: prisma menguraikan cahaya putih menjadi
spektrum (biru paling dideviasikan), jantung spektroskop prisma.
\end{remark}

\section{Pelangi}

\begin{example}[Pelangi Descartes]\label{ex:b1:rays-reflection-refraction:rainbow}
\index{pelangi}
Sinar matahari memasuki tetes hujan berbentuk bola dengan \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang}
$i$, dibiaskan ($\sin i = n\sin r$), memantul satu kali pada permukaan
belakangnya, lalu keluar setelah pembiasan kedua. Tiap pembiasan
mendeviasikannya sebesar $i - r$, pemantulan dalamnya sebesar
$\pi - 2r$: deviasi totalnya $D(i) = \pi + 2i - 4r$. Sinar memenuhi
tetesnya pada setiap $i$; pengamat melihat cahaya paling terang di
tempat banyak sinar keluar dengan arah yang nyaris sama, yaitu di tempat
$D$ stasioner: $\dd D/\dd i = 2 - 4\,\dd r/\dd i
= 0$. Dengan menurunkan \omterm{thm:b1:rays-reflection-refraction:snell}{hukum Snell}, $\cos i = n\cos r\,\dd r/\dd i$,
sehingga syaratnya berbunyi $2\cos i = n\cos r$; dikuadratkan dan dengan
memakai $n^2\cos^2 r = n^2 - \sin^2 i = n^2 - 1 + \cos^2 i$ diperoleh
\[
\cos^2 i = \frac{n^2 - 1}{3} .
\]
Untuk air, $n = 1.333$: $i = 59.4^\circ$, $r = 40.2^\circ$,
$D_{\min} = 137.9^\circ$. Cahayanya kembali ke sisi Matahari
pada $180^\circ - 137.9^\circ = 42^\circ$ dari arah antimatahari:
pelangi adalah kerucut bersetengah sudut $42^\circ$ di sekitar bayang-bayang
kepala pengamat. Merah ($n = 1.331$) duduk pada $42.4^\circ$, ungu
($n = 1.343$) pada $40.6^\circ$ --- merah di luar, ungu di dalam.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:1}
Hitunglah laju cahaya di dalam air ($n = 1.33$), di dalam kaca ($1.50$)
dan di dalam intan ($2.42$). Berapa lama cahaya menempuh \qty{1.0}{km}
serat berindeks $1.48$?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:2}
Sinar di udara menemui air ($n = 1.33$) pada $30^\circ$, lalu pada
$60^\circ$ dari garis normalnya: carilah sudut biasnya. Sinar di dalam
air menemui permukaannya pada $50^\circ$: apa yang terjadi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:3}
Hitunglah \omterm{thm:b1:rays-reflection-refraction:tir}{sudut kritis} untuk kaca--udara ($n = 1.50$), intan--udara
($2.42$) dan air--udara ($1.33$). Berapa garis tengah sudut cakram
langit yang dilihat seorang penyelam?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:rays-reflection-refraction:4}
Seseorang setinggi \qty{1.70}{m}, matanya \qty{10}{cm} di bawah puncak
kepalanya, berdiri di depan cermin dinding yang tegak. Carilah cermin
terkecil (tingginya, dan letak kedua tepinya di atas lantai) yang
memperlihatkan seluruh tubuhnya. Apakah jawabannya bergantung pada
jaraknya ke cermin?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:5}
Sekeping uang logam terletak \qty{1.2}{m} di dalam kolam; pada
kedalaman berapa ia tampak bagi orang yang melihat dari atas? Mata
seorang pemancing berada \qty{1.8}{m} di atas air: setinggi apa
matanya tampak bagi ikan yang menengadah? Berikan alasan keduanya
dengan sinar bersudut kecil.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:6}
Sebuah prisma mempunyai $A = 60^\circ$ dan $n = 1.52$. Hitunglah
\omterm{prop:b1:rays-reflection-refraction:prism}{deviasi minimumnya}. Tunjukkan bahwa sinar hanya dapat keluar dari muka
kedua jika $A \leq 2 i_c$, lalu periksalah bahwa prisma ini memenuhinya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:7}
Serat indeks tangga mempunyai $n_1 = 1.50$, $n_2 = 1.48$. Hitunglah
\omterm{def:b1:rays-reflection-refraction:fiber}{apertur numeriknya}, \omterm{prop:b1:rays-reflection-refraction:fiber}{sudut penerimaannya}, selisih waktu antarmoda tiap
kilometer, dan laju bit maksimum yang dihasilkannya sepanjang
\qty{1}{km} (ambil $B_{\max} \approx 1/(2\Delta t)$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:8}
Prisma tipis bersudut puncak $10^\circ$ mempunyai $n = 1.510$ untuk
merah dan $1.530$ untuk biru. Hitunglah deviasi tiap warnanya dan lebar
sudut spektrum yang dihasilkannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:rays-reflection-refraction:9}
Sinar menemui pelat kaca bermuka sejajar ($n = 1.50$, tebal
$e = \qty{2.0}{cm}$) pada $45^\circ$. Tunjukkan bahwa ia keluar sejajar
dengan arah datangnya, tergeser ke samping sebesar
$d = e\,\sin(i - r)/\cos r$, lalu hitunglah $d$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:10}
Pelangi sekunder berasal dari sinar yang dipantulkan \emph{dua kali} di
dalam tetesnya. Tunjukkan bahwa $D = 2\pi + 2i - 6r$, bahwa deviasi
stasionernya memenuhi $\cos^2 i = (n^2 - 1)/8$, lalu hitunglah sudut
pelangi sekunder dari arah antimatahari untuk $n = 1.333$. Di mana
letaknya relatif terhadap pelangi utama, dan dalam urutan apa warnanya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:11}
Di atas jalan yang panas, udara tepat di atas aspalnya berindeks
$n_{\text{panas}} = 1.00026$, udara yang lebih sejuk di atasnya
$n_{\text{sejuk}} =
1.00029$. Modelkan lapisannya sebagai antarmuka tajam: carilah sudut
susur kritis (diukur dari jalannya) yang di bawahnya sinar dari langit
dipantulkan sempurna ke atas. Mata pengemudi berada \qty{1.2}{m} di atas
jalan: di luar jarak berapa jalannya tampak seperti air?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:rays-reflection-refraction:12}
Titik $A$ (setinggi $a$ di atas sebuah antarmuka) dan $B$ (sedalam $b$
di bawahnya) terpisah mendatar sejauh $d$; cahaya merambat dengan $v_1$
di atas dan $v_2$ di bawah. Tuliskan waktu tempuh lintasan patah yang
melalui titik antarmuka pada kedudukan mendatar $x$, lalu tunjukkan
bahwa waktunya stasioner tepat ketika $\sin i_1/v_1 = \sin i_2/v_2$ ---
\omterm{thm:b1:rays-reflection-refraction:snell}{hukum Snell}.
\end{exercise}

\section{Soal: Sambungan serat optik}

\begin{problem}[{Soal akhir pekan --- seutas benang kaca di bawah laut:
sejauh dan secepat apa sebuah pulsa cahaya dapat membawa satu bit
sebelum geometri, kaca dan dispersi mengaburkannya}]
\label{pb:b1:rays-reflection-refraction:1}
Sebuah sambungan memakai serat indeks tangga dengan \omterm{def:b1:rays-reflection-refraction:fiber}{inti} $n_1 = 1.480$,
\omterm{def:b1:rays-reflection-refraction:fiber}{selubung} $n_2 = 1.465$, jari-jari \omterm{def:b1:rays-reflection-refraction:fiber}{inti} $a = \qty{25}{\micro m}$, pada
panjang gelombang $\lambda_0 = \qty{1550}{nm}$. Data: $c = \qty{3.00e8}{m/s}$.

\medskip
\noindent\textbf{Bagian I --- Cahaya di dalam kaca.}
\begin{enumerate}
  \item Definisikan indeksnya lalu hitunglah laju cahaya di dalam \omterm{def:b1:rays-reflection-refraction:fiber}{intinya}.
  \item Berapa lama sebuah pulsa menempuh \qty{50}{km} menyusuri
        sumbunya?
  \item Hitunglah frekuensi cahayanya dan panjang gelombangnya di dalam
        \omterm{def:b1:rays-reflection-refraction:fiber}{inti}.
  \item Mengapa \omterm{def:b1:rays-reflection-refraction:fiber}{intinya} dibungkus kaca berindeks lebih rendah dan bukan
        dibiarkan telanjang di udara, yang justru memberi tangga indeks
        lebih besar?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Penuntunan.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{thm:b1:rays-reflection-refraction:tir}{sudut kritis} pada permukaan inti--selubung.
  \item Simpulkan sudut terbesar yang dapat dibentuk sinar terpandu
        terhadap sumbunya.
  \item Sinar masuk lewat muka ujung yang datar dari udara dengan sudut
        $\theta$ terhadap sumbunya. Tunjukkan bahwa ia terpandu jika
        $\sin\theta \leq
        \sqrt{n_1^2 - n_2^2}$, lalu hitunglah \omterm{prop:b1:rays-reflection-refraction:fiber}{sudut penerimaannya}
        $\theta_a$.
  \item Berikan \omterm{def:b1:rays-reflection-refraction:fiber}{apertur numeriknya}. Mengapa tangga indeks yang lebih
        besar membuat serat itu lebih mudah diberi cahaya?
  \item Seratnya dilengkungkan menyusuri lingkaran berjari-jari $R_b$.
        Tinjaulah sinar yang berjalan sejajar sumbunya menyusuri tepi
        dalam \omterm{def:b1:rays-reflection-refraction:fiber}{intinya}: tunjukkan bahwa ia menemui dinding luarnya dengan
        \omterm{def:b1:rays-reflection-refraction:incidence}{sudut datang} $i$ yang memenuhi $\sin i = R_b/(R_b + 2a)$, lalu
        carilah $R_b$ terkecil yang masih menuntunnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- \omterm{prop:b1:rays-reflection-refraction:fiber}{Dispersi antarmoda}.}
\begin{enumerate}[resume]
  \item Nyatakan waktu tempuh sinar sumbu dan sinar terpandu yang paling
        curam sepanjang panjang $L$.
  \item Simpulkan pemuluran $\Delta t$ tiap kilometer, dan nilainya
        sepanjang $L = \qty{2}{km}$.
  \item Satu bit adalah pulsa pendek; dua pulsa berurutan harus tetap
        terpisah sekurang-kurangnya $2\Delta t$ agar dapat dibedakan.
        Taksirlah laju bit maksimum sepanjang \qty{2}{km}.
  \item Tunjukkan bahwa hasil kali (laju bit) $\times$ (panjang) tetap
        untuk serat ini, lalu hitunglah dalam \unit{Mbit.s^{-1}.km}.
  \item Dalam serat indeks berjenjang, indeksnya menurun mulus dari
        sumbu ke luar. Jelaskan secara kualitatif mengapa hal ini
        mengurangi pemuluran antarmoda.
  \item Serat moda tunggal bergaris tengah \omterm{def:b1:rays-reflection-refraction:fiber}{inti} sekitar \qty{9}{\micro m}.
        Bandingkan dengan $\lambda_0$ lalu jelaskan mengapa model sinar
        tak dapat menggambarkannya; apa yang diperoleh?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Redaman dan dispersi kromatik.}
Seratnya meredam dayanya sebesar $\alpha = \qty{0.20}{dB/km}$ pada
\qty{1550}{nm}: $P(L) = P_0\,10^{-\alpha L/10}$. Pemancarnya
melepaskan $P_0 = \qty{1.0}{mW}$; penerimanya memerlukan sekurang-kurangnya
$P_{\min} = \qty{1.0}{\micro W}$. Karena $n$ bergantung pada $\lambda$, dua
panjang gelombang merambat dengan laju yang sedikit berbeda: sumber berlebar
spektrum $\Delta\lambda$ memberi pemuluran pulsa $\Delta t_c = D\,\Delta\lambda\,L$
dengan $D = \qty{17}{ps/(nm.km)}$.
\begin{enumerate}[resume]
  \item Berapa bagian dayanya yang bertahan sepanjang \qty{100}{km}?
  \item Carilah panjang maksimum sebelum sebuah penguat diperlukan.
  \item Penguat dipasang tiap \qty{100}{km}: berapa margin, dalam dB
        dan sebagai nisbah daya, yang tersisa bagi konektor dan
        sambungannya?
  \item Hitunglah $\Delta t_c$ sepanjang \qty{100}{km} untuk sumber LED
        ($\Delta\lambda = \qty{40}{nm}$) dan untuk laser
        ($\Delta\lambda = \qty{0.10}{nm}$); simpulkan laju bit maksimum
        masing-masing sepanjang jarak itu (kriteria yang sama seperti di atas).
  \item Pada \qty{850}{nm} silika meredam \qty{2.0}{dB/km}. Jarak berapa
        yang diizinkan anggaran daya yang sama di sana? Mengapa
        \qty{1550}{nm} menjadi panjang gelombang telekomunikasi?
  \item Kabel koaksial tembaga meredam sekitar \qty{30}{dB/km} pada
        frekuensi yang diperlukan untuk \qty{1}{Gbit/s}. Bandingkan
        jarak antarpenguatnya dengan jarak antarpenguat seratnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian V --- Seluruh sambungannya.}
\begin{enumerate}[resume]
  \item Untuk serat indeks tangga sepanjang \qty{2}{km}, pengaruh mana
        yang membatasi laju bitnya: pemuluran antarmoda, pemuluran
        kromatik (sumber laser) atau redaman? Berikan laju bit
        pembatasnya.
  \item Untuk serat moda tunggal sepanjang \qty{100}{km} dengan sumber
        laser: pengaruh mana yang membatasi, dan pada laju bit berapa?
  \item Sambungan moda tunggal itu membawa $80$ laser pada panjang
        gelombang yang berjarak \qty{0.8}{nm} (multipleksing panjang
        gelombang), masing-masing pada laju bit pertanyaan sebelumnya.
        Berapa kapasitas totalnya?
  \item Bentuklah hasil kali laju bit dan panjang bagi sambungan indeks
        tangga dan bagi satu kanal moda tunggal; dengan faktor berapa
        serat moda tunggal menang, dan fakta fisis tunggal apa yang
        bertanggung jawab?
\end{enumerate}
\end{problem}
