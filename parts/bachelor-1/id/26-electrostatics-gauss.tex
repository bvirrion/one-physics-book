\chapter{Elektrostatika: Medan dan Hukum Gauss}\label{ch:b1:electrostatics-gauss}

Gosokkan sebuah balon pada baju hangat dan ia melekat di dinding; sisirlah rambut
yang kering dan sisirnya membelokkan seurai tipis air. Di balik keduanya ada gaya antara
\omterm{def:b1:electrostatics-gauss:charge}{muatan listrik} yang diam, yaitu \omterm{thm:b1:electrostatics-gauss:coulomb}{hukum Coulomb} --- sebuah hukum kebalikan kuadrat seperti
gravitasi, tetapi $10^{36}$ kali lebih kuat di antara dua proton dan bertanda
dua-duanya, sehingga materi bersifat netral dengan ketelitian yang menakjubkan dan
sisa pengaruhnya adalah yang kita sebut kimia, bahan dan petir.
Bab ini membangun gagasan yang membuat hukum itu dapat dikerjakan, yaitu
\omterm{def:b1:electrostatics-gauss:field}{medan listrik}, memperlihatkan bagaimana membaca sebuah medan dari simetrinya, lalu
menurunkan satu teorema yang menghitung medan distribusi
yang simetris dalam tiga baris: \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss}. Kembarannya dalam gravitasi
datang cuma-cuma di akhirnya.

\section{Muatan dan hukum Coulomb}

\begin{definition}[Muatan listrik]\label{def:b1:electrostatics-gauss:charge}
\index{muatan listrik}\index{muatan elementer}\index{rapat muatan}
Yang disebut \emph{muatan listrik} $q$ (coulomb, \unit{C}) adalah sifat materi
yang bertanda salah satu dari dua; ia kekal (muatan total sebuah sistem terkucil
tak pernah berubah) dan terkuantisasi dalam satuan \emph{muatan elementer}
$e = \qty{1.602e-19}{C}$ (proton $+e$, elektron $-e$). Sebuah distribusi
makroskopik dilukiskan oleh \emph{rapat muatan}: yang linear $\lambda$
(\unit{C/m}), yang permukaan $\sigma$ (\unit{C/m^2}) atau yang volume $\rho$
(\unit{C/m^3}), sehingga sebuah unsur kecil mengemban $\dd q = \lambda\,\dd l$,
$\sigma\,\dd S$ atau $\rho\,\dd\tau$.
\end{definition}

\begin{theorem}[Hukum Coulomb]\label{thm:b1:electrostatics-gauss:coulomb}
\index{hukum Coulomb}\index{permitivitas hampa}
Dua muatan titik $q_1$ di $M_1$ dan $q_2$ di $M_2$ yang diam dalam hampa
mengerahkan gaya satu sama lain
\[
\vect F_{1 \to 2} = \frac{q_1q_2}{4\pi\varepsilon_0\,r^2}\,\vect e_{12} = -\vect F_{2 \to 1},
\qquad r = M_1M_2,\quad \vect e_{12} = \frac{\vect{M_1M_2}}{r},
\]
yang tolak-menolak bagi muatan sejenis, tarik-menarik bagi yang tak sejenis, dengan
\emph{\omterm{thm:b1:electrostatics-gauss:coulomb}{permitivitas hampa}} $\varepsilon_0 = \qty{8.854e-12}{F/m}$, yakni
$1/4\pi\varepsilon_0 = \qty{8.99e9}{N.m^2/C^2}$. Gaya dari beberapa muatan berjumlah
(superposisi).
\end{theorem}
\admitted

\begin{example}[Seberapa kuat]\label{ex:b1:electrostatics-gauss:strong}
Dua proton yang terpisah \qty{1}{fm} tolak-menolak dengan $8.99 \times 10^9 \times (1.6 \times
10^{-19})^2/10^{-30} = \qty{230}{N}$ --- seberat seorang anak, di antara dua
partikel bermassa $10^{-27}$~kg; tarikan gravitasinya
$10^{36}$ kali lebih kecil. Dua gram elektron di satu ujung sebuah ruangan
dan proton yang menyertainya di ujung lainnya akan saling menarik dengan $10^{25}$~N.
Materi netral karena ia tak punya pilihan: setiap ketakseimbangan diperbaiki
oleh gaya yang tak dapat ditahan struktur apa pun.
\end{example}

\section{Medan listrik}

\begin{definition}[Medan elektrostatik]\label{def:b1:electrostatics-gauss:field}
\index{medan listrik}\index{garis medan}
Sebuah distribusi muatan yang diam menciptakan di setiap titik $M$ ruang
sebuah \emph{medan listrik} $\vect E(M)$, yang didefinisikan oleh gaya yang dikerahkannya pada
sebuah muatan uji $q$ yang diletakkan di $M$: $\vect F = q\vect E(M)$ (\unit{V/m} $=$
\unit{N/C}). Sebuah muatan titik $q_1$ di $O$ menciptakan
\[
\vect E(M) = \frac{q_1}{4\pi\varepsilon_0 r^2}\,\vect e_r , \qquad r = OM ,
\]
yang radial, mengarah keluar bagi $q_1 > 0$; sebuah distribusi menciptakan jumlah vektor
medan unsur-unsurnya. Yang disebut \emph{garis medan} adalah kurva yang menyinggung
$\vect E$ di tiap titiknya, berarah menyusurinya.
\end{definition}

\begin{proposition}[Membaca peta medan]\label{prop:b1:electrostatics-gauss:lines}
\omterm{def:b1:electrostatics-gauss:field}{Garis medan} meninggalkan muatan positif dan berakhir pada muatan negatif (atau di
tak hingga); ia tak pernah berpotongan (medannya bernilai tunggal), dan ia
merapat di tempat medannya kuat. Bila ada dua muatan,
banyaknya garis yang digambar dari masing-masing sebanding dengan muatannya.
\end{proposition}

\begin{example}[Cincin dan ruas garis dengan penjumlahan langsung]\label{ex:b1:electrostatics-gauss:ring}
Pada sumbu sebuah cincin berjari-jari $R$ yang mengemban $Q$ secara seragam, pada jarak
$z$ dari pusatnya, tiap unsur $\dd q$ menyumbang $\dd q/4\pi\varepsilon_0(R^2 +
z^2)$ menyusuri garis yang menghubungkannya dengan $M$; komponen yang tegak lurus
pada sumbunya saling meniadakan berpasangan dan komponen aksialnya berjumlah dengan faktor
$z/\sqrt{R^2 + z^2}$:
\[
E_z = \frac{Qz}{4\pi\varepsilon_0(R^2 + z^2)^{3/2}} ,
\]
yang nol di pusatnya, maksimal pada $z = R/\sqrt2$, dan $Q/4\pi\varepsilon_0z^2$ jauh
di kejauhan. Sebuah cakram bermuatan seragam adalah tumpukan cincin (\cref{pb:b1:electrostatics-gauss:1}).
Penjumlahan langsung berhasil bagi geometri yang mengizinkan integralnya;
sisa bab ini adalah tentang menghindarinya.
\end{example}

\section{Simetri dan invariansi}

\begin{proposition}[Asas simetri]\label{prop:b1:electrostatics-gauss:symmetry}
\index{simetri (sebuah medan)}\index{bidang simetri}\index{bidang antisimetri}
Medannya mempunyai simetri sumbernya. Bila sebuah bidang $\Pi$ merupakan
\emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{bidang simetri}} distribusi muatannya (bayangan cerminnya
sama dengan dirinya sendiri), maka di setiap titik $\Pi$ medannya
terletak \emph{di dalam} $\Pi$; bila $\Pi$ merupakan \emph{\omterm{prop:b1:electrostatics-gauss:symmetry}{bidang antisimetri}}
(bayangan cerminnya adalah distribusi itu dengan semua muatannya dibalik), maka
medan di titik-titik $\Pi$ \emph{tegak lurus} pada $\Pi$. Bila
distribusinya invarian terhadap sebuah translasi atau rotasi, medannya pun
demikian: komponennya tak bergantung pada koordinat yang bersangkutan.
\end{proposition}

\begin{proof}
$\vect E$ adalah jumlah suku $\dd q\,\vect e/4\pi\varepsilon_0r^2$; bayangan cermin
jumlahnya adalah jumlah bayangannya, dan bayangan medan
sebuah distribusi adalah medan distribusi bayangannya (sebuah vektor
yang dibangun dari posisi dan muatan bertransformasi seperti posisinya). Pada
\omterm{prop:b1:electrostatics-gauss:symmetry}{bidang simetri} medannya sama dengan bayangan cerminnya sendiri, sehingga terletak di dalam
bidangnya; pada \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang antisimetri} ia sama dengan lawan bayangannya,
sehingga tegak lurus padanya. Invariansinya adalah alasan yang sama dengan sebuah
translasi atau rotasi.
\end{proof}

\begin{method}[Menemukan bentuk sebuah medan sebelum menghitungnya]\label{meth:b1:electrostatics-gauss:symmetry}
Bagi titik $M$ tempat medannya dicari: (1) daftarlah \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang
simetri} sumbernya yang memuat $M$ --- medannya terletak pada
perpotongannya; dua bidang semacam itu menetapkan arahnya. (2) Daftarlah
invariansinya --- keduanya memberitahukan koordinat mana yang menjadi tempat bergantung $\vect E$. Maka sebuah
kawat tak hingga yang bermuatan seragam memberikan $\vect E = E(r)\,\vect e_r$ dalam
koordinat tabung, sebuah bola $\vect E = E(r)\,\vect e_r$ dalam \omterm{def:b1:point-kinematics:spherical}{koordinat
bola}, dan sebuah bidang tak hingga $\vect E = E(z)\,\vect e_z$ dengan $E(-z) = -E(z)$.
Barulah setelah itu hitung.
\end{method}

\section{Fluks dan hukum Gauss}

\begin{definition}[Fluks medannya]\label{def:b1:electrostatics-gauss:flux}
\index{fluks medan listrik}
Potonglah sebuah permukaan $S$ menjadi unsur-unsur kecil seluas $\dd S$, masing-masing dengan
normal satuan $\vect n$ (bagi permukaan tertutup, normal yang mengarah keluar). Yang disebut
\emph{fluks} $\vect E$ menembus $S$ adalah jumlah
\[
\Phi = \sum \vect E\cdot\vect n\,\dd S = \sum E_n\,\dd S \qquad (\unit{V.m}):
\]
yaitu komponen normal medannya yang diberi bobot luas --- ``banyaknya \omterm{def:b1:electrostatics-gauss:field}{garis
medan}'' yang menyeberangi $S$, dihitung positif ketika ia menyeberang menyusuri
$\vect n$.
\end{definition}

\begin{theorem}[Hukum Gauss]\label{thm:b1:electrostatics-gauss:gauss}
\index{hukum Gauss}
\omterm{def:b1:electrostatics-gauss:flux}{Fluks} medan elektrostatik menembus sebarang permukaan tertutup $S$
sama dengan muatan total yang dilingkupi $S$ dibagi $\varepsilon_0$:
\[
\Phi_{\text{keluar}} = \frac{Q_{\text{dalam}}}{\varepsilon_0} .
\]
Muatan di luar $S$ tak menyumbang apa pun pada fluksnya (garisnya masuk
lalu keluar lagi).
\end{theorem}

\begin{proof}
Bagi sebuah muatan titik $q$ di pusat sebuah bola berjari-jari $r$:
$\vect E$ bersifat radial, berbesar $q/4\pi\varepsilon_0r^2$ di mana-mana pada
bolanya, sehingga $\Phi = E \times 4\pi r^2 = q/\varepsilon_0$ --- tak bergantung pada $r$,
karena medannya jatuh sebagai $1/r^2$ tepat seperti luasnya tumbuh. Bahwa
hal yang sama berlaku bagi sebarang permukaan tertutup di sekitar $q$ (\omterm{def:b1:electrostatics-gauss:flux}{fluks} menembus sebuah
unsur permukaan hanya bergantung pada sudut ruang yang dibentangnya dari $q$),
bahwa muatan di luar memberikan nol (ia membentang tiap sudut ruang dua kali,
dengan tanda berlawanan), dan bahwa \omterm{def:b1:electrostatics-gauss:flux}{fluks} beberapa muatan berjumlah, diterima
tanpa bukti di sini; bukti lengkapnya ada dalam volume Tahun ke-2.
\end{proof}

\begin{method}[Memakai hukum Gauss]\label{meth:b1:electrostatics-gauss:gauss}
(1) Dari simetrinya, tuliskan bentuk $\vect E$ (arahnya dan
variabel tempatnya bergantung). (2) Pilihlah sebuah \emph{permukaan Gauss} tertutup
yang melalui titik $M$ dan yang di atasnya $\vect E$ di mana-mana entah tegak lurus dengan
besar yang sama $E(M)$, entah menyinggung: fluksnya lalu $E(M)$ dikali
luas bagian normalnya. (3) Hitunglah muatan di dalamnya. (4) Selesaikan untuk
$E(M)$. Cara ini berhasil bagi bola, tabung tak hingga dan bidang tak hingga
--- dan tak bagi yang lain tanpa perkakas tambahan.
\end{method}

\begin{proposition}[Ketiga medan klasiknya]\label{prop:b1:electrostatics-gauss:three}
\index{medan sebuah bola bermuatan}\index{medan sebuah kawat bermuatan}\index{medan sebuah bidang bermuatan}
\begin{enumerate}
  \item Bola berjari-jari $R$, bermuatan $Q$ yang tersebar seragam dalam
        volumenya: $E(r) = \dfrac{Q}{4\pi\varepsilon_0r^2}$ di luarnya ($r > R$) ---
        seolah seluruh muatannya duduk di pusatnya --- dan $E(r) =
        \dfrac{Qr}{4\pi\varepsilon_0R^3}$ di dalamnya, yang tumbuh linear dari nol;
        kontinu di $r = R$. Bila muatannya duduk di permukaannya saja,
        medan di dalamnya nol.
  \item Kawat tak hingga bermuatan linear $\lambda$: $E(r) =
        \dfrac{\lambda}{2\pi\varepsilon_0 r}$, radial.
  \item Bidang tak hingga bermuatan permukaan $\sigma$: $E = \dfrac{\sigma}{2\varepsilon_0}$
        di kedua sisinya, tegak lurus pada bidangnya, mengarah menjauhinya bagi
        $\sigma > 0$, dan \emph{tak bergantung pada jaraknya}. Menyeberangi bidangnya
        komponen normalnya melompat sebesar $\sigma/\varepsilon_0$.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Bola berjari-jari $r$: fluksnya $4\pi r^2E(r)$; muatan di dalamnya $Q$ bila
$r > R$, dan $Q(r/R)^3$ bila $r < R$. (2) Tabung berjari-jari $r$, sepanjang $h$:
\omterm{def:b1:electrostatics-gauss:flux}{fluks} menembus permukaan sisinya $2\pi rh\,E(r)$, tak ada yang menembus tutupnya
(medannya menyinggung); muatannya $\lambda h$. (3) Kotak pil berpenampang $S$
yang mengangkangi bidangnya: fluksnya $2SE$ menembus kedua tutupnya (medannya ganjil
dalam $z$, sehingga keduanya terhitung positif), tak ada yang menembus sisinya; muatannya
$\sigma S$.
\end{proof}

\begin{example}[Dua bidang: kapasitor keping sejajar]\label{ex:b1:electrostatics-gauss:twoplanes}
\index{kapasitor keping!medannya}
Dua bidang sejajar yang mengemban $+\sigma$ dan $-\sigma$: masing-masing menciptakan
$\sigma/2\varepsilon_0$ yang mengarah menjauhinya (menujunya bagi yang negatif).
Di antara kedua bidangnya kedua medannya berjumlah menjadi $\sigma/\varepsilon_0$, terarah dari
$+$ ke $-$; di luarnya keduanya saling meniadakan. Dengan $\sigma = \qty{1}{\micro C/m^2}$, $E =
10^{-6}/8.85 \times 10^{-12} = \qty{1.1e5}{V/m}$ di antaranya, dan nol di luarnya: medan
sebuah kapasitor bermuatan terkurung dalam celahnya
(\cref{ch:b1:potential-capacitors}).
\end{example}

\section{Kemiripan dengan gravitasi}

\begin{proposition}[Hukum Gauss bagi gravitasi]\label{prop:b1:electrostatics-gauss:gravity}
\index{hukum Gauss!bagi gravitasi}\index{medan gravitasi}
\omterm{thm:b1:newton-dynamics:laws}{Hukum Newton} $\vect F = -Gm_1m_2\,\vect e_{12}/r^2$ berbentuk seperti \omterm{thm:b1:electrostatics-gauss:coulomb}{hukum
Coulomb} dengan $q \to m$ dan $1/4\pi\varepsilon_0 \to -G$. \omterm{prop:b1:newton-dynamics:weight}{Medan
gravitasi} $\vect g$ (gaya tiap satuan massa) karenanya menuruti aturan simetri
yang sama dan
\[
\Phi_{\text{keluar}}(\vect g) = -4\pi G\,M_{\text{dalam}} .
\]
Khususnya sebuah benda yang setangkup bola menarik titik-titik di luarnya seolah
massanya berada di pusatnya, dan medan di dalam bola seragam
berjari-jari $R$ dan bermassa $M$ adalah $g(r) = GMr/R^3$, linear dalam $r$.
\end{proposition}

\begin{proof}
Gantilah $Q/\varepsilon_0$ dengan $-4\pi GM$ dalam bukti
\cref{prop:b1:electrostatics-gauss:three}; tanda minusnya mengatakan bahwa fluksnya
masuk (gravitasi menarik).
\end{proof}

\begin{example}[Di dalam Bumi]\label{ex:b1:electrostatics-gauss:earth}
Bagi Bumi yang seragam, $g$ akan turun linear dari \qty{9.8}{m/s^2} di
permukaannya sampai nol di pusatnya; sebuah batu yang dijatuhkan ke dalam terowongan menembus
planetnya akan berosilasi harmonik dengan periode $2\pi\sqrt{R/g} =
\qty{84}{min}$ --- yaitu periode sebuah satelit yang menyerempet. Bumi yang sebenarnya mempunyai
inti yang padat, dan $g$ sesungguhnya \emph{naik} sampai \qty{10.7}{m/s^2} di
batas inti--mantelnya sebelum turun (\cref{pb:b1:electrostatics-gauss:1}).
Newton memerlukan teorema kulit bola untuk menerapkan hukumnya pada planet; \omterm{thm:b1:electrostatics-gauss:gauss}{hukum
Gauss} memberikannya dalam satu baris.
\end{example}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:1}
Gaya listrik dan gaya gravitasi di antara dua proton yang terpisah
\qty{2.0}{fm}; berapa nisbahnya. Lalu mengapa inti atom dapat bertahan bersama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:2}
Medan pada jarak \qty{1.0}{m} dari muatan titik \qty{1.0}{\micro C}. Permukaan Bumi
mengemban medan \qty{100}{V/m} yang mengarah ke bawah: berapa tanda dan nilai
muatan Bumi, dengan memperlakukannya sebagai bola berjari-jari \qty{6370}{km}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:3}
Dua muatan $+q$ di $x = \pm a$ pada sumbu $x$. Berapa medannya pada sumbu $x$
untuk $|x| > a$ dan pada sumbu $y$; tunjukkan bahwa pada sumbu $y$ nilai $E_y$
maksimal pada $y = a/\sqrt2$. Sketsalah garisnya di dekat titik asalnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:electrostatics-gauss:4}
Muatan $+2q$ di $x = 0$ dan $-q$ di $x = a$. Di titik mana pada sumbu $x$
medannya lenyap? Berapa banyak \omterm{def:b1:electrostatics-gauss:field}{garis medan} yang meninggalkan $+2q$ bagi tiap satu garis
yang tiba di $-q$, dan ke mana yang lain pergi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:5}
Sebuah cakram berjari-jari $R$ mengemban $\sigma$ secara seragam. Dari hasil cincinnya,
tunjukkan bahwa pada sumbunya $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 +
R^2}}\Bigr)$ untuk $z > 0$; periksalah kedua limitnya $z \ll R$ dan $z \gg R$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:6}
Sebuah inti emas ($Z = 79$, $R = \qty{7.0}{fm}$) sebagai bola bermuatan
seragam: berapa medan di permukaannya, pada $R/2$, dan pada \qty{1}{pm}. Bandingkan dengan
medan dadal udara, \qty{3e6}{V/m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:7}
Medan sebuah kawat tak hingga dengan \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss}. Sebuah penghantar bertegangan tinggi
berjari-jari \qty{1.0}{cm}: pada muatan linear berapa medan di permukaannya
mencapai nilai dadalnya \qty{3e6}{V/m} (awal korona). Lalu berapa medannya pada
\qty{5}{m}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:8}
Dua keping sejajar seluas \qty{1.0}{m^2} mengemban $\pm\qty{1.0}{\micro C}$.
Berapa medan di antaranya; berapa gaya pada satu keping (medan yang bekerja padanya adalah medan
keping yang lain saja --- mengapa?); berapa tekanan pada kepingnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:electrostatics-gauss:9}
Kabel koaksial: penghantar dalam berjari-jari $a$ dengan $+\lambda$, dan selubung
luar berjari-jari $b$ dengan $-\lambda$. Berapa medan di ketiga daerahnya menurut
\omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss}. Dengan $a = \qty{0.5}{mm}$, $b = \qty{2.5}{mm}$, $\lambda = \qty{10}{nC/m}$:
berapa medan di permukaan penghantar dalamnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:10}
Sebuah kulit bola tebal $R_1 < r < R_2$ mengemban $Q$ secara seragam dalam
volumenya. Berapa medan di ketiga daerahnya; periksalah kekontinuannya di $R_1$ dan
$R_2$; di mana ia maksimal? Sketsalah $E(r)$. Sebuah muatan positif yang diletakkan
di dalam rongganya: gaya apa yang dirasakannya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:11}
\emph{Gravitasi di dalam Bumi.} Inti: berjari-jari \qty{3480}{km}, bermassa jenis
\qty{11000}{kg/m^3}; mantel sampai \qty{6370}{km}, bermassa jenis \qty{4500}{kg/m^3}.
(a) Berapa massa intinya, mantelnya, dan totalnya (bandingkan dengan \qty{5.97e24}{kg}).
(b) Berapa $g$ di permukaannya dan di batas inti--mantelnya. (c) Tunjukkan bahwa
$g$ naik terhadap kedalaman di dalam mantelnya bila massa jenis mantelnya di bawah
$\tfrac23$ massa jenis rata-rata yang ada di bawahnya, lalu periksalah.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:electrostatics-gauss:12}
\emph{Atom Thomson.} Modelkan atom hidrogen sebagai bola positif
seragam bermuatan $+e$ dan berjari-jari $R = \qty{0.10}{nm}$ dengan elektronnya
bebas di dalamnya. (a) Berapa gaya pada elektronnya pada jarak $r$ dari pusatnya.
(b) Tunjukkan bahwa ia berosilasi harmonik lalu hitunglah frekuensinya dan
panjang gelombang cahaya yang akan dipancarkannya. (c) Bandingkan dengan
garis ultraungu hidrogen (\qty{122}{nm} dan yang lebih pendek) lalu berilah komentar tentang
apa yang benar dan apa yang keliru pada modelnya (\cref{ch:b1:quantum-introduction}).
\end{exercise}

\section{Soal: Bumi yang bermuatan listrik}

\begin{problem}[{Soal akhir pekan --- medan cuaca cerah, awan
guruh, sambaran petir, dan hukum Gauss yang diarahkan pada gravitasi}]
\label{pb:b1:electrostatics-gauss:1}
Data: $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $1/4\pi\varepsilon_0 = \qty{8.99e9}{SI}$,
$e = \qty{1.6e-19}{C}$, jari-jari Bumi $R_E = \qty{6.37e6}{m}$, permukaannya
$\qty{5.1e14}{m^2}$, $G = \qty{6.67e-11}{SI}$, $M_E = \qty{5.97e24}{kg}$.

\medskip
\noindent\textbf{Bagian I --- Medan cuaca cerah.}
Pada cuaca cerah medan di permukaan tanah adalah $E_0 = \qty{100}{V/m}$,
yang mengarah \emph{ke bawah}.
\begin{enumerate}
  \item Ke mana arah gaya pada sebuah elektron bebas di permukaan tanah; berapa tanda
        muatan Bumi.
  \item Perlakukan Bumi sebagai bola bermuatan seragam: dari \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss},
        berapa muatan totalnya.
  \item Berapa \omterm{def:b1:electrostatics-gauss:charge}{rapat muatan} permukaan tanahnya; berapa banyak kelebihan elektron
        tiap sentimeter persegi.
  \item Berapa gaya pada sebutir debu bermassa \qty{1}{\micro g} yang mengemban $100$
        \omterm{def:b1:electrostatics-gauss:charge}{muatan elementer}; bandingkan dengan beratnya.
  \item Pengukuran memperlihatkan medannya turun sampai kira-kira \qty{5}{V/m} pada
        \qty{10}{km}. Terapkan \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss} pada sebuah kolom tegak berpenampang
        \qty{1}{m^2} di antara tanah dan \qty{10}{km}: berapa tanda
        dan besar muatan yang dimuat udaranya tiap meter persegi.
  \item Berapa \omterm{def:b1:electrostatics-gauss:charge}{rapat muatan} volume rata-rata udara itu, dan berapa
        banyak \omterm{def:b1:electrostatics-gauss:charge}{muatan elementer} yang bersesuaian tiap meter kubik. (Udaranya memuat
        kira-kira $10^9$ ion tiap tanda tiap meter kubik: berapa pecahan
        ketakseimbangan yang ini?)
  \item Udaranya mempunyai konduktivitas kecil $\gamma = \qty{2e-14}{S/m}$, sehingga
        rapat arus $\gamma E_0$ mengalir turun. Berapa arus cuaca cerah
        total di seluruh dunia; berapa lama waktu yang diperlukan untuk menetralkan
        muatan Bumi. Simpulkan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Awan guruhnya.}
Modelkan sebuah awan badai sebagai dua cakram mendatar berjari-jari $R = \qty{3}{km}$:
$-Q$ pada ketinggian \qty{3}{km}, $+Q$ pada \qty{9}{km}, dengan $Q = \qty{40}{C}$.
\begin{enumerate}[resume]
  \item Berapa medan pada sumbu sebuah cincin berjari-jari $a$ yang mengemban $q$, pada
        jarak $z$ dari pusatnya (simetrinya dulu, lalu penjumlahannya).
  \item Simpulkan, dengan menjumlahkan atas cincin-cincinnya, medan pada sumbu sebuah cakram
        berjari-jari $R$ dan bermuatan permukaan $\sigma$:
        $E_z = \dfrac{\sigma}{2\varepsilon_0}\Bigl(1 - \dfrac{z}{\sqrt{z^2 + R^2}}\Bigr)$.
  \item Berapa limitnya untuk $z \ll R$ dan $z \gg R$; tafsirkan masing-masing.
  \item Berapa muatan permukaan cakram bawahnya; berapa medan yang diciptakannya di tanah
        tepat di bawahnya (arah dan nilainya).
  \item Berapa medan cakram atasnya di titik yang sama; berapa medan totalnya di
        tanah di bawah badainya. Tanahnya penghantar dan muatan
        imbasnya kira-kira melipatduakan hasilnya (diterima tanpa bukti): bandingkan dengan
        \qty{10}{kV/m} yang terukur di bawah badai dan dengan nilai cuaca
        cerahnya.
  \item Berapa \omterm{def:b1:electrostatics-gauss:flux}{fluks} medannya menembus permukaan tertutup yang melingkupi seluruh
        awannya; dan menembus permukaan yang melingkupi cakram bawahnya saja.
  \item Berapa medan di tengah-tengah kedua cakramnya (\qty{6}{km}), dan ke mana
        arahnya; bandingkan dengan medan dadal udara pada ketinggian
        itu, kira-kira \qty{1.5e6}{V/m}.
  \item Seandainya \qty{40}{C} itu dipusatkan dalam sebuah bola berjari-jari
        \qty{500}{m}: berapa medan di permukaannya. Mengapa pemicuan petir
        masih menjadi pertanyaan terbuka?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Sambarannya.}
Sebuah sambaran balik membawa \qty{5}{C} ke tanah dalam \qty{50}{\micro s}
melalui saluran sepanjang \qty{3}{km}; ambillah medan menyusuri salurannya
sebesar $10^5$~V/m.
\begin{enumerate}[resume]
  \item Berapa arus rata-ratanya.
  \item Berapa usaha gaya listrik pada muatan yang dipindahkan; berapa dayanya
        selama sambarannya; bandingkan dengan pemakaian \omterm{thm:b1:sinusoidal-impedance:power}{daya rata-rata}
        umat manusia, kira-kira \qty{2e13}{W}.
  \item Salurannya berjari-jari \qty{2}{cm}: berapa massa udara di dalamnya
        ($\rho = \qty{1.2}{kg/m^3}$), dan berapa suhu yang akan dicapainya bila
        seluruh energinya memanaskannya pada volume tetap ($c_V =
        \qty{720}{J/(kg.K)}$). Suhu saluran yang sebenarnya dekat
        \qty{30000}{K}: ke mana sisanya pergi?
  \item Guruh: berapa selangnya tiap kilometer, dan mengapa bunyi
        saluran \qty{3}{km} bergemuruh selama beberapa sekon.
  \item Di seluruh dunia kira-kira $50$ sambaran awan-ke-tanah terjadi tiap sekon:
        berapa arus yang dibawanya ke bawah; bandingkan dengan Bagian~I lalu lengkapilah
        gambaran ``rangkaian global''-nya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- \omterm{thm:b1:electrostatics-gauss:gauss}{Hukum Gauss} yang diarahkan pada gravitasi.}
\begin{enumerate}[resume]
  \item Tuliskan \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss} bagi \omterm{prop:b1:newton-dynamics:weight}{medan gravitasi} $\vect g$ lalu
        benarkan tetapannya dengan kasus sebuah massa titik.
  \item Bumi yang seragam: berapa $g(r)$ di dalamnya; berapa nilainya di $r = R_E/2$.
  \item Bumi yang sebenarnya, dengan inti berjari-jari \qty{3480}{km} dan bermassa jenis
        \qty{11000}{kg/m^3}: berapa $g$ di batas inti--mantelnya; bandingkan dengan
        permukaannya.
  \item Sebuah batu yang dijatuhkan ke dalam terowongan tanpa gesekan menembus Bumi
        yang seragam: bagaimana persamaan geraknya, berapa periode osilasinya, dan berapa kelajuannya di
        pusatnya.
  \item Ringkaslah: pada masing-masing keempat bagiannya, apa yang digantikan \omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss},
        dan empat angka mana yang layak diingat.
\end{enumerate}
\end{problem}
