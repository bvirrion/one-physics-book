\chapter{Sistem Titik; Pengantar Benda Tegar}\label{ch:b1:systems-of-points}

Seorang peloncat indah meninggalkan papannya, melipat badan, bersalto dua
kali, lalu meluruskan diri; sepanjang semua itu satu titik tubuhnya ---
\omterm{def:b1:systems-of-points:cm}{pusat massanya} --- melukiskan parabola sederhana yang sama seperti sebuah
batu. Sebuah yoyo yang dijatuhkan dari tangan memerlukan waktu tiga kali
lebih lama untuk mencapai ujung talinya daripada sebuah batu, dan tiba
sambil berputar dua ribu kali tiap menit. Bola pejal mengalahkan bola
berongga menuruni bidang miring mana pun, berapa pun ukuran atau massanya.
Mekanika satu titik, yang dibawa melalui bab-bab sebelumnya, meluas ke
sistem banyak titik --- dan ke \omterm{def:b1:systems-of-points:solid}{benda tegar} sehari-hari --- melalui
segenggam teorema: \omterm{def:b1:systems-of-points:cm}{pusat massanya} bergerak seperti sebuah titik, energi
dan \omterm{def:b1:angular-momentum:L}{momentum sudutnya} terbelah menjadi bagian \omterm{def:b1:systems-of-points:cm}{pusat massa} dan bagian ``di
sekitarnya'', dan benda yang berputar terhadap sumbu tetap adalah sistem
berderajat kebebasan satu yang diatur \omterm{prop:b1:angular-momentum:inertia}{momen inersianya}.

\section{Pusat massa dan momentum}

\begin{definition}[Pusat massa; momentum sebuah sistem]\label{def:b1:systems-of-points:cm}
\index{pusat massa}\index{barisentrum}
Untuk partikel bermassa $m_i$ di $M_i$, dengan massa total $M = \sum m_i$,
\emph{pusat massa} (barisentrum) $G$ didefinisikan oleh
\[
\sum_i m_i\,\vect{GM_i} = \vect 0, \qquad\text{yakni}\qquad
\vect{OG} = \frac{1}{M}\sum_i m_i\,\vect{OM_i}
\]
untuk titik asal $O$ mana pun. \emph{Momentum} sistemnya adalah $\vect P =
\sum_i m_i\vect v_i = M\vect v_G$.
\end{definition}

\begin{theorem}[Teorema pusat massa]\label{thm:b1:systems-of-points:cmtheorem}
\index{teorema pusat massa}\index{kekekalan momentum}
Dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial},
\[
M\,\vect a_G = \frac{\dd\vect P}{\dd t} = \sum\vect F_{\mathrm{ext}} :
\]
\omterm{def:b1:systems-of-points:cm}{pusat massanya} bergerak seperti \omterm{def:b1:point-kinematics:frame}{partikel titik} bermassa $M$ yang dikenai
resultan gaya \emph{luar} saja. Bagi \omterm{def:b1:newton-dynamics:conservation}{sistem tertutup} tanpa
gaya luar, $\vect P$ kekal dan $G$ bergerak beraturan.
\end{theorem}

\begin{proof}
Jumlahkan hukum kedua Newton atas partikel-partikelnya; gaya dalamnya
saling meniadakan berpasangan aksi--reaksi; $\sum m_i\vect a_i = M\vect a_G$ dengan
menurunkan definisi $G$ dua kali.
\end{proof}

\begin{example}[Peloncat indah, sentakan mundur, tumbukan]\label{ex:b1:systems-of-points:cm}
\index{tumbukan}
Titik $G$ peloncatnya menuruti parabola di bawah beratnya saja, bagaimanapun ia
berputar --- ototnya adalah gaya dalam. Sebuah senapan \qty{4.0}{kg}
yang menembakkan peluru \qty{10}{g} pada \qty{800}{m/s}: $\vect P = \vect 0$ sebelum
dan sesudahnya, sehingga senapannya tersentak mundur pada \qty{2.0}{m/s}. Dua mobil, \qty{1000}{kg}
pada \qty{20}{m/s} dan \qty{1500}{kg} yang diam, yang saling mengunci: $v =
20000/2500 = \qty{8.0}{m/s}$ --- momentumnya kekal, sementara $60\%$
\omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} berubah menjadi logam remuk dan kalor. Momentum kekal
dalam setiap tumbukan; \omterm{thm:b1:work-and-energy:ke}{energi kinetik} hanya dalam tumbukan \emph{lenting}
(\cref{exo:b1:systems-of-points:12}).
\end{example}

\section{Teorema Koenig}

\begin{definition}[Kerangka barisentrum]\label{def:b1:systems-of-points:barycentric}
\index{kerangka barisentrum}
Yang disebut \emph{kerangka \omterm{def:b1:systems-of-points:cm}{barisentrum}} $\mathcal R^*$ bertitik asal di $G$ dan
bertranslasi terhadap \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial} $\mathcal R$ (sumbu-sumbunya
menjaga arahnya tetap). Besaran yang diukur di dalamnya diberi bintang: $\vect v_i^*
= \vect v_i - \vect v_G$, dan $\sum m_i\vect v_i^* = \vect 0$.
\end{definition}

\begin{theorem}[Teorema Koenig]\label{thm:b1:systems-of-points:koenig}
\index{teorema Koenig}
\omterm{def:b1:angular-momentum:L}{Momentum sudut} terhadap sebuah titik $O$ dan \omterm{thm:b1:work-and-energy:ke}{energi kinetik} sebuah
sistem terbelah menjadi sumbangan \omterm{def:b1:systems-of-points:cm}{pusat massanya}, yang mengemban
seluruh massa, dan sumbangan dalam \omterm{def:b1:systems-of-points:barycentric}{kerangka barisentrumnya}:
\[
\vect L_O = \vect{OG}\wedge M\vect v_G + \vect L^*, \qquad
E_k = \tfrac12 Mv_G^2 + E_k^*,
\]
dengan $\vect L^* = \sum\vect{GM_i}\wedge m_i\vect v_i^*$ dan $E_k^* = \sum\tfrac12
m_iv_i^{*2}$ yang tak bergantung pada titik asal kerangkanya.
\end{theorem}

\begin{proof}
$\vect{OM_i} = \vect{OG} + \vect{GM_i}$ dan $\vect v_i = \vect v_G + \vect v_i^*$;
jabarkan $\sum\vect{OM_i}\wedge m_i\vect v_i$ dan $\sum\tfrac12 m_iv_i^2$: suku
silangnya memuat $\sum m_i\vect{GM_i} = \vect 0$ atau $\sum m_i\vect v_i^* = \vect 0$
dan lenyap.
\end{proof}

\begin{example}[Sebuah roda yang menggelinding]\label{ex:b1:systems-of-points:rolling}
\index{menggelinding tanpa tergelincir}
Sebuah roda bermassa $M$, berjari-jari $R$, bermomen inersia $J$ terhadap gandarnya,
yang \omterm{ex:b1:systems-of-points:rolling}{menggelinding tanpa tergelincir} dengan laju $v$: titik singgungnya diam, sehingga
$\omega = v/R$, dan $E_k = \tfrac12 Mv^2 + \tfrac12 J\omega^2 = \tfrac12 Mv^2(1 +
J/MR^2)$ --- tiga perempat $Mv^2$ bagi cakram seragam ($J = \tfrac12 MR^2$),
$Mv^2$ bagi gelang. Suku keduanya adalah alasan roda sepeda dibuat
ringan: rotasinya memakan energi yang tak dimakan massa rangkanya.
\end{example}

\section{Masalah dua benda}

\begin{proposition}[Massa tereduksi]\label{prop:b1:systems-of-points:twobody}
\index{masalah dua benda}\index{massa tereduksi}
Dua partikel yang hanya berinteraksi satu sama lain (gaya $\vect F$ pada $m_2$
dari $m_1$, $-\vect F$ pada $m_1$): $G$ bergerak beraturan, dan posisi
relatifnya $\vect r = \vect{M_1M_2}$ menuruti
\[
\mu\,\frac{\dd^2\vect r}{\dd t^2} = \vect F, \qquad \mu = \frac{m_1m_2}{m_1 + m_2},
\]
yaitu persamaan sebuah partikel khayal tunggal dengan \emph{\omterm{prop:b1:systems-of-points:twobody}{massa tereduksi}} $\mu$
yang bergerak di bawah $\vect F$ terhadap sebuah pusat tetap. Dalam $\mathcal R^*$, $E_k^* =
\tfrac12\mu\dot{\vect r}^2$ dan $\vect L^* = \vect r\wedge\mu\dot{\vect r}$.
\end{proposition}

\begin{proof}
$\ddot{\vect r} = \vect a_2 - \vect a_1 = \vect F/m_2 + \vect F/m_1 = \vect F(m_1 +
m_2)/m_1m_2$. Dalam $\mathcal R^*$, $\vect{GM_2} = m_1\vect r/(m_1 + m_2)$ dan
$\vect{GM_1} = -m_2\vect r/(m_1 + m_2)$; substitusikan ke dalam $E_k^*$ dan $\vect L^*$.
\end{proof}

\begin{example}[Ketika pusatnya ikut bergerak]\label{ex:b1:systems-of-points:reduced}
Dalam \cref{ch:b1:central-forces} Matahari dianggap tetap: tepat hanya dalam
limit $m \ll M$; secara umum \omterm{thm:b1:central-forces:circular}{hukum ketiga Kepler} berbunyi $T^2 = 4\pi^2a^3/G(M +
m)$. Bagi molekul HCl yang bergetar pada \cref{pb:b1:work-and-energy:1},
massa yang berosilasi di dalam sumurnya adalah $\mu = m_Hm_{Cl}/(m_H + m_{Cl}) =
0.972\,m_H$, yang menaikkan frekuensinya sebesar $1.4\%$; bagi atom
hidrogen, $\mu$ elektronnya berbeda dari $m_e$ sebesar satu bagian dalam $1836$ ---
terukur dalam spektrumnya.
\end{example}

\section{Benda tegar yang berputar terhadap sumbu tetap}

\begin{definition}[Benda tegar; rotasi terhadap sumbu tetap]\label{def:b1:systems-of-points:solid}
\index{benda tegar}\index{rotasi terhadap sumbu tetap}
Yang disebut \emph{benda tegar} (benda pejal) adalah sistem yang titik-titiknya
menjaga jaraknya satu sama lain tetap. Bila ia berputar terhadap sumbu $\Delta$ yang tetap dalam kerangkanya, setiap
titiknya melukiskan lingkaran berpusat pada $\Delta$ dengan \omterm{cor:b1:point-kinematics:circular}{kecepatan sudut} yang sama
$\omega = \dot\theta$; titik yang berjarak $r_i$ dari sumbunya berkelajuan
$r_i\omega$.
\end{definition}

\begin{proposition}[Momen inersia; momentum sudut; energi kinetik]\label{prop:b1:systems-of-points:inertia}
\index{momen inersia!benda tegar}\index{teorema Huygens}
Dengan \emph{\omterm{prop:b1:angular-momentum:inertia}{momen inersia}} $J_\Delta = \sum m_ir_i^2$ (sebuah integral
$\int r^2\,\dd m$ bagi benda kontinu):
\[
L_\Delta = J_\Delta\,\omega, \qquad E_k = \tfrac12 J_\Delta\omega^2 .
\]
Nilai baku (massa $M$): cincin tipis atau tabung berongga terhadap sumbunya,
$MR^2$; cakram seragam atau tabung pejal, $\tfrac12 MR^2$; batang tipis
sepanjang $L$ terhadap pusatnya, $\tfrac{1}{12}ML^2$, terhadap salah satu ujungnya,
$\tfrac13 ML^2$; bola pejal terhadap sebuah garis tengahnya, $\tfrac25 MR^2$.
\emph{Teorema Huygens}: terhadap sumbu $\Delta$ yang sejajar dengan sumbu
$\Delta_G$ melalui $G$ pada jarak $d$, $J_\Delta = J_{\Delta_G} + Md^2$.
\end{proposition}

\begin{proof}
$L_\Delta$ dan $E_k$: \cref{prop:b1:angular-momentum:inertia}. Batang terhadap
ujungnya: $\int_0^L x^2(M/L)\,\dd x = ML^2/3$; terhadap pusatnya, $\int_{-L/2}^{L/2}
x^2(M/L)\dd x = ML^2/12$. Cakram: cincin berjari-jari $r$, berlebar $\dd r$, bermassa
$(2Mr/R^2)\dd r$: $\int_0^R r^2(2Mr/R^2)\dd r = MR^2/2$; cincin dan
bolanya diterima tanpa bukti (bolanya menuntut integral rangkap). Huygens: dengan
$r_i^2 = \abs{\vect{GM_i}_\perp + \vect d}^2 = r_{G,i}^2 + d^2 + 2\vect d\cdot
\vect{GM_i}_\perp$, suku silangnya berjumlah nol menurut definisi $G$.
\end{proof}

\begin{theorem}[Dinamika benda tegar terhadap sumbu tetap]\label{thm:b1:systems-of-points:rotation}
\index{poros}\index{daya sebuah momen}
Bagi \omterm{def:b1:systems-of-points:solid}{benda tegar} yang berputar terhadap sumbu tetap $\Delta$ dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial},
\[
J_\Delta\,\ddot\theta = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}}),
\qquad
\frac{\dd}{\dd t}\Big(\tfrac12 J_\Delta\dot\theta^2\Big) = \sum\mathcal M_\Delta(\vect F_{\mathrm{ext}})\,\dot\theta :
\]
momen gaya-gaya luarnya terhadap sumbunya menggerakkan percepatan
sudutnya, dan sebuah momen $\mathcal M_\Delta$ menyampaikan daya
$\mathcal M_\Delta\omega$. Sebuah \emph{poros} ideal (bantalan tanpa gesekan) tak
mengerahkan momen terhadap sumbunya sehingga tak mempercepat maupun mengerem
rotasinya.
\end{theorem}

\begin{proof}
\cref{thm:b1:angular-momentum:system} yang diproyeksikan pada $\Delta$, dengan $L_\Delta
= J_\Delta\dot\theta$ dan $J_\Delta$ tetap bagi \omterm{def:b1:systems-of-points:solid}{benda tegar}; kalikan dengan
$\dot\theta$ untuk bentuk energinya (gaya dalam sebuah \omterm{def:b1:systems-of-points:solid}{benda tegar} tak
melakukan usaha: jaraknya tak berubah).
\end{proof}

\begin{example}[Bandul fisis; bandul torsi]\label{ex:b1:systems-of-points:pendulums}
\index{bandul fisis}\index{bandul torsi}
Sebuah benda bermassa $M$ yang berporos pada sumbu mendatar berjarak $d$
dari $G$: momen beratnya $-Mgd\sin\theta$, sehingga $J_\Delta\ddot\theta =
-Mgd\sin\theta$ dan ayunan kecilnya berperiode
\[
T = 2\pi\sqrt{\frac{J_\Delta}{Mgd}} ,
\]
yaitu periode \omterm{ex:b1:newton-dynamics:pendulum}{bandul sederhana} sepanjang $J_\Delta/Md$ --- bagi batang seragam
yang berporos di salah satu ujungnya, $2L/3$. Sebuah cakram yang digantung pada kawat bertetapan \omterm{def:b1:angular-momentum:moment}{torsi}
$C$ (momen pemulih $-C\theta$): $J\ddot\theta = -C\theta$, $T = 2\pi\sqrt{J/C}$
--- yaitu neraca \omterm{def:b1:angular-momentum:moment}{torsi} yang mengukur $G$ dan hukum Coulomb.
\end{example}

\begin{proposition}[Menggelinding tanpa tergelincir menuruni bidang miring]\label{prop:b1:systems-of-points:rollingincline}
Sebuah benda putar ($J = kMR^2$ terhadap sumbunya) yang dilepaskan pada
bidang miring bersudut $\alpha$ \omterm{ex:b1:systems-of-points:rolling}{menggelinding tanpa tergelincir} dengan percepatan
\[
a = \frac{g\sin\alpha}{1 + k} :
\]
$\tfrac57 g\sin\alpha$ bagi bola pejal, $\tfrac23 g\sin\alpha$ bagi cakram,
$\tfrac12 g\sin\alpha$ bagi gelang --- tak bergantung pada massa dan jari-jarinya.
\end{proposition}

\begin{proof}
Energi: $Mgh = \tfrac12 Mv^2(1 + k)$ menyusuri lerengnya (gesekan statis
di titik singgungnya yang diam tak melakukan usaha), sehingga $v^2 = 2ah$ dengan $a$ seperti
yang dinyatakan; atau Newton untuk $G$ dan teorema momen terhadap $G$ dengan
gesekan $f$: $Ma = Mg\sin\alpha - f$, $kMR^2\dot\omega = fR$, $a = R\dot\omega$.
\end{proof}

\begin{remark}[Sistem yang dapat berubah bentuk]\label{rem:b1:systems-of-points:deformable}
\index{sistem yang dapat berubah bentuk}
Bagi sistem yang tidak tegar, teorema \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} berbunyi
$\Delta E_k = W_{\mathrm{ext}} + W_{\mathrm{int}}$: gaya dalamnya melakukan
usaha (otot, pegas, gesekan di dalam sistemnya). Peseluncur yang menarik
lengannya masuk, peloncat indah yang melipat badan, pesepeda yang berdiri di atas pedalnya
semuanya memperoleh \omterm{thm:b1:work-and-energy:ke}{energi kinetik} tanpa usaha luar --- teorema \omterm{def:b1:systems-of-points:cm}{pusat
massanya} tetap benar, energinya datang dari dalam.
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:1}
Massa \qty{2.0}{kg} di $x = 0$ dan \qty{3.0}{kg} di $x = \qty{1.0}{m}$:
posisi $G$. Sistem Bumi--Bulan ($81:1$, \qty{384000}{km}):
jarak $G$ dari pusat Bumi; bandingkan dengan jari-jari
Bumi.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:2}
Sebuah senapan \qty{4.0}{kg} menembakkan peluru \qty{10}{g} pada \qty{800}{m/s}.
Kelajuan sentakan mundurnya; \omterm{thm:b1:work-and-energy:ke}{energi kinetik} peluru dan senapannya; dari mana
energinya datang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:3}
Sebuah mobil \qty{1000}{kg} pada \qty{20}{m/s} menabrak mobil \qty{1500}{kg} yang diam;
keduanya saling mengunci. Kelajuan bersamanya; pecahan \omterm{thm:b1:work-and-energy:ke}{energi kinetik} yang hilang.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:systems-of-points:4}
Turunkan \omterm{prop:b1:angular-momentum:inertia}{momen inersia} sebuah batang seragam bermassa $M$, sepanjang $L$,
terhadap sumbu tegak lurus yang melalui ujungnya, lalu yang melalui pusatnya;
periksalah teorema Huygens di antara keduanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:systems-of-points:5}
\omterm{thm:b1:work-and-energy:ke}{Energi kinetik} sebuah tabung seragam yang menggelinding dan sebuah gelang yang menggelinding
dengan massa dan laju yang sama, sebagai kelipatan $\tfrac12 Mv^2$. Mana yang berhenti
lebih dulu di lantai datar yang kasar, dan mana yang memanjat lebih tinggi di sebuah lereng?
\end{exercise}

\section{Soal: Yoyo}

\begin{problem}[{Soal akhir pekan --- sebuah cakram pada poros tipis di ujung
seutas tali: mengapa ia jatuh begitu lambat, seberapa cepat ia berputar di
bawah, mengapa ia ``tidur'' di sana, dan ke arah mana sebuah gelendong
menggelinding ketika benangnya ditarik}]
\label{pb:b1:systems-of-points:1}
Sebuah yoyo adalah cakram seragam bermassa $M = \qty{50}{g}$ dan berjari-jari $R =
\qty{3.0}{cm}$ dengan poros tipis berjari-jari $r = \qty{0.40}{cm}$ yang padanya
digulung tali sepanjang $h = \qty{1.0}{m}$; ujung atas talinya
dipegang tetap. $g = \qty{9.81}{m/s^2}$; $J = \tfrac12 MR^2$ terhadap
sumbunya.

\medskip
\noindent\textbf{Bagian I --- Kinematika dan energi.}
\begin{enumerate}
  \item Di mana \omterm{def:b1:systems-of-points:cm}{pusat massa} $G$ yoyonya? Mengapa?
  \item Tunjukkan dari teorema Koenig bahwa \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} adalah $\tfrac12
        Mv_G^2 + \tfrac12 J\omega^2$.
  \item Turunkan $J = \tfrac12 MR^2$ dengan mengintegralkan cincin demi cincin, lalu hitunglah
        $J$.
  \item Ketika talinya terurai tanpa tergelincir dari porosnya, hubungkan
        $v_G$ dengan $\omega$.
  \item Simpulkan $E_k = \tfrac12 Mv_G^2(1 + R^2/2r^2)$ lalu hitunglah
        faktor dalam kurungnya. Berilah komentar.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Jatuhnya.}
\begin{enumerate}[resume]
  \item Gaya-gaya pada yoyonya; tuliskan teorema \omterm{def:b1:systems-of-points:cm}{pusat massanya}
        (sumbu tegak ke bawah).
  \item Tuliskan teorema \omterm{def:b1:angular-momentum:L}{momentum sudut} terhadap sumbu yang melalui $G$
        (gaya mana yang bermomen?).
  \item Gabungkan dengan pertanyaan 4 untuk menemukan percepatannya $a_G = g/(1 +
        R^2/2r^2)$; hitunglah.
  \item Hitunglah tegangan talinya lalu bandingkan dengan beratnya.
  \item Waktu untuk mengurai \qty{1.0}{m} talinya, dan kelajuannya ketika itu;
        bandingkan dengan sebuah batu yang dijatuhkan dari ketinggian yang sama.
  \item \omterm{cor:b1:point-kinematics:circular}{Kecepatan sudut} di bawah, dalam rad/s dan putaran tiap menit.
  \item Belahlah \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya} di bawah menjadi bagian translasi
        dan bagian rotasinya.
  \item Periksalah neraca energinya: bandingkan $Mgh$ dengan \omterm{thm:b1:work-and-energy:ke}{energi kinetiknya}
        di bawah, lalu katakan mengapa tegangan talinya, walaupun besar,
        tak melakukan usaha.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Di bawah: si tidur.}
Ketika talinya terurai seluruhnya, ia menghentikan \omterm{def:b1:systems-of-points:cm}{pusat massanya} dalam beberapa
milisekon; yoyonya terus berputar di dalam simpul di ujung
talinya (``tidur'').
\begin{enumerate}[resume]
  \item Mengapa putarannya bertahan setelah $G$ berhenti? (Teorema mana, dan
        gaya mana yang akan diperlukan untuk menghentikannya?)
  \item Taksirlah tegangan rata-rata selama perhentiannya bila $G$ dihentikan
        dalam \qty{5}{ms}.
  \item Simpulnya bergesek pada porosnya dengan momen gesek $\Gamma =
        \qty{1.0e-4}{N.m}$: berapa lama yoyonya tidur?
  \item Sentakan tajam ke atas pada talinya membuat talinya mencengkeram porosnya
        lagi; yoyo yang berputar itu lalu memanjat. Dengan memakai energi, sampai
        ketinggian berapa ia dapat naik bila tak ada yang hilang? Mengapa, dalam
        praktiknya, kurang dari itu?
  \item Selama pemanjatannya, apakah tegangannya lebih besar atau lebih kecil daripada
        beratnya? (Tanda $a_G$.)
  \item Apa yang akan terjadi pada yoyo yang jari-jari porosnya sama dengan $R$
        (talinya digulung pada tepinya)? Hitunglah percepatan jatuhnya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Gelendong di atas meja.}
Benda yang sama terletak di atas meja yang kasar, porosnya mendatar, dan talinya
meninggalkan \emph{bagian bawah} porosnya secara mendatar; talinya ditarik dengan
gaya $F$. Ia \omterm{ex:b1:systems-of-points:rolling}{menggelinding tanpa tergelincir}.
\begin{enumerate}[resume]
  \item Di titik singgung $C$ dengan mejanya, berapa kecepatan
        bahan gelendongnya? Simpulkan bahwa geraknya, pada tiap
        saat, adalah rotasi terhadap $C$.
  \item Hitunglah momen $F$ terhadap $C$ (\omterm{def:b1:angular-momentum:moment}{lengan gaya} $R - r$) dan momen
        beratnya serta momen reaksi mejanya.
  \item Dengan memakai teorema \omterm{def:b1:angular-momentum:L}{momentum sudut} terhadap $C$ dengan $J_C = J +
        MR^2$ (Huygens), temukan $a_G$ lalu tunjukkan bahwa gelendongnya menggelinding
        \emph{menuju} tangannya.
  \item Kini talinya ditarik pada sudut $\beta$ di atas arah
        mendatar. Tunjukkan bahwa gelendongnya tak menggelinding ketika garis
        kerja $F$ melalui $C$, yakni $\cos\beta = r/R$, dan
        menggelinding menjauhi tangannya untuk tarikan yang lebih curam. Hitunglah sudut itu.
  \item Hitunglah $a_G$ untuk $F = \qty{0.20}{N}$ yang ditarik mendatar, dan
        gaya gesek yang diperlukan; apa yang membatasi $F$ sebelum gelendongnya
        tergelincir?
  \item Ringkaslah: ketiga teorema yang dipakai (\omterm{def:b1:systems-of-points:cm}{pusat massa}, \omterm{def:b1:angular-momentum:L}{momentum
        sudut} terhadap sebuah sumbu, energi) dan apa yang ditentukan masing-masing.
\end{enumerate}
\end{problem}
