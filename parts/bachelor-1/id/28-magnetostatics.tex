\chapter{Magnetostatika: Medan, Gaya, dan Dipol}\label{ch:b1:magnetostatics}

Sebuah jarum kompas berayun ketika kawat di dekatnya mengalirkan arus:
pengamatan Ørsted pada tahun 1820 itu menyatukan dua ilmu yang tumbuh
terpisah, kelistrikan dan kemagnetan, dan dalam hitungan bulan Ampère
memperoleh hukum gaya antara arus, sedangkan Biot dan Savart memperoleh medan
sebuah kawat. Bab ini adalah statika \omterm{def:b1:magnetostatics:field}{medan magnet}: apa yang diciptakan arus
(peta medan, \omterm{thm:b1:magnetostatics:biotsavart}{hukum Biot--Savart}, dan teorema Ampère, kembaran magnetik
\omterm{thm:b1:electrostatics-gauss:gauss}{hukum Gauss}), apa yang dilakukan medan pada arus (\omterm{thm:b1:magnetostatics:laplace}{gaya Laplace}, yang
menjalankan setiap motor dan setiap alat ukur), serta \omterm{def:b1:magnetostatics:moment}{dipol magnetik}, yaitu
lilitan arus yang menjelaskan kompas, magnet, dan medan Bumi sendiri.

\section{Sumber dan peta medan}

\begin{definition}[Medan magnetostatik]\label{def:b1:magnetostatics:field}
\index{medan magnet}\index{garis medan!magnetik}\index{tesla}
Arus tunak (muatan dalam gerak tunak) menciptakan di setiap titik sebuah
\emph{medan magnet} $\vect B$ (tesla, \unit{T}), yang didefinisikan oleh gaya
$q\vect v\wedge\vect B$ yang dikerahkannya pada muatan uji yang bergerak
(\cref{def:b1:charged-particles:lorentz}). Yang disebut \emph{\omterm{def:b1:electrostatics-gauss:field}{garis medan}}-nya
menyinggung $\vect B$ dan berarah menyusurinya. Magnet tetap menciptakan
medan yang sehakikat, dari arus mikroskopik atom-atomnya. \omterm{def:b1:units-dimensions:oom}{Orde besarnya}:
Bumi \qty{5e-5}{T}; magnet \omterm{def:b1:heat-engines:machine}{lemari es} \qty{0.01}{T}; sebuah pemindai MRI
\qty{1.5}{T} sampai \qty{3}{T}; medan tunak laboratorium yang terbesar
\qty{45}{T}.
\end{definition}

\begin{proposition}[Peta medan]\label{prop:b1:magnetostatics:maps}
\index{aturan tangan kanan}\index{kekekalan fluks (magnetik)}
\begin{enumerate}
  \item Sebuah kawat lurus: lingkaran yang berpusat pada kawatnya, pada bidang
        yang tegak lurus terhadapnya, berarah menurut \emph{aturan tangan kanan}
        (ibu jari menyusuri arusnya, jari-jari lain menyusuri $\vect B$).
  \item Sebuah lilitan melingkar: garis yang menembus lilitannya menyusuri
        sumbunya lalu menutup di luar; dari jauh, peta sebuah magnet kecil.
  \item Sebuah solenoida panjang: garis yang sejajar, rapat dan seragam di
        dalam; hampir tak ada medan di luar; peta sebuah magnet batang, yang
        muka ``utara''-nya adalah muka yang ditinggalkan garisnya.
  \item Garis \omterm{def:b1:magnetostatics:field}{medan magnet} tak berpangkal dan tak berujung: ia menutup pada
        dirinya sendiri atau lari ke tak hingga. Setara dengan itu, \emph{fluks
        $\vect B$ menembus sebarang permukaan tertutup bernilai nol} --- tak ada
        muatan magnetik.
\end{enumerate}
\end{proposition}
\admitted

\begin{proposition}[Simetri $\vect B$]\label{prop:b1:magnetostatics:symmetry}
\index{simetri (sebuah medan)!magnetik}
$\vect B$ berperilaku di bawah pencerminan \emph{berlawanan} dengan $\vect E$
(ia sebuah \emph{pseudovektor}, yang dibangun dari hasil kali silang): pada sebuah
titik bidang \emph{simetri} arusnya, $\vect B$ \emph{tegak lurus} pada
bidangnya; pada sebuah titik bidang \emph{antisimetri} (bayangan cerminnya
membalik arusnya), $\vect B$ terletak \emph{di dalam} bidangnya. Invariansinya
berpindah seperti bagi $\vect E$.
\end{proposition}

\begin{proof}
Medan sebuah unsur arus $\propto I\,\dd\vect l\wedge\vect e_r$
(di bawah); sebuah cermin membalik satu faktor hasil kali silang relatif
terhadap bayangan sebuah vektor sejati, sehingga bayangan $\vect B$ adalah
lawan bayangan cermin anak panahnya. Pada \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang simetri} $\vect B$ harus sama
dengan lawan bayangan cerminnya sendiri, jadi normal terhadap bidangnya; pada
\omterm{prop:b1:electrostatics-gauss:symmetry}{bidang antisimetri}, lawan dari lawannya: di dalam bidangnya.
\end{proof}

\begin{example}[Memakai simetrinya]\label{ex:b1:magnetostatics:symmetryuse}
Bagi sebuah kawat lurus, setiap bidang yang memuat kawatnya adalah \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang simetri},
sehingga di setiap titik $\vect B$ tegak lurus pada bidang yang melalui
kawatnya dan titiknya: ortoradial, $\vect B = B(r)\,\vect e_\theta$ menurut
invariansi menyusuri dan mengelilingi kawatnya. Bagi sebuah lilitan, bidang lilitannya
adalah \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang simetri}: padanya $\vect B$ normal terhadap lilitannya; sumbunya
terletak di dalam setiap \omterm{prop:b1:electrostatics-gauss:symmetry}{bidang antisimetri} yang memuatnya: pada sumbunya $\vect B$
menyusuri sumbunya.
\end{example}

\section{Hukum Biot--Savart}

\begin{theorem}[Biot--Savart]\label{thm:b1:magnetostatics:biotsavart}
\index{hukum Biot--Savart}\index{permeabilitas hampa}
Sebuah rangkaian yang mengalirkan arus $I$ menciptakan di $M$ medan
\[
\vect B(M) = \frac{\mu_0}{4\pi}\sum_{\text{rangkaian}} \frac{I\,\dd\vect l\wedge\vect e_{PM}}{PM^2} ,
\]
yaitu jumlah atas unsur $\dd\vect l$ rangkaiannya di titik $P$,
yang berarah menyusuri arusnya, dengan $\vect e_{PM}$ vektor satuan dari
$P$ ke $M$, dan $\mu_0 = 4\pi \times 10^{-7}\,\unit{T.m/A}$ yang disebut
\emph{\omterm{thm:b1:magnetostatics:biotsavart}{permeabilitas hampa}}. Tiap unsurnya menyumbang medan yang
tegak lurus pada dirinya sendiri dan pada garis $PM$, meluruh sebanding $1/PM^2$.
\end{theorem}
\admitted

\begin{proposition}[Kawat, lilitan, solenoida]\label{prop:b1:magnetostatics:three}
\index{medan sebuah kawat lurus}\index{medan sebuah lilitan}\index{medan sebuah solenoida}
\begin{enumerate}
  \item Kawat lurus tak hingga, pada jarak $r$: $B = \dfrac{\mu_0I}{2\pi r}$.
  \item Lilitan melingkar berjari-jari $R$, pada sumbunya sejauh $z$ dari
        pusatnya: $B_z = \dfrac{\mu_0IR^2}{2(R^2 + z^2)^{3/2}} = \dfrac{\mu_0I}{2R}
        \sin^3\alpha$, dengan $\alpha$ setengah sudut tempat lilitannya
        terlihat dari $M$; di pusatnya $B = \mu_0I/2R$.
  \item Solenoida berlilitan $n$ tiap meter, pada sumbunya: $B = \tfrac12\mu_0nI
        (\cos\alpha_1 - \cos\alpha_2)$, dengan $\alpha_1, \alpha_2$ sudut tempat
        kedua ujungnya terlihat; bagi solenoida yang panjang, $B = \mu_0nI$
        di dalam, separuhnya di sebuah ujung.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) $M$ berjarak $r$ dari kawatnya, $P$ berabsis $z$ menyusurinya:
$\dd\vect l\wedge\vect e_{PM}$ bernilai $\dd z\,\sin\beta = \dd z\cdot r/PM$
dengan $PM = \sqrt{r^2 + z^2}$; semuanya ortoradial:
$B = \dfrac{\mu_0I}{4\pi}\displaystyle\int_{-\infty}^{\infty}\dfrac{r\,\dd z}{(r^2 + z^2)^{3/2}}
= \dfrac{\mu_0I}{4\pi}\,\dfrac{2}{r}$. (2) Tiap unsur lilitannya berjarak
sama $\sqrt{R^2 + z^2}$ dari $M$, tegak lurus pada
$\vect e_{PM}$; medannya bernilai $\mu_0I\,\dd l/4\pi(R^2 + z^2)$, dan
hanya komponen aksialnya, yaitu pecahan $R/\sqrt{R^2 + z^2}$ darinya,
yang bertahan terhadap penjumlahan mengelilingi lilitannya: $B_z = \mu_0I\cdot2\pi R\cdot R/4\pi(R^2 +
z^2)^{3/2}$. (3) Sebuah irisan $\dd z'$ solenoidanya adalah sebuah lilitan berarus
$nI\,\dd z'$; dengan $z' - z = R/\tan\alpha$, $\dd z' = -R\,\dd\alpha/\sin^2\alpha$,
dan $B = \dfrac{\mu_0nI}{2}\displaystyle\int\sin^3\alpha\,\dfrac{\dd\alpha}{\sin^2\alpha}$ atas
solenoidanya $= \tfrac12\mu_0nI(\cos\alpha_1 - \cos\alpha_2)$; bila panjangnya
tak hingga, $\alpha_1 \to 0$, $\alpha_2 \to \pi$.
\end{proof}

\begin{example}[Orde besarnya]\label{ex:b1:magnetostatics:orders}
Sebuah kawat \qty{10}{A} pada \qty{1}{cm}: $B = 2 \times 10^{-7} \times 10/0.01 =
\qty{2e-4}{T}$, empat kali medan Bumi --- kompasnya memang berayun.
Sebuah kumparan berlilitan $100$, berjari-jari \qty{5}{cm}, \qty{2}{A}: $\mu_0NI/2R =
\qty{2.5}{mT}$ di pusatnya. Sebuah solenoida berlilitan $1000$ tiap meter pada
\qty{5}{A}: \qty{6.3}{mT}; dengan teras besi berpermeabilitas nisbi
$1000$, beberapa \omterm{def:b1:magnetostatics:field}{tesla} akan menyusul seandainya besinya tak menjenuh dekat
\qty{2}{T} --- langit-langit setiap elektromagnet, dan itulah sebabnya
medan terkuat datang dari kumparan superkonduktor tanpa besi.
\end{example}

\section{Hukum Ampère}

\begin{theorem}[Ampère]\label{thm:b1:magnetostatics:ampere}
\index{hukum Ampère}\index{sirkulasi!medan magnet}
Sirkulasi $\vect B$ menyusuri sebarang kurva tertutup berarah $\Gamma$
sama dengan $\mu_0$ dikalikan arus total yang menembus sebarang permukaan yang
dibatasi $\Gamma$, dihitung positif bila ia menembus dalam arah yang diberikan
aturan tangan kanan dari arah $\Gamma$:
\[
\oint_\Gamma \vect B\cdot\dd\vect l = \mu_0 I_{\text{dalam}} .
\]
\end{theorem}

\begin{proof}
Bagi sebuah lingkaran berjari-jari $r$ yang berpusat pada kawat lurus, $\vect B$
menyinggung dan bernilai $\mu_0I/2\pi r$ di mana-mana, sehingga sirkulasinya
$2\pi r \times \mu_0I/2\pi r = \mu_0I$, tak bergantung pada $r$; bagi sebuah kurva yang tak
melingkupi kawatnya sumbangannya saling meniadakan. Kasus umumnya
diterima tanpa bukti (jilid Tahun ke-2).
\end{proof}

\begin{method}[Memakai hukum Ampère]\label{meth:b1:magnetostatics:ampere}
Seperti bagi Gauss: (1) simetrinya memberi arah $\vect B$ dan variabel
yang menjadi gantungannya; (2) pilihlah sebuah \emph{\omterm{def:b1:point-kinematics:frame}{lintasan} Ampère} lewat $M$
yang padanya $\vect B$ menyinggung dengan nilai tetap atau
tegak lurus; (3) cacahlah arus yang menembusnya; (4) selesaikanlah. Cara ini berhasil
bagi kawat dan silinder tak hingga, solenoida dan toroida tak hingga, serta
lembar arus tak hingga.
\end{method}

\begin{proposition}[Silinder, solenoida, toroida]\label{prop:b1:magnetostatics:ampereapps}
\begin{enumerate}
  \item Sebuah kawat silinder berjari-jari $a$ yang mengalirkan $I$ secara seragam:
        $B = \dfrac{\mu_0Ir}{2\pi a^2}$ di dalam, $\dfrac{\mu_0I}{2\pi r}$ di luar
        --- maksimumnya di permukaannya.
  \item Sebuah solenoida tak hingga berlilitan $n$ tiap meter: $\vect B = \mu_0nI\,
        \vect e_z$ seragam di dalam, $\vect 0$ di luar, apa pun bentuk
        penampangnya.
  \item Sebuah toroida berlilitan $N$: $B = \dfrac{\mu_0NI}{2\pi r}$ di dalam
        gulungannya, nol di luar --- sebuah solenoida yang menutup pada dirinya
        sendiri, tanpa medan sasar.
\end{enumerate}
\end{proposition}

\begin{proof}
(1) Lingkaran berjari-jari $r$: $2\pi rB = \mu_0I(r/a)^2$ atau $\mu_0I$. (2) $\vect B$
aksial menurut simetrinya dan, menurut Ampère pada sebuah persegi panjang yang kedua
sisi panjangnya di luar, seragam di luar, jadi nol (ia lenyap di tak hingga); sebuah
persegi panjang dengan satu sisi panjangnya di dalam, sepanjang $\ell$, memberi $B\ell =
\mu_0n\ell I$. (3) Lingkaran berjari-jari $r$ di dalam torusnya: $2\pi rB =
\mu_0NI$; di luar, arus terlingkupinya nol.
\end{proof}

\section{Gaya Laplace}

\begin{theorem}[Gaya Laplace]\label{thm:b1:magnetostatics:laplace}
\index{gaya Laplace}
Sebuah unsur penghantar $\dd\vect l$ yang mengalirkan $I$ dalam medan $\vect B$
menerima gaya $\dd\vect F = I\,\dd\vect l\wedge\vect B$; sebuah ruas
lurus $\vect L$ dalam medan seragam menerima $\vect F = I\vect L\wedge\vect B$,
yang bernilai $ILB\sin\theta$, tegak lurus baik pada kawatnya maupun pada
medannya.
\end{theorem}

\begin{proof}
Pengemban muatan dalam $\dd\vect l$ (jumlahnya $n\,S\,\dd l$, muatannya $q$,
kecepatan hanyutnya $\vect v$) masing-masing merasakan $q\vect v\wedge\vect B$; jumlahnya
adalah $nSq\,\dd l\,\vect v\wedge\vect B = I\,\dd\vect l\wedge\vect B$ karena $I = nqvS$
dan $\vect v$ menyusuri $\dd\vect l$. Pengembannya meneruskan gayanya ke
kisi kawatnya lewat tumbukannya.
\end{proof}

\begin{proposition}[Gaya antara kawat sejajar]\label{prop:b1:magnetostatics:parallel}
\index{ampere (definisi)}
Dua kawat sejajar yang panjang dan terpisah sejauh $d$ yang mengalirkan $I_1$ dan $I_2$
saling menarik bila arusnya searah, saling menolak bila berlawanan arah,
dengan gaya tiap satuan panjang
\[
\frac{F}{\ell} = \frac{\mu_0I_1I_2}{2\pi d} .
\]
Dua kawat yang terpisah \qty{1}{m} dan mengalirkan \qty{1}{A} merasakan $2 \times 10^{-7}$~N tiap
meter --- definisi \omterm{def:b1:dc-circuits:current}{ampere} dari tahun 1948 sampai 2019, dan asal
nilai $\mu_0 = 4\pi \times 10^{-7}$.
\end{proposition}

\begin{proof}
Kawat~1 menciptakan $\mu_0I_1/2\pi d$ di kawat~2, tegak lurus terhadapnya;
\omterm{thm:b1:magnetostatics:laplace}{gaya Laplace} pada panjang $\ell$ kawat~2 adalah $I_2\ell \times \mu_0I_1/2\pi d$,
terarah menuju kawat~1 bila arusnya searah (periksalah dengan
aturan tangan kanan).
\end{proof}

\section{Dipol magnetik}

\begin{definition}[Momen magnetik]\label{def:b1:magnetostatics:moment}
\index{momen magnetik}\index{dipol magnetik}
Sebuah lilitan datar berluas $S$ yang mengalirkan $I$, berarah menurut aturan tangan kanan
(normal $\vect n$ menyusuri ibu jari ketika jari-jari lain mengikuti $I$), memiliki
\emph{momen magnetik} $\vect m = IS\,\vect n$ (\unit{A.m^2}); bagi $N$
lilitan, $NIS\,\vect n$. Dilihat dari jarak $r \gg \sqrt S$ lilitannya adalah sebuah
\emph{dipol magnetik}: medannya berbentuk persis seperti medan \omterm{def:b1:potential-capacitors:dipole}{dipol
listrik} dengan $\vect p/\varepsilon_0 \to \mu_0\vect m$:
\[
B_r = \frac{\mu_0}{4\pi}\,\frac{2m\cos\theta}{r^3}, \qquad
B_\theta = \frac{\mu_0}{4\pi}\,\frac{m\sin\theta}{r^3} .
\]
Sebuah magnet batang, sebuah atom, Bumi adalah dipol magnetik: momen
Bumi \qty{8e22}{A.m^2}, momen sebuah elektron \qty{9.3e-24}{A.m^2} (yaitu
magneton Bohr).
\end{definition}

\begin{proof}
Pada sumbunya medan lilitan $\mu_0IR^2/2z^3$ untuk $z \gg R$ sama dengan
$\mu_0 \cdot 2m/4\pi z^3$ dengan $m = I\pi R^2$, cocok dengan $B_r$ pada $\theta = 0$;
bentuk sudut lengkapnya diterima tanpa bukti.
\end{proof}

\begin{theorem}[Dipol dalam sebuah medan]\label{thm:b1:magnetostatics:dipoleinfield}
\index{dipol magnetik!momen gaya}\index{dipol magnetik!energi}
Sebuah \omterm{def:b1:magnetostatics:moment}{momen magnetik} $\vect m$ dalam medan seragam $\vect B$ tak merasakan
gaya total, merasakan \omterm{def:b1:angular-momentum:moment}{momen gaya} $\vect\Gamma = \vect m\wedge\vect B$ yang menyearahkannya
dengan medannya, dan memiliki \omterm{def:b1:work-and-energy:conservative}{energi potensial} $E_p = -\vect m\cdot\vect B$. Dalam
medan tak seragam ia ditarik ke daerah bermedan kuat ketika
searah (dan didorong menjauh ketika berlawanan arah).
\end{theorem}

\begin{proof}
Bagi sebuah lilitan persegi panjang $a \times b$ dengan $\vect n$ bersudut $\theta$ terhadap
$\vect B$: \omterm{thm:b1:magnetostatics:laplace}{gaya Laplace} pada kedua sisi sepanjang $b$ yang sejajar dengan
sumbu putarnya adalah $\pm IbB$, berlawanan, terpisah sejauh $a\sin\theta$:
sebuah kopel bermomen $IabB\sin\theta = mB\sin\theta$, yang cenderung
memperkecil $\theta$; gaya pada kedua sisi lainnya berlawanan dan
segaris. Energinya: $\Gamma = -\dd E_p/\dd\theta$ dengan $\Gamma = -mB\sin\theta$
memberi $E_p = -mB\cos\theta$. Setiap lilitan datar adalah jumlah persegi panjang kecil;
kasus tak seragamnya diterima tanpa bukti.
\end{proof}

\section{Soal: Kabel, saluran, dan alat ukur}

\begin{problem}[{Soal akhir pekan --- hukum Ampère pada sebuah kabel sesumbu,
gaya Laplace pada sebuah saluran daya, dan rancangan sebuah
galvanometer kumparan putar}]
\label{pb:b1:magnetostatics:1}
$\mu_0 = 4\pi \times 10^{-7}$~SI; resistivitas tembaga $\rho = \qty{1.7e-8}{\ohm.m}$.

\medskip
\noindent\textbf{Bagian I --- Kabel sesumbunya.}
Sebuah kabel mengalirkan $I = \qty{1.0}{A}$ menyusuri teras tembaga pejal berjari-jari
$a = \qty{0.50}{mm}$ dan kembali menyusuri sebuah selongsong berjari-jari dalam $b =
\qty{2.5}{mm}$ dan berjari-jari luar $c = \qty{3.0}{mm}$; arusnya seragam
pada penampang tiap penghantarnya.
\begin{enumerate}
  \item Simetri dan invariansinya: arah $\vect B$ dan variabel
        yang menjadi gantungannya.
  \item $B(r)$ di dalam terasnya, di antara kedua penghantarnya, dan di luar
        kabelnya, dengan \omterm{thm:b1:magnetostatics:ampere}{hukum Ampère}.
  \item Nilainya pada $r = a$ dan $r = b$.
  \item Terasnya diganti oleh sebuah tabung tembaga tipis berjari-jari sama
        $a$ yang mengalirkan arus sama: apa yang berubah di tiap daerahnya?
  \item $B(r)$ di dalam selongsongnya ($b < r < c$); periksalah nilainya di $b$
        dan $c$; sketsalah $B(r)$ dari $0$ sampai \qty{4}{mm}.
  \item \omterm{thm:b1:magnetostatics:laplace}{Gaya Laplace} pada selongsongnya: arahnya (tarikan atau
        tolakan dari terasnya?) dan nilainya tiap satuan luas, dengan mengambil
        medan pada selongsongnya sebagai rata-rata nilainya di $b$ dan $c$.
        Serupa bagi arus denyut \qty{10}{kA}.
  \item Mengapa geometri sesumbu dipakai bagi \omterm{def:b1:wave-propagation:signal}{sinyal}? Bandingkan dengan
        sepasang kawat sejajar yang terpisah $d = \qty{2}{mm}$: medan pada jarak
        $D = \qty{10}{cm}$ dari pasangannya (superposisikan dua kawat).
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Saluran dayanya.}
Dua penghantar mendatar yang terpisah $d = \qty{1.0}{m}$, $h = \qty{10}{m}$ di atas
tanah, mengalirkan $I = \qty{500}{A}$ dalam arah berlawanan.
\begin{enumerate}[resume]
  \item Medan salah satu penghantarnya di titik tanah yang tepat di bawah
        titik tengahnya; tunjukkan bahwa kedua medannya berpadu menjadi medan tegak
        lalu berikan nilainya; bandingkan dengan medan Bumi dan dengan
        pedoman paparan \qty{100}{\micro T}.
  \item Tunjukkan bahwa pada jarak $D \gg d$ medan pasangannya meluruh sebanding
        $\mu_0Id/2\pi D^2$: mengapa lebih cepat daripada sebuah kawat tunggal?
  \item Medan pada \qty{1.0}{m} di bawah salah satu penghantarnya, tempat seorang teknisi
        mungkin bekerja.
  \item Gaya tiap meter antara kedua penghantarnya; tarikan atau
        tolakan? Serupa di bawah arus korsleting \qty{20}{kA}.
  \item Gaya pada rentang \qty{300}{m}, dalam keadaan biasa dan dalam korsleting;
        bandingkan dengan berat rentangnya (\qty{1.0}{kg/m}); apa yang dilakukan salurannya

  \item Sebuah saluran tiga fase mengalirkan tiga arus beramplitudo sama
        yang bergeser \qty{120}{\degree}: tunjukkan bahwa jumlahnya lenyap pada

  \item Sebuah kompas di titik pertanyaan~8, tempat medan mendatar

\end{enumerate}

\noindent\textbf{Bagian III --- Galvanometer kumparan putarnya.}
Sebuah kumparan persegi panjang, $N = 200$ lilitan, bersisi $a = \qty{2.0}{cm}$ (sejajar
dengan sumbu putarnya) dan $b = \qty{3.0}{cm}$, berputar terhadap sumbunya dalam
celah di antara sepatu kutub silinder yang medannya, $B = \qty{0.20}{T}$,
bersifat \emph{radial}: selalu di dalam bidang kumparannya, tegak lurus pada
kedua sisi $a$-nya. Sebuah pegas spiral mengerahkan \omterm{def:b1:angular-momentum:moment}{momen gaya} pemulih $-C\theta$ dengan
$C = \qty{1.0e-5}{N.m/rad}$. \omterm{prop:b1:angular-momentum:inertia}{Momen inersia} kumparan dan jarumnya:
$J = \qty{6.7e-8}{kg.m^2}$.
\begin{enumerate}[resume]
  \item \omterm{thm:b1:magnetostatics:laplace}{Gaya Laplace} pada keempat sisinya; yang mana menghasilkan \omterm{def:b1:angular-momentum:moment}{momen gaya}
        terhadap sumbunya?
  \item \omterm{def:b1:angular-momentum:moment}{Momen gayanya} bagi arus $I$; mengapa medan radialnya membuatnya
        tak bergantung pada $\theta$?
  \item Simpangan $\theta$ pada kesetimbangan; kepekaannya dalam radian tiap
        miliampere; arus bagi simpangan skala penuh \qty{1.2}{rad}.
  \item Kumparannya digulung dengan kawat tembaga berdiameter \qty{0.10}{mm}:
        hambatan kumparannya; \omterm{def:b1:dc-circuits:voltage}{tegangan} padanya dan daya yang dilesapkan pada
        skala penuh.
  \item Persamaan gerak kumparannya; periode alami
        osilasi terhadap kesetimbangannya.
  \item Gerak kumparannya di dalam medan mengimbaskan \omterm{def:b1:angular-momentum:moment}{momen gaya} pengerem
        $-\alpha\dot\theta$ (\cref{ch:b1:induction}); nilai $\alpha$ bagi
        peredaman kritis, dan mengapa orang menginginkannya.
  \item \omterm{def:b1:dc-circuits:resistor}{Resistor} seri untuk membuat voltmeter berskala penuh \qty{10}{V}; \omterm{def:b1:dc-circuits:resistor}{resistor}
        pirau untuk membuat amperemeter berskala penuh \qty{1.0}{A}.
  \item \omterm{def:b1:magnetostatics:moment}{Momen magnetik} kumparannya pada skala penuh; energi $-\vect m\cdot
        \vect B$ yang akan dimilikinya, bila searah, dalam medan seragam \qty{0.20}{T};
        bandingkan dengan energi yang tersimpan dalam pegasnya pada skala penuh.
  \item Seandainya medannya \emph{seragam} alih-alih radial, tunjukkan bahwa
        skalanya tak lagi linear: tuliskan syarat kesetimbangannya
        dengan $\theta$ diukur dari kedudukan yang padanya bidang kumparannya
        memuat $\vect B$.
  \item Arus \qty{50}{Hz} beramplitudo \qty{0.5}{mA} dialirkan ke
        alat ukurnya: apa yang ditunjukkan jarumnya, dan mengapa? (Bandingkan \qty{50}{Hz}
        dengan \omterm{def:b1:oscillators-resonance:harmonic}{frekuensi alaminya}.)
  \item Ringkaslah ketiga adegannya dalam tiga angka, dan sebutkan hukum tunggal
        mana yang menghasilkan masing-masingnya.
\end{enumerate}
