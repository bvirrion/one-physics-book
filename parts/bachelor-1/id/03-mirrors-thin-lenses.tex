\chapter{Cermin dan Lensa Tipis}\label{ch:b1:mirrors-thin-lenses}

Cermin rias memperlihatkan wajah dua kali lebih besar; punggung sendok
memperlihatkannya mungil dan tegak, cekungannya memperlihatkannya mungil
dan terbalik; kaca pembesar yang dipegang sepanjang lengan menjungkirkan
seisi ruangan, dan bila dipegang dekat ia memperbesar sekeping
perangko; \omterm{def:b1:mirrors-thin-lenses:lens}{lensa}
telepon sebesar kacang polong melukis seluruh jalan ke atas sebuah
sensor. Satu hubungan aljabar antara tiga jarak mengatur setiap \omterm{def:b1:mirrors-thin-lenses:image}{bayangan}
itu, dan bab ini menegakkannya --- lebih dulu untuk cermin lengkung,
lalu untuk \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis} yang menjadi bahan hampir semua alat optik.

\section{Bayangan dan syarat Gauss}

\begin{definition}[Benda, bayangan, stigmatisme]\label{def:b1:mirrors-thin-lenses:image}
\index{bayangan (optik)}\index{benda (optik)}\index{stigmatisme}\index{bayangan nyata}\index{bayangan maya}
Sebuah sistem optik menerima sinar dari titik $A$ (\emph{titik benda}).
Jika semua sinar yang keluar melalui satu titik $A'$, atau semuanya
seolah datang dari satu titik $A'$, sistemnya bersifat
\emph{stigmatik} bagi $A$ dan $A'$ adalah \emph{bayangan} $A$.
Bayangannya \emph{nyata} jika sinar yang keluar benar-benar berpotongan
di $A'$ (ia dapat ditangkap layar), \emph{maya} jika hanya perpanjangan
mundurnya yang berpotongan (ia dilihat lewat sistemnya, bukan
diproyeksikan). Begitu pula bendanya nyata jika sinar yang masuk
menyebar dari $A$, dan maya jika sinar itu memusat menuju $A$ di
belakang pintu masuk sistemnya. Pasangan $(A, A')$ disebut
\emph{konjugat}.
\end{definition}

\begin{definition}[Sistem terpusat; syarat Gauss]\label{def:b1:mirrors-thin-lenses:gauss}
\index{sumbu optik}\index{syarat Gauss}\index{sinar paraksial}
\emph{Sistem terpusat} mempunyai sumbu kesetangkupan putar, yaitu
\emph{sumbu optik}. Sinar yang tetap dekat dengan sumbunya dan
membentuk sudut kecil terhadapnya disebut \emph{paraksial}; sistem yang
hanya dipakai dengan sinar paraksial dikatakan bekerja dalam
\emph{syarat Gauss}. Dalam syarat itu sistem terpusat bersifat hampir
stigmatik bagi setiap titik, dan bidang benda yang tegak lurus sumbunya
mempunyai bidang \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} yang tegak lurus sumbunya pula
(\emph{aplanatisme}).
\end{definition}
\admitted

\begin{remark}[Mengapa sudut kecil]\label{rem:b1:mirrors-thin-lenses:why}
\omterm{prop:b1:rays-reflection-refraction:mirror}{Cermin datar} adalah satu-satunya sistem sederhana yang stigmatik secara
ketat bagi setiap titik. Cermin atau \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} sferis stigmatik hanya
sejauh $\sin\theta \approx \tan\theta \approx \theta$: hampiran itulah
yang mengubah \omterm{thm:b1:rays-reflection-refraction:snell}{hukum Snell} menjadi hubungan linear antara jarak. Di luar
itu \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} sebuah titik menjadi noda kabur (\emph{aberasi}), dan
tugas perancangnya --- diafragma, gabungan \omterm{def:b1:mirrors-thin-lenses:lens}{lensa}, permukaan asferis ---
adalah menekan kekaburan itu di bawah apa yang dapat diuraikan mata
atau sensor.
\end{remark}

\begin{notation}[Ukuran aljabar dan perbesaran]\label{not:b1:mirrors-thin-lenses:signs}
\index{ukuran aljabar}\index{perbesaran melintang}
\omterm{def:b1:mirrors-thin-lenses:gauss}{Sumbu optiknya} diarahkan searah cahaya yang datang; arah melintangnya
diarahkan ke atas. Untuk titik $P$, $Q$ pada sumbunya,
$\overline{PQ}$ menyatakan \omterm{rem:b1:mirrors-thin-lenses:why}{ukuran aljabar} ($\overline{PQ} > 0$
jika $Q$ berada di hilir $P$); untuk benda $AB$ yang tegak lurus
sumbunya dan \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} $A'B'$, \emph{\omterm{rem:b1:mirrors-thin-lenses:why}{perbesaran melintang}}nya adalah
\[
\gamma = \frac{\overline{A'B'}}{\overline{AB}} ,
\]
negatif untuk \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} terbalik, dan bernilai mutlak lebih besar
daripada $1$ untuk \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} yang diperbesar.
\end{notation}

\section{Cermin sferis}

\begin{definition}[Cermin sferis]\label{def:b1:mirrors-thin-lenses:mirror}
\index{cermin sferis}\index{cermin cekung}\index{cermin cembung}\index{titik fokus}
\emph{Cermin sferis} adalah tembereng bola yang memantulkan, berpusat
$C$ dan berpuncak $S$ (perpotongannya dengan sumbu). Ia \emph{cekung}
jika cahayanya menemui sisi cekungnya ($C$ berada di sisi cahaya yang
datang), dan \emph{cembung} jika sebaliknya. \emph{Titik fokus}nya $F$
adalah \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} sebuah titik di tak hingga pada sumbunya; jarak fokusnya
$f = \overline{SF}$.
\end{definition}

\begin{proposition}[Titik fokus cermin sferis]\label{prop:b1:mirrors-thin-lenses:focus}
Dalam \omterm{def:b1:mirrors-thin-lenses:gauss}{syarat Gauss}, titik fokusnya adalah titik tengah $SC$:
$\overline{SF} = \overline{SC}/2$. \omterm{def:b1:mirrors-thin-lenses:mirror}{Cermin cekung} mempunyai $f < 0$
(fokus nyata, di depan); \omterm{def:b1:mirrors-thin-lenses:mirror}{cermin cembung} mempunyai $f > 0$ (fokus maya,
di belakang).
\end{proposition}

\begin{proof}
Sinar yang sejajar sumbu pada ketinggian $h$ menemui cerminnya di $I$;
garis normal di $I$ adalah jari-jari $CI$, yang membentuk sudut $\alpha$
dengan sumbunya, $\sin\alpha = h/R$. Dipantulkan dengan sudut $\alpha$
yang sama terhadap garis normalnya, sinar itu memotong sumbunya di $F$;
segitiga $CIF$ mempunyai sudut $\alpha$ yang sama besar di $C$ dan di
$I$, sehingga $CF = IF$ dan, dengan memproyeksikan,
$CF = R/(2\cos\alpha)$. Untuk \omterm{def:b1:mirrors-thin-lenses:gauss}{sinar paraksial} $\cos\alpha \to 1$ dan
$CF = R/2$: semua sinar semacam itu memotong sumbunya di titik yang
sama, yaitu titik tengah $SC$. Tandanya: untuk \omterm{def:b1:mirrors-thin-lenses:mirror}{cermin cekung} $C$ dan
$F$ terletak di depan (di hulu), $\overline{SC} < 0$.
\end{proof}

\begin{theorem}[Konjugasi untuk cermin sferis]\label{thm:b1:mirrors-thin-lenses:mirrorconj}
\index{hubungan konjugasi!cermin sferis}
Dalam \omterm{def:b1:mirrors-thin-lenses:gauss}{syarat Gauss}, \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} $A'$ sebuah titik $A$ pada sumbunya
memenuhi
\[
\frac{1}{\overline{SA'}} + \frac{1}{\overline{SA}} = \frac{2}{\overline{SC}}
= \frac{1}{\overline{SF}},
\qquad
\gamma = -\frac{\overline{SA'}}{\overline{SA}} .
\]
\end{theorem}

\begin{proof}
Ambil \omterm{def:b1:mirrors-thin-lenses:mirror}{cermin cekung} dan benda nyata $A$ di luar $C$; bentuk aljabarnya
kemudian mencakup setiap kasus. Sinar dari $A$ mengenai cerminnya di $I$
(ketinggian $h$) lalu kembali ke $A'$ pada sumbunya. Sebut $u$, $u'$,
$\omega$ sudut $AI$, $A'I$ dan $CI$ terhadap sumbunya; secara paraksial
$u = h/AS$, $u' = h/A'S$, $\omega = h/CS$ (jarak). Hukum pemantulan
mengatakan bahwa garis normal $CI$ membagi dua sudut antara $AI$ dan
$A'I$: $\omega - u = u' - \omega$, yaitu $u + u' = 2\omega$, yaitu
$1/AS + 1/A'S = 2/CS$; dengan $\overline{SA}$, $\overline{SA'}$,
$\overline{SC}$ semuanya negatif, inilah hubungan yang dipajang. Untuk
perbesarannya, sinar $B \to S \to B'$ yang dipantulkan di puncaknya
membentuk sudut yang sama besar dengan sumbunya: $A'B'/SA' = AB/SA$
dalam panjang, dan \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} terbalik,
$\gamma = -\overline{SA'}/\overline{SA}$.
\end{proof}

\begin{method}[Melukis bayangan pada cermin]\label{met:b1:mirrors-thin-lenses:mirrorrays}
Untuk titik benda $B$ di luar sumbunya, lukislah dua dari empat sinar
istimewa; perpotongannya (atau perpotongan perpanjangannya) adalah $B'$:
\begin{enumerate}
  \item sejajar sumbu $\to$ dipantulkan melalui $F$;
  \item melalui $F$ $\to$ dipantulkan sejajar sumbu;
  \item melalui $C$ $\to$ dipantulkan kembali menyusuri dirinya sendiri;
  \item menuju puncak $S$ $\to$ dipantulkan setangkup terhadap sumbunya.
\end{enumerate}
Lalu $A'$ adalah kaki garis tegak lurus dari $B'$ ke sumbunya.
\end{method}

\begin{example}[Cermin cukur, cermin spion]\label{ex:b1:mirrors-thin-lenses:mirrors}
\omterm{def:b1:mirrors-thin-lenses:mirror}{Cermin cekung} berjari-jari \qty{60}{cm} ($f = \qty{-30}{cm}$) dengan
wajah pada \qty{20}{cm}: $1/\overline{SA'} = -1/30 + 1/20 = 1/60$,
$\overline{SA'} = \qty{+60}{cm}$ --- di belakang cerminnya, maya ---
dan $\gamma = -60/(-20) = +3$: tegak, tiga kali lebih besar, itulah
cermin cukur. \omterm{def:b1:mirrors-thin-lenses:mirror}{Cermin cembung} berjari-jari \qty{2.0}{m}
($f = \qty{+1.0}{m}$) dengan sebuah mobil \qty{20}{m} di belakangnya:
$1/\overline{SA'} = 1 + 1/20$, $\overline{SA'} = \qty{0.95}{m}$ (maya,
di belakang), $\gamma = +0.048$: tegak, dua puluh kali lebih kecil, dan
karena itu bermedan pandang luas --- cermin spion, lengkap dengan
peringatannya bahwa benda lebih dekat daripada yang tampak.
\end{example}

\section{Lensa tipis}

\begin{definition}[Lensa tipis]\label{def:b1:mirrors-thin-lenses:lens}
\index{lensa tipis}\index{lensa konvergen}\index{lensa divergen}\index{pusat optik}\index{jarak fokus}\index{vergensi}
Yang disebut \emph{lensa} adalah \omterm{def:b1:rays-reflection-refraction:index}{medium bening} yang dibatasi dua permukaan bola
(atau datar); ia \emph{tipis} bila tebalnya dapat diabaikan terhadap
jari-jarinya dan terhadap jarak yang terlibat, sehingga kedua puncaknya
melebur menjadi satu titik, yaitu \emph{pusat optik} $O$. Sinar yang
melalui $O$ tidak dibelokkan. \emph{Fokus \omterm{def:b1:mirrors-thin-lenses:image}{bayangan}} $F'$ adalah
\omterm{def:b1:mirrors-thin-lenses:image}{bayangan} sebuah titik di tak hingga pada sumbunya; \emph{fokus benda}
$F$ adalah titik yang \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} di tak hingga; keduanya berjarak sama
dari $O$: $\overline{OF'} = -\overline{OF} = f'$, yaitu \emph{jarak
fokus}. Lensanya \emph{konvergen} bila $f' > 0$ ($F'$ nyata, di hilir;
tepinya lebih tipis daripada tengahnya), dan \emph{divergen} bila
$f' < 0$. \emph{Vergensi}nya $V = 1/f'$ diukur dalam \emph{dioptri}
($\qty{1}{\delta} = \qty{1}{m^{-1}}$).
\end{definition}

\begin{theorem}[Konjugasi untuk lensa tipis]\label{thm:b1:mirrors-thin-lenses:lensconj}
\index{hubungan konjugasi!lensa tipis}\index{hubungan Descartes}\index{hubungan Newton}
Dalam \omterm{def:b1:mirrors-thin-lenses:gauss}{syarat Gauss}, \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} $A'$ sebuah titik $A$ pada sumbunya
memenuhi \omterm{thm:b1:mirrors-thin-lenses:lensconj}{hubungan Descartes}
\[
\frac{1}{\overline{OA'}} - \frac{1}{\overline{OA}} = \frac{1}{f'},
\qquad
\gamma = \frac{\overline{OA'}}{\overline{OA}},
\]
dan, secara setara, \omterm{thm:b1:mirrors-thin-lenses:lensconj}{hubungan Newton}
\[
\overline{FA}\cdot\overline{F'A'} = -f'^2,
\qquad
\gamma = \frac{f'}{\overline{FA}} = -\frac{\overline{F'A'}}{f'} .
\]
\end{theorem}

\begin{proof}[Bukti sebagian]
Fakta kuncinya: \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis} mendeviasikan sinar yang melintasinya pada
ketinggian $h$ sebesar sudut $h/f'$ ke arah sumbunya, apa pun arah
sinarnya --- tiap daerah \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} berlaku sebagai prisma tipis
(\cref{rem:b1:rays-reflection-refraction:thin}) yang sudut puncaknya
tumbuh linear terhadap $h$ untuk permukaan bola; kelinearan ini, yang
eksak untuk \omterm{def:b1:mirrors-thin-lenses:gauss}{sinar paraksial}, diterima saja di sini dan diturunkan dari
\omterm{thm:b1:rays-reflection-refraction:snell}{hukum Snell} pada dua permukaan bola dalam volume Tahun ke-2. Dengan
menerimanya: sinar dari $A$ (pada jarak $p = -\overline{OA} > 0$ untuk
benda nyata) yang mencapai \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} pada ketinggian $h$ tiba dengan
kemiringan $h/p$, dibelokkan sebesar $h/f'$ dan keluar dengan kemiringan
$h/p - h/f'$; ia memotong sumbunya di $A'$ dengan
$h/\overline{OA'} = h/f' - h/p$, yaitu $1/\overline{OA'}
- 1/\overline{OA} = 1/f'$, bebas dari $h$: semua sinar dari $A$ bertemu
di $A'$. Sinar $B \to O \to B'$ yang melalui pusatnya berupa garis
lurus, sehingga $\overline{A'B'}/\overline{AB} = \overline{OA'}/\overline{OA}$.
Bentuk Newton menyusul dengan menyubstitusikan $\overline{OA} = \overline{OF} + \overline{FA}
= -f' + \overline{FA}$ dan $\overline{OA'} = f' + \overline{F'A'}$.
\end{proof}

\begin{method}[Melukis bayangan lewat lensa tipis]\label{met:b1:mirrors-thin-lenses:lensrays}
Tiga sinar istimewa dari $B$:
\begin{enumerate}
  \item melalui $O$: tak dibelokkan;
  \item sejajar sumbu: keluar melalui $F'$;
  \item melalui $F$: keluar sejajar sumbu.
\end{enumerate}
Dua di antaranya sudah cukup; untuk \omterm{def:b1:mirrors-thin-lenses:lens}{lensa divergen}, pakailah
perpanjangannya (sinar yang dibidikkan ke $F$ di belakang \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} keluar
sejajar; sinar sejajar keluar seolah dari $F'$ di depan).
\end{method}

\begin{example}[Lensa yang sama, tiga cara]\label{ex:b1:mirrors-thin-lenses:lensuses}
Sebuah \omterm{def:b1:mirrors-thin-lenses:lens}{lensa konvergen}, $f' = \qty{5.0}{cm}$. Perangko pada
\qty{3.0}{cm}: $1/\overline{OA'} = 1/5 - 1/3 = -2/15$,
$\overline{OA'} = \qty{-7.5}{cm}$, maya, $\gamma = +2.5$: itulah kaca
pembesar. Lampu pada \qty{20}{cm}: $\overline{OA'} = \qty{6.7}{cm}$,
nyata, $\gamma = -1/3$: \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} terbalik yang diperkecil pada sehelai
kartu. Jendela sejauh \qty{4}{m}: $\overline{OA'}
\approx f'$, \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} terbalik yang mungil di bidang fokusnya --- itulah
kamera. Tiga kegunaan, satu hubungan.
\end{example}

\begin{proposition}[Perbesaran membujur]\label{prop:b1:mirrors-thin-lenses:longitudinal}
\index{perbesaran membujur}
Jika bendanya bergerak menyusuri sumbunya sejauh $\dd\overline{OA}$,
\omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} bergerak sejauh $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$,
dengan arah yang sama.
\end{proposition}

\begin{proof}
Turunkan \omterm{thm:b1:mirrors-thin-lenses:lensconj}{hubungan Descartes} pada $f'$ tetap:
$-\dd\overline{OA'}/\overline{OA'}^2 + \dd\overline{OA}/\overline{OA}^2 = 0$,
sehingga $\dd\overline{OA'} = (\overline{OA'}/\overline{OA})^2\,\dd\overline{OA}$.
\end{proof}

\begin{remark}[Kedalaman fokus]\label{rem:b1:mirrors-thin-lenses:depth}
Proyektor dengan $\gamma = -50$ mempunyai $\gamma^2 = 2500$: slide yang
melengkung sebesar \qty{0.1}{mm} oleh panas lampunya melemparkan
\omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} \qty{25}{cm} keluar dari bidang layarnya. Sebaliknya kamera
($\abs\gamma
\ll 1$) menoleransi perpindahan benda yang besar demi geseran \omterm{def:b1:mirrors-thin-lenses:image}{bayangan}
yang mungil --- itulah kedalaman medan.
\end{remark}

\section{Mengukur sebuah jarak fokus}

\begin{method}[Autokolimasi]\label{met:b1:mirrors-thin-lenses:autocollimation}
\index{autokolimasi}
Tempatkan \omterm{prop:b1:rays-reflection-refraction:mirror}{cermin datar} tepat di belakang \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} lalu geser benda yang
disinari (sebuah tanda silang pada kaca buram) menyusuri sumbunya sampai
\omterm{def:b1:mirrors-thin-lenses:image}{bayangannya}, yang dipantulkan kembali lewat \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya}, terbentuk tajam
pada benda itu sendiri, terbalik dan sama besar. Bendanya lalu berada
di bidang fokus: sinarnya meninggalkan \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} sejajar, kembali sejajar
dari cerminnya, dan difokuskan kembali di bidang fokusnya. Jarak
benda--lensanya adalah $f'$.
\end{method}

\begin{proposition}[Metode Bessel]\label{prop:b1:mirrors-thin-lenses:bessel}
\index{metode Bessel}\index{metode Silbermann}
Sebuah benda dan sebuah layar dipasang tetap berjarak $D$. \omterm{def:b1:mirrors-thin-lenses:lens}{Lensa
konvergen} membentuk \omterm{def:b1:mirrors-thin-lenses:image}{bayangan nyata} yang tajam pada layarnya tepat pada
dua kedudukan jika $D > 4f'$, pada satu kedudukan jika $D = 4f'$, dan
pada tak satu pun jika $D < 4f'$. Kedua kedudukan itu berjarak $d$
dengan
\[
f' = \frac{D^2 - d^2}{4D} .
\]
Pada $D = 4f'$ (Silbermann) kedudukan tunggalnya memberi $\gamma = -1$.
\end{proposition}

\begin{proof}
Misalkan $p = -\overline{OA} > 0$ jarak benda--lensa; maka
$\overline{OA'} = D - p$ dan Descartes memberi $1/(D - p) + 1/p = 1/f'$,
yaitu $p^2 - Dp + Df' = 0$: diskriminannya $D^2 - 4Df'$, positif jika
dan hanya jika $D > 4f'$. Kedua akarnya $p_{1,2} = (D \mp \sqrt{D^2 - 4Df'})/2$
berjumlah $D$ (kedudukannya setangkup: jarak benda dan jarak \omterm{def:b1:mirrors-thin-lenses:image}{bayangan}
saling bertukar) dan berselisih $d = \sqrt{D^2 - 4Df'}$; selesaikan
untuk $f'$. Pada $D = 4f'$, $p = 2f' = \overline{OA'}$ dan $\gamma = -1$.
\end{proof}

\section{Dua lensa tipis}

\begin{proposition}[Lensa bersinggungan; doublet afokal]\label{prop:b1:mirrors-thin-lenses:doublet}
\index{sistem afokal}\index{perbesaran sudut}
Dua \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis} yang bersinggungan berlaku sebagai satu \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis}
bervergensi $V = V_1 + V_2$. Dua \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} yang ditempatkan sehingga
$F_1' = F_2$ membentuk sistem \emph{afokal}: berkas sejajar keluar
sejajar, dan berkas yang miring sebesar $\alpha$ keluar miring sebesar
$\alpha' = G\alpha$ dengan \emph{perbesaran sudut} $G = -f_1'/f_2'$.
\end{proposition}

\begin{proof}
\omterm{def:b1:mirrors-thin-lenses:image}{Bayangan} $A_1$ yang diberikan $L_1$ menjadi benda bagi $L_2$. Ketika
bersinggungan, $O_1 = O_2 = O$: $1/\overline{OA_1} - 1/\overline{OA} = V_1$ dan
$1/\overline{OA'} - 1/\overline{OA_1} = V_2$; jumlahkan. Afokal: berkas
sejajar bersudut $\alpha$ memfokus di bidang fokus bersamanya pada
ketinggian $h = f_1'\alpha$ dari sumbunya (sinar melalui $O_1$); dilihat
dari $L_2$ titik ini berada di bidang fokus bendanya, sehingga berkasnya
keluar sejajar, dengan sudut sinar dari titik itu melalui $O_2$:
$\alpha' = -h/f_2'$.
\end{proof}

\begin{example}[Sebuah teleskop dalam satu baris]\label{ex:b1:mirrors-thin-lenses:afocal}
$f_1' = \qty{1000}{mm}$, $f_2' = \qty{25}{mm}$: $G = -40$; Bulan, yang
lebarnya setengah derajat, menjadi cakram $20^\circ$ --- itulah teleskop
astronomi bab berikutnya. Dengan okuler divergen, $f_2' = \qty{-25}{mm}$,
$G = +40$: tegak, teropong Galileo.
\end{example}

\section{Soal: Proyektor slide}

\begin{problem}[{Soal akhir pekan --- satu lensa, diukur di atas meja
kerja, lalu diminta melemparkan sebuah slide 35 mm menyeberangi aula:
bagaimana sebuah proyektor dirancang, difokuskan, dan dipasangi
pengubah lempar jauh}]
\label{pb:b1:mirrors-thin-lenses:1}
\omterm{def:b1:mirrors-thin-lenses:lens}{Lensa} proyeksi $L_1$ adalah \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis} konvergen dengan \omterm{def:b1:mirrors-thin-lenses:lens}{jarak fokus}
$f_1'$ yang belum diketahui. Slidenya setinggi \qty{24}{mm} dan selebar \qty{36}{mm}.

\medskip
\noindent\textbf{Bagian I --- \omterm{def:b1:mirrors-thin-lenses:lens}{Lensa} di atas meja kerja.}
\begin{enumerate}
  \item Nyatakan \omterm{thm:b1:mirrors-thin-lenses:lensconj}{hubungan Descartes} dan perbesaran untuk \omterm{def:b1:mirrors-thin-lenses:lens}{lensa tipis},
        beserta kesepakatan tandanya.
  \item Benda dan layar dipasang tetap berjarak $D$. Tunjukkan bahwa
        \omterm{def:b1:mirrors-thin-lenses:image}{bayangan nyata} pada layarnya menuntut $D \geq 4f_1'$.
  \item Tunjukkan bahwa, ketika $D > 4f_1'$, kedua kedudukan \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} yang
        memberi \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} tajam setangkup terhadap titik tengah benda
        dan layarnya, dan bahwa $f_1' = (D^2 - d^2)/4D$ dengan $d$
        jarak antarkeduanya.
  \item Dengan $D = \qty{80.0}{cm}$ dan $d = \qty{40.0}{cm}$, hitunglah
        $f_1'$.
  \item \omterm{def:b1:mirrors-thin-lenses:lens}{Lensanya} duduk dalam tabung tebal yang \omterm{def:b1:mirrors-thin-lenses:lens}{pusat optiknya} tak dapat
        ditentukan dengan tepat. Mengapa hal ini tidak memengaruhi
        metodenya?
  \item Jaraknya membawa $u(D) = \qty{0.2}{cm}$ dan
        $u(d) = \qty{0.5}{cm}$. Hitunglah $u(f_1')$ lalu tuliskan hasilnya.
  \item Berikan perbesaran pada masing-masing kedudukannya. Yang mana
        yang bersesuaian dengan sebuah proyektor?
  \item Berapa $D$ terkecil yang masih memberi \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} tajam, dan
        berapa perbesarannya ketika itu (Silbermann)?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Melemparkan gambarnya.}
Gambar pada layarnya harus selebar \qty{1.80}{m}. Ambil
$f_1' = \qty{15.0}{cm}$.
\begin{enumerate}[resume]
  \item Berapa perbesaran yang diperlukan (beserta tandanya)?
  \item Hitunglah jarak lensa--layar dan jarak slide--lensa.
  \item Mengapa slidenya dimasukkan terbalik atas--bawah sekaligus
        terbalik kiri--kanan?
  \item Layarnya dipindahkan ke \qty{10.0}{m}. Sejauh apa, dan ke arah
        mana, \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} harus digeser relatif terhadap slidenya untuk
        memfokuskan kembali? Berapa lebar gambarnya sekarang?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Toleransi fokus dan cahaya.}
\begin{enumerate}[resume]
  \item Tunjukkan bahwa perpindahan kecil $\dd\overline{OA}$ slidenya
        menggeser \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} sebesar $\dd\overline{OA'} = \gamma^2\,\dd\overline{OA}$.
  \item Filmnya melengkung sebesar \qty{0.10}{mm} oleh panas lampunya.
        Sejauh apa bidang \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya} bergerak (layar pada
        \qty{7.65}{m})? Berikan komentar.
  \item Bukaan \omterm{def:b1:mirrors-thin-lenses:lens}{lensanya} bergaris tengah \qty{30}{mm}. Jika layarnya
        berada \qty{20}{cm} di luar bidang \omterm{def:b1:mirrors-thin-lenses:image}{bayangannya}, berapa garis
        tengah noda kabur sebuah titik slidenya? Dilihat dari
        \qty{5}{m}, apakah mata (daya urai sudut sekitar
        $\qty{3e-4}{rad}$) menyadarinya?
  \item Slidenya menerima cahaya \qty{2.0}{W}, yang seluruhnya mencapai
        layarnya. Hitunglah iradiansi (daya tiap satuan luas) pada
        slidenya dan pada layarnya, lalu tunjukkan bahwa nisbahnya
        adalah $\gamma^2$.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pengubah lempar jauh.}
Di aula yang lebih besar layarnya berada \qty{12.0}{m} jauhnya. \omterm{def:b1:mirrors-thin-lenses:lens}{Lensa
divergen} $L_2$, $f_2' = \qty{-30.0}{cm}$, dijepitkan \qty{8.0}{cm} di
belakang $L_1$; layarnya \qty{12.0}{m} dari $L_2$.
\begin{enumerate}[resume]
  \item Agar \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} akhirnya nyata dan berada pada layarnya, di mana
        (dari $L_2$) \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} antara yang diberikan $L_1$ harus
        terletak? Apakah ia benda nyata atau benda maya bagi $L_2$?
  \item Simpulkan kedudukan slidenya $\overline{O_1A}$.
  \item Hitunglah $\gamma_1$, $\gamma_2$ dan perbesaran totalnya.
  \item Berikan lebar gambarnya, lalu bandingkan dengan lebar yang akan
        diberikan $L_1$ sendirian pada layar yang sama (\qty{12.08}{m}
        dari $L_1$).
  \item Untuk benda di tak hingga, tunjukkan bahwa sistemnya setara
        dengan satu \omterm{def:b1:mirrors-thin-lenses:lens}{lensa} berjarak fokus $f_{\mathrm{eq}}$ dengan
        \[
        \frac{1}{f_{\mathrm{eq}}} = \frac{1}{f_1'} + \frac{1}{f_2'} - \frac{e}{f_1' f_2'},
        \]
        $e = \overline{O_1O_2}$, dalam arti bahwa berkas yang miring
        sebesar $\alpha$ memfokus pada ketinggian
        $f_{\mathrm{eq}}\,\alpha$ dari sumbunya. Hitunglah
        $f_{\mathrm{eq}}$.
  \item Periksalah bahwa nisbah lebar gambarnya (dengan/tanpa pengubah)
        dekat dengan $f_1'/f_{\mathrm{eq}}$. Mengapa hanya hampir?
  \item Mengapa \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} antaranya harus terletak kurang dari
        $\abs{f_2'}$ di luar $L_2$ agar \omterm{def:b1:mirrors-thin-lenses:image}{bayangan} akhirnya nyata?
  \item Dengan lampu yang sama, dengan faktor berapa iradiansi layarnya
        berubah ketika pengubahnya dipasang (jarak layar yang sama)?
  \item Ringkaslah: pengubah itu mengalikan jarak fokusnya dengan faktor
        berapa, dan apa pengaruhnya terhadap ukuran serta kecerahan
        gambar pada lemparan tertentu?
\end{enumerate}
\end{problem}
