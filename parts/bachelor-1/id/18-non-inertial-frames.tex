\chapter{Kerangka Tak Inersial; Dinamika di Bumi}\label{ch:b1:non-inertial-frames}

Berdiri di dalam bus yang membelok ke kiri, Anda condong ke kanan,
terdorong oleh gaya yang tak dikerahkan siapa pun. Seember air yang
diputar di atas meja putar mencekungkan permukaannya menjadi sebuah
mangkuk. Di bawah kubah sebuah museum sebuah bandul berayun berjam-jam,
dan garis ayunannya merayap berputar perlahan, beberapa derajat tiap jam,
seolah-olah sebuah tangan tak terlihat memutarnya. Tak satu pun di
antaranya gaya baru: begitulah rupa \omterm{thm:b1:newton-dynamics:laws}{hukum Newton} ketika kerangka tempat
kita melukiskan geraknya sendiri sedang dipercepat atau berputar. Bab ini
memperlihatkan bagaimana kecepatan dan percepatan bertransformasi antar
kerangka, bagaimana hukum dinamikanya diperbaiki dengan menambahkan
\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{gaya inersial}}, dan apa yang dilakukan gaya-gaya itu pada kerangka
berputar yang paling penting --- Bumi.

\section{Berganti kerangka: kinematika}

\begin{definition}[Gerak relatif; kerangka bertranslasi, kerangka berotasi]\label{def:b1:non-inertial-frames:frames}
\index{gerak relatif}\index{kerangka bertranslasi}\index{kerangka berotasi}
Misalkan $\mathcal R$ sebuah kerangka (titik asal $O$) dan $\mathcal R'$ kerangka lain (titik asal
$O'$) yang bergerak terhadapnya. $\mathcal R'$ disebut \emph{bertranslasi} bila
sumbu-sumbunya menjaga arahnya tetap dalam $\mathcal R$ (setiap titik $\mathcal R'$
berkecepatan sama dengan $O'$); ia disebut \emph{berotasi terhadap sumbu tetap}
$\Delta$ bila $\Delta$ tetap dalam kedua kerangka dan $\mathcal R'$ berputar terhadapnya
dengan \omterm{cor:b1:point-kinematics:circular}{kecepatan sudut} $\Omega(t)$. Kecepatan dan percepatan sebuah
titik $M$ \emph{relatif} terhadap $\mathcal R'$, yaitu $\vect v'$ dan $\vect a'$, adalah
yang diukur seorang pengamat yang diam di $\mathcal R'$.
\end{definition}

\begin{theorem}[Penyusunan kecepatan dan percepatan]\label{thm:b1:non-inertial-frames:composition}
\index{kecepatan seret}\index{percepatan seret}\index{percepatan Coriolis}
\begin{itemize}
  \item $\mathcal R'$ bertranslasi dengan kecepatan $\vect v_{O'}$ dan
        percepatan $\vect a_{O'}$:
        \[
        \vect v = \vect v' + \vect v_{O'}, \qquad \vect a = \vect a' + \vect a_{O'} .
        \]
  \item $\mathcal R'$ berputar terhadap sumbu tetap $\Delta$ pada $\Omega$ (vektor
        $\vect\Omega = \Omega\vect e_z$ menyusuri sumbunya); $H$ proyeksi
        $M$ pada sumbunya:
        \[
        \vect v = \vect v' + \vect\Omega\wedge\vect{HM}, \qquad
        \vect a = \vect a' + \underbrace{\big(-\Omega^2\vect{HM} + \dot\Omega\,\vect e_z\wedge\vect{HM}\big)}_{\vect a_e}
        + \underbrace{2\vect\Omega\wedge\vect v'}_{\vect a_C} .
        \]
\end{itemize}
$\vect a_e$ adalah \emph{\omterm{thm:b1:non-inertial-frames:composition}{percepatan seret}} (percepatan titik
$\mathcal R'$ yang berimpit dengan $M$: sentripetal menuju sumbunya
bagi rotasi beraturan), $\vect a_C = 2\vect\Omega\wedge\vect v'$ adalah
\emph{\omterm{thm:b1:non-inertial-frames:composition}{percepatan Coriolis}}, yang hadir hanya bila $M$ bergerak relatif terhadap
kerangka berputarnya.
\end{theorem}

\begin{proof}
Translasi: $\vect{OM} = \vect{OO'} + \vect{O'M}$ dengan komponen
$\vect{O'M}$ dalam basis bersamanya; turunkan dua kali. Rotasi: pakailah
koordinat tabung terhadap $\Delta$ di $\mathcal R$, $(r, \theta, z)$, dan
di $\mathcal R'$, $(r, \theta', z)$ dengan $\theta = \theta' + \int\Omega\,\dd t$.
Lalu \cref{thm:b1:point-kinematics:cylindrical} di $\mathcal R$ dengan
$\dot\theta = \dot\theta' + \Omega$:
$\vect v = \dot r\vect e_r + r(\dot\theta' + \Omega)\vect e_\theta + \dot z\vect e_z
= \vect v' + r\Omega\vect e_\theta$, dan $r\Omega\vect e_\theta = \vect\Omega\wedge\vect{HM}$.
Untuk percepatannya, $\ddot\theta = \ddot\theta' + \dot\Omega$:
$\vect a = [\ddot r - r(\dot\theta' + \Omega)^2]\vect e_r + [r(\ddot\theta' + \dot\Omega)
+ 2\dot r(\dot\theta' + \Omega)]\vect e_\theta + \ddot z\vect e_z$; jabarkan lalu
kelompokkan, $\vect a = \vect a' - r\Omega^2\vect e_r + r\dot\Omega\vect e_\theta
+ 2\Omega(-r\dot\theta'\vect e_r + \dot r\vect e_\theta)$, dan kurung terakhirnya adalah
$2\vect\Omega\wedge\vect v'$ karena $\vect e_z\wedge\vect e_r = \vect e_\theta$,
$\vect e_z\wedge\vect e_\theta = -\vect e_r$.
\end{proof}

\begin{example}[Berjalan di atas meja putar]\label{ex:b1:non-inertial-frames:turntable}
\cref{exo:b1:point-kinematics:12} menghitung percepatan seseorang yang
berjalan secara radial dengan $u$ di atas meja putar yang berputar pada $\Omega$: $-ut\Omega^2\vect e_r
+ 2u\Omega\vect e_\theta$. Dalam bahasa yang baru: percepatan relatifnya nol,
seretnya $-r\Omega^2\vect e_r$, Coriolisnya $2\vect\Omega\wedge u\vect e_r = 2u\Omega\vect e_\theta$.
\end{example}

\section{Dinamika dalam kerangka tak inersial: gaya inersial}

\begin{theorem}[Hukum Newton dalam kerangka tak inersial]\label{thm:b1:non-inertial-frames:dynamics}
\index{gaya inersial}\index{gaya sentrifugal}\index{gaya Coriolis}
Dalam kerangka $\mathcal R'$ yang tak inersial, gerak sebuah partikel
menuruti
\[
m\vect a' = \sum\vect F + \vect F_e + \vect F_C, \qquad
\vect F_e = -m\vect a_e, \qquad \vect F_C = -m\vect a_C = -2m\vect\Omega\wedge\vect v' :
\]
Hukum kedua Newton berlaku asalkan pada gaya-gaya nyatanya ditambahkan
\emph{gaya seret} (bagi kerangka yang berputar beraturan, yaitu
\emph{\omterm{thm:b1:non-inertial-frames:dynamics}{gaya sentrifugal}} $+m\Omega^2\vect{HM}$, terarah menjauhi
sumbunya; bagi \omterm{def:b1:non-inertial-frames:frames}{kerangka bertranslasi}, $-m\vect a_{O'}$) dan \emph{\omterm{thm:b1:non-inertial-frames:dynamics}{gaya
Coriolis}}. \emph{\omterm{thm:b1:non-inertial-frames:dynamics}{Gaya inersial}} ini bukan interaksi: tak ada benda yang
mengerahkannya, ia tak mempunyai reaksi, dan ia lenyap dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka
inersial}.
\end{theorem}

\begin{proof}
$m\vect a = \sum\vect F$ dalam \omterm{thm:b1:newton-dynamics:laws}{kerangka inersial} $\mathcal R$; substitusikan
$\vect a = \vect a' + \vect a_e + \vect a_C$ lalu pindahkan kedua sukunya ke
ruas kanan.
\end{proof}

\begin{example}[Penumpang yang condong, unting-unting]\label{ex:b1:non-inertial-frames:bus}
Di dalam bus yang mengerem pada $a = \qty{3.0}{m/s^2}$, sebuah pegangan
gantung merasakan, dalam kerangka busnya, gravitasi ditambah \omterm{thm:b1:non-inertial-frames:dynamics}{gaya inersial}
ke depan $ma$: ia menggantung menyusuri \emph{gravitasi semu} $\vect g' = \vect g - \vect a_{O'}$, yang condong
ke depan sebesar $\arctan(a/g) = 17^\circ$. Penumpang yang ingin berdiri
diam harus mencondongkan badannya ke arah vertikal semu itu. Di dalam lift yang
jatuh bebas, $\vect g' = \vect 0$: ketakberbobotan, dilihat dari dalam.
\end{example}

\begin{example}[Ember yang berputar]\label{ex:b1:non-inertial-frames:bucket}
Air di dalam ember yang berputar pada $\Omega$ terhadap sumbu tegaknya diam
dalam kerangka berputarnya di bawah gravitasi dan \omterm{thm:b1:non-inertial-frames:dynamics}{gaya sentrifugal}: permukaan
bebasnya tegak lurus pada gravitasi semu $\vect g' = -g\vect e_z +
\Omega^2r\vect e_r$ (permukaan fluida yang diam tegak lurus pada medannya,
\cref{ch:b1:fluid-statics}). Kemiringannya $\dd z/\dd r = \Omega^2r/g$, sehingga
$z = \Omega^2r^2/2g$: sebuah paraboloid. Pada \qty{10}{rad/s} tepi ember
\qty{10}{cm} berdiri \qty{5}{cm} di atas pusatnya --- asas
teleskop bercermin cair, yang raksa berputarnya mengambil tepat
bentuk parabolik yang dituntut sebuah cermin.
\end{example}

\begin{method}[Bekerja dalam kerangka tak inersial]\label{met:b1:non-inertial-frames:method}
Pilihlah kerangka yang membuat pertanyaannya paling sederhana (busnya, meja
putarnya, Buminya); daftarlah gaya-gaya nyatanya; tambahkan $-m\vect a_e$ dan, bila
bendanya bergerak dalam kerangka itu, $-2m\vect\Omega\wedge\vect v'$; lalu lanjutkan seperti pada
\cref{met:b1:newton-dynamics:method}. Periksa: \omterm{thm:b1:non-inertial-frames:dynamics}{gaya inersial} tak pernah
muncul dalam neraca energinya sebagai ``usaha yang dilakukan seseorang'' --- \omterm{thm:b1:non-inertial-frames:dynamics}{gaya
Coriolis}, yang tegak lurus pada $\vect v'$, tak melakukan usaha; \omterm{thm:b1:non-inertial-frames:dynamics}{gaya sentrifugal}
diturunkan dari \omterm{def:b1:work-and-energy:conservative}{energi potensial} $-\tfrac12 m\Omega^2r^2$ dalam kerangka yang
berputar beraturan.
\end{method}

\section{Kerangka Bumi}

\begin{definition}[Kerangka geosentris dan kerangka Bumi]\label{def:b1:non-inertial-frames:earth}
\index{kerangka geosentris}\index{kerangka Bumi}
Yang disebut \emph{kerangka geosentris} berpusat di pusat Bumi dan
bersumbu menunjuk ke bintang-bintang tetap; ia inersial dengan hampiran
yang sangat baik bagi gerak sehari atau kurang (percepatannya sendiri
mengelilingi Matahari hanya \qty{6e-3}{m/s^2}, dan hampir seragam di seluruh Bumi).
Yang disebut \emph{kerangka Bumi}, yang melekat pada tanah, berputar
terhadapnya pada sumbu kutub dengan
\[
\Omega = \frac{2\pi}{\qty{86164}{s}} = \qty{7.29e-5}{rad/s} .
\]
\end{definition}

\begin{proposition}[Berat, dan perubahan $g$]\label{prop:b1:non-inertial-frames:weight}
\index{berat!dan rotasi Bumi}
Dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka Bumi} sebuah benda yang diam merasakan tarikan
gravitasi $m\vect{\mathcal G}$ dan \omterm{thm:b1:non-inertial-frames:dynamics}{gaya sentrifugal} $m\Omega^2\vect{HM}$; jumlah keduanya
adalah apa yang diukur sebuah neraca dan ditunjukkan garis unting-unting: \emph{berat}
$m\vect g = m\vect{\mathcal G} + m\Omega^2\vect{HM}$. Suku sentrifugalnya terbesar
di khatulistiwa, $\Omega^2R = \qty{0.034}{m/s^2}$, $0.35\%$ dari $g$; ia
mengurangi $g$ dari kutub ke khatulistiwa dan memiringkan vertikalnya menjauhi
pusat Bumi sampai $0.1^\circ$ pada lintang menengah.
\end{proposition}

\begin{proof}
Terapkan \cref{thm:b1:non-inertial-frames:dynamics} dengan $\vect v' = \vect 0$
(tak ada \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis}); $\Omega^2R = (7.29\times10^{-5})^2 \times 6.37\times10^6$.
Sudut simpangannya menyusul dari segitiga $m\vect{\mathcal G}$ dan \omterm{thm:b1:non-inertial-frames:dynamics}{gaya
sentrifugalnya} (\cref{exo:b1:non-inertial-frames:6}).
\end{proof}

\begin{proposition}[Gaya Coriolis di Bumi]\label{prop:b1:non-inertial-frames:coriolis}
Pada lintang $\lambda$, sebuah benda yang bergerak mendatar dengan kelajuan $v'$
merasakan \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis} mendatar yang besarnya
\[
F_C = 2m\Omega v'\sin\lambda ,
\]
terarah ke \emph{kanan} geraknya di belahan bumi utara dan
ke kiri di belahan selatan, ke mana pun arah perjalanannya; benda
yang jatuh tegak dibelokkan ke \emph{timur} sebesar $2m\Omega v'\cos\lambda$.
\end{proposition}

\begin{proof}
Uraikan $\vect\Omega = \Omega(\cos\lambda\,\vect e_N + \sin\lambda\,\vect e_z)$
(utara dan vertikal setempat). Untuk $\vect v'$ yang mendatar, $-2m\vect\Omega\wedge\vect v'$
mempunyai bagian mendatar $-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$ --- terputar
$90^\circ$ searah jarum jam dari $\vect v'$ dilihat dari atas, untuk $\lambda > 0$ ---
dan sebuah bagian tegak (dapat diabaikan terhadap gravitasi). Untuk $\vect v' =
-v'\vect e_z$ yang tegak: $-2m\Omega\cos\lambda\,\vect e_N\wedge(-v'\vect e_z) = 2m\Omega v'\cos\lambda\,
\vect e_E$, ke timur.
\end{proof}

\begin{example}[Gaya kecil, akibat besar]\label{ex:b1:non-inertial-frames:coriolisnum}
Sebuah kereta pada \qty{100}{km/h} di lintang $45^\circ$: $a_C = 2 \times 7.29\times
10^{-5} \times 27.8 \times 0.71 = \qty{2.9e-3}{m/s^2}$, tiga perseribu dari
$g$ --- rel kanan sebuah jalur searah aus sedikit lebih cepat. Udara
yang mengalir pada \qty{20}{m/s} menuju pusat bertekanan rendah dibelokkan ke kanan
sepanjang jalan masuknya dan berakhir mengitarinya berlawanan arah jarum jam: setiap
siklon di belahan bumi utara berputar begitu, setiap siklon selatan sebaliknya. Sebuah batu
yang dijatuhkan ke dalam sumur \qty{100}{m} di lintang $45^\circ$ mendarat \qty{1.6}{cm} di timur
garis unting-untingnya (\cref{exo:b1:non-inertial-frames:7}).
\end{example}

\begin{proposition}[Bandul Foucault]\label{prop:b1:non-inertial-frames:foucault}
\index{bandul Foucault}
Sebuah bandul yang berayun dengan \omterm{def:b1:wave-propagation:signal}{amplitudo} kecil di lintang $\lambda$ menjaga
bidang osilasinya tetap terhadap \omterm{def:b1:non-inertial-frames:earth}{kerangka geosentris}, kecuali rotasi
vertikal setempatnya: dilihat dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka Bumi}, bidang itu berputar
\emph{searah jarum jam} (dari atas, di belahan bumi utara) dengan laju sudut
$\Omega\sin\lambda$, yakni satu putaran penuh dalam
\[
T_F = \frac{\qty{23.93}{h}}{\sin\lambda} .
\]
\end{proposition}

\begin{proof}[Bukti sebagian]
Di kutub ($\lambda = 90^\circ$) alasannya geometris: tak ada gaya mendatar
yang bekerja pada bebannya selain tegangan talinya, yang komponen mendatarnya
menyusuri arah ayunannya, sehingga dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka geosentris} bidang ayunnya tetap
sementara Bumi berputar di bawahnya pada $\Omega$; dilihat dari tanah ia berputar
pada $-\Omega$. Di lintang $\lambda$, \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis} mendatarnya adalah
$-2m\Omega\sin\lambda\,\vect e_z\wedge\vect v'$: tepat apa yang akan dihasilkan kerangka yang berputar pada
$\Omega\sin\lambda$ terhadap vertikal setempatnya. Dalam kerangka yang ikut berputar
bersama bidangnya pada $-\Omega\sin\lambda$, gaya itu lenyap sampai orde pertama
dan bandulnya menjadi osilator bidang biasa; maka bidangnya
berpresesi pada $\Omega\sin\lambda$ dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka Bumi}. Perhitungan
lengkapnya, dengan suku orde kedua yang diabaikan, adalah soal akhir
pekannya.
\end{proof}

\begin{remark}[Pasang surut, dalam dua baris]\label{rem:b1:non-inertial-frames:tides}
\index{pasang surut}
Dalam kerangka pusat Bumi, yang jatuh bebas menuju Bulan,
tarikan Bulan tidak seragam: sedikit lebih kuat di sisi dekatnya,
lebih lemah di sisi jauhnya. Selisihnya, yang berorde $2GM_{\text{Bulan}}R/d^3
\approx \qty{1e-6}{m/s^2}$, meregangkan lautan menjadi dua tonjolan --- dua
kali pasang sehari --- dan sumbangan Matahari, $46\%$ dari sumbangan Bulan, membuat
pasang purnama dan pasang perbani ketika keduanya sejajar atau bersilang
(\cref{exo:b1:non-inertial-frames:12}).
\end{remark}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:1}
Sebuah unting-unting menggantung di dalam mobil yang dipercepat pada \qty{2.5}{m/s^2}: sudutnya
terhadap vertikal, dan tegangan talinya dalam satuan $mg$. Mobil yang sama
mengerem pada laju yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:2}
Seorang anak duduk \qty{3.0}{m} dari sumbu komidi putar yang berputar sekali
tiap \qty{4.0}{s}. Percepatan sentrifugal dalam kerangka berputarnya, sebagai
pecahan $g$; gaya nyata apa yang memasoknya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:3}
Hitunglah \omterm{cor:b1:point-kinematics:circular}{kecepatan sudut} Bumi $\Omega$, percepatan sentrifugal
di khatulistiwa, dan pecahannya terhadap $g$. Pada periode rotasi berapa
benda di khatulistiwa akan melayang lepas?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b1:non-inertial-frames:4}
Sebuah kereta \qty{400}{t} berjalan pada \qty{100}{km/h} di lintang $45^\circ$.
Percepatan Coriolisnya dan gaya menyampingnya pada rel; rel mana
yang terdorong bila menuju utara? bila menuju timur?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:5}
Sebuah ember berjari-jari \qty{10}{cm} berputar pada \qty{2.0}{rad/s}, lalu
\qty{10}{rad/s}. Kenaikan airnya di tepi relatif terhadap pusatnya.
Mengapa bak raksa yang berputar merupakan cermin teleskop yang baik?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:6}
Tunjukkan bahwa garis unting-unting di lintang $\lambda$ menyimpang dari pusat
Bumi kira-kira sebesar $\Omega^2R\sin\lambda\cos\lambda/g$ radian, dan hitunglah
pada $45^\circ$ dalam menit busur. Berapa lebih kecil $g$ di khatulistiwa
daripada di kutub \omterm{def:b1:optical-instruments:eye}{semata-mata} karena rotasinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:7}
Sebuah batu jatuh dari keadaan diam ke dalam sumur \qty{100}{m} di lintang $45^\circ$.
Perlakukan \omterm{thm:b1:non-inertial-frames:composition}{percepatan Coriolis} $2\Omega v\cos\lambda$ (ke timur) sebagai usikan
kecil dengan $v = gt$; integralkan dua kali untuk menemukan simpangan ke
timur di dasarnya. Mengapa ke timur?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:8}
Periode presesi sebuah \omterm{prop:b1:non-inertial-frames:foucault}{bandul Foucault} di Paris ($\lambda = 48.9^\circ$),
di kutub, dan di khatulistiwa. Sebesar sudut berapa bidangnya berputar
dalam satu jam di Paris?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b1:non-inertial-frames:9}
Udara mengalir pada \qty{20}{m/s} menuju pusat bertekanan rendah di $45^\circ$
utara. Percepatan Coriolisnya; ke arah mana udara itu akhirnya
mengitari pusatnya? Mengapa pusaran kecil (wastafel, tornado) tidak menuruti
aturan ini?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:10}
Sebuah balon helium bertali melayang di dalam sebuah mobil. Ketika mobilnya
dipercepat maju, ke arah mana balonnya condong? Jelaskan dengan
gravitasi semu dan gaya apung.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:11}
Sebuah stasiun luar angkasa berupa cincin berjari-jari \qty{100}{m} yang diputar untuk memberi $1g$
gravitasi buatan di tepinya. Berapa periode rotasinya? Seorang antariksawan berjalan
menyusuri tepinya pada \qty{1.5}{m/s}: percepatan Coriolisnya, dan pengaruhnya pada
berat semunya ketika berjalan searah dan berlawanan dengan rotasinya. Ia menjatuhkan
sebuah bola dari \qty{2.0}{m}: kira-kira berapa jauh dari kakinya bola itu mendarat?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b1:non-inertial-frames:12}
Tunjukkan bahwa selisih antara percepatan gravitasi Bulan
di pusat Bumi dan di titik permukaan yang terdekat ke Bulan
kira-kira $2GM_Mr/d^3$ ($r$ jari-jari Bumi, $d$ jarak Bumi--Bulan);
hitunglah, lalu hitung pula yang sama untuk Matahari; simpulkan nisbah \omterm{rem:b1:non-inertial-frames:tides}{pasang
surut} matahari terhadap \omterm{rem:b1:non-inertial-frames:tides}{pasang surut} bulan.
\end{exercise}

\section{Soal: Bandul Foucault}

\begin{problem}[{Soal akhir pekan --- sebuah beban 28 kilogram pada kawat
sepanjang 67 meter di bawah sebuah kubah: mengapa bidang ayunnya berputar,
seberapa cepat, apa yang dibuktikannya, dan sekecil apa gaya yang melakukannya}]
\label{pb:b1:non-inertial-frames:1}
Pada tahun 1851 sebuah bandul sepanjang $\ell = \qty{67}{m}$ dan bermassa $m = \qty{28}{kg}$
berayun di bawah kubah Pantheon di Paris, lintang $\lambda =
48.85^\circ$, dengan \omterm{def:b1:wave-propagation:signal}{amplitudo} kira-kira $x_0 = \qty{3.0}{m}$. $\Omega =
\qty{7.29e-5}{rad/s}$, $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{Bagian I --- Bandulnya sendiri.}
\begin{enumerate}
  \item Periode \omterm{prop:b1:work-and-energy:stability}{osilasi kecilnya}; periksalah bahwa \omterm{def:b1:wave-propagation:signal}{amplitudonya} kecil
        ($x_0/\ell$).
  \item Kelajuan maksimal bebannya, dan tinggi maksimalnya di atas
        titik terendah.
  \item Tegangan tali di titik terendah, dalam satuan $mg$.
  \item Taksirlah \omterm{prop:b1:newton-dynamics:drag}{gaya hambat} udara pada bebannya pada kelajuan maksimal (hukum
        kuadratik, $C_xS \approx \qty{0.05}{m^2}$, $\rho = \qty{1.2}{kg/m^3}$) lalu
        bandingkan dengan gaya pemulih maksimal $m\omega_0^2x_0$.
  \item Mengapa kawatnya panjang dan bebannya berat? (Dua alasan, satu tentang
        periodenya, satu tentang peredamannya.)
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Di kutub.}
Bayangkan bandul yang sama di kutub utara.
\begin{enumerate}[resume]
  \item Dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka geosentris}, daftarlah gaya-gaya pada bebannya dan
        tunjukkan bahwa tak ada yang dapat memutar bidang ayunnya.
  \item Simpulkan apa yang dilihat seorang pengamat yang berdiri di atas esnya
        terhadap bidang itu, dan dalam berapa lama ia kembali ke arah semula.
  \item Lukiskan \omterm{def:b1:point-kinematics:frame}{lintasan} bebannya di tanah (sebuah roset): berapa
        kelopak dalam sehari?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Di lintang $\lambda$: suku Coriolisnya.}
Pakailah sumbu setempat: $x$ ke timur, $y$ ke utara, $z$ ke atas; osilasinya kecil, sehingga
geraknya hampir mendatar, $\vect v' \approx (\dot x, \dot y, 0)$; gaya
pemulihnya $-m\omega_0^2(x, y, 0)$ dengan $\omega_0 = \sqrt{g/\ell}$.
\begin{enumerate}[resume]
  \item Tuliskan $\vect\Omega$ dalam sumbu-sumbu ini.
  \item Hitunglah $-2m\vect\Omega\wedge\vect v'$ dan pertahankan komponen
        mendatarnya; tunjukkan bahwa ia hanya melibatkan $\Omega\sin\lambda$.
  \item Tuliskan kedua persamaan geraknya untuk $x$ dan $y$; tetapkan
        $\omega_F = \Omega\sin\lambda$ lalu bandingkan $\omega_F$ dengan $\omega_0$.
  \item Perkenalkan $u = x + jy$ dan tunjukkan bahwa $\ddot u + 2j\omega_F\dot u +
        \omega_0^2u = 0$.
  \item Cobalah $u = \eu^{j\alpha t}$: temukan kedua nilai $\alpha$, dan
        tunjukkan bahwa untuk $\omega_F \ll \omega_0$ keduanya $\alpha \approx \pm\omega_0
        - \omega_F$.
  \item Simpulkan bahwa $u \approx \eu^{-j\omega_Ft}(A\eu^{j\omega_0t} + B\eu^{-j\omega_0t})$:
        tafsirkan faktor $\eu^{-j\omega_Ft}$ --- ke arah mana, dengan laju
        berapa, bidang ayunnya berputar?
  \item Hitunglah periode presesinya di Paris dan sudut yang ditempuhnya dalam
        satu jam.
  \item Mengapa bidangnya tak berputar di khatulistiwa?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sekecil apa gayanya.}
\begin{enumerate}[resume]
  \item Hitunglah \omterm{thm:b1:non-inertial-frames:dynamics}{gaya Coriolis} maksimal pada bebannya lalu bandingkan
        dengan beratnya dan dengan gaya pemulih maksimalnya.
  \item Sejauh mana titik ujung ayunannya bergeser ke samping selama
        setengah periode? (Bidangnya berputar sebesar $\omega_FT/2$.)
  \item Pada tahun 1851 bebannya membawa sebuah jarum yang merobohkan pasak kecil
        yang diletakkan melingkar pada \omterm{def:b1:wave-propagation:signal}{amplitudonya}: kira-kira berapa lama sebelum
        pasak berikutnya jatuh?
  \item Bandulnya diluncurkan dengan membakar seutas benang yang menahan bebannya
        ke samping. Mengapa tidak didorong dengan tangan saja?
  \item Hambatan udara menghentikan sebuah bandul setelah beberapa jam. Bandul
        museum modern memakai elektromagnet kecil di bawahnya untuk memulihkan
        energinya tanpa menyentuh bidang ayunnya: mengapa tendangan itu harus
        benar-benar radial?
\end{enumerate}

\medskip
\noindent\textbf{Bagian V --- Apa yang dibuktikannya, dan apa lagi yang dilakukannya.}
\begin{enumerate}[resume]
  \item Peragaan Foucault meyakinkan khalayak akan rotasi Bumi.
        Apa persisnya yang diukur laju presesinya, dan
        dapatkah percobaannya menentukan lintang?
  \item Pada lintang berapa bidangnya akan berputar sekali tepat dalam \qty{48}{h}?
  \item Sebuah giroskop (roda yang berputar cepat dalam gimbal) menjaga sumbunya
        tetap dalam \omterm{def:b1:non-inertial-frames:earth}{kerangka geosentris}. Jelaskan dalam satu kalimat mengapa ia pun
        menyingkapkan rotasinya, dan sebutkan sebuah penerapan.
  \item Ringkaslah: satu \omterm{thm:b1:non-inertial-frames:dynamics}{gaya inersial} yang bertanggung jawab, kebergantungannya
        pada lintang, dan periode yang diberikannya di Paris.
\end{enumerate}
\end{problem}
