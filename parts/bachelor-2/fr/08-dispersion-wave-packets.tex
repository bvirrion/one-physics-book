\chapter{Dispersion et paquets d'ondes}\label{ch:b2:dispersion-wave-packets}

Jetez une pierre dans un étang et observez l'anneau de rides~: de nouvelles crêtes
naissent à son bord intérieur, courent vers l'extérieur à travers l'anneau et meurent à son
bord extérieur --- l'anneau tout entier avance moins vite que les crêtes qu'il contient.
Asseyez-vous sur une plage après une tempête lointaine et la houle longue et lente arrive
un jour avant le clapot court~: la mer trie les vagues par longueur d'onde. Le
premier câble télégraphique transatlantique, en 1858, transformait des points et des
traits nets en une bouillie qu'il fallait des minutes pour lire. Dans ces trois cas la
vitesse d'une onde dépend de sa fréquence~: le milieu est
\emph{dispersif}. Ce chapitre introduit la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion}, le
\omterm{def:b2:dispersion-wave-packets:complex}{nombre d'onde complexe} qui porte à la fois la propagation et l'absorption, la
\omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe} à laquelle un paquet --- et son énergie, et
l'information qu'il porte --- se déplace réellement, et la ligne le long de laquelle la télégraphie,
la télévision et l'internet ont voyagé~: le \omterm{prop:b2:dispersion-wave-packets:coax}{câble coaxial}.

\section{La relation de dispersion}

\begin{definition}[Relation de dispersion~; vitesse de phase]\label{def:b2:dispersion-wave-packets:relation}
\index{relation de dispersion}\index{vitesse de phase}\index{milieu dispersif}
Un milieu linéaire, homogène et invariant dans le temps admet les ondes planes
monochromatiques $\underline s = A\,\eu^{\iu(\omega t - kx)}$ (notation
complexe~; le signal physique en est la partie réelle) pourvu que $\omega$ et $k$
vérifient la \emph{relation de dispersion} du milieu, $\mathcal D(\omega, k)
= 0$, résolue en $k(\omega)$ ou en $\omega(k)$. Pour $k$ réel l'onde se propage
sans se déformer à la \emph{vitesse de phase}
\[
v_\varphi = \frac\omega k .
\]
Le milieu est \emph{non dispersif} lorsque $v_\varphi$ ne dépend pas de
$\omega$ (alors $\omega = ck$ et tout signal se propage sans déformation --- c'est
l'\omterm{thm:b2:waves-on-strings:equation}{équation de d'Alembert}) et \emph{dispersif} sinon.
\end{definition}

\begin{proof}
En reportant $\eu^{\iu(\omega t - kx)}$ dans une équation linéaire à coefficients
constants, chaque $\partial_t$ devient $\iu\omega$ et chaque $\partial_x$ devient
$-\iu k$, ce qui laisse une relation algébrique. La corde, l'onde sonore et
la barre ont donné $\omega^2 = c^2k^2$.
\end{proof}

\begin{example}[Dispersif et non dispersif]\label{ex:b2:dispersion-wave-packets:examples}
\index{équation de Klein--Gordon}\index{ondes de gravité en eau profonde}
(i) Une corde sur un lit élastique (force de rappel $-Ky$ par unité de longueur,
ou une chaîne de pendules), $\mu\partial_t^2y = T\partial_x^2y - Ky$~: $\omega^2 = \omega_c^2
+ c^2k^2$ avec $\omega_c = \sqrt{K/\mu}$ --- la relation de \emph{Klein--Gordon},
la même que celle d'un plasma et d'un guide d'ondes
(\cref{ch:b2:waves-in-media,ch:b2:guided-waves})~: $v_\varphi = c/\sqrt{1 -
\omega_c^2/\omega^2} > c$, et aucun $k$ réel en dessous de la \emph{pulsation de coupure} $\omega_c$.
(ii) \omterm{ex:b2:dispersion-wave-packets:examples}{Ondes de gravité en eau profonde}~: $\omega^2 = gk$ (admis), $v_\varphi =
\sqrt{g/k} = \sqrt{g\lambda/2\pi}$~: les grandes longueurs d'onde vont plus vite --- \qty{12.5}{m/s} pour
\qty{100}{m}, \qty{40}{m/s} pour \qty{1}{km}. (iii) La \omterm{prop:b2:waves-on-strings:chain}{chaîne d'atomes}, $\omega =
2\omega_0|\sin(ka/2)|$ (\cref{ch:b2:waves-on-strings}). (iv) La lumière dans le verre,
$k = n(\omega)\omega/c$, d'où le spectre étalé par un prisme.
\end{example}

\begin{definition}[Nombre d'onde complexe~: propagation et atténuation]\label{def:b2:dispersion-wave-packets:complex}
\index{nombre d'onde complexe}\index{atténuation}\index{onde évanescente}\index{épaisseur de peau}
Lorsque la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion} donne, pour $\omega$ réel, un $\underline k = k' - \iu k''$
complexe, l'onde s'écrit
\[
\underline s = A\,\eu^{-k''x}\,\eu^{\iu(\omega t - k'x)} :
\]
elle se propage à $v_\varphi = \omega/k'$ et son amplitude décroît en
$\eu^{-x/\delta}$ avec la \emph{longueur d'atténuation} $\delta = 1/k''$ (elle doit
décroître dans son sens de propagation~: $k'$ et $k''$ de même signe).
Si $\underline k$ est imaginaire pur, $\underline k = -\iu\kappa$, l'onde est
\emph{évanescente}~: $A\,\eu^{-\kappa x}\eu^{\iu\omega t}$, une oscillation stationnaire
dont l'amplitude s'éteint sur la distance $1/\kappa$ sans rien
propager --- le cas du milieu de Klein--Gordon en dessous de sa coupure,
$\kappa = \sqrt{\omega_c^2 - \omega^2}/c$, et d'un métal aux fréquences optiques.
\end{definition}

\begin{example}[Une corde amortie]\label{ex:b2:dispersion-wave-packets:damped}
Une corde dans un fluide visqueux, $\mu\partial_t^2y + \alpha\partial_ty = T\partial_x^2y$~:
$\underline k^2 = (\omega^2 - \iu\alpha\omega/\mu)/c^2$. Pour un amortissement faible ($\alpha \ll \mu
\omega$), $\underline k \approx \dfrac\omega c\Bigl(1 - \iu\dfrac{\alpha}{2\mu\omega}\Bigr)$~: l'onde
garde sa célérité et perd de l'amplitude sur $\delta = 2\mu c/\alpha = 2Z/\alpha$,
indépendante de la fréquence --- une perte \emph{dissipative} et non dispersive,
de \qty{8.7}{dB} par longueur $\delta$.
\end{example}

\section{Paquets d'ondes et vitesse de groupe}

\begin{proposition}[Battements~; vitesse de groupe]\label{prop:b2:dispersion-wave-packets:group}
\index{vitesse de groupe}\index{paquet d'ondes}\index{battements}
Deux ondes de même amplitude et de fréquences voisines $\omega \pm
\delta\omega$, de nombres d'onde $k \pm \delta k$, se superposent en
\[
s = 2A\cos(\delta\omega\,t - \delta k\,x)\cos(\omega t - kx) :
\]
une porteuse en $(\omega, k)$ se déplaçant à $v_\varphi = \omega/k$, modulée par une lente
enveloppe se déplaçant à $\delta\omega/\delta k$. Un \emph{\omterm{prop:b2:dispersion-wave-packets:group}{paquet d'ondes}} --- une
superposition d'ondes monochromatiques dont les nombres d'onde occupent une bande étroite
autour de $k_0$ --- se déplace, dans son ensemble, à la \emph{\omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe}}
\[
v_g = \frac{\dd\omega}{\dd k}\Bigr|_{k_0} ,
\]
qui est la vitesse de son énergie et de l'information qu'il porte.
Dans un milieu non dispersif $v_g = v_\varphi = c$~; sinon le paquet
se déforme en chemin (il \emph{s'étale}), d'autant plus vite que
$\dd^2\omega/\dd k^2$ est grand et que sa largeur spatiale, complément de son spectre,
est petite.
\end{proposition}

\begin{proof}
La formule de transformation en produit donne les battements. Pour un paquet, écrivons $s =
\int a(k)\eu^{\iu(\omega(k)t - kx)}\dd k$ (une superposition de Fourier --- l'intégrale
de Fourier est étudiée dans le volume de mathématiques de troisième année~; seule est utilisée ici
son interprétation comme somme continue de sinusoïdes), avec
$a(k)$ piqué en $k_0$. Développons $\omega(k) \approx \omega_0 + v_g(k - k_0)$~: alors
$s \approx \eu^{\iu(\omega_0t - k_0x)}\int a(k)\eu^{\iu(k - k_0)(v_gt - x)}\dd k$, une
porteuse multipliée par une enveloppe qui n'est fonction que de $x - v_gt$ --- l'enveloppe
se déplace à $v_g$. Le terme suivant, $\tfrac12\omega''(k_0)(k - k_0)^2t$,
déphase les composantes au cours du temps et étale l'enveloppe. L'énergie
d'un paquet se trouve là où est son enveloppe~; tout signal aussi.
\end{proof}

\begin{example}[Eau profonde~: les crêtes devancent le groupe]\label{ex:b2:dispersion-wave-packets:deepwater}
$\omega = \sqrt{gk}$~: $v_g = \tfrac12\sqrt{g/k} = \tfrac12v_\varphi$. Un train de
houle avance à la moitié de la vitesse de ses crêtes~: en observant un train de vagues,
on voit des crêtes apparaître à l'arrière du groupe, le traverser et
disparaître à l'avant --- exactement l'anneau de l'étang. Une tempête à \qty{3000}{km}
envoie sa houle de \qty{15}{s} ($\lambda = gT^2/2\pi = \qty{350}{m}$, $v_g =
gT/4\pi = \qty{11.7}{m/s}$) en trois jours, ses vagues de \qty{8}{s} ($v_g =
\qty{6.2}{m/s}$) en cinq jours et demi~: à partir des heures d'arrivée des
différentes périodes, les océanographes localisent la tempête.
\end{example}

\begin{example}[Klein--Gordon~: plus vite que la lumière, et non]\label{ex:b2:dispersion-wave-packets:superluminal}
Pour $\omega^2 = \omega_c^2 + c^2k^2$~: $v_\varphi = \omega/k$ et $v_g = c^2k/\omega$, donc
$v_\varphi v_g = c^2$~: la \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} dépasse $c$ (dans l'ionosphère, les crêtes
d'une onde à \qty{10}{MHz} vont plus vite que la lumière) mais la \omterm{prop:b2:dispersion-wave-packets:group}{vitesse
de groupe}, qui porte le signal, reste en dessous. Une \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase}
ne transporte ni énergie ni information --- les crêtes sont comme la
tache d'un faisceau de phare balayant un nuage lointain.
\end{example}

\begin{remark}[Étalement]\label{rem:b2:dispersion-wave-packets:spreading}
Un paquet de largeur spatiale $\Delta x$ contient des nombres d'onde sur $\Delta k \sim
1/\Delta x$ (la réciprocité de Fourier, admise)~; les \omterm{prop:b2:dispersion-wave-packets:group}{vitesses de groupe} de ses
composantes diffèrent de $\Delta v_g \approx |\omega''|\Delta k$, si bien qu'au bout d'un temps $t$ il
s'est étalé de $|\omega''|\Delta k\,t$~: il double de largeur au bout de $t_{\text{ét}}
\sim \Delta x^2/|\omega''|$. Un paquet de dix vagues de \qty{100}{m} en eau profonde
($\Delta x \approx \qty{1}{km}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-50}{m^2/s}$)
double en six heures environ~; une impulsion lumineuse de \qty{1}{ns} dans une
fibre optique s'étale de quelques dizaines de picosecondes par kilomètre --- la
limite du débit des liaisons longues. Le \omterm{prop:b2:dispersion-wave-packets:group}{paquet d'ondes} quantique du
\cref{ch:b2:schrodinger-wave-functions} s'étale pour la même raison.
\end{remark}

\section{Le câble coaxial}

\begin{proposition}[Équations des télégraphistes~; ligne sans pertes]\label{prop:b2:dispersion-wave-packets:coax}
\index{câble coaxial}\index{équations des télégraphistes}\index{impédance caractéristique}\index{ligne de transmission}
Un \omterm{prop:b2:dispersion-wave-packets:coax}{câble coaxial} (ou toute ligne bifilaire) possède, par unité de longueur, une
inductance $\Lambda$ et une capacité $\Gamma$. La tension $v(x, t)$ entre
les conducteurs et le courant $i(x, t)$ dans le conducteur intérieur obéissent aux
\emph{\omterm{prop:b2:dispersion-wave-packets:coax}{équations des télégraphistes}}
\[
\frac{\partial v}{\partial x} = -\Lambda\frac{\partial i}{\partial t} , \qquad
\frac{\partial i}{\partial x} = -\Gamma\frac{\partial v}{\partial t} ,
\]
d'où l'\omterm{thm:b2:waves-on-strings:equation}{équation de d'Alembert} pour $v$ et $i$ avec
\[
c = \frac1{\sqrt{\Lambda\Gamma}} , \qquad
v = Z_c\,i \ \text{pour une onde vers les } +x, \quad Z_c = \sqrt{\frac\Lambda\Gamma} ,
\]
$Z_c$ étant l'\emph{\omterm{prop:b2:dispersion-wave-packets:coax}{impédance caractéristique}} de la ligne. Pour un
\omterm{prop:b2:dispersion-wave-packets:coax}{câble coaxial} de rayons $a < b$ rempli d'un diélectrique de permittivité
relative $\varepsilon_r$, $\Lambda = \dfrac{\mu_0}{2\pi}\ln\dfrac ba$, $\Gamma =
\dfrac{2\pi\varepsilon_0\varepsilon_r}{\ln(b/a)}$, de sorte que $c = c_0/\sqrt{\varepsilon_r}$ (environ
\qty{2e8}{m/s}, les deux tiers de la vitesse de la lumière) et $Z_c = \dfrac{60\,\Omega}
{\sqrt{\varepsilon_r}}\ln\dfrac ba$~: \qty{50}{\ohm} ou \qty{75}{\ohm} pour les câbles usuels.
\end{proposition}

\begin{proof}
Modélisons la tranche $[x, x + \dd x]$ par une inductance série $\Lambda\dd x$ et une
capacité parallèle $\Gamma\dd x$~: la chute de tension le long de la tranche vaut
$\Lambda\dd x\,\partial_ti$, et le courant perdu dans la capacité vaut $\Gamma\dd x
\,\partial_tv$. On dérive croisé. Pour $v = f(x - ct)$, la première équation
donne $i = f(x - ct)/\Lambda c = v/\sqrt{\Lambda/\Gamma}$. Les valeurs de $\Lambda$ et de
$\Gamma$ sont celles du condensateur cylindrique et de l'inductance
coaxiale calculées dans le volume de première année.
\end{proof}

\begin{proposition}[Ligne avec pertes~; condition de non-distorsion]\label{prop:b2:dispersion-wave-packets:lossy}
\index{condition de Heaviside}\index{loi des carrés de Kelvin}
Avec une résistance série $R$ et une conductance parallèle $G$ par unité de
longueur, les équations deviennent $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi =
-\Gamma\partial_tv - Gv$, et la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion} $\underline k^2 = (\Lambda\omega -
\iu R)(\Gamma\omega - \iu G)$~: la ligne est dispersive et atténuante. Deux
cas limites~: (i) $R/\Lambda = G/\Gamma$ (\omterm{prop:b2:dispersion-wave-packets:lossy}{condition de Heaviside}, atteinte en ajoutant
de l'inductance)~: $\underline k = \omega\sqrt{\Lambda\Gamma} - \iu\sqrt{RG}$, toutes les fréquences
se propagent à la même vitesse avec la même \omterm{def:b2:dispersion-wave-packets:complex}{atténuation} --- la ligne
\emph{atténue} mais ne \emph{déforme} pas. (ii) $\Lambda \approx 0$, $G \approx 0$
(les premiers câbles sous-marins)~: $\partial_x^2v = R\Gamma\,\partial_tv$, une
équation de \emph{diffusion} --- une impulsion ne se propage pas mais s'étale,
atteignant la distance $L$ au bout d'un temps $\sim R\Gamma L^2$ (\omterm{prop:b2:dispersion-wave-packets:lossy}{loi des carrés
de Kelvin}), cent fois plus long pour un câble dix fois plus long.
\end{proposition}

\begin{proof}
Reportons $\eu^{\iu(\omega t - kx)}$~: $-\iu kv = -(\iu\omega\Lambda + R)i$, $-\iu ki = -(\iu
\omega\Gamma + G)v$~; on multiplie. Sous la \omterm{prop:b2:dispersion-wave-packets:lossy}{condition de Heaviside} le produit est un
carré parfait. La limite diffusive laisse tomber les termes $\iu\omega\Lambda$ et $G$.
\end{proof}

\begin{example}[Trois câbles]\label{ex:b2:dispersion-wave-packets:cables}
Le câble transatlantique de 1858~: un unique fil de cuivre sous gutta-percha,
$R \approx \qty{3}{\ohm/km}$, $\Gamma \approx \qty{0.3}{\micro F/km}$, $L = \qty{3000}{km}$~:
$R\Gamma = \qty{9e-10}{s/m^2}$ et $R\Gamma L^2 \approx \qty{8}{s}$ par point~: un mot à la
minute, et les tentatives des opérateurs pour forcer le signal
avec \qty{2}{kV} détruisirent l'isolant en quelques semaines. Une ligne téléphonique
chargée de bobines tous les \qty{2}{km} pour satisfaire la \omterm{prop:b2:dispersion-wave-packets:lossy}{condition de Heaviside}
(1900)~: une parole claire sur des centaines de kilomètres. Un \omterm{prop:b2:dispersion-wave-packets:coax}{câble coaxial}
moderne de \qty{50}{\ohm}~: $c = \qty{2e8}{m/s}$, $\Lambda = \qty{0.25}{\micro H/m}$, $\Gamma =
\qty{100}{pF/m}$, et une \omterm{def:b2:dispersion-wave-packets:complex}{atténuation} due à l'effet de peau dans les
conducteurs (\cref{ch:b2:waves-in-media}), croissant comme $\sqrt f$~: quelques
\omterm{def:b2:sound-waves:decibel}{décibels} par centaine de mètres à \qty{100}{MHz}.
\end{example}

\begin{method}[Lire une relation de dispersion]\label{met:b2:dispersion-wave-packets:method}
(1) Écrire l'\omterm{thm:b2:waves-on-strings:equation}{équation d'onde} du milieu en notation complexe et
résoudre en $\underline k(\omega)$. (2) $k$ réel~: calculer $v_\varphi = \omega/k$ et
$v_g = \dd\omega/\dd k$ (ou $1/(\dd k/\dd\omega)$)~; comparer~: dispersif ou non,
dispersion normale ($v_g < v_\varphi$) ou anomale. (3) $k$ complexe~: séparer
propagation ($k'$) et \omterm{def:b2:dispersion-wave-packets:complex}{atténuation} ($k''$, longueur $1/k''$)~; imaginaire
pur signifie évanescent, sans transport. (4) Une pulsation de coupure
sépare une bande propagative d'une bande évanescente. (5) Pour un signal,
raisonner sur l'enveloppe à $v_g$, jamais sur les crêtes.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:1}
Ondes en eau profonde, $\omega^2 = gk$. (a) \omterm{def:b2:dispersion-wave-packets:relation}{Vitesse de phase}, \omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe} et
période d'une houle de \qty{100}{m}. (b) Une houle de période \qty{14}{s}~:
longueur d'onde, $v_\varphi$, $v_g$~; temps mis par son énergie pour parcourir \qty{5000}{km}.
(c) Pourquoi le clapot de période \qty{2}{s} n'y parvient-il jamais~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:2}
Un milieu obéit à $\omega^2 = \omega_c^2 + c^2k^2$ avec $\omega_c = 2\pi \times \qty{1e7}{rad/s}$
et $c = \qty{3e8}{m/s}$. Pour $f = \qty{20}{MHz}$~: $k$, $v_\varphi$, $v_g$, et
leur produit. Pour $f = \qty{5}{MHz}$~: $\kappa$ et la profondeur de pénétration.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:3}
Deux diapasons à \qty{440}{Hz} et \qty{444}{Hz}~: fréquence de battement, et
période des variations d'intensité sonore. Deux vagues de longueurs d'onde
\qty{100}{m} et \qty{110}{m}~: fréquences, longueur d'onde et vitesse de
l'enveloppe~; comparer à $v_g$ à \qty{105}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:4}
Un \omterm{prop:b2:dispersion-wave-packets:coax}{câble coaxial}~: conducteur intérieur de \qty{0.9}{mm} de diamètre, extérieur
\qty{3.0}{mm}, polyéthylène ($\varepsilon_r = 2.3$). $\Lambda$, $\Gamma$, célérité,
\omterm{prop:b2:dispersion-wave-packets:coax}{impédance caractéristique}, retard par mètre~; longueur de câble équivalente
à un retard de \qty{10}{ns}. Même câble avec de l'air à la place du polyéthylène.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:5}
\emph{La corde raide.} Une corde de piano réelle obéit à $\mu\partial_t^2y = T\partial_x^2y
- EI\,\partial_x^4y$ ($EI$ étant la raideur de flexion). (a) \omterm{def:b2:dispersion-wave-packets:relation}{Relation de dispersion},
\omterm{def:b2:dispersion-wave-packets:relation}{vitesses de phase} et de groupe. (b) Pour la corde de la$_4$ du
\cref{ch:b2:waves-on-strings} ($T = \qty{685}{N}$, $\mu = \qty{6.1}{g/m}$, $EI =
\qty{9.8e-3}{N.m^2}$), augmentation relative de $v_\varphi$ pour la huitième
harmonique ($k = 8\pi/L$, $L = \qty{0.38}{m}$)~; vérifier avec l'inharmonicité
qui y a été trouvée. (c) La dispersion est-elle normale ou anomale~? À quoi
ressemble un pincement net après quelques allers-retours~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:6}
\emph{Corde amortie.} $\mu\partial_t^2y + \alpha\partial_ty = T\partial_x^2y$. (a) Établir
la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion}. (b) Pour $\alpha \ll \mu\omega$, montrer que $k' \approx \omega/c$ et
$k'' \approx \alpha/2Z$~; \omterm{def:b2:dispersion-wave-packets:complex}{atténuation} en \omterm{def:b2:sound-waves:decibel}{décibels} par mètre. (c) Applications numériques~: $\mu =
\qty{5}{g/m}$, $T = \qty{50}{N}$, $\alpha = \qty{0.02}{kg.m^{-1}.s^{-1}}$, $f =
\qty{100}{Hz}$~: vérifier l'approximation et donner $\delta$. (d) La corde amortie
est-elle dispersive~? Dans quel régime de $\alpha$ le deviendrait-elle~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:7}
\emph{Rides.} Les vagues courtes sont gouvernées par la \omterm{rem:b2:viscous-flows:capillarity}{tension superficielle}~: $\omega^2 =
\gamma k^3/\rho$ ($\gamma = \qty{0.072}{N/m}$)~; la relation complète est $\omega^2 = gk +
\gamma k^3/\rho$. (a) $v_\varphi$ et $v_g$ pour des ondes capillaires pures~; laquelle est
la plus grande~? (b) Montrer que la \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} de la relation complète est
minimale en $\lambda_m = 2\pi\sqrt{\gamma/\rho g}$ et calculer $\lambda_m$ et $v_{\min}$.
(c) Un bateau plus lent que $v_{\min}$ ne fait pas de sillage~: pourquoi~? (d) Les gouttes de pluie sur un
étang font des anneaux dont les crêtes sont en retard sur l'anneau~: dans quel régime~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:8}
\emph{Étalement.} Un paquet de largeur initiale $\Delta x_0$ double de largeur
au bout de $t_{\text{ét}} \approx \Delta x_0^2/|\omega''(k_0)|$. (a) Un train de vagues en eau
profonde, $\lambda = \qty{50}{m}$, $\Delta x_0 = \qty{500}{m}$. (b) Une impulsion lumineuse dans
une fibre de verre~: $\omega'' \approx \qty{-0.16}{m^2/s}$~; une impulsion de \qty{10}{ps} ($\Delta x_0 =
\qty{2}{mm}$)~: temps d'étalement et distance parcourue pendant ce temps~; idem
pour une impulsion de \qty{1}{ns}. Pourquoi les longues liaisons optiques utilisent-elles des impulsions plus longues ou
une fibre à compensation de dispersion~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:9}
\emph{Le câble sous-marin.} $R = \qty{3}{\ohm/km}$, $\Gamma = \qty{0.3}{\micro F/km}$,
$\Lambda$ et $G$ négligeables. (a) Montrer que $v$ obéit à une équation de diffusion et
donner sa diffusivité. (b) \omterm{def:b2:dispersion-wave-packets:relation}{Relation de dispersion} $\underline k(\omega)$~; \omterm{def:b2:dispersion-wave-packets:relation}{vitesse
de phase} et longueur d'\omterm{def:b2:dispersion-wave-packets:complex}{atténuation} à \qty{1}{Hz} et à \qty{10}{Hz}. (c)
Temps mis par un signal pour «~arriver~» sur \qty{3000}{km} ($\sim R\Gamma L^2$)~; mots par
minute. (d) Le remède de Heaviside~: l'inductance par kilomètre nécessaire pour
$R/\Lambda = G/\Gamma$ si $G = \qty{1e-7}{S/km}$~; pourquoi était-elle difficile à ajouter~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:10}
\emph{L'énergie voyage à la \omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe}.} Corde sur un lit élastique~:
$\mu\partial_t^2y = T\partial_x^2y - Ky$, onde $y = A\cos(\omega t - kx)$ avec $\omega^2 =
\omega_c^2 + c^2k^2$. (a) Densité d'énergie $e = \tfrac12\mu(\partial_ty)^2 + \tfrac12T
(\partial_xy)^2 + \tfrac12Ky^2$ et flux $\mathcal P = -T\partial_xy\,\partial_ty$~: vérifier
$\partial_te + \partial_x\mathcal P = 0$. (b) Moyennes temporelles $\langle e\rangle$ et
$\langle\mathcal P\rangle$ pour l'onde. (c) Montrer que $\langle\mathcal P\rangle/
\langle e\rangle = c^2k/\omega = v_g$, et non $v_\varphi$. (d) Qu'advient-il du
flux d'énergie en dessous de la coupure~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:11}
\emph{Houle et tsunami.} Les vagues de longueur d'onde $\lambda$ sur une profondeur $h$
obéissent à $\omega^2 = gk\tanh kh$ (admis). (a) Retrouver les limites d'eau profonde et
d'eau peu profonde ($kh \ll 1$)~; célérité des ondes en eau peu profonde~; sont-elles
dispersives~? (b) Un tsunami de $\lambda = \qty{200}{km}$ sur \qty{4}{km} d'océan~:
quelle limite, célérité, période~; temps pour traverser \qty{8000}{km}. (c) Son
amplitude au large vaut \qty{0.5}{m}~: énergie par unité de longueur de crête
(admettre $E = \tfrac12\rho gA^2$ par unité d'aire) pour un front de \qty{1000}{km}~; que
se passe-t-il quand la profondeur tombe à \qty{10}{m}, si le flux d'énergie $Ev_g$ se
conserve (loi de Green, $A \propto h^{-1/4}$)~? (d) Une houle de \qty{100}{m}
arrivant sur une plage~: à quelle profondeur commence-t-elle à sentir le fond,
et pourquoi ses crêtes deviennent-elles parallèles au rivage~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:12}
\emph{Une impulsion sur une ligne.} Un câble de \qty{50}{\ohm} et de longueur $\ell = \qty{100}{m}$
($c = \qty{2e8}{m/s}$) est alimenté en $x = 0$ par un générateur de résistance
interne \qty{50}{\ohm} et fermé en $x = \ell$ sur une charge $R_L$. (a) Montrer
qu'une onde arrivant à l'extrémité est réfléchie avec le coefficient
en tension $r = (R_L - Z_c)/(R_L + Z_c)$ (écrire $v$ et $i$ comme somme d'une onde incidente
et d'une onde réfléchie et imposer la loi d'Ohm dans la charge). (b) Valeurs pour
$R_L = 50$, $0$, $\infty$, \qty{150}{\ohm}. (c) On envoie une impulsion de \qty{1}{ns} et \qty{1}{V}~:
ce que montre l'oscilloscope placé au générateur, et à quel instant, pour chaque
charge~; pourquoi les \qty{50}{\ohm} du générateur comptent. (d) Un défaut dans un câble
enterré renvoie un écho après \qty{1.8}{\micro s}~: où se trouve-t-il~? (Réflectométrie
temporelle.)
\end{exercise}

\section{Problème~: le câble et la mer}

\begin{problem}[{Problème du week-end --- la dispersion à l'œuvre~: le câble
télégraphique qui brouillait les points, la houle qui annonce une tempête, le
tsunami qui traverse un océan, et le paquet qui s'étale}]
\label{pb:b2:dispersion-wave-packets:1}

\medskip
\noindent\textbf{Partie I --- Le câble télégraphique.}
Le câble transatlantique de 1866~: $R = \qty{2.0}{\ohm/km}$, $\Gamma = \qty{0.25}{\micro F/km}$,
$\Lambda = \qty{0.5}{mH/km}$, $G$ négligeable, longueur $L = \qty{3500}{km}$.
\begin{enumerate}
  \item Écrire les \omterm{prop:b2:dispersion-wave-packets:coax}{équations des télégraphistes} avec $R$ et en déduire la
        \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion} $\underline k^2 = \Gamma\omega(\Lambda\omega - \iu R)$.
  \item À quelle fréquence $\Lambda\omega$ égale-t-il $R$~? En conclure que pour
        les fréquences télégraphiques (quelques hertz) la ligne est en
        régime diffusif.
  \item Dans ce régime, $\underline k = \sqrt{-\iu R\Gamma\omega}$~: donner $k'$ et $k''$
        (rappel~: $\sqrt{-\iu} = (1 - \iu)/\sqrt2$)~; \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} et
        longueur d'\omterm{def:b2:dispersion-wave-packets:complex}{atténuation} à \qty{1}{Hz}~; à \qty{4}{Hz}.
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Atténuation} en \omterm{def:b2:sound-waves:decibel}{décibels} sur tout le câble à \qty{1}{Hz} et
        à \qty{4}{Hz}. Quelle fréquence survit~? Qu'advient-il alors d'un
        point net~?
  \item Estimation de Kelvin du temps de transit, $R\Gamma L^2$~; débit pratique
        du câble en mots par minute si un mot compte environ
        $10$ points et traits et que les points ne doivent pas se recouvrir.
  \item Heaviside proposait d'augmenter $\Lambda$~: quelle valeur rend $\Lambda\omega = R$
        à \qty{4}{Hz}, et que vaudraient alors la vitesse et l'\omterm{def:b2:dispersion-wave-packets:complex}{atténuation}
        (prendre $G = 0$, de sorte que la condition $R/\Lambda = G/\Gamma$ ne peut être
        satisfaite exactement --- utiliser la relation générale à cette fréquence)~?
  \item Une fibre optique moderne transporte \qty{1e10}{bit/s}~: comparer à la
        question 5, en puissances de dix.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La houle.}
Eau profonde~: $\omega^2 = gk$.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Vitesses de phase} et de groupe en fonction de la période $T$.
  \item Une tempête à \qty{4000}{km} d'une côte engendre des vagues de périodes
        allant de \qty{6}{s} à \qty{18}{s}~: heure d'arrivée de chaque extrême~;
        durée de l'épisode de houle.
  \item Montrer que sur une côte située à la distance $D$ la période reçue obéit à
        $1/T = gt/4\pi D$, $t$ étant compté depuis la tempête~: $1/T$ croît
        linéairement dans le temps. Une bouée enregistre $T = \qty{16}{s}$ un jour à midi
        et $T = \qty{12}{s}$ exactement \qty{24}{h} plus tard~: distance de la
        tempête.
  \item L'énergie par unité d'aire d'une houle d'amplitude $A$ vaut $\tfrac12\rho
        gA^2$~; puissance traversant une ligne de \qty{1}{m} de crête, pour $A =
        \qty{1.5}{m}$, $T = \qty{12}{s}$. (Utiliser la \omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe} --- pourquoi~?)
  \item Une ferme houlomotrice de \qty{1}{km} de front, de rendement $30\%$~:
        puissance électrique~; comparer à une éolienne (\qty{3}{MW}).
  \item Un bateau à \qty{5}{m/s} en eau profonde laisse derrière lui un sillage
        stationnaire~: quelle longueur d'onde a une \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} égale à
        la vitesse du bateau et le suit donc~? Pourquoi le motif traîne-t-il en
        V, plus étroit que ne le laisserait croire l'étalement propre des crêtes
        (penser à $v_g = v_\varphi/2$)~?
  \item En observant un train de houle depuis une falaise, décrire ce que fait une
        crête particulière pendant que le groupe passe devant une
        bouée fixe~; combien de crêtes contient un groupe de durée
        \qty{2}{min} à $T = \qty{12}{s}$, et combien de «~vies~» de crête
        cela représente-t-il~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le tsunami.}
$\omega^2 = gk\tanh kh$~; profondeur du Pacifique $h = \qty{4.0}{km}$.
\begin{enumerate}[resume]
  \item Un séisme soulève une portion de fond marin de \qty{100}{km} de
        \qty{2}{m}~; l'onde a $\lambda \approx \qty{200}{km}$~: vérifier que c'est une
        onde d'eau peu profonde et donner sa célérité, sa période et le temps pour
        atteindre une côte à \qty{6000}{km}.
  \item Est-elle dispersive~? Comparer à la houle de la partie II~: arrive-t-elle
        avant ou après la houle~? Laquelle arrive comme
        une impulsion unique~?
  \item Sa hauteur au large vaut \qty{0.6}{m}~: énergie par mètre carré~;
        énergie totale d'un front long de \qty{500}{km} étalé sur une
        longueur d'onde~; comparer à une explosion de \qty{1}{Mt} (\qty{4.2e15}{J}).
  \item Loi de Green~: hauteur lorsque la profondeur tombe à \qty{20}{m} près de la
        côte (flux d'énergie $\tfrac12\rho gA^2v_g$ conservé, $v_g = \sqrt{gh}$)~;
        célérité et longueur d'onde en ce point~; pourquoi l'onde se redresse alors et
        déferle comme un mur.
  \item Un navire au large ne remarque pas le passage du tsunami~: pourquoi~?
  \item Des marégraphes distants de \qty{6000}{km} enregistrent l'onde~: ce que donne
        le temps de trajet, et pourquoi la célérité mesurée est légèrement
        inférieure à $\sqrt{gh}$ sur un océan de profondeur variable.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le paquet qui s'étale.}
\begin{enumerate}[resume]
  \item Pour l'eau profonde, calculer $\omega''(k) = \dd^2\omega/\dd k^2$ et sa valeur
        pour $\lambda = \qty{100}{m}$.
  \item Un paquet de \qty{10}{} vagues de ce type ($\Delta x_0 \approx \qty{1}{km}$) est
        lâché~: temps d'étalement $\Delta x_0^2/|\omega''|$~; distance parcourue
        par le paquet pendant ce temps.
  \item Expliquer avec des mots pourquoi un paquet de spectre plus étroit
        (train plus long) s'étale plus lentement, et pourquoi le tsunami de la
        partie III ne s'étale presque pas.
  \item La même estimation pour un électron de longueur d'onde \qty{0.1}{nm}
        localisé sur \qty{1}{nm}, avec $\omega = \hbar k^2/2m$ ($\hbar =
        \qty{1.05e-34}{J.s}$, $m = \qty{9.1e-31}{kg}$)~: temps d'étalement et
        distance parcourue --- le premier contact avec le \omterm{prop:b2:dispersion-wave-packets:group}{paquet d'ondes}
        quantique du \cref{ch:b2:schrodinger-wave-functions}.
  \item Récapituler en un tableau~: pour le câble, la houle, le tsunami et
        l'électron, la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion}, $v_\varphi$, $v_g$ et
        le fait que le milieu étale ou se contente d'atténuer.
\end{enumerate}
\end{problem}
