\chapter{L'équation de Schrödinger et les fonctions d'onde}\label{ch:b2:schrodinger-wave-functions}

Envoyez des électrons un par un sur une paire de fentes et notez où chacun
atterrit~: un point ici, un point là, apparemment au hasard --- et après dix
mille points, les franges de l'expérience d'Young, comme si chaque électron
était passé par les deux fentes et avait interféré avec lui-même. Le dernier
chapitre du volume de première année a rencontré cette étrangeté~: la lumière vient par
photons, la matière a une longueur d'onde $\lambda = h/p$, et position et
quantité de mouvement ne peuvent être nettes toutes deux. Ce qu'il n'a pas donné, c'est
l'\emph{équation} --- la règle qui dit comment évolue l'onde associée à une
particule, comme la loi de Newton dit comment évolue une \omterm{def:b2:fluid-kinematics:euler}{trajectoire}. Cette
équation est celle de Schrödinger (1926), et avec elle la description quantique
devient une théorie calculable~: on écrit une \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{fonction d'onde}}
$\psi(x,t)$, dont le module carré est la probabilité de trouver la
particule, et on résout une équation aux dérivées partielles linéaire. Ce
chapitre énonce l'équation, explique ce que $\psi$ signifie et ce qu'elle ne
signifie pas, trouve ses \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} et les \omterm{prop:b2:dispersion-wave-packets:group}{paquets d'ondes} de la particule
libre, et montre où la mécanique de Newton réapparaît --- pour
tout ce qui est plus lourd qu'une molécule.

\section{Ce que le volume de première année nous a laissé}

\begin{proposition}[Les particules sont des ondes]\label{prop:b2:schrodinger-wave-functions:recap}
Un photon de fréquence $\nu$ a l'énergie $E = h\nu = \hbar\omega$ et la
quantité de mouvement $p = h/\lambda = \hbar k$~; une particule matérielle de quantité de mouvement $p$
se comporte comme une onde de longueur d'onde $\lambda = h/p$ (de Broglie), avec $E =
\hbar\omega$ et $p = \hbar k$ également --- la \omterm{def:b2:diffraction:huygens}{diffraction} des électrons par les
cristaux, l'interférométrie de neutrons et même de molécules le confirment~; et aucun
état ne peut avoir une position et une quantité de mouvement plus nettes que $\Delta x\,\Delta p
\ge \hbar/2$ (Heisenberg). Avec $\hbar = h/2\pi = \qty{1.05e-34}{J.s}$~: un
électron à \qty{100}{eV} a $\lambda = \qty{0.12}{nm}$, la taille d'un
atome~; un grain de poussière de \qty{1e-15}{kg} à \qty{1}{mm/s} a $\lambda =
\qty{7e-16}{m}$, plus petit qu'un noyau --- il se comporte classiquement.
\end{proposition}

\begin{proof}
Rappelé du volume de première année~; la tâche est maintenant la dynamique de
l'onde.
\end{proof}

\section{La fonction d'onde}

\begin{definition}[Fonction d'onde et règle de Born]\label{def:b2:schrodinger-wave-functions:psi}
L'état d'une particule se déplaçant selon $x$ est décrit par une
\emph{fonction d'onde}\index{fonction d'onde} complexe $\psi(x,t)$ telle que
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
soit la probabilité de trouver la particule entre $x$ et $x + \dd x$
à l'instant $t$ (\emph{règle de Born}\index{règle de Born})~; d'où la
\emph{normalisation} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, et les
valeurs moyennes $\langle x\rangle = \int x|\psi|^2\dd x$, $\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$. Les fonctions d'onde obéissent au
\emph{principe de superposition}\index{principe de superposition}~: si
$\psi_1$ et $\psi_2$ sont des états possibles, $\alpha\psi_1 +
\beta\psi_2$ l'est aussi (une fois normalisé), et les probabilités contiennent alors le
terme d'interférence $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$. Un
facteur de phase global $\eu^{\iu\theta}$ ne change rien~; une phase relative
entre deux composantes superposées change l'interférence.
\end{definition}

\begin{remark}[Ce que $\psi$ n'est pas]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ n'est pas un champ classique étalé dans l'espace comme $\vect E$, ni un
nuage de matière~: l'électron est toujours trouvé entier, en un point.
$\psi$ est une \emph{amplitude de probabilité}, l'objet qui
interfère~; seul $|\psi|^2$ s'observe, et seulement statistiquement, en
répétant l'expérience sur des particules préparées à l'identique. Les points
des \omterm{prop:b2:two-wave-interference:young}{fentes d'Young} sont des électrons uniques~; les franges sont $|\psi_1 +
\psi_2|^2$.
\end{remark}

\section{L'équation de Schrödinger}

\begin{theorem}[Équation de Schrödinger]\label{thm:b2:schrodinger-wave-functions:equation}
Une particule de masse $m$ dans l'énergie potentielle $V(x)$ a une \omterm{def:b2:schrodinger-wave-functions:psi}{fonction
d'onde} qui obéit à
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
Elle est linéaire (superposition), du premier ordre en temps (la \omterm{def:b2:schrodinger-wave-functions:psi}{fonction d'onde}
à un instant la détermine à tout instant ultérieur), et complexe ($\psi$
ne peut être réelle~: le $\iu$ est essentiel). C'est un \emph{postulat},
justifié par ses conséquences.
\end{theorem}

\begin{proof}
Une motivation plutôt qu'une démonstration. Une particule libre de quantité de mouvement nette
$p = \hbar k$ et d'énergie $E = \hbar\omega$ devrait être une onde plane $\psi =
A\eu^{\iu(kx - \omega t)}$, pour laquelle $\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ et $-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$.
La relation classique $E = p^2/2m$ est ainsi reproduite par l'équation
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$, et la généralisation
naturelle pour $E = p^2/2m + V$ ajoute $V\psi$. Remarquons que
l'équation doit être linéaire pour valoir sur des superpositions d'ondes planes, et
du premier ordre en temps avec un $\iu$ pour que $\eu^{-\iu\omega t}$ et non
$\cos\omega t$ soit la dépendance temporelle --- une \omterm{thm:b2:waves-on-strings:equation}{équation d'onde} réelle
donnerait $E^2 \propto p^4$.
\end{proof}

\begin{proposition}[Conservation de la probabilité]\label{prop:b2:schrodinger-wave-functions:current}
La densité de probabilité $\rho = |\psi|^2$ et le \emph{courant de
probabilité}\index{courant de probabilité}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
obéissent à $\partial_t\rho + \partial_xj = 0$~: la probabilité n'est ni créée ni
détruite, et $\int|\psi|^2\dd x$ reste égal à $1$. Pour une onde plane
$A\eu^{\iu kx}$, $j = |A|^2\hbar k/m = \rho v$~: densité fois vitesse, comme pour
tout écoulement.
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$~; à partir de
l'équation et de sa conjuguée, $\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$, $\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$~; les termes en $V$ se compensent et le reste vaut $(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$.
\end{proof}

\begin{proposition}[Quantité de mouvement, valeurs moyennes et limite classique]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
La quantité de mouvement d'un état s'obtient par l'opérateur $\hat p =
-\iu\hbar\,\partial_x$ (qui donne $\hbar k$ sur une onde plane)~: $\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, et l'énergie par $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$. Les valeurs moyennes obéissent
(Ehrenfest) à
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
le centre du \omterm{prop:b2:dispersion-wave-packets:group}{paquet d'ondes} suit exactement la loi de Newton quand $V$ est
au plus quadratique, et approximativement dès que le paquet est étroit
devant l'échelle sur laquelle $V$ varie --- la limite classique,
valable pour tout objet macroscopique parce que $\hbar$ est si petit.
\end{proposition}

\begin{proof}
Dériver $\langle x\rangle = \int x|\psi|^2$, utiliser $\partial_t\rho =
-\partial_xj$ et intégrer par parties~: $\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$. La seconde relation est le même calcul un
ordre plus haut (admis ici).
\end{proof}

\section{États stationnaires}

\begin{theorem}[États stationnaires]\label{thm:b2:schrodinger-wave-functions:stationary}
Si $V$ ne dépend pas du temps, l'équation a des solutions de la
forme séparée
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(l'\emph{équation de Schrödinger indépendante du temps}\index{équation de Schrödinger!indépendante du temps})~: les \emph{états stationnaires}\index{état stationnaire}, d'énergie bien définie $E$, dont la densité de probabilité
$|\varphi|^2$ ne bouge pas. Les énergies $E$ pour lesquelles $\varphi$ est
acceptable (bornée, normalisable pour un état lié) sont les
\emph{niveaux d'énergie}~; la solution générale est une superposition $\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$, et une superposition de deux niveaux
a une densité qui oscille à la \emph{fréquence de Bohr}\index{fréquence de Bohr} $\nu_{21} = (E_2 - E_1)/h$ --- la fréquence de la lumière que
l'atome émet ou absorbe.
\end{theorem}

\begin{proof}
Insérons $\varphi(x)f(t)$~: $\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
une fonction de $t$ égale à une fonction de $x$, donc une constante $E$~; $f
= \eu^{-\iu Et/\hbar}$. Pour deux niveaux, $|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$.
La complétude des \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} est admise.
\end{proof}

\section{La particule libre~: ondes planes et paquets d'ondes}

\begin{proposition}[Particule libre]\label{prop:b2:schrodinger-wave-functions:free}
Pour $V = 0$ les \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} sont les ondes planes $\eu^{\iu(kx -
\omega t)}$ avec
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{g}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
les ondes de matière sont \emph{dispersives}~; la \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} n'a pas de sens
physique, la \emph{\omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe}} est la vitesse de la particule. Une onde
plane n'est pas normalisable --- une particule de quantité de mouvement parfaitement nette est
étalée sur tout l'espace --- de sorte qu'une particule réelle est un \emph{paquet
d'ondes}\index{paquet d'ondes}, une superposition d'ondes planes sur une bande
$\Delta k$, localisée sur $\Delta x \sim 1/\Delta k$, se déplaçant à $v_{\text{g}}$ et
s'\emph{étalant}~: un paquet gaussien de largeur initiale $\sigma_0$ a à
l'instant $t$ la largeur
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
et vérifie $\Delta x\,\Delta p = \hbar/2$ en $t = 0$ (le minimum permis)
et davantage ensuite.
\end{proposition}

\begin{proof}
La \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion} découle de l'équation~; $v_{\text{g}}$ comme au
\cref{ch:b2:dispersion-wave-packets}. La formule d'étalement vient de
la synthèse de Fourier du paquet --- chaque composante $\eu^{\iu kx}$ prend
la phase $\eu^{-\iu\hbar k^2t/2m}$, et l'intégrale gaussienne à
coefficient complexe donne une gaussienne de largeur croissante~; l'intégrale
de Fourier est l'outil du volume de mathématiques de troisième année et nous en admettons le
résultat. Physiquement~: le paquet contient des quantités de mouvement étalées sur $\Delta p
= \hbar/2\sigma_0$, donc des vitesses étalées sur $\Delta v = \hbar/2m\sigma_0$,
et après $t$ les composantes rapides ont distancé les lentes de $\Delta v\,t$
--- la formule dit exactement cela.
\end{proof}

\begin{example}[Qui s'étale]\label{ex:b2:schrodinger-wave-functions:spreading}
Un électron localisé dans $\sigma_0 = \qty{1}{nm}$~: $\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- après une nanoseconde son paquet est large de \qty{60}{\micro m}~:
un électron laissé seul ne reste pas en place~; lié dans un atome il
ne s'étale pas parce que le potentiel le retient (un \omterm{thm:b2:schrodinger-wave-functions:stationary}{état stationnaire}).
Un grain de poussière de \qty{1e-15}{kg} localisé à \qty{1}{\micro m}~: $\tau =
\qty{2e7}{s}$, huit mois~; une balle de fusil~: plus que l'âge de l'univers.
L'étalement quantique est réel, et invisible pour tout ce qu'on peut voir.
\end{example}

\begin{remark}[Énergie et temps]\label{rem:b2:schrodinger-wave-functions:energytime}
Un état qui n'existe que pendant $\Delta t$ --- un atome excité de durée de vie
$\tau$, un \omterm{prop:b2:scalar-light-model:coherence}{train d'ondes} de longueur $c\tau$ --- est une superposition d'énergies
étalées sur $\Delta E \sim \hbar/\Delta t$~: la relation de Fourier du
\cref{ch:b2:scalar-light-model} entre la durée d'un train et sa
\omterm{prop:b2:scalar-light-model:coherence}{largeur spectrale}, appliquée maintenant à $E = \hbar\omega$. Un niveau atomique de
durée de vie \qty{10}{ns} a une largeur $\Delta E = \qty{7e-8}{eV}$, une largeur
naturelle de raie de \qty{16}{MHz}~; la largeur Doppler du \cref{ch:b2:laser}
(\qty{1.5}{GHz}) la masque dans un gaz, mais non dans un atome froid ou un ion au
repos.
\end{remark}

\begin{method}[Travailler avec les fonctions d'onde]\label{meth:b2:schrodinger-wave-functions:method}
(1) Normaliser~; les probabilités sont $\int|\psi|^2$ sur la région. (2)
$V$ indépendant du temps~: chercher $\varphi(x)\eu^{-\iu Et/\hbar}$, résoudre
l'équation stationnaire, imposer les conditions aux limites --- les niveaux.
(3) Superposer des \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} pour l'évolution~; deux niveaux battent à
$(E_2 - E_1)/h$. (4) Mouvement libre~: $v_{\text{g}} = p/m$, temps d'étalement
$2m\sigma_0^2/\hbar$. (5) Courants~: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$~;
pour $A\eu^{\iu kx} + B\eu^{-\iu kx}$, $j = (|A|^2 - |B|^2)\hbar k/m$ (pas de terme croisé).
(6) Limite classique~: Ehrenfest, et $\lambda$ face à la taille du
dispositif.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:1}
Longueurs d'onde de de Broglie~: un électron à \qty{100}{eV}, \qty{10}{keV},
\qty{100}{keV} (utiliser $p = \sqrt{2mE}$, puis dire pourquoi la dernière demande une
correction)~; un neutron thermique (\qty{25}{meV}, $m = \qty{1.67e-27}{kg}$)~; une
molécule de C$_{60}$ ($m = \qty{1.2e-24}{kg}$) à \qty{200}{m/s}~; une balle de
tennis. Lesquels montrent des interférences~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:2}
$\psi(x) = A\eu^{-x^2/2a^2}$~: $A$ (utiliser $\int\eu^{-u^2}\dd u = \sqrt\pi$)~;
$\langle x\rangle$, $\Delta x$~; probabilité de trouver la particule dans
$|x| < a$ ($\operatorname{erf}(1) = 0.84$) et dans $|x| < 2a$ ($0.9995$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:3}
Vérifier que $\varphi(x)\eu^{-\iu Et/\hbar}$ résout l'équation quand $\varphi$
résout la stationnaire. Fréquence du phaseur pour $E = \qty{1}{eV}$~;
pour une superposition de deux niveaux distants de \qty{2}{eV}, fréquence et
longueur d'onde de l'oscillation de $|\psi|^2$ --- et de la lumière
émise.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:4}
Temps d'étalement $\tau = 2m\sigma_0^2/\hbar$ et largeur après \qty{1}{ns} et
\qty{1}{s}~: un électron avec $\sigma_0 = \qty{1}{nm}$ et $\sigma_0 = \qty{1}{mm}$~;
un proton avec $\sigma_0 = \qty{1}{nm}$~; un grain de \qty{1e-12}{kg} avec
$\sigma_0 = \qty{1}{\micro m}$. Pourquoi les objets restent-ils là où on les pose~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:5}
\emph{Courant.} (a) Établir l'\omterm{thm:b2:fluid-kinematics:continuity}{équation de continuité}. (b) $j$ pour
$A\eu^{\iu kx}$, pour $A\sin kx$, pour $A\eu^{\iu kx} + B\eu^{-\iu kx}$. (c) Pourquoi
l'absence de terme croisé dans le dernier cas revient-elle à dire que
les flux réfléchi et incident se soustraient simplement~? (d) Montrer que pour une \omterm{def:b2:schrodinger-wave-functions:psi}{fonction
d'onde} réelle $j = 0$~: un \omterm{thm:b2:schrodinger-wave-functions:stationary}{état stationnaire} lié ne porte aucun courant.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:6}
\emph{Un paquet gaussien.} $\psi(x,0) = A\eu^{\iu k_0x}\eu^{-x^2/4\sigma^2}$. (a)
$A$, $\langle x\rangle$, $\Delta x$. (b) $\langle p\rangle$ avec $\hat p =
-\iu\hbar\partial_x$. (c) En admettant $\Delta p = \hbar/2\sigma$, vérifier
$\Delta x\Delta p = \hbar/2$. (d) Démontrer $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$ à partir de l'\omterm{thm:b2:fluid-kinematics:continuity}{équation de continuité}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:7}
\emph{Des électrons dans un tube.} Des électrons accélérés sous \qty{100}{V}
traversent \qty{30}{cm}. (a) Vitesse, longueur d'onde, temps de vol. (b) \omterm{def:b2:dispersion-wave-packets:relation}{Vitesses de phase} et
de groupe~; laquelle est celle de l'électron~? (c) Un paquet initialement
large de \qty{10}{\micro m}~: temps d'étalement, largeur à l'arrivée. (d) \omterm{def:b2:diffraction:huygens}{Diffraction}
par une ouverture de \qty{0.3}{mm} en chemin~: étalement angulaire et tache sur
l'écran. L'image des rayons est-elle valable~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:8}
\emph{Phases.} (a) Montrer que $\psi$ et $\eu^{\iu\theta}\psi$ donnent les mêmes
prédictions. (b) Pour $\psi = (\psi_1 + \eu^{\iu\alpha}\psi_2)/\sqrt2$, écrire
$|\psi|^2$ et montrer que $\alpha$ compte. (c) Dans les \omterm{prop:b2:two-wave-interference:young}{fentes d'Young}, que
sont $\psi_1$ et $\psi_2$, et qu'est-ce qui fixe leur phase relative en un point
de l'écran~? (d) Pourquoi détecter par quelle fente l'électron est passé
détruit-il les franges (qualitativement)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:9}
\emph{Ehrenfest.} (a) Pour $V = \tfrac12m\omega^2x^2$, montrer que $\langle
x\rangle$ obéit exactement à $\ddot{\langle x\rangle} = -\omega^2\langle x\rangle$.
(b) Pour $V = mgx$, $\langle x\rangle$ tombe comme une pierre. (c) Pour un
$V$ général, quand a-t-on $\langle V'(x)\rangle \approx V'(\langle x\rangle)$~? (d) Une
bille de \qty{1}{g} dans une cuvette~: estimer la taille de paquet qu'il lui faudrait
pour être «~quantique~» et comparer à un atome.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:10}
\emph{Ballottement.} Dans une boîte $0 < x < L$ à parois impénétrables les
\omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} sont $\varphi_n = \sqrt{2/L}\sin(n\pi x/L)$, $E_n =
n^2h^2/8mL^2$ (\cref{ch:b2:potential-wells-tunneling}). Prenons $\psi =
(\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/\sqrt2$. (a)
$|\psi|^2$ en fonction du temps. (b) Montrer $\langle x\rangle = L/2 -
(16L/9\pi^2)\cos\omega_{21}t$ (utiliser $\int_0^L x\sin(\pi x/L)\sin(2\pi x/L)\dd x
= -8L^2/9\pi^2$). (c) Électron, $L = \qty{1}{nm}$~: $E_2 - E_1$, la
fréquence et la longueur d'onde de l'oscillation. (d) Que fait une charge
oscillante (\cref{ch:b2:dipole-radiation})~? Conclure sur les
raies spectrales.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{L'étalement, établi.} Le paquet $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
est la superposition $\int g(k)\eu^{\iu kx}\dd k$ avec $g(k) \propto
\eu^{-\sigma_0^2k^2}$ (admis~: la transformée de Fourier d'une gaussienne est une
gaussienne, de largeur $1/2\sigma_0$ en $k$). (a) Écrire $\psi(x,t)$ en donnant à chaque
composante sa phase $\eu^{-\iu\hbar k^2t/2m}$. (b) Compléter le carré dans
l'exposant et admettre $\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$ pour $\alpha$ complexe de partie réelle positive~: montrer que
$|\psi|^2$ est une gaussienne de largeur $\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$. (c) Interpréter $\tau$ par $\Delta v = \Delta p/m$. (d) Pourquoi
un paquet de photons dans le vide ne s'étale-t-il pas~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{Énergie--temps.} (a) Un état décroissant comme $\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$
(probabilité $\eu^{-t/\tau}$) est une superposition d'énergies~: en admettant
que sa distribution en énergie est une lorentzienne de largeur totale $\Gamma =
\hbar/\tau$, donner $\Gamma$ pour $\tau = \qty{10}{ns}$ (atome), \qty{1e-8}{s},
\qty{1e-23}{s} (une résonance hadronique, en MeV). (b) Largeur naturelle du
niveau de \qty{10}{ns} en fréquence~; comparer à la largeur Doppler
\qty{1.5}{GHz}. (c) Comment voit-on la largeur naturelle (atomes froids,
ions piégés)~? (d) La relier à la \omterm{prop:b2:scalar-light-model:coherence}{longueur de cohérence} du \omterm{prop:b2:scalar-light-model:coherence}{train d'ondes}
émis.
\end{exercise}

\section{Problème~: l'électron dans un tube et l'étalement d'un paquet d'ondes}

\begin{problem}[{Problème du week-end --- quand un électron est-il une particule,
quand une onde}]\label{pb:b2:schrodinger-wave-functions:1}
Données~: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $m_{\text{p}} = \qty{1.67e-27}{kg}$.

\textbf{Partie I --- Le tube cathodique.}
Des électrons quittent une cathode chaude avec une énergie négligeable, sont accélérés
sous $U = \qty{20}{kV}$, passent une ouverture de \qty{0.3}{mm}, et frappent un
écran à \qty{40}{cm}.
\begin{enumerate}
  \item Vitesse et quantité de mouvement (non relativistes)~; longueur d'onde de de Broglie.
  \item L'énergie cinétique vaut $4\%$ de $mc^2$~: le traitement non relativiste
        est-il acceptable pour ce problème~?
  \item Temps de vol jusqu'à l'écran.
  \item \omterm{def:b2:diffraction:huygens}{Diffraction} par l'ouverture~: étalement angulaire $\sim\lambda/a$ et
        tache produite sur l'écran~; comparer à un pixel
        (\qty{0.3}{mm}).
  \item Heisenberg à l'ouverture~: $\Delta p_y \gtrsim \hbar/2a$ donne le
        même étalement --- vérifier.
  \item Un paquet large de \qty{0.3}{mm} le long du faisceau~: temps d'étalement
        $2m\sigma_0^2/\hbar$ et largeur à l'écran.
  \item Des plaques de déviation appliquent une force transverse~: d'après Ehrenfest, comment
        $\langle y\rangle$ se déplace-t-il~? Conclure que le tube est un
        dispositif classique.
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Vitesse de phase} de l'onde électronique~; pourquoi n'est-ce pas un paradoxe
        qu'elle diffère de la vitesse de l'électron~?
\end{enumerate}

\textbf{Partie II --- \omterm{prop:b2:dispersion-wave-packets:group}{Paquets d'ondes}.}
\begin{enumerate}[resume]
  \item Pour un paquet gaussien de largeur $\sigma_0$~: $\Delta p$ et
        l'étalement en vitesse $\Delta v$.
  \item Temps d'étalement pour un électron avec $\sigma_0 = \qty{0.1}{nm}$,
        \qty{1}{nm}, \qty{1}{\micro m}~; pour un proton avec \qty{1}{nm}~; pour un
        virus (\qty{1e-17}{kg}) avec \qty{10}{nm}.
  \item Un électron dans un atome d'hydrogène est localisé à \qty{0.1}{nm}
        et ne s'étale pas~: pourquoi~?
  \item Un électron libre à \qty{1}{eV} avec $\sigma_0 = \qty{1}{nm}$~: quelle distance
        parcourt-il avant que sa largeur double~? Comparer à la
        longueur d'onde.
  \item Dans un métal, les électrons de conduction sont des ondes de $\lambda \approx
        \qty{0.5}{nm}$ sur tout le cristal~: en quel sens sont-ils
        «~délocalisés~»~?
  \item Un microscope électronique à \qty{100}{keV} a $\lambda \approx
        \qty{4}{pm}$, et pourtant il ne résout que \qty{0.1}{nm}~: pourquoi (ses lentilles
        magnétiques acceptent un demi-angle d'environ \qty{10}{mrad})~?
  \item Donner le critère qui décide si un objet doit être
        traité par une \omterm{def:b2:schrodinger-wave-functions:psi}{fonction d'onde} ou par Newton.
\end{enumerate}

\textbf{Partie III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{États stationnaires} et transitions.}
Un électron est confiné dans $0 < x < L = \qty{1}{nm}$~; $\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$.
\begin{enumerate}[resume]
  \item $E_1$, $E_2$, $E_3$ en électronvolts~; les \omterm{thm:b2:schrodinger-wave-functions:stationary}{fréquences de Bohr}
        $\nu_{21}$, $\nu_{31}$ et les longueurs d'onde.
  \item Vérifier que $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ est indépendant du temps~;
        où l'électron est-il le plus probable dans l'état fondamental~?
  \item Pour $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$~: $|\psi|^2$ en $t = 0$ et $t = T_{21}/2$~; esquisser.
  \item $\langle x\rangle(t)$ (avec $\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$)~:
        amplitude du ballottement.
  \item Le \omterm{thm:b2:dipole-radiation:field}{dipôle oscillant} $e\langle x\rangle$ rayonne~: à quelle
        fréquence, et que cela dit-il des raies spectrales et
        de la stabilité des \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires}~?
  \item Énergie de l'état $\psi$~: est-ce $E_1$, $E_2$, ou autre
        chose~? Que donne une mesure~?
\end{enumerate}

\textbf{Partie IV --- Interprétation.}
\begin{enumerate}[resume]
  \item Dans l'expérience des \omterm{prop:b2:two-wave-interference:young}{fentes d'Young} à un électron, qu'y a-t-il d'aléatoire
        et qu'y a-t-il qui ne l'est pas~?
  \item Écrire le \omterm{prop:b2:schrodinger-wave-functions:current}{courant de probabilité} de $A\eu^{\iu kx}$ et le relier à
        un faisceau de $n$ électrons par unité de longueur à la vitesse $v$.
  \item Pourquoi l'équation de Schrödinger doit-elle être complexe et du premier ordre
        en temps (deux raisons)~?
  \item Résumer en cinq lignes~: la \omterm{def:b2:schrodinger-wave-functions:psi}{fonction d'onde}, l'équation, les
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires}, le paquet, la limite classique.
\end{enumerate}
\end{problem}
