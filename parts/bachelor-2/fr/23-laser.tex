\chapter{Le laser~: émission stimulée et faisceaux gaussiens}\label{ch:b2:laser}

Un point rouge sur l'écran d'un amphithéâtre, le lecteur de codes-barres à la caisse,
le faisceau qui lit un disque, soude une caisse d'automobile, porte l'internet
sous les océans, corrige une cornée et mesure la distance à la
Lune au millimètre~: chacun d'eux est un \emph{\omterm{def:b2:laser:cavity}{laser}}, une source de
lumière comme aucune flamme ni aucune lampe. Sa lumière est d'une seule couleur à une partie
sur un milliard, reste en un faisceau de quelques millimètres à travers une pièce, et peut être
focalisée en une tache large de quelques longueurs d'onde, où son éclairement dépasse
un million de fois celui de la surface du Soleil. Rien de cela ne vient d'un
nouveau type d'atome~: cela vient d'un processus qu'Einstein a identifié en 1917
--- l'\emph{\omterm{def:b2:laser:processes}{émission stimulée}}, l'émission d'un photon identique à
un photon déjà présent --- et de l'art d'agencer la matière pour que ce processus
l'emporte sur l'\omterm{def:b2:laser:processes}{absorption}, puis de placer l'amplificateur entre deux miroirs
pour qu'il s'alimente lui-même. Ce chapitre construit le \omterm{def:b2:laser:cavity}{laser} en trois étapes~:
comment la lumière et une population d'atomes échangent de l'énergie (les \omterm{def:b2:laser:processes}{coefficients
d'Einstein}), quand l'échange amplifie (\omterm{prop:b2:laser:gain}{inversion de population} et
gain), et quand l'amplificateur devient un oscillateur (la cavité, son
seuil et ses modes). Il finit par la forme du faisceau qui en
sort, le \omterm{prop:b2:laser:gaussian}{faisceau gaussien}, dont le \omterm{prop:b2:laser:gaussian}{col} et la \omterm{def:b2:maxwell-equations:operators}{divergence} décident de ce qu'un \omterm{def:b2:laser:cavity}{laser}
peut faire à distance et au foyer.

\section{Lumière et matière~: les trois processus}

\begin{definition}[Absorption, émission spontanée et stimulée]\label{def:b2:laser:processes}
Considérons des atomes à deux niveaux d'énergie $E_1 < E_2$, de populations (densités
en nombre) $N_1$, $N_2$, et de la lumière de fréquence $\nu$ avec
$h\nu = E_2 - E_1$ et densité spectrale d'énergie $u(\nu)$ (énergie par unité de
volume et par unité de fréquence). Trois processus échangent de l'énergie~:
\begin{itemize}
\item l'\emph{absorption}\index{absorption (de la lumière)}~: un atome dans
      $E_1$ absorbe un photon et monte en $E_2$, au taux
      $B_{12}\,u(\nu)\,N_1$ par unité de volume et de temps~;
\item l'\emph{émission spontanée}\index{émission spontanée}~: un atome
      dans $E_2$ retombe en $E_1$ de lui-même, en émettant un photon dans une direction
      aléatoire avec une phase aléatoire, au taux $A_{21}N_2$ ---
      $1/A_{21} = \tau$ est la durée de vie du niveau~;
\item l'\emph{émission stimulée}\index{émission stimulée}~: un photon
      de fréquence $\nu$ passant près d'un atome dans $E_2$ l'incite à retomber,
      en émettant un second photon \emph{identique} au premier (même
      fréquence, même direction, même phase et même polarisation), au taux
      $B_{21}\,u(\nu)\,N_2$.
\end{itemize}
$A_{21}$, $B_{12}$, $B_{21}$ sont les \emph{coefficients
d'Einstein}\index{coefficients d'Einstein} de la transition.
\end{definition}

\begin{proposition}[Relations d'Einstein]\label{prop:b2:laser:einstein}
Pour des niveaux non dégénérés,
\[
B_{12} = B_{21} \equiv B, \qquad
\frac{A_{21}}{B_{21}} = \frac{8\pi h\nu^3}{c^3}.
\]
\omterm{def:b2:laser:processes}{Émission stimulée} et \omterm{def:b2:laser:processes}{absorption} ont le même coefficient~; le
taux spontané croît en $\nu^3$ relativement au taux stimulé.
\end{proposition}

\begin{proof}
Plaçons les atomes en équilibre avec le rayonnement à la température $T$. Deux
faits sont empruntés à des chapitres ultérieurs~: les populations de deux niveaux
à l'équilibre sont dans le rapport $N_2/N_1 = \eu^{-h\nu/k_BT}$ (le
facteur de Boltzmann, \cref{ch:b2:boltzmann-factor}), et le rayonnement d'équilibre
a la densité spectrale de Planck
$u(\nu) = (8\pi h\nu^3/c^3)/(\eu^{h\nu/k_BT} - 1)$
(\cref{ch:b2:thermal-radiation}). À l'équilibre les populations des niveaux
sont stationnaires~: $B_{12}uN_1 = A_{21}N_2 + B_{21}uN_2$, d'où
\[
u = \frac{A_{21}}{B_{12}\,N_1/N_2 - B_{21}}
  = \frac{A_{21}}{B_{12}\eu^{h\nu/k_BT} - B_{21}}.
\]
Ceci doit égaler l'expression de Planck à toute $T$~: la limite $T \to \infty$
(où $u \to \infty$) impose $B_{12} = B_{21}$, et la
comparaison donne alors $A_{21}/B_{21} = 8\pi h\nu^3/c^3$. Les
coefficients sont des propriétés de l'atome, de sorte que les relations valent aussi hors
d'équilibre.
\end{proof}

\begin{remark}[Pourquoi les lampes n'amplifient pas]\label{rem:b2:laser:lamps}
Dans un gaz à \qty{300}{K}, pour une transition visible ($h\nu \approx
\qty{2}{eV}$, $k_BT \approx \qty{0.025}{eV}$), $N_2/N_1 = \eu^{-80} \approx
10^{-35}$~: pratiquement aucun atome n'est excité, et la lumière n'est qu'absorbée. Une
décharge ou une flamme peuple $E_2$, mais toujours avec $N_2 < N_1$, et
chaque photon émis est débordé par les \omterm{def:b2:laser:processes}{absorptions}~; de plus le rapport
de l'\omterm{def:b2:laser:processes}{émission spontanée} à l'\omterm{def:b2:laser:processes}{émission stimulée} à l'équilibre,
$A_{21}/(B_{21}u) = \eu^{h\nu/k_BT} - 1$, est astronomiquement grand dans le
visible --- la lumière d'une lampe est spontanée~: phases aléatoires, toutes
les directions, toute la largeur de la raie. Ce n'est qu'aux fréquences radio et micro-ondes
($h\nu \ll k_BT$) que l'\omterm{def:b2:laser:processes}{émission stimulée} rivalise
naturellement~; le premier dispositif de ce genre fut d'ailleurs le maser, à
\qty{24}{GHz} (1954), avant le \omterm{def:b2:laser:cavity}{laser} optique (1960).
\end{remark}

\section{Amplification~: inversion de population et gain}

\begin{proposition}[Gain d'un milieu]\label{prop:b2:laser:gain}
Un faisceau d'intensité $I$ à la fréquence $\nu$ traversant le milieu
gagne de l'énergie par \omterm{def:b2:laser:processes}{émission stimulée} et en perd par \omterm{def:b2:laser:processes}{absorption}
(l'\omterm{def:b2:laser:processes}{émission spontanée} part dans toutes les directions et ne contribue guère
au faisceau). Sur une longueur $\dd z$,
\[
\dd I = \sigma\,(N_2 - N_1)\,I\,\dd z, \qquad
I(z) = I(0)\,\eu^{g z}, \qquad g = \sigma(N_2 - N_1),
\]
où $\sigma$ (une aire, la \emph{section efficace}\index{section efficace (transition)} de la transition, $\sigma \propto B\,h\nu/c$) mesure
la force de la transition. Le milieu \emph{amplifie} ($g > 0$)
seulement si
\[
N_2 > N_1 :
\]
c'est une \emph{inversion de population}\index{inversion de population}. Sans elle
($N_2 < N_1$, cas de tout milieu à l'équilibre) le faisceau est
absorbé.
\end{proposition}

\begin{proof}
Par unité de volume et de temps, $B u (N_2 - N_1)$ transitions ajoutent (ou retirent)
un photon $h\nu$ au faisceau~; avec $u = I/c$ (par unité de fréquence, sur
la largeur de la raie), la puissance gagnée par unité de volume vaut
$B h\nu (N_2 - N_1) I/c$, et c'est $\dd I/\dd z$~; on pose $\sigma =
Bh\nu/c$ (multiplié par le facteur de forme de raie).
\end{proof}

\begin{remark}[Pourquoi deux niveaux ne peuvent être inversés]\label{rem:b2:laser:twolevel}
Pompez un système à deux niveaux aussi fort que vous voudrez avec de la lumière à $\nu$~: le
pompage absorbe quand $N_1 > N_2$ et stimule l'émission quand $N_2 > N_1$,
et les populations tendent au mieux vers l'égalité, $N_2 = N_1$, où le
milieu est transparent mais ne gagne rien. Une inversion demande au moins
trois niveaux~: pomper depuis le niveau fondamental vers un niveau de courte vie qui
se désexcite vite vers le niveau \omterm{def:b2:laser:cavity}{laser} supérieur, lequel est de \emph{longue vie}
(métastable) et emmagasine donc de la population. Dans un
schéma à \emph{trois niveaux}\index{laser à trois niveaux} (rubis, le premier
\omterm{def:b2:laser:cavity}{laser}) le niveau \omterm{def:b2:laser:cavity}{laser} inférieur est le fondamental, de sorte que plus de la moitié
des atomes doivent être élevés avant l'inversion~: le pompage est brutal. Dans un
schéma à \emph{quatre niveaux}\index{laser à quatre niveaux} (He--Ne, Nd:YAG, la plupart
des \omterm{def:b2:laser:cavity}{lasers}) le niveau \omterm{def:b2:laser:cavity}{laser} inférieur est un niveau excité qui se vide vite
vers le fondamental~; il est presque vide à tout instant, et une faible
population dans le niveau supérieur est déjà une inversion --- le seuil
est bas.
\end{remark}

\begin{proposition}[Saturation du gain]\label{prop:b2:laser:saturation}
L'inversion est entretenue par un pompage qui fournit des atomes au niveau
supérieur au taux $R$ par unité de volume, contre la désexcitation $1/\tau$ et
l'\omterm{def:b2:laser:processes}{émission stimulée} $\sigma I (N_2 - N_1)/h\nu$. Dans un milieu à quatre
niveaux ($N_1 \approx 0$) le régime permanent est
\[
N_2 = \frac{R\tau}{1 + I/I_{\text{s}}}, \qquad
g = \frac{g_0}{1 + I/I_{\text{s}}}, \qquad
I_{\text{s}} = \frac{h\nu}{\sigma\tau} :
\]
le \emph{gain petit signal} $g_0 = \sigma R\tau$ est réduit par le faisceau
lui-même dès que $I$ approche l'\emph{intensité de
saturation}\index{intensité de saturation} $I_{\text{s}}$ --- l'amplificateur
est non linéaire, et c'est cette non-linéarité qui fixera la puissance du
\omterm{def:b2:laser:cavity}{laser}.
\end{proposition}

\begin{proof}
$\dd N_2/\dd t = R - N_2/\tau - \sigma I N_2/h\nu = 0$.
\end{proof}

\section{L'oscillateur~: cavité, seuil et modes}

\begin{definition}[Cavité laser]\label{def:b2:laser:cavity}
Un \emph{laser}\index{laser} est un milieu amplificateur de longueur $\ell$
placé entre deux miroirs (réflectances $R_1$, $R_2$) distants de $L$~:
une cavité de Fabry--Pérot (\cref{ch:b2:gratings}). Un miroir, le
\emph{coupleur de sortie}, est partiellement transmetteur ($T_2 = 1 - R_2$ de quelques
pour cent) et laisse sortir le faisceau. La lumière qui fait le tour de la cavité est
amplifiée deux fois par le milieu et atténuée par les miroirs et par les
autres pertes (diffusion, \omterm{def:b2:laser:processes}{absorption}, \omterm{def:b2:diffraction:huygens}{diffraction}).
\end{definition}

\begin{theorem}[Condition d'oscillation]\label{thm:b2:laser:threshold}
Une onde d'amplitude complexe $\underline s$ se reproduit après un
aller-retour si
\[
\underline s\,\cdot\, \eu^{g\ell}\sqrt{R_1}\,\eu^{g\ell}\sqrt{R_2}\,
\eu^{-\alpha\cdot 2L}\,\eu^{2\iu kL} = \underline s
\]
($\alpha$~: les autres pertes par unité de longueur). D'où deux conditions.
\begin{itemize}
\item \emph{Amplitude}~: $R_1R_2\,\eu^{2(g\ell - \alpha L)} \ge 1$, c'est-à-dire que le
      gain doit atteindre le \emph{seuil}\index{seuil (laser)}
      \[
      g_{\text{th}} = \frac{1}{\ell}\Big(\alpha L - \tfrac12\ln(R_1R_2)\Big)
      \approx \frac{1}{2\ell}\,(T_2 + \text{pertes par aller-retour}) ;
      \]
\item \emph{Phase}~: $2kL = 2\pi p$, c'est-à-dire $\nu_p = p\,c/2L$ --- le \omterm{def:b2:laser:cavity}{laser}
      n'oscille que sur les \emph{modes longitudinaux}\index{modes longitudinaux} de la cavité, espacés de $c/2L$, et parmi eux seulement sur
      ceux qui tombent sous la courbe de gain là où $g(\nu) \ge g_{\text{th}}$.
\end{itemize}
\end{theorem}

\begin{proof}
Le module et la phase du facteur d'aller-retour doivent tous deux être triviaux~;
l'approximation utilise $-\ln R \approx 1 - R = T$ pour $T \ll 1$.
\end{proof}

\begin{proposition}[Régime permanent au-dessus du seuil]\label{prop:b2:laser:steady}
Parti de l'\omterm{def:b2:laser:processes}{émission spontanée} d'un seul atome, tout mode pour
lequel $g_0 > g_{\text{th}}$ croît exponentiellement, aller-retour après
aller-retour~; à mesure que son intensité croît, le gain sature
(\cref{prop:b2:laser:saturation}) jusqu'à égaler exactement les pertes~:
\[
g(I) = g_{\text{th}} \quad\Longrightarrow\quad
I_{\text{cav}} = I_{\text{s}}\Big(\frac{g_0}{g_{\text{th}}} - 1\Big),
\qquad P_{\text{out}} = T_2\,I_{\text{cav}}\,A .
\]
Le gain se \emph{verrouille} à la valeur de seuil~; chaque atome supplémentaire
pompé au-dessus du seuil devient un photon de sortie, si bien que la puissance de sortie
croît linéairement avec le pompage au-dessus du pompage de seuil.
\end{proposition}

\begin{remark}[Le laser est un oscillateur à rétroaction]\label{rem:b2:laser:feedback}
Comparez à l'\omterm{prop:b2:feedback-oscillators:wien}{oscillateur à pont de Wien} du
\cref{ch:b2:feedback-oscillators}~: un amplificateur (le milieu inversé),
une \omterm{def:b2:feedback-oscillators:loop}{rétroaction} sélective en fréquence (la cavité, qui ne laisse passer que ses modes),
une condition de démarrage (\omterm{def:b2:feedback-oscillators:loop}{gain de boucle} $> 1$, c'est-à-dire $g_0 > g_{\text{th}}$),
un démarrage sur le bruit (ici l'\omterm{def:b2:laser:processes}{émission spontanée}), et une amplitude fixée
par la non-linéarité de l'amplificateur (saturation du gain). Le \omterm{def:b2:laser:cavity}{laser} est
le membre optique de la famille.
\end{remark}

\begin{example}[Le laser hélium--néon]\label{ex:b2:laser:hene}
Un tube de verre long de \qty{30}{cm}, d'alésage \qty{1}{mm}, contenant de l'hélium et du
néon à environ un millième d'atmosphère, traversé par une décharge de
quelques milliampères. Les électrons portent l'hélium à un niveau métastable à
\qty{20.6}{eV}, qui cède son énergie par collision à un niveau du néon à
presque exactement la même hauteur~; de là le néon retombe sur un niveau inférieur
(se vidant vite vers le fondamental --- quatre niveaux) en émettant à
\qty{632.8}{nm}. Le gain petit signal est minuscule, quelques pour cent par
passage, de sorte que les miroirs doivent être excellents ($R_1 = 0.999$, $R_2 = 0.99$)
et le tube propre~; la raie de gain est élargie par \omterm{thm:b2:sound-waves:doppler}{effet Doppler} jusqu'à
$\Delta\nu \approx \qty{1.5}{GHz}$, les modes sont espacés de $c/2L = \qty{500}{MHz}$,
de sorte que deux ou trois modes oscillent à la fois. Sortie \qty{1}{mW} pour
plusieurs watts de décharge~: rendement $10^{-4}$~; mais une longueur d'onde
définie à $10^{-6}$ près et un faisceau qui reste large de \qty{1}{mm} à travers le
laboratoire.
\end{example}

\section{Le faisceau gaussien}

\begin{proposition}[Faisceau gaussien]\label{prop:b2:laser:gaussian}
Le faisceau qu'émet une cavité stable à miroirs courbes est (dans le
mode transverse fondamental) un \emph{faisceau gaussien}\index{faisceau gaussien}~: à la distance $z$ de sa section la plus étroite (le
\emph{col}\index{col (d'un faisceau)}, de rayon $w_0$) le profil d'intensité
vaut
\[
I(r,z) = \frac{2P}{\pi w(z)^2}\,\exp\Big(-\frac{2r^2}{w(z)^2}\Big),
\qquad
w(z) = w_0\sqrt{1 + \Big(\frac{z}{z_{\text{R}}}\Big)^2},
\qquad
z_{\text{R}} = \frac{\pi w_0^2}{\lambda},
\]
où $P$ est la puissance totale et $z_{\text{R}}$ la \emph{longueur de
Rayleigh}\index{longueur de Rayleigh}~: sur $\pm z_{\text{R}}$ le faisceau reste
à moins de $\sqrt2$ de son \omterm{prop:b2:laser:gaussian}{col}, et bien au-delà il s'étale avec le
demi-angle
\[
\theta = \frac{\lambda}{\pi w_0} .
\]
Le faisceau est une solution de l'\omterm{thm:b2:waves-on-strings:equation}{équation d'onde} dans l'approximation
paraxiale~; nous en admettons la forme et l'utilisons. Ses fronts d'onde sont plans au
\omterm{prop:b2:laser:gaussian}{col} et sphériques au loin.
\end{proposition}

\begin{proof}
Admis à ce niveau (le calcul est la propagation, en optique de Fourier,
d'une distribution gaussienne sur une ouverture)~; sa propriété clé est
celle que la \omterm{def:b2:diffraction:huygens}{diffraction} a déjà donnée au \cref{ch:b2:diffraction}~: une
ouverture de taille $w_0$ étale sur $\lambda/w_0$, ici avec le facteur exact
$1/\pi$.
\end{proof}

\begin{proposition}[Focaliser et élargir un faisceau]\label{prop:b2:laser:focus}
Un \omterm{prop:b2:laser:gaussian}{faisceau gaussien} de rayon $w$ (bien plus grand que ce que son champ lointain
donnerait à la lentille, c'est-à-dire à peu près collimaté) tombant sur une lentille de distance
focale $f$ est focalisé en un \omterm{prop:b2:laser:gaussian}{col}
\[
w_0' = \frac{\lambda f}{\pi w}
\]
au foyer, de \omterm{prop:b2:laser:gaussian}{longueur de Rayleigh} $\pi w_0'^2/\lambda$. L'éclairement
maximal y vaut $2P/\pi w_0'^2$. Inversement une lunette afocale de
grandissement $M$ transforme un faisceau de rayon $w$ en un faisceau de rayon $Mw$
dont la \omterm{def:b2:maxwell-equations:operators}{divergence} est $M$ fois plus petite~: pour envoyer un faisceau loin, il faut d'abord le
faire large.
\end{proposition}

\begin{proof}
La lentille convertit le front d'onde plan en une sphère convergeant en $f$~;
la \omterm{def:b2:diffraction:huygens}{diffraction} d'une ouverture de rayon $w$ donne l'étalement angulaire
$\lambda/\pi w$, d'où la tache focale $f\lambda/\pi w$ --- ce qui s'accorde avec
$\theta = \lambda/\pi w_0'$ lu à l'envers.
\end{proof}

\begin{example}[Couper, lire, pointer]\label{ex:b2:laser:focus}
Un \omterm{def:b2:laser:cavity}{laser} à dioxyde de carbone de \qty{1}{kW} à \qty{10.6}{\micro m}, de rayon de faisceau
\qty{10}{mm}, focalisé par $f = \qty{100}{mm}$~: $w_0' = \qty{34}{\micro m}$,
éclairement maximal $2P/\pi w_0'^2 = \qty{5e11}{W/m^2}$ --- l'acier bout.
Un pointeur de \qty{1}{mW}, $w_0 = \qty{0.5}{mm}$ à \qty{633}{nm}~: $\theta =
\qty{0.4}{mrad}$, une tache de \qty{4}{cm} à \qty{50}{m}, un éclairement de
\qty{2.5}{kW/m^2} à la sortie --- au-dessus de la lumière solaire, ce qui explique pourquoi même un
milliwatt ne doit jamais entrer dans un œil~: le cristallin le focaliserait en une
tache de \qty{10}{\micro m} sur la rétine, à des millions de watts par mètre
carré. Un lecteur de disque à \qty{650}{nm} avec une lentille d'\omterm{prop:b2:guided-waves:fibre}{ouverture numérique}
$0.6$ focalise en environ $\lambda/2\mathrm{NA} \approx \qty{0.5}{\micro m}$,
la taille d'une cuvette.
\end{example}

\begin{remark}[Ce qui rend la lumière laser spéciale]\label{rem:b2:laser:special}
\emph{Directivité}~: un seul mode transverse, une \omterm{def:b2:maxwell-equations:operators}{divergence} à la limite de
\omterm{def:b2:diffraction:huygens}{diffraction}. \emph{Monochromatisme}~: un ou quelques modes de cavité, chacun
plus étroit qu'un mégahertz --- des \omterm{prop:b2:scalar-light-model:coherence}{longueurs de cohérence} de plusieurs mètres à plusieurs kilomètres
(\cref{ch:b2:scalar-light-model}). \emph{\omterm{prop:b2:two-wave-interference:spatial}{Cohérence spatiale}}~: tout le
faisceau est une seule onde, de sorte qu'il interfère avec lui-même n'importe où (holographie,
interférométrie). \emph{Brillance}~: un milliwatt dans un faisceau limité par la \omterm{def:b2:diffraction:huygens}{diffraction}
éclipse le Soleil par unité d'angle solide et de bande passante. \emph{Puissance}
en impulsions~: un joule en une nanoseconde fait un gigawatt, en une femtoseconde un
pétawatt. Et, puisque le faisceau est la sortie d'un oscillateur, il peut être
modulé au gigahertz --- la porteuse de la \omterm{prop:b2:guided-waves:fibre}{fibre optique}
(\cref{ch:b2:guided-waves}).
\end{remark}

\begin{method}[Estimations sur les lasers]\label{meth:b2:laser:method}
(1) Énergie du photon $h\nu = hc/\lambda$ et débit de photons $P/h\nu$.
(2) Seuil~: $g_{\text{th}}\ell \approx \tfrac12(T_2 + \text{pertes})$~;
comparer au gain petit signal $g_0\ell$ du milieu.
(3) Modes~: $c/2L$ contre la largeur de gain~; compter les modes qui oscillent.
(4) Puissance~: le gain se verrouille au seuil~; sortie $\propto$ (pompage $-$
pompage de seuil). (5) Faisceau~: $z_{\text{R}} = \pi w_0^2/\lambda$, $\theta =
\lambda/\pi w_0$, tache focale $\lambda f/\pi w$, éclairement maximal
$2P/\pi w_0^2$. (6) Sécurité~: comparer l'éclairement rétinien à celui du
Soleil.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:laser:1}
Un \omterm{def:b2:laser:cavity}{laser} He--Ne émet \qty{1}{mW} à \qty{632.8}{nm}. Fréquence, énergie du
photon en joules et en électronvolts, photons par seconde. Cavité de
\qty{30}{cm}~: espacement des modes~; combien de modes tiennent sous une courbe de gain
large de \qty{1.5}{GHz}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:laser:2}
Deux niveaux distants de \qty{2}{eV} à \qty{300}{K}~: rapport $N_2/N_1$ (utiliser
$k_BT = \qty{0.025}{eV}$). À quelle température les populations seraient-elles
égales~? Montrer qu'une inversion correspond à une température formellement \emph{négative}
dans le rapport de Boltzmann. Même rapport pour une transition micro-onde
à \qty{24}{GHz}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:laser:3}
\omterm{prop:b2:laser:gaussian}{Faisceau gaussien}, $w_0 = \qty{0.5}{mm}$, $\lambda = \qty{633}{nm}$~: \omterm{prop:b2:laser:gaussian}{longueur de
Rayleigh}, \omterm{def:b2:maxwell-equations:operators}{divergence}, rayon après \qty{100}{m} et à la Lune
(\qty{3.8e8}{m}). Même faisceau après un élargisseur $\times 20$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:laser:4}
Cavité avec $R_1 = 1$, $R_2 = 0.98$, milieu long de \qty{20}{cm}, autres
pertes $0.5\%$ par passage. Coefficient de gain au seuil. Le gain petit
signal du milieu vaut \qty{0.2}{m^{-1}}~: y a-t-il effet \omterm{def:b2:laser:cavity}{laser}~? Avec
$R_2 = 0.90$~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:laser:5}
\emph{Relations d'Einstein.} (a) Écrire le bilan d'une population à deux
niveaux en équilibre avec le rayonnement $u(\nu)$ et en tirer les deux
relations à partir de la loi de Planck (donnée). (b) Calculer
$A_{21}/B_{21}u = \eu^{h\nu/k_BT} - 1$ à \qty{300}{K} pour
$\lambda = \qty{600}{nm}$ et pour $\lambda = \qty{1}{cm}$. (c) Pourquoi les
masers ont-ils été inventés avant les \omterm{def:b2:laser:cavity}{lasers}~? (d) La durée de vie d'un niveau vaut
$\tau = 1/A_{21}$~; si $\tau = \qty{10}{ns}$ à \qty{600}{nm}, que vaut
$\tau$ pour une transition de même $B$ à \qty{6}{\micro m}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:laser:6}
\emph{Saturation et puissance de sortie.} Milieu à quatre niveaux, $\sigma =
\qty{3e-17}{m^2}$, $\tau = \qty{100}{ns}$, $\lambda = \qty{633}{nm}$.
(a) \omterm{prop:b2:laser:saturation}{Intensité de saturation}. (b) Taux de pompage $R$ pour un gain petit signal
$g_0 = \qty{0.1}{m^{-1}}$. (c) Le seuil de la cavité vaut
$g_{\text{th}} = \qty{0.02}{m^{-1}}$~: intensité intracavité en régime
permanent~; sortie à travers $T_2 = 1\%$ d'un faisceau d'aire \qty{1}{mm^2}.
(d) Montrer que la puissance de sortie est linéaire en $R$ au-dessus du seuil.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:laser:7}
\emph{Modes et stabilité.} (a) Longueur de cavité pour un fonctionnement monomode
sous une courbe de gain de \qty{1.5}{GHz}. (b) Une cavité de \qty{30}{cm} s'allonge
de \qty{1}{\micro m}~: décalage de la fréquence d'un mode, comparé à
l'espacement des modes et à la largeur de gain. (c) Pourquoi les He--Ne stabilisés
du commerce verrouillent-ils la longueur du tube en le chauffant~? (d) La largeur propre du mode est
de quelques kilohertz~: \omterm{prop:b2:scalar-light-model:coherence}{longueur de cohérence}~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:laser:8}
\emph{Focalisation.} (a) Faisceau $w = \qty{1}{mm}$, $\lambda = \qty{633}{nm}$,
$f = \qty{50}{mm}$~: \omterm{prop:b2:laser:gaussian}{col}, \omterm{prop:b2:laser:gaussian}{longueur de Rayleigh}, éclairement maximal pour
\qty{1}{mW}. (b) Le même faisceau dans un œil (focale \qty{17}{mm})~:
tache et éclairement rétiniens~; comparer à l'image du Soleil

(c) Pourquoi un pointeur vert de \qty{5}{mW} est-il plus dangereux qu'une
ampoule de \qty{100}{W}~? (d) Profondeur de foyer de la lentille de \qty{50}{mm}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:laser:9}

ions chrome, durée de vie du niveau supérieur \qty{3}{ms}, $\lambda =

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:laser:10}
\emph{La diode \omterm{def:b2:laser:cavity}{laser}.} La région émettrice d'une diode \omterm{def:b2:laser:cavity}{laser} fait environ
\qty{3}{\micro m} sur \qty{1}{\micro m}, $\lambda = \qty{780}{nm}$. (a)
Demi-angles de \omterm{def:b2:maxwell-equations:operators}{divergence} dans les deux directions. (b) Quelle direction
diverge le plus, et pourquoi le faisceau est-il elliptique~? (c) Une lentille de $f =

rendre l'ellipse ronde (deux idées)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:laser:11}
\emph{Démarrage.} (a) Écrire la condition d'amplitude sur un aller-retour avec un
gain net $G = \eu^{2(g_0 - g_{\text{th}})\ell} = 1.04$ par aller-retour

les mêmes atomes~: expliquer le saut de mode.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:laser:12}
\emph{Télémétrie lunaire.} Impulsions de \qty{100}{mJ} à \qty{532}{nm},

(d) Datation à \qty{100}{ps} près~: précision en distance par impulsion, et après

\end{exercise}

\section{Problème~: un laser hélium--néon, du tube à la Lune}

\begin{problem}[{Problème du week-end --- un laser construit, caractérisé et
utilisé}]\label{pb:b2:laser:1}
Le \omterm{def:b2:laser:cavity}{laser} sur la paillasse~: tube de verre $L = \qty{30}{cm}$, alésage de diamètre
\qty{1}{mm}, mélange hélium--néon, décharge de \qty{5}{mA} sous
\qty{1.5}{kV}~; miroirs $R_1 = 0.999$, $R_2 = 0.99$~; longueur d'onde
\qty{632.8}{nm}~; sortie \qty{1}{mW}. Données~: le niveau métastable de He se situe
à \qty{20.61}{eV}, le niveau \omterm{def:b2:laser:cavity}{laser} supérieur de Ne à \qty{20.66}{eV} avec une
durée de vie de \qty{100}{ns}, le niveau \omterm{def:b2:laser:cavity}{laser} inférieur à \qty{18.70}{eV} avec une
durée de vie de \qty{10}{ns}~; température du gaz \qty{400}{K}~; masse molaire du néon
\qty{20}{g/mol}~; \omterm{prop:b2:laser:gain}{section efficace} $\sigma = \qty{3e-17}{m^2}$~; $h =
\qty{6.63e-34}{J.s}$, $k_B = \qty{1.38e-23}{J/K}$, $k_BT$ à \qty{400}{K}
$= \qty{0.0345}{eV}$.

\textbf{Partie I --- Le milieu.}
\begin{enumerate}
  \item Énergie du photon et fréquence de la raie \omterm{def:b2:laser:cavity}{laser}.
  \item Les niveaux de He et de Ne diffèrent de \qty{0.05}{eV}~: comparer à
        $k_BT$ et expliquer pourquoi le transfert par collision He $\to$ Ne est
        efficace.
  \item Rapport de Boltzmann du niveau supérieur du Ne au fondamental à
        \qty{400}{K} sans la décharge~: commenter.
  \item Expliquer, à partir des deux durées de vie, pourquoi une inversion entre les
        niveaux du Ne est facile à maintenir dès que le niveau supérieur est alimenté.
  \item Largeur Doppler de la raie~: vitesse la plus probable du néon à
        \qty{400}{K} et $\Delta\nu \approx \nu\,v/c$ en ordre de
        grandeur (précisément $\Delta\nu = 2\nu\sqrt{2k_BT\ln2/Mc^2}$).
  \item Avec $\Delta N = N_2 - N_1 = \qty{3e15}{m^{-3}}$~: coefficient de gain
        petit signal et gain par passage sur \qty{30}{cm}.
\end{enumerate}

\textbf{Partie II --- La cavité.}
\begin{enumerate}[resume]
  \item Coefficient de gain au seuil~; marge par rapport à la question 6.
  \item Espacement des modes~; nombre de modes susceptibles d'osciller.
  \item Puissance intracavité et éclairement intracavité dans l'alésage.
  \item Qu'est-ce qui fixe le gain en régime permanent, et où passe l'excédent
        d'énergie de pompage~?
  \item Puissance électrique consommée et puissance optique sortie~: rendement~; deux
        raisons de sa faiblesse.
  \item Démarrage~: gain net par aller-retour $2(g_0 - g_{\text{th}})L$,
        nombre de photons dans la cavité en régime permanent, nombre
        d'allers-retours depuis un photon, et durée du démarrage.
  \item Le tube s'échauffe et s'allonge de \qty{1}{\micro m}~: décalage en fréquence
        d'un mode~; conséquence pour un \omterm{def:b2:laser:cavity}{laser} monomode.
  \item Le tube est fermé par des fenêtres à l'\omterm{prop:b2:wave-interfaces:brewster}{angle de Brewster}~: qu'est-ce que
        cela fait à la polarisation de la sortie, et pourquoi cela n'ajoute-t-il pas
        de perte~?
\end{enumerate}

\textbf{Partie III --- Le faisceau.}
\begin{enumerate}[resume]
  \item La cavité produit un \omterm{prop:b2:laser:gaussian}{col} $w_0 = \qty{0.3}{mm}$~: \omterm{prop:b2:laser:gaussian}{longueur de
        Rayleigh}, \omterm{def:b2:maxwell-equations:operators}{divergence}, diamètre du faisceau à \qty{10}{m}.
  \item Éclairement maximal à la sortie~; comparer à la lumière solaire
        (\qty{1}{kW/m^2}).
  \item Une lentille de $f = \qty{50}{mm}$~: \omterm{prop:b2:laser:gaussian}{col} et éclairement maximal au
        foyer.
  \item Le faisceau entre dans un œil (focale \qty{17}{mm})~: tache et éclairement
        rétiniens~; pourquoi un clignement n'est pas trop lent pour \qty{1}{mW},
        et ce que signifie «~classe 2~».
  \item Un élargisseur $\times 10$~: nouvelle \omterm{def:b2:maxwell-equations:operators}{divergence} et diamètre de tache
        à \qty{1}{km}.
  \item \omterm{prop:b2:scalar-light-model:coherence}{Longueur de cohérence} si le \omterm{def:b2:laser:cavity}{laser} fonctionne sur un seul mode de largeur
        \qty{1}{MHz}~; s'il fonctionne sur trois modes couvrant \qty{1}{GHz}.
\end{enumerate}

\textbf{Partie IV --- Jusqu'à la Lune.}
Une station de télémétrie tire des impulsions de \qty{100}{mJ} de \qty{100}{ps} à
\qty{532}{nm} par un télescope de \qty{1}{m}~; l'atmosphère élargit
le faisceau sortant à \qty{1}{''} ($\qty{4.8e-6}{rad}$).
\begin{enumerate}[resume]
  \item Photons par impulsion et puissance crête.
  \item Rayon de la tache sur la Lune (\qty{3.8e8}{m})~; comparer à la
        tache limitée par la \omterm{def:b2:diffraction:huygens}{diffraction}.
  \item Un réseau de réflecteurs de \qty{0.1}{m^2} intercepte quelle fraction de
        l'impulsion, combien de photons~?
  \item Les coins de cube de \qty{4}{cm} du réseau renvoient la lumière avec
        l'étalement $\lambda/d$~: rayon de la tache sur Terre, fraction captée par le
        télescope, photons détectés par impulsion (prendre un rendement optique
        global de $0.1$).
  \item Le temps d'aller-retour vaut environ \qty{2.56}{s}, mesuré à
        \qty{100}{ps} près~: précision en distance par impulsion~; après $10^4$
        retours~; comparer aux \qty{3.8}{cm} par an dont la
        Lune s'éloigne.
\end{enumerate}
\end{problem}
