\chapter{Fluides parfaits~: Euler et Bernoulli}\label{ch:b2:euler-bernoulli}

Tenez une feuille de papier par le bord et soufflez au-dessus~: elle se
soulève. Ouvrez le robinet de la douche et le rideau se rabat vers
l'intérieur. Une aile porte trois cents tonnes parce que l'air s'écoule
plus vite sur son extrados que sous son intrados. Ces trois faits sont
un seul énoncé~: \emph{là où un fluide va plus vite, sa pression est
plus basse}. Ce chapitre écrit la deuxième loi de Newton pour une
\omterm{def:b2:fluid-kinematics:particle}{particule de fluide} dans le modèle le plus simple --- le fluide
\emph{parfait}, qui subit la pression mais aucun frottement --- et en
tire le théorème de Bernoulli, bilan énergétique d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant},
dont les applications vont du débitmètre d'une conduite à
l'indicateur de vitesse d'un avion.

\section{L'équation d'Euler}

\begin{proposition}[Force de pression sur une particule de fluide]\label{prop:b2:euler-bernoulli:pressure}
\index{force de pression (densité volumique)}
La résultante des forces de pression exercées par le fluide environnant
sur une particule de volume $\dd\tau$ vaut
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
la pression agit comme une force volumique de densité $-\operatorname{\vect{grad}}P$,
qui pousse des hautes vers les basses pressions.
\end{proposition}

\begin{proof}
Sur la boîte du \cref{thm:b2:fluid-kinematics:continuity}, les faces en
$x$ et $x + \dd x$ reçoivent $+P(x)\dd y\dd z$ et $-P(x + \dd x)\dd y\dd z$
selon $x$~: soit $-\partial_xP\,\dd x\dd y\dd z$~; de même selon $y$ et $z$.
\end{proof}

\begin{definition}[Fluide parfait]\label{def:b2:euler-bernoulli:perfect}
\index{fluide parfait}
Un \emph{fluide parfait} est un fluide dans lequel la seule force de contact entre
particules voisines est la pression --- normale à toute surface, sans
contrainte tangentielle (visqueuse). C'est une idéalisation valable, comme l'explique le chapitre
suivant, loin des parois et à grand nombre de Reynolds~: l'eau dans
une conduite loin de la paroi, l'air autour d'une aile hors de la mince couche
limite, une vague sur la mer.
\end{definition}

\begin{theorem}[Équation d'Euler]\label{thm:b2:euler-bernoulli:euler}
\index{équation d'Euler}
Dans un référentiel galiléen, un \omterm{def:b2:euler-bernoulli:perfect}{fluide parfait} de masse volumique $\rho$ soumis à la
pesanteur obéit, en tout point, à
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
à quoi s'ajoute au second membre toute autre densité volumique de force $\vect f_v$ (électrique, d'inertie
dans un référentiel non galiléen). Au repos elle se réduit à
la loi de l'hydrostatique $\operatorname{\vect{grad}}P = \rho\vect g$ du
volume de première année.
\end{theorem}

\begin{proof}
Deuxième loi de Newton pour la particule de masse $\rho\,\dd\tau$~: son
accélération est la \omterm{thm:b2:fluid-kinematics:material}{dérivée particulaire}
(\cref{thm:b2:fluid-kinematics:material}), les forces sont la résultante des
pressions et le poids $\rho\vect g\,\dd\tau$~; on divise par $\dd\tau$.
\end{proof}

\begin{example}[Une cuve qui accélère]\label{ex:b2:euler-bernoulli:tank}
L'eau d'une cuve portée par un camion qui accélère de $\vect a$ est au repos
dans le référentiel du camion~; là, Euler avec la force d'inertie $-\rho\vect a$
donne $\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$~: la pression
croît le long de la «~pesanteur effective~» $\vect g - \vect a$, la surface
libre (une isobare) lui est perpendiculaire et s'incline vers l'arrière de
l'angle $\arctan(a/g)$ --- \qty{11}{\degree} pour $a = \qty{2}{m/s^2}$. Dans un
seau tournant à $\Omega$, la force centrifuge $\rho\Omega^2r\,\vect e_r$
donne $P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$ et la surface libre est
le paraboloïde $z = \Omega^2r^2/2g$.
\end{example}

\section{Le théorème de Bernoulli}

\begin{theorem}[Bernoulli]\label{thm:b2:euler-bernoulli:bernoulli}
\index{théorème de Bernoulli}\index{pression d'arrêt}\index{pression dynamique}
Pour un fluide \emph{parfait}, \emph{incompressible} et de masse volumique uniforme en
écoulement \emph{stationnaire}, dans un référentiel galiléen où la pesanteur est uniforme, la quantité
\[
P + \tfrac12\rho v^2 + \rho gz
\]
est constante \emph{le long de chaque \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}}. Si l'écoulement est de plus
\emph{irrotationnel}, c'est la même constante dans tout le fluide. Le
terme $\tfrac12\rho v^2$ est la \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique}}~; $P + \tfrac12\rho v^2$
est la \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{pression d'arrêt}}, celle qu'atteindrait le fluide
s'il était amené au repos le long de sa \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}.
\end{theorem}

\begin{proof}
Euler stationnaire avec $(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$ et $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$, $\rho$ uniforme~:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
Le second membre est perpendiculaire à $\vect v$, donc le gradient de la
quantité de Bernoulli n'a pas de composante le long de la \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}~: la quantité
est constante le long de celle-ci. Si $\vect\omega = \vect 0$ le gradient est nul
partout.
\end{proof}

\begin{remark}[Ce que dit Bernoulli, et ce qu'il ne dit pas]\label{rem:b2:euler-bernoulli:meaning}
Divisé par $\rho$, le théorème énonce~: l'énergie mécanique massique,
$v^2/2 + gz$, augmentée de $P/\rho$, se conserve le long du chemin d'une particule.
Le terme de pression est le travail fourni \emph{à} la particule par la
pression du fluide qui la suit, diminué du travail qu'elle fournit au fluide
qui la précède~: c'est le théorème de l'énergie pour une particule poussée à travers un champ
de pression sans frottement. Quatre conditions~: \omterm{def:b2:euler-bernoulli:perfect}{fluide parfait} (aucune perte
visqueuse), stationnaire, incompressible, et \emph{le long d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}}. Comparer
deux points de \omterm{def:b2:fluid-kinematics:euler}{lignes de courant} différentes n'est permis qu'en \omterm{def:b2:fluid-kinematics:vorticity}{écoulement irrotationnel}.
Il ne dit pas que «~l'air rapide aspire~»~: il dit qu'une particule qui
a été \emph{accélérée} (par une chute de pression) est maintenant à plus basse
pression --- la cause est le champ de pression, la vitesse en est l'effet.
\end{remark}

\begin{remark}[Fluides compressibles]\label{rem:b2:euler-bernoulli:compressible}
Pour un gaz la masse volumique varie, mais si l'écoulement est stationnaire, parfait et
isentropique la même démonstration donne la conservation de $h + v^2/2 + gz$
le long d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}, où $h$ est l'enthalpie massique ($h = c_PT$
pour un gaz parfait)~: c'est la forme utilisée pour les tuyères et les turbines au
\cref{ch:b2:open-systems}. À des vitesses très inférieures à celle du son
($v \lesssim \qty{100}{m/s}$ dans l'air) la masse volumique varie de moins de $5\%$
et le théorème incompressible est exact.
\end{remark}

\section{Applications}

\begin{proposition}[Le débitmètre de Venturi]\label{prop:b2:euler-bernoulli:venturi}
\index{effet Venturi}\index{débitmètre de Venturi}
Dans une conduite horizontale qui se rétrécit de la section $S_1$ à $S_2 < S_1$, la
vitesse croît ($v_2 = v_1S_1/S_2$, conservation de la masse) et la pression
chute~:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
Mesurer la différence de pression au manomètre donne le débit~:
c'est un \emph{\omterm{prop:b2:euler-bernoulli:venturi}{débitmètre de Venturi}}. Le même effet fait fonctionner un carburateur, un
pistolet à peinture, l'arrivée d'air d'un bec Bunsen et « l'aspiration » d'un
train qui passe.
\end{proposition}

\begin{proof}
Bernoulli le long d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant} centrale à hauteur constante, avec $v_2 =
v_1S_1/S_2$~; on résout en $v_1$.
\end{proof}

\begin{proposition}[Le tube de Pitot]\label{prop:b2:euler-bernoulli:pitot}
\index{tube de Pitot}
Un tube face à l'écoulement, fermé à l'autre bout, amène le fluide au repos
à son embouchure (un \emph{point d'arrêt})~; une seconde ouverture latérale,
parallèle à l'écoulement, relève la pression statique $P$. La différence est
la \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique}~:
\[
P_{\text{arr}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{arr}} - P)/\rho} .
\]
Tout avion mesure ainsi sa vitesse par rapport à l'air.
\end{proposition}

\begin{proof}
Bernoulli le long de la \omterm{def:b2:fluid-kinematics:euler}{ligne de courant} qui aboutit au point d'arrêt, où
$v = 0$~; les trous latéraux ne perturbent pas l'écoulement, qui passe aux $v$ et $P$
non perturbés.
\end{proof}

\begin{proposition}[Formule de Torricelli]\label{prop:b2:euler-bernoulli:torricelli}
\index{formule de Torricelli}\index{vidange}
Un grand réservoir ouvert se vide par un petit trou situé à la profondeur $h$ sous la
surface libre~; le jet sort à
\[
v = \sqrt{2gh} ,
\]
la vitesse de chute libre depuis la surface. Le débit vaut $D_V \approx
\alpha s\sqrt{2gh}$ avec $s$ l'aire du trou et $\alpha \approx 0.6$ le
\emph{coefficient de contraction} d'un orifice à bord vif (les \omterm{def:b2:fluid-kinematics:euler}{lignes de courant}
continuent de converger après le trou).
\end{proposition}

\begin{proof}
Bernoulli de la surface libre (au repos, puisque le réservoir est grand~:
$v_{\text{surf}} = v\,s/S \ll v$~; pression $P_0$) au jet (pression $P_0$
juste à la sortie du trou)~: $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$. L'écoulement
est quasi stationnaire~: $h$ varie lentement.
\end{proof}

\begin{example}[Applications numériques]\label{ex:b2:euler-bernoulli:numbers}
Un Venturi de \qty{5}{cm} qui se rétrécit à \qty{2.5}{cm} sur une conduite d'eau, avec
$\Delta h = \qty{20}{cm}$ d'eau (\qty{2.0}{kPa})~: $v_1 = \sqrt{2 \times 2000/
(1000 \times 15)} = \qty{0.52}{m/s}$, $D_V = \qty{1.0}{L/s}$. Un avion de ligne à
\qty{11}{km} ($\rho = \qty{0.36}{kg/m^3}$) à \qty{250}{m/s}~: \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique}
\qty{11}{kPa}, soit $11\%$ de la pression ambiante --- le \omterm{prop:b2:euler-bernoulli:pitot}{tube de Pitot} fonctionne
encore, avec une correction de compressibilité. Un château d'eau \qty{40}{m}
au-dessus des robinets~: \qty{4}{bar} en statique, un jet à \qty{28}{m/s} si l'on ouvre
en grand (bien moins en pratique~: les conduites ne sont pas parfaites).
\end{example}

\begin{proposition}[Portance d'une aile]\label{prop:b2:euler-bernoulli:lift}
\index{portance}\index{théorème de Kutta--Joukowski}\index{effet Magnus}
Dans l'écoulement autour d'une aile l'air passe plus vite sur l'extrados
que sous l'intrados~; la pression est donc plus basse
dessus que dessous, et la résultante est la \emph{portance} $F_L =
\tfrac12\rho v^2SC_L$ ($S$ la surface de l'aile, $C_L \approx 0.3$--$1.5$ le coefficient
de portance, selon le profil et l'incidence). L'écart
de vitesse équivaut à une \emph{circulation} $\Gamma$ de la
vitesse autour du profil, et la portance par unité d'envergure vaut $F_L' =
\rho v\Gamma$ (Kutta--Joukowski, admis). Une balle en rotation entraîne l'air
avec elle et se trouve déviée latéralement de la même façon~: c'est l'\emph{\omterm{prop:b2:euler-bernoulli:lift}{effet
Magnus}} d'une balle de tennis coupée ou d'un coup franc brossé.
\end{proposition}

\begin{proof}
Bernoulli sur les \omterm{def:b2:fluid-kinematics:euler}{lignes de courant} juste au-dessus et juste au-dessous, qui partent des
mêmes conditions amont~: $P_{\text{inf}} - P_{\text{sup}} = \tfrac12\rho
(v_{\text{sup}}^2 - v_{\text{inf}}^2)$~; on intègre sur la corde. Pourquoi l'écoulement
supérieur est plus rapide --- le bord de fuite effilé y impose le
point d'arrêt arrière et fixe la circulation (condition de Kutta)
--- dépasse le seul modèle du \omterm{def:b2:euler-bernoulli:perfect}{fluide parfait}~; il faut la couche limite
du chapitre suivant.
\end{proof}

\begin{proposition}[Un écoulement non stationnaire~: le tube en U]\label{prop:b2:euler-bernoulli:utube}
\index{oscillation dans un tube en U}\index{Bernoulli non stationnaire}
Une colonne liquide de longueur totale $L$ remplit un tube en U de section uniforme~;
écartée de $z$ de l'équilibre, elle oscille selon
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
comme un pendule de longueur $L/2$. Plus généralement, pour un écoulement non stationnaire mais
irrotationnel de potentiel $\varphi$, Euler s'intègre en
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$, la même
constante dans tout le fluide (\omterm{prop:b2:euler-bernoulli:utube}{Bernoulli non stationnaire}).
\end{proposition}

\begin{proof}
La vitesse $v = \dot z$ est uniforme le long du tube (incompressible,
section uniforme). Projetons Euler sur l'axe du tube et intégrons le long
de la colonne d'une surface libre à l'autre~: les termes de pression se
compensent ($P_0$ aux deux extrémités), $\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$, le
terme convectif $\rho\,\Delta(v^2/2)$ est nul (même vitesse aux deux extrémités),
et le terme de pesanteur donne $\rho g\cdot2z$. D'où $\rho L\ddot z = -2\rho gz$.
L'énoncé général découle de $\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ pour $\vect v = \operatorname{\vect{grad}}\varphi$.
\end{proof}

\begin{example}[Mise en route d'une conduite]\label{ex:b2:euler-bernoulli:start}
Une conduite horizontale de longueur $\ell$ mène d'un réservoir de charge
constante $h$ à une extrémité ouverte~; la vanne est ouverte à $t = 0$. Avec $v$ uniforme
le long de la conduite, Euler intégré de la surface du réservoir à la
sortie donne $\ell\,\dd v/\dd t = gh - \tfrac12v^2$, dont la solution est
$v = V\tanh(Vt/2\ell)$ avec $V = \sqrt{2gh}$~: l'écoulement atteint la vitesse de
Torricelli avec la constante de temps $2\ell/V$ --- \qty{13}{s} pour une conduite forcée de
\qty{400}{m} sous \qty{200}{m} de charge.
\end{example}

\begin{method}[Utiliser Bernoulli]\label{met:b2:euler-bernoulli:method}
(1) Vérifier les quatre conditions~; décider si l'écoulement est irrotationnel
(écoulement uniforme en amont autour d'un obstacle~: oui~; écoulement en conduite avec un
profil de vitesse~: non --- rester sur une seule \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}). (2) Choisir deux points
d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant} où le plus de grandeurs possible sont connues~: une surface
libre ($P = P_0$, $v \approx 0$ si la section est grande), un jet à
l'air libre ($P = P_0$), un point d'arrêt ($v = 0$). (3) Ajouter la conservation
de la masse $vS = $ const pour relier les vitesses. (4) Pour les écoulements non
stationnaires, intégrer plutôt Euler le long de la ligne.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:1}
De l'eau s'écoule dans une conduite horizontale de diamètre \qty{8}{cm} à \qty{1.5}{m/s}
sous \qty{3.0}{bar}. Elle passe dans une section de \qty{4}{cm}~: vitesse
et pression~; puis dans une section de \qty{2}{cm}~: pression --- que se passe-t-il
si elle devait être négative~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:2}
Le \omterm{prop:b2:euler-bernoulli:pitot}{tube de Pitot} d'un avion à \qty{5}{km} ($\rho = \qty{0.74}{kg/m^3}$)
lit une \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique} de \qty{15}{kPa}. Vitesse vraie. L'anémomètre
est étalonné pour la masse volumique au niveau de la mer (\qty{1.22}{kg/m^3})~: quelle
«~vitesse indiquée~» affiche-t-il, et pourquoi les pilotes trouvent-ils l'écart
utile plutôt que gênant~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:3}
Un réservoir contient de l'eau sur \qty{3.0}{m} de profondeur~; un trou de \qty{1.0}{cm^2} est percé
à \qty{1.0}{m} au-dessus du sol. Vitesse de sortie~; débit (contraction $0.6$)~;
où le jet touche-t-il le sol~? À quelle profondeur faudrait-il percer un second trou
pour atteindre le même point~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:4}
Un vent de \qty{30}{m/s} souffle sur un toit plat de \qty{100}{m^2}~; l'air
sous le toit est immobile à la pression atmosphérique ($\rho = \qty{1.2}{kg/m^3}$).
Différence de pression et force résultante sur le toit~; comparer au poids
du toit (\qty{50}{kg/m^2}). Dans quel sens est-il poussé~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:5}
Une cuve parallélépipédique fermée de \qty{2.0}{m} de long, à moitié pleine d'eau, est posée sur un
camion. (a) Accélération de \qty{3.0}{m/s^2}~: angle de la surface libre,
et différence de hauteur d'eau entre les parois arrière et avant.
(b) Différence de pression entre les deux coins du fond. (c)
Freinage à \qty{6}{m/s^2} --- l'eau atteint-elle le couvercle situé \qty{0.8}{m} au-dessus
du fond~? (d) Même camion abordant un virage de rayon \qty{40}{m} à
\qty{15}{m/s}~: inclinaison de la surface.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
Un seau de rayon \qty{15}{cm} contenant de l'eau sur \qty{20}{cm} tourne
autour de son axe à \qty{2.0}{tr/s}. (a) Forme et équation de la surface
libre. (b) Différence de hauteur entre le bord et le centre. (c) Le volume étant
conservé, hauteur de l'eau au centre et au bord. (d) À
quelle vitesse de rotation le fond se découvre-t-il en son centre~? (e) Pourquoi le
même paraboloïde fait-il un miroir de télescope parfait (un bain de mercure en rotation)~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}
\emph{Siphon.} Un tube amène de l'eau depuis un réservoir par-dessus une crête située \qty{2.0}{m}
au-dessus de la surface libre jusqu'à une sortie \qty{3.0}{m} au-dessous d'elle. (a) Vitesse de sortie
et, pour un tube de \qty{2}{cm}, débit. (b) Pression à la crête. (c)
Hauteur maximale de la crête pour que le siphon fonctionne (la pression de vapeur
de l'eau, \qty{2.3}{kPa}, ne doit pas être atteinte). (d) Pourquoi le siphon
ne fonctionne-t-il pas si le tube est plein d'air~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}
Un avion léger de \qty{1000}{kg} croise à \qty{60}{m/s} au niveau de la mer sur
une aile de \qty{15}{m^2}. (a) Différence de pression moyenne entre les deux
faces de l'aile~; coefficient de portance. (b) Si la vitesse sous l'aile vaut
\qty{58}{m/s}, que vaut-elle au-dessus (mêmes conditions
amont)~? (c) Circulation autour du profil par la formule de
Kutta--Joukowski, pour une envergure de \qty{10}{m}. (d) Vitesse de décrochage si
$C_L$ ne peut dépasser $1.4$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{Sténose.} Le sang ($\rho = \qty{1060}{kg/m^3}$) s'écoule à \qty{0.50}{m/s}
dans une artère de section \qty{0.50}{cm^2} à \qty{13}{kPa} au-dessus de la pression
atmosphérique. Une plaque la rétrécit à \qty{0.10}{cm^2}. (a) Vitesse et pression
dans la sténose. (b) Le tissu qui entoure l'artère est à environ
\qty{1}{kPa}~: l'artère peut-elle s'affaisser~? (c) Si elle se rétrécit, la vitesse
croît encore~: expliquer l'emballement et pourquoi un tel rétrécissement peut se mettre à battre.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{Température d'arrêt.} Pour un gaz parfait en écoulement isentropique
stationnaire, $c_PT + v^2/2$ se conserve le long d'une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant}. (a) Montrer que
l'air amené au repos au nez d'un avion atteint $T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$, avec $M = v/c$ le nombre de Mach et $c =
\sqrt{\gamma RT/M_{\text{mol}}}$ la vitesse du son. (b) Température au nez
d'un avion de ligne à $M = 0.85$ dans un air à \qty{220}{K}~; d'un avion
supersonique à $M = 2$~; d'une capsule de rentrée à $M = 25$ (la formule
a-t-elle encore un sens~?). (c) Montrer que la formule de Pitot incompressible
surestime la vitesse à $M = 0.85$, et d'à peu près combien (développer
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ au second ordre en $M^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{Régime transitoire de démarrage.} Une conduite de longueur $\ell = \qty{50}{m}$ et de section
uniforme mène d'un grand réservoir (charge $h = \qty{5.0}{m}$ au-dessus de la
sortie) à une vanne située à l'extrémité ouverte. La vanne est ouverte à $t = 0$. (a)
Intégrer Euler le long de la conduite pour obtenir $\ell\,\dd v/\dd t = gh - v^2/2$.
(b) La résoudre~: $v = V\tanh(Vt/2\ell)$ avec $V = \sqrt{2gh}$. (c) Temps pour
atteindre $90\%$ de la vitesse finale. (d) La vanne est maintenant fermée brutalement en
\qty{0.5}{s} depuis le plein débit~: estimer la décélération moyenne et la
surpression à la vanne par $\rho\ell\,\dd v/\dd t$ --- et pourquoi une fermeture réelle
en produit bien davantage (chapitres suivants).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{La surface libre d'un tourbillon.} De l'eau tourne comme le \omterm{ex:b2:fluid-kinematics:planeflows}{tourbillon ponctuel}
du \cref{ch:b2:fluid-kinematics}, $v_\theta = \Gamma/2\pi r$, irrotationnel,
avec une surface libre en $z = 0$ au loin. (a) Pourquoi peut-on appliquer Bernoulli
entre deux points quelconques~? (b) Forme de la surface libre $z(r)$. (c)
À l'intérieur d'un cœur de Rankine ($r < a$, $v_\theta = \Omega r$, rotationnel), utiliser
plutôt Euler dans le référentiel tournant, ou directement~: montrer que $z = z(a) +
\Omega^2(r^2 - a^2)/2g$. (d) Profondeur totale de l'entonnoir pour un tourbillon
de baignoire avec $\Gamma = \qty{0.03}{m^2/s}$ et $a = \qty{5}{mm}$~; pour une tornade
($\Gamma = \qty{3e4}{m^2/s}$, $a = \qty{60}{m}$, dans l'air~: exprimer plutôt le
résultat comme une chute de pression au centre).
\end{exercise}

\section{Problème~: le barrage, la conduite forcée et la turbine}

\begin{problem}[{Problème du week-end --- une centrale hydroélectrique du
réservoir au jet~: pressions, vitesses, puissances, pertes et le
coup de bélier}]
\label{pb:b2:euler-bernoulli:1}

Un lac de retenue contient de l'eau sur \qty{150}{m} de profondeur derrière un barrage~; sa surface libre
est $H = \qty{200}{m}$ au-dessus des injecteurs d'une turbine. Une conduite forcée (un
tuyau d'acier) de diamètre $D = \qty{3.0}{m}$ et de longueur $\ell = \qty{400}{m}$
achemine le débit nominal $D_V = \qty{60}{m^3/s}$ jusqu'à la centrale,
où un injecteur le transforme en un jet libre à la pression atmosphérique $P_0 =
\qty{1.0}{bar}$ qui frappe les augets d'une roue Pelton. $\rho =
\qty{1000}{kg/m^3}$, $g = \qty{9.81}{m/s^2}$~; le fluide est parfait sauf mention
contraire.

\medskip
\noindent\textbf{Partie I --- Au repos et au débit nominal.}
\begin{enumerate}
  \item Pression à la base du barrage, et force sur une bande
        verticale de \qty{1}{m} de large du barrage, de haut en bas.
  \item Les vannes de la turbine étant fermées, pression dans la conduite forcée à
        la centrale.
  \item Vitesse dans la conduite au débit nominal~; \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique} correspondante~;
        pression à l'extrémité aval de la conduite (avant
        l'injecteur), par Bernoulli depuis la surface du lac.
  \item Vitesse du jet libre~; section et diamètre de l'injecteur.
  \item Puissance cinétique portée par le jet~; montrer qu'elle vaut $\rho gD_VH$
        et la calculer.
  \item Masse et énergie cinétique de l'eau contenue dans la conduite
        au débit nominal~; temps de transit d'une particule dans la conduite.
  \item Montrer que la charge $H$ se partage, au débit nominal, entre la
        pression au bas de la conduite et l'énergie cinétique
        de l'eau, et que l'injecteur convertit la première en la
        seconde~: que vaut la pression juste après l'injecteur~?
  \item Un \omterm{def:b2:euler-bernoulli:perfect}{fluide parfait} n'a pas de perte de charge, un fluide réel en a~: la perte dans
        une conduite vaut $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ avec $\lambda
        \approx 0.015$ pour de l'acier lisse. Perte de charge en mètres d'eau, et
        fraction de la puissance perdue.
  \item Un ingénieur propose une conduite de \qty{2}{m} pour économiser l'acier. Reprendre
        la question 7~; commenter.
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Mesurer le débit.}
Un Venturi est intégré à la conduite forcée, avec un col de \qty{2.0}{m}.
\begin{enumerate}[resume]
  \item Vitesse au col et chute de pression entre la conduite et le
        col au débit nominal.
  \item La chute est lue sur un manomètre à mercure ($\rho_{\text{Hg}} =
        \qty{13600}{kg/m^3}$)~: différence de hauteur.
  \item Montrer que le débit est proportionnel à la racine carrée de
        la lecture du manomètre, et donner la lecture à la moitié du débit
        nominal.
  \item Pourquoi les prises de pression doivent-elles affleurer la paroi et non
        dépasser dans l'écoulement~?
  \item Une portion de la conduite passe, pour des raisons topographiques, par une
        crête située \qty{6}{m} \emph{au-dessus} de la surface du lac. Pression en ce point
        au débit nominal~; que se passe-t-il, et que doit faire le concepteur~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Régimes non stationnaires.}
Le lac a une superficie $A = \qty{2.0}{km^2}$.
\begin{enumerate}[resume]
  \item Vitesse à laquelle le niveau du lac baisse au débit nominal, en
        centimètres par heure.
  \item La turbine est arrêtée et l'injecteur (de section trouvée à la question 4)
        est laissé ouvert à l'air libre~: le lac se vide par gravité. Écrire
        le bilan de masse et trouver le niveau $h(t)$ (écoulement de Torricelli
        quasi stationnaire).
  \item Temps pour que le niveau baisse de \qty{10}{m}~; comparer au temps
        obtenu à débit nominal constant.
  \item Démarrage de la centrale~: la conduite est pleine, l'injecteur est ouvert
        à $t = 0$. En prenant la vitesse uniforme le long de la conduite et en
        négligeant la géométrie de l'injecteur, intégrer Euler le long de
        la conduite pour obtenir $\ell\,\dd v/\dd t = gH - v^2/2$, la résoudre et donner
        la constante de temps.
  \item Pendant ce démarrage, à l'instant où $v$ vaut la moitié de sa valeur
        finale, que vaut la pression au bas de la conduite~?
        Expliquer le signe de l'écart avec la valeur en régime établi.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Fermer la vanne.}
\begin{enumerate}[resume]
  \item La vanne de la turbine est fermée en $\tau = \qty{10}{s}$~; on suppose que
        l'eau de la conduite décélère uniformément. À l'aide d'Euler le long
        de la conduite, estimer la surpression à la vanne, en bars.
  \item Pour une fermeture rapide l'eau n'est plus incompressible~: une onde de
        pression remonte la conduite à $c = \qty{1400}{m/s}$ (la vitesse du
        son dans l'eau au sein d'un tuyau d'acier --- \cref{ch:b2:sound-waves}),
        et la surpression vaut $\Delta P = \rho c\,\Delta v$ (Joukowsky).
        Valeur pour un arrêt brutal depuis le débit nominal~; comparer à la
        pression statique. Combien de temps l'onde met-elle pour atteindre le
        lac et revenir, et que cela dit-il du sens du mot
        «~rapide~»~?
  \item Pour protéger la conduite on installe une \emph{cheminée d'équilibre} --- un puits
        vertical ouvert de section $A_s = \qty{100}{m^2}$ --- au point où une
        galerie horizontale de longueur $L = \qty{2000}{m}$ et de section $A =
        \qty{7.0}{m^2}$ venant du lac rejoint la conduite forcée. Après une
        fermeture brutale, l'eau de la galerie continue d'avancer et le
        niveau $z$ dans le puits monte. Écrire la conservation de la masse entre
        la galerie et le puits.
  \item Intégrer Euler le long de la galerie (\omterm{def:b2:euler-bernoulli:perfect}{fluide parfait}) pour montrer que
        $z$ obéit à $\ddot z + \dfrac{gA}{LA_s}z = 0$~; période de
        l'oscillation.
  \item Montée maximale du niveau dans le puits après une fermeture depuis le
        débit nominal (argument énergétique~: l'énergie cinétique de l'eau de la galerie
        devient de l'énergie potentielle dans le puits).
  \item Récapituler~: les cinq pressions rencontrées dans cette centrale (statique au
        barrage, statique à la turbine, dynamique dans la conduite, coup de bélier,
        cheminée d'équilibre) et la loi qui donne chacune.
\end{enumerate}
\end{problem}
