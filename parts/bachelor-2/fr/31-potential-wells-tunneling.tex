\chapter{Puits de potentiel, barrières et effet tunnel}\label{ch:b2:potential-wells-tunneling}

Une bille dans une cuvette va et vient avec l'énergie qu'on lui a
donnée~; un électron dans un atome, un nucléon dans un noyau, un électron dans un
cristal de taille nanométrique ne peuvent avoir que certaines énergies, et la plus basse
d'entre elles n'est pas nulle. Une bille ne peut franchir une colline plus haute que son énergie
ne le permet~; une particule $\alpha$ quitte un noyau à travers un mur coulombien
qu'elle ne pourrait jamais escalader, la pointe d'un microscope à \omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel} fait passer
un courant à travers un intervalle de vide, et l'atome d'azote de l'ammoniac
traverse le plan de ses trois hydrogènes vingt-quatre milliards
de fois par seconde. Ce chapitre résout l'équation de Schrödinger dans les
potentiels les plus simples --- puits à parois infinies et finies, marche,
barrière, double puits --- et y trouve les deux grands faits quantiques
que la mécanique classique ne peut produire~: la \emph{quantification}
des énergies liées, et l'\emph{\omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel}} à travers des régions classiquement
interdites, avec sa sensibilité exponentielle à la largeur et à la hauteur de
la barrière, d'où sa capacité à servir de règle au picomètre et l'étendue des
durées de vie radioactives sur trente ordres de grandeur.

\section{Le puits infini}

\begin{proposition}[Puits carré infini]\label{prop:b2:potential-wells-tunneling:infinite}
Une particule confinée dans $0 < x < L$ par des parois impénétrables ($V = 0$
à l'intérieur, $V = \infty$ à l'extérieur, donc $\varphi = 0$ aux parois) a les
\omterm{thm:b2:schrodinger-wave-functions:stationary}{états stationnaires} et les énergies
\[
\varphi_n(x) = \sqrt{\frac{2}{L}}\sin\frac{n\pi x}{L}, \qquad
E_n = \frac{n^2h^2}{8mL^2} = n^2E_1, \qquad n = 1, 2, 3, \dots
\]
Les énergies sont \emph{quantifiées}\index{quantification de l'énergie}, croissent
comme $n^2$, et la plus basse n'est pas nulle~: l'\emph{énergie de
confinement}\index{énergie de confinement} $E_1 = h^2/8mL^2$ est le prix de la
localisation ($\Delta x \sim L$ impose $\Delta p \sim h/2L$). Les
$\varphi_n$ sont orthogonales, $\int\varphi_n\varphi_m\dd x = \delta_{nm}$, et tout
état du puits est une superposition de celles-ci.
\end{proposition}

\begin{proof}
À l'intérieur, $\varphi'' = -k^2\varphi$ avec $k^2 = 2mE/\hbar^2$~: $\varphi = A\sin kx
+ B\cos kx$~; $\varphi(0) = 0$ annule $B$, $\varphi(L) = 0$ exige $kL = n\pi$
--- les \omterm{prop:b2:waves-on-strings:modes}{ondes stationnaires} d'une corde, avec $E = \hbar^2k^2/2m$. La
normalisation donne $A = \sqrt{2/L}$~; l'orthogonalité est celle des
sinus.
\end{proof}

\begin{example}[Ordres de grandeur, et couleurs]\label{ex:b2:potential-wells-tunneling:orders}
$E_1 = h^2/8mL^2$~: un électron dans \qty{1}{nm}, \qty{0.38}{eV}~; dans
\qty{0.1}{nm} (un atome), \qty{38}{eV} --- l'échelle des énergies atomiques~; un
nucléon dans \qty{5}{fm}, \qty{8}{MeV} --- l'échelle des énergies nucléaires~; une
bille dans une boîte, $10^{-63}\,\unit{J}$ --- personne ne le remarquera. Un
nanocristal semi-conducteur de quelques nanomètres (une \emph{boîte
quantique}) confine ses électrons~: l'\omterm{prop:b2:potential-wells-tunneling:infinite}{énergie de confinement} s'ajoute à la
bande interdite du cristal, si bien que plus la boîte est petite, plus la lumière émise est bleue ---
le même matériau brille rouge à \qty{6}{nm} et vert à \qty{3}{nm},
réglé par la seule taille.
\end{example}

\section{Le puits fini}

\begin{proposition}[Puits carré fini]\label{prop:b2:potential-wells-tunneling:finite}
Pour $V = 0$ dans $|x| < a$ et $V = V_0$ à l'extérieur, un état d'énergie $0 < E < V_0$
oscille à l'intérieur ($k = \sqrt{2mE}/\hbar$) et décroît à l'extérieur ($\kappa =
\sqrt{2m(V_0 - E)}/\hbar$)~: $\varphi \propto \eu^{-\kappa|x|}$ au-delà des parois.
Le raccordement de $\varphi$ et $\varphi'$ en $x = \pm a$ donne, pour les états pairs et
impairs,
\[
k\tan ka = \kappa \quad\text{(pair)}, \qquad
-k\cot ka = \kappa \quad\text{(impair)}, \qquad\text{avec}\quad
(ka)^2 + (\kappa a)^2 = \frac{2mV_0a^2}{\hbar^2} \equiv R^2 .
\]
Graphiquement, les solutions sont les intersections des courbes $\eta =
\xi\tan\xi$, $\eta = -\xi\cot\xi$ avec le cercle $\xi^2 + \eta^2 = R^2$
($\xi = ka$, $\eta = \kappa a$)~: il y a \emph{toujours au moins un état
lié} (l'état fondamental pair), et $1 + \lfloor 2R/\pi\rfloor$ en tout.
La particule \emph{pénètre} dans la région interdite sur la profondeur
$1/\kappa$, et les niveaux sont plus bas que ceux du puits infini de
même largeur.
\end{proposition}

\begin{proof}
La solution paire vaut $A\cos kx$ à l'intérieur et $B\eu^{-\kappa|x|}$ à l'extérieur~; la
continuité de $\varphi'/\varphi$ en $x = a$ donne $-k\tan ka = -\kappa$. De même
pour l'impaire avec $\sin$. Le cercle est la définition de $k$ et de
$\kappa$. Chaque branche de $\xi\tan\xi$ partant de $\xi = n\pi$ et de
$-\xi\cot\xi$ partant de $(n + \tfrac12)\pi$ coupe le cercle une fois si elle
part à l'intérieur~: d'où le compte.
\end{proof}

\begin{example}[Un électron dans un puits nanométrique]\label{ex:b2:potential-wells-tunneling:finite}
$V_0 = \qty{1}{eV}$, largeur $2a = \qty{1}{nm}$~: $R = a\sqrt{2mV_0}/\hbar = 0.5 \times
10^{-9} \times 5.1 \times 10^9 = 2.6$, donc $1 + \lfloor 2R/\pi\rfloor = 2$ états
liés~; l'état fondamental se situe vers \qty{0.26}{eV} au lieu des \qty{0.38}{eV}
du puits infini, et déborde de \qty{0.2}{nm} dans les parois
($1/\kappa = \hbar/\sqrt{2m(V_0 - E)}$). De tels puits, obtenus en couches de
semi-conducteurs de quelques nanomètres d'épaisseur, sont le cœur des diodes
laser du \cref{ch:b2:laser}.
\end{example}

\section{Marche et barrière~: l'effet tunnel}

\begin{proposition}[Marche de potentiel]\label{prop:b2:potential-wells-tunneling:step}
Une particule d'énergie $E$ venant de $x < 0$ sur la marche $V = V_0$
pour $x > 0$~:
\begin{itemize}
\item $E > V_0$~: $\varphi = \eu^{\iu k_1x} + r\eu^{-\iu k_1x}$ à gauche, $t\eu^{\iu
      k_2x}$ à droite, avec $k_{1,2} = \sqrt{2m(E - V_{0,\text{gauche/droite}})}/
      \hbar$~; la continuité de $\varphi$ et de $\varphi'$ donne $r = (k_1 - k_2)/(k_1
      + k_2)$ et la probabilité de réflexion (rapport des courants) $R = r^2$,
      $T = 1 - R = 4k_1k_2/(k_1 + k_2)^2$ --- une particule \emph{peut être
      réfléchie par une marche qu'elle a l'énergie de gravir}, exactement comme une onde sur
      une corde par un changement d'impédance (\cref{ch:b2:wave-interfaces})~;
\item $E < V_0$~: à droite $\varphi = t\eu^{-\kappa x}$, $\kappa = \sqrt{2m(V_0
      - E)}/\hbar$ --- une \emph{onde évanescente}\index{onde évanescente (quantique)}~; $|r| = 1$ (\omterm{thm:b2:wave-interfaces:snell}{réflexion totale}, avec un déphasage), aucun
      courant n'entre dans la marche, mais la particule s'y trouve avec une
      probabilité décroissant sur $1/\kappa$.
\end{itemize}
\end{proposition}

\begin{proof}
Écrire les \omterm{thm:b2:fluid-kinematics:continuity}{équations de continuité} en $x = 0$ et résoudre~; pour les courants
utiliser $j = (|A|^2 - |B|^2)\hbar k/m$ (\cref{ch:b2:schrodinger-wave-functions}).
Pour $E < V_0$, $r = (k - \iu\kappa)/(k + \iu\kappa)$, de module un.
\end{proof}

\begin{theorem}[Effet tunnel à travers une barrière]\label{thm:b2:potential-wells-tunneling:tunnel}
Une particule d'énergie $E < V_0$ rencontrant une barrière rectangulaire de hauteur
$V_0$ et de largeur $a$ est transmise avec la probabilité
\[
T = \Big[1 + \frac{V_0^2\sinh^2(\kappa a)}{4E(V_0 - E)}\Big]^{-1}
\ \approx\ 16\,\frac{E}{V_0}\Big(1 - \frac{E}{V_0}\Big)\eu^{-2\kappa a}
\quad (\kappa a \gg 1), \qquad
\kappa = \frac{\sqrt{2m(V_0 - E)}}{\hbar} .
\]
C'est l'\emph{effet tunnel}\index{effet tunnel}~: classiquement
impossible, quantiquement exponentiellement petit --- et
exponentiellement sensible à la largeur $a$, à la hauteur $V_0 - E$ et à la
masse $m$. Pour un électron à \qty{1}{eV} sous le sommet, $\kappa = \qty{5.1}{nm^{-1}}$~:
$T \sim 10^{-2}$ pour $a = \qty{0.5}{nm}$, $10^{-4}$ pour \qty{1}{nm}, $10^{-9}$ pour
\qty{2}{nm}~; pour un proton, $\kappa$ est $43$ fois plus grand et rien ne passe
à ces largeurs.
\end{theorem}

\begin{proof}
$\varphi = \eu^{\iu kx} + r\eu^{-\iu kx}$ pour $x < 0$, $A\eu^{\kappa x} + B\eu^{-\kappa x}$
à l'intérieur, $t\eu^{\iu kx}$ pour $x > a$~; quatre conditions de continuité (sur $\varphi$
et $\varphi'$ en $0$ et en $a$) pour quatre inconnues~; en éliminant $A$, $B$, $r$
on obtient $1/|t|^2 = 1 + (k^2 + \kappa^2)^2\sinh^2(\kappa a)/4k^2\kappa^2$, qui est
la formule avec $k^2 = 2mE/\hbar^2$, $\kappa^2 = 2m(V_0 - E)/\hbar^2$. Pour
$\kappa a \gg 1$, $\sinh\kappa a \approx \eu^{\kappa a}/2$. Pour une barrière de
forme quelconque l'exposant devient $2\int\kappa(x)\dd x$ à travers la
région interdite (le facteur de Gamow, admis).
\end{proof}

\begin{example}[Le microscope à effet tunnel]\label{ex:b2:potential-wells-tunneling:stm}
Une pointe métallique acérée est amenée à $d \approx \qty{0.5}{nm}$ d'une
surface conductrice et une petite tension est appliquée~: les électrons passent par \omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel}
à travers l'intervalle de vide, dont la hauteur de barrière est le travail de sortie $\Phi \approx
\qty{4.5}{eV}$, si bien que $\kappa = \sqrt{2m\Phi}/\hbar = \qty{1.1e10}{m^{-1}}$ et
le courant $I \propto \eu^{-2\kappa d}$ change d'un facteur $\eu^{2.2} \approx
10$ tous les \qty{0.1}{nm}. Une boucle d'asservissement déplace la pointe pour maintenir $I$
constant, et la hauteur de la pointe, enregistrée pendant le balayage, dessine la surface
avec une résolution verticale du picomètre --- et latéralement atome par
atome, parce que le dernier atome de la pointe, étant le plus proche, porte l'essentiel
du courant (Binnig et Rohrer, 1981).
\end{example}

\begin{example}[La désintégration alpha]\label{ex:b2:potential-wells-tunneling:alpha}
Une particule $\alpha$ de \qty{5}{MeV} à l'intérieur d'un noyau lourd affronte la
barrière coulombienne de la charge restante $Ze$, haute de plusieurs dizaines de MeV à la
surface nucléaire et s'étendant jusqu'au rayon $b = 2Ze^2/4\pi\varepsilon_0E
\approx \qty{50}{fm}$ où $V = E$. Le facteur de Gamow $2\int\kappa\,\dd r$ est
de l'ordre de $80$~: $T \sim \eu^{-80} \sim 10^{-35}$. La particule frappe la paroi
quelque $10^{21}$ fois par seconde, de sorte qu'elle s'échappe au rythme $\sim 10^{-14}
\,\unit{s^{-1}}$~: une durée de vie d'un million d'années. Comme l'exposant
varie en $Z/\sqrt E$, un changement de $\qty{2}{MeV}$ de $E$ décale la durée de vie
de vingt ordres de grandeur --- la loi de Geiger--Nuttall, de la
microseconde à l'âge de l'univers (Gamow, 1928~: la première
application de l'\omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel}).
\end{example}

\section{Le double puits}

\begin{proposition}[Levée de dégénérescence par effet tunnel]\label{prop:b2:potential-wells-tunneling:double}
Deux puits identiques séparés par une barrière~: si la barrière était
impénétrable chaque puits aurait le même niveau fondamental $E_0$, deux fois.
L'\omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel} les couple et les deux états les plus bas du double puits
sont les combinaisons \emph{symétrique} et \emph{antisymétrique},
$\varphi_\pm \approx (\varphi_{\text{L}} \pm \varphi_{\text{R}})/\sqrt2$, d'énergies
$E_0 \mp \Delta/2$~: le niveau est \emph{dédoublé} d'une quantité $\Delta \propto
\eu^{-\kappa a}$ (l'amplitude de passage, non son carré). Une particule
placée dans le puits de gauche, $\psi = (\varphi_+ + \varphi_-)/\sqrt2$, n'est pas
stationnaire~: elle oscille entre les puits à la fréquence
$\nu = \Delta/h$ --- les \emph{oscillations tunnel}, les battements à deux niveaux du
\cref{ch:b2:schrodinger-wave-functions}.
\end{proposition}

\begin{proof}
La combinaison symétrique n'a pas de nœud sous la barrière et une courbure
plus faible, donc une énergie plus basse~; l'antisymétrique a un nœud et
se situe plus haut~; le dédoublement est proportionnel au recouvrement des fonctions
localisées sous la barrière, $\eu^{-\kappa a}$. Le battement est
la superposition à deux niveaux déjà calculée.
\end{proof}

\begin{example}[L'ammoniac]\label{ex:b2:potential-wells-tunneling:ammonia}
Dans NH$_3$ l'atome d'azote se tient d'un côté du plan des trois
hydrogènes ou de l'autre --- un double puits avec une barrière d'environ
\qty{0.25}{eV}. Le dédoublement tunnel du niveau fondamental vaut $\Delta =
\qty{1e-4}{eV}$~: $\nu = \Delta/h = \qty{24}{GHz}$, la fréquence d'\emph{inversion},
la transition du premier maser et de la première horloge
atomique (1949). Remplacez l'hydrogène par du deutérium et la molécule, plus lourde,
passe moins~: \qty{1.6}{GHz}. Dans des pyramides plus lourdes (PH$_3$, AsH$_3$) le
dédoublement s'effondre au kilohertz et au-dessous~: la molécule reste de son
côté des années durant --- d'où l'existence même de molécules gauches et droites,
et le fait que le sucre ne se racémise pas sur l'étagère.
\end{example}

\begin{method}[Puits et barrières]\label{meth:b2:potential-wells-tunneling:method}
(1) Écrire $\varphi$ région par région~: $\eu^{\pm\iu kx}$ là où $E > V$,
$\eu^{\pm\kappa x}$ là où $E < V$, avec $k, \kappa = \sqrt{2m|E - V|}/\hbar$.
(2) Raccorder $\varphi$ et $\varphi'$ à chaque frontière (à une paroi infinie,
$\varphi = 0$). (3) États liés~: le raccordement n'a de solutions que pour des
$E$ discrets --- les compter graphiquement. (4) Diffusion~: les courants
$(|A|^2 - |B|^2)\hbar k/m$ donnent $R$ et $T$. (5) \omterm{thm:b2:potential-wells-tunneling:tunnel}{Effet tunnel}~: $T \sim
\eu^{-2\kappa a}$~; l'ordre de grandeur d'abord, le préfacteur ensuite. (6)
Double puits~: dédoublement $\propto\eu^{-\kappa a}$, battements à $\Delta/h$.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:1}
Puits infini~: un électron dans \qty{1}{nm} ($E_1$, $E_2$, $E_3$, longueur d'onde du

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:2}
Boîtes quantiques~: un nanocristal de bande interdite \qty{1.74}{eV} confine une
paire électron--trou dont l'\omterm{prop:b2:potential-wells-tunneling:infinite}{énergie de confinement} vaut $h^2/8m^*L^2$ avec
$m^* = 0.1\,m_{\text{e}}$. Longueur d'onde d'émission pour $L = 6$, $4$, $3$,
\qty{2.5}{nm}~; quelles couleurs~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:3}
\omterm{thm:b2:potential-wells-tunneling:tunnel}{Effet tunnel}~: un électron avec $V_0 - E = \qty{4.5}{eV}$~: $\kappa$~; $\eu^{-2\kappa a}$
pour $a = 0.3$, $0.5$, $1$, \qty{2}{nm}~; facteur par \qty{0.1}{nm}~; de même pour
un proton à $a = \qty{0.3}{nm}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:4}
Marche~: un électron de \qty{2}{eV} sur une marche de \qty{1}{eV}~: $k_1$, $k_2$,
$R$, $T$. De même pour une marche descendante de \qty{1}{eV}. Pourquoi est-ce impossible
classiquement, et quel est l'analogue pour la lumière~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:5}
\emph{Le puits infini en détail.} (a) Établir les niveaux et normaliser.
(b) Montrer l'orthogonalité. (c) Pour $n = 1$~: $\langle x\rangle$, $\Delta x =
L\sqrt{1/12 - 1/2\pi^2}$, $\langle p\rangle = 0$, $\Delta p = \pi\hbar/L$~; vérifier
Heisenberg. (d) Esquisser $|\varphi_n|^2$ pour $n$ grand et comparer à la
probabilité classique de trouver une particule qui rebondit.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:6}
\emph{Puits fini.} (a) Établir $k\tan ka = \kappa$ pour les états pairs.
(b) Montrer graphiquement qu'il y a toujours un état lié et les compter pour
$R = 1$, $4$, $10$. (c) Électron, $V_0 = \qty{1}{eV}$, $2a = \qty{1}{nm}$~:
$R$, nombre d'états, profondeur de pénétration de l'état fondamental (prendre
$E \approx \qty{0.26}{eV}$). (d) Qu'advient-il du nombre d'états liés
quand $V_0 \to \infty$, et de leurs énergies~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:7}
\emph{Marche, $E < V_0$.} (a) Écrire $\varphi$ des deux côtés et trouver $r =
(k - \iu\kappa)/(k + \iu\kappa)$. (b) Montrer $|r| = 1$ et calculer la phase
de $r$. (c) Montrer que le courant s'annule pour $x > 0$ bien que $|\varphi|^2 \ne
0$ là. (d) Un électron à \qty{1}{eV} sous le sommet~: profondeur de pénétration~;
comparer à l'\omterm{def:b2:dispersion-wave-packets:complex}{onde évanescente} de la \omterm{thm:b2:wave-interfaces:snell}{réflexion totale}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:8}
\emph{La barrière.} (a) Écrire les quatre conditions de continuité et en
tirer $T$. (b) L'évaluer exactement puis avec l'approximation pour $E =
V_0/2$, $\kappa a = 3$~; pour $\kappa a = 1$. (c) Pour $E > V_0$ montrer $T = [1 +
V_0^2\sin^2(k_2a)/4E(E - V_0)]^{-1}$ et trouver les énergies de transmission
parfaite. (d) Les interpréter (penser à la couche antireflet du
\cref{ch:b2:wave-interfaces}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:9}
\emph{Gamow.} Pour la barrière coulombienne du rayon nucléaire $R_{\text{n}}$
à $b = 2Ze^2/4\pi\varepsilon_0E$, l'exposant vaut $G = 2\int\kappa\,\dd r =
(2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b} - \sqrt{(R_{\text{n}}/b)(1 -
R_{\text{n}}/b)}]$. (a) $Z = 90$, $E = \qty{5}{MeV}$, $R_{\text{n}} = \qty{8}{fm}$~:
$b$, $G$, $T$. (b) Fréquence des chocs $v/2R_{\text{n}}$ et durée de vie. (c)
Refaire pour $E = 4$ et \qty{6}{MeV}~: rapport des durées de vie. (d) Pourquoi les
noyaux émettent-ils des particules $\alpha$ plutôt que des protons ou du $^{12}$C~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:10}
\emph{Ammoniac.} Dédoublement $\Delta = \qty{1e-4}{eV}$. (a) Fréquence d'inversion
et période de l'oscillation tunnel. (b) L'atome de N est
préparé d'un côté~: écrire $\psi(t)$ et la probabilité de le trouver
de l'autre côté. (c) ND$_3$ passe à \qty{1.6}{GHz}~: en déduire $\kappa a$
à partir du rapport des dédoublements si $\Delta \propto \eu^{-\kappa a}$ et $\kappa
\propto\sqrt m$ ($m_{\text{D}} = 2m_{\text{H}}$, barrière inchangée). (d) Pourquoi une
molécule chirale peut-elle rester gauche des années~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:11}
\emph{Le STM.} $\Phi = \qty{4.5}{eV}$, $d = \qty{0.5}{nm}$, tension \qty{0.1}{V},
$I = \qty{1}{nA}$. (a) $\kappa$, le facteur sur $I$ par \qty{0.1}{nm}, la
résolution en hauteur si $I$ est mesuré à $1\%$ près. (b) Électrons par seconde
dans le courant. (c) L'atome d'apex de la pointe est \qty{0.25}{nm} plus près que ses
voisins (à \qty{0.25}{nm} latéralement, donc $d' = \sqrt{d^2 + 0.25^2}$)~:
fraction du courant qu'il porte. (d) Un bâti d'acier de \qty{1}{cm} se dilate
de $\alpha L\Delta T$ avec $\alpha = \qty{1e-5}{K^{-1}}$~: quelle stabilité en température

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:12}

\end{exercise}

\section{Problème~: le STM, la particule alpha et le maser à ammoniac}

\begin{problem}[{Problème du week-end --- trois tunnels}]\label{pb:b2:potential-wells-tunneling:1}
Données~: $\hbar = \qty{1.05e-34}{J.s}$, $h = \qty{6.63e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $m_\alpha = \qty{6.64e-27}{kg}$, $e^2/4\pi\varepsilon_0 =
\qty{1.44}{MeV.fm}$, \qty{1}{eV} $= \qty{1.60e-19}{J}$.

\textbf{Partie I --- Le microscope à \omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel}.}
Pointe et échantillon sont en tungstène ($\Phi = \qty{4.5}{eV}$)~; intervalle $d$~; tension
\qty{50}{mV}~; le courant vaut $I = I_0\eu^{-2\kappa d}$.
\begin{enumerate}
  \item $\kappa$ pour des électrons au niveau de Fermi face à la barrière de
        vide $\Phi$.
  \item Facteur de variation de $I$ pour $\Delta d = \qty{0.1}{nm}$~; pour
        \qty{1}{pm}.
  \item L'asservissement tient $I$ constant à $1\%$ près~: bruit en hauteur.
  \item $I = \qty{1}{nA}$~: électrons par seconde~; temps moyen entre deux
        électrons, comparé au «~temps~» de passage $\sim\hbar/\Phi$.
  \item L'atome d'apex et ses voisins à \qty{0.25}{nm} latéralement,
        $d = \qty{0.5}{nm}$~: part de l'atome d'apex~; pourquoi un seul atome
        suffit à imager des atomes.
  \item Un atome adsorbé haut de \qty{0.1}{nm}~: variation de $I$ en
        mode hauteur constante~; pourquoi le mode courant constant est préféré.
  \item Un bâti métallique de \qty{2}{cm}, $\alpha = \qty{1e-5}{K^{-1}}$~: variation de
        température qui déplace la pointe de \qty{1}{pm}~; comment les instruments
        s'en accommodent-ils (symétrie, rapidité, matériaux à faible dilatation)~?
  \item Sous une tension de \qty{1}{V} la barrière n'est plus rectangulaire~:
        l'esquisser et dire si $I$ croît plus vite ou moins vite que
        linéairement avec $V$.
  \item Sur une plage oxydée le travail de sortie vaut \qty{3}{eV} au lieu
        de \qty{4.5}{eV}~: de combien la pointe recule-t-elle en
        mode courant constant au-dessus d'une surface parfaitement plane~? Qu'image
        alors le STM~?
\end{enumerate}

\textbf{Partie II --- Désintégration alpha d'un noyau $Z = 92$, $A = 238$.}
L'$\alpha$ ($E = \qty{4.2}{MeV}$) se déplace dans un noyau de rayon
$R_{\text{n}} = \qty{8}{fm}$~; à l'extérieur, l'énergie coulombienne vaut $V(r) =
2(Z - 2)e^2/4\pi\varepsilon_0r$.
\begin{enumerate}[resume]
  \item Hauteur de la barrière en $R_{\text{n}}$~; point de rebroussement classique
        $b$.
  \item Vitesse de l'$\alpha$ à l'intérieur et sa fréquence de chocs sur la
        paroi.
  \item Exposant de Gamow $G = (2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b}
        - \sqrt{(R_{\text{n}}/b)(1 - R_{\text{n}}/b)}]$ et $T = \eu^{-G}$.
  \item Taux de désintégration et demi-vie~; comparer aux $4.5 \times
        10^9$ ans mesurés.
  \item La même chose avec $E = \qty{5.2}{MeV}$ (un autre isotope)~: demi-vie~;
        la sensibilité de Geiger--Nuttall.
  \item En prenant $R_{\text{n}} = \qty{9}{fm}$ au lieu de \qty{8}{fm}~: effet sur
        la demi-vie~; commenter la précision de telles estimations.
  \item Pourquoi le passage par \omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel} d'un proton (charge $1$, masse $1/4$)
        n'est-il pas observé depuis le même noyau, ni celui d'un $^{12}$C
        (charge $6$)~?
  \item L'$\alpha$ émerge avec la totalité des \qty{4.2}{MeV} bien qu'elle soit
        «~passée sous~» un mur de \qty{30}{MeV}~: l'énergie est-elle conservée~?
\end{enumerate}

\textbf{Partie III --- Le maser à ammoniac.}
L'atome de N passe par \omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel} à travers le plan des H$_3$~; le niveau fondamental est dédoublé
de $\Delta = \qty{9.8e-5}{eV}$.
\begin{enumerate}[resume]
  \item Fréquence et longueur d'onde de la transition d'inversion.
  \item Écrire les états symétrique et antisymétrique en fonction de
        «~N en haut~» et «~N en bas~»~; lequel est le plus bas, et pourquoi~?
  \item Une molécule préparée «~N en haut~»~: $\psi(t)$, la probabilité d'être
        «~en bas~» à l'instant $t$, la période.
  \item Rapport de Boltzmann des deux niveaux à \qty{300}{K}~: la
        population thermique peut-elle donner un maser~? Comment obtient-on l'inversion
        (un champ électrique sélecteur d'état)~?
  \item Énergie du photon en joules~; nombre de molécules qui doivent émettre
        par seconde pour une sortie de \qty{1e-10}{W}.
  \item ND$_3$ passe à \qty{1.6}{GHz}~: en déduire $\kappa a$ pour NH$_3$
        en supposant $\Delta \propto \eu^{-\kappa a}$ avec $\kappa \propto \sqrt{m}$.
  \item Le maser fut la première horloge atomique~: pourquoi une fréquence de
        passage tunnel est-elle une bonne horloge et quels effets la décalent~?
  \item Résumer~: la quantification vient du confinement, l'\omterm{thm:b2:potential-wells-tunneling:tunnel}{effet tunnel} de
        l'évanescence, et ce qui fixe les exposants.
\end{enumerate}
\end{problem}
