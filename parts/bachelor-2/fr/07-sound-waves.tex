\chapter{Ondes sonores dans les fluides}\label{ch:b2:sound-waves}

Comptez les secondes entre l'éclair et le tonnerre et divisez par
trois~: c'est la distance en kilomètres. Le son traverse l'air à un
tiers de kilomètre par seconde, l'eau à un kilomètre et demi, et
un échographe chronomètre au millionième de seconde les échos venus de l'intérieur du corps
pour en dessiner l'image. Ce chapitre applique la
mécanique des fluides des \cref{ch:b2:euler-bernoulli,ch:b2:fluid-kinematics}
à un fluide au repos troublé par une petite ondulation de pression, et y retrouve
l'\omterm{thm:b2:waves-on-strings:equation}{équation de d'Alembert}~: la \omterm{thm:b2:sound-waves:equation}{célérité du son}, ce qui transporte
l'énergie, ce que veut dire «~fort~», comment le son se répand dans l'espace, et pourquoi une sirène
change de hauteur en passant.

\section{L'approximation acoustique}

\begin{definition}[Perturbation acoustique]\label{def:b2:sound-waves:approximation}
\index{approximation acoustique}\index{surpression}
Un fluide au repos, de pression uniforme $P_0$ et de masse volumique $\rho_0$, est
perturbé~: $P = P_0 + p(M, t)$, $\rho = \rho_0 + \rho_1(M, t)$, vitesse
$\vect v(M, t)$. L'\emph{approximation acoustique} ne garde que le premier
ordre en les petites grandeurs $p$ (la \emph{surpression}, ou pression
acoustique), $\rho_1$ et $\vect v$~: $|p| \ll P_0$, $|\rho_1| \ll \rho_0$, $v \ll c$,
et la transformation subie par chaque \omterm{def:b2:fluid-kinematics:particle}{particule de fluide} est si rapide
qu'elle n'échange aucune chaleur~: elle est \emph{isentropique}, avec la
compressibilité $\chi_S = \frac1\rho\bigl(\frac{\partial\rho}{\partial P}\bigr)_S$.
La pesanteur est négligée.
\end{definition}

\begin{theorem}[Ondes sonores]\label{thm:b2:sound-waves:equation}
\index{célérité du son}\index{équation d'onde (acoustique)}
Au premier ordre, l'\omterm{thm:b2:euler-bernoulli:euler}{équation d'Euler}, la conservation de la masse et la loi
isentropique s'écrivent
\[
\rho_0\frac{\partial\vect v}{\partial t} = -\operatorname{\vect{grad}}p , \qquad
\frac{\partial\rho_1}{\partial t} + \rho_0\operatorname{div}\vect v = 0 , \qquad
\rho_1 = \rho_0\chi_Sp ,
\]
et ensemble elles donnent l'\omterm{thm:b2:waves-on-strings:equation}{équation d'onde} pour la \omterm{def:b2:sound-waves:approximation}{surpression},
\[
\Delta p = \frac1{c^2}\frac{\partial^2p}{\partial t^2} , \qquad
c = \frac1{\sqrt{\rho_0\chi_S}} ,
\]
(la même pour $\rho_1$ et pour $\vect v$). Pour un gaz parfait, $\chi_S =
1/\gamma P_0$ et
\[
c = \sqrt{\frac{\gamma P_0}{\rho_0}} = \sqrt{\frac{\gamma RT}{M}} :
\]
\qty{343}{m/s} dans l'air à \qty{20}{\degreeCelsius}, en hausse de \qty{0.6}{m/s}
par kelvin~; \qty{1000}{m/s} dans l'hélium~; \qty{1480}{m/s} dans l'eau ($\chi_S =
\qty{4.5e-10}{Pa^{-1}}$).
\end{theorem}

\begin{proof}
Euler~: le terme convectif $\rho(\vect v\cdot\operatorname{\vect{grad}})\vect v$
est du second ordre, $\rho\,\partial_t\vect v \approx \rho_0\,\partial_t\vect v$, et
$\operatorname{\vect{grad}}P = \operatorname{\vect{grad}}p$. Masse~: $\partial_t\rho +
\operatorname{div}(\rho\vect v) \approx \partial_t\rho_1 + \rho_0\operatorname{div}\vect v$.
Loi isentropique~: $\dd\rho = \rho\chi_S\,\dd P$ au premier ordre. Prenons la
divergence de la première équation et la dérivée temporelle de la seconde~:
$\rho_0\partial_t\operatorname{div}\vect v = -\Delta p = -\partial_t^2\rho_1 = -\rho_0\chi_S
\partial_t^2p$. Pour le gaz parfait, $PV^\gamma$ constant le long d'une transformation
isentropique donne $\dd\rho/\rho = \dd P/\gamma P$, d'où $\chi_S = 1/\gamma P$,
et $P_0/\rho_0 = RT/M$.
\end{proof}

\begin{remark}[Pourquoi isentropique]\label{rem:b2:sound-waves:laplace}
Newton avait calculé la \omterm{thm:b2:sound-waves:equation}{célérité du son} avec la compressibilité isotherme,
$\sqrt{P_0/\rho_0} = \qty{290}{m/s}$, $15\%$ trop bas~; Laplace vit que les
compressions sont trop rapides pour que la chaleur passe des crêtes chaudes aux
creux froids --- sur une période, la chaleur ne diffuse que de quelques micromètres
(\cref{ch:b2:heat-conduction}), rien au regard d'une longueur d'onde ---
si bien que la transformation est adiabatique et réversible, et que $\gamma$ apparaît.
La \omterm{thm:b2:sound-waves:equation}{célérité du son} est ainsi une mesure directe de $\gamma$~: $1.40$ pour
l'air, $1.67$ pour l'argon.
\end{remark}

\section{Ondes planes, impédance, intensité}

\begin{proposition}[Onde sonore plane progressive]\label{prop:b2:sound-waves:plane}
\index{impédance acoustique}\index{intensité acoustique}\index{densité d'énergie (acoustique)}
Pour une onde plane $p = f(x - ct)$ se propageant vers les $+x$, la vitesse du
fluide est selon $x$, en phase avec la \omterm{def:b2:sound-waves:approximation}{surpression}, et
\[
p = Z\,v , \qquad Z = \rho_0c = \sqrt{\rho_0/\chi_S} ,
\]
où $Z$ est l'\emph{\omterm{prop:b2:sound-waves:plane}{impédance acoustique}} du milieu
(\unit{kg.m^{-2}.s^{-1}})~: \qty{410}{kg.m^{-2}.s^{-1}} pour l'air, \qty{1.5e6}{kg.m^{-2}.s^{-1}}
pour l'eau, $\sim\qty{1.6e6}{kg.m^{-2}.s^{-1}}$ pour les tissus mous,
\qty{4e7}{kg.m^{-2}.s^{-1}} pour l'acier. L'onde transporte la densité d'énergie et
l'intensité (puissance par unité d'aire, vers les $+x$)
\[
e = \tfrac12\rho_0v^2 + \tfrac12\chi_Sp^2 , \qquad I = p\,v ,
\]
qui vérifient $\partial_te + \partial_xI = 0$~; pour l'onde progressive les deux
moitiés de $e$ sont égales et $I = p^2/Z = Zv^2$. Une onde sinusoïdale
d'amplitude de pression $p_m$ a l'intensité moyenne $\langle I\rangle = p_m^2/2Z$.
\end{proposition}

\begin{proof}
Pour $p = f(x - ct)$, $\rho_0\partial_tv = -\partial_xp = f'/c\cdot c\dots$~: précisément,
$\rho_0\partial_tv = -f'(x - ct)$, et $v = f(x - ct)/\rho_0c$ convient. Le
terme potentiel de $e$ est le travail emmagasiné en comprimant le fluide
(une compression isentropique d'un volume unité de $\dd V/V = -\chi_S\,\dd p$
emmagasine $\int p\,\chi_S\dd p = \tfrac12\chi_Sp^2$)~; $I$ est la puissance de la
force de pression $pS$ sur le fluide qui précède, animé de $v$, par unité d'aire.
Bilan~: $\partial_te = \rho_0v\partial_tv + \chi_Sp\partial_tp = -v\partial_xp - p\partial_xv =
-\partial_x(pv)$ d'après les deux équations linéarisées.
\end{proof}

\begin{definition}[Niveau sonore]\label{def:b2:sound-waves:decibel}
\index{décibel}\index{niveau sonore}\index{seuil d'audition}
Le \emph{niveau sonore} d'une onde d'intensité moyenne $I$ vaut
\[
L = 10\log_{10}\frac I{I_0} \ \text{dB} , \qquad I_0 = \qty{1e-12}{W/m^2}
\]
--- le seuil d'audition à \qty{1}{kHz}, qui correspond dans l'air à une
amplitude de pression $p_0 \approx \qty{2e-5}{Pa}$ (efficace) $= 2 \times 10^{-10}$
atmosphère. Doubler l'intensité ajoute \qty{3}{dB}~; la multiplier par dix, \qty{10}{dB}~;
multiplier par cent l'amplitude de pression, \qty{40}{dB}. Pièce calme
\qty{30}{dB}, conversation \qty{60}{dB}, rue passante \qty{80}{dB}, concert de rock
\qty{110}{dB}, douleur \qty{120}{dB} ($I = \qty{1}{W/m^2}$, $p_m = \qty{29}{Pa}$).
\end{definition}

\begin{example}[Combien un son est petit]\label{ex:b2:sound-waves:small}
Au seuil, à \qty{1}{kHz}~: $p_m = \qty{2.8e-5}{Pa}$, $v_m = p_m/Z =
\qty{7e-8}{m/s}$, et l'amplitude de déplacement $\xi_m = v_m/\omega =
\qty{1e-11}{m}$ --- le dixième d'un diamètre atomique~: le tympan détecte des
mouvements plus petits qu'un atome. À \qty{120}{dB}, $\xi_m = \qty{11}{\micro m}$,
encore invisible, et $p_m/P_0 = 3 \times 10^{-4}$~: même les sons les plus intenses
sont de minuscules perturbations, d'où l'excellence de la théorie linéaire.
\end{example}

\section{Ondes sphériques}

\begin{proposition}[Ondes sphériques et loi en carré inverse]\label{prop:b2:sound-waves:spherical}
\index{onde sphérique}\index{loi en carré inverse (son)}
Une petite source rayonnant également dans toutes les directions produit
l'\omterm{prop:b2:sound-waves:spherical}{onde sphérique}
\[
p(r, t) = \frac{A}{r}\,f\Bigl(t - \frac rc\Bigr) ,
\]
dont l'amplitude décroît en $1/r$ et l'intensité en $1/r^2$~:
pour une source de puissance acoustique $\mathcal P$, $I = \mathcal P/4\pi r^2$, soit
$-\qty{6}{dB}$ chaque fois que la distance double. Loin de la source, l'onde
est localement plane, avec $v = p/Z$ radiale.
\end{proposition}

\begin{proof}
En coordonnées sphériques, pour une fonction de $r$ seul, $\Delta p =
\frac1r\frac{\partial^2(rp)}{\partial r^2}$ (admis~; \cref{ch:b2:maxwell-equations}),
de sorte que $u = rp$ obéit à l'\omterm{thm:b2:waves-on-strings:equation}{équation d'onde} unidimensionnelle $\partial_r^2u = \partial_t^2u/c^2$,
d'où $u = Af(t - r/c)$ pour l'onde sortante. La puissance qui traverse une
sphère de rayon $r$ vaut $4\pi r^2I$, constante si rien n'est absorbé.
\end{proof}

\begin{example}[Une sirène, une conversation]\label{ex:b2:sound-waves:siren}
Une sirène de \qty{10}{W}~: $I = 10/4\pi r^2$, \qty{0.8}{W/m^2} à \qty{1}{m} (\qty{119}{dB},
douleur), \qty{79}{dB} à \qty{100}{m}, \qty{59}{dB} à \qty{1}{km} --- encore audible dans
la ville. Une voix rayonne environ \qty{10}{\micro W}~: \qty{60}{dB} à \qty{1}{m},
et \qty{40}{dB} à \qty{10}{m}. L'air absorbe aussi le son (davantage aux hautes
fréquences, quelques \omterm{def:b2:sound-waves:decibel}{décibels} par centaine de mètres à \qty{10}{kHz}), d'où le
grondement grave du tonnerre lointain.
\end{example}

\section{L'effet Doppler}

\begin{theorem}[Effet Doppler]\label{thm:b2:sound-waves:doppler}
\index{effet Doppler}
Une source émet à la fréquence $f$ et se déplace à la vitesse $v_s$ selon la droite
qui la joint à un observateur se déplaçant à $v_o$, les deux vitesses étant comptées
positivement quand ils se rapprochent~; le son se propage à $c$ dans
l'air au repos. L'observateur reçoit la fréquence
\[
f' = f\,\frac{c + v_o}{c - v_s} .
\]
Une source en mouvement resserre ses fronts d'onde devant elle ($\lambda' = (c - v_s)/f$)
et les écarte derrière~; un observateur en mouvement rencontre les fronts d'onde au
rythme $(c + v_o)/\lambda$. Pour $v_s, v_o \ll c$ les deux donnent $\Delta f/f \approx
(v_s + v_o)/c$.
\end{theorem}

\begin{proof}
Source en mouvement~: les fronts émis en $t$ et en $t + 1/f$ sont séparés
de $c/f - v_s/f$ dans la direction du mouvement, si bien que la longueur d'onde en avant vaut
$\lambda' = (c - v_s)/f$ et que la fréquence reçue par un observateur fixe vaut
$c/\lambda'$. Observateur en mouvement~: la longueur d'onde vaut $\lambda = c/f$ mais les fronts passent
à la vitesse relative $c + v_o$. On combine.
\end{proof}

\begin{example}[Sirènes, cinémomètres, globules rouges]\label{ex:b2:sound-waves:dopplerex}
Une sirène d'ambulance à \qty{700}{Hz} qui passe à \qty{25}{m/s}~: \qty{755}{Hz}
à l'approche, \qty{651}{Hz} à l'éloignement --- une chute d'un sixième d'octave,
le familier «~pin-pon~». Un cinémomètre de police mesure le même
effet sur une onde radio de \qty{24}{GHz} réfléchie par une voiture (l'onde est
décalée deux fois, à la réception et à la réémission par la voiture)~: $\Delta f = 2fv/c =
\qty{4.8}{kHz}$ à \qty{30}{m/s}. Une sonde à ultrasons de \qty{5}{MHz} braquée sur
une artère entend les globules à $\Delta f = 2fv\cos\theta/c \approx \qty{1.6}{kHz}$
pour \qty{0.5}{m/s} à \qty{60}{\degree}~: un sifflement audible dont la hauteur donne la
vitesse du sang. Et les raies spectrales d'une galaxie qui s'éloigne sont décalées
vers le rouge de $\Delta\lambda/\lambda = v/c$ --- la formule vaut pour la lumière à faible vitesse,
la correction relativiste prenant le relais au-delà.
\end{example}

\begin{remark}[Supersonique]\label{rem:b2:sound-waves:mach}
\index{cône de Mach}\index{bang supersonique}
Lorsque $v_s > c$ la formule tombe en défaut ($f' < 0$ en avant)~: aucun son ne précède
la source. Les fronts d'onde émis le long de la \omterm{def:b2:fluid-kinematics:euler}{trajectoire} ont une enveloppe
conique de demi-angle $\theta$ avec $\sin\theta = c/v_s$ (le \emph{\omterm{rem:b2:sound-waves:mach}{cône de Mach}})~;
le saut de pression porté par le cône est le \omterm{rem:b2:sound-waves:mach}{bang supersonique}, qui n'atteint
un auditeur au sol qu'après le passage de l'avion --- à
Mach $2$ et \qty{10}{km} d'altitude, $\theta = \qty{30}{\degree}$ et le bang
arrive $10\,\text{km}/\tan\theta/v_s \approx \qty{25}{s}$ après le passage
à la verticale. Le claquement d'une balle et la pointe d'un fouet forment le même cône.
\end{remark}

\begin{method}[Ordres de grandeur en acoustique]\label{met:b2:sound-waves:method}
Étant donné un \omterm{def:b2:sound-waves:decibel}{niveau sonore} $L$~: $I = I_0\,10^{L/10}$~; puis $p_m = \sqrt{2ZI}$,
$v_m = p_m/Z$, $\xi_m = v_m/\omega$~; la puissance d'une source vaut $4\pi r^2I$ si elle
rayonne dans toutes les directions (deux fois moins au-dessus d'un sol dur, qui
réfléchit dans un demi-espace). Passer de $r_1$ à $r_2$ change $L$ de
$20\log_{10}(r_1/r_2)$. Ajouter $n$ sources égales incohérentes ajoute
$10\log_{10}n$~; deux sources cohérentes en phase en un point ajoutent \qty{6}{dB}.
\end{method}

\section{Exercices}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:1}
\omterm{thm:b2:sound-waves:equation}{Célérité du son} dans l'air ($\gamma = 1.40$, $M = \qty{29}{g/mol}$) à
\qty{0}{\degreeCelsius}, \qty{20}{\degreeCelsius} et à \qty{-50}{\degreeCelsius}
(\qty{10}{km} d'altitude)~; dans l'hélium ($\gamma = 1.67$, $M = \qty{4}{g/mol}$) et
le dioxyde de carbone ($\gamma = 1.30$, $M = \qty{44}{g/mol}$) à \qty{20}{\degreeCelsius}.
Pourquoi une voix devient-elle aiguë après avoir respiré de l'hélium~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:2}
Une conversation à \qty{60}{dB}, \qty{500}{Hz}~: intensité, amplitude de
pression, amplitude de vitesse et amplitude de déplacement dans l'air
($Z = \qty{410}{kg.m^{-2}.s^{-1}}$). Idem pour \qty{120}{dB}. Niveau de deux
personnes parlant en même temps~; de cent.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:3}
Un haut-parleur rayonne \qty{1.0}{W} de son également dans toutes les directions.
Intensité et niveau à \qty{1}{m}, \qty{10}{m}, \qty{100}{m}~; distance à laquelle
le niveau tombe à \qty{60}{dB}~; puissance nécessaire pour \qty{100}{dB} à \qty{30}{m}
(un concert en plein air).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:4}
Le tonnerre s'entend \qty{4.5}{s} après l'éclair~: distance. La corne d'un navire
renvoie un écho d'une falaise après \qty{3.0}{s}~: distance. Sonar~: un écho du
fond marin revient après \qty{2.4}{s} ($c = \qty{1500}{m/s}$)~: profondeur. Un
écho ultrasonore de \qty{5}{MHz} revient après \qty{130}{\micro s} dans les tissus
(\qty{1540}{m/s})~: profondeur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:5}
La corne d'un train à \qty{500}{Hz}~; $c = \qty{340}{m/s}$. (a) Fréquence entendue par une
personne sur le quai à l'approche du train à \qty{40}{m/s}, puis
après son passage. (b) La personne est dans un train qui se dirige vers la
corne à \qty{40}{m/s}, la corne étant au repos~; puis les deux vont l'un vers
l'autre à \qty{40}{m/s}~: comparer avec une vitesse relative de \qty{80}{m/s} et
un corps au repos. (c) Fréquence entendue par le conducteur du train porteur de la corne
dans l'écho renvoyé par un mur situé devant.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:6}
Pour une amplitude de pression de \qty{1.0}{Pa} à \qty{1}{kHz}, calculer les
amplitudes de vitesse et de déplacement, l'intensité et le niveau dans
l'air, dans l'eau ($Z = \qty{1.5e6}{kg.m^{-2}.s^{-1}}$) et dans l'acier
($\qty{4e7}{kg.m^{-2}.s^{-1}}$). Même pression, intensités très différentes~:
commenter. (Sous l'eau les niveaux sont rapportés à \qty{1}{\micro Pa}~: que vaut
\qty{1}{Pa} sur cette échelle~?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:7}
(a) Célérité isotherme de Newton $\sqrt{P_0/\rho_0}$ pour l'air à
\qty{20}{\degreeCelsius} ($\rho_0 = \qty{1.20}{kg/m^3}$) et correction de Laplace.
(b) Mesures~: \qty{343}{m/s} dans l'air, \qty{323}{m/s} dans l'argon ($M = \qty{40}{g/mol}$)
à \qty{20}{\degreeCelsius}~: en déduire $\gamma$ pour chacun~; interpréter avec le
nombre de degrés de liberté. (c) Estimer sur quelle distance la chaleur diffuse en
une période à \qty{1}{kHz} (diffusivité thermique de l'air
\qty{2e-5}{m^2/s}) et comparer à la longueur d'onde.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:8}
\emph{Tuyaux d'orgue.} Dans un tuyau la \omterm{def:b2:sound-waves:approximation}{surpression} est nulle à une extrémité ouverte
et la vitesse est nulle à une extrémité fermée. (a) \omterm{prop:b2:waves-on-strings:modes}{Ondes stationnaires} dans un tuyau
ouvert aux deux bouts~: montrer que les modes sont $f_n = nc/2L$~; dans un tuyau fermé à
une extrémité, $f_n = (2n - 1)c/4L$. (b) Longueurs pour un fondamental à \qty{440}{Hz}
dans chaque cas. (c) Quelles harmoniques chaque tuyau contient-il~? (d) L'extrémité
ouverte réelle se situe environ $0.6$ rayon au-delà du tuyau (correction d'extrémité)~: pour
un tuyau de \qty{4}{cm}, la correction sur $L$ et sur $f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:9}
Une sirène électrique de \qty{2.0}{kW} convertit $20\%$ de sa puissance en son
rayonné dans un demi-espace (elle est posée sur un toit). (a) Intensité, niveau
et amplitude de pression à \qty{100}{m}. (b) Distance du seuil de douleur
(\qty{120}{dB})~; du contour \qty{70}{dB} où elle cesse d'alerter.
(c) L'absorption de l'air ajoute \qty{1}{dB} par \qty{100}{m} à sa fréquence~:
distance du contour \qty{70}{dB} désormais (à résoudre numériquement). (d) Combien
de telles sirènes couvrent une ville de \qty{10}{km} de rayon~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:10}
\emph{Énergie d'une onde sonore stationnaire.} Dans un tube fermé de longueur $L$
la \omterm{def:b2:sound-waves:approximation}{surpression} vaut $p = p_m\cos(kx)\cos(\omega t)$, $k = n\pi/L$. (a) Trouver $v(x,
t)$ par l'\omterm{thm:b2:euler-bernoulli:euler}{équation d'Euler} linéarisée. (b) Densités d'énergie cinétique et
potentielle~; montrer qu'elles oscillent en quadrature dans le temps et
sont décalées d'un quart de longueur d'onde dans l'espace. (c) Énergie totale du
tube (section $S$)~; vérifier qu'elle est constante. (d) Intensité moyenne~: nulle
partout --- et pourtant l'énergie se déplace~: décrire comment.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:11}
\emph{\omterm{rem:b2:sound-waves:mach}{Bang supersonique}.} Un avion vole horizontalement à $v = 1.6\,c$ à
l'altitude $h = \qty{12}{km}$, où $c = \qty{300}{m/s}$. (a) Établir l'angle de
Mach à partir de l'enveloppe des fronts d'\omterm{prop:b2:sound-waves:spherical}{onde sphériques}. (b) Temps entre
le passage à la verticale et le bang au sol (négliger la variation
de $c$ avec l'altitude). (c) Largeur au sol de la bande qui entend
le bang au passage de l'avion, si la trace du cône s'entend à
$\pm\qty{30}{km}$ de la route~: pourquoi le vol supersonique au-dessus des terres
est-il interdit~? (d) Une balle de fusil à \qty{800}{m/s}~: angle de Mach~; ce qu'entend un observateur
placé à \qty{10}{m} de la \omterm{def:b2:fluid-kinematics:euler}{trajectoire}, et dans quel ordre.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:12}
\emph{Doppler ultrasonore.} Une sonde émet à $f = \qty{5.0}{MHz}$ dans les tissus
($c = \qty{1540}{m/s}$)~; les globules rouges se déplacent à $v$ le long d'un vaisseau faisant
l'angle $\theta$ avec le faisceau. (a) Fréquence reçue par un globule (observateur
en mouvement). (b) Il réémet (diffuse) cette fréquence en source mobile~:
fréquence de retour à la sonde~; montrer que $\Delta f = 2fv\cos\theta/c$ pour $v \ll c$. (c)
$\Delta f$ pour $v = \qty{0.50}{m/s}$, $\theta = \qty{60}{\degree}$~; pour $\theta = \qty{90}{\degree}$
--- que fait alors l'opérateur~? (d) La sonde résout \qty{50}{Hz}~:
résolution en vitesse. Pourquoi rend-on le signal Doppler audible au lieu de
l'afficher seulement~?
\end{exercise}

\section{Problème~: du tonnerre à l'échographe}

\begin{problem}[{Problème du week-end --- le son en plein air, dans une salle de
concert, à l'intérieur du corps et venant d'une sirène qui passe}]
\label{pb:b2:sound-waves:1}

Air~: $\gamma = 1.40$, $M = \qty{29}{g/mol}$, $Z = \qty{410}{kg.m^{-2}.s^{-1}}$ à
\qty{15}{\degreeCelsius}~; $R = \qty{8.31}{J/(mol.K)}$.

\medskip
\noindent\textbf{Partie I --- L'orage.}
\begin{enumerate}
  \item \omterm{thm:b2:sound-waves:equation}{Célérité du son} à \qty{15}{\degreeCelsius} et à \qty{30}{\degreeCelsius}~;
        justifier la règle des «~trois secondes par kilomètre~».
  \item Le tonnerre arrive \qty{4.0}{s} après l'éclair~: distance de
        l'impact. Le tonnerre dure plusieurs secondes alors que l'éclair est
        instantané~: pourquoi~?
  \item Près de l'impact le niveau atteint \qty{120}{dB}~: amplitude de pression
        et amplitude de vitesse de l'air.
  \item Énergie reçue par mètre carré de mur pendant un coup de tonnerre
        de \qty{2}{s} à \qty{120}{dB}.
  \item Par nuit claire le sol se refroidit et l'air est plus froid en bas
        qu'en haut. Les rayons sonores obéissent à la loi de Descartes entre couches,
        $\sin i/c = $ const~: les rayons lancés vers le haut s'incurvent-ils vers le bas ou vers le haut~?
        Pourquoi entend-on si bien les bruits lointains la nuit et au-dessus de l'eau~?
  \item Le jour, l'inverse~: un auditeur à \qty{3}{km} n'entend rien.
        Estimer l'angle dont un rayon horizontal s'est incurvé après
        \qty{3}{km} si $c$ décroît de \qty{1}{\%} par \qty{500}{m} d'altitude
        (traiter la courbure comme uniforme~: le rayon est un cercle).

  \item L'air absorbe les hautes fréquences bien plus que les basses~: pourquoi
        le tonnerre lointain gronde-t-il alors qu'un impact proche claque~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- Le concert.}
Un haut-parleur reçoit \qty{100}{W} de puissance électrique et en convertit
$2\%$ en son, rayonné également dans le demi-espace situé devant
lui.
\begin{enumerate}[resume]
  \item Puissance acoustique~; intensité et niveau à \qty{10}{m}.
  \item Amplitude de pression en ce point~; amplitudes de vitesse et de déplacement
        à \qty{100}{Hz}.
  \item Niveau à \qty{1}{m} (est-ce sans danger~?) et distance à laquelle la musique
        tombe à \qty{60}{dB}.
  \item Dix haut-parleurs identiques répartis autour de la scène, non
        en phase~: niveau à \qty{10}{m}. Deux haut-parleurs alimentés en phase, à la
        même distance d'un auditeur placé sur leur axe~: niveau en ce point.
  \item La salle (volume \qty{10000}{m^3}) présente \qty{2000}{m^2} de parois et de public qui absorbent $30\%$ du
        son qui les frappe~: estimer l'énergie acoustique emmagasinée dans la
        salle à l'équilibre et le temps de réverbération (temps que met le
        niveau pour chuter de \qty{60}{dB} après l'arrêt de la musique), à l'aide du
        bilan entre la puissance émise et la puissance absorbée
        ($I_{\text{paroi}} \approx ce/4$ pour un champ diffus, admis).
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- L'échographe.}
Tissus mous~: $c = \qty{1540}{m/s}$, $Z = \qty{1.6e6}{kg.m^{-2}.s^{-1}}$~; os
$Z = \qty{7e6}{kg.m^{-2}.s^{-1}}$~; air $\qty{400}{kg.m^{-2}.s^{-1}}$. Sous
incidence normale l'amplitude de pression réfléchie à une interface vaut $r = (Z_2 -
Z_1)/(Z_2 + Z_1)$ fois l'amplitude incidente (établi au
\cref{ch:b2:wave-interfaces})~; les tissus atténuent le son d'environ
\qty{0.5}{dB} par centimètre et par mégahertz.
\begin{enumerate}[resume]
  \item Longueur d'onde dans les tissus à \qty{3}{MHz}, \qty{5}{MHz}, \qty{10}{MHz}~; le
        plus petit détail que chacune résout vaut environ une longueur d'onde.
  \item Retard de l'écho d'un organe situé à \qty{12}{cm} de profondeur~; cadence maximale à laquelle
        on peut envoyer des impulsions sans confondre les échos~; temps pour former une
        image de $200$ lignes.
  \item Fraction de la pression, puis de l'énergie, réfléchie à une interface
        tissu--air~; à une interface tissu--os~; entre la graisse
        ($Z = \qty{1.38e6}{kg.m^{-2}.s^{-1}}$) et le muscle
        ($\qty{1.70e6}{kg.m^{-2}.s^{-1}}$). Pourquoi le gel, et pourquoi les os
        projettent-ils des ombres~?
  \item À \qty{5}{MHz} le faisceau traverse \qty{2}{cm} de graisse avant de rencontrer le
        muscle~: fraction de l'énergie émise qui revient à la
        sonde depuis cette interface (réflexion et atténuation
        aller-retour), en \omterm{def:b2:sound-waves:decibel}{décibels}.
  \item Atténuation aller-retour en \omterm{def:b2:sound-waves:decibel}{décibels} pour l'organe de la question 12
        à chacune des trois fréquences~; si l'appareil supporte
        \qty{80}{dB} de perte, quelles fréquences conviennent~? Énoncer le compromis
        entre résolution et profondeur.
  \item La sonde émet des impulsions de \qty{1}{\micro s} à \qty{5}{MHz}~: combien de
        longueurs d'onde dure une impulsion, et en quoi cela limite-t-il la résolution en
        profondeur~?
  \item Mode Doppler à \qty{5}{MHz} sur une artère à \qty{0.40}{m/s}, faisceau à
        \qty{60}{\degree}~: décalage~; ce qu'entend l'opérateur.
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- La sirène.}
\begin{enumerate}[resume]
  \item Une ambulance passe à \qty{90}{km/h} avec une sirène à \qty{700}{Hz}~:
        fréquences entendues à l'approche et à l'éloignement~; l'intervalle
        qui les sépare en demi-tons (un demi-ton est un rapport $2^{1/12}$).
  \item L'auditeur est dans une voiture qui roule à \qty{15}{m/s} vers
        l'ambulance qui approche~: fréquence entendue.
  \item À quel instant un auditeur sur le trottoir entend-il exactement
        \qty{700}{Hz}~?
  \item Les fronts d'onde de l'ambulance devant elle~: longueur d'onde~; vitesse du
        son par rapport à l'ambulance, devant et derrière.
  \item Un cinémomètre à \qty{24}{GHz} relève un décalage de \qty{3.2}{kHz} sur une voiture~:
        vitesse de la voiture (l'onde est décalée deux fois).
  \item Récapituler~: les six vitesses de ce problème (son dans l'air, dans
        les tissus, l'ambulance, la voiture, les globules rouges, la lumière) et la
        formule unique qui les traite toutes.
\end{enumerate}
\end{problem}
