\chapter{Ondes électromagnétiques dans les plasmas, les conducteurs et les diélectriques}\label{ch:b2:waves-in-media}

Une émission en ondes courtes traverse un océan en rebondissant sur la haute
atmosphère, qui la réfléchit comme un miroir --- et pourtant la même couche laisse
passer la télévision et les signaux des satellites. Une cuillère métallique est brillante, une
grille métallique dans la porte d'un four à micro-ondes retient les ondes pendant
que vous regardez vos aliments au travers, et une plaque de verre est transparente
à la lumière mais opaque à l'ultraviolet. Dans chaque cas une onde entre dans un
matériau, les charges du matériau répondent, et leur réponse modifie
l'onde~: sa vitesse, son \omterm{def:b2:dispersion-wave-packets:complex}{atténuation}, voire son existence même.
Ce chapitre traite les trois modèles usuels de milieu --- un
gaz d'électrons libres, un conducteur ohmique et des électrons liés --- par
la seule méthode qui les traite tous~: ajouter le courant induit aux
\omterm{thm:b2:maxwell-equations:maxwell}{équations de Maxwell} et y lire la \omterm{def:b2:dispersion-wave-packets:relation}{relation de dispersion}.

\section{Ondes dans un plasma}

\begin{definition}[Plasma~; le modèle de l'électron libre]\label{def:b2:waves-in-media:plasma}
\index{plasma}\index{pulsation plasma}
Un \emph{plasma} est un gaz globalement neutre d'électrons libres (densité
$n$, masse $m$, charge $-e$) et d'ions positifs qui, étant des milliers de
fois plus lourds, sont supposés fixes. La haute atmosphère (ionosphère,
$n \sim 10^{11}$ à $10^{12}$ m$^{-3}$), un tube à décharge, la couronne solaire
et --- aux fréquences optiques --- les électrons de conduction d'un métal
sont des plasmas. Les chocs sont négligés (la période de l'onde est courte
devant le temps entre deux chocs). Sa fréquence caractéristique
est la \emph{fréquence plasma}
\[
\omega_p = \sqrt{\frac{ne^2}{\varepsilon_0m}} , \qquad
f_p = \frac{\omega_p}{2\pi} \approx 9\,\text{Hz}\times\sqrt{n\,[\text{m}^{-3}]} :
\]
de \qty{3}{MHz} à \qty{9}{MHz} pour l'ionosphère, \qty{2e15}{Hz} (ultraviolet) pour
un métal.
\end{definition}

\begin{theorem}[Relation de dispersion d'un plasma]\label{thm:b2:waves-in-media:plasmadispersion}
\index{pulsation de coupure (plasma)}\index{onde évanescente (plasma)}
Pour une OPPH transversale $\underline{\vect E} = \underline{\vect E}_0\eu^{\iu(\omega t - kx)}$ dans
un \omterm{def:b2:waves-in-media:plasma}{plasma}, les électrons oscillent selon $\underline{\vect v} = -e\underline{\vect E}/\iu m\omega$
et portent la \omterm{def:b2:charges-currents-conduction:densities}{densité de courant} $\underline{\vect j} = \underline\gamma\underline{\vect E}$ avec
$\underline\gamma = ne^2/\iu m\omega$~; les \omterm{thm:b2:maxwell-equations:maxwell}{équations de Maxwell} donnent alors
\[
k^2 = \frac{\omega^2 - \omega_p^2}{c^2} .
\]
Au-dessus de $\omega_p$ l'onde se propage, avec $v_\varphi = c/\sqrt{1 - \omega_p^2/\omega^2}
> c$ et $v_g = c\sqrt{1 - \omega_p^2/\omega^2} < c$, $v_\varphi v_g = c^2$~; en dessous
de $\omega_p$, $k = -\iu\kappa$ avec $\kappa = \sqrt{\omega_p^2 - \omega^2}/c$~: l'onde est
\emph{évanescente}, pénètre sur $1/\kappa$ et est totalement réfléchie
(\cref{ch:b2:wave-interfaces})~; le \omterm{def:b2:waves-in-media:plasma}{plasma} est alors un miroir. Pour $\omega
\gg \omega_p$ les électrons ne peuvent plus suivre et le \omterm{def:b2:waves-in-media:plasma}{plasma} est transparent.
\end{theorem}

\begin{proof}
Équation du mouvement $m\,\dd\vect v/\dd t = -e\vect E$ (la force magnétique
$ev B \sim evE/c$ est négligeable pour $v \ll c$, et le mouvement des ions est
$m/M$ fois plus petit)~: $\iu m\omega\underline{\vect v} = -e\underline{\vect E}$. Puis
$\underline{\vect j} = -ne\underline{\vect v}$. Le \omterm{def:b2:waves-in-media:plasma}{plasma} reste neutre pour une
onde transversale ($\vect k\cdot\vect E = 0$ donne $\rho = 0$ par Gauss), de sorte que
Maxwell--Ampère s'écrit $-\iu\vect k\wedge\underline{\vect B} = \mu_0\underline\gamma\underline{\vect E}
+ \iu\omega\mu_0\varepsilon_0\underline{\vect E}$ et Maxwell--Faraday $\underline{\vect B} = \vect k
\wedge\underline{\vect E}/\omega$~; en combinant, $k^2\underline{\vect E} = \mu_0(\iu\omega\underline\gamma -
\omega^2\varepsilon_0)(-1)\underline{\vect E}$, c'est-à-dire $k^2 = \omega^2\mu_0\varepsilon_0 - \iu\omega\mu_0
\underline\gamma = (\omega^2 - ne^2/\varepsilon_0m)/c^2$. Les vitesses s'en déduisent comme au
\cref{ch:b2:dispersion-wave-packets}.
\end{proof}

\begin{remark}[Où passe l'énergie]\label{rem:b2:waves-in-media:plasmaenergy}
La conductivité est imaginaire~: $\langle\vect j\cdot\vect E\rangle = 0$, les
électrons prennent de l'énergie à l'onde pendant une demi-période et la lui rendent
pendant la suivante --- un milieu sans pertes, dans lequel l'énergie de l'onde
se partage entre les champs et l'énergie cinétique des électrons, et
se déplace à $v_g$. Les chocs ajoutent une petite partie réelle à $\underline\gamma$ et une
faible absorption~: la couche D de l'ionosphère absorbe les ondes moyennes le
jour, d'où les stations AM lointaines que l'on capte la nuit.
\end{remark}

\begin{example}[L'ionosphère et la radio]\label{ex:b2:waves-in-media:ionosphere}
Avec $n = \qty{1e12}{m^{-3}}$ dans la couche F, $f_p = \qty{9}{MHz}$~: les émissions
en ondes courtes situées en dessous sont réfléchies (et peuvent faire des bonds autour de la Terre
entre l'ionosphère et le sol), la FM (\qty{100}{MHz}), la télévision
et les liaisons satellites passent. Une onde à \qty{3}{MHz} ne pénètre que de
$1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{6}{m}$ dans la couche avant de
faire demi-tour. Un engin qui rentre dans l'atmosphère est enveloppé d'un
\omterm{def:b2:waves-in-media:plasma}{plasma} de $n \sim 10^{18}$ m$^{-3}$, $f_p \sim \qty{10}{GHz}$~: le contact radio est
perdu pendant plusieurs minutes.
\end{example}

\section{Ondes dans un conducteur ohmique~: l'effet de peau}

\begin{theorem}[Effet de peau]\label{thm:b2:waves-in-media:skin}
\index{effet de peau}\index{épaisseur de peau}
Dans un conducteur ohmique ($\vect j = \gamma\vect E$, $\gamma$ réel, aux fréquences
telles que $\omega\tau \ll 1$ et $\varepsilon_0\omega \ll \gamma$, c'est-à-dire en dessous de $\sim\qty{1e16}{Hz}$ pour le
cuivre) le \omterm{thm:b2:maxwell-equations:maxwell}{courant de déplacement} est négligeable et les champs obéissent à
l'équation de diffusion $\Delta\vect E = \mu_0\gamma\,\partial_t\vect E$. Une OPPH qui entre dans
le conducteur en $x = 0$ a
\[
\underline k^2 = -\iu\mu_0\gamma\omega , \qquad \underline k = \frac{1 - \iu}\delta , \qquad
\delta = \sqrt{\frac2{\mu_0\gamma\omega}} ,
\]
de sorte que $\underline E = E_0\,\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$~: elle est amortie
sur l'\emph{\omterm{def:b2:dispersion-wave-packets:complex}{épaisseur de peau}} $\delta$, avec une phase qui tourne aussi d'un
radian par $\delta$ --- une longueur d'onde $2\pi\delta$, une \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} $\omega\delta \ll
c$. Le courant $\vect j = \gamma\vect E$ est confiné dans une couche d'épaisseur
$\delta$ sous la surface, et $\vect B$ est en retard de \qty{45}{\degree} sur $\vect E$ avec
$|B| = |E|\sqrt2/\omega\delta$. Pour le cuivre, $\delta = \qty{9.2}{mm}$ à \qty{50}{Hz},
\qty{0.21}{mm} à \qty{100}{kHz}, \qty{2.1}{\micro m} à \qty{1}{GHz}.
\end{theorem}

\begin{proof}
$\operatorname{\vect{curl}}\vect B = \mu_0\gamma\vect E$ (déplacement abandonné) et
$\operatorname{\vect{curl}}\vect E = -\partial_t\vect B$~: en prenant le rotationnel de la seconde,
$-\Delta\vect E = -\mu_0\gamma\partial_t\vect E$ (le conducteur est neutre, $\operatorname{div}
\vect E = 0$). Pour $\eu^{\iu(\omega t - kx)}$~: $k^2 = -\iu\mu_0\gamma\omega$, et
$\sqrt{-\iu} = (1 - \iu)/\sqrt2$ donne $\underline k = (1 - \iu)/\delta$ (la racine qui
décroît vers l'intérieur du conducteur). Faraday~: $\underline B = \underline k\underline E/\omega = (1 - \iu)
\underline E/\omega\delta$, de module $\sqrt2E/\omega\delta$ et de phase $-\pi/4$.
\end{proof}

\begin{example}[Conséquences de l'effet de peau]\label{ex:b2:waves-in-media:skinuses}
\index{résistance en alternatif}\index{cage de Faraday}
(i) Un fil de rayon $a \gg \delta$ transporte son courant dans une couronne
d'épaisseur $\delta$~: sa résistance passe de $1/\gamma\pi a^2$ à environ $1/\gamma
2\pi a\delta$ par mètre --- pour un fil de cuivre de \qty{1}{mm}, quatre fois la valeur
en continu à \qty{1}{MHz}, cent fois à \qty{1}{GHz}. Les bobines haute fréquence
utilisent du fil «~de Litz~» (de nombreux brins isolés) ou des tubes argentés~; les
conducteurs du réseau électrique sont câblés avec une gaine d'aluminium. (ii) Une
boîte métallique atténue une onde de $\eu^{-d/\delta}$ par épaisseur $d$~: une
enceinte d'aluminium de \qty{1}{mm} bloque \qty{1}{MHz} de \qty{100}{dB} mais \qty{50}{Hz}
presque pas --- les champs magnétiques de basse fréquence exigent du fer. (iii)
Les sous-marins ne reçoivent que la radio à très grandes ondes ($\delta = \qty{2.3}{m}$ à
\qty{10}{kHz} dans l'eau de mer)~; les mines se détectent par les courants de Foucault qu'une
bobine y induit (\cref{ch:b2:maxwell-equations}, le temps de
diffusion $\mu_0\gamma d^2$).
\end{example}

\section{Ondes dans un diélectrique~: l'électron lié}

\begin{proposition}[Électron élastiquement lié~; indice complexe]\label{prop:b2:waves-in-media:lorentz}
\index{diélectrique}\index{polarisation (d'un milieu)}\index{susceptibilité}\index{indice complexe}\index{modèle de Lorentz}
Dans un isolant chaque électron est lié à son atome~: un déplacement
$\vect r$ hors de l'équilibre appelle la force de rappel $-m\omega_0^2\vect r$ et
un amortissement $-m\Gamma\,\dd\vect r/\dd t$ (l'énergie rayonnée ou cédée au
réseau). Excité par l'onde, $m\ddot{\vect r} = -m\omega_0^2\vect r - m\Gamma\dot{\vect r}
- e\vect E$~; les électrons déplacés donnent au milieu la \emph{polarisation}
(moment dipolaire par unité de volume) $\vect P = -ne\vect r = \varepsilon_0\underline\chi(\omega)
\underline{\vect E}$ avec la \emph{susceptibilité}
\[
\underline\chi(\omega) = \frac{\omega_p^2}{\omega_0^2 - \omega^2 + \iu\Gamma\omega} ,
\qquad \omega_p^2 = \frac{ne^2}{\varepsilon_0m} ,
\]
et le courant lié $\vect j = \partial_t\vect P$. Les \omterm{thm:b2:maxwell-equations:maxwell}{équations de Maxwell} donnent alors
\[
\underline k^2 = \frac{\omega^2}{c^2}\,(1 + \underline\chi) = \frac{\omega^2}{c^2}\,\underline n^2 ,
\qquad \underline n = n' - \iu n'' ,
\]
où $n'$ est l'indice de réfraction (\omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} $c/n'$) et $n''$
l'extinction (l'intensité décroît en $\eu^{-2n''\omega x/c}$). Très en dessous d'une
résonance ($\omega \ll \omega_0$, $\Gamma$ négligeable), $n'^2 \approx 1 + \omega_p^2/\omega_0^2 +
\omega_p^2\omega^2/\omega_0^4$~: l'indice \emph{croît} avec la fréquence --- c'est la
dispersion normale, loi de Cauchy $n \approx A + B/\lambda^2$ --- et l'absorption est
négligeable~; près de $\omega_0$ le milieu absorbe, et juste au-dessus $n'$
décroît avec $\omega$ (dispersion anomale).
\end{proposition}

\begin{proof}
Amplitudes complexes~: $(\omega_0^2 - \omega^2 + \iu\Gamma\omega)\underline{\vect r} = -e\underline{\vect E}
/m$~; $\underline{\vect P} = -ne\underline{\vect r}$. Puis, comme pour le \omterm{def:b2:waves-in-media:plasma}{plasma}, $k^2 = \omega^2\mu_0
\varepsilon_0 - \iu\omega\mu_0\underline\gamma$ avec $\underline{\vect j} = \iu\omega\underline{\vect P} = \iu\omega\varepsilon_0
\underline\chi\underline{\vect E}$, c'est-à-dire $\underline\gamma = \iu\omega\varepsilon_0\underline\chi$~: $k^2 = (\omega^2/c^2)
(1 + \underline\chi)$. (Le \omterm{def:b2:waves-in-media:plasma}{plasma} est le cas $\omega_0 = 0$, $\Gamma = 0$.)
Développement~: $1/(\omega_0^2 - \omega^2) \approx (1 + \omega^2/\omega_0^2)/\omega_0^2$.
\end{proof}

\begin{example}[Verre, eau, air]\label{ex:b2:waves-in-media:dielectrics}
Le verre a ses résonances électroniques dans l'ultraviolet ($\omega_0 \sim 10^{16}$
rad/s)~: dans le visible il est transparent et dispersif, $n = 1.51$ à
\qty{700}{nm}, $1.53$ à \qty{400}{nm}, assez pour qu'un prisme étale un
arc-en-ciel et pour que les impulsions d'une fibre s'étalent (\cref{ch:b2:dispersion-wave-packets})~;
dans l'ultraviolet il absorbe --- on ne bronze pas derrière une vitre.
L'eau possède en outre une résonance de rotation de ses molécules polaires vers
\qty{20}{GHz} (une relaxation de Debye plutôt qu'une raie fine)~: elle absorbe fortement
les micro-ondes, ce qui chauffe les aliments et aveugle les radars sous la pluie,
et elle a $n'' \approx 0$ dans une étroite fenêtre visible --- la fenêtre par
laquelle les yeux ont évolué. L'air, avec $n - 1 = 2.9 \times 10^{-4}$, courbe encore le
Soleil couchant d'un demi-degré.
\end{example}

\begin{remark}[Une équation, trois milieux]\label{rem:b2:waves-in-media:summary}
Tout milieu linéaire entre dans les \omterm{thm:b2:maxwell-equations:maxwell}{équations de Maxwell} par son
courant induit $\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$, et la \omterm{def:b2:dispersion-wave-packets:relation}{relation de
dispersion} est toujours $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma(\omega)$~:
\begin{center}\small
\begin{tabular}{lll}
milieu & $\underline\gamma(\omega)$ & $\underline k^2$\\ \hline
\omterm{def:b2:waves-in-media:plasma}{plasma} & $ne^2/\iu m\omega$ (imaginaire) & $(\omega^2 - \omega_p^2)/c^2$~: coupure\\
conducteur ohmique & $\gamma$ (réel) & $-\iu\mu_0\gamma\omega$~: \omterm{thm:b2:waves-in-media:skin}{effet de peau}\\
diélectrique & $\iu\omega\varepsilon_0\underline\chi(\omega)$ & $(\omega^2/c^2)(1 + \underline\chi)$~: indice
\end{tabular}
\end{center}
Un métal est tour à tour les trois~: ohmique en dessous de \qty{1e13}{Hz}, \omterm{def:b2:waves-in-media:plasma}{plasma}
(réfléchissant) dans le visible, transparent dans l'ultraviolet lointain.
\end{remark}

\begin{method}[Les ondes dans un milieu]\label{met:b2:waves-in-media:method}
(1) Écrire l'équation du mouvement des charges et en tirer $\underline{\vect j} =
\underline\gamma\underline{\vect E}$. (2) Reporter dans Maxwell--Ampère avec l'hypothèse
OPPH~: $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma$. (3) Prendre la racine
telle que $\operatorname{Im}k \le 0$ (décroissance dans le sens de propagation), séparer
en parties réelle (propagation, $v_\varphi = \omega/k'$) et imaginaire (\omterm{def:b2:dispersion-wave-packets:complex}{atténuation}).
(4) Repérer les régimes en fréquence --- propagatif,
évanescent, absorbant --- et la fréquence caractéristique ($\omega_p$,
$1/\mu_0\gamma\delta^2$, $\omega_0$). (5) Trouver $\vect B$ par Faraday et, s'il le faut,
le flux de Poynting et la puissance dissipée.
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:1}
\omterm{def:b2:waves-in-media:plasma}{Fréquences plasma} pour $n = \qty{1e11}{m^{-3}}$ (ionosphère de nuit),
\qty{1e12}{m^{-3}} (de jour), \qty{1e18}{m^{-3}} (gaine de rentrée), \qty{8.5e28}{m^{-3}}
(cuivre), \qty{1e5}{m^{-3}} (espace interstellaire). Pour chacune, laquelle des
ondes suivantes passe~: une station AM à \qty{1}{MHz}, une station FM à \qty{100}{MHz}, un
signal GPS à \qty{1.5}{GHz}, la lumière visible~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:2}
\omterm{def:b2:dispersion-wave-packets:complex}{Épaisseur de peau} du cuivre ($\gamma = \qty{6.0e7}{S/m}$) à \qty{50}{Hz}, \qty{1}{kHz},
\qty{1}{MHz}, \qty{1}{GHz}~; de l'aluminium (\qty{3.6e7}{S/m}) à \qty{1}{MHz}~; de l'eau
de mer (\qty{5}{S/m}) à \qty{10}{kHz} et \qty{1}{GHz}~; d'un sol humide (\qty{1e-2}{S/m})
à \qty{1}{MHz}. Quelles fréquences radio atteignent un sous-marin à \qty{10}{m}~?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:3}
Un verre suit $n = 1.50 + \qty{4.5e3}{nm^2}/\lambda^2$. Indice à \qty{400}{nm},
\qty{550}{nm}, \qty{700}{nm}~; \omterm{def:b2:dispersion-wave-packets:relation}{vitesses de phase}~; indice de groupe $n_g = n -
\lambda\,\dd n/\dd\lambda$ à \qty{550}{nm}~; étalement angulaire d'un faisceau blanc
réfracté sous incidence \qty{45}{\degree} dans le verre.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:4}
Une onde à \qty{5}{MHz} rencontre une couche de $n = \qty{2e11}{m^{-3}}$~: est-elle
réfléchie~? Profondeur de pénétration~; idem à \qty{2}{MHz}~; fréquence minimale
qui traverse une couche de $n = \qty{1.5e12}{m^{-3}}$. Pourquoi les bandes en ondes courtes
«~s'ouvrent~» et «~se ferment~» avec l'heure et le cycle solaire~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:5}
\emph{\omterm{ex:b2:waves-in-media:skinuses}{Résistance en alternatif}.} Un fil de cuivre de rayon $a = \qty{1.0}{mm}$. (a) Résistance
par mètre en continu. (b) À \qty{1}{MHz}~: $\delta$~; résistance approchée
par mètre avec le courant confiné dans une couronne d'épaisseur $\delta$~;
rapport au continu. (c) À \qty{100}{MHz}. (d) Pourquoi une bobine de $N$ spires de
ce fil a-t-elle un facteur de qualité qui cesse de s'améliorer avec $N$ à haute
fréquence, et qu'est-ce que le fil de Litz~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:6}
\emph{Blindage.} Une boîte d'aluminium ($\gamma = \qty{3.6e7}{S/m}$) de \qty{1.0}{mm}
d'épaisseur. \omterm{def:b2:dispersion-wave-packets:complex}{Atténuation} de l'amplitude du champ à la traversée de la paroi, en dB, à
\qty{50}{Hz}, \qty{10}{kHz}, \qty{1}{MHz} (le facteur $\eu^{-d/\delta}$, réflexions aux
surfaces négligées). Pourquoi blinde-t-on le champ magnétique du secteur
avec du fer (haute perméabilité) plutôt qu'avec du cuivre~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:7}
\emph{Énergie dans un \omterm{def:b2:waves-in-media:plasma}{plasma}.} Pour l'onde transversale du
\cref{thm:b2:waves-in-media:plasmadispersion}~: (a) amplitude de la vitesse des
électrons et densité d'énergie cinétique $\tfrac12nm\langle v^2\rangle$. (b)
Densités d'énergie électrique et magnétique moyennes. (c) Montrer que la densité
d'énergie totale vaut $\tfrac12\varepsilon_0E_0^2(1 + \dots)$ --- calculer la parenthèse ---
et que $\langle\Pi\rangle$ divisé par elle vaut $v_g$. (d) Quelle fraction de
l'énergie est dans les électrons pour $\omega = 2\omega_p$~?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:8}
\emph{Dispersion d'un pulsar.} Les impulsions radio d'un pulsar traversent une distance
$L$ de \omterm{def:b2:waves-in-media:plasma}{plasma} interstellaire ($n \ll$ tout le reste, $\omega \gg \omega_p$). (a) Montrer
que le retard de groupe vaut $t \approx (L/c)(1 + \omega_p^2/2\omega^2)$. (b) Le retard
entre les arrivées à $f_1$ et $f_2$ vaut $\Delta t = \dfrac{e^2}{8\pi^2
\varepsilon_0mc}\,nL\Bigl(\dfrac1{f_1^2} - \dfrac1{f_2^2}\Bigr)$~: calculer la constante.
(c) Un pulsar à $L = \qty{1}{kpc} = \qty{3.1e19}{m}$ à travers $n = \qty{3e4}{m^{-3}}$~:
retard entre \qty{400}{MHz} et \qty{1.4}{GHz}. (d) Les astronomes mesurent
$\Delta t$ et connaissent $n$~: qu'obtiennent-ils~? (La mesure de dispersion, la
jauge de distance standard des pulsars.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:9}
\emph{Une raie spectrale.} Dans un gaz dilué ($\underline n \approx 1 + \underline\chi/2$) le
\omterm{prop:b2:waves-in-media:lorentz}{modèle de Lorentz} donne $n'' = \tfrac12\operatorname{Im}\underline\chi$. (a) Montrer qu'au voisinage de
$\omega_0$, $n'' \approx \dfrac{\omega_p^2}{4\omega_0}\,\dfrac{\Gamma/2}{(\omega_0 - \omega)^2 + \Gamma^2/4}$, une
lorentzienne de largeur totale $\Gamma$. (b) Coefficient d'absorption $\alpha = 2n''\omega/c$
au centre. (c) Vapeur de sodium, $n = \qty{1e17}{m^{-3}}$, $\lambda_0 = \qty{589}{nm}$,
$\Gamma = \qty{6e7}{s^{-1}}$~: $\alpha$ au centre et épaisseur qui absorbe
$90\%$ (négliger l'élargissement Doppler, qui en réalité élargit la raie
d'un facteur cent). (d) Tracer $n' - 1$ en travers de la raie~; où la
dispersion est-elle anomale~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:10}
\emph{Échauffement Joule dans la peau.} Une onde $E_0\eu^{\iu\omega t}$ (selon $y$)
entre dans un conducteur qui occupe $x > 0$. (a) Écrire $\underline E(x, t)$ et en déduire
$\underline B(x, t)$ (selon $z$) par Faraday. (b) \omterm{thm:b2:poynting-vector:poynting}{Vecteur de Poynting} moyen à la
surface~: montrer que $\langle\Pi_x(0)\rangle = E_0^2/2\mu_0\omega\delta$. (c) Puissance Joule
moyenne par unité d'aire, $\int_0^\infty\tfrac12\gamma|\underline E|^2\dd x$~: montrer qu'elle vaut
(b). (d) Exprimer le résultat sous la forme $\tfrac12R_sH_0^2$ avec $H_0 = B_0/\mu_0$ le
champ magnétique de surface et $R_s = 1/\gamma\delta$ la \emph{résistance
de surface}~; valeur pour le cuivre à \qty{1}{GHz}, et perte d'une
cavité hyperfréquence dont les parois portent $H_0 = \qty{100}{A/m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:11}
\emph{Pourquoi les métaux brillent.} Traiter les électrons de conduction de l'argent ($n =
\qty{5.9e28}{m^{-3}}$) comme un \omterm{def:b2:waves-in-media:plasma}{plasma} sans chocs aux fréquences optiques.
(a) $\omega_p$ et la longueur d'onde correspondante. (b) À \qty{500}{nm}~: $k$ est
imaginaire --- profondeur de pénétration~; l'onde est totalement réfléchie (aucune
absorption dans ce modèle)~: l'argent est un miroir. (c) Et à \qty{200}{nm}~? D'où
la «~transparence ultraviolette~» des métaux alcalins. (d) L'argent réel
réfléchit $95\%$, l'or paraît jaune~: qu'oublie le modèle (penser aux
chocs, et aux électrons liés dont les résonances tombent dans le
visible pour l'or)~?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:12}
\emph{L'eau dans le four à micro-ondes.} L'orientation des molécules polaires de l'eau
donne une susceptibilité $\underline\chi = \chi_s/(1 + \iu\omega\tau)$ (relaxation
de Debye), $\chi_s \approx 80$, $\tau = \qty{8}{ps}$, s'ajoutant à une constante
$\chi_\infty \approx 4$. (a) Parties réelle et imaginaire de $\underline\varepsilon_r = 1 + \chi_\infty
+ \underline\chi$ à \qty{2.45}{GHz} et à \qty{20}{GHz}. (b) \omterm{prop:b2:waves-in-media:lorentz}{Indice complexe} et
longueur d'absorption $1/(2n''\omega/c)$ de l'intensité à \qty{2.45}{GHz}~: sur quelle
profondeur un four cuit-il~? (c) Puissance absorbée par unité de volume pour une
amplitude de champ $E_0$ dans l'eau~: $\tfrac12\omega\varepsilon_0\varepsilon''E_0^2$~; valeur pour
$E_0 = \qty{2}{kV/m}$. (d) Pourquoi utilise-t-on \qty{2.45}{GHz} plutôt que les
\qty{20}{GHz} d'absorption maximale~?
\end{exercise}

\section{Problème~: l'ionosphère, la porte du four et le prisme}

\begin{problem}[{Problème du week-end --- trois matériaux vus par une onde~: le
plasma qui courbe les signaux GPS, le métal qui retient les micro-ondes, et
le verre qui étale la lumière blanche}]
\label{pb:b2:waves-in-media:1}

\medskip
\noindent\textbf{Partie I --- L'ionosphère et le GPS.}
Densité électronique dans la couche F $n = \qty{1e12}{m^{-3}}$~; la colonne totale
d'électrons sur un trajet vertical vaut $N_T = \int n\,\dd z = \qty{2e17}{m^{-2}}$ de
jour. GPS~: $f_1 = \qty{1575}{MHz}$, $f_2 = \qty{1228}{MHz}$.
\begin{enumerate}
  \item \omterm{def:b2:waves-in-media:plasma}{Fréquence plasma} de la couche F~; le GPS passe-t-il~? Et une
        onde courte à \qty{5}{MHz}~?
  \item Montrer que pour $\omega \gg \omega_p$ la \omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe} vaut $v_g \approx c(1 -
        \omega_p^2/2\omega^2)$ et la \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} $c(1 + \omega_p^2/2\omega^2)$.
  \item Retard de groupe supplémentaire du signal GPS à la traversée de la couche~: montrer que
        $\Delta t = \dfrac{e^2}{8\pi^2\varepsilon_0mc}\,\dfrac{N_T}{f^2}$ et le calculer pour
        $f_1$~; erreur de distance $c\,\Delta t$.
  \item Le récepteur utilise les deux fréquences~: des deux retards il
        élimine l'erreur. Exprimer $N_T$ en fonction de $\Delta t_1 - \Delta t_2$~;
        quelle précision sur l'écart des retards donne $N_T$ à $1\%$ près~?
  \item Combien de périodes de porteuse supplémentaires le retard de la question 3
        représente-t-il à $f_1$~?
  \item La phase de la porteuse est \emph{avancée} de la même
        quantité que le groupe est retardé~: pourquoi (signes des deux
        corrections)~? Laquelle importe pour un récepteur qui compte les
        périodes de la porteuse~?
  \item La nuit $N_T$ est dix fois plus faible~: erreur pour un récepteur
        monofréquence de jour et de nuit.
  \item Une éruption solaire porte $n$ dans la couche D (\qty{80}{km}) à
        \qty{1e10}{m^{-3}} avec de nombreux chocs~: quels services radio sont
        touchés, et pourquoi le GPS ne se dégrade-t-il que légèrement~?
\end{enumerate}

\medskip
\noindent\textbf{Partie II --- La porte du four.}
Les parois de la cavité et la grille de la porte sont en acier ($\gamma = \qty{1.0e6}{S/m}$)~;
les trous de la grille font \qty{1.0}{mm} de large dans une tôle de \qty{0.5}{mm} d'épaisseur~;
$f = \qty{2.45}{GHz}$, $P = \qty{800}{W}$.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Épaisseur de peau} dans l'acier à \qty{2.45}{GHz}~; la paroi est-elle «~épaisse~»~?
  \item Résistance de surface $R_s = 1/\gamma\delta$~; avec un champ magnétique
        de surface d'amplitude $H_0 = \qty{30}{A/m}$ sur les parois (\qty{1}{m^2}),
        puissance perdue dans les parois, $\tfrac12R_sH_0^2$ par unité d'aire~; fraction
        de $P$.
  \item Dans un trou de largeur $a \ll \lambda$ le champ ne peut pas se propager (le
        trou est un guide d'ondes sous la coupure, \cref{ch:b2:guided-waves})~:
        il décroît en $\eu^{-\pi z/a}$ sur l'épaisseur $z$. \omterm{def:b2:dispersion-wave-packets:complex}{Atténuation}
        de l'amplitude à travers la tôle de \qty{0.5}{mm}, en dB.
  \item Pourquoi voit-on à travers la grille (longueur d'onde de la lumière
        comparée à la taille des trous) alors que les micro-ondes ne peuvent pas sortir~?
  \item La vitre est un diélectrique de $\underline\varepsilon_r = 4 - 0.02\iu$~:
        indice, et longueur d'\omterm{def:b2:dispersion-wave-packets:complex}{atténuation} de l'intensité dans le verre~; le
        verre chauffe-t-il~?
  \item Une fourchette métallique dans le four~: le champ à ses pointes est renforcé~;
        avec l'\omterm{def:b2:dispersion-wave-packets:complex}{épaisseur de peau} de la question 8, estimer la \omterm{def:b2:charges-currents-conduction:densities}{densité de
        courant} à la surface pour $H_0 = \qty{300}{A/m}$ (utiliser $j \approx H_0/\delta$
        en surface) et l'échauffement local $j^2/\gamma$ --- pourquoi des étincelles~?
\end{enumerate}

\medskip
\noindent\textbf{Partie III --- Le fil à haute fréquence.}
Un fil de cuivre de rayon $a = \qty{0.5}{mm}$ ($\gamma = \qty{6e7}{S/m}$) dans une
bobine de chauffage par induction à \qty{13.56}{MHz} transporte \qty{20}{A} efficaces.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Épaisseur de peau}~; section conductrice effective~; \omterm{ex:b2:waves-in-media:skinuses}{résistance en alternatif} par
        mètre comparée au continu.
  \item Puissance dissipée par mètre de fil de bobine~; température que
        le fil atteindrait avec un refroidissement par convection $h = \qty{20}{W/(m^2.K)}$.
  \item La bobine induit des courants dans une pièce d'acier ($\gamma =
        \qty{1e6}{S/m}$, $\mu_r \approx 1$ à chaud)~: \omterm{def:b2:dispersion-wave-packets:complex}{épaisseur de peau} en ce point~; pourquoi le
        chauffage par induction ne chauffe-t-il qu'une couche superficielle, et comment cela
        sert-il à durcir les engrenages~?
  \item Pour réduire les pertes de la bobine on remplace le fil par un tube
        de cuivre de \qty{3}{mm} avec de l'eau à l'intérieur~: nouvelle résistance par mètre~;
        l'intérieur du tube transporte-t-il du courant~?
  \item À quelle fréquence l'\omterm{def:b2:dispersion-wave-packets:complex}{épaisseur de peau} dans le cuivre égalerait-elle le
        rayon du fil~? En dessous, quelle approximation remplace la
        formule du «~fil épais~»~?
\end{enumerate}

\medskip
\noindent\textbf{Partie IV --- Le prisme.}
Un verre suit $n(\lambda) = 1.500 + \qty{5.0e3}{nm^2}/\lambda^2$, sa résonance étant
en $\lambda_0 = \qty{100}{nm}$.
\begin{enumerate}[resume]
  \item Vérifier que la forme de Cauchy découle du \omterm{prop:b2:waves-in-media:lorentz}{modèle de Lorentz} très
        en dessous de la résonance, et que $\omega_p^2/\omega_0^2 = n^2(\infty) - 1$~: valeur
        de $\omega_p^2/\omega_0^2$ et du $n$ effectif de ce verre.
  \item Indice et \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} à \qty{400}{nm} et \qty{700}{nm}~; indice de
        groupe à \qty{550}{nm}~; retard entre les deux couleurs sur
        \qty{1}{km} d'un tel verre dans une fibre.
  \item Un prisme de \qty{60}{\degree} au minimum de déviation pour \qty{550}{nm}~:
        déviation (utiliser $n\sin(A/2) = \sin((A + D)/2)$)~; écart angulaire
        entre \qty{400}{nm} et \qty{700}{nm} (par différentiation)~; sur un écran à
        \qty{2}{m}, largeur du spectre.
  \item Pourquoi le verre est-il opaque à \qty{200}{nm}, et quelle est sa
        couleur si une trace de fer ajoute une faible résonance à \qty{700}{nm}~?
  \item Dans l'image de Lorentz, pourquoi le bleu est-il plus dévié que le rouge par
        le prisme~?
  \item Donner en un tableau, pour la couche F, l'acier, le cuivre et
        le verre à leurs fréquences de travail, la nature de
        $\underline\gamma$ et le sort de l'onde.
\end{enumerate}
\end{problem}
