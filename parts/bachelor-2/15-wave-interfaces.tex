\chapter{Reflection and Transmission at Interfaces}\label{ch:b2:wave-interfaces}

The sonographer squeezes a gel between the probe and the skin: without
it the sound would bounce off the film of air and never enter the
body. A camera lens is coated with a film a quarter of a wavelength
thick, and its surfaces stop reflecting. A radio wave meets the
ionosphere and turns back; light meets a silver mirror and turns back;
a pulse reaches the end of a cable and comes back inverted or upright
according to what is connected there. Every wave, meeting a change of
medium, splits into a reflected and a transmitted part, and one idea
decides the shares: the \emph{continuity conditions} at the interface,
written with the impedances of the two media. This chapter applies it
to sound, to light between two dielectrics, to a \omterm{def:b2:waves-in-media:plasma}{plasma}, to a metal,
and to a cable, and draws from it the laws of reflection and
refraction and the one angle at which glass reflects nothing.

\section{Normal incidence: the impedance rule}

\begin{theorem}[Reflection and transmission at normal incidence]\label{thm:b2:wave-interfaces:normal}
\index{reflection coefficient}\index{transmission coefficient}\index{impedance matching}
A plane wave travels in medium 1 toward the plane interface $x = 0$ with
medium 2; each medium is characterized by an impedance $Z_i$ (the ratio
of the two field quantities continuous at the interface --- pressure and
velocity for sound, $E$ and $H = B/\mu_0$ for light, voltage and current for
a line). Continuity of both quantities at $x = 0$ gives, for the
"force-like" quantity (pressure, $E$, $v$),
\[
r = \frac{Z_2 - Z_1}{Z_2 + Z_1} , \qquad t = \frac{2Z_2}{Z_2 + Z_1} ,
\qquad R = r^2 , \quad T = \frac{4Z_1Z_2}{(Z_1 + Z_2)^2} , \quad R + T = 1 .
\]
(For the "flow-like" quantity --- velocity, $H$, current --- $r$ changes
sign.) Nothing is reflected when $Z_1 = Z_2$ (\emph{matched} media);
almost everything when $Z_2 \gg Z_1$ ($r \to +1$) or $Z_2 \ll Z_1$ ($r \to -1$).
\end{theorem}

\begin{proof}
Incident $p_i = A\cos(\omega t - k_1x)$, reflected $p_r = rA\cos(\omega t + k_1x)$,
transmitted $p_t = tA\cos(\omega t - k_2x)$, same frequency. The associated
flow quantities are $p/Z_1$, $-p_r/Z_1$ (a wave going backward carries
$v = -p/Z$) and $p_t/Z_2$. Continuity at $x = 0$: $1 + r = t$ and $(1 - r)/Z_1
= t/Z_2$; solve. Energies: the intensities are $p^2/Z$, so $R = r^2$ and
$T = t^2Z_1/Z_2$.
\end{proof}

\begin{example}[Sound at three interfaces]\label{ex:b2:wave-interfaces:sound}
Air ($Z = \qty{410}{kg.m^{-2}.s^{-1}}$) to water ($1.5 \times 10^6$): $r = 0.9995$,
$T = 1.1 \times 10^{-3}$ --- $99.9\%$ reflected; a swimmer hears almost nothing
of the world above, and the ultrasound probe needs its gel. Tissue
($1.6 \times 10^6$) to bone ($7 \times 10^6$): $r = 0.63$, $T = 0.6$. Fat ($1.38 \times
10^6$) to muscle ($1.70 \times 10^6$): $r = 0.10$, $R = 1\%$ --- the faint echoes an
image is made of. A string tied to a wall ($Z_2 \to \infty$): $r = +1$ for the
force, $-1$ for the velocity, the pulse returns inverted
(\cref{ch:b2:waves-on-strings}).
\end{example}

\begin{proposition}[Light at normal incidence; anti-reflection coating]\label{prop:b2:wave-interfaces:optics}
\index{reflectance (normal incidence)}\index{anti-reflection coating}\index{quarter-wave layer}
For light between two transparent media of indices $n_1$, $n_2$ (the
impedance of a dielectric is $Z = \sqrt{\mu_0/\varepsilon_0}/n$, inversely
proportional to $n$), for the electric field
\[
r = \frac{n_1 - n_2}{n_1 + n_2} , \qquad R = \Bigl(\frac{n_1 - n_2}{n_1 + n_2}\Bigr)^2 :
\]
$4\%$ at each air--glass surface, $2\%$ for water, $17\%$ for diamond. A
layer of index $n = \sqrt{n_1n_2}$ and thickness $\lambda/4n$ between the media
cancels the reflection at $\lambda$ (\emph{quarter-wave \omterm{prop:b2:wave-interfaces:optics}{anti-reflection
coating}}): the waves reflected at its two faces have equal amplitudes
and opposite phases. The same layer with a high index, or a stack of
alternating \omterm{prop:b2:wave-interfaces:optics}{quarter-wave layers}, does the opposite and builds a
mirror that reflects $99.99\%$ --- the mirrors of lasers.
\end{proposition}

\begin{proof}
Continuity of the tangential $\vect E$ and $\vect B$ (no \omterm{prop:b2:maxwell-equations:boundary}{surface
current}): $E_i + E_r = E_t$ and $B_i - B_r = B_t$ with $B = nE/c$ in each
medium, i.e.\ the impedance rule with $Z \propto 1/n$. Coating: the two
reflected amplitudes are $(n_1 - n)/(n_1 + n)$ and $(n - n_2)/(n + n_2)$,
equal when $n^2 = n_1n_2$; a round trip in the layer adds the phase
$2\times2\pi n(\lambda/4n)/\lambda = \pi$, so they cancel (multiple reflections,
small, included in the exact treatment of \cref{exo:b2:wave-interfaces:5}).
\end{proof}

\section{Oblique incidence: the laws of Descartes and the Brewster angle}

\begin{theorem}[Laws of reflection and refraction]\label{thm:b2:wave-interfaces:snell}
\index{Snell--Descartes laws}\index{total internal reflection}\index{evanescent wave (total reflection)}
A plane wave of wavevector $\vect k_i$ meets the plane interface between
two media in which waves travel at $v_1$ and $v_2$. The continuity
conditions must hold at every point of the interface and every time:
the reflected and transmitted waves must have the same frequency and
the same \emph{tangential wavevector} as the incident one. Hence the
three wavevectors are coplanar (the plane of incidence), the angle of
reflection equals the angle of incidence, and
\[
\frac{\sin\theta_1}{v_1} = \frac{\sin\theta_2}{v_2} \qquad (n_1\sin\theta_1 = n_2\sin\theta_2 \text{ for light}) .
\]
When $v_2 > v_1$ and $\sin\theta_1 > v_1/v_2$ no real $\theta_2$ exists:
\emph{\omterm{thm:b2:wave-interfaces:snell}{total internal reflection}}; the field in medium 2 is then an
\emph{\omterm{def:b2:dispersion-wave-packets:complex}{evanescent wave}}, $\propto\eu^{-x/\ell}\eu^{\iu(\omega t - k_ty)}$ travelling
along the interface and decaying away from it over $\ell = \lambda_2/2\pi
\sqrt{\sin^2\theta_1(v_2/v_1)^2 - 1}$, a fraction of a wavelength; it
carries no energy away --- total reflection.
\end{theorem}

\begin{proof}
With $\vect k = (k_x, k_y, 0)$ and the interface $x = 0$, the phases
$\omega t - k_yy$ of the three waves must agree for all $y$, $t$: same
$\omega$, same $k_y = (\omega/v_1)\sin\theta_1 = (\omega/v_1)\sin\theta_r = (\omega/v_2)\sin
\theta_2$. If $(\omega/v_2)^2 - k_y^2 < 0$, $k_{2x}$ is imaginary, $k_{2x} = -\iu/\ell$:
evanescence; its \omterm{thm:b2:poynting-vector:poynting}{Poynting vector} along $x$ averages to zero.
\end{proof}

\begin{proposition}[Brewster's angle; polarization by reflection]\label{prop:b2:wave-interfaces:brewster}
\index{Brewster angle}\index{polarization by reflection}\index{Fresnel coefficients}
For light falling on a dielectric, the reflected amplitude depends on
the polarization (Fresnel's coefficients, admitted in general): for the
polarization \emph{in} the plane of incidence (p) it vanishes at the
\emph{\omterm{prop:b2:wave-interfaces:brewster}{Brewster angle}}
\[
\tan\theta_B = \frac{n_2}{n_1} ,
\]
at which the reflected and refracted rays are perpendicular; light
reflected at $\theta_B$ is then completely polarized perpendicular to the
plane of incidence (s), and partially so around it --- the glare on
water and roads that polarizing sunglasses remove. For glass ($n = 1.5$)
$\theta_B = \qty{56}{\degree}$, for water \qty{53}{\degree}.
\end{proposition}

\begin{proof}
The reflected wave is radiated by the dipoles of medium 2, set
oscillating by the refracted wave along its $\vect E$, which for p
polarization lies in the plane of incidence, perpendicular to the
refracted ray. A dipole does not radiate along its own axis
(\cref{ch:b2:dipole-radiation}); when the reflected direction is
parallel to that axis --- i.e.\ perpendicular to the refracted ray, $\theta_1
+ \theta_2 = \pi/2$ --- nothing is reflected. Then $\sin\theta_2 = \cos\theta_1$ and
Descartes gives $\tan\theta_1 = n_2/n_1$. The s polarization has its $\vect E$
perpendicular to the plane of incidence, never along the reflected
ray: it is always partly reflected.
\end{proof}

\section{Plasmas, metals and the perfect conductor}

\begin{proposition}[Reflection on a plasma and on a metal]\label{prop:b2:wave-interfaces:conductors}
\index{perfect conductor}\index{surface current (reflection)}\index{Hagen--Rubens relation}
(i) A wave of $\omega < \omega_p$ at normal incidence on a \omterm{def:b2:waves-in-media:plasma}{plasma} is totally
reflected: the "index" $\underline n = -\iu\sqrt{\omega_p^2/\omega^2 - 1}$ is imaginary,
$r = (1 - \underline n)/(1 + \underline n)$ has modulus $1$, and the field penetrates
as the evanescent $\eu^{-\kappa x}$ of \cref{ch:b2:waves-in-media}. (ii) A
good conductor has $\underline n = (1 - \iu)c/\omega\delta$, of huge modulus: $r \approx
-1$, the reflectance is $R \approx 1 - 2\omega\delta/c = 1 - \sqrt{8\varepsilon_0\omega/\gamma}$
(Hagen--Rubens): $0.99999$ for copper at \qty{10}{GHz}, $0.99$ in the far
infrared. (iii) The \emph{\omterm{prop:b2:wave-interfaces:conductors}{perfect conductor}} ($\gamma \to \infty$, $\delta \to 0$):
$r = -1$ exactly for $\vect E$, the total tangential $\vect E$ vanishes at
the surface, the incident and reflected waves form the \omterm{prop:b2:waves-on-strings:modes}{standing wave}
\[
\vect E = -2E_0\sin kx\sin\omega t\,\vect e_y , \qquad
\vect B = 2\frac{E_0}c\cos kx\cos\omega t\,\vect e_z ,
\]
with nodes of $\vect E$ (antinodes of $\vect B$) at $x = 0, -\lambda/2, \dots$; the
field is carried by a \emph{\omterm{prop:b2:maxwell-equations:boundary}{surface current}} $\vect j_s = \vect n\wedge\vect B/
\mu_0 = (2E_0/\mu_0c)\cos\omega t\,\vect e_y$, and the Laplace force on it is
the \omterm{prop:b2:poynting-vector:wave}{radiation pressure} $2I/c$.
\end{proposition}

\begin{proof}
(i) and (ii): the impedance rule with $Z \propto 1/\underline n$ and the complex
indices of \cref{ch:b2:waves-in-media}; for (ii), $r = (1 - \underline n)/(1 +
\underline n) \approx -1 + 2/\underline n$ and $R = |r|^2 \approx 1 - 4\operatorname{Re}(1/
\underline n) = 1 - 2\omega\delta/c$. (iii) With $\vect E = \vect 0$ inside, the
boundary relation $\vect E_{2t} = \vect E_{1t}$ gives $E_{\text{tot}}(0) = 0$, so $r =
-1$; sum the incident $E_0\cos(\omega t - kx)$ and reflected $-E_0\cos(\omega t + kx)$;
$\vect B$ from Faraday; the jump of $\vect B$ at the surface is $\mu_0\vect j_s
\wedge\vect n$; the mean force per unit area $\tfrac12\langle j_sB\rangle$ on
the sheet is $\varepsilon_0E_0^2 = 2I/c$ (\cref{ch:b2:poynting-vector}).
\end{proof}

\begin{example}[Mirrors, radar, microwave ovens]\label{ex:b2:wave-interfaces:conductoruses}
A metal mirror reflects by this mechanism: the incoming field drives
a current in a skin a few nanometres deep (\qty{23}{nm} for silver at
\qty{500}{nm}), which re-radiates the reflected wave; the $5\%$ missing
heats the metal --- which is why a high-power laser mirror is cooled,
and why \cref{ch:b2:waves-in-media}'s oven wall takes $6\%$ of the
power. A radar sees an aircraft by the wave its skin re-radiates; a
stealth aircraft replaces flat metal by absorbers and by shapes that
reflect away from the radar. The nodes of the \omterm{prop:b2:waves-on-strings:modes}{standing wave}, $\lambda/2 =
\qty{6}{cm}$ apart in an oven, are the cold spots.
\end{example}

\section{The end of a cable}

\begin{proposition}[Termination of a transmission line]\label{prop:b2:wave-interfaces:coax}
\index{matched load}\index{standing wave ratio}\index{time-domain reflectometry}
A line of \omterm{prop:b2:dispersion-wave-packets:coax}{characteristic impedance} $Z_c$ ends at $x = \ell$ in a load of
impedance $\underline Z_L$: the voltage wave is reflected with $\underline r =
(\underline Z_L - Z_c)/(\underline Z_L + Z_c)$ --- nothing for a \emph{matched} load
$Z_L = Z_c$, $r = -1$ for a short circuit, $r = +1$ for an open end. A
mismatched line carries a partial \omterm{prop:b2:waves-on-strings:modes}{standing wave} whose ratio of maximum
to minimum voltage is the \emph{\omterm{prop:b2:wave-interfaces:coax}{standing wave ratio}} $(1 + |r|)/(1 - |r|)$.
Sending a short pulse and timing the echo locates a fault (\omterm{prop:b2:wave-interfaces:coax}{time-domain
reflectometry}) and reads its nature from the echo's sign.
\end{proposition}

\begin{proof}
At the load, $v = Z_Li$ with $v = v_i + v_r$ and $i = (v_i - v_r)/Z_c$
(\cref{ch:b2:dispersion-wave-packets}); the impedance rule once more.
Standing-wave ratio: $|v|$ varies between $|v_i|(1 + |r|)$ and $|v_i|(1 -
|r|)$ along the line.
\end{proof}

\begin{method}[Any interface]\label{met:b2:wave-interfaces:method}
(1) Identify the two quantities continuous across the interface and
the impedance of each medium as their ratio in a progressive wave.
(2) Write incident $+$ reflected on one side, transmitted on the other,
with equal frequencies and equal tangential wavevectors (Descartes).
(3) Impose continuity: two equations, two unknowns $r$, $t$. (4) Energy:
$R = |r|^2$, $T = 1 - R$ for lossless media. (5) Check the limits: equal
impedances (no reflection), infinite or zero impedance ($|r| = 1$, sign
of $r$), and for layers, the phases of the successive reflections.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b2:wave-interfaces:1}
\omterm{prop:b2:sound-waves:plane}{Acoustic impedances}: air \qty{410}{kg.m^{-2}.s^{-1}}, water \qty{1.5e6}{kg.m^{-2}.s^{-1}},

of energy reflected for air--water, water--steel, tissue--bone,
tissue--air. Why does a fish not hear you, and why does a diver hear

\end{exercise}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:wave-interfaces:5}
\emph{\omterm{prop:b2:wave-interfaces:optics}{Anti-reflection coating}.} A glass of $n_2 = 1.52$ is coated with

thickness for \qty{550}{nm}. (b) Residual reflectance with MgF$_2$ at

\section{Problem: Coatings, gel and the radar absorber}

\begin{problem}[{Weekend problem --- making interfaces disappear: the
coated lens, the matched transducer, the radar-absorbing screen, and the
mirror that cannot be made to vanish}]
\label{pb:b2:wave-interfaces:1}

\medskip
\noindent\textbf{Part I --- The coated lens.}
A zoom lens has $16$ air--glass surfaces of index $1.52$; light at
\qty{550}{nm}.
\begin{enumerate}
  \item Reflectance of one uncoated surface; transmission of the whole
        lens; fraction of the light wandering as stray reflections.
  \item Ideal single-layer coating: index and thickness; why is no
        common solid of that index available, and what is the
        residual reflectance with MgF$_2$ ($1.38$)?
  \item With $16$ surfaces at $1.3\%$ each: transmission; at $0.3\%$
        (multilayer): transmission. Why do old lenses give "flat" images
        with low contrast?
  \item The coating is designed for \qty{550}{nm}: show that at
        \qty{440}{nm} the phase $\varphi$ of the round trip is $1.25\pi$ and
        estimate the reflectance there (two-reflection formula, $R
        \approx r_1^2 + r_2^2 + 2r_1r_2\cos\varphi$, with $r_1 = -0.16$, $r_2 =
        -0.05$). Colour of the reflection?
  \item A dielectric mirror stacks \omterm{prop:b2:wave-interfaces:optics}{quarter-wave layers} of $1.38$ and
        $2.35$: the reflected amplitudes from successive interfaces
        now add in phase; reflection of one pair (two interfaces) and
        why $20$ pairs reach $99.9\%$.
  \item In the coated lens the reflected wave is "cancelled": where
        has its energy gone?
  \item Such a mirror reflects one band of wavelengths and transmits
        the rest: why, and what colour does it look in transmission
        if it reflects green?
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The probe and the gel.}
Ceramic $Z_1 = \qty{30e6}{kg.m^{-2}.s^{-1}}$, matching layer, gel, skin $Z =
\qty{1.6e6}{kg.m^{-2}.s^{-1}}$, $f = \qty{5}{MHz}$.
\begin{enumerate}[resume]
  \item Direct transmission ceramic--tissue in energy; with the
        \omterm{prop:b2:wave-interfaces:optics}{quarter-wave layer} of ideal impedance (value) at $f$.
  \item The layer is a quarter wave at \qty{5}{MHz}: transmission
        at \qty{3}{MHz} and \qty{7}{MHz} (use the two-reflection estimate
        with the actual round-trip phase): what does this do to the
        probe's bandwidth, and why is a short pulse (needed for depth
        resolution) in conflict with it?
  \item A \qty{5}{\micro m} air film between probe and skin: amplitude
        transmission through the two interfaces (multiply the $t$'s,
        ignore the resonance); energy transmitted; in dB.
  \item The same with gel ($Z = \qty{1.5e6}{kg.m^{-2}.s^{-1}}$).
  \item The ceramic's own reverberation: the pulse bounces inside
        the ceramic with $r \approx 0.9$ per reflection at the tissue face
        (and $\approx 1$ at the back); number of round trips for the
        amplitude to fall to $1\%$, and why the ceramic is backed with
        an absorber.
  \item Echo from a \qty{1}{mm} bone surface at \qty{5}{cm} depth: energy
        fraction returning to the probe (reflection at bone, two
        passes of \qty{0.5}{dB/cm/MHz} \omterm{def:b2:dispersion-wave-packets:complex}{attenuation}); why bone casts an
        acoustic shadow.
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The radar absorber.}
A radar at \qty{10}{GHz} illuminates a metal plate. To hide it, a thin
resistive sheet of surface resistance $R_s$ (\unit{\ohm} per square) is
placed at a distance $d$ in front of the plate, with air between.
The impedance of free space is $Z_0 = \sqrt{\mu_0/\varepsilon_0} = \qty{377}{\ohm}$.
\begin{enumerate}[resume]
  \item Reflectance of the bare metal plate (\omterm{prop:b2:wave-interfaces:conductors}{perfect conductor}), and
        the phase of $r$.
  \item Treat the air gap as a \omterm{prop:b2:dispersion-wave-packets:coax}{transmission line} of impedance $Z_0$
        shorted by the plate: show that the impedance seen at the
        sheet, looking toward the plate, is $\underline Z_{\text{in}} = \iu Z_0
        \tan(2\pi d/\lambda)$ (write the \omterm{prop:b2:waves-on-strings:modes}{standing wave} $v = v_i(\eu^{-\iu kx} -
        \eu^{\iu kx})$ from the short and take $v/i$ at $x = -d$); for $d =
        \lambda/4$ it is infinite (an open circuit).
  \item The sheet of resistance $R_s$ in parallel with that open
        circuit presents $R_s$ to the incoming wave: \omterm{thm:b2:wave-interfaces:normal}{reflection
        coefficient} $(R_s - Z_0)/(R_s + Z_0)$; conclude that a sheet of
        \qty{377}{\ohm} per square at $\lambda/4$ reflects nothing (the
        Salisbury screen). Value of $d$ at \qty{10}{GHz}.
  \item Where does the energy go? Check with the power dissipated
        in the sheet, $|v|^2/2R_s$ per unit area, against the incident
        intensity.
  \item At \qty{8}{GHz} and \qty{12}{GHz} the same screen: $\underline Z_{\text{in}}$,
        the total impedance $R_s \parallel \underline Z_{\text{in}}$, and the
        reflectance; bandwidth of the trick.
  \item The radar now illuminates the screen at \qty{30}{\degree} from
        the normal: what changes in the phase of the round trip
        through the gap, and what does that do to the matching?
  \item Why do real stealth coatings use several lossy layers and
        shaped surfaces rather than one Salisbury screen?
\end{enumerate}

\medskip
\noindent\textbf{Part IV --- The mirror that stays.}
\begin{enumerate}[resume]
  \item Silver at \qty{500}{nm} behaves as a \omterm{def:b2:waves-in-media:plasma}{plasma} ($\omega_p = \qty{1.4e16}{rad/s}$):
        $\underline n$, $r$ (modulus and phase), penetration depth. Why can no
        quarter-wave coating make a metal stop reflecting?
  \item A \qty{1}{kW} green laser on a silver mirror ($R = 0.97$): absorbed
        power and the temperature rise rate of a \qty{10}{g} mirror
        ($c = \qty{235}{J/(kg.K)}$) if nothing cools it.
  \item \omterm{prop:b2:waves-on-strings:modes}{Standing wave} in front of the silver: spacing of the nodes of
        $\vect E$ and position of the first one (use the phase of $r$);
        what does a dust grain sitting at a node receive?
  \item The beam of question 22 is \qty{1}{cm^2}: intensity, field
        amplitude, and the surface \omterm{def:b2:charges-currents-conduction:densities}{current density} the mirror must
        carry (perfect-conductor estimate $j_s = 2E_0/\mu_0c$).
  \item Compare the three "mirrors" of this problem --- the metal, the
        \omterm{def:b2:waves-in-media:plasma}{plasma}, the dielectric stack --- by the mechanism that sends
        the wave back, the phase of $r$, and what limits their
        reflectance.
\end{enumerate}
\end{problem}
