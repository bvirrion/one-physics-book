\chapter{Volmaakte fluïda: Euler en Bernoulli}\label{ch:b2:euler-bernoulli}

Houd een vel papier aan de rand vast en blaas er overheen: het komt omhoog.
Zet de douche aan en het gordijn buigt naar binnen. Een vleugel houdt
driehonderd ton in de lucht omdat de lucht sneller over haar bovenkant
stroomt dan eronder. Alle drie zeggen hetzelfde: \emph{waar een fluïdum
sneller beweegt, is de druk lager}. Dit hoofdstuk schrijft de tweede wet van
Newton op voor een \omterm{def:b2:fluid-kinematics:particle}{fluïdumdeeltje} in het eenvoudigste model --- het
\emph{volmaakte} fluïdum, dat druk ondervindt maar geen wrijving --- en
leidt er de \omterm{thm:b2:euler-bernoulli:bernoulli}{stelling van Bernoulli} uit af, de energiebalans van een
\omterm{def:b2:fluid-kinematics:euler}{stroomlijn}, met toepassingen van de debietmeter in een buis tot de
snelheidsmeter van een vliegtuig.

\section{De vergelijking van Euler}

\begin{proposition}[Drukkracht op een fluïdumdeeltje]\label{prop:b2:euler-bernoulli:pressure}
\index{drukkracht (volumedichtheid)}
De resultante van de drukkrachten die het omringende fluïdum uitoefent op
een deeltje met volume $\dd\tau$ is
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
druk werkt als een volumekracht met dichtheid
$-\operatorname{\vect{grad}}P$, die van hoge naar lage druk duwt.
\end{proposition}

\begin{proof}
Op het blokje van \cref{thm:b2:fluid-kinematics:continuity} ontvangen de
zijvlakken in $x$ en $x + \dd x$ de krachten $+P(x)\dd y\dd z$ en
$-P(x + \dd x)\dd y\dd z$ langs $x$: netto $-\partial_xP\,\dd x\dd y\dd z$; evenzo
voor $y$ en $z$.
\end{proof}

\begin{definition}[Volmaakt fluïdum]\label{def:b2:euler-bernoulli:perfect}
\index{volmaakt fluïdum}
Een \emph{volmaakt fluïdum} is er een waarin de druk de enige contactkracht
tussen naburige deeltjes is --- loodrecht op elk oppervlak, zonder
tangentiële (viskeuze) spanning. Het is een idealisatie die, zoals het
volgende hoofdstuk uitlegt, geldig is ver van wanden en bij een hoog
reynoldsgetal: water in een buis weg van de wand, lucht om een vleugel
buiten de dunne grenslaag, een golf op zee.
\end{definition}

\begin{theorem}[Vergelijking van Euler]\label{thm:b2:euler-bernoulli:euler}
\index{vergelijking van Euler}
In een galileïsch stelsel voldoet een \omterm{def:b2:euler-bernoulli:perfect}{volmaakt fluïdum} met dichtheid $\rho$
onder de zwaartekracht in elk punt aan
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
waaraan rechts elke andere volumekrachtdichtheid $\vect f_v$ (elektrisch, of
traagheidskracht in een niet-galileïsch stelsel) wordt toegevoegd. In rust
herleidt zij zich tot de hydrostatische wet
$\operatorname{\vect{grad}}P = \rho\vect g$ van het volume van jaar 1.
\end{theorem}

\begin{proof}
Tweede wet van Newton voor het deeltje met massa $\rho\,\dd\tau$: zijn
versnelling is de \omterm{thm:b2:fluid-kinematics:material}{materiële afgeleide}
(\cref{thm:b2:fluid-kinematics:material}), de krachten zijn de resultante
van de druk en het gewicht $\rho\vect g\,\dd\tau$; deel door $\dd\tau$.
\end{proof}

\begin{example}[Een tank die versnelt]\label{ex:b2:euler-bernoulli:tank}
Water in een tank op een vrachtwagen die met $\vect a$ versnelt, is in rust
in het stelsel van de vrachtwagen; daar geeft Euler met de traagheidskracht
$-\rho\vect a$ dat $\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$:
de druk neemt toe langs de ``effectieve zwaartekracht'' $\vect g - \vect a$,
het vrije oppervlak (een isobaar) staat er loodrecht op en helt naar
achteren over de hoek $\arctan(a/g)$ --- \qty{11}{\degree} voor
$a = \qty{2}{m/s^2}$. In een emmer die met $\Omega$ draait, geeft de
middelpuntvliedende kracht $\rho\Omega^2r\,\vect e_r$ dat
$P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$, en is het vrije oppervlak de
paraboloïde $z = \Omega^2r^2/2g$.
\end{example}

\section{De stelling van Bernoulli}

\begin{theorem}[Bernoulli]\label{thm:b2:euler-bernoulli:bernoulli}
\index{stelling van Bernoulli}\index{stuwdruk}\index{dynamische druk}
Voor een \emph{volmaakt}, \emph{onsamendrukbaar} fluïdum met uniforme
dichtheid in \emph{stationaire} stroming is, in een galileïsch stelsel met
uniforme zwaartekracht, de grootheid
\[
P + \tfrac12\rho v^2 + \rho gz
\]
constant \emph{langs elke \omterm{def:b2:fluid-kinematics:euler}{stroomlijn}}. Is de stroming bovendien
\emph{rotatievrij}, dan is het dezelfde constante in het hele fluïdum. De
term $\tfrac12\rho v^2$ is de \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{dynamische druk}}; $P + \tfrac12\rho v^2$
is de \emph{stuwdruk}, de druk die het fluïdum zou bereiken als het langs
zijn \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} tot stilstand werd gebracht.
\end{theorem}

\begin{proof}
Stationaire Euler met
$(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$
en $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$, met $\rho$ uniform:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
Het rechterlid staat loodrecht op $\vect v$, zodat de gradiënt van de
grootheid van Bernoulli geen component langs de \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} heeft: de
grootheid is er constant langs. Is $\vect\omega = \vect 0$, dan verdwijnt de
gradiënt overal.
\end{proof}

\begin{remark}[Wat Bernoulli zegt en niet zegt]\label{rem:b2:euler-bernoulli:meaning}
Gedeeld door $\rho$ luidt de stelling: de mechanische energie per
massa-eenheid, $v^2/2 + gz$, plus $P/\rho$ blijft behouden langs de baan van
een deeltje. De drukterm is de arbeid die de druk van het fluïdum erachter
\emph{op} het deeltje verricht, min de arbeid die het op het fluïdum ervoor
verricht: de energiestelling voor een deeltje dat zonder wrijving door een
drukveld wordt geduwd. Vier voorwaarden: \omterm{def:b2:euler-bernoulli:perfect}{volmaakt fluïdum} (geen viskeus
verlies), stationair, onsamendrukbaar, en \emph{langs een \omterm{def:b2:fluid-kinematics:euler}{stroomlijn}}. Twee
punten op verschillende \omterm{def:b2:fluid-kinematics:euler}{stroomlijnen} mag men alleen in een \omterm{def:b2:fluid-kinematics:vorticity}{rotatievrije
stroming} vergelijken. De stelling zegt niet ``snelle lucht zuigt'': zij zegt
dat een deeltje dat (door een drukval) is \emph{versneld}, nu een lagere
druk heeft --- de oorzaak is het drukveld, de snelheid is het gevolg.
\end{remark}

\begin{remark}[Samendrukbare fluïda]\label{rem:b2:euler-bernoulli:compressible}
Voor een gas verandert de dichtheid, maar is de stroming stationair,
volmaakt en isentroop, dan geeft hetzelfde bewijs het behoud van
$h + v^2/2 + gz$ langs een \omterm{def:b2:fluid-kinematics:euler}{stroomlijn}, waarin $h$ de enthalpie per
massa-eenheid is ($h = c_PT$ voor een ideaal gas): de vorm die in
\cref{ch:b2:open-systems} voor mondstukken en turbines wordt gebruikt. Bij
snelheden ver onder de geluidssnelheid ($v \lesssim \qty{100}{m/s}$ in
lucht) verandert de dichtheid met minder dan $5\%$ en is de onsamendrukbare
stelling nauwkeurig.
\end{remark}

\section{Toepassingen}

\begin{proposition}[De venturimeter]\label{prop:b2:euler-bernoulli:venturi}
\index{venturi-effect}\index{venturimeter}
In een horizontale buis die van doorsnede $S_1$ naar $S_2 < S_1$ vernauwt,
stijgt de snelheid ($v_2 = v_1S_1/S_2$, behoud van massa) en daalt de druk:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
Het drukverschil met een manometer meten geeft het debiet: een
\emph{\omterm{prop:b2:euler-bernoulli:venturi}{venturimeter}}. Hetzelfde effect drijft een carburateur aan, een
verfspuit, de luchtinlaat van een bunsenbrander en de ``zuiging'' van een
voorbijrijdende trein.
\end{proposition}

\begin{proof}
Bernoulli langs een centrale \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} op constante hoogte, met
$v_2 =
v_1S_1/S_2$; los op naar $v_1$.
\end{proof}

\begin{proposition}[De pitotbuis]\label{prop:b2:euler-bernoulli:pitot}
\index{pitotbuis}
Een buis die naar de stroming gericht is en aan het verre uiteinde gesloten,
brengt het fluïdum aan haar monding tot stilstand (een \emph{stuwpunt}); een
tweede opening opzij, evenwijdig aan de stroming, leest de statische druk
$P$ af. Het verschil is de \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamische druk}:
\[
P_{\text{stuw}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{stuw}} - P)/\rho} .
\]
Elk vliegtuig meet zo zijn luchtsnelheid.
\end{proposition}

\begin{proof}
Bernoulli langs de \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} die in het stuwpunt eindigt, waar $v = 0$; de
zijopeningen verstoren de stroming niet, die er met de onverstoorde $v$ en
$P$ passeert.
\end{proof}

\begin{proposition}[Formule van Torricelli]\label{prop:b2:euler-bernoulli:torricelli}
\index{formule van Torricelli}\index{uitstroming}
Een groot open vat loopt leeg door een klein gat op diepte $h$ onder het
vrije oppervlak; de straal vertrekt met
\[
v = \sqrt{2gh} ,
\]
de snelheid van de vrije val vanaf het oppervlak. Het debiet is
$D_V \approx
\alpha s\sqrt{2gh}$ met $s$ de oppervlakte van het gat en
$\alpha \approx 0.6$ de \emph{contractiecoëfficiënt} van een scherprandige
opening (de \omterm{def:b2:fluid-kinematics:euler}{stroomlijnen} blijven voorbij het gat naar elkaar toe lopen).
\end{proposition}

\begin{proof}
Bernoulli van het vrije oppervlak (in rust, want het vat is groot:
$v_{\text{opp}} = v\,s/S \ll v$; druk $P_0$) naar de straal (druk $P_0$ vlak
buiten het gat): $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$. De stroming is
quasi-stationair: $h$ verandert langzaam.
\end{proof}

\begin{example}[Getallen]\label{ex:b2:euler-bernoulli:numbers}
Een venturi van \qty{5}{cm} die tot \qty{2.5}{cm} vernauwt op een
waterleiding, met $\Delta h = \qty{20}{cm}$ water (\qty{2.0}{kPa}):
$v_1 = \sqrt{2 \times 2000/
(1000 \times 15)} = \qty{0.52}{m/s}$,
$D_V = \qty{1.0}{L/s}$. Een verkeersvliegtuig op \qty{11}{km}
($\rho = \qty{0.36}{kg/m^3}$) bij \qty{250}{m/s}: \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamische druk}
\qty{11}{kPa}, $11\%$ van de omgevingsdruk --- de \omterm{prop:b2:euler-bernoulli:pitot}{pitotbuis} werkt nog
altijd, met een correctie voor samendrukbaarheid. Een watertoren \qty{40}{m}
boven de kranen: \qty{4}{bar} bij gesloten kraan, een straal van
\qty{28}{m/s} als zij wijd opengaat (in de praktijk veel minder: de
leidingen zijn niet volmaakt).
\end{example}

\begin{proposition}[Draagkracht op een vleugel]\label{prop:b2:euler-bernoulli:lift}
\index{draagkracht}\index{stelling van Kutta--Joukowski}\index{magnuseffect}
In de stroming om een vleugel gaat de lucht sneller over de bovenkant dan
onder de onderkant; de druk is daarom boven lager dan onder, en de
resultante is de \emph{\omterm{prop:b2:euler-bernoulli:lift}{draagkracht}} $F_L =
\tfrac12\rho v^2SC_L$ ($S$ het
vleugeloppervlak, $C_L \approx 0.3$--$1.5$ de draagkrachtcoëfficiënt,
afhankelijk van het profiel en de invalshoek). Het snelheidsverschil komt
neer op een \emph{circulatie} $\Gamma$ van de snelheid om het profiel, en de
\omterm{prop:b2:euler-bernoulli:lift}{draagkracht} per eenheid spanwijdte is $F_L' =
\rho v\Gamma$
(Kutta--Joukowski, aangenomen). Een tollende bal sleept lucht met zich mee
en wordt op dezelfde manier zijwaarts gedragen: het \emph{\omterm{prop:b2:euler-bernoulli:lift}{magnuseffect}} van
een gesneden tennisbal of een gekrulde vrije trap.
\end{proposition}

\begin{proof}
Bernoulli op de \omterm{def:b2:fluid-kinematics:euler}{stroomlijnen} vlak boven en vlak onder, die van dezelfde
stroomopwaartse toestand vertrekken:
$P_{\text{onder}} - P_{\text{boven}} = \tfrac12\rho
(v_{\text{boven}}^2 - v_{\text{onder}}^2)$;
integreer over de koorde. Waarom de bovenstroming sneller is --- de scherpe
achterrand dwingt het achterste stuwpunt daarheen en legt de circulatie vast
(de voorwaarde van Kutta) --- ligt buiten het bereik van het model van het
volmaakte fluïdum alleen; daarvoor is de grenslaag van het volgende
hoofdstuk nodig.
\end{proof}

\begin{proposition}[Een niet-stationaire stroming: de U-buis]\label{prop:b2:euler-bernoulli:utube}
\index{oscillatie in een U-buis}\index{niet-stationaire Bernoulli}
Een vloeistofkolom met totale lengte $L$ vult een U-buis met uniforme
doorsnede; over $z$ uit haar evenwicht gebracht, oscilleert zij volgens
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
als een slinger met lengte $L/2$. Algemener integreert Euler voor een
niet-stationaire maar \omterm{def:b2:fluid-kinematics:vorticity}{rotatievrije stroming} met potentiaal $\varphi$ tot
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$, met
dezelfde constante in het hele fluïdum (\omterm{prop:b2:euler-bernoulli:utube}{niet-stationaire Bernoulli}).
\end{proposition}

\begin{proof}
De snelheid $v = \dot z$ is uniform langs de buis (onsamendrukbaar, uniforme
doorsnede). Projecteer Euler op de as van de buis en integreer langs de
kolom van het ene vrije oppervlak naar het andere: de druktermen vallen weg
($P_0$ aan beide uiteinden), $\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$,
de convectieve term $\rho\,\Delta(v^2/2)$ verdwijnt (dezelfde snelheid aan
beide uiteinden), en de zwaartekrachtterm geeft $\rho g\cdot2z$. Vandaar
$\rho L\ddot z = -2\rho gz$. De algemene uitspraak volgt uit
$\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ voor
$\vect v = \operatorname{\vect{grad}}\varphi$.
\end{proof}

\begin{example}[Een leiding op gang brengen]\label{ex:b2:euler-bernoulli:start}
Een horizontale buis met lengte $\ell$ voert van een reservoir met constante
vervalhoogte $h$ naar een open uiteinde; de afsluiter gaat open op $t = 0$.
Met $v$ uniform langs de buis geeft Euler, geïntegreerd van het
reservoiroppervlak tot de uitmonding,
$\ell\,\dd v/\dd t = gh - \tfrac12v^2$, met als oplossing
$v = V\tanh(Vt/2\ell)$ met $V = \sqrt{2gh}$: de stroming bereikt de snelheid
van Torricelli met de tijdconstante $2\ell/V$ --- \qty{13}{s} voor een
drukleiding van \qty{400}{m} onder \qty{200}{m} vervalhoogte.
\end{example}

\begin{method}[Bernoulli gebruiken]\label{met:b2:euler-bernoulli:method}
(1) Ga de vier voorwaarden na; beslis of de stroming rotatievrij is
(uniforme aanstroming om een hindernis: ja; stroming in een buis met een
snelheidsprofiel: nee --- blijf op één \omterm{def:b2:fluid-kinematics:euler}{stroomlijn}). (2) Kies twee punten op
een \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} waar zo veel mogelijk grootheden bekend zijn: een vrij
oppervlak ($P = P_0$, $v \approx 0$ als de doorsnede groot is), een straal
in de open lucht ($P = P_0$), een stuwpunt ($v = 0$). (3) Voeg het behoud
van massa $vS = $ const toe om de snelheden te koppelen. (4) Integreer voor
niet-stationaire stromingen in plaats daarvan Euler langs de lijn.
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:1}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:2}
De \omterm{prop:b2:euler-bernoulli:pitot}{pitotbuis} van een vliegtuig op \qty{5}{km} ($\rho = \qty{0.74}{kg/m^3}$)
leest een \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamische druk} van \qty{15}{kPa} af. Ware luchtsnelheid. De
snelheidsmeter is geijkt op de dichtheid op zeeniveau (\qty{1.22}{kg/m^3}):
welke ``aangewezen snelheid'' toont hij, en waarom vinden piloten dat
verschil nuttig in plaats van hinderlijk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:3}
In een vat staat het water \qty{3.0}{m} diep; op \qty{1.0}{m} boven de vloer
wordt een gat van \qty{1.0}{cm^2} geprikt. Uitstroomsnelheid; debiet
(contractie $0.6$); waar treft de straal de vloer? Op welke diepte moet een
tweede gat worden geprikt om hetzelfde punt te bereiken?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:4}
Een wind van \qty{30}{m/s} waait over een plat dak van \qty{100}{m^2}; de
lucht onder het dak staat stil op atmosferische druk
($\rho = \qty{1.2}{kg/m^3}$). Drukverschil en resulterende kracht op het
dak; vergelijk met het gewicht van het dak (\qty{50}{kg/m^2}). Welke kant op
wordt het geduwd?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:5}
Een gesloten rechthoekige tank van \qty{2.0}{m} lang, half met water gevuld,
staat op een vrachtwagen. (a) Bij een versnelling van \qty{3.0}{m/s^2}: hoek
van het vrije oppervlak en het hoogteverschil van het water tussen de
achter- en de voorwand. (b) Drukverschil tussen de twee hoeken op de bodem.
(c) Bij remmen met \qty{6}{m/s^2} --- bereikt het water het deksel
\qty{0.8}{m} boven de bodem? (d) Dezelfde vrachtwagen die een bocht met
straal \qty{40}{m} neemt bij \qty{15}{m/s}: helling van het oppervlak.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
Een emmer met straal \qty{15}{cm} met water tot \qty{20}{cm} diep wordt om
zijn as gedraaid met \qty{2.0}{omw/s}. (a) Vorm en vergelijking van het
vrije oppervlak. (b) Hoogteverschil tussen rand en middelpunt. (c) Hoogte
van het water in het middelpunt en aan de rand, met behoud van volume. (d)
Bij welk toerental valt het midden van de bodem droog? (e) Waarom vormt
dezelfde paraboloïde een volmaakte telescoopspiegel (een draaiende plas
kwik)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}

niet als de buis vol lucht zit?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}
Een licht vliegtuig van \qty{1000}{kg} vliegt op zeeniveau met \qty{60}{m/s}

vlakken van de vleugel; draagkrachtcoëfficiënt. (b) Is de snelheid onder de
vleugel \qty{58}{m/s}, hoeveel is zij dan erboven (neem dezelfde
stroomopwaartse toestand)? (c) Circulatie om het profiel uit de formule van
Kutta--Joukowski, voor een spanwijdte van \qty{10}{m}. (d) Overtreksnelheid
als $C_L$ niet groter dan $1.4$ kan worden.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{Stenose.} Bloed ($\rho = \qty{1060}{kg/m^3}$) stroomt met
\qty{0.50}{m/s} in een slagader met doorsnede \qty{0.50}{cm^2} bij
\qty{13}{kPa} boven de atmosferische druk. Een plaque vernauwt haar tot
\qty{0.10}{cm^2}. (a) Snelheid en druk in de vernauwing. (b) Het weefsel
buiten de slagader staat op ongeveer \qty{1}{kPa}: kan de slagader
dichtklappen? (c) Vernauwt zij verder, dan stijgt de snelheid nog meer:
verklaar het op hol slaan en waarom zo'n vernauwing kan gaan fladderen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{Stuwtemperatuur.} Voor een ideaal gas in een stationaire isentrope
stroming blijft $c_PT + v^2/2$ langs een \omterm{def:b2:fluid-kinematics:euler}{stroomlijn} behouden. (a) Toon aan
dat de lucht die aan de neus van een vliegtuig tot stilstand komt,
$T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$ bereikt, met $M = v/c$ het machgetal
en $c =
\sqrt{\gamma RT/M_{\text{mol}}}$ de geluidssnelheid. (b)
Neustemperatuur van een verkeersvliegtuig bij $M = 0.85$ in lucht van
\qty{220}{K}; van een supersonisch vliegtuig bij $M = 2$; van een
terugkeercapsule bij $M = 25$ (heeft de formule dan nog zin?). (c) Toon aan
dat de onsamendrukbare formule van Pitot de snelheid bij $M = 0.85$
overschat, en ruwweg met hoeveel (ontwikkel
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ tot tweede orde in $M^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{Aanloopverschijnsel.} Een buis met lengte $\ell = \qty{50}{m}$ en
uniforme doorsnede voert van een groot reservoir (vervalhoogte
$h = \qty{5.0}{m}$ boven de uitmonding) naar een afsluiter aan het open
uiteinde. De afsluiter gaat open op $t = 0$. (a) Integreer Euler langs de
buis tot $\ell\,\dd v/\dd t = gh - v^2/2$. (b) Los die op:
$v = V\tanh(Vt/2\ell)$ met $V = \sqrt{2gh}$. (c) Tijd om $90\%$ van de
eindsnelheid te bereiken. (d) De afsluiter wordt nu vanuit volle stroming
abrupt in \qty{0.5}{s} gesloten: schat de gemiddelde vertraging en de
overdruk aan de afsluiter uit $\rho\ell\,\dd v/\dd t$ --- en waarom een
echte sluiting veel meer oplevert (volgende hoofdstukken).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{Het vrije oppervlak van een wervel.} Water draait als de \omterm{ex:b2:fluid-kinematics:planeflows}{puntwervel}

twee willekeurige punten worden toegepast? (b) Vorm van het vrije oppervlak
$z(r)$. (c) Gebruik binnen een kern van Rankine ($r < a$,
$v_\theta = \Omega r$, met rotatie) in plaats daarvan Euler in het
meedraaiende stelsel, of rechtstreeks: toon aan dat
$z = z(a) +
\Omega^2(r^2 - a^2)/2g$. (d) Totale diepte van de trechter voor
een badwervel met $\Gamma = \qty{0.03}{m^2/s}$ en $a = \qty{5}{mm}$; voor
een tornado ($\Gamma = \qty{3e4}{m^2/s}$, $a = \qty{60}{m}$, in lucht: druk
het resultaat dan liever uit als een drukval in het middelpunt).
\end{exercise}

\section{Vraagstuk: De stuwdam, de drukleiding en de turbine}

\begin{problem}[{Weekendvraagstuk --- een waterkrachtcentrale van het
stuwmeer tot de straal: drukken, snelheden, vermogens, verliezen en de
waterslag}]
\label{pb:b2:euler-bernoulli:1}

In een stuwmeer staat het water \qty{150}{m} diep achter een dam; het vrije
oppervlak ligt $H = \qty{200}{m}$ boven de mondstukken van een turbine. Een
drukleiding (een stalen buis) met diameter $D = \qty{3.0}{m}$ en lengte
$\ell = \qty{400}{m}$ voert het ontwerpdebiet $D_V = \qty{60}{m^3/s}$ naar
het turbinehuis, waar een mondstuk het omzet in een vrije straal op
atmosferische druk $P_0 =
\qty{1.0}{bar}$, die op de schoepen van een
peltonwiel slaat. $\rho =
\qty{1000}{kg/m^3}$, $g = \qty{9.81}{m/s^2}$; het
fluïdum is volmaakt tenzij anders vermeld.

\medskip
\noindent\textbf{Deel I --- In rust en bij het ontwerpdebiet.}
\begin{enumerate}
  \item Druk aan de voet van de dam, en de kracht op een verticale strook
        van de dam van \qty{1}{m} breed, van boven tot onder.
  \item Druk in de drukleiding bij het turbinehuis, met de turbineafsluiters dicht.
  \item Snelheid in de drukleiding bij het ontwerpdebiet; \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamische druk}
        daar; druk aan het uiteinde van de drukleiding bij het turbinehuis
        (vóór het mondstuk), met Bernoulli vanaf het oppervlak van het
        stuwmeer.
  \item Snelheid van de vrije straal; doorsnede en diameter van het mondstuk.
  \item Kinetisch vermogen dat de straal meevoert; toon aan dat het gelijk
        is aan $\rho gD_VH$ en bereken het.
  \item Massa en kinetische energie van het water dat bij het ontwerpdebiet
        in de drukleiding zit; doorlooptijd van een deeltje door de buis.
  \item Toon aan dat de vervalhoogte $H$ bij het ontwerpdebiet verdeeld is
        over de druk onder aan de drukleiding en de kinetische energie van
        het water, en dat het mondstuk de eerste in de tweede omzet: wat is
        de druk vlak na het mondstuk?
  \item Een \omterm{def:b2:euler-bernoulli:perfect}{volmaakt fluïdum} heeft geen drukverlies, een echt fluïdum wel:
        het verlies in een buis is
        $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ met
        $\lambda
        \approx 0.015$ voor glad staal. Drukverlies in meters
        waterkolom, en de fractie van het vermogen die verloren gaat.
  \item Een ingenieur stelt een drukleiding van \qty{2}{m} voor om staal te
        sparen. Doe vraag 7 opnieuw; geef commentaar.
\end{enumerate}

\medskip
\noindent\textbf{Deel II --- Het debiet meten.}
In de drukleiding is een venturi ingebouwd, met een keel van \qty{2.0}{m}.
\begin{enumerate}[resume]
  \item Snelheid in de keel en drukval tussen de buis en de keel bij het
        ontwerpdebiet.
  \item De val wordt op een kwikmanometer afgelezen
        ($\rho_{\text{Hg}} =
        \qty{13600}{kg/m^3}$): hoogteverschil.
  \item Toon aan dat het debiet evenredig is met de wortel uit de
        manometeraflezing, en geef de aflezing bij de helft van het
        ontwerpdebiet.
  \item Waarom moeten de drukaftakkingen gelijk liggen met de wand en niet
        in de stroming uitsteken?
  \item Een deel van de buis loopt om topografische redenen over een top
        \qty{6}{m} \emph{boven} het oppervlak van het stuwmeer. Druk daar
        bij het ontwerpdebiet; wat gebeurt er, en wat moet de ontwerper
        doen?
\end{enumerate}

\medskip
\noindent\textbf{Deel III --- Niet-stationaire regimes.}
Het stuwmeer heeft een oppervlakte $A = \qty{2.0}{km^2}$.
\begin{enumerate}[resume]
  \item Snelheid waarmee het peil van het stuwmeer bij het ontwerpdebiet
        daalt, in centimeters per uur.
  \item De turbine wordt stilgelegd en het mondstuk (doorsnede uit vraag 4)
        blijft open naar de lucht: het stuwmeer loopt door de zwaartekracht
        leeg. Schrijf de massabalans op en bepaal het peil $h(t)$
        (quasi-stationaire stroming volgens Torricelli).
  \item Tijd waarin het peil \qty{10}{m} daalt; vergelijk met de tijd bij
        constant ontwerpdebiet.
  \item De centrale opstarten: de drukleiding is vol, het mondstuk gaat open
        op $t = 0$. Neem de snelheid uniform langs de drukleiding,
        verwaarloos de vorm van het mondstuk, integreer Euler langs de buis
        tot $\ell\,\dd v/\dd t = gH - v^2/2$, los die op, en geef de
        tijdconstante.
  \item Wat is tijdens dat opstarten, op het ogenblik dat $v$ de helft van
        haar eindwaarde is, de druk onder aan de drukleiding? Verklaar het
        teken van het verschil met de stationaire waarde.
\end{enumerate}

\medskip
\noindent\textbf{Deel IV --- De afsluiter dichtdoen.}
\begin{enumerate}[resume]
  \item De afsluiter bij de turbine wordt in $\tau = \qty{10}{s}$ gesloten;
        neem aan dat het water in de drukleiding eenparig vertraagt. Schat
        met Euler langs de buis de overdruk aan de afsluiter, in bar.
  \item Bij een snelle sluiting is het water niet onsamendrukbaar: een
        drukgolf loopt met $c = \qty{1400}{m/s}$ de buis op (de
        geluidssnelheid in water in een stalen buis ---
        \cref{ch:b2:sound-waves}), en de overdruk is
        $\Delta P = \rho c\,\Delta v$ (Joukowsky). Waarde bij een
        plotselinge stop vanuit het ontwerpdebiet; vergelijk met de
        statische druk. Hoe lang doet de golf erover om het stuwmeer te
        bereiken en terug te komen, en wat zegt dat over de betekenis van
        ``snel''?
  \item Om de drukleiding te beschermen wordt een \emph{waterslottoren} ---
        een verticale open schacht met doorsnede $A_s = \qty{100}{m^2}$ ---
        aangesloten waar een horizontale tunnel met lengte
        $L = \qty{2000}{m}$ en doorsnede $A =
        \qty{7.0}{m^2}$ vanuit het
        stuwmeer op de drukleiding uitkomt. Na een plotselinge sluiting
        blijft het water in de tunnel bewegen en stijgt het peil $z$ in de
        schacht. Schrijf het behoud van massa tussen tunnel en schacht op.
  \item Integreer Euler langs de tunnel (\omterm{def:b2:euler-bernoulli:perfect}{volmaakt fluïdum}) om aan te tonen
        dat $z$ voldoet aan $\ddot z + \dfrac{gA}{LA_s}z = 0$; periode van
        de oscillatie.
  \item Maximale peilstijging in de schacht na een sluiting vanuit het
        ontwerpdebiet (energieredenering: de kinetische energie van het
        tunnelwater wordt potentiële energie in de schacht).
  \item Vat samen: de vijf drukken die men in deze centrale tegenkomt
        (statisch bij de dam, statisch bij de turbine, dynamisch in de buis,
        waterslag, waterslot) en de wet die elk ervan geeft.
\end{enumerate}
\end{problem}
