\chapter{De boltzmannfactor}\label{ch:b2:boltzmann-factor}

Beklim een berg en de lucht wordt ijler: op drieduizend meter is de
druk tot \qty{0.7}{bar} gezakt, op de top van de Everest tot een derde
van haar waarde op zeeniveau. Niets houdt de dampkring van bovenaf
tegen; hij wordt door de zwaartekracht vastgehouden en door de
warmtebeweging uitgespreid, en het compromis tussen die twee is een
exponentiële, $n \propto
\eu^{-mgz/k_BT}$ --- de verhouding van de
potentiële energie van een molecuul tot $k_BT$ beslist hoe zeldzaam het
daarboven is. Dezelfde exponentiële beheerst het deel van de atomen in
een aangeslagen niveau, het aantal moleculen dat snel genoeg is om een
planeet te ontvluchten, de magnetisatie van een paramagneet, de
bevolkingen die een laser moet inverteren, en de quanta die de wet van
Planck telt: het is de \emph{boltzmannfactor}, de nuttigste formule van
de statistische fysica. Dit hoofdstuk motiveert hem met de dampkring,
formuleert hem in het algemeen, en leidt eruit af, met niets dan sommen
en gaussische integralen, het tweeniveaustelsel en zijn
warmtecapaciteit, de gelijke verdeling van de energie, de verdeling van
de molecuulsnelheden en de \omterm{prop:b2:boltzmann-factor:curie}{wet van Curie} voor het paramagnetisme ---
onderweg ook de gemiddelde energie van een kwantumoscillator, die de
formule van Planck is.

\section{De isotherme dampkring}

\begin{proposition}[Barometrische wet]\label{prop:b2:boltzmann-factor:barometric}
In een dampkring van volmaakt gas (molecuulmassa $m$) op gelijkmatige
temperatuur $T$ in de gelijkmatige zwaartekracht $g$ dalen de druk en de
\omterm{def:b2:particle-diffusion:flux}{aantalsdichtheid} exponentieel met de hoogte:
\[
P(z) = P_0\,\eu^{-z/H}, \qquad n(z) = n_0\,\eu^{-mgz/k_BT}, \qquad
H = \frac{k_BT}{mg} ,
\]
met de \emph{schaalhoogte}\index{schaalhoogte} $H \approx \qty{8.4}{km}$
voor lucht bij \qty{288}{K}. De exponent is de verhouding van de potentiële
energie $mgz$ van een molecuul tot de warmte-energie $k_BT$.
\end{proposition}

\begin{proof}
Hydrostatica, $\dd P/\dd z = -\rho g = -nmg$, en de wet van het volmaakte
gas $P
= nk_BT$: $\dd P/\dd z = -(mg/k_BT)P$, een exponentiële. De
werkelijke
dampkring koelt af met de hoogte (ongeveer \qty{6.5}{K/km} in de onderste tien
kilometer), wat de daling wat sneller maakt; het isotherme
model klopt tot op $20\%$.
\end{proof}

\section{De boltzmannfactor}

\begin{theorem}[Boltzmannfactor]\label{thm:b2:boltzmann-factor:factor}
Een systeem in warmte-evenwicht met een thermostaat op de temperatuur
$T$ wordt in een gegeven microscopische toestand met energie $E$
aangetroffen met een
kans evenredig met
\[
\eu^{-E/k_BT} .
\]
Dus geldt voor een stel afzonderlijke toestanden met energieën $E_i$
\[
p_i = \frac{\eu^{-E_i/k_BT}}{Z}, \qquad Z = \sum_i\eu^{-E_i/k_BT} ,
\]
waarbij de som $Z$ (de \emph{toestandssom}\index{toestandssom})
de kansen normeert; twee toestanden (of twee niveaus met gelijke
ontaarding) zijn bevolkt in de verhouding $N_2/N_1 = \eu^{-(E_2 - E_1)/k_BT}$;
en voor een continue veranderlijke (een plaats, een snelheid) is de kansdichtheid
evenredig met $\eu^{-E/k_BT}$ in die veranderlijke. Bij
\qty{300}{K} is $k_BT = \qty{4.1e-21}{J} = \qty{0.026}{eV}$: alles wat
veel meer kost dan dat is zeldzaam, alles wat veel minder kost wordt
vrijelijk aangeslagen.
\end{theorem}

\begin{proof}
De barometrische wet is de factor voor de potentiële energie $mgz$; wij
nemen zijn algemeenheid aan, maar hier is het argument (in het volume van
jaar 3 streng gemaakt). De thermostaat $R$ is groot; wanneer ons kleine systeem in
een toestand met energie $E$ is, heeft de thermostaat de energie
$E_{\text{tot}} -
E$, en kan hij in $\Omega_R(E_{\text{tot}} - E)$
microscopische toestanden zijn, alle
even waarschijnlijk (het grondpostulaat voor een geïsoleerd geheel). Dus is
$p(E) \propto \Omega_R(E_{\text{tot}} - E)$. Met de definitie van Boltzmann
voor de entropie, $S_R = k_B\ln\Omega_R$, en $1/T = \dd S_R/\dd E_R$ voor de
thermostaat, is
$\ln\Omega_R(E_{\text{tot}} - E) \approx \ln\Omega_R(E_{\text{tot}})
- E/k_BT$
omdat $E \ll E_{\text{tot}}$; neem de exponentiële.
\end{proof}

\begin{example}[Bevolkingen van atoomniveaus]\label{ex:b2:boltzmann-factor:levels}
Het eerste aangeslagen niveau van waterstof ligt \qty{10.2}{eV} boven de
grondtoestand: bij \qty{300}{K} $\eu^{-395}$ --- geen enkel atoom in het heelal; bij
\qty{6000}{K}, het zonneoppervlak, $\eu^{-19.7} \approx 3 \times 10^{-9}$ (maal
een ontaarding $4$): genoeg, in de reusachtige gaskolom van de zon, voor de
balmerabsorptielijnen, die het sterkst zijn in sterren nabij \qty{10000}{K}.
De D-lijn van natrium (\qty{2.1}{eV}) in een vlam van \qty{2500}{K}:
$\eu^{-9.8} = 6
\times 10^{-5}$ van de atomen is aangeslagen, wat de gele
vlam van een
snuifje zout geeft. En het lasermidden van \cref{ch:b2:laser}: de factor
zegt dat zijn bovenste niveau in evenwicht leeg is, en de pomp moet daartegen
vechten.
\end{example}

\section{Afzonderlijke niveaus: het tweeniveaustelsel en de oscillator}

\begin{proposition}[Gemiddelde energie uit de toestandssom]\label{prop:b2:boltzmann-factor:meanE}
Met $\beta = 1/k_BT$ geldt
\[
\langle E\rangle = \sum_ip_iE_i = -\frac{\partial\ln Z}{\partial\beta}, \qquad
C = \frac{\dd\langle E\rangle}{\dd T} .
\]
\end{proposition}

\begin{proof}
$\partial_\beta Z = -\sum E_i\eu^{-\beta E_i}$, deel door $Z$.
\end{proof}

\begin{proposition}[Tweeniveaustelsel]\label{prop:b2:boltzmann-factor:twolevel}
$N$ onafhankelijke stelsels met twee niveaus $0$ en $\varepsilon$ (spins in een
veld, atomen met één aangeslagen niveau, defecten met twee standen):
\[
\frac{N_2}{N} = \frac{1}{\eu^{\varepsilon/k_BT} + 1}, \qquad
\langle E\rangle = \frac{N\varepsilon}{\eu^{\varepsilon/k_BT} + 1}, \qquad
C = Nk_B\,\frac{x^2\eu^x}{(\eu^x + 1)^2}, \quad x = \frac{\varepsilon}{k_BT} .
\]
Het bovenste niveau is leeg bij lage temperatuur en hoogstens halfgevuld bij
hoge temperatuur (in evenwicht nooit geïnverteerd); de warmtecapaciteit
verdwijnt aan beide uiteinden en piekt bij $0.44\,Nk_B$ nabij
$k_BT = 0.42
\varepsilon$ --- de \emph{schottkyanomalie}\index{schottky
anomalie}, het
kenmerk van een kloof in het spectrum.
\end{proposition}

\begin{proof}
$Z = 1 + \eu^{-x}$; $p_2 = \eu^{-x}/(1 + \eu^{-x})$; differentieer
$\langle
E\rangle$ naar $T$.
\end{proof}

\begin{proposition}[De kwantumoscillator en de formule van Planck]\label{prop:b2:boltzmann-factor:oscillator}
Een harmonische oscillator met frequentie $\nu$ heeft de niveaus $E_n = nh\nu$ (op
een constante na); bij temperatuur $T$ geldt
\[
Z = \frac{1}{1 - \eu^{-h\nu/k_BT}}, \qquad
\langle E\rangle = \frac{h\nu}{\eu^{h\nu/k_BT} - 1}
\ \longrightarrow\
\begin{cases} k_BT & (h\nu \ll k_BT), \\ h\nu\,\eu^{-h\nu/k_BT} & (h\nu \gg k_BT). \end{cases}
\]
Dit is de gemiddelde energie van een mode van het stralingsveld die de wet
van Planck (\cref{ch:b2:thermal-radiation}) met het aantal modes vermenigvuldigt,
en de gemiddelde energie van een trilling in een vaste stof: bij hoge temperatuur de
klassieke $k_BT$, bij lage temperatuur een bevroren mode.
\end{proposition}

\begin{proof}
Meetkundige reeks, dan $-\partial_\beta\ln Z$.
\end{proof}

\begin{example}[Warmtecapaciteit van vaste stoffen]\label{ex:b2:boltzmann-factor:einstein}
Behandel elk atoom van een vaste stof als drie oscillatoren met dezelfde frequentie
$\nu$ (Einstein, 1907): $C = 3Nk_B\,x^2\eu^x/(\eu^x - 1)^2$ met
$x =
h\nu/k_BT$. Boven $\theta_{\text{E}} = h\nu/k_B$ gaat
$C \to 3Nk_B = 3R$ per mol
--- de wet van Dulong en Petit (\qty{25}{J/mol/K}), gevolgd door lood,
koper en ijzer bij kamertemperatuur; ver daaronder stort $C$ in. Diamant
($\theta_{\text{E}} \approx \qty{1300}{K}$) heeft bij \qty{300}{K} maar
\qty{6}{J/mol/K},
wat de klassieke natuurkunde niet kon verklaren, en de warmtecapaciteit van
elke vaste stof verdwijnt bij lage temperatuur --- het eerste succes van het
kwantum buiten de straling.
\end{example}

\section{Gelijke verdeling en de verdeling van Maxwell}

\begin{theorem}[Gelijke verdeling van de energie]\label{thm:b2:boltzmann-factor:equipartition}
Elke veranderlijke $q$ die kwadratisch in de energie voorkomt,
$E = aq^2 +
\dots$, en die alle reële waarden doorloopt, draagt
$\tfrac12k_BT$ bij aan
de gemiddelde energie in evenwicht:
\[
\langle aq^2\rangle = \tfrac12k_BT .
\]
Een eenatomig gas (drie translaties) heeft $U = \tfrac32Nk_BT$ en
$C_V =
\tfrac32R$ per mol; een tweeatomig gas voegt twee rotaties toe,
$C_V = \tfrac52R$,
en zijn trilling (twee kwadratische termen, maar gekwantiseerd met
$h\nu \gg
k_BT$ bij kamertemperatuur) blijft bevroren tot enkele duizenden
kelvin.
\end{theorem}

\begin{proof}
De kansdichtheid in $q$ is $\propto\eu^{-aq^2/k_BT}$, een gaussische:
\[
\langle aq^2\rangle = \frac{\int aq^2\,\eu^{-aq^2/k_BT}\,\dd q}{\int\eu^{-aq^2/k_BT}\,\dd q} = \tfrac12k_BT
\]
(differentieer $\int\eu^{-\lambda q^2}\dd q = \sqrt{\pi/\lambda}$ naar
$\lambda$). De uitkomst vergt het continuüm: wanneer de
niveaus verder dan $k_BT$ uit elkaar liggen, bevriest de factor de veranderlijke,
zoals bij de oscillator hierboven.
\end{proof}

\begin{theorem}[Snelheidsverdeling van Maxwell]\label{thm:b2:boltzmann-factor:maxwell}
In een gas op $T$ zijn de snelheidscomponenten onafhankelijke gaussische
verdelingen met
variantie $k_BT/m$, en de snelheid $v = |\vect v|$ heeft de kans\-dichtheid
\[
f(v) = 4\pi\Big(\frac{m}{2\pi k_BT}\Big)^{3/2}v^2\,\eu^{-mv^2/2k_BT} ,
\]
met de meest waarschijnlijke, de gemiddelde en de kwadratisch gemiddelde snelheid
\[
v_{\text{p}} = \sqrt{\frac{2k_BT}{m}}, \qquad
\langle v\rangle = \sqrt{\frac{8k_BT}{\pi m}}, \qquad
v_{\text{rms}} = \sqrt{\frac{3k_BT}{m}} ,
\]
in de verhoudingen $1 : 1.13 : 1.22$. Voor stikstof bij \qty{300}{K}: \qty{422}{},
\qty{476}{}, \qty{517}{m/s}; voor waterstof $3.7$ keer meer; de
staart van de verdeling, $\propto\eu^{-mv^2/2k_BT}$, is wat de
snelste moleculen laat verdampen, reageren of een planeet verlaten.
\end{theorem}

\begin{proof}
De kinetische energie $\tfrac12m(v_x^2 + v_y^2 + v_z^2)$ geeft een factor
$\eu^{-mv_x^2/2k_BT}\eu^{-mv_y^2/2k_BT}\eu^{-mv_z^2/2k_BT}$: drie onafhankelijke
gaussische verdelingen, elk genormeerd door $\sqrt{m/2\pi k_BT}$. De
snelheidsverdeling verzamelt alle snelheden in de schil met straal $v$ en
volume $4\pi v^2\dd v$. Dan volgt $v_{\text{p}}$ uit $\dd f/\dd v = 0$, en de
momenten uit $\int_0^\infty v^3\eu^{-av^2}\dd v = 1/2a^2$,
$\int_0^\infty
v^4\eu^{-av^2}\dd v = \tfrac38\sqrt{\pi/a^5}$;
$v_{\text{rms}}$ volgt ook
uit de gelijke verdeling, $\tfrac12m\langle v^2\rangle = \tfrac32k_BT$.
\end{proof}

\begin{example}[Wie de planeet verlaat]\label{ex:b2:boltzmann-factor:escape}
De ontsnappingssnelheid van de aarde is \qty{11.2}{km/s}. In de exosfeer, nabij
\qty{1000}{K}, heeft helium $v_{\text{p}} = \qty{2.0}{km/s}$: het deel van de
atomen boven de ontsnappingssnelheid is van de orde
$\eu^{-(11.2/2.0)^2} = \eu^{-31}
\approx 10^{-13}$ --- klein, maar bij elke
botsing vernieuwd, miljarden
jaren lang: helium lekt weg, en zijn aandeel in de lucht is maar
\qty{5}{ppm} ondanks zijn gestage aanmaak door radioactief verval. Voor
stikstof is de exponent zeven keer groter, $\eu^{-220}$: stikstof blijft.
Op de maan ($v_{\text{ontsn}} = \qty{2.4}{km/s}$) heeft zelfs stikstof bij
\qty{400}{K} ($v_{\text{p}} = \qty{490}{m/s}$) $\eu^{-24}$ per botsingstijd:
over de leeftijd van het zonnestelsel overleeft geen dampkring.
\end{example}

\section{Paramagnetisme en de wet van Curie}

\begin{proposition}[Paramagneet met spin 1/2]\label{prop:b2:boltzmann-factor:curie}
$N$ onafhankelijke magnetische momenten die alleen met of tegen
het veld $B$ kunnen wijzen, met energieën $\mp\mu B$, hebben bij temperatuur
$T$ het gemiddelde
moment en de magnetisatie
\[
\langle\mu_z\rangle = \mu\tanh\frac{\mu B}{k_BT}, \qquad
M = n\mu\tanh\frac{\mu B}{k_BT}
\ \longrightarrow\
\begin{cases} n\mu^2B/k_BT & (\mu B \ll k_BT):\ \text{wet van Curie}, \\ n\mu & (\mu B \gg k_BT):\ \text{verzadiging}. \end{cases}
\]
De susceptibiliteit $\chi = \mu_0M/B = \mu_0n\mu^2/k_BT$ verandert als $1/T$
(de \emph{wet van Curie}\index{wet van Curie}); met
$\mu = \mu_B = \qty{9.27e-24}{J/T}$
is $\mu B/k_BT = 2.2 \times 10^{-3}$ bij \qty{1}{T} en \qty{300}{K}, zodat gewone
paramagneten nauwelijks gemagnetiseerd zijn; bij \qty{1}{K} geeft hetzelfde veld
$0.67$ en begint de verzadiging.
\end{proposition}

\begin{proof}
$p_\pm \propto\eu^{\pm\mu B/k_BT}$:
$\langle\mu_z\rangle = \mu(\eu^x - \eu^{-x})/(\eu^x +
\eu^{-x})$ met
$x = \mu B/k_BT$; $\tanh x \approx x$ voor kleine $x$.
\end{proof}

\begin{remark}[Wat de factor niet doet]\label{rem:b2:boltzmann-factor:limits}
De boltzmannfactor beschrijft het \emph{evenwicht} bij een temperatuur; hij
zegt niets over snelheden (hoe snel het evenwicht wordt bereikt --- al
geeft een activeringsenergie $E_{\text{a}}$ snelheden
$\propto\eu^{-E_{\text{a}}/k_BT}$,
de wet van Arrhenius die wij bij diffusie in vaste stoffen tegenkwamen). Hij
geldt voor
onafhankelijke deelsystemen of voor het geheel; voor gelijke kwantumdeeltjes
bij hoge dichtheid (elektronen in een metaal, fotonen, helium nabij
het absolute nulpunt) wordt hij vervangen door de verdelingen van Fermi--Dirac en
Bose--Einstein van het volume van jaar 3, waarvan hij de ijle grens is.
\end{remark}

\begin{method}[Boltzmannschattingen]\label{meth:b2:boltzmann-factor:method}
(1) Bereken $E/k_BT$; $k_BT = \qty{0.026}{eV}$ bij \qty{300}{K}, \qty{1}{eV}
bij \qty{11600}{K}. (2) Verhoudingen van bevolkingen: $\eu^{-\Delta E/k_BT}$ (maal
ontaardingen). (3) Afzonderlijke niveaus: $Z$, dan
$\langle E\rangle =
-\partial_\beta\ln Z$, dan $C$. (4) Kwadratische
continue veranderlijken:
elk $\tfrac12k_BT$, indien niet bevroren. (5) Snelheden:
$v_{\text{p}} = \sqrt{2k_BT/m}$
en de gaussische staart. (6) Spins: $\tanh(\mu B/k_BT)$, de $1/T$ van Curie.
\end{method}

\qty{0.26}{bar}); \omterm{prop:b2:boltzmann-factor:barometric}{schaalhoogte} van helium alleen; van de marsdampkring
(CO$_2$, \qty{210}{K}, $g = \qty{3.7}{m/s^2}$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:boltzmann-factor:4}
Momenten met spin 1/2 $\mu = \mu_B$, $n = \qty{1e28}{m^{-3}}$, $B = \qty{1}{T}$:

\begin{exercise}[$\star\star$]\label{exo:b2:boltzmann-factor:5}
\emph{Twee niveaus.} (a) Leid $p_1$, $p_2$, $\langle E\rangle$ af uit $Z$. (b)
Toon aan dat $N_2 < N_1$ bij elke positieve temperatuur; wat zou $N_2 >
N_1$
voor $T$ betekenen? (c) Entropie $S = -Nk_B\sum p_i\ln p_i$ van het stelsel
bij
$T \to 0$ en $T \to \infty$. (d) Een defect in een kristal kan in twee
standen zitten die \qty{0.05}{eV} uit elkaar liggen: bezetting van de bovenste bij

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:boltzmann-factor:6}

\end{exercise}

een eenatomig en van een

\qty{3000}{K}. (d) Einsteinvaste stof: $C$ bij $T = \theta_{\text{E}}$,

$4\theta_{\text{E}}$ in eenheden van $3R$; diamant bij kamertemperatuur.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:boltzmann-factor:8}
\emph{Maxwell in bijzonderheden.} (a) Toon aan dat de snelheidscomponenten
onafhankelijke gaussische verdelingen zijn en leid $f(v)$ af. (b) Bereken

$\langle v\rangle$, $v_{\text{rms}}$. (c) Gemiddelde kinetische energie: ga na met

$\tfrac14n\langle v\rangle$ per oppervlakte-eenheid en per tijdseenheid
(aangenomen): aantal
moleculen N$_2$ dat per seconde op \qty{1}{cm^2} botst bij \qty{1}{bar},
\qty{300}{K}; en het tempo waarmee een gaatje van \qty{1}{\micro m} lucht
in een vacuümkamer laat (effusie).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:boltzmann-factor:9}
\emph{Ontsnapping.} (a) Deel van een maxwellgas sneller dan $v$: toon aan dat het
ongeveer $(2/\sqrt\pi)\,x\,\eu^{-x^2}$ is voor $x = v/v_{\text{p}} \gg 1$.
(b) Helium bij \qty{1000}{K} boven \qty{11.2}{km/s}; stikstof. (c) Maan,
$v_{\text{ontsn}} = \qty{2.4}{km/s}$, \qty{400}{K}: stikstof, en welke
temperatuur haar zou vasthouden gedurende de leeftijd van het zonnestelsel
(neem $10^{17}$
botsingstijden: ontsnappingsdeel onder $10^{-17}$). (d) Waarom houdt Titan
(\qty{2.6}{km/s}, \qty{94}{K}) een dikke stikstofdampkring?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:boltzmann-factor:10}
\emph{Andere potentialen.} (a) In een centrifuge die met $\omega$ draait, is de
effectieve potentiële energie $-\tfrac12m\omega^2r^2$: dichtheidsprofiel
$n(r)$. (b) Uraniumhexafluoride, twee isotopen $\Delta m = \qty{3}{u}$,
$\omega
= 2\pi \times \qty{1000}{Hz}$, $r = \qty{10}{cm}$, \qty{300}{K}:
verrijkingsfactor
van één trap. (c) Colloïdale bolletjes (opdrijfmassa
$m' =
\qty{2.6e-17}{kg}$) in water bij \qty{293}{K}: \omterm{prop:b2:boltzmann-factor:barometric}{schaalhoogte} van hun
bezinkingsevenwicht; hoe gaf dat het getal van Avogadro? (d) Een
elektronengas in een veld $E$: waarom is het boltzmannprofiel
$\eu^{eEx/k_BT}$ de grondslag van de debyeafschermingslengte
$\lambda_{\text{D}} = \sqrt{\varepsilon_0k_BT/ne^2}$ van een \omterm{def:b2:waves-in-media:plasma}{plasma} (schets het
argument)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:boltzmann-factor:11}
\emph{Planck uit Boltzmann.} (a) Bereken $Z$ en $\langle E\rangle$ voor
de niveaus $nh\nu$. (b) De twee grensgevallen en hun betekenis. (c)

(d) Waarom faalt de klassieke telling, in één zin?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:boltzmann-factor:12}
\emph{Adiabatische demagnetisatie.} De entropie van $N$ spins 1/2 is
$S =
Nk_B[\ln(2\cosh x) - x\tanh x]$, $x = \mu B/k_BT$. (a) Toon aan dat zij
alleen
van $B/T$ afhangt, met $S \to Nk_B\ln 2$ voor $B/T \to 0$ en $S \to 0$ voor
$B/T
\to \infty$. (b) Een zout op \qty{1}{K} wordt isotherm gemagnetiseerd
van $0$ tot
\qty{1}{T}: weggenomen entropie (per spin, met $\mu = \mu_B$). (c) Het veld
wordt vervolgens adiabatisch tot \qty{0.01}{T} verlaagd: eindtemperatuur. (d)
Wat begrenst de methode, en hoe dient dezelfde $\tanh$-kromme als
thermometer?
\end{exercise}

\section{Vraagstuk: de isotherme dampkring, de ontsnapping van helium en een
spinthermometer}

\begin{problem}[{Weekendvraagstuk --- drie toepassingen van één exponentiële}]\label{pb:b2:boltzmann-factor:1}
Gegevens: $k_B = \qty{1.38e-23}{J/K}$, $N_A = \qty{6.02e23}{mol^{-1}}$,
$g =
\qty{9.8}{m/s^2}$, $R_{\oplus} = \qty{6.37e6}{m}$,
$\mu_B = \qty{9.27e-24}{J/T}$;
lucht $M = \qty{29}{g/mol}$, helium \qty{4}{g/mol}, stikstof \qty{28}{g/mol}.

\textbf{Deel I --- De isotherme dampkring.}
\begin{enumerate}
  \item Leid uit de hydrostatica en de wet van het volmaakte gas
        $P(z) =
        P_0\eu^{-z/H}$ af en geef $H$ voor lucht bij
        \qty{288}{K}.
  \item Wijs de boltzmannfactor in het resultaat aan; welke energie,
        welke temperatuur?
  \item Druk op \qty{3000}{m}, \qty{5500}{m} en \qty{8849}{m}; de
        hoogte waar de druk gehalveerd is.
  \item Massa van de dampkring per vierkante meter (uit
        $P_0 =
        \qty{1.013e5}{Pa}$), en de totale massa; ga na dat
        $\int_0^\infty
        \rho\,\dd z = \rho_0H$.
  \item Aantal moleculen in de dampkring.
  \item Als elk gas zijn eigen $H$ volgde, wat zou dan de verhouding
        O$_2$/N$_2$ op \qty{50}{km} zijn vergeleken met de grond? Waarom is de
        samenstelling in werkelijkheid gelijkmatig tot \qty{100}{km}?
  \item De werkelijke troposfeer koelt met \qty{6.5}{K/km} af: is de druk
        op \qty{10}{km} hoger of lager dan de isotherme schatting?
        Verklaar.
  \item \omterm{def:b2:particle-diffusion:flux}{Aantalsdichtheid} op zeeniveau en op \qty{100}{km} (isotherm
        model, \qty{288}{K}): bespreek ``de rand van de ruimte''.
\end{enumerate}

\textbf{Deel II --- De ontsnapping van helium.}
De exosfeer, boven \qty{500}{km}, staat op ongeveer \qty{1000}{K} en is
botsingsvrij: een molecuul dat sneller dan de ontsnappingssnelheid omhoog beweegt,
vertrekt.
\begin{enumerate}[resume]
  \item Ontsnappingssnelheid van de aarde op die hoogte.
  \item Meest waarschijnlijke en gemiddelde snelheden van helium en stikstof bij
        \qty{1000}{K}.
  \item Deel van de heliumatomen met $v > v_{\text{ontsn}}$ (gebruik
        $(2/\sqrt\pi)x\eu^{-x^2}$); idem voor stikstof.
  \item De ontsnappingsflux van helium is ruwweg
        $\tfrac14n_{\text{He}}\langle
        v\rangle \times$ (dat deel)
        $\times\tfrac12$; schat met
        $n_{\text{He}}
        = \qty{1e12}{m^{-3}}$ aan de exobasis het
        verlies per vierkante
        meter per seconde en per jaar voor de hele aarde.
  \item Helium in de lucht is \qty{5}{ppm} naar volume: totaal helium in de
        dampkring; zijn verblijftijd tegenover het berekende verlies.
  \item Radioactiviteit in de aardkorst maakt ongeveer \qty{3e6}{kg} helium
        per jaar: is het helium van de dampkring in stationaire toestand, en wat
        leert de vergelijking met vraag 11?
  \item Bij zonnemaximum haalt de exosfeer \qty{2000}{K}: met welke
        factor stijgt het ontsnappingsdeel van helium?
  \item Waarom heeft de maan geen dampkring, en Titan (ontsnapping
        \qty{2.6}{km/s}, \qty{94}{K}) een dikke?
\end{enumerate}

\textbf{Deel III --- Een spinthermometer.}
Een paramagnetisch zout bevat $n = \qty{2e27}{m^{-3}}$ spins 1/2 met
$\mu
= \mu_B$, in een veld $B$.
\begin{enumerate}[resume]
  \item Bevolkingen van de twee niveaus en de magnetisatie
        $M(B,T)$.
  \item Bij $B = \qty{0.1}{T}$: $M$ bij \qty{300}{K}, \qty{4}{K}, \qty{0.05}{K};
        waar geldt de \omterm{prop:b2:boltzmann-factor:curie}{wet van Curie}?
  \item Waarom is $M$ een thermometer, en in welk bereik is zij het
        gevoeligst? Wat wordt er in de praktijk gemeten?
  \item Gemiddelde energie en warmtecapaciteit van de spins (Schottky);
        temperatuur van de piek bij \qty{0.1}{T}.
  \item Entropie van de spins bij hoge en bij nultemperatuur; welke
        ``ordening'' treedt op als $T \to 0$?
  \item Adiabatische demagnetisatie van \qty{1}{K}, \qty{1}{T} tot \qty{0.01}{T}:
        eindtemperatuur; waarom kan $B \to 0$ geen $T \to 0$ geven?
  \item Een kernspinthermometer gebruikt $\mu_{\text{N}} = \mu_B/1836$: tot
        welke temperatuur geldt bij \qty{1}{T} de \omterm{prop:b2:boltzmann-factor:curie}{wet van Curie}, en
        waarom wordt zo'n thermometer in het microkelvinbereik gebruikt?
  \item Vergelijk de dopplerbreedte van een spectraallijn (uit de verdeling
        van Maxwell) als thermometer voor gassen: welke grootheid, en
        hoe schaalt zij met $T$?
  \item Vat samen: de drie stelsels en de ene formule.
\end{enumerate}
\end{problem}
