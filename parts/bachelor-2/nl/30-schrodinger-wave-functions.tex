\chapter{De schrödingervergelijking en
golffuncties}\label{ch:b2:schrodinger-wave-functions}

Schiet elektronen één voor één op een paar spleten af en teken op waar
elk landt: een stip hier, een stip daar, schijnbaar willekeurig --- en
na tienduizend stippen de strepen van de proef van Young, alsof elk
elektron door beide spleten was gegaan en met zichzelf had
geïnterfereerd. Het laatste hoofdstuk van het volume van jaar 1 kwam
die vreemdheid tegen: licht komt in fotonen, materie heeft een
golflengte $\lambda = h/p$, en plaats en impuls kunnen niet allebei
scherp zijn. Wat het niet gaf, is de \emph{vergelijking} --- de regel
die zegt hoe de golf die bij een deeltje hoort evolueert, zoals de wet
van Newton zegt hoe een baan evolueert. Die vergelijking is die van
Schrödinger (1926), en daarmee wordt de kwantumbeschrijving een
berekenbare theorie: men schrijft een \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{golffunctie}} $\psi(x,t)$,
waarvan de gekwadrateerde modulus de kans is om het deeltje aan te
treffen, en men lost een lineaire partiële differentiaalvergelijking
op. Dit hoofdstuk formuleert de vergelijking, verklaart wat $\psi$
betekent en wat niet, vindt haar \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden} en de
golfpakketten van het vrije deeltje, en toont waar de mechanica van
Newton weer opduikt --- voor alles wat zwaarder is dan een molecuul.

\section{Wat het volume van jaar 1 ons naliet}

\begin{proposition}[Deeltjes zijn golven]\label{prop:b2:schrodinger-wave-functions:recap}
Een foton met frequentie $\nu$ heeft de energie $E = h\nu = \hbar\omega$ en de
impuls $p = h/\lambda = \hbar k$; een materieel deeltje met impuls $p$
gedraagt zich als een golf met golflengte $\lambda = h/p$ (De Broglie),
eveneens met $E =
\hbar\omega$ en $p = \hbar k$ --- elektronenbuiging door
kristallen en interferometrie met neutronen en zelfs met moleculen
bevestigen het; en geen enkele
toestand kan een plaats en een impuls scherper hebben dan
$\Delta x\,\Delta p
\ge \hbar/2$ (Heisenberg). Met
$\hbar = h/2\pi = \qty{1.05e-34}{J.s}$: een
elektron van \qty{100}{eV} heeft $\lambda = \qty{0.12}{nm}$, de grootte van een
atoom; een stofkorrel van \qty{1e-15}{kg} bij \qty{1}{mm/s} heeft
$\lambda =
\qty{7e-16}{m}$, kleiner dan een kern --- zij gedraagt zich
klassiek.
\end{proposition}

\begin{proof}
Herhaald uit het volume van jaar 1; de taak is nu de dynamica van de
golf.
\end{proof}

\section{De golffunctie}

\begin{definition}[Golffunctie en de regel van Born]\label{def:b2:schrodinger-wave-functions:psi}
De toestand van een deeltje dat langs $x$ beweegt, wordt beschreven door een
complexe
\emph{golffunctie}\index{golffunctie} $\psi(x,t)$ zodanig dat
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
de kans is om het deeltje tussen $x$ en $x + \dd x$ aan te treffen
op het tijdstip $t$ (de \emph{regel van Born}\index{regel van Born}); vandaar de
\emph{normering} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, en de
verwachtingswaarden $\langle x\rangle = \int x|\psi|^2\dd x$,
$\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$. Golffuncties
gehoorzamen aan het
\emph{superpositiebeginsel}\index{superpositiebeginsel}: als
$\psi_1$ en $\psi_2$ mogelijke toestanden zijn, dan ook
$\alpha\psi_1 +
\beta\psi_2$ (genormeerd), en de kansen bevatten dan de
interferentieterm $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$. Een
overkoepelende fasefactor $\eu^{\iu\theta}$ verandert niets; een betrekkelijke fase
tussen twee opgetelde componenten verandert de interferentie.
\end{definition}

\begin{remark}[Wat $\psi$ niet is]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ is geen klassiek veld dat zich zoals $\vect E$ in de ruimte
uitspreidt, noch een
wolk materie: het elektron wordt altijd heel aangetroffen, in één punt.
$\psi$ is een \emph{kansamplitude}, het voorwerp dat
interfereert; alleen $|\psi|^2$ wordt waargenomen, en alleen statistisch, door
de proef op gelijk voorbereide deeltjes te herhalen. De stippen
van de dubbele spleet zijn afzonderlijke elektronen; de strepen zijn
$|\psi_1 +
\psi_2|^2$.
\end{remark}

\section{De schrödingervergelijking}

\begin{theorem}[Schrödingervergelijking]\label{thm:b2:schrodinger-wave-functions:equation}
Een deeltje met massa $m$ in de potentiële energie $V(x)$ heeft een golf\-functie
die voldoet aan
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
Zij is lineair (superpositie), van eerste orde in de tijd (de \omterm{def:b2:schrodinger-wave-functions:psi}{golffunctie}
op één ogenblik legt haar op alle latere tijdstippen vast), en complex ($\psi$
kan niet reëel zijn: de $\iu$ is wezenlijk). Zij is een \emph{postulaat},
verantwoord door haar gevolgen.
\end{theorem}

\begin{proof}
Eerder een motivering dan een bewijs. Een vrij deeltje met scherpe impuls
$p = \hbar k$ en energie $E = \hbar\omega$ zou een vlakke golf
$\psi =
A\eu^{\iu(kx - \omega t)}$ moeten zijn, waarvoor
$\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ en
$-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$.
De klassieke betrekking $E = p^2/2m$ wordt dus weergegeven door de vergelijking
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$, en de natuurlijke
veralgemening voor $E = p^2/2m + V$ voegt $V\psi$ toe. Merk op dat de
vergelijking lineair moet zijn om voor superposities van vlakke golven te gelden, en
van eerste orde in de tijd met een $\iu$ opdat $\eu^{-\iu\omega t}$ en niet
$\cos\omega t$ de tijdafhankelijkheid is --- een reële \omterm{thm:b2:waves-on-strings:equation}{golfvergelijking} zou
$E^2 \propto p^4$ geven.
\end{proof}

\begin{proposition}[Behoud van de kans]\label{prop:b2:schrodinger-wave-functions:current}
De kansdichtheid $\rho = |\psi|^2$ en de \emph{kansstroom}\index{kans
stroom}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
voldoen aan $\partial_t\rho + \partial_xj = 0$: kans wordt noch gemaakt noch
vernietigd, en $\int|\psi|^2\dd x$ blijft gelijk aan $1$. Voor een vlakke golf
$A\eu^{\iu kx}$ is $j = |A|^2\hbar k/m = \rho v$: dichtheid maal snelheid, zoals bij
elke stroming.
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$; uit de
vergelijking en haar toegevoegde,
$\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$,
$\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$;
de termen met $V$ vallen weg; de rest is
$(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$.
\end{proof}

\begin{proposition}[Impuls, verwachtingswaarden en de klassieke grens]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
De impuls van een toestand wordt verkregen met de operator
$\hat p =
-\iu\hbar\,\partial_x$ (die op een vlakke golf $\hbar k$ geeft):
$\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, en de energie
met $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$. De verwachtingswaarden voldoen
aan
(Ehrenfest)
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
het midden van het \omterm{prop:b2:schrodinger-wave-functions:free}{golfpakket} volgt de wet van Newton precies wanneer $V$
hoogstens kwadratisch is, en bij benadering telkens wanneer het pakket smal is
vergeleken met de schaal waarover $V$ verandert --- de klassieke grens,
die voor elk macroscopisch voorwerp geldt omdat $\hbar$ zo klein is.
\end{proposition}

\begin{proof}
Differentieer $\langle x\rangle = \int x|\psi|^2$, gebruik
$\partial_t\rho =
-\partial_xj$ en integreer partieel:
$\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$. De tweede
betrekking is dezelfde berekening één
orde hoger (hier aangenomen).
\end{proof}

\section{Stationaire toestanden}

\begin{theorem}[Stationaire toestanden]\label{thm:b2:schrodinger-wave-functions:stationary}
Als $V$ niet van de tijd afhangt, heeft de vergelijking oplossingen van de
gescheiden vorm
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(de \emph{tijdonafhankelijke schrödingervergelijking}\index{schrödinger
vergelijking!tijdonafhankelijk}): de \emph{stationaire toestanden}\index{stationaire
toestand}, met welbepaalde energie $E$, waarvan de kansdichtheid
$|\varphi|^2$ niet beweegt. De energieën $E$ waarvoor $\varphi$
aanvaardbaar is (begrensd, normeerbaar voor een gebonden toestand) zijn de
\emph{energieniveaus}; de algemene oplossing is een superpositie
$\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$, en een superpositie van
twee niveaus
heeft een dichtheid die trilt op de \emph{bohrfrequentie}\index{bohr
frequentie} $\nu_{21} = (E_2 - E_1)/h$ --- de frequentie van het licht dat het
atoom uitzendt of opneemt.
\end{theorem}

\begin{proof}
Vul $\varphi(x)f(t)$ in:
$\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
een functie van $t$ gelijk aan een functie van $x$, dus een constante $E$;
$f
= \eu^{-\iu Et/\hbar}$. Voor twee niveaus is
$|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$.
De volledigheid van de \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden} wordt aangenomen.
\end{proof}

\section{Het vrije deeltje: vlakke golven en golfpakketten}

\begin{proposition}[Vrij deeltje]\label{prop:b2:schrodinger-wave-functions:free}
Voor $V = 0$ zijn de \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden} de vlakke golven
$\eu^{\iu(kx -
\omega t)}$ met
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{g}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
materiegolven zijn \emph{dispersief}; de \omterm{def:b2:dispersion-wave-packets:relation}{fasesnelheid} heeft geen fysische
betekenis, de \emph{\omterm{prop:b2:dispersion-wave-packets:group}{groepssnelheid}} is de snelheid van het deeltje. Een vlakke
golf is niet normeerbaar --- een deeltje met een volmaakt scherpe impuls
is over de hele ruimte uitgespreid --- zodat een werkelijk deeltje een
\emph{golfpakket}\index{golfpakket} is, een superpositie van vlakke golven
over een band
$\Delta k$, gelokaliseerd over $\Delta x \sim 1/\Delta k$, dat met
$v_{\text{g}}$ beweegt en
zich \emph{uitspreidt}: een gaussisch pakket met beginbreedte $\sigma_0$ heeft op
tijdstip $t$ de breedte
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
en voldoet aan $\Delta x\,\Delta p = \hbar/2$ op $t = 0$ (het kleinst toegelaten)
en meer daarna.
\end{proposition}

\begin{proof}
De \omterm{def:b2:dispersion-wave-packets:relation}{dispersierelatie} volgt uit de vergelijking; $v_{\text{g}}$ zoals in
\cref{ch:b2:dispersion-wave-packets}. De uitspreidingsformule komt uit
de fouriersynthese van het pakket --- elke component $\eu^{\iu kx}$ krijgt
de fase $\eu^{-\iu\hbar k^2t/2m}$, en de gaussische integraal met een
complexe coëfficiënt geeft een gaussische functie met groeiende breedte; de
fourier\-integraal
is het gereedschap van het wiskundevolume van jaar 3 en wij nemen het
resultaat aan. Fysisch: het pakket bevat impulsen gespreid over
$\Delta p
= \hbar/2\sigma_0$, dus snelheden gespreid over
$\Delta v = \hbar/2m\sigma_0$,
en na $t$ zijn de snellere componenten de tragere $\Delta v\,t$ voorbij
--- de formule zegt precies dat.
\end{proof}

\begin{example}[Wie zich uitspreidt]\label{ex:b2:schrodinger-wave-functions:spreading}
Een elektron gelokaliseerd binnen $\sigma_0 = \qty{1}{nm}$:
$\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- na één nanoseconde is
zijn pakket \qty{60}{\micro m}
breed: een elektron dat met rust wordt gelaten blijft niet zitten; in een
atoom gebonden
spreidt het zich niet uit omdat de potentiaal het vasthoudt (een \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire
toestand}).
Een stofkorrel van \qty{1e-15}{kg} tot \qty{1}{\micro m} gelokaliseerd:
$\tau =
\qty{2e7}{s}$, acht maanden; een kogel: langer dan de leeftijd van
het heelal.
De kwantumuitspreiding is echt, en onzichtbaar voor alles wat men kan zien.
\end{example}

\begin{remark}[Energie en tijd]\label{rem:b2:schrodinger-wave-functions:energytime}
Een toestand die maar gedurende $\Delta t$ bestaat --- een aangeslagen atoom
met levensduur
$\tau$, een \omterm{prop:b2:scalar-light-model:coherence}{golftrein} met lengte $c\tau$ --- is een superpositie van energieën
gespreid over $\Delta E \sim \hbar/\Delta t$: de fourierbetrekking van
\cref{ch:b2:scalar-light-model} tussen de duur van een trein en zijn
\omterm{prop:b2:scalar-light-model:coherence}{spectrale breedte}, nu op $E = \hbar\omega$ toegepast. Een atoomniveau met
levensduur \qty{10}{ns} heeft een breedte $\Delta E = \qty{7e-8}{eV}$, een
natuurlijke
lijnbreedte van \qty{16}{MHz}; de dopplerbreedte van \cref{ch:b2:laser}
(\qty{1.5}{GHz}) verbergt haar in een gas, maar niet in een koud atoom of een ion in
rust.
\end{remark}

\begin{method}[Werken met golffuncties]\label{meth:b2:schrodinger-wave-functions:method}
(1) Normeer; kansen zijn $\int|\psi|^2$ over het gebied. (2)
Tijdonafhankelijke $V$: zoek $\varphi(x)\eu^{-\iu Et/\hbar}$, los de
stationaire vergelijking op, leg de randvoorwaarden op --- de niveaus.
(3) Tel \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden} op voor de evolutie; twee niveaus zweven op
$(E_2 - E_1)/h$. (4) Vrije beweging: $v_{\text{g}} = p/m$, uitspreidingstijd
$2m\sigma_0^2/\hbar$. (5) Stromen: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$;
voor $A\eu^{\iu kx} + B\eu^{-\iu kx}$ is $j = (|A|^2 - |B|^2)\hbar k/m$
(geen kruisterm).
(6) Klassieke grens: Ehrenfest, en $\lambda$ tegen de grootte van het
toestel.
\end{method}

\section{Opgaven}

Golflengten van De Broglie: een elektron bij \qty{100}{eV}, \qty{10}{keV},

verbetering vergt); een thermisch neutron (\qty{25}{meV},

\end{exercise}

$\langle x\rangle$, $\Delta x$; kans om het deeltje aan te treffen in

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:3}
Ga na dat $\varphi(x)\eu^{-\iu Et/\hbar}$ de vergelijking oplost wanneer $\varphi$
de stationaire oplost. Frequentie van de fasor voor $E = \qty{1}{eV}$;
voor een superpositie van twee niveaus \qty{2}{eV} uit elkaar, frequentie en
golflengte van de trilling van $|\psi|^2$ --- en van het uitgezonden
licht.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:4}
Uitspreidingstijd $\tau = 2m\sigma_0^2/\hbar$ en breedte na \qty{1}{ns} en
\qty{1}{s}: een elektron met $\sigma_0 = \qty{1}{nm}$ en $\sigma_0 = \qty{1}{mm}$;
een proton met $\sigma_0 = \qty{1}{nm}$; een korrel van \qty{1e-12}{kg} met
$\sigma_0 = \qty{1}{\micro m}$. Waarom blijven voorwerpen waar men ze neerzet?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:5}
\emph{Stroom.} (a) Leid de \omterm{thm:b2:fluid-kinematics:continuity}{continuïteitsvergelijking} af. (b) $j$ voor
$A\eu^{\iu kx}$, voor $A\sin kx$, voor $A\eu^{\iu kx} + B\eu^{-\iu kx}$. (c)
Waarom is
het ontbreken van een kruisterm in het laatste geval de uitspraak dat
weerkaatste en invallende fluxen eenvoudig van elkaar worden afgetrokken?
(d) Toon aan dat voor een reële
\omterm{def:b2:schrodinger-wave-functions:psi}{golffunctie} $j = 0$: een gebonden \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestand} draagt geen stroom.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:6}
\emph{Een gaussisch pakket.} $\psi(x,0) = A\eu^{\iu k_0x}\eu^{-x^2/4\sigma^2}$. (a)
$A$, $\langle x\rangle$, $\Delta x$. (b) $\langle p\rangle$ met
$\hat p =
-\iu\hbar\partial_x$. (c) Neem $\Delta p = \hbar/2\sigma$ aan en
ga
$\Delta x\Delta p = \hbar/2$ na. (d) Bewijs
$\dd\langle x\rangle/\dd t = \langle
p\rangle/m$ uit de
\omterm{thm:b2:fluid-kinematics:continuity}{continuïteitsvergelijking}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:7}
\emph{Elektronen in een buis.} Elektronen die door \qty{100}{V} zijn versneld
doorlopen \qty{30}{cm}. (a) Snelheid, golflengte, vluchttijd. (b) Fase- en
\omterm{prop:b2:dispersion-wave-packets:group}{groepssnelheid}; welke is die van het elektron? (c) Een pakket dat aanvankelijk
\qty{10}{\micro m} breed is: uitspreidingstijd, breedte bij aankomst. (d) \omterm{def:b2:diffraction:huygens}{Buiging}
door een opening van \qty{0.3}{mm} onderweg: hoekspreiding en vlek op het
scherm. Is het stralenbeeld geldig?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:8}
\emph{Fasen.} (a) Toon aan dat $\psi$ en $\eu^{\iu\theta}\psi$ dezelfde
voorspellingen geven. (b) Schrijf voor
$\psi = (\psi_1 + \eu^{\iu\alpha}\psi_2)/\sqrt2$
$|\psi|^2$ en toon aan dat $\alpha$ ertoe doet. (c) Wat zijn in de dubbele spleet
$\psi_1$ en $\psi_2$, en wat legt hun betrekkelijke fase in een punt
van het scherm vast? (d) Waarom vernietigt het opsporen van de spleet waar
het elektron doorging
de strepen (kwalitatief)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:9}
\emph{Ehrenfest.} (a) Toon voor $V = \tfrac12m\omega^2x^2$ aan dat
$\langle
x\rangle$ precies aan
$\ddot{\langle x\rangle} = -\omega^2\langle x\rangle$ voldoet.
(b) Voor $V = mgx$ valt $\langle x\rangle$ als een steen. (c) Wanneer geldt
voor een algemene
$V$ dat $\langle V'(x)\rangle \approx V'(\langle x\rangle)$? (d) Een
kraaltje van \qty{1}{g} in een kom: schat de grootte van zijn pakket die
nodig is opdat
het ``kwantum'' zou zijn, en vergelijk met een atoom.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:10}
\emph{Klotsen.} In een doos $0 < x < L$ met ondoordringbare wanden zijn de
\omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden} $\varphi_n = \sqrt{2/L}\sin(n\pi x/L)$,
$E_n =
n^2h^2/8mL^2$ (\cref{ch:b2:potential-wells-tunneling}). Neem
$\psi =
(\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/\sqrt2$.
(a)
$|\psi|^2$ als functie van de tijd. (b) Toon aan dat
$\langle x\rangle = L/2 -
(16L/9\pi^2)\cos\omega_{21}t$ (gebruik
$\int_0^L x\sin(\pi x/L)\sin(2\pi x/L)\dd x
= -8L^2/9\pi^2$). (c) Elektron,
$L = \qty{1}{nm}$: $E_2 - E_1$, de
frequentie en de golflengte van de trilling. (d) Wat doet een
trillende lading (\cref{ch:b2:dipole-radiation})? Besluit over
spectraallijnen.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{Uitspreiding, afgeleid.} Het pakket $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
is de superpositie $\int g(k)\eu^{\iu kx}\dd k$ met
$g(k) \propto
\eu^{-\sigma_0^2k^2}$ (aangenomen: de fouriergetransformeerde
van een gaussische functie is
gaussisch, breedte $1/2\sigma_0$ in $k$). (a) Schrijf $\psi(x,t)$ door elke
component haar fase $\eu^{-\iu\hbar k^2t/2m}$ te geven. (b) Maak het kwadraat in
de exponent volledig en neem
$\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$
aan voor complexe $\alpha$ met positief reëel deel: toon aan dat
$|\psi|^2$ een gaussische functie is met breedte
$\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$. (c) Duid $\tau$
via $\Delta v = \Delta p/m$. (d) Waarom spreidt
een pakket fotonen zich in vacuüm niet uit?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{Energie--tijd.} (a) Een toestand die als
$\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$ vervalt
(kans $\eu^{-t/\tau}$) is een superpositie van energieën: neem aan
dat haar energieverdeling een lorentziaan met volle breedte
$\Gamma =
\hbar/\tau$ is, en geef $\Gamma$ voor $\tau = \qty{10}{ns}$
(atoom), \qty{1e-8}{s},
\qty{1e-23}{s} (een hadronische resonantie, in MeV). (b) Natuurlijke lijnbreedte van
het niveau van \qty{10}{ns} in frequentie; vergelijk met de dopplerbreedte
\qty{1.5}{GHz}. (c) Hoe ziet men de natuurlijke breedte (koude atomen,
gevangen ionen)? (d) Breng dat in verband met de \omterm{prop:b2:scalar-light-model:coherence}{coherentielengte} van de uitgezonden
\omterm{prop:b2:scalar-light-model:coherence}{golftrein}.
\end{exercise}

\section{Vraagstuk: het elektron in een buis en de uitspreiding van een golfpakket}

\begin{problem}[{Weekendvraagstuk --- wanneer een elektron een deeltje is,
wanneer een golf}]\label{pb:b2:schrodinger-wave-functions:1}
Gegevens: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$,
$m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$,
$m_{\text{p}} = \qty{1.67e-27}{kg}$.

\textbf{Deel I --- De kathodestraalbuis.}
Elektronen verlaten een hete kathode met verwaarloosbare energie, worden
door $U = \qty{20}{kV}$ versneld, gaan door een opening van \qty{0.3}{mm} en
treffen een
scherm op \qty{40}{cm} afstand.
\begin{enumerate}
  \item Snelheid en impuls (niet-relativistisch); golflengte van De Broglie.
  \item De kinetische energie is $4\%$ van $mc^2$: is de niet-relativistische
        behandeling voor dit vraagstuk aanvaardbaar?
  \item Vluchttijd tot het scherm.
  \item \omterm{def:b2:diffraction:huygens}{Buiging} door de opening: hoekspreiding $\sim\lambda/a$ en
        de vlek die zij op het scherm maakt; vergelijk met een pixel
        (\qty{0.3}{mm}).
  \item Heisenberg bij de opening: $\Delta p_y \gtrsim \hbar/2a$ geeft dezelfde
        spreiding --- ga dat na.
  \item Een pakket van \qty{0.3}{mm} breed langs de bundel: uitspreidingstijd
        $2m\sigma_0^2/\hbar$ en breedte bij het scherm.
  \item Afbuigplaten oefenen een dwarskracht uit: hoe beweegt volgens Ehrenfest
        $\langle y\rangle$? Besluit dat de buis een
        klassiek toestel is.
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Fasesnelheid} van de elektronengolf; waarom is het geen tegenspraak dat
        zij van de snelheid van het elektron verschilt?
\end{enumerate}

\textbf{Deel II --- Golfpakketten.}
\begin{enumerate}[resume]
  \item Voor een gaussisch pakket met breedte $\sigma_0$: $\Delta p$ en de
        snelheidsspreiding $\Delta v$.
  \item Uitspreidingstijd voor een elektron met $\sigma_0 = \qty{0.1}{nm}$,
        \qty{1}{nm}, \qty{1}{\micro m}; voor een proton met \qty{1}{nm}; voor een
        virus (\qty{1e-17}{kg}) met \qty{10}{nm}.
  \item Een elektron in een waterstofatoom is tot \qty{0.1}{nm} gelokaliseerd
        en spreidt zich niet uit: waarom?
  \item Een vrij elektron van \qty{1}{eV} met $\sigma_0 = \qty{1}{nm}$: hoe ver
        reist het voordat zijn breedte verdubbelt? Vergelijk met de
        golflengte.
  \item In een metaal zijn de geleidingselektronen golven met
        $\lambda \approx
        \qty{0.5}{nm}$ over het hele kristal: in
        welke zin zijn zij
        ``gedelokaliseerd''?
  \item Een elektronenmicroscoop bij \qty{100}{keV} heeft
        $\lambda \approx
        \qty{4}{pm}$, en lost toch maar
        \qty{0.1}{nm} op: waarom (zijn magnetische
        lenzen nemen een halve hoek van ongeveer \qty{10}{mrad} aan)?
  \item Geef het criterium dat beslist of een voorwerp met een \omterm{def:b2:schrodinger-wave-functions:psi}{golffunctie}
        dan wel met Newton moet worden behandeld.
\end{enumerate}

\textbf{Deel III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{Stationaire toestanden} en overgangen.}
Een elektron is opgesloten in $0 < x < L = \qty{1}{nm}$;
$\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$.
\begin{enumerate}[resume]
  \item $E_1$, $E_2$, $E_3$ in elektronvolt; de \omterm{thm:b2:schrodinger-wave-functions:stationary}{bohrfrequenties}
        $\nu_{21}$, $\nu_{31}$ en de golflengten.
  \item Ga na dat $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ tijdonafhankelijk is;
        waar bevindt het elektron zich in de grondtoestand het waarschijnlijkst?
  \item Voor
        $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$:
        $|\psi|^2$ bij $t = 0$ en $t = T_{21}/2$; schets.
  \item $\langle x\rangle(t)$ (gegeven
        $\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$):
        amplitude van het klotsen.
  \item De trillende dipool $e\langle x\rangle$ straalt: op welke
        frequentie, en wat zegt dat over spectraallijnen en
        over de stabiliteit van \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden}?
  \item Energie van de toestand $\psi$: is zij $E_1$, $E_2$, of iets
        anders? Wat geeft een meting?
\end{enumerate}

\textbf{Deel IV --- Duiding.}
\begin{enumerate}[resume]
  \item Wat is er in de proef met de dubbele spleet en één elektron willekeurig
        en wat niet?
  \item Schrijf de \omterm{prop:b2:schrodinger-wave-functions:current}{kansstroom} van $A\eu^{\iu kx}$ en breng haar in verband met
        een bundel van $n$ elektronen per lengte-eenheid met snelheid $v$.
  \item Waarom moet de schrödingervergelijking complex en van eerste orde
        in de tijd zijn (twee redenen)?
  \item Vat in vijf regels samen: de \omterm{def:b2:schrodinger-wave-functions:psi}{golffunctie}, de vergelijking, de
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationaire toestanden}, het pakket, de klassieke grens.
\end{enumerate}
\end{problem}
