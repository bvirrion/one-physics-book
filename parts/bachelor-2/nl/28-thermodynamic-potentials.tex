\chapter{Thermodynamische potentialen}\label{ch:b2:thermodynamic-potentials}

Rek een elastiekje snel uit en houd het tegen uw lip: het is warm;
laat het samentrekken en het is koel. Hang er een gewicht aan en verwarm
het met een haardroger: het \emph{wordt korter}. Geen stalen veer gedraagt
zich zo, en de reden is dat de spanning van een elastiekje geen zaak van
energie is maar van entropie --- de uitgerekte ketens hebben minder manieren
om zich te schikken. Om over zulke dingen te redeneren moet men weten wat
een systeem bij vaste temperatuur, of bij vaste temperatuur en druk, geneigd
is te doen, en hoeveel arbeid eruit kan worden gehaald: vragen die de
inwendige energie en de entropie maar onhandig beantwoorden, omdat zij de
taal van een geïsoleerd systeem spreken. Dit hoofdstuk voert de twee functies
in die voor de omstandigheden van het laboratorium zijn gebouwd, de
\emph{\omterm{def:b2:thermodynamic-potentials:FG}{vrije energie} van Helmholtz} $F = U -
TS$ en de \emph{\omterm{def:b2:thermodynamic-potentials:FG}{vrije enthalpie}
van Gibbs} $G = H - TS$, toont dat zij de terugwinbare arbeid meten en naar
het evenwicht toe dalen, haalt uit hun differentialen de betrekkingen van
Maxwell die grootheden verbinden waarvan niemand een verband zou vermoeden,
en past dat alles toe op het evenwicht van twee fasen, waar $G$ beslist welke
fase wint en de formule van Clapeyron teruggeeft.

\section{De grondidentiteit en haar veranderlijken}

\begin{proposition}[Grondidentiteit]\label{prop:b2:thermodynamic-potentials:identity}
Voor een gesloten systeem met vaste samenstelling voldoet de inwendige energie,
opgevat als functie van de entropie en het volume, aan
\[
\dd U = T\,\dd S - P\,\dd V, \qquad
T = \Big(\frac{\partial U}{\partial S}\Big)_V, \quad
P = -\Big(\frac{\partial U}{\partial V}\Big)_S ,
\]
en de \omterm{thm:b2:open-systems:firstlaw}{enthalpie} $H = U + PV$ voldoet aan $\dd H = T\,\dd S + V\,\dd P$.
Elke functie bevat, in haar \emph{natuurlijke veranderlijken}\index{natuurlijke
veranderlijken} --- $(S,V)$ voor $U$, $(S,P)$ voor $H$ --- de hele
thermodynamica van het systeem: haar partiële afgeleiden geven de overige
toestandsveranderlijken. De tweede hoofdwet kiest het evenwicht van een geïsoleerd
systeem (vaste $U$, $V$) als de toestand van maximale $S$.
\end{proposition}

\begin{proof}
De eerste hoofdwet voor een omkeerbare verandering,
$\dd U = \delta Q_{\text{omk}}
+ \delta W_{\text{omk}} = T\,\dd S - P\,\dd V$,
bevat alleen toestands\-functies
en geldt daarom voor elke oneindig kleine verandering tussen
evenwichtstoestanden. Tel $\dd(PV)$ erbij voor $H$.
\end{proof}

\begin{remark}[Waarom nieuwe functies]\label{rem:b2:thermodynamic-potentials:why}
Proeven worden niet bij vaste entropie gedaan. De kolf van een scheikundige staat op
de temperatuur van de kamer en de druk van de dampkring; een
zuiger in een thermostaat werkt bij vaste $T$. Men wil functies waarvan de
\omterm{prop:b2:thermodynamic-potentials:identity}{natuurlijke veranderlijken} $(T,V)$ en $(T,P)$ zijn, en die onder die
omstandigheden de rol spelen die $S$ voor een geïsoleerd systeem speelt: zeggen
welke kant het systeem uit gaat en hoeveel arbeid het kan geven. Een
\emph{legendretransformatie} --- $TS$ aftrekken om $S$ voor $T$ als
onafhankelijke veranderlijke te ruilen --- bouwt ze.
\end{remark}

\section{Vrije energie en vrije enthalpie}

\begin{definition}[Vrije energie van Helmholtz, vrije enthalpie van Gibbs]\label{def:b2:thermodynamic-potentials:FG}
\[
F = U - TS \quad\text{(\emph{vrije energie}\index{vrije energie})}, \qquad
G = H - TS = U + PV - TS \quad\text{(\emph{vrije enthalpie}\index{vrije
enthalpie})}.
\]
Beide zijn extensieve toestandsfuncties, gemeten in joule.
\end{definition}

\begin{theorem}[Differentialen en afgeleide grootheden]\label{thm:b2:thermodynamic-potentials:dFdG}
\[
\dd F = -S\,\dd T - P\,\dd V, \qquad
\dd G = -S\,\dd T + V\,\dd P ;
\]
dus, in de \omterm{prop:b2:thermodynamic-potentials:identity}{natuurlijke veranderlijken} $(T,V)$ van $F$ en $(T,P)$ van $G$,
\[
S = -\Big(\frac{\partial F}{\partial T}\Big)_V, \quad
P = -\Big(\frac{\partial F}{\partial V}\Big)_T, \qquad
S = -\Big(\frac{\partial G}{\partial T}\Big)_P, \quad
V = \Big(\frac{\partial G}{\partial P}\Big)_T ,
\]
en (Gibbs--Helmholtz)
$U = F - T(\partial F/\partial T)_V =
-T^2\,\partial(F/T)/\partial T$,
$H = -T^2\,\partial(G/T)/\partial T$.
$F(T,V)$ of $G(T,P)$ kennen is alles kennen: de toestandsvergelijking,
de entropie, de energie, de warmtecapaciteiten.
\end{theorem}

\begin{proof}
$\dd F = \dd U - T\dd S - S\dd T = -S\dd T - P\dd V$; evenzo voor $G$
uit $\dd H$. Lees de partiële afgeleiden af; voor Gibbs--Helmholtz geldt
$F - T\,\partial_TF = F + TS = U$ en
$\partial_T(F/T) = (T\partial_TF -
F)/T^2 = -U/T^2$.
\end{proof}

\begin{example}[Het volmaakte gas]\label{ex:b2:thermodynamic-potentials:perfect}
Voor $n$ mol geven $U = nc_{V}T + U_0$ en $S = nc_V\ln T + nR\ln V + S_0$
dat $F(T,V) = -nRT\ln V + f(T)$: dan is $P = -\partial_VF = nRT/V$
--- de toestandsvergelijking komt uit $F$ --- en $S = -\partial_TF$
geeft de entropie terug. In de veranderlijken $(T,P)$ is
$G(T,P) = nRT\ln P +
g(T)$, zodat $V = \partial_PG = nRT/P$, en de
\emph{molaire} \omterm{def:b2:thermodynamic-potentials:FG}{vrije enthalpie}
$\mu(T,P) = \mu^\circ(T) + RT\ln(P/P^\circ)$ --- de \emph{chemische
potentiaal}\index{chemische potentiaal} van het gas, die met zijn
druk groeit: gas stroomt van hoge naar lage $\mu$, hier van hoge naar lage
druk.
\end{example}

\section{Maximale arbeid en de richting van de evolutie}

\begin{theorem}[Monotherme veranderingen]\label{thm:b2:thermodynamic-potentials:monothermal}
Een systeem dat warmte uitwisselt met één enkele thermostaat op $T_0$, en waarvan
de begin- en eindtemperatuur gelijk zijn aan $T_0$, ontvangt tijdens elke
verandering de arbeid
\[
W \ge \Delta F , \qquad\text{d.w.z.}\qquad W_{\text{terug}} = -W \le -\Delta F :
\]
de daling van $F$ is de \emph{maximale terugwinbare arbeid} uit het
systeem bij vaste temperatuur (bereikt voor een omkeerbare verandering) ---
vandaar ``vrije'' energie: het deel van $U$ dat in arbeid kan worden omgezet, de
rest, $TS$, is gebonden. Is de verandering bovendien monobaar
(druk $P_0$ buiten, begin- en einddruk $P_0$), dan voldoet de arbeid
\emph{anders dan die van de dampkring} (\omterm{def:b2:open-systems:cv}{nuttige arbeid}, bijvoorbeeld elektrische)
aan $W_{\text{u}} \ge \Delta G$: $-\Delta G$ is de maximale \omterm{def:b2:open-systems:cv}{nuttige arbeid} die
bij vaste $T$ en $P$ terugwinbaar is (de arbeid van een brandstofcel, van een
batterij).
\end{theorem}

\begin{proof}
Eerste hoofdwet: $\Delta U = W + Q$. Tweede hoofdwet met één thermostaat:
$\Delta S
= Q/T_0 + S_{\text{c}}$, $S_{\text{c}} \ge 0$, dus
$Q \le T_0\Delta S$. Vandaar
$W = \Delta U - Q \ge \Delta U - T_0\Delta S = \Delta(U - T_0S) = \Delta F$, met
$T = T_0$ aan beide uiteinden. Met de arbeid van de dampkring $-P_0\Delta V$
afgezonderd is $W = W_{\text{u}} - P_0\Delta V$ en
$W_{\text{u}} \ge \Delta U + P_0
\Delta V - T_0\Delta S = \Delta G$.
\end{proof}

\begin{theorem}[Criteria voor evolutie en evenwicht]\label{thm:b2:thermodynamic-potentials:criteria}
Een systeem dat bij vaste $T$ en $V$ wordt gehouden en geen arbeid ontvangt,
kan alleen
zo evolueren dat \emph{$F$ daalt}; zijn evenwicht is de toestand van
minimale $F$. Een systeem dat bij vaste $T$ en $P$ wordt gehouden en geen \omterm{def:b2:open-systems:cv}{nuttige
arbeid} ontvangt, kan alleen zo evolueren dat \emph{$G$ daalt}; zijn evenwicht is
de toestand van minimale $G$. (Meer in het algemeen daalt, voor een systeem
in aanraking
met een omgeving op $T_0$, $P_0$, de grootheid $U - T_0S + P_0V$.)
\end{theorem}

\begin{proof}
Neem $W = 0$ (respectievelijk $W_{\text{u}} = 0$) in de vorige stelling:
$\Delta F
\le 0$ (respectievelijk $\Delta G \le 0$), met gelijkheid voor een
omkeerbare, dat wil zeggen
evenwichtige, evolutie.
\end{proof}

\begin{example}[Isotherme uitzetting, werkelijk en ideaal]\label{ex:b2:thermodynamic-potentials:expansion}
Eén mol gas op \qty{300}{K} zet uit van \qty{1}{L} tot \qty{10}{L} in
aanraking met de thermostaat: $\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$. Een
omkeerbare uitzetting wint \qty{5.7}{kJ} terug; een uitzetting tegen de
dampkring op \qty{1}{bar} wint maar $P_0\Delta V = \qty{0.9}{kJ}$ terug; een
vrije uitzetting in vacuüm niets. De \qty{5.7}{kJ} is het maximum dat
het gas en zijn thermostaat voor deze toestandsverandering kunnen geven, hoe
vernuftig de machine ook is.
\end{example}

\section{Betrekkingen van Maxwell}

\begin{theorem}[Betrekkingen van Maxwell]\label{thm:b2:thermodynamic-potentials:maxwell}
Omdat $\dd F$ en $\dd G$ exacte differentialen zijn (de stelling van Schwarz
over gemengde tweede afgeleiden), geldt
\[
\Big(\frac{\partial S}{\partial V}\Big)_T = \Big(\frac{\partial P}{\partial T}\Big)_V,
\qquad
\Big(\frac{\partial S}{\partial P}\Big)_T = -\Big(\frac{\partial V}{\partial T}\Big)_P .
\]
Zij drukken het onmeetbare (hoe de entropie met volume of
druk bij vaste temperatuur verandert) uit door de toestandsvergelijking. Twee
gevolgen:
\[
\Big(\frac{\partial U}{\partial V}\Big)_T = T\Big(\frac{\partial P}{\partial T}\Big)_V - P,
\qquad
C_P - C_V = T\Big(\frac{\partial P}{\partial T}\Big)_V\Big(\frac{\partial V}{\partial T}\Big)_P
= \frac{TV\alpha^2}{\kappa_T} ,
\]
met $\alpha = V^{-1}(\partial V/\partial T)_P$ de uitzettingscoëfficiënt en
$\kappa_T = -V^{-1}(\partial V/\partial P)_T$ de isotherme samendrukbaarheid.
\end{theorem}

\begin{proof}
$\partial^2F/\partial T\partial V$ in beide volgorden berekend:
$-\partial_VS =
-\partial_TP$. Idem voor $G$. Dan geeft
$\dd U = T\dd S - P\dd V$ bij vaste $T$
dat $(\partial_VU)_T = T(\partial_VS)_T - P$; en $C_P - C_V$ volgt
uit het schrijven van $S(T,V(T,P))$:
$(\partial_TS)_P = (\partial_TS)_V +
(\partial_VS)_T(\partial_TV)_P$, maal
$T$.
\end{proof}

\begin{example}[Wat de betrekkingen onthullen]\label{ex:b2:thermodynamic-potentials:reveal}
Voor een volmaakt gas is $T(\partial_TP)_V - P = 0$: de energie hangt niet
van het volume af (de wet van Joule teruggevonden) en $C_P - C_V = nR$. Voor een
vanderwaalsgas is $(\partial_VU)_T = a/V_{\text{m}}^2$: de inwendige druk van
de aantrekkingen. Voor vloeibaar water is
$(\partial_TP)_V = \alpha/\kappa_T = 2.1
\times 10^{-4}/4.6 \times 10^{-10} = \qty{4.6e5}{Pa/K}$:
water in een
star, vol vat één kelvin verwarmen verhoogt de druk met \qty{4.6}{bar}
(sluit nooit een volle fles af), en $(\partial_VU)_T \approx \qty{1400}{bar}$ ---
de samenhang van de vloeistof; toch is
$c_P - c_V = Tv\alpha^2/\kappa_T \approx
\qty{30}{J/kg/K}$, minder dan $1\%$
van $c_P$: voor gecondenseerde materie vallen de twee
warmtecapaciteiten vrijwel samen.
\end{example}

\begin{proposition}[Entropische elasticiteit van rubber]\label{prop:b2:thermodynamic-potentials:rubber}
Voor een band met lengte $L$ onder de spanning $f$ geldt $\dd U = T\dd S + f\dd L$
en $\dd F = -S\dd T + f\dd L$, zodat
$(\partial S/\partial L)_T =
-(\partial f/\partial T)_L$ en
$(\partial U/\partial L)_T = f - T(\partial f/
\partial T)_L$. Rubber
voldoet, in goede benadering, aan $f = T\,\varphi(L)$
met $\varphi > 0$ voor $L > L_0$: dan is $(\partial U/\partial L)_T = 0$ --- de
energie verandert niet bij het uitrekken --- en
$(\partial S/\partial L)_T =
-\varphi(L) < 0$: de spanning is
\emph{entropisch}, $f = -T(\partial S/\partial
L)_T$; de band trekt terug
omdat zijn uitgerekte ketens minder
schikkingen hebben. Gevolgen: isotherm uitgerekt geeft hij warmte af
($Q = T\Delta S < 0$); adiabatisch uitgerekt wordt hij warmer; onder
constante last verwarmd trekt hij samen (vaste $f/T$ betekent vaste
$\varphi(L)$, zodat bij
vaste $f$ $\varphi$ moet dalen als $T$ stijgt).
\end{proposition}

\begin{proof}
De betrekking van Maxwell uit $\dd F$; de rest is invullen. De
microscopische oorsprong is de toevalswandeling van \cref{ch:b2:particle-diffusion}:
een keten van $N$ schakels met lengte $a$ heeft haar afstand van uiteinde
tot uiteinde
verdeeld als een gaussische met variantie $Na^2$, zodat het aantal
schikkingen bij uitrekking $L$ gelijk is aan
$\Omega \propto \eu^{-L^2/2Na^2}$, $S = S_0 -
k_BL^2/2Na^2$ en
$f = k_BTL/Na^2$: een veer waarvan de stijfheid
\emph{evenredig is met de temperatuur}.
\end{proof}

\section{Evenwicht tussen twee fasen}

\begin{theorem}[Voorwaarde voor naast elkaar bestaan]\label{thm:b2:thermodynamic-potentials:coexistence}
Een zuivere stof bij vaste $T$ en $P$, verdeeld over twee fasen met
massa's $m_1$, $m_2$ en vrije enthalpieën per massa-eenheid $g_1(T,P)$,
$g_2(T,P)$, heeft $G = m_1g_1 + m_2g_2$; in evenwicht is $G$ minimaal, dus
\begin{itemize}
\item als $g_1 < g_2$ gaat alle materie naar fase 1 (de stabiele
      fase is die met de \emph{lagere} $g$);
\item de twee fasen bestaan alleen naast elkaar waar
      \[
      g_1(T,P) = g_2(T,P) ,
      \]
      een betrekking tussen $T$ en $P$: de coëxistentiekromme van het
      fasediagram.
\end{itemize}
Langs die kromme geldt (Clapeyron)
\[
\frac{\dd P}{\dd T} = \frac{s_2 - s_1}{v_2 - v_1} = \frac{L_{1\to2}}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
$\dd G = (g_1 - g_2)\dd m_1$ bij vaste $T$, $P$ (met $m_1 + m_2$ vast):
$G$ daalt door massa naar de fase met de lagere $g$ te verplaatsen, tot die
fase alleen overblijft of tot $g_1 = g_2$. Differentieer $g_1 = g_2$ langs de
kromme met $\dd g = -s\,\dd T + v\,\dd P$ voor elke fase:
$-s_1\dd T +
v_1\dd P = -s_2\dd T + v_2\dd P$; en $s_2 - s_1 = L/T$ omdat de
overgang
bij vaste $T$, $P$ omkeerbaar is. Het volume van jaar 1 verkreeg deze formule
uit een carnotkringloop; hier valt zij uit $G$.
\end{proof}

\begin{example}[Water: hellingen en metastabiliteit]\label{ex:b2:thermodynamic-potentials:water}
Smelten: $L = \qty{334}{kJ/kg}$,
$v_{\text{vl}} - v_{\text{ijs}} = -\qty{9.1e-5}{m^3/kg}$:
$\dd P/\dd T = \qty{-1.3e7}{Pa/K}$, \emph{negatief} --- ijs smelt onder
druk, maar \qty{130}{bar} per kelvin: de \qty{70}{bar} van een schaatser verlagen
het smeltpunt maar met een halve graad (schaatsen werkt door wrijving en
een voorgesmolten oppervlaktelaagje), terwijl de \qty{270}{bar} onder drie
kilometer ijskap haar bodem smelten en haar laten glijden. Verdampen bij
\qty{100}{\celsius}: $L = \qty{2.26}{MJ/kg}$, $\Delta v = \qty{1.67}{m^3/kg}$:
\qty{3.6}{kPa/K} --- \qty{92}{\celsius} bij \qty{0.7}{bar} op een berg,
\qty{120}{\celsius} in een snelkookpan bij \qty{2}{bar}. Onder
\qty{0}{\celsius} heeft vloeibaar water
$g_{\text{vl}} - g_{\text{ijs}} \approx
(L/T_{\text{smelt}})(T_{\text{smelt}} - T) = \qty{1.2}{kJ/kg}$
per kelvin
onderkoeling: het is metastabiel, niet onstabiel, en zuivere druppeltjes in
wolken blijven bij gebrek aan een kiem vloeibaar tot \qty{-40}{\celsius}.
\end{example}

\begin{remark}[Van der Waals en de constructie van Maxwell]\label{rem:b2:thermodynamic-potentials:maxwellconstruction}
De vergelijking van Van der Waals $(P + a/v^2)(v - b) = RT$ geeft, onder haar
kritieke temperatuur $T_{\text{c}} = 8a/27Rb$, isothermen met een lus. Het
werkelijke fluïdum volgt de lus niet: bij $P_{\text{verz}}(T)$ springt het van
de vloeistof- naar de damptak, en $P_{\text{verz}}$ wordt vastgelegd door
$g_{\text{vl}}
= g_{\text{damp}}$, dat wil zeggen
$\int_{\text{vl}}^{\text{damp}}v\,\dd P = 0$ langs de
lus: de horizontale lijn snijdt \emph{gelijke oppervlakken} af. De delen van de
lus waar $(\partial P/\partial v)_T > 0$ zijn mechanisch onstabiel; de
delen tussen die en de lijn zijn metastabiel --- de oververhitte
vloeistof die in een magnetron ``opspat'', de oververzadigde damp van de
nevelkamer.
\end{remark}

\begin{method}[De potentialen gebruiken]\label{meth:b2:thermodynamic-potentials:method}
(1) Stel de randvoorwaarden vast: geïsoleerd $\to$ $S$ maximaal; vaste $T,V$ $\to$
$F$ minimaal; vaste $T,P$ $\to$ $G$ minimaal. (2) Maximale arbeid:
$-\Delta F$ bij vaste
$T$, $-\Delta G$ aan \omterm{def:b2:open-systems:cv}{nuttige arbeid} bij vaste $T,P$. (3) Een afgeleide van
$S$ bij vaste $T$ nodig? Gebruik een betrekking van Maxwell en de
toestandsvergelijking.
(4) Twee fasen: vergelijk $g$; naast elkaar bestaan bij $g_1 = g_2$;
Clapeyron voor de
helling; teken van $\Delta v$ voor het teken van de helling. (5) \omterm{ex:b2:thermodynamic-potentials:perfect}{Chemische
potentiaal} $\mu = g$ per mol: materie stroomt naar lagere $\mu$.
\end{method}

\section{Opgaven}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:1}
Schrijf voor $n$ mol volmaakt gas $F(T,V)$ en $G(T,P)$ op een
functie van $T$ alleen na; vind uit elk de toestandsvergelijking terug; toon
aan dat $S = -\partial_TF$ de bekende entropie geeft.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:2}
Eén mol gas, \qty{300}{K}, van \qty{1}{L} tot \qty{10}{L} in een
thermostaat: $\Delta F$, $\Delta U$, $\Delta S$; teruggewonnen arbeid indien
omkeerbaar;
indien tegen \qty{1}{bar}; indien in vacuüm; de entropie die in elk
geval wordt gemaakt.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:3}
Water en ijs bij \qty{1}{bar}: toon aan dat
$g_{\text{vl}} - g_{\text{ijs}} \approx
(L/T_{\text{smelt}})(T_{\text{smelt}} - T)$
nabij \qty{0}{\celsius} en bereken het
bij \qty{-5}{} en $+\qty{5}{\celsius}$ ($L = \qty{334}{kJ/kg}$). Welke fase is
in elk geval stabiel; is de andere onmogelijk?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:4}
Hellingen van Clapeyron voor water: smelten ($\Delta v = -\qty{9.1e-5}{m^3/kg}$) en
verdampen bij \qty{100}{\celsius} ($L = \qty{2.26}{MJ/kg}$,
$\Delta v =
\qty{1.67}{m^3/kg}$). Smeltpunt onder \qty{70}{bar}; kookpunt
bij
\qty{0.7}{bar} en bij \qty{2}{bar}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:5}
\emph{Maxwell.} (a) Leid de twee betrekkingen van Maxwell van het hoofdstuk
af en de twee uit $\dd U$ en $\dd H$. (b) Toon aan dat
$(\partial U/\partial V)_T =
T(\partial P/\partial T)_V - P$; bereken haar
voor een volmaakt gas en een
vanderwaalsgas. (c) Vloeibaar water: $\alpha = \qty{2.1e-4}{K^{-1}}$,
$\kappa_T =
\qty{4.6e-10}{Pa^{-1}}$: drukstijging per kelvin in een star,
vol vat,
en $(\partial U/\partial V)_T$. (d) Een volle, afgesloten glazen fles wordt
\qty{10}{K} verwarmd: bespreek.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:6}
\emph{$C_P - C_V$.} (a) Leid $C_P - C_V = TV\alpha^2/\kappa_T$ af. (b)
Volmaakt gas. (c) Water bij \qty{300}{K}; koper ($\alpha = \qty{5e-5}{K^{-1}}$,
$\kappa_T = \qty{7e-12}{Pa^{-1}}$, $\rho = \qty{8900}{kg/m^3}$); vergelijk met
$c_P$ (\qty{4180}{} en \qty{385}{J/kg/K}). (d) Waarom kan men van
``de'' warmtecapaciteit van een vaste stof spreken?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:7}
\emph{Rubber.} Een band voldoet aan $f = CT(L - L_0)$ met $C = \qty{1}{N/m/K}$.
(a) $(\partial S/\partial L)_T$ en $(\partial U/\partial L)_T$. (b)
Entropie\-verandering,
warmte en arbeid bij isotherm uitrekken op \qty{300}{K} van
$L_0$ tot $L_0 + \qty{0.1}{m}$. (c) In plaats daarvan adiabatisch uitgerekt
(warmte\-capaciteit
\qty{4}{J/K}): temperatuurverandering. (d) Met een vast gewicht opgehangen
en \qty{30}{K} verwarmd: betrekkelijke verandering van $L - L_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:8}
\emph{Verdamping van een plas.} De \omterm{ex:b2:thermodynamic-potentials:perfect}{chemische potentiaal} van waterdamp
bij partiële druk $p$ is
$\mu_{\text{v}} = \mu_{\text{v}}^\circ(T) +
RT\ln(p/p^\circ)$, en is gelijk
aan die van de vloeistof wanneer $p = p_{\text{verz}}(T)$.
(a) Toon aan dat vloeistof verdampt wanneer $p < p_{\text{verz}}$ en damp
condenseert wanneer $p > p_{\text{verz}}$; definieer de betrekkelijke
vochtigheid. (b)
$\mu_{\text{vl}} - \mu_{\text{v}}$ bij $50\%$ vochtigheid, \qty{293}{K}. (c) Waarom
droogt was zelfs bij \qty{20}{\celsius}, en nooit bij $100\%$ vochtigheid?
(d) Dauw: op welke oppervlakken vormt hij zich eerst, en waarom?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:9}
\emph{Van der Waals.} $(P + a/v^2)(v - b) = RT$ (molair). (a) Vind het
kritieke punt ($\partial_vP = \partial_v^2P = 0$): $v_{\text{c}} = 3b$,
$T_{\text{c}} = 8a/27Rb$, $P_{\text{c}} = a/27b^2$. (b) Koolstofdioxide,
$T_{\text{c}} = \qty{304}{K}$, $P_{\text{c}} = \qty{73.8}{bar}$: $a$ en $b$. (c)
Verantwoord de regel van de gelijke oppervlakken van Maxwell uit $g$. (d)
Welke delen van de lus
zijn onstabiel, welke metastabiel, en welke verschijnselen tonen de metastabiele?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:10}
\emph{Clausius--Clapeyron geïntegreerd.} Verwaarloos voor verdamping
$v_{\text{vl}}$
en behandel de damp als volmaakt. (a) Toon aan dat
$\dd\ln P/\dd T = L_{\text{m}}/RT^2$
($L_{\text{m}}$ molair) en integreer voor constante $L_{\text{m}}$. (b) Water,
$L_{\text{m}} = \qty{40.7}{kJ/mol}$: $P_{\text{verz}}$ bij \qty{20}{\celsius}
vanuit de
\qty{1}{bar} bij \qty{100}{\celsius}; vergelijk met de gemeten \qty{23}{mbar}.
(c) Voorspel de druk van het tripelpunt (\qty{0.01}{\celsius}) en vergelijk
met \qty{611}{Pa}. (d) Toon aan dat dezelfde wet volgt uit Gibbs--Helmholtz
toegepast op $\Delta g$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:11}
\emph{Osmose.} De \omterm{ex:b2:thermodynamic-potentials:perfect}{chemische potentiaal} van een oplosmiddel wordt door een verdunde
opgeloste stof verlaagd: $\mu = \mu^\circ - RTx_{\text{s}}$ ($x_{\text{s}}$
de molfractie
van de opgeloste stof). Een membraan laat alleen het oplosmiddel door;
zuiver oplosmiddel aan
de ene kant, oplossing aan de andere, dezelfde $T$. (a) Toon aan dat evenwicht
een extra druk $\Pi$ op de oplossing vergt met
$\Pi v_{\text{m}} =
RTx_{\text{s}}$, dat wil zeggen $\Pi = cRT$ (Van 't
Hoff). (b) Zeewater, $c \approx
\qty{1.1}{mol/L}$ opgeloste deeltjes,
\qty{293}{K}: $\Pi$. (c) Kleinste
arbeid om een kubieke meter zoet water te winnen; vergelijk met werkelijke
installaties
(\qty{3}{kWh/m^3}). (d) Waarom barsten plantencellen in zuiver water en
verschrompelen zij in pekel?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:12}
\emph{De keten.} Een polymeerketen van $N$ schakels met lengte $a$: het
aantal schikkingen met een vector van uiteinde tot uiteinde van lengte $L$ langs
de band is $\Omega \propto \eu^{-L^2/2Na^2}$ (de toevalswandeling). (a) $S(L)$
en, bij vaste $U$, de spanning $f = -T\,\partial S/\partial L$. (b) De
stijfheid van de keten en haar temperatuurafhankelijkheid. (c) $N = 1000$,
$a =
\qty{0.5}{nm}$, \qty{300}{K}, $L = \qty{10}{nm}$: $f$; hoeveel ketens
in
parallel geven \qty{30}{N}? (d) Schat de elasticiteitsmodulus van een rubber
met $10^{26}$ ketens per kubieke meter, elk met $N = 100$.
\end{exercise}

\section{Vraagstuk: het elastiekje en het fasediagram van water}

\begin{problem}[{Weekendvraagstuk --- entropie die trekt, en fasen
die met elkaar wedijveren}]\label{pb:b2:thermodynamic-potentials:1}
\textbf{Deel I --- Het elastiekje.}
Een band met massa \qty{2}{g} (warmtecapaciteit \qty{4}{J/K}) heeft de spanning
$f = CT(L - L_0)$ met $C = \qty{1}{N/m/K}$; bij \qty{300}{K} en
$L - L_0 =
\qty{0.1}{m}$ is $f = \qty{30}{N}$.
\begin{enumerate}
  \item Schrijf $\dd U$ en $\dd F$ voor de band ($\dd U = T\dd S + f\dd L$).
  \item Leid de betrekking van Maxwell
        $(\partial S/\partial L)_T = -(\partial
        f/\partial T)_L$ af.
  \item Toon aan dat $(\partial U/\partial L)_T = 0$ voor deze band: wat
        betekent dat?
  \item Isotherm uitrekken van $L_0$ tot $L_0 + \qty{0.1}{m}$ bij
        \qty{300}{K}: $\Delta S$, warmte $Q$, arbeid $W$; ga na dat $\Delta U = 0$.
  \item Hetzelfde uitrekken adiabatisch gedaan: temperatuurverandering.
  \item De band houdt een vast gewicht; hij wordt \qty{30}{K} verwarmd: nieuwe
        $L - L_0$. Beschrijf de ``rubbermotor'' (een wiel met rubberen
        spaken dat aan één kant wordt verwarmd).
  \item Een keten van $N$ schakels met lengte $a$ heeft
        $\Omega \propto \eu^{-L^2/2Na^2}$:
        $S(L)$ en $f(L,T)$; waarom groeit de stijfheid met $T$?
  \item $N = 1000$, $a = \qty{0.5}{nm}$, $L = \qty{10}{nm}$, \qty{300}{K}: $f$
        voor één keten en het aantal ketens voor \qty{30}{N}.
  \item Waarom gedraagt een stalen veer zich omgekeerd (spanning die bij
        vaste lengte met $T$ daalt)?
\end{enumerate}

\textbf{Deel II --- Het fasediagram van water.}
Gegevens: $L_{\text{smelt}} = \qty{334}{kJ/kg}$,
$v_{\text{ijs}} = \qty{1.091e-3}{m^3/kg}$,
$v_{\text{vl}} = \qty{1.000e-3}{m^3/kg}$; $L_{\text{damp}} = \qty{2.26}{MJ/kg}$ bij
\qty{100}{\celsius}, $v_{\text{damp}} = \qty{1.67}{m^3/kg}$;
$L_{\text{m}} = \qty{40.7}{kJ/mol}$;
tripelpunt \qty{0.01}{\celsius}, \qty{611}{Pa}; kritiek punt \qty{647}{K},
\qty{221}{bar}.
\begin{enumerate}[resume]
  \item Schets $g(T)$ voor ijs, vloeistof en damp bij \qty{1}{bar}; verantwoord
        de volgorde van de hellingen en wijs de overgangen aan.
  \item Leid de formule van Clapeyron af uit $g_1 = g_2$.
  \item Helling van de smeltkromme; haar teken, en waarom dat ongewoon is.
  \item Een schaatser: \qty{700}{N} op \qty{1}{cm^2}: verandering van het
        smeltpunt; besluit over de volkse verklaring van het schaatsen.
  \item Onder \qty{3}{km} ijs ($\rho = \qty{917}{kg/m^3}$): druk en
        smeltpunt; gevolg voor de ijskap.
  \item Helling van de verdampingskromme bij \qty{100}{\celsius}; kookpunt
        bij \qty{0.7}{bar} en bij \qty{2}{bar}.
  \item Integreer Clausius--Clapeyron met een volmaakte damp en
        constante $L_{\text{m}}$; $P_{\text{verz}}$ bij \qty{20}{\celsius} en bij
        het tripelpunt; bespreek de nauwkeurigheid.
  \item Vergelijk in het tripelpunt de hellingen van de sublimatie- en de
        verdampingskromme ($L_{\text{sub}} = L_{\text{smelt}} + L_{\text{damp}}$).
  \item Onderkoeld water op \qty{-10}{\celsius}: $g_{\text{vl}} - g_{\text{ijs}}$;
        waarom kunnen wolkendruppeltjes tot \qty{-40}{\celsius} vloeibaar blijven?
  \item Wat gebeurt er met $\Delta v$, $L$ en de helling in het kritieke
        punt; wat is een superkritisch fluïdum?
\end{enumerate}

\textbf{Deel III --- Toepassingen.}
\begin{enumerate}[resume]
  \item Verklaar de constructie van de gelijke oppervlakken van Maxwell op een
        vanderwaalsisotherm uit de continuïteit van $g$.
  \item Oververhit water in een magnetron ``spat op'': welke tak van de
        isotherm, en waarom het gevaar?
  \item Vriesdrogen: beschrijf de weg in het $(P,T)$-diagram en
        waarom het water vertrekt zonder te smelten.
  \item Waarom verdampt een schaal water bij \qty{20}{\celsius} in droge lucht
        en houdt zij op in verzadigde lucht --- in termen van chemische potentialen?
  \item Een snelkookpan bij \qty{2}{bar}: waarom gaart voedsel zoveel
        sneller (de snelheid van de scheikunde verdubbelt ruwweg om de
        \qty{10}{K})?
  \item Vat samen: welke potentiaal bij welke randvoorwaarde, en wat de
        betrekkingen van Maxwell opleveren.
\end{enumerate}
\end{problem}
