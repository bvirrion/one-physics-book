\chapter{Warmtegeleiding}\label{ch:b2:heat-conduction}

Raak op dezelfde koude ochtend een metalen rail en een houten bank aan:
de rail voelt kouder, hoewel een thermometer voor beide dezelfde
temperatuur geeft. Zet een pan op het fornuis en het handvat wordt pas
na een tijdje warm; graaf in de zomer twee meter diep en de grond is
nog koud van de winter; een processor die honderd watt op een vierkante
centimeter afvoert smelt niet omdat er een blok geribd aluminium op
staat. Dat alles is \emph{warmtegeleiding}, het transport van inwendige
energie door de materie heen door de beweging van haar moleculen,
zonder enige stroming van materie --- en dat alles gehoorzaamt aan één
wet, die van Fourier, en één vergelijking, de warmtevergelijking, die
de \omterm{thm:b2:particle-diffusion:equation}{diffusievergelijking} van het vorige hoofdstuk is met de temperatuur
in plaats van de dichtheid. Dit hoofdstuk stelt de wet en de
vergelijking op uit een energiebalans, lost ze op in de gevallen die
ertoe doen (stationaire muren en hun \omterm{prop:b2:heat-conduction:resistance}{warmteweerstanden}, periodieke
verwarming en de warmtegolf, koelribben), voert de uitwisseling met een
fluïdum in (de wet van Newton en het \omterm{def:b2:heat-conduction:biot}{biotgetal}), en berekent de
entropie die geleiding maakt --- want warmte stroomt alleen bergaf.

\section{De wet van Fourier}

\begin{definition}[Warmtestroomdichtheid]\label{def:b2:heat-conduction:flux}
De \emph{warmtestroomdichtheid}\index{warmtestroomdichtheid}
$\vect{j}_Q$ (in \unit{W/m^2}) is de vector zodanig dat de energie
die door geleiding door een georiënteerd oppervlakte-element
$\dd\vect S$ gedurende $\dd t$ wordt overgedragen
$\vect{j}_Q\cdot\dd\vect S\,\dd t$ is; de
\emph{warmteflux}\index{warmteflux} (een vermogen, in watt) door een
oppervlak $S$ is $\Phi = \iint_S\vect{j}_Q\cdot\dd\vect S$.
\end{definition}

\begin{theorem}[Wet van Fourier]\label{thm:b2:heat-conduction:fourier}
In een midden in rust waarvan de temperatuur niet gelijkmatig is, geldt
\[
\vect{j}_Q = -\lambda\,\vect{\operatorname{grad}}\,T ,
\]
waarin de \emph{warmtegeleidingscoëfficiënt}\index{warmtegeleidingscoëfficiënt}
$\lambda > 0$ (in \unit{W/m/K}) van het materiaal afhangt en, zwak, van
de temperatuur. Warmte stroomt de temperatuurgradiënt af, van warm naar
koud --- de tweede hoofdwet ingebouwd in een lineaire wet.
\end{theorem}

\begin{proof}
Fenomenologisch, zoals de wet van Fick, en met dezelfde microscopische
verantwoording: de moleculen (of de vrije elektronen van een metaal, of de
roostertrillingen) dragen hun energie op een toevalswandeling mee, en er
komt meer van de warme kant.
\end{proof}

\begin{example}[Geleidingscoëfficiënten]\label{ex:b2:heat-conduction:values}
Metalen (elektronen dragen zowel de warmte als de stroom ---
de geleidingen volgen elkaar): koper \qty{400}{W/m/K}, aluminium
\qty{240}{W/m/K}, staal \qty{50}{W/m/K}. Isolerende vaste stoffen: beton
\qty{1.5}{W/m/K}, glas \qty{1}{W/m/K}, baksteen \qty{0.8}{W/m/K}, hout
\qty{0.15}{W/m/K}. Vloeistoffen: water \qty{0.6}{W/m/K}. Gassen: lucht
\qty{0.026}{W/m/K} --- en de beste isolatoren, glaswol of schuim bij
\qty{0.04}{W/m/K}, zijn grotendeels stilstaande lucht, zo ingesloten dat zij niet
kan stromen. Vier ordes van grootte van koper tot lucht.
\end{example}

\section{De energiebalans en de warmtevergelijking}

\begin{theorem}[Plaatselijke energiebalans]\label{thm:b2:heat-conduction:balance}
In een vaste stof (of een fluïdum in rust) met dichtheid $\rho$ en
soortelijke warmte
$c$, met een vermogen $p$ dat per volume-eenheid vrijkomt (joulewarmte, een
scheikundige of kernreactie, opgenomen straling), geldt
\[
\rho c\,\frac{\partial T}{\partial t} = -\operatorname{div}\vect{j}_Q + p ,
\qquad\text{in één dimensie}\quad
\rho c\,\frac{\partial T}{\partial t} = -\frac{\partial j_Q}{\partial x} + p .
\]
\end{theorem}

\begin{proof}
Eerste hoofdwet voor de plak tussen $x$ en $x + \dd x$ (doorsnede $S$,
vast volume, dus geen arbeid): haar inwendige energie $\rho c\,T\,S\,\dd x$
verandert in $\dd t$ met de ontvangen warmte,
$[j_Q(x) - j_Q(x+\dd x)]S\,\dd t
= -\partial_x j_Q\,\dd x\,S\,\dd t$, plus
$p\,S\,\dd x\,\dd t$. In drie
dimensies is de warmte die door een gesloten oppervlak een vast volume binnenkomt
$-\iint\vect{j}_Q\cdot\dd\vect S = -\iiint\operatorname{div}\vect{j}_Q\,\dd\tau$.
\end{proof}

\begin{theorem}[Warmtevergelijking]\label{thm:b2:heat-conduction:equation}
Voor gelijkmatige $\lambda$, $\rho$, $c$ geldt
\[
\frac{\partial T}{\partial t} = a\,\Delta T + \frac{p}{\rho c},
\qquad a = \frac{\lambda}{\rho c} \quad\text{(de \emph{thermische diffusiviteit}\index{thermische diffusiviteit}, in \unit{m^2/s})} .
\]
Alles wat over de \omterm{thm:b2:particle-diffusion:equation}{diffusievergelijking} is gezegd, geldt: lineariteit,
onomkeerbaarheid, gladstrijken, en de schalen $L \sim \sqrt{at}$, $t \sim
L^2/a$.
\end{theorem}

\begin{proof}
Vul de wet van Fourier in de balans in:
$\rho c\,\partial_t T =
\lambda\Delta T + p$.
\end{proof}

\begin{example}[Diffusiviteiten en tijden]\label{ex:b2:heat-conduction:times}
Koper $a = \qty{1.1e-4}{m^2/s}$, lucht \qty{2e-5}{m^2/s}, beton en
baksteen $\approx \qty{5e-7}{m^2/s}$, water \qty{1.4e-7}{m^2/s}, hout
\qty{1e-7}{m^2/s}. Warmte doorloopt een centimeter koper in een seconde, een
centimeter baksteen in drie minuten, een muur van \qty{20}{cm} in een dag
(en daarom vlakken dikke muren de dag-nachtcyclus af), de
halve dikte van \qty{1.5}{cm} van een biefstuk in een kwartier (baktijden
schalen als het kwadraat van de dikte), en een kilometer rots
in $10^{12}\,\unit{s}$, dertigduizend jaar --- de aarde koelt nog altijd
af sinds haar vorming.
\end{example}

\section{Stationair regime: warmteweerstanden}

\begin{proposition}[Warmteweerstand van een plaat]\label{prop:b2:heat-conduction:resistance}
In het stationaire regime zonder bronnen heeft een plaat met dikte $e$,
oppervlak $S$ en geleidingscoëfficiënt $\lambda$, tussen de temperaturen $T_1$ en
$T_2$, een lineair profiel en draagt zij de flux
\[
\Phi = \frac{T_1 - T_2}{R_{\text{th}}}, \qquad
R_{\text{th}} = \frac{e}{\lambda S}\quad\text{(in \unit{K/W})} :
\]
haar \emph{warmteweerstand}\index{warmteweerstand}. De temperatuur
speelt de rol van potentiaal, de flux die van stroom: weerstanden in
serie (lagen van een muur) tellen op, weerstanden in parallel (muur en
raam) tellen hun omgekeerden op. Voor een cilinderschil (de isolatie
van een buis) tussen de stralen $r_1 < r_2$ en met lengte $\ell$ is
$R_{\text{th}} = \ln(r_2/r_1)/2\pi\lambda\ell$; voor een bolschil,
$(1/r_1 - 1/r_2)/4\pi\lambda$.
\end{proposition}

\begin{proof}
$\dd^2T/\dd x^2 = 0$: affien profiel, $j_Q = \lambda(T_1 - T_2)/e$
gelijkmatig, $\Phi = j_QS$. In cilindermeetkunde is
$\Phi = -\lambda\,2\pi
r\ell\,\dd T/\dd r$ bij elke $r$ hetzelfde (geen
opeenhoping), zodat
$T = A - (\Phi/2\pi\lambda\ell)\ln r$; in bolmeetkunde geeft
$\Phi =
-4\pi r^2\lambda\,\dd T/\dd r$ dat $T = A + \Phi/4\pi\lambda r$.
\end{proof}

\begin{proposition}[Uitwisseling met een fluïdum: de wet van Newton]\label{prop:b2:heat-conduction:newton}
Een vast oppervlak op $T_{\text{s}}$ in aanraking met een fluïdum op $T_{\text{f}}$
(weg van het oppervlak) verliest de fluxdichtheid
\[
j_Q = h\,(T_{\text{s}} - T_{\text{f}}) ,
\]
waarin de \emph{warmteoverdrachtscoëfficiënt}\index{warmteoverdrachts
coëfficiënt} $h$ (in \unit{W/m^2/K}) de geleiding door
het dunne fluïdumlaagje dat aan de wand kleeft en de convectie die het
vernieuwt samenvat: $h \approx 5$--\qty{25}{W/m^2/K} voor lucht bij natuurlijke of
lichte convectie, $50$--\qty{500}{W/m^2/K} voor gedwongen lucht, $10^3$--$10^4$
voor water. Het oppervlak voegt dan een \emph{filmweerstand} $1/hS$ in
serie met de muur toe.
\end{proposition}

\begin{proof}
Fenomenologisch: de beweging van het fluïdum wordt niet berekend, zij wordt
samengevat
in $h$ (gemeten, of gegeven door de correlaties van de vloeistofmechanica).
\end{proof}

\begin{definition}[Biotgetal]\label{def:b2:heat-conduction:biot}
Voor een vaste stof van grootte $L$ die met een fluïdum uitwisselt, vergelijkt het
\emph{biotgetal}\index{biotgetal}
\[
\mathrm{Bi} = \frac{hL}{\lambda} = \frac{\text{inwendige weerstand } L/\lambda S}{\text{filmweerstand } 1/hS}
\]
die twee. Is $\mathrm{Bi} \ll 1$, dan is de vaste stof op elk ogenblik
gelijkmatig van temperatuur en koelt zij als geheel af, exponentieel met
de tijdconstante $\tau = \rho c V/hS$; is $\mathrm{Bi} \gg 1$, dan staat het
oppervlak op de temperatuur van het fluïdum en geleidt het inwendige in zijn
eigen tempo $L^2/a$.
\end{definition}

\section{Periodieke verwarming: de warmtegolf}

\begin{proposition}[Warmtegolf in een halfruimte]\label{prop:b2:heat-conduction:wave}
Als het oppervlak $x = 0$ van een halfruimte op
$T(0,t) = T_0 +
\theta_0\cos\omega t$ wordt gehouden, is de temperatuur
binnenin, in het
ingestelde regime,
\[
T(x,t) = T_0 + \theta_0\,\eu^{-x/\delta}\cos\Big(\omega t - \frac{x}{\delta}\Big),
\qquad \delta = \sqrt{\frac{2a}{\omega}} :
\]
een golf die met de snelheid $\omega\delta =
\sqrt{2a\omega}$ naar binnen
loopt en per radiaal fase over één
\emph{\omterm{def:b2:dispersion-wave-packets:complex}{indringdiepte}} $\delta$ wordt gedempt --- met
$\eu^{-2\pi} = 2 \times 10^{-3}$ per
golflengte. Snelle trillingen dringen minder ver door dan trage.
\end{proposition}

\begin{proof}
Zoek $\underline T = \theta_0\eu^{\iu(\omega t - \underline k x)}$:
$\iu\omega = -a\underline k^2$, dus $\underline k^2 = -\iu\omega/a$,
$\underline k = (1 - \iu)\sqrt{\omega/2a} = (1 - \iu)/\delta$ (de wortel
die voor $x > 0$ uitdooft). Neem het reële deel.
\end{proof}

\section{Ribben}

\begin{proposition}[De koelrib]\label{prop:b2:heat-conduction:fin}
Een dunne staaf of plaat (doorsnede $S$, omtrek $p$, geleidingscoëfficiënt
$\lambda$)
die bij $x = 0$ aan een muur op $T_0$ vastzit, in een fluïdum op $T_{\text{f}}$ met
de coëfficiënt $h$, heeft een overtemperatuur $\theta = T - T_{\text{f}}$
die voldoet aan
\[
\frac{\dd^2\theta}{\dd x^2} = m^2\theta, \qquad m = \sqrt{\frac{hp}{\lambda S}} ;
\]
voor een lange rib is $\theta = \theta_0\eu^{-mx}$ en voert de rib
$\Phi =
\sqrt{hp\lambda S}\,\theta_0$ af --- evenveel als een kaal oppervlak
van $\sqrt{\lambda
S/hp} = 1/m$ lang, met een temperatuur die tot
$\theta_0/\eu$ is gedaald bij $x = 1/m$. Voor een rib van eindige lengte $L$
(adiabatische tip) is $\theta =
\theta_0\cosh m(L-x)/\cosh mL$ en is haar
\emph{rendement}\index{rib
rendement} --- de flux gedeeld door die van een isotherme rib op $\theta_0$ ---
gelijk aan $\eta = \tanh(mL)/mL$.
\end{proposition}

\begin{proof}
Stationaire balans van de schijf $[x, x + \dd x]$: geleid naar binnen,
$-\lambda S\theta'(x)$; naar buiten, $-\lambda S\theta'(x + \dd x)$;
verloren aan het
fluïdum, $hp\,\theta\,\dd x$: $\lambda S\theta'' = hp\theta$. De flux aan de
voet is $-\lambda S\theta'(0)$: $\lambda Sm\theta_0$ voor de lange rib,
$\lambda Sm\theta_0\tanh mL$ voor de eindige, te vergelijken met
$hpL\theta_0$. Het \omterm{def:b2:heat-conduction:biot}{biotgetal} over de dikte van de rib, $ht/\lambda$, moet
klein zijn voor de eendimensionale behandeling.
\end{proof}

\qty{30}{mm} hoog, in gedwongen lucht $h = \qty{50}{W/m^2/K}$:
$m =
\sqrt{2h/\lambda t} = \qty{20}{m^{-1}}$, $mL = 0.6$, $\eta = 0.89$ ---
de
ribben werken op $89\%$ van hun oppervlak. Dertig ervan, \qty{60}{mm} breed,
bieden \qty{0.11}{m^2} waar het kale vierkant van \qty{60}{mm}
\qty{0.0036}{m^2} bood: een weerstand $1/\eta hA = \qty{0.2}{K/W}$ in plaats van
\qty{5.6}{K/W}; \qty{100}{W} verhogen de voet met \qty{20}{K} in plaats van
\qty{560}{K}. Langere ribben winnen weinig ($\tanh$ verzadigt), dunnere
verliezen rendement: $mL \approx 1$ is het compromis van de ingenieur.
\end{example}

\section{Entropie die door geleiding wordt gemaakt}

\begin{proposition}[Entropieproductie]\label{prop:b2:heat-conduction:entropy}
Een flux $\Phi$ die van een lichaam op $T_{\text{h}}$ naar een lichaam op
$T_{\text{c}} < T_{\text{h}}$ door een stationaire muur wordt geleid, maakt
entropie met
het tempo
\[
\dot S_{\text{c}} = \Phi\Big(\frac{1}{T_{\text{c}}} - \frac{1}{T_{\text{h}}}\Big) > 0 ;
\]
plaatselijk maakt geleiding
$\sigma_s = \lambda\,(\vect{\operatorname{grad}}\,T)^2/T^2
\ge 0$ per
volume-eenheid en per tijdseenheid --- nul alleen waar de temperatuur
gelijkmatig is. Warmtegeleiding is onomkeerbaar: dezelfde energie, geleverd bij
een lagere temperatuur, kan minder arbeid verrichten.
\end{proposition}

\begin{proof}
De toestand van de muur verandert niet, dus haar entropie is constant: het warme
lichaam verliest $\Phi/T_{\text{h}}$ per tijdseenheid, het koude wint
$\Phi/T_{\text{c}}$; het verschil wordt gemaakt. Plaatselijk geeft de entropiebalans
$\partial_t(\rho s) + \operatorname{div}(\vect{j}_Q/T) = \sigma_s$
met $\rho T\partial_t s = -\operatorname{div}\vect{j}_Q$ dat
$\sigma_s =
\vect{j}_Q\cdot\vect{\operatorname{grad}}(1/T) = -\vect{j}_Q\cdot
\vect{\operatorname{grad}}\,T/T^2 = \lambda(\vect{\operatorname{grad}}\,T)^2/T^2$.
\end{proof}

\begin{method}[Geleidingsschattingen]\label{meth:b2:heat-conduction:method}
(1) Stationair: bouw de warmtekring --- platen $e/\lambda S$, films
$1/hS$, schillen $\ln(r_2/r_1)/2\pi\lambda\ell$; serie en parallel.
(2) Overgangsverschijnsel: $\mathrm{Bi} = hL/\lambda$; is die klein, dan
$\tau = \rho cV/hS$; is hij
groot, dan $t \sim L^2/a$. (3) Periodiek: $\delta = \sqrt{2a/\omega}$, demping
$\eu^{-x/\delta}$, achterstand $x/\delta$. (4) Bronnen: $p$ per
volume-eenheid, parabolische
profielen. (5) Ribben: $m = \sqrt{hp/\lambda S}$, $\eta = \tanh(mL)/mL$. (6)
Entropie: $\Phi(1/T_{\text{c}} - 1/T_{\text{h}})$.
\end{method}

\section{Vraagstuk: een huis isoleren, een processor koelen}

\begin{problem}[{Weekendvraagstuk --- geleiding op de schaal van een
gebouw en van een chip}]\label{pb:b2:heat-conduction:1}
Gegevens: baksteen $\lambda = \qty{0.8}{W/m/K}$,
$\rho = \qty{1800}{kg/m^3}$, $c =
\qty{840}{J/kg/K}$; glaswol
\qty{0.04}{W/m/K}; glas \qty{1}{W/m/K}; stilstaande
lucht \qty{0.026}{W/m/K}; aluminium \qty{240}{W/m/K}, $\rho = \qty{2700}{kg/m^3}$,
$c = \qty{900}{J/kg/K}$; silicium \qty{150}{W/m/K}; warmtegeleidende pasta
\qty{5}{W/m/K}; bodem $a = \qty{5e-7}{m^2/s}$, $\lambda = \qty{1}{W/m/K}$;
films $h_{\text{i}} = 8$, $h_{\text{o}} = \qty{25}{W/m^2/K}$.

\textbf{Deel I --- Het huis.}
Een huis heeft \qty{100}{m^2} muren (baksteen \qty{20}{cm}) en \qty{20}{m^2}
enkel glas (\qty{4}{mm}); binnen \qty{20}{\celsius},
buiten \qty{0}{\celsius}.
\begin{enumerate}
  \item Weerstand per vierkante meter van de baksteen alleen; fluxdichtheid
        als haar vlakken op de twee temperaturen stonden.
  \item Met de twee films: $R$ en $U = 1/R$ per vierkante meter; idem
        met \qty{10}{cm} glaswol erbij.
  \item $U$ van het enkele glas; van een dubbel glas met \qty{12}{mm}
        stilstaande lucht.
  \item Totaal verlies van het huis vóór en na het isoleren van de muren
        en het verdubbelen van de ramen.
  \item Over een stookseizoen van $2500$ graaddagen (de integraal van
        het temperatuurverschil over het seizoen, in \unit{K.dag}),
        verloren energie vóór en na, in kilowattuur.
  \item Temperatuur van het binnenoppervlak van de muur, kaal en geïsoleerd;
        waarom voelt een koude muur oncomfortabel en beschimmelt hij?
  \item Diffusietijd door de baksteen; wat doet die met de
        dag-nachtcyclus?
\end{enumerate}

\textbf{Deel II --- De grond onder het huis.}
\begin{enumerate}[resume]
  \item Leid de \omterm{def:b2:dispersion-wave-packets:complex}{indringdiepte} $\delta = \sqrt{2a/\omega}$ van een
        warmtegolf af.
  \item Dagelijkse en jaarlijkse $\delta$ in de bodem.
  \item Diepte waarop de dagelijkse schommeling tot $1\%$ is gedaald; jaarlijkse
        amplitude op \qty{2.2}{m} en op \qty{5}{m} voor een oppervlakteschommeling
        van \qty{10}{K}.
  \item Fase-achterstand op \qty{2.2}{m}, in dagen: wanneer is de kelder het koudst?
  \item \omterm{def:b2:heat-conduction:flux}{Warmteflux} aan het oppervlak van de jaargolf (amplitude en fase);
        vergelijk haar amplitude met de geothermische flux \qty{0.06}{W/m^2}.
  \item De geothermische \omterm{def:b2:maxwell-equations:operators}{gradiënt} is \qty{30}{K/km}: ga hem na met de
        flux en $\lambda$, en verklaar waarom hij in de seizoenen van de
        kelder onzichtbaar is.
\end{enumerate}

\textbf{Deel III --- De processor.}
Een chip voert \qty{100}{W} af in een die van \qty{1}{cm^2}, \qty{0.5}{mm}
dik; een laag pasta van \qty{50}{\micro m} over \qty{10}{cm^2} verbindt hem
met een aluminium koellichaam: voet \qty{60}{mm} $\times$ \qty{60}{mm} $\times$
\qty{5}{mm}, dertig ribben van \qty{60}{mm} breed, \qty{30}{mm} hoog, \qty{1}{mm}
dik; een ventilator geeft $h = \qty{50}{W/m^2/K}$; omgeving \qty{25}{\celsius}.
\begin{enumerate}[resume]
  \item Fluxdichtheid door de die; weerstand van de die.
  \item Weerstand van de pasta; en als een luchtspleet van \qty{50}{\micro m}
        haar verving.
  \item Ribparameter $m$, $mL$, rendement; ga het \omterm{def:b2:heat-conduction:biot}{biotgetal} over de
        dikte van de rib na.
  \item Totaal ribbenoppervlak en de weerstand van de ribben; weerstand van
        de voet (geleiding door \qty{5}{mm}); junctietemperatuur.
  \item Idem zonder ribben (kaal vierkant van \qty{60}{mm}): besluit.
  \item De ventilator valt stil ($h = \qty{10}{W/m^2/K}$): junctietemperatuur;
        wat doet de processor bij \qty{100}{\celsius}?
  \item Massa en warmtecapaciteit van het koellichaam; zijn tijdconstante $RC$;
        en die van de die alleen (\qty{0.1}{g}, \qty{700}{J/kg/K}).
  \item Waarom hebben goede koellichamen een koperen voet of heatpipes?
\end{enumerate}

\textbf{Deel IV --- Entropie.}
\begin{enumerate}[resume]
  \item Entropie die per seconde wordt gemaakt door de kale schil van het huis
        (vraag 4) tussen \qty{293}{K} en \qty{273}{K}; na
        isolatie.
  \item Entropie die wordt gemaakt doordat de \qty{100}{W} van de chip van de
        junctie naar de omgeving gaat; en door de omzetting van elektrische
        arbeid in warmte in de die.
  \item Toon aan dat voor een plaat
        $\int\sigma_s\,\dd V = \Phi(1/T_{\text{c}} -
        1/T_{\text{h}})$.
  \item Vat in vijf regels samen wat een muur, een rib, een kelder
        en een chip bepaalt.
\end{enumerate}
\end{problem}
