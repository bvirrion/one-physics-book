\chapter{Thermodynamic Potentials}\label{ch:b2:thermodynamic-potentials}

Stretch a rubber band quickly and touch it to your lip: it is warm;
let it contract and it is cool. Hang a weight on it and warm it with a
hair-dryer: it \emph{shortens}. No steel spring behaves so, and the
reason is that a rubber band's tension is not a matter of energy but of
entropy --- the stretched chains have fewer ways to arrange themselves.
To reason about such things one needs to know what a system at fixed
temperature, or at fixed temperature and pressure, tends to do, and
how much work can be drawn from it: questions the internal energy and
the entropy answer only clumsily, because they speak the language of an
isolated system. This chapter introduces the two functions built for
the laboratory's conditions, the \emph{Helmholtz \omterm{def:b2:thermodynamic-potentials:FG}{free energy}} $F = U -
TS$ and the \emph{Gibbs \omterm{def:b2:thermodynamic-potentials:FG}{free enthalpy}} $G = H - TS$, shows that they
measure the recoverable work and decrease toward equilibrium, extracts
from their differentials the Maxwell relations that link quantities
no one would have thought related, and applies the whole to the
equilibrium of two phases, where $G$ decides which phase wins and
gives back Clapeyron's formula.

\section{The fundamental identity and its variables}

\begin{proposition}[Fundamental identity]\label{prop:b2:thermodynamic-potentials:identity}
For a closed system of fixed composition, the internal energy,
regarded as a function of the entropy and the volume, satisfies
\[
\dd U = T\,\dd S - P\,\dd V, \qquad
T = \Big(\frac{\partial U}{\partial S}\Big)_V, \quad
P = -\Big(\frac{\partial U}{\partial V}\Big)_S ,
\]
and the \omterm{thm:b2:open-systems:firstlaw}{enthalpy} $H = U + PV$ satisfies $\dd H = T\,\dd S + V\,\dd P$.
Each function, in its \emph{natural variables}\index{natural
variables} --- $(S,V)$ for $U$, $(S,P)$ for $H$ --- contains the whole
thermodynamics of the system: its partial derivatives give the other
state variables. The second law selects the equilibrium of an isolated
system (fixed $U$, $V$) as the state of maximum $S$.
\end{proposition}

\begin{proof}
The first law for a reversible transformation, $\dd U = \delta Q_{\text{rev}}
+ \delta W_{\text{rev}} = T\,\dd S - P\,\dd V$, involves only state
functions and therefore holds for any infinitesimal change between
equilibrium states. Add $\dd(PV)$ for $H$.
\end{proof}

\begin{remark}[Why new functions]\label{rem:b2:thermodynamic-potentials:why}
Experiments are not done at fixed entropy. A chemist's flask sits at
the temperature of the room and the pressure of the atmosphere; a
piston in a thermostat works at fixed $T$. One wants functions whose
\omterm{prop:b2:thermodynamic-potentials:identity}{natural variables} are $(T,V)$ and $(T,P)$, and which play, under those
conditions, the role that $S$ plays for an isolated system: to tell
which way the system evolves and how much work it can give. A
\emph{Legendre transform} --- subtracting $TS$ to trade $S$ for $T$ as
the independent variable --- builds them.
\end{remark}

\section{Free energy and free enthalpy}

\begin{definition}[Helmholtz free energy, Gibbs free enthalpy]\label{def:b2:thermodynamic-potentials:FG}
\[
F = U - TS \quad\text{(\emph{free energy}\index{free energy})}, \qquad
G = H - TS = U + PV - TS \quad\text{(\emph{free enthalpy}\index{free enthalpy})}.
\]
Both are extensive state functions, measured in joules.
\end{definition}

\begin{theorem}[Differentials and derived quantities]\label{thm:b2:thermodynamic-potentials:dFdG}
\[
\dd F = -S\,\dd T - P\,\dd V, \qquad
\dd G = -S\,\dd T + V\,\dd P ;
\]
hence, in the \omterm{prop:b2:thermodynamic-potentials:identity}{natural variables} $(T,V)$ of $F$ and $(T,P)$ of $G$,
\[
S = -\Big(\frac{\partial F}{\partial T}\Big)_V, \quad
P = -\Big(\frac{\partial F}{\partial V}\Big)_T, \qquad
S = -\Big(\frac{\partial G}{\partial T}\Big)_P, \quad
V = \Big(\frac{\partial G}{\partial P}\Big)_T ,
\]
and (Gibbs--Helmholtz) $U = F - T(\partial F/\partial T)_V =
-T^2\,\partial(F/T)/\partial T$, $H = -T^2\,\partial(G/T)/\partial T$.
Knowing $F(T,V)$ or $G(T,P)$ is knowing everything: the equation of
state, the entropy, the energy, the heat capacities.
\end{theorem}

\begin{proof}
$\dd F = \dd U - T\dd S - S\dd T = -S\dd T - P\dd V$; likewise for $G$
from $\dd H$. Read off the partial derivatives; for Gibbs--Helmholtz,
$F - T\,\partial_TF = F + TS = U$ and $\partial_T(F/T) = (T\partial_TF -
F)/T^2 = -U/T^2$.
\end{proof}

\begin{example}[The perfect gas]\label{ex:b2:thermodynamic-potentials:perfect}
For $n$ moles, $U = nc_{V}T + U_0$ and $S = nc_V\ln T + nR\ln V + S_0$
give $F(T,V) = -nRT\ln V + f(T)$: then $P = -\partial_VF = nRT/V$
--- the equation of state comes out of $F$ --- and $S = -\partial_TF$
gives back the entropy. In the variables $(T,P)$, $G(T,P) = nRT\ln P +
g(T)$, so $V = \partial_PG = nRT/P$, and the \emph{molar} \omterm{def:b2:thermodynamic-potentials:FG}{free enthalpy}
$\mu(T,P) = \mu^\circ(T) + RT\ln(P/P^\circ)$ --- the \emph{chemical
potential}\index{chemical potential} of the gas, which grows with its
pressure: gas flows from high to low $\mu$, here from high to low
pressure.
\end{example}

\section{Maximum work and the direction of evolution}

\begin{theorem}[Monothermal transformations]\label{thm:b2:thermodynamic-potentials:monothermal}
A system exchanging heat with a single thermostat at $T_0$, and whose
initial and final temperatures equal $T_0$, receives during any
transformation the work
\[
W \ge \Delta F , \qquad\text{i.e.}\qquad W_{\text{recovered}} = -W \le -\Delta F :
\]
the decrease of $F$ is the \emph{maximum work recoverable} from the
system at fixed temperature (reached for a reversible transformation) ---
hence ``free'' energy: the part of $U$ that can be turned into work, the
rest, $TS$, being bound. If moreover the transformation is monobaric
(pressure $P_0$ outside, initial and final pressure $P_0$), the work
\emph{other than that of the atmosphere} (\omterm{def:b2:open-systems:cv}{useful work}, e.g. electrical)
satisfies $W_{\text{u}} \ge \Delta G$: $-\Delta G$ is the maximum \omterm{def:b2:open-systems:cv}{useful work}
recoverable at fixed $T$ and $P$ (the work of a fuel cell, of a
battery).
\end{theorem}

\begin{proof}
First law: $\Delta U = W + Q$. Second law with one thermostat: $\Delta S
= Q/T_0 + S_{\text{c}}$, $S_{\text{c}} \ge 0$, so $Q \le T_0\Delta S$. Hence
$W = \Delta U - Q \ge \Delta U - T_0\Delta S = \Delta(U - T_0S) = \Delta F$, using
$T = T_0$ at both ends. With the atmosphere's work $-P_0\Delta V$
separated, $W = W_{\text{u}} - P_0\Delta V$ and $W_{\text{u}} \ge \Delta U + P_0
\Delta V - T_0\Delta S = \Delta G$.
\end{proof}

\begin{theorem}[Criteria of evolution and equilibrium]\label{thm:b2:thermodynamic-potentials:criteria}
A system held at fixed $T$ and $V$ and receiving no work can only
evolve so that \emph{$F$ decreases}; its equilibrium is the state of
minimum $F$. A system held at fixed $T$ and $P$ and receiving no \omterm{def:b2:open-systems:cv}{useful
work} can only evolve so that \emph{$G$ decreases}; its equilibrium is
the state of minimum $G$. (More generally, for a system in contact with
a surrounding at $T_0$, $P_0$, the quantity $U - T_0S + P_0V$ decreases.)
\end{theorem}

\begin{proof}
Take $W = 0$ (resp.\ $W_{\text{u}} = 0$) in the previous theorem: $\Delta F
\le 0$ (resp.\ $\Delta G \le 0$), with equality for a reversible, i.e.
equilibrium, evolution.
\end{proof}

\begin{example}[Isothermal expansion, real and ideal]\label{ex:b2:thermodynamic-potentials:expansion}
One mole of gas at \qty{300}{K} expands from \qty{1}{L} to \qty{10}{L} in
contact with the thermostat: $\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$. A
reversible expansion recovers \qty{5.7}{kJ}; an expansion against the
atmosphere at \qty{1}{bar} recovers only $P_0\Delta V = \qty{0.9}{kJ}$; a
free expansion into vacuum, nothing. The \qty{5.7}{kJ} is the maximum
the gas and its thermostat can give for this change of state, however
clever the machine.
\end{example}

\section{Maxwell relations}

\begin{theorem}[Maxwell relations]\label{thm:b2:thermodynamic-potentials:maxwell}
Because $\dd F$ and $\dd G$ are exact differentials (Schwarz's theorem
on mixed second derivatives),
\[
\Big(\frac{\partial S}{\partial V}\Big)_T = \Big(\frac{\partial P}{\partial T}\Big)_V,
\qquad
\Big(\frac{\partial S}{\partial P}\Big)_T = -\Big(\frac{\partial V}{\partial T}\Big)_P .
\]
They express the unmeasurable (how the entropy varies with volume or
pressure at fixed temperature) through the equation of state. Two
consequences:
\[
\Big(\frac{\partial U}{\partial V}\Big)_T = T\Big(\frac{\partial P}{\partial T}\Big)_V - P,
\qquad
C_P - C_V = T\Big(\frac{\partial P}{\partial T}\Big)_V\Big(\frac{\partial V}{\partial T}\Big)_P
= \frac{TV\alpha^2}{\kappa_T} ,
\]
with $\alpha = V^{-1}(\partial V/\partial T)_P$ the expansion coefficient and
$\kappa_T = -V^{-1}(\partial V/\partial P)_T$ the isothermal compressibility.
\end{theorem}

\begin{proof}
$\partial^2F/\partial T\partial V$ computed in both orders: $-\partial_VS =
-\partial_TP$. Same for $G$. Then $\dd U = T\dd S - P\dd V$ at fixed $T$
gives $(\partial_VU)_T = T(\partial_VS)_T - P$; and $C_P - C_V$ follows
from writing $S(T,V(T,P))$: $(\partial_TS)_P = (\partial_TS)_V +
(\partial_VS)_T(\partial_TV)_P$, multiplied by $T$.
\end{proof}

\begin{example}[What the relations reveal]\label{ex:b2:thermodynamic-potentials:reveal}
For a perfect gas, $T(\partial_TP)_V - P = 0$: the energy does not depend
on the volume (Joule's law recovered) and $C_P - C_V = nR$. For a van
der Waals gas, $(\partial_VU)_T = a/V_{\text{m}}^2$: the internal pressure of
the attractions. For liquid water, $(\partial_TP)_V = \alpha/\kappa_T = 2.1
\times 10^{-4}/4.6 \times 10^{-10} = \qty{4.6e5}{Pa/K}$: heating water in a
rigid full vessel by one kelvin raises the pressure by \qty{4.6}{bar}
(never seal a full bottle), and $(\partial_VU)_T \approx \qty{1400}{bar}$ ---
the cohesion of the liquid; yet $c_P - c_V = Tv\alpha^2/\kappa_T \approx
\qty{30}{J/kg/K}$, less than $1\%$ of $c_P$: for condensed matter the two
heat capacities nearly coincide.
\end{example}

\begin{proposition}[Entropic elasticity of rubber]\label{prop:b2:thermodynamic-potentials:rubber}
For a band of length $L$ under the tension $f$, $\dd U = T\dd S + f\dd L$
and $\dd F = -S\dd T + f\dd L$, so that $(\partial S/\partial L)_T =
-(\partial f/\partial T)_L$ and $(\partial U/\partial L)_T = f - T(\partial f/
\partial T)_L$. Rubber obeys, to a good approximation, $f = T\,\varphi(L)$
with $\varphi > 0$ for $L > L_0$: then $(\partial U/\partial L)_T = 0$ --- the
energy does not change on stretching --- and $(\partial S/\partial L)_T =
-\varphi(L) < 0$: the tension is \emph{entropic}, $f = -T(\partial S/\partial
L)_T$, the band pulls back because its stretched chains have fewer
configurations. Consequences: stretched isothermally it gives off heat
($Q = T\Delta S < 0$); stretched adiabatically it warms; heated under
constant load it contracts ($f/T$ fixed means $\varphi(L)$ fixed, so at
fixed $f$, $\varphi$ must fall as $T$ rises).
\end{proposition}

\begin{proof}
Maxwell relation from $\dd F$; the rest is substitution. The
microscopic origin is the random walk of \cref{ch:b2:particle-diffusion}:
a chain of $N$ links of length $a$ has its end-to-end distance
distributed as a Gaussian of variance $Na^2$, so the number of
configurations at extension $L$ is $\Omega \propto \eu^{-L^2/2Na^2}$, $S = S_0 -
k_BL^2/2Na^2$ and $f = k_BTL/Na^2$: a spring whose stiffness is
\emph{proportional to the temperature}.
\end{proof}

\section{Equilibrium between two phases}

\begin{theorem}[Condition of coexistence]\label{thm:b2:thermodynamic-potentials:coexistence}
A pure substance at fixed $T$ and $P$, split between two phases of
masses $m_1$, $m_2$ with free enthalpies per unit mass $g_1(T,P)$,
$g_2(T,P)$, has $G = m_1g_1 + m_2g_2$; at equilibrium $G$ is minimal, so
\begin{itemize}
\item if $g_1 < g_2$ all the matter goes into phase 1 (the stable
      phase is the one of \emph{lower} $g$);
\item the two phases coexist only where
      \[
      g_1(T,P) = g_2(T,P) ,
      \]
      a relation between $T$ and $P$: the coexistence curve of the phase
      diagram.
\end{itemize}
Along that curve (Clapeyron),
\[
\frac{\dd P}{\dd T} = \frac{s_2 - s_1}{v_2 - v_1} = \frac{L_{1\to2}}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
$\dd G = (g_1 - g_2)\dd m_1$ at fixed $T$, $P$ (with $m_1 + m_2$ fixed):
$G$ decreases by moving mass to the phase of lower $g$, until that
phase is alone or until $g_1 = g_2$. Differentiate $g_1 = g_2$ along the
curve with $\dd g = -s\,\dd T + v\,\dd P$ for each phase: $-s_1\dd T +
v_1\dd P = -s_2\dd T + v_2\dd P$; and $s_2 - s_1 = L/T$ since the transition
at fixed $T$, $P$ is reversible. The Year 1 volume obtained this formula
from a Carnot cycle; here it falls out of $G$.
\end{proof}

\begin{example}[Water: slopes and metastability]\label{ex:b2:thermodynamic-potentials:water}
Fusion: $L = \qty{334}{kJ/kg}$, $v_{\text{liq}} - v_{\text{ice}} = -\qty{9.1e-5}{m^3/kg}$:
$\dd P/\dd T = \qty{-1.3e7}{Pa/K}$, \emph{negative} --- ice melts under
pressure, but \qty{130}{bar} per kelvin: a skater's \qty{70}{bar} lower
the melting point by half a degree only (skating works by friction and
a premelted surface layer), while the \qty{270}{bar} under three
kilometres of ice sheet melt its base and let it slide. Vaporisation at
\qty{100}{\celsius}: $L = \qty{2.26}{MJ/kg}$, $\Delta v = \qty{1.67}{m^3/kg}$:
\qty{3.6}{kPa/K} --- \qty{92}{\celsius} at \qty{0.7}{bar} on a mountain,
\qty{120}{\celsius} in a pressure cooker at \qty{2}{bar}. Below
\qty{0}{\celsius}, liquid water has $g_{\text{liq}} - g_{\text{ice}} \approx
(L/T_{\text{fus}})(T_{\text{fus}} - T) = \qty{1.2}{kJ/kg}$ per kelvin of
supercooling: it is metastable, not unstable, and pure droplets in
clouds stay liquid down to \qty{-40}{\celsius} for want of a nucleus.
\end{example}

\begin{remark}[Van der Waals and Maxwell's construction]\label{rem:b2:thermodynamic-potentials:maxwellconstruction}
The van der Waals equation $(P + a/v^2)(v - b) = RT$ gives, below its
critical temperature $T_{\text{c}} = 8a/27Rb$, isotherms with a loop. The
real fluid does not follow the loop: at $P_{\text{sat}}(T)$ it jumps from
the liquid to the vapour branch, and $P_{\text{sat}}$ is fixed by $g_{\text{liq}}
= g_{\text{vap}}$, i.e. $\int_{\text{liq}}^{\text{vap}}v\,\dd P = 0$ along the
loop: the horizontal line cuts off \emph{equal areas}. The parts of the
loop where $(\partial P/\partial v)_T > 0$ are mechanically unstable; the
parts between them and the line are metastable --- the superheated
liquid that ``bumps'' in a microwave, the supersaturated vapour of the
cloud chamber.
\end{remark}

\begin{method}[Using the potentials]\label{meth:b2:thermodynamic-potentials:method}
(1) Identify the constraints: isolated $\to$ $S$ max; fixed $T,V$ $\to$
$F$ min; fixed $T,P$ $\to$ $G$ min. (2) Maximum work: $-\Delta F$ at fixed
$T$, $-\Delta G$ of \omterm{def:b2:open-systems:cv}{useful work} at fixed $T,P$. (3) Need a derivative of
$S$ at fixed $T$? Use a Maxwell relation and the equation of state.
(4) Two phases: compare $g$; coexistence $g_1 = g_2$; Clapeyron for the
slope; sign of $\Delta v$ for the sign of the slope. (5) \omterm{ex:b2:thermodynamic-potentials:perfect}{Chemical
potential} $\mu = g$ per mole: matter flows toward lower $\mu$.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:1}
For $n$ moles of perfect gas, write $F(T,V)$ and $G(T,P)$ up to
functions of $T$ alone; recover the equation of state from each; show
that $S = -\partial_TF$ gives the known entropy.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:2}
One mole of gas, \qty{300}{K}, from \qty{1}{L} to \qty{10}{L} in a
thermostat: $\Delta F$, $\Delta U$, $\Delta S$; work recovered if reversible;
if against \qty{1}{bar}; if into vacuum; the entropy created in each
case.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:3}
Water and ice at \qty{1}{bar}: show that $g_{\text{liq}} - g_{\text{ice}} \approx
(L/T_{\text{fus}})(T_{\text{fus}} - T)$ near \qty{0}{\celsius} and compute it
at \qty{-5}{} and $+\qty{5}{\celsius}$ ($L = \qty{334}{kJ/kg}$). Which phase is
stable in each case; is the other impossible?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:4}
Clapeyron slopes of water: fusion ($\Delta v = -\qty{9.1e-5}{m^3/kg}$) and
vaporisation at \qty{100}{\celsius} ($L = \qty{2.26}{MJ/kg}$, $\Delta v =
\qty{1.67}{m^3/kg}$). Melting point under \qty{70}{bar}; boiling point at
\qty{0.7}{bar} and at \qty{2}{bar}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:5}
\emph{Maxwell.} (a) Derive the two Maxwell relations of the chapter
and the two from $\dd U$ and $\dd H$. (b) Show $(\partial U/\partial V)_T =
T(\partial P/\partial T)_V - P$; evaluate it for a perfect gas and a van
der Waals gas. (c) Liquid water: $\alpha = \qty{2.1e-4}{K^{-1}}$, $\kappa_T =
\qty{4.6e-10}{Pa^{-1}}$: pressure rise per kelvin in a rigid full vessel,
and $(\partial U/\partial V)_T$. (d) A full, sealed glass bottle is warmed
by \qty{10}{K}: comment.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:6}
\emph{$C_P - C_V$.} (a) Derive $C_P - C_V = TV\alpha^2/\kappa_T$. (b)
Perfect gas. (c) Water at \qty{300}{K}; copper ($\alpha = \qty{5e-5}{K^{-1}}$,
$\kappa_T = \qty{7e-12}{Pa^{-1}}$, $\rho = \qty{8900}{kg/m^3}$); compare with
$c_P$ (\qty{4180}{} and \qty{385}{J/kg/K}). (d) Why can one speak of
``the'' heat capacity of a solid?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:7}
\emph{Rubber.} A band obeys $f = CT(L - L_0)$ with $C = \qty{1}{N/m/K}$.
(a) $(\partial S/\partial L)_T$ and $(\partial U/\partial L)_T$. (b) Entropy
change, heat and work when stretched isothermally at \qty{300}{K} from
$L_0$ to $L_0 + \qty{0.1}{m}$. (c) Stretched adiabatically instead (heat
capacity \qty{4}{J/K}): temperature change. (d) Hung with a fixed weight
and warmed by \qty{30}{K}: relative change of $L - L_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:8}
\emph{Evaporation of a puddle.} The \omterm{ex:b2:thermodynamic-potentials:perfect}{chemical potential} of water vapour
at partial pressure $p$ is $\mu_{\text{v}} = \mu_{\text{v}}^\circ(T) +
RT\ln(p/p^\circ)$, and equals that of the liquid when $p = p_{\text{sat}}(T)$.
(a) Show that liquid evaporates when $p < p_{\text{sat}}$ and vapour
condenses when $p > p_{\text{sat}}$; define the relative humidity. (b)
$\mu_{\text{liq}} - \mu_{\text{v}}$ at $50\%$ humidity, \qty{293}{K}. (c) Why
does washing dry even at \qty{20}{\celsius}, and never at $100\%$ humidity?
(d) Dew: on what surfaces does it form first, and why?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:9}
\emph{Van der Waals.} $(P + a/v^2)(v - b) = RT$ (molar). (a) Find the
critical point ($\partial_vP = \partial_v^2P = 0$): $v_{\text{c}} = 3b$,
$T_{\text{c}} = 8a/27Rb$, $P_{\text{c}} = a/27b^2$. (b) Carbon dioxide,
$T_{\text{c}} = \qty{304}{K}$, $P_{\text{c}} = \qty{73.8}{bar}$: $a$ and $b$. (c)
Justify Maxwell's equal-area rule from $g$. (d) Which parts of the loop
are unstable, which metastable, and what phenomena show the metastable
ones?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:10}
\emph{Clausius--Clapeyron integrated.} For vaporisation, neglect $v_{\text{liq}}$
and treat the vapour as perfect. (a) Show $\dd\ln P/\dd T = L_{\text{m}}/RT^2$
($L_{\text{m}}$ molar) and integrate for constant $L_{\text{m}}$. (b) Water,
$L_{\text{m}} = \qty{40.7}{kJ/mol}$: $P_{\text{sat}}$ at \qty{20}{\celsius} from the
\qty{1}{bar} at \qty{100}{\celsius}; compare with the measured \qty{23}{mbar}.
(c) Predict the triple-point pressure (\qty{0.01}{\celsius}) and compare
with \qty{611}{Pa}. (d) Show the same law follows from Gibbs--Helmholtz
applied to $\Delta g$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:11}
\emph{Osmosis.} A solvent's \omterm{ex:b2:thermodynamic-potentials:perfect}{chemical potential} is lowered by a dilute
solute: $\mu = \mu^\circ - RTx_{\text{s}}$ ($x_{\text{s}}$ the solute mole
fraction). A membrane lets only the solvent through; pure solvent on
one side, solution on the other, same $T$. (a) Show that equilibrium
requires an extra pressure $\Pi$ on the solution with $\Pi v_{\text{m}} =
RTx_{\text{s}}$, i.e. $\Pi = cRT$ (van 't Hoff). (b) Sea water, $c \approx
\qty{1.1}{mol/L}$ of dissolved particles, \qty{293}{K}: $\Pi$. (c) Minimum
work to extract a cubic metre of fresh water; compare with real plants
(\qty{3}{kWh/m^3}). (d) Why do plant cells burst in pure water and
shrivel in brine?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:12}
\emph{The chain.} A polymer chain of $N$ links of length $a$: the
number of configurations with end-to-end vector of length $L$ along
the band is $\Omega \propto \eu^{-L^2/2Na^2}$ (the random walk). (a) $S(L)$
and, at fixed $U$, the tension $f = -T\,\partial S/\partial L$. (b) The
chain's stiffness and its temperature dependence. (c) $N = 1000$, $a =
\qty{0.5}{nm}$, \qty{300}{K}, $L = \qty{10}{nm}$: $f$; how many chains in
parallel give \qty{30}{N}? (d) Estimate the \omterm{prop:b2:waves-on-strings:chain}{Young's modulus} of a rubber
with $10^{26}$ chains per cubic metre, each of $N = 100$.
\end{exercise}

\section{Problem: The rubber band and the phase diagram of water}

\begin{problem}[{Weekend problem --- entropy that pulls, and phases
that compete}]\label{pb:b2:thermodynamic-potentials:1}
\textbf{Part I --- The rubber band.}
A band of mass \qty{2}{g} (heat capacity \qty{4}{J/K}) has the tension
$f = CT(L - L_0)$ with $C = \qty{1}{N/m/K}$; at \qty{300}{K} and $L - L_0 =
\qty{0.1}{m}$, $f = \qty{30}{N}$.
\begin{enumerate}
  \item Write $\dd U$ and $\dd F$ for the band ($\dd U = T\dd S + f\dd L$).
  \item Derive the Maxwell relation $(\partial S/\partial L)_T = -(\partial
        f/\partial T)_L$.
  \item Show that $(\partial U/\partial L)_T = 0$ for this band: what does
        it mean?
  \item Isothermal stretching from $L_0$ to $L_0 + \qty{0.1}{m}$ at
        \qty{300}{K}: $\Delta S$, heat $Q$, work $W$; check $\Delta U = 0$.
  \item Same stretching done adiabatically: temperature change.
  \item The band holds a fixed weight; it is warmed by \qty{30}{K}: new
        $L - L_0$. Describe the ``rubber engine'' (a wheel with rubber
        spokes heated on one side).
  \item A chain of $N$ links of length $a$ has $\Omega \propto \eu^{-L^2/2Na^2}$:
        $S(L)$ and $f(L,T)$; why does the stiffness grow with $T$?
  \item $N = 1000$, $a = \qty{0.5}{nm}$, $L = \qty{10}{nm}$, \qty{300}{K}: $f$
        for one chain and the number of chains for \qty{30}{N}.
  \item Why does a steel spring behave oppositely (tension falling
        with $T$ at fixed length)?
\end{enumerate}

\textbf{Part II --- The phase diagram of water.}
Data: $L_{\text{fus}} = \qty{334}{kJ/kg}$, $v_{\text{ice}} = \qty{1.091e-3}{m^3/kg}$,
$v_{\text{liq}} = \qty{1.000e-3}{m^3/kg}$; $L_{\text{vap}} = \qty{2.26}{MJ/kg}$ at
\qty{100}{\celsius}, $v_{\text{vap}} = \qty{1.67}{m^3/kg}$; $L_{\text{m}} = \qty{40.7}{kJ/mol}$;
triple point \qty{0.01}{\celsius}, \qty{611}{Pa}; critical point \qty{647}{K},
\qty{221}{bar}.
\begin{enumerate}[resume]
  \item Sketch $g(T)$ for ice, liquid and vapour at \qty{1}{bar}; justify
        the order of the slopes and locate the transitions.
  \item Derive Clapeyron's formula from $g_1 = g_2$.
  \item Slope of the fusion curve; its sign, and why it is unusual.
  \item A skater: \qty{700}{N} on \qty{1}{cm^2}: change of the melting
        point; conclude about the popular explanation of skating.
  \item Under \qty{3}{km} of ice ($\rho = \qty{917}{kg/m^3}$): pressure and
        melting point; consequence for the ice sheet.
  \item Slope of the vaporisation curve at \qty{100}{\celsius}; boiling
        point at \qty{0.7}{bar} and at \qty{2}{bar}.
  \item Integrate Clausius--Clapeyron with a perfect vapour and
        constant $L_{\text{m}}$; $P_{\text{sat}}$ at \qty{20}{\celsius} and at
        the triple point; comment on the accuracy.
  \item At the triple point, compare the slopes of the sublimation and
        vaporisation curves ($L_{\text{sub}} = L_{\text{fus}} + L_{\text{vap}}$).
  \item Supercooled water at \qty{-10}{\celsius}: $g_{\text{liq}} - g_{\text{ice}}$;
        why can cloud droplets stay liquid to \qty{-40}{\celsius}?
  \item What happens to $\Delta v$, $L$ and the slope at the critical
        point; what is a supercritical fluid?
\end{enumerate}

\textbf{Part III --- Applications.}
\begin{enumerate}[resume]
  \item Explain Maxwell's equal-area construction on a van der Waals
        isotherm from the continuity of $g$.
  \item Superheated water in a microwave ``bumps'': which branch of the
        isotherm, and why the danger?
  \item Freeze-drying: describe the path in the $(P,T)$ diagram and
        why the water leaves without melting.
  \item Why does a dish of water evaporate at \qty{20}{\celsius} in dry air
        and stop in saturated air --- in terms of \omterm{ex:b2:thermodynamic-potentials:perfect}{chemical potentials}?
  \item A pressure cooker at \qty{2}{bar}: why does food cook so much
        faster (the rate of the chemistry roughly doubles every
        \qty{10}{K})?
  \item Summarise: which potential for which constraint, and what the
        Maxwell relations buy.
\end{enumerate}
\end{problem}
