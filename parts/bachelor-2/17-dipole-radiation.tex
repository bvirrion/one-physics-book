\chapter{Dipole Radiation and Scattering}\label{ch:b2:dipole-radiation}

Why is the sky blue, and why is it bluest at right angles to the Sun
--- and polarized there, as bees and photographers know? Why does a
radio mast have to be tens of metres tall, and why does it radiate
nothing straight up? Both questions have the same answer: an
oscillating electric charge radiates \omterm{thm:b2:plane-waves-polarization:wave}{electromagnetic waves}, with a
power that grows as the fourth power of the frequency and a pattern
that vanishes along the axis of oscillation. This chapter states the
field of the \emph{\omterm{thm:b2:dipole-radiation:field}{oscillating dipole}}, the simplest radiating system,
draws from it the antenna and the radiation of an accelerated charge,
and applies it to the molecules of the air, each a tiny antenna
re-emitting the sunlight: \omterm{prop:b2:dipole-radiation:rayleigh}{Rayleigh scattering}.

\section{The field of an oscillating dipole}

\begin{theorem}[Radiation field of an electric dipole]\label{thm:b2:dipole-radiation:field}
\index{oscillating dipole}\index{radiation zone}\index{retardation}
A dipole of moment $\vect p(t) = p_0\cos\omega t\,\vect e_z$, of size $a$, at the
origin, radiates. In the \emph{\omterm{thm:b2:dipole-radiation:field}{radiation zone}} --- at distances $r \gg
\lambda = 2\pi c/\omega \gg a$ --- its field, in spherical coordinates $(r,
\theta, \varphi)$ about the dipole's axis, is
\[
\vect E(M, t) = -\frac{\mu_0\,\omega^2p_0}{4\pi}\,\frac{\sin\theta}r\,\cos\bigl(\omega(t - r/c)\bigr)\,\vect e_\theta ,
\qquad
\vect B = \frac{\vect e_r\wedge\vect E}c :
\]
a \omterm{prop:b2:sound-waves:spherical}{spherical wave}, locally plane --- $(\vect E, \vect B, \vect e_r)$ a
right-handed triad with $B = E/c$ --- \emph{retarded} by the travel time
$r/c$, of amplitude falling as $1/r$, proportional to the acceleration
$\ddot p = -\omega^2p$ of the dipole and to $\sin\theta$: maximal in the
equatorial plane, zero along the axis; polarized in the plane
containing the axis.
\end{theorem}
\admitted

\begin{remark}[Reading the formula]\label{rem:b2:dipole-radiation:reading}
The exact solution of \omterm{thm:b2:maxwell-equations:maxwell}{Maxwell's equations} (from the retarded
potentials, beyond this volume) contains also terms in $1/r^2$ and
$1/r^3$; the last is the quasi-static dipole field of the Year 1
volume, $\vect E \propto p/4\pi\varepsilon_0r^3$, which dominates in the \emph{near
zone} $r \ll \lambda$; at $r \sim \lambda/2\pi$ the two are comparable, beyond it
only the $1/r$ term survives. Each factor has a reason. $1/r$: the
power through a sphere, $\propto E^2r^2$, must not depend on $r$. $\omega^2p_0$:
the field of a charge at rest or in uniform motion is not radiative; it
is the \emph{acceleration} that radiates. $\sin\theta$: an observer on the
axis sees the charge come and go along the line of sight, with no
transverse acceleration; in the equatorial plane the whole
acceleration is transverse. Retardation: the field at $M$ at time $t$
reflects what the dipole did at $t - r/c$.
\end{remark}

\section{Radiated power}

\begin{proposition}[Power of a dipole; Larmor's formula]\label{prop:b2:dipole-radiation:power}
\index{radiated power (dipole)}\index{Larmor formula}\index{radiation pattern}
The mean \omterm{thm:b2:poynting-vector:poynting}{Poynting vector} of the dipole's field is radial,
\[
\langle\vect\Pi\rangle = \frac{\mu_0\,\omega^4p_0^2}{32\pi^2c}\,\frac{\sin^2\theta}{r^2}\,\vect e_r ,
\]
and the total mean power radiated is
\[
\mathcal P = \frac{\mu_0\,\omega^4p_0^2}{12\pi c} = \frac{p_0^2\omega^4}{12\pi\varepsilon_0c^3} .
\]
For a single charge $q$ with acceleration $a$ (non-relativistic), the
instantaneous power is $\mathcal P = q^2a^2/6\pi\varepsilon_0c^3$ (Larmor).
\end{proposition}

\begin{proof}
$\langle\Pi\rangle = \langle E^2\rangle/\mu_0c$ with $\langle\cos^2\rangle = \tfrac12$;
integrate over the sphere:
\[
\int\sin^2\theta\,r^2\dd\Omega = 2\pi r^2\int_0^\pi\sin^3\theta\,\dd\theta = \tfrac83\pi r^2 ,
\qquad
\mathcal P = \frac{\mu_0\omega^4p_0^2}{32\pi^2c}\cdot\frac{8\pi}3 = \frac{\mu_0\omega^4p_0^2}{12\pi c} .
\]
Larmor: $p = qz$, so $\ddot p = qa$;
replacing the mean $\tfrac12\omega^4p_0^2 = \langle\ddot p^2\rangle$ by the
instantaneous $q^2a^2$ gives $\mathcal P = \mu_0q^2a^2/6\pi c$.
\end{proof}

\begin{example}[An antenna and a wire]\label{ex:b2:dipole-radiation:antenna}
\index{antenna}\index{radiation resistance}\index{half-wave dipole}
A short wire of length $\ell \ll \lambda$ carrying the current $I_0\cos\omega t$ is a
dipole with $\dot p = I_0\ell\cos\omega t$, i.e.\ $p_0 = I_0\ell/\omega$: it radiates
$\mathcal P = \mu_0\omega^2I_0^2\ell^2/12\pi c = \tfrac12R_{\text{rad}}I_0^2$ with the
\emph{\omterm{ex:b2:dipole-radiation:antenna}{radiation resistance}} $R_{\text{rad}} = \dfrac{2\pi}3\sqrt{\dfrac{\mu_0}{\varepsilon_0}}
\Bigl(\dfrac\ell\lambda\Bigr)^2 \approx 790\,(\ell/\lambda)^2\ \Omega$ --- the power leaves the
circuit as if through a resistor, without heating anything. A \qty{1}{m}
wire at \qty{1}{MHz} ($\lambda = \qty{300}{m}$): $R_{\text{rad}} = \qty{9}{m\ohm}$, less than
its copper resistance, a poor antenna; at \qty{100}{MHz}: \qty{88}{\ohm} (the
short-dipole formula is stretched here), a good one. The \emph{\omterm{ex:b2:dipole-radiation:antenna}{half-wave
dipole}}, $\ell = \lambda/2$ with a sinusoidal current distribution, has
$R_{\text{rad}} = \qty{73}{\ohm}$ (admitted) and a pattern close to the
elementary dipole's --- the standard antenna, from radio masts to the
rabbit ears on old televisions. A \qty{1}{m} mains wire at \qty{50}{Hz}:
$R_{\text{rad}} \approx \qty{2e-11}{\ohm}$ --- the quasi-stationary circuits of
\cref{ch:b2:maxwell-equations} radiate nothing.
\end{example}

\begin{example}[An electron that radiates]\label{ex:b2:dipole-radiation:electron}
In an X-ray tube an electron of \qty{50}{keV} ($v = \qty{1.2e8}{m/s}$) stops
in \qty{1}{\micro m} of tungsten: $a \sim v^2/2d = \qty{7e21}{m/s^2}$, Larmor power
$\qty{1.2e-4}{W}$ over $\qty{1.6e-14}{s}$: \qty{2e-18}{J}, about $0.02\%$ of its
energy radiated as the "braking radiation" (bremsstrahlung) of the
tube --- the rest is heat, and the anode is cooled. The classical
electron of a hydrogen atom, accelerated at $v^2/r = \qty{9e22}{m/s^2}$,
would radiate \qty{5e-8}{W} and spiral into the nucleus in \qty{1e-11}{s}:
classical physics forbids atoms, and \cref{ch:b2:schrodinger-wave-functions}
rescues them.
\end{example}

\section{Rayleigh scattering: the blue sky}

\begin{proposition}[Scattering by a bound electron]\label{prop:b2:dipole-radiation:rayleigh}
\index{Rayleigh scattering}\index{scattering cross-section}\index{Thomson scattering}
A wave of amplitude $E_0$ and frequency $\omega$ falls on an atom modelled
as an elastically bound electron of resonance $\omega_0 \gg \omega$
(\cref{ch:b2:waves-in-media}): the electron oscillates with amplitude
$eE_0/m\omega_0^2$, the atom becomes a dipole $p_0 = e^2E_0/m\omega_0^2$ in phase
with the field, and re-radiates the power $\mathcal P = p_0^2\omega^4/12\pi
\varepsilon_0c^3$. The ratio of that power to the incident intensity is the
\emph{\omterm{prop:b2:dipole-radiation:rayleigh}{scattering cross-section}}
\[
\sigma = \frac{\mathcal P}{\varepsilon_0cE_0^2/2} = \frac{8\pi}3\,r_e^2\Bigl(\frac\omega{\omega_0}\Bigr)^4 ,
\qquad r_e = \frac{e^2}{4\pi\varepsilon_0mc^2} = \qty{2.8e-15}{m} :
\]
Rayleigh's law, $\sigma \propto \omega^4 \propto 1/\lambda^4$. For a \emph{free}
electron ($\omega_0 = 0$) the cross-section is $\sigma_T = 8\pi r_e^2/3 =
\qty{6.65e-29}{m^2}$, independent of frequency (\omterm{prop:b2:dipole-radiation:rayleigh}{Thomson scattering}).
\end{proposition}

\begin{proof}
From the \omterm{prop:b2:waves-in-media:lorentz}{Lorentz model} with $\omega \ll \omega_0$ and negligible damping,
$\underline{\vect r} = -e\underline{\vect E}/m\omega_0^2$; insert $p_0$ into the power and
divide by $I = \varepsilon_0cE_0^2/2$; $r_e$ collects the constants. For the free
electron, $\vect r = e\vect E/m\omega^2$ (in antiphase), $p_0 = e^2E_0/m\omega^2$, and
the $\omega^4$ cancel.
\end{proof}

\begin{example}[Blue sky, red sunset, white clouds]\label{ex:b2:dipole-radiation:sky}
Violet light ($\qty{400}{nm}$) is scattered $(700/400)^4 = 9.4$ times more
than red: the sky, lit by scattered sunlight, is blue (not violet: the
Sun emits less violet and the eye sees it poorly). At sunset the direct
light crosses forty times more air than at noon and loses its blue to
the sky of other places: it reddens. Each air molecule scatters with
$\sigma \approx \qty{5e-31}{m^2}$ at \qty{550}{nm}; with $n = \qty{2.5e25}{m^{-3}}$ the
mean free path $1/n\sigma$ is \qty{80}{km}: a tenth of the green light is
scattered on its way down through the atmosphere, a third of the
violet. Cloud droplets, far larger than $\lambda$, are not Rayleigh
scatterers: they scatter all colours alike --- white.
\end{example}

\begin{proposition}[Polarization by scattering]\label{prop:b2:dipole-radiation:polarization}
\index{polarization by scattering}
\omterm{def:b2:plane-waves-polarization:states}{Natural light} travelling along $x$ induces dipoles in the $yz$ plane.
An observer looking along $y$ (at \qty{90}{\degree} from the beam) receives
no radiation from the dipole components along $y$ (his line of sight)
and full radiation from those along $z$: the scattered light is
\emph{linearly polarized} along $z$, perpendicular to the plane
containing the beam and the line of sight. At other angles it is
partially polarized, with the degree $(1 - \cos^2\chi)/(1 + \cos^2\chi)$ for
the scattering angle $\chi$.
\end{proposition}

\begin{proof}
The incident \omterm{def:b2:plane-waves-polarization:states}{natural light} has $\vect E$ with equal mean components
along $y$ and $z$; the dipole along $y$ does not radiate toward $y$ ($\sin
\theta = 0$), the one along $z$ radiates $\propto\sin\theta = 1$. At angle $\chi$
the $y$ component's contribution is reduced by $\cos^2\chi$ in intensity.
\end{proof}

\begin{remark}[Why the blue light reaches us at all]\label{rem:b2:dipole-radiation:why}
In a perfectly uniform medium the waves scattered by neighbouring
molecules would interfere destructively in every direction but forward
(the index of refraction is that forward-scattered wave). It is the
\emph{fluctuations} of the density of air --- molecules are not on a
lattice --- that leave a net scattered intensity, proportional to the
number of molecules, as computed above for one (Einstein, 1910). The
same randomness makes the sky's light incoherent from point to point.
\end{remark}

\begin{method}[Radiation estimates]\label{met:b2:dipole-radiation:method}
(1) Identify the dipole: $p_0 = q\,\times$ amplitude, or $I_0\ell/\omega$ for a
wire. (2) Field at distance $r$: $E \approx \mu_0\omega^2p_0\sin\theta/4\pi r$; power:
$\mu_0\omega^4p_0^2/12\pi c$. (3) For an accelerated charge use Larmor; for
an antenna the \omterm{ex:b2:dipole-radiation:antenna}{radiation resistance}. (4) For scattering, compute the
induced dipole from the medium's response (\omterm{prop:b2:waves-in-media:lorentz}{Lorentz model}) and divide
the radiated power by the incident intensity. (5) Remember the zeros:
nothing along the axis of oscillation, hence the polarization at
\qty{90}{\degree} and the \omterm{prop:b2:wave-interfaces:brewster}{Brewster angle} of \cref{ch:b2:wave-interfaces}.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:1}
A short antenna of \qty{1.0}{m} carries \qty{2.0}{A} (amplitude) at
\qty{100}{MHz}. Dipole amplitude $p_0$; field amplitude at \qty{1}{km} in the
equatorial plane and at \qty{30}{\degree} from the axis; radiated power;
\omterm{ex:b2:dipole-radiation:antenna}{radiation resistance}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:2}
\omterm{ex:b2:dipole-radiation:antenna}{Radiation resistance} of a \qty{1}{m} wire at \qty{100}{kHz}, \qty{1}{MHz},
\qty{10}{MHz}, \qty{100}{MHz}; compare with its copper resistance
(\qty{1}{mm} diameter, with the \omterm{thm:b2:waves-in-media:skin}{skin effect} at the higher frequencies,
$\delta = \qty{0.2}{mm}$ at \qty{100}{kHz}); efficiency (radiated over total)
in each case. Why do long-wave transmitters have masts hundreds of
metres tall?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:3}
Ratio of the \omterm{prop:b2:dipole-radiation:rayleigh}{Rayleigh scattering} of \qty{400}{nm}, \qty{550}{nm} and
\qty{700}{nm} light; of the \qty{3}{cm} and \qty{10}{cm} radar wavelengths by
raindrops (if they were small enough --- they are not quite); why does
weather radar use short wavelengths to see rain and long ones to see
through it?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:4}
Degree of polarization of skylight at \qty{30}{\degree}, \qty{60}{\degree},
\qty{90}{\degree}, \qty{120}{\degree} from the Sun; where in the sky should a
photographer point a polarizer for the darkest blue, and what
happens when the Sun is overhead? Why is the real maximum only about
$75\%$ (think of multiple scattering and of what the ground reflects)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:5}
(a) Integrate the \omterm{thm:b2:poynting-vector:poynting}{Poynting vector} of the dipole over a sphere and
check $\mathcal P = \mu_0\omega^4p_0^2/12\pi c$. (b) Fraction of the power
radiated within \qty{30}{\degree} of the equatorial plane. (c) Half-power
angle: at what $\theta$ is the intensity half its maximum? (d) The
"gain" of the dipole, maximum intensity over the isotropic average:
show it is $1.5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:6}
\emph{Near and far.} (a) Write the quasi-static field of a dipole at
$\theta = 90^\circ$, $E_{\text{ns}} = p/4\pi\varepsilon_0r^3$, and the radiative one; at
what distance $r_0$ are their amplitudes equal? Express $r_0$ in terms of
$\lambda$. (b) For a \qty{1}{MHz} antenna, $r_0$; which field does a receiver
at \qty{10}{m} feel, at \qty{10}{km}? (c) A mobile phone at \qty{1}{GHz} held
\qty{2}{cm} from the head: near or far zone? (d) Why does the near field
carry no net power away?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:7}
\emph{Thomson and the corona.} (a) Compute $\sigma_T$. (b) The solar corona
has $n_e \approx \qty{1e14}{m^{-3}}$ over $10^9$ m: fraction of the photospheric
light scattered toward us; why is the corona white and only visible
during eclipses? (c) X-rays of \qty{10}{keV} on a carbon atom ($6$
electrons, binding energies below \qty{300}{eV}): why do the electrons
scatter as if free, and what is the atom's cross-section? (d) Why
does a medical X-ray of bone differ from flesh mostly through
absorption, not \omterm{prop:b2:dipole-radiation:rayleigh}{Thomson scattering}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:8}
\emph{Sunset.} The optical depth of the atmosphere for \omterm{prop:b2:dipole-radiation:rayleigh}{Rayleigh
scattering} at the zenith is $\tau(\lambda) = 0.1\,(550\,\text{nm}/\lambda)^4$ (the
direct light is attenuated by $\eu^{-\tau}$). (a) Transmission at the
zenith for \qty{400}{nm}, \qty{550}{nm}, \qty{700}{nm}. (b) At sunset the path
is $38$ times longer: transmissions; ratio red/blue. (c) Why does the
Moon turn red during a lunar eclipse? (d) Why is the sky near the
horizon whitish rather than deep blue?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:9}
\emph{Two antennas.} Two vertical dipoles a distance $d$ apart along
$x$ are fed with the same amplitude. (a) In phase, $d = \lambda/2$: in which
horizontal directions do their far fields add, in which cancel?
(broadside array). (b) In antiphase, $d = \lambda/2$: the same (end-fire).
(c) In quadrature, $d = \lambda/4$: show the array radiates toward one
side only. (d) Why do AM broadcast stations use several masts, and
how does a phased-array radar steer its beam without moving?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:10}
\emph{Radiation damping.} The Lorentz electron (\cref{ch:b2:waves-in-media})
oscillating at $\omega_0$ with amplitude $x_0$ radiates by Larmor's formula.
(a) Mean radiated power; energy of the oscillator $\tfrac12m\omega_0^2x_0^2$;
show that the energy decays with the rate $\Gamma = e^2\omega_0^2/6\pi\varepsilon_0mc^3$.
(b) Value of $\Gamma$ for the sodium line (\qty{589}{nm}): compare with the
$\Gamma$ used in \cref{exo:b2:waves-in-media:9}. (c) Lifetime $1/\Gamma$ and the
natural width of the line, $\Delta\lambda = \lambda^2\Gamma/2\pi c$. (d) Quality factor
$\omega_0/\Gamma$ of the atomic oscillator.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:11}
\emph{The mast.} An AM station radiates \qty{50}{kW} at \qty{1}{MHz} from a
vertical mast of height $\lambda/4$ on a conducting ground (which acts as
a mirror, making the mast half of a \omterm{ex:b2:dipole-radiation:antenna}{half-wave dipole}: $R_{\text{rad}} \approx
\qty{36}{\ohm}$, and the power goes into the upper half-space only). (a)
Height; current at the base. (b) Field amplitude at \qty{10}{km} in the
horizontal plane (use the dipole pattern, $\mathcal P$ over the upper
hemisphere $= \int\langle\Pi\rangle\dd S$; take $\langle\Pi\rangle = 1.5\,
\mathcal P/2\pi r^2$ at the horizon as a fair estimate). (c) Emf in a
\qty{1}{m} receiving whip there. (d) Why does the ground's conductivity
matter, and why are radial copper wires buried under the mast?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:12}
\emph{Scattering and the index.} For a dilute gas the forward-scattered
waves of all molecules add coherently and build the refractive index,
$n - 1 = N\alpha/2\varepsilon_0$ with $\alpha = e^2/m\omega_0^2$ the polarizability; the
Rayleigh cross-section is $\sigma = (8\pi/3)r_e^2(\omega/\omega_0)^4$. (a) Express
$\sigma$ in terms of $n$, $N$ and $\lambda$: $\sigma = 8\pi^3(n^2 - 1)^2/3N^2\lambda^4$. (b)
For air ($n - 1 = 2.9 \times 10^{-4}$, $N = \qty{2.5e25}{m^{-3}}$) at \qty{550}{nm}:
$\sigma$ and the mean free path $1/N\sigma$. (c) Vertical optical depth of the
atmosphere (equivalent height \qty{8}{km}); compare with the $0.1$ of
\cref{exo:b2:dipole-radiation:8}. (d) Why did this relation (Rayleigh,
1899) give a value of Avogadro's number?
\end{exercise}

\section{Problem: The mast, the sky and the electron}

\begin{problem}[{Weekend problem --- three antennas: a broadcasting
mast, an air molecule in sunlight, and a free electron}]
\label{pb:b2:dipole-radiation:1}

\medskip
\noindent\textbf{Part I --- The mast.}
A medium-wave station at $f = \qty{1.0}{MHz}$ radiates $\mathcal P = \qty{100}{kW}$
from a quarter-wave mast on a conducting ground ($R_{\text{rad}} = \qty{36}{\ohm}$,
pattern of a vertical dipole over the upper half-space, as in
\cref{exo:b2:dipole-radiation:11}).
\begin{enumerate}
  \item Wavelength and height of the mast; current amplitude at its
        base; voltage across $R_{\text{rad}}$.
  \item Equivalent dipole moment amplitude, treating the mast as a
        short dipole with $p_0 = I_0\ell_{\text{eff}}/\omega$, $\ell_{\text{eff}} \approx
        \lambda/2\pi$ for the quarter-wave mast (admitted).
  \item Mean intensity at \qty{10}{km} and \qty{100}{km} in the horizontal
        plane (take the upper-hemisphere pattern, maximal at the
        horizon, $\langle\Pi\rangle = 1.5\mathcal P/2\pi r^2$); field amplitudes.
  \item Emf in a \qty{1}{m} vertical whip at \qty{100}{km}; in a ferrite
        rod antenna (a coil of $N = 100$ turns, \qty{1}{cm^2}, effective
        permeability $100$) facing the wave's $\vect B$: compare.
  \item Power received by a matched receiver with an antenna of
        effective area $A = 1.5\lambda^2/4\pi$ (the dipole's) at \qty{100}{km}.
  \item The mast's conductors have a total loss resistance of
        \qty{2}{\ohm}: efficiency; power lost as heat.
  \item Is a receiver at \qty{10}{km} in the \omterm{thm:b2:dipole-radiation:field}{radiation zone}? One at
        \qty{100}{m} from the mast?
  \item Why do medium-wave signals reach farther by night? (Recall the
        D layer of \cref{ch:b2:waves-in-media}.)
\end{enumerate}

\medskip
\noindent\textbf{Part II --- The sky.}
Air at sea level: $N = \qty{2.5e25}{m^{-3}}$, Rayleigh cross-section $\sigma =
\qty{4.6e-31}{m^2}\,(550\,\text{nm}/\lambda)^4$; sunlight \qty{1.0}{kW/m^2} at the
ground; the atmosphere's equivalent thickness at sea-level density is
\qty{8}{km}.
\begin{enumerate}[resume]
  \item Cross-section at \qty{450}{nm} and \qty{650}{nm}; ratio.
  \item Mean free path of the three colours; vertical optical depth
        $\tau = N\sigma H$; fraction of each colour scattered out of the
        direct beam at the zenith.
  \item Power scattered per cubic metre of air at the ground by the
        green light (take \qty{300}{W/m^2} of sunlight around
        \qty{550}{nm}); into what solid angle, with what pattern?
  \item A column of air of \qty{1}{m^2} section viewed at \qty{90}{\degree} from
        the Sun: estimate the intensity of skylight arriving at the
        observer from that column (power scattered toward him per
        unit solid angle, over the \qty{8}{km}), and compare with the
        direct Sun; is the sky's brightness of the right order
        (about \qty{1e-5}{} of the Sun's)?
  \item Degree of polarization of the light from that column; why is
        it lower in reality, and how do some insects use it?
  \item At sunset the path is $38$ atmospheres: transmitted fractions
        of blue and red; colour of the Sun; why the sky overhead stays
        blue.
  \item Why is the sky black on the Moon even in full daylight?
  \item On Mars the air is a hundred times thinner but full of fine
        dust (grains of a micrometre): why is the Martian sky
        butterscotch and its sunset blue?
\end{enumerate}

\medskip
\noindent\textbf{Part III --- The electron.}
\begin{enumerate}[resume]
  \item Thomson cross-section from $r_e$.
  \item In an X-ray tube an electron of \qty{100}{keV} ($v = \qty{1.6e8}{m/s}$,
        treat it non-relativistically) is stopped over \qty{2}{\micro m} in
        tungsten: acceleration, Larmor power, duration, energy
        radiated and its fraction of the kinetic energy.
  \item Shortest wavelength the tube can emit (a photon taking the
        electron's whole energy).
  \item The tube runs at \qty{20}{mA}: radiated power; heat in the anode;
        why the anode rotates.
  \item A classical electron circling a proton at the Bohr radius
        (\qty{5.3e-11}{m}, speed \qty{2.2e6}{m/s}): acceleration, Larmor
        power, and the time to radiate away its \qty{13.6}{eV} --- the
        classical atom's lifetime.
  \item Real atoms last forever: name what classical physics misses
        here, and what \cref{ch:b2:schrodinger-wave-functions} will say
        about a stationary state.
  \item An electron in a synchrotron of radius \qty{100}{m} at $v \approx c$:
        the non-relativistic Larmor formula gives what power? (The
        true power is $\gamma^4$ times larger, with $\gamma \sim 10^4$: the
        synchrotron light source.)
  \item The same electron at rest in the \qty{1}{kW/m^2} of sunlight:
        power it re-radiates (Thomson); number of such electrons
        needed to scatter a watt.
  \item Summarize the three antennas in a table: dipole moment,
        frequency, power, pattern.
\end{enumerate}
\end{problem}
