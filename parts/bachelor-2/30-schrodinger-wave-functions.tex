\chapter{The Schrödinger Equation and Wave Functions}\label{ch:b2:schrodinger-wave-functions}

Fire electrons one at a time at a pair of slits and record where each
lands: a dot here, a dot there, apparently at random --- and after ten
thousand dots, the fringes of Young's experiment, as if each electron
had passed through both slits and interfered with itself. The last
chapter of the Year 1 volume met this strangeness: light comes in
photons, matter has a wavelength $\lambda = h/p$, and position and
momentum cannot both be sharp. What it did not give is the
\emph{equation} --- the rule that says how the wave associated with a
particle evolves, as Newton's law says how a trajectory evolves. That
equation is Schrödinger's (1926), and with it the quantum description
becomes a calculable theory: one writes a \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{wave function}}
$\psi(x,t)$, whose squared modulus is the probability of finding the
particle, and one solves a linear partial differential equation. This
chapter states the equation, explains what $\psi$ means and what it
does not, finds its \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states} and the free particle's \omterm{prop:b2:dispersion-wave-packets:group}{wave
packets}, and shows where Newton's mechanics re-emerges --- for
everything heavier than a molecule.

\section{What the Year 1 volume left us}

\begin{proposition}[Particles are waves]\label{prop:b2:schrodinger-wave-functions:recap}
A photon of frequency $\nu$ has the energy $E = h\nu = \hbar\omega$ and the
momentum $p = h/\lambda = \hbar k$; a material particle of momentum $p$
behaves as a wave of wavelength $\lambda = h/p$ (de Broglie), with $E =
\hbar\omega$ and $p = \hbar k$ as well --- electron \omterm{def:b2:diffraction:huygens}{diffraction} by
crystals, neutron and even molecule interferometry confirm it; and no
state can have a position and a momentum sharper than $\Delta x\,\Delta p
\ge \hbar/2$ (Heisenberg). With $\hbar = h/2\pi = \qty{1.05e-34}{J.s}$: an
electron at \qty{100}{eV} has $\lambda = \qty{0.12}{nm}$, the size of an
atom; a dust grain of \qty{1e-15}{kg} at \qty{1}{mm/s} has $\lambda =
\qty{7e-16}{m}$, smaller than a nucleus --- it behaves classically.
\end{proposition}

\begin{proof}
Recalled from the Year 1 volume; the task now is the dynamics of the
wave.
\end{proof}

\section{The wave function}

\begin{definition}[Wave function and Born's rule]\label{def:b2:schrodinger-wave-functions:psi}
The state of a particle moving along $x$ is described by a complex
\emph{wave function}\index{wave function} $\psi(x,t)$ such that
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
is the probability of finding the particle between $x$ and $x + \dd x$
at the time $t$ (\emph{Born's rule}\index{Born's rule}); hence the
\emph{normalisation} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, and the
expectation values $\langle x\rangle = \int x|\psi|^2\dd x$, $\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$. Wave functions obey the
\emph{superposition principle}\index{superposition principle}: if
$\psi_1$ and $\psi_2$ are possible states, so is $\alpha\psi_1 +
\beta\psi_2$ (normalised), and the probabilities then contain the
interference term $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$. A
global phase factor $\eu^{\iu\theta}$ changes nothing; a relative phase
between two superposed components changes the interference.
\end{definition}

\begin{remark}[What $\psi$ is not]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ is not a classical field spread in space like $\vect E$, nor a
cloud of matter: the electron is always found whole, at one point.
$\psi$ is an \emph{amplitude of probability}, the object that
interferes; only $|\psi|^2$ is observed, and only statistically, by
repeating the experiment on identically prepared particles. The dots
of the double slit are single electrons; the fringes are $|\psi_1 +
\psi_2|^2$.
\end{remark}

\section{The Schrödinger equation}

\begin{theorem}[Schrödinger equation]\label{thm:b2:schrodinger-wave-functions:equation}
A particle of mass $m$ in the potential energy $V(x)$ has a \omterm{def:b2:schrodinger-wave-functions:psi}{wave
function} obeying
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
It is linear (superposition), first order in time (the \omterm{def:b2:schrodinger-wave-functions:psi}{wave function}
at one instant determines it at all later times), and complex ($\psi$
cannot be real: the $\iu$ is essential). It is a \emph{postulate},
justified by its consequences.
\end{theorem}

\begin{proof}
A motivation rather than a proof. A free particle of sharp momentum
$p = \hbar k$ and energy $E = \hbar\omega$ should be a plane wave $\psi =
A\eu^{\iu(kx - \omega t)}$, for which $\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ and $-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$.
The classical relation $E = p^2/2m$ is thus reproduced by the equation
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$, and the natural
generalisation for $E = p^2/2m + V$ adds $V\psi$. Notice that the
equation must be linear to hold for superpositions of plane waves, and
first order in time with an $\iu$ so that $\eu^{-\iu\omega t}$ and not
$\cos\omega t$ is the time dependence --- a real \omterm{thm:b2:waves-on-strings:equation}{wave equation} would
give $E^2 \propto p^4$.
\end{proof}

\begin{proposition}[Conservation of probability]\label{prop:b2:schrodinger-wave-functions:current}
The probability density $\rho = |\psi|^2$ and the \emph{probability
current}\index{probability current}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
obey $\partial_t\rho + \partial_xj = 0$: probability is neither created nor
destroyed, and $\int|\psi|^2\dd x$ stays equal to $1$. For a plane wave
$A\eu^{\iu kx}$, $j = |A|^2\hbar k/m = \rho v$: density times velocity, as for
any flow.
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$; from the
equation and its conjugate, $\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$, $\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$; the $V$ terms cancel and the rest is $(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$.
\end{proof}

\begin{proposition}[Momentum, expectation values and the classical limit]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
The momentum of a state is obtained by the operator $\hat p =
-\iu\hbar\,\partial_x$ (which gives $\hbar k$ on a plane wave): $\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, and the energy by $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$. The expectation values obey
(Ehrenfest)
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
the centre of the \omterm{prop:b2:dispersion-wave-packets:group}{wave packet} follows Newton's law exactly when $V$ is
at most quadratic, and approximately whenever the packet is narrow
compared with the scale over which $V$ varies --- the classical limit,
which holds for any macroscopic object because $\hbar$ is so small.
\end{proposition}

\begin{proof}
Differentiate $\langle x\rangle = \int x|\psi|^2$, use $\partial_t\rho =
-\partial_xj$ and integrate by parts: $\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$. The second relation is the same computation one
order higher (admitted here).
\end{proof}

\section{Stationary states}

\begin{theorem}[Stationary states]\label{thm:b2:schrodinger-wave-functions:stationary}
If $V$ does not depend on time, the equation has solutions of the
separated form
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(the \emph{time-independent Schrödinger equation}\index{Schrödinger
equation!time-independent}): the \emph{stationary states}\index{stationary
state}, of well-defined energy $E$, whose probability density
$|\varphi|^2$ does not move. The energies $E$ for which $\varphi$ is
acceptable (bounded, normalisable for a bound state) are the
\emph{energy levels}; the general solution is a superposition $\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$, and a superposition of two levels
has a density that oscillates at the \emph{Bohr frequency}\index{Bohr
frequency} $\nu_{21} = (E_2 - E_1)/h$ --- the frequency of the light the
atom emits or absorbs.
\end{theorem}

\begin{proof}
Insert $\varphi(x)f(t)$: $\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
a function of $t$ equal to a function of $x$, hence a constant $E$; $f
= \eu^{-\iu Et/\hbar}$. For two levels, $|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$.
Completeness of the \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states} is admitted.
\end{proof}

\section{The free particle: plane waves and wave packets}

\begin{proposition}[Free particle]\label{prop:b2:schrodinger-wave-functions:free}
For $V = 0$ the \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states} are the plane waves $\eu^{\iu(kx -
\omega t)}$ with
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{g}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
matter waves are \emph{dispersive}; the \omterm{def:b2:dispersion-wave-packets:relation}{phase velocity} has no physical
meaning, the \emph{\omterm{prop:b2:dispersion-wave-packets:group}{group velocity}} is the particle's velocity. A plane
wave is not normalisable --- a particle of perfectly sharp momentum is
spread over all space --- so a real particle is a \emph{wave
packet}\index{wave packet}, a superposition of plane waves over a band
$\Delta k$, localised over $\Delta x \sim 1/\Delta k$, moving at $v_{\text{g}}$ and
\emph{spreading}: a Gaussian packet of initial width $\sigma_0$ has at
time $t$ the width
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
and satisfies $\Delta x\,\Delta p = \hbar/2$ at $t = 0$ (the minimum allowed)
and more later.
\end{proposition}

\begin{proof}
The \omterm{def:b2:dispersion-wave-packets:relation}{dispersion relation} follows from the equation; $v_{\text{g}}$ as in
\cref{ch:b2:dispersion-wave-packets}. The spreading formula comes from
the Fourier synthesis of the packet --- each component $\eu^{\iu kx}$ picks
up the phase $\eu^{-\iu\hbar k^2t/2m}$, and the Gaussian integral with a
complex coefficient gives a Gaussian of growing width; the Fourier
integral is the tool of the Year 3 mathematics volume and we admit the
result. Physically: the packet contains momenta spread over $\Delta p
= \hbar/2\sigma_0$, hence velocities spread over $\Delta v = \hbar/2m\sigma_0$,
and after $t$ the faster components have outrun the slower by $\Delta v\,t$
--- the formula says exactly this.
\end{proof}

\begin{example}[Who spreads]\label{ex:b2:schrodinger-wave-functions:spreading}
An electron localised within $\sigma_0 = \qty{1}{nm}$: $\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- after one nanosecond its packet is \qty{60}{\micro m}
wide: an electron left alone does not stay put; bound in an atom it
does not spread because the potential holds it (a \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary state}).
A dust grain of \qty{1e-15}{kg} localised to \qty{1}{\micro m}: $\tau =
\qty{2e7}{s}$, eight months; a bullet: longer than the age of the universe.
The quantum spreading is real, and invisible for anything one can see.
\end{example}

\begin{remark}[Energy and time]\label{rem:b2:schrodinger-wave-functions:energytime}
A state that exists only during $\Delta t$ --- an excited atom of lifetime
$\tau$, a \omterm{prop:b2:scalar-light-model:coherence}{wave train} of length $c\tau$ --- is a superposition of energies
spread over $\Delta E \sim \hbar/\Delta t$: the Fourier relation of
\cref{ch:b2:scalar-light-model} between the duration of a train and its
\omterm{prop:b2:scalar-light-model:coherence}{spectral width}, now applied to $E = \hbar\omega$. An atomic level of
lifetime \qty{10}{ns} has a width $\Delta E = \qty{7e-8}{eV}$, a natural
linewidth of \qty{16}{MHz}; the Doppler width of \cref{ch:b2:laser}
(\qty{1.5}{GHz}) hides it in a gas, but not in a cold atom or an ion at
rest.
\end{remark}

\begin{method}[Working with wave functions]\label{meth:b2:schrodinger-wave-functions:method}
(1) Normalise; probabilities are $\int|\psi|^2$ over the region. (2)
Time-independent $V$: look for $\varphi(x)\eu^{-\iu Et/\hbar}$, solve the
stationary equation, impose the boundary conditions --- the levels.
(3) Superpose \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states} for the evolution; two levels beat at
$(E_2 - E_1)/h$. (4) Free motion: $v_{\text{g}} = p/m$, spreading time
$2m\sigma_0^2/\hbar$. (5) Currents: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$;
for $A\eu^{\iu kx} + B\eu^{-\iu kx}$, $j = (|A|^2 - |B|^2)\hbar k/m$ (no cross term).
(6) Classical limit: Ehrenfest, and $\lambda$ against the size of the
apparatus.
\end{method}

\section{Exercises}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:1}
De Broglie wavelengths: an electron at \qty{100}{eV}, \qty{10}{keV},
\qty{100}{keV} (use $p = \sqrt{2mE}$, then say why the last needs a
correction); a thermal neutron (\qty{25}{meV}, $m = \qty{1.67e-27}{kg}$); a
C$_{60}$ molecule ($m = \qty{1.2e-24}{kg}$) at \qty{200}{m/s}; a tennis
ball. Which of these show interference?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:2}
$\psi(x) = A\eu^{-x^2/2a^2}$: $A$ (use $\int\eu^{-u^2}\dd u = \sqrt\pi$);
$\langle x\rangle$, $\Delta x$; probability of finding the particle in
$|x| < a$ ($\operatorname{erf}(1) = 0.84$) and in $|x| < 2a$ ($0.9995$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:3}
Check that $\varphi(x)\eu^{-\iu Et/\hbar}$ solves the equation when $\varphi$
solves the stationary one. Frequency of the phasor for $E = \qty{1}{eV}$;
for a superposition of two levels \qty{2}{eV} apart, frequency and
wavelength of the oscillation of $|\psi|^2$ --- and of the light
emitted.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:4}
Spreading time $\tau = 2m\sigma_0^2/\hbar$ and width after \qty{1}{ns} and
\qty{1}{s}: an electron with $\sigma_0 = \qty{1}{nm}$ and $\sigma_0 = \qty{1}{mm}$;
a proton with $\sigma_0 = \qty{1}{nm}$; a grain of \qty{1e-12}{kg} with
$\sigma_0 = \qty{1}{\micro m}$. Why do objects stay where they are put?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:5}
\emph{Current.} (a) Derive the \omterm{thm:b2:fluid-kinematics:continuity}{continuity equation}. (b) $j$ for
$A\eu^{\iu kx}$, for $A\sin kx$, for $A\eu^{\iu kx} + B\eu^{-\iu kx}$. (c) Why is
the absence of a cross term in the last case the statement that
reflected and incident fluxes simply subtract? (d) Show that for a real
\omterm{def:b2:schrodinger-wave-functions:psi}{wave function} $j = 0$: a bound \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary state} carries no current.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:6}
\emph{A Gaussian packet.} $\psi(x,0) = A\eu^{\iu k_0x}\eu^{-x^2/4\sigma^2}$. (a)
$A$, $\langle x\rangle$, $\Delta x$. (b) $\langle p\rangle$ with $\hat p =
-\iu\hbar\partial_x$. (c) Admitting $\Delta p = \hbar/2\sigma$, check
$\Delta x\Delta p = \hbar/2$. (d) Prove $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$ from the \omterm{thm:b2:fluid-kinematics:continuity}{continuity equation}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:7}
\emph{Electrons in a tube.} Electrons accelerated through \qty{100}{V}
cross \qty{30}{cm}. (a) Speed, wavelength, time of flight. (b) Phase and
group velocities; which is the electron's? (c) A packet initially
\qty{10}{\micro m} wide: spreading time, width at arrival. (d) \omterm{def:b2:diffraction:huygens}{Diffraction}
by a \qty{0.3}{mm} aperture on the way: angular spread and spot on the
screen. Is the ray picture valid?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:8}
\emph{Phases.} (a) Show that $\psi$ and $\eu^{\iu\theta}\psi$ give the same
predictions. (b) For $\psi = (\psi_1 + \eu^{\iu\alpha}\psi_2)/\sqrt2$, write
$|\psi|^2$ and show that $\alpha$ matters. (c) In the double slit, what
are $\psi_1$ and $\psi_2$, and what fixes their relative phase at a point
of the screen? (d) Why does detecting which slit the electron passed
destroy the fringes (qualitative)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:9}
\emph{Ehrenfest.} (a) For $V = \tfrac12m\omega^2x^2$, show that $\langle
x\rangle$ obeys exactly $\ddot{\langle x\rangle} = -\omega^2\langle x\rangle$.
(b) For $V = mgx$, $\langle x\rangle$ falls like a stone. (c) For a general
$V$, when is $\langle V'(x)\rangle \approx V'(\langle x\rangle)$? (d) A
\qty{1}{g} bead in a bowl: estimate the size of its packet needed for
it to be ``quantum'' and compare with an atom.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:10}
\emph{Sloshing.} In a box $0 < x < L$ with impenetrable walls the
\omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states} are $\varphi_n = \sqrt{2/L}\sin(n\pi x/L)$, $E_n =
n^2h^2/8mL^2$ (\cref{ch:b2:potential-wells-tunneling}). Take $\psi =
(\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/\sqrt2$. (a)
$|\psi|^2$ as a function of time. (b) Show $\langle x\rangle = L/2 -
(16L/9\pi^2)\cos\omega_{21}t$ (use $\int_0^L x\sin(\pi x/L)\sin(2\pi x/L)\dd x
= -8L^2/9\pi^2$). (c) Electron, $L = \qty{1}{nm}$: $E_2 - E_1$, the
frequency and wavelength of the oscillation. (d) What does an
oscillating charge do (\cref{ch:b2:dipole-radiation})? Conclude on
spectral lines.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{Spreading, derived.} The packet $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
is the superposition $\int g(k)\eu^{\iu kx}\dd k$ with $g(k) \propto
\eu^{-\sigma_0^2k^2}$ (admitted: the Fourier transform of a Gaussian is a
Gaussian, width $1/2\sigma_0$ in $k$). (a) Write $\psi(x,t)$ by giving each
component its phase $\eu^{-\iu\hbar k^2t/2m}$. (b) Complete the square in
the exponent and admit $\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$ for complex $\alpha$ with positive real part: show
$|\psi|^2$ is a Gaussian of width $\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$. (c) Interpret $\tau$ via $\Delta v = \Delta p/m$. (d) Why does
a packet of photons in vacuum not spread?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{Energy--time.} (a) A state decaying as $\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$
(probability $\eu^{-t/\tau}$) is a superposition of energies: admitting
that its energy distribution is a Lorentzian of full width $\Gamma =
\hbar/\tau$, give $\Gamma$ for $\tau = \qty{10}{ns}$ (atom), \qty{1e-8}{s},
\qty{1e-23}{s} (a hadronic resonance, in MeV). (b) Natural linewidth of
the \qty{10}{ns} level in frequency; compare with the Doppler width
\qty{1.5}{GHz}. (c) How does one see the natural width (cold atoms,
trapped ions)? (d) Relate to the \omterm{prop:b2:scalar-light-model:coherence}{coherence length} of the emitted \omterm{prop:b2:scalar-light-model:coherence}{wave
train}.
\end{exercise}

\section{Problem: The electron in a tube and the spreading of a wave packet}

\begin{problem}[{Weekend problem --- when is an electron a particle,
when a wave}]\label{pb:b2:schrodinger-wave-functions:1}
Data: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $m_{\text{p}} = \qty{1.67e-27}{kg}$.

\textbf{Part I --- The cathode-ray tube.}
Electrons leave a hot cathode with negligible energy, are accelerated
through $U = \qty{20}{kV}$, pass a \qty{0.3}{mm} aperture, and hit a
screen \qty{40}{cm} away.
\begin{enumerate}
  \item Speed and momentum (non-relativistic); de Broglie wavelength.
  \item The kinetic energy is $4\%$ of $mc^2$: is the non-relativistic
        treatment acceptable for this problem?
  \item Time of flight to the screen.
  \item \omterm{def:b2:diffraction:huygens}{Diffraction} by the aperture: angular spread $\sim\lambda/a$ and
        the spot it produces on the screen; compare with a pixel
        (\qty{0.3}{mm}).
  \item Heisenberg at the aperture: $\Delta p_y \gtrsim \hbar/2a$ gives the
        same spread --- check.
  \item A packet \qty{0.3}{mm} wide along the beam: spreading time
        $2m\sigma_0^2/\hbar$ and width at the screen.
  \item Deflection plates apply a transverse force: by Ehrenfest, how
        does $\langle y\rangle$ move? Conclude that the tube is a
        classical device.
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Phase velocity} of the electron wave; why is it no paradox that
        it differs from the electron's speed?
\end{enumerate}

\textbf{Part II --- \omterm{prop:b2:dispersion-wave-packets:group}{Wave packets}.}
\begin{enumerate}[resume]
  \item For a Gaussian packet of width $\sigma_0$: $\Delta p$ and the
        velocity spread $\Delta v$.
  \item Spreading time for an electron with $\sigma_0 = \qty{0.1}{nm}$,
        \qty{1}{nm}, \qty{1}{\micro m}; for a proton with \qty{1}{nm}; for a
        virus (\qty{1e-17}{kg}) with \qty{10}{nm}.
  \item An electron in a hydrogen atom is localised to \qty{0.1}{nm}
        and does not spread: why?
  \item A free electron at \qty{1}{eV} with $\sigma_0 = \qty{1}{nm}$: how far
        does it travel before its width doubles? Compare with the
        wavelength.
  \item In a metal, conduction electrons are waves of $\lambda \approx
        \qty{0.5}{nm}$ over the whole crystal: in what sense are they
        ``delocalised''?
  \item An electron microscope at \qty{100}{keV} has $\lambda \approx
        \qty{4}{pm}$, yet resolves only \qty{0.1}{nm}: why (its magnetic
        lenses accept a half-angle of about \qty{10}{mrad})?
  \item Give the criterion that decides whether an object must be
        treated with a \omterm{def:b2:schrodinger-wave-functions:psi}{wave function} or with Newton.
\end{enumerate}

\textbf{Part III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{Stationary states} and transitions.}
An electron is confined in $0 < x < L = \qty{1}{nm}$; $\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$.
\begin{enumerate}[resume]
  \item $E_1$, $E_2$, $E_3$ in electronvolts; the Bohr frequencies
        $\nu_{21}$, $\nu_{31}$ and the wavelengths.
  \item Check that $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ is time-independent;
        where is the electron most likely in the ground state?
  \item For $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$: $|\psi|^2$ at $t = 0$ and $t = T_{21}/2$; sketch.
  \item $\langle x\rangle(t)$ (given $\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$):
        amplitude of the sloshing.
  \item The \omterm{thm:b2:dipole-radiation:field}{oscillating dipole} $e\langle x\rangle$ radiates: at which
        frequency, and what does that say about spectral lines and
        about the stability of \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states}?
  \item Energy of the state $\psi$: is it $E_1$, $E_2$, or something
        else? What does a measurement give?
\end{enumerate}

\textbf{Part IV --- Interpretation.}
\begin{enumerate}[resume]
  \item In the single-electron double-slit experiment, what is random
        and what is not?
  \item Write the \omterm{prop:b2:schrodinger-wave-functions:current}{probability current} of $A\eu^{\iu kx}$ and relate it to
        a beam of $n$ electrons per unit length at speed $v$.
  \item Why must the Schrödinger equation be complex and first order
        in time (two reasons)?
  \item Summarise in five lines: the \omterm{def:b2:schrodinger-wave-functions:psi}{wave function}, the equation, the
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{stationary states}, the packet, the classical limit.
\end{enumerate}
\end{problem}
