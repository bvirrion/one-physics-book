\chapter{الطاقة الكهرمغناطيسية ومتجهة بوينتنغ}\label{ch:b2:poynting-vector}

تضيء بطارية مصباحًا عبر سلكين نحاسيين. فأين تنتقل
الطاقة؟ لا داخل النحاس، كما يتبيّن: فالإلكترونات
هناك تنساق بميليمتر في الثانية ولا تكاد تحمل شيئًا. بل
تجري الطاقة في \emph{الفضاء حول السلكين}، في المجال
الكهرمغناطيسي، وتدخل شعيرة المصباح من
جوانبها. وتسلّم الشمس كيلوواطًا لكل متر مربع من سطح مشمس
بالطريقة نفسها، عبر مئة وخمسين مليون كيلومتر من
الخلاء. وهذا الفصل يستخرج من \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} ميزان
الطاقة الكهرمغناطيسية --- كم منها يُخزَّن في مجال، و
المتجهة، \omterm{thm:b2:poynting-vector:poynting}{متجهة بوينتنغ}، التي تقول أين تذهب --- ويطبّقه على
سلك ومكثف ووشيعة وحزمة ضوء.

\section{ميزان طاقة المجال}

\begin{theorem}[مبرهنة بوينتنغ]\label{thm:b2:poynting-vector:poynting}
\index{متجهة بوينتنغ}\index{مبرهنة بوينتنغ}\index{كثافة الطاقة الكهرمغناطيسية}
لنعرّف \emph{\omterm{thm:b2:poynting-vector:poynting}{كثافة الطاقة الكهرمغناطيسية}} و\emph{\omterm{thm:b2:poynting-vector:poynting}{متجهة
بوينتنغ}}
\[
u = \frac{\varepsilon_0E^2}2 + \frac{B^2}{2\mu_0} \quad (\unit{J/m^3}) , \qquad
\vect\Pi = \frac{\vect E\wedge\vect B}{\mu_0} \quad (\unit{W/m^2}) .
\]
وتستلزم \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل}، عند كل نقطة،
\[
\frac{\partial u}{\partial t} + \operatorname{div}\vect\Pi = -\vect j\cdot\vect E ,
\]
ومن أجل أي حجم ثابت $V$ يحدّه $\Sigma$،
\[
\frac{\dd}{\dd t}\iiint_Vu\,\dd\tau = -\iint_\Sigma\vect\Pi\cdot\vect n\,\dd S - \iiint_V\vect j\cdot\vect E\,\dd\tau :
\]
أي إن طاقة المجال داخل $V$ تتناقص بما يجري خارجًا عبر
الحدّ --- أي تدفق $\vect\Pi$ --- وبما يعطيه المجال
للشحن ($\vect j\cdot\vect E$، أي استطاعة جول في ناقل). و $\vect\Pi$
هي كثافة تدفق الطاقة: أي الاستطاعة العابرة لواحدة سطح ناظمية
عليها.
\end{theorem}

\begin{proof}
من ماكسويل--أمبير، $\vect j = \operatorname{\vect{curl}}\vect B/\mu_0 - \varepsilon_0\partial_t
\vect E$، ومنه $\vect j\cdot\vect E = \vect E\cdot\operatorname{\vect{curl}}\vect B/\mu_0 -
\partial_t(\varepsilon_0E^2/2)$. وتعطي المتطابقة $\operatorname{div}(\vect E\wedge\vect B) = \vect B
\cdot\operatorname{\vect{curl}}\vect E - \vect E\cdot\operatorname{\vect{curl}}\vect B$ (وتحقق منها
على المركّبات) وماكسويل--فاراداي $\vect E\cdot\operatorname{\vect{curl}}
\vect B = -\operatorname{div}(\vect E\wedge\vect B) - \vect B\cdot\partial_t\vect B = -\mu_0
\operatorname{div}\vect\Pi - \partial_t(B^2/2)$. ونجمع: $\vect j\cdot\vect E = -\operatorname{div}
\vect\Pi - \partial_tu$. ثم نكامل على $V$ ونطبّق مبرهنة \omterm{def:b2:maxwell-equations:operators}{التباعد}.
\end{proof}

\begin{remark}[ما يتحدد وما لا يتحدد]\label{rem:b2:poynting-vector:interpretation}
لا تحدّد المبرهنة إلا تدفق $\vect\Pi$ عبر السطوح المغلقة
والمقدار الكلي $u$؛ وإضافة أي مجال معدوم \omterm{def:b2:maxwell-equations:operators}{التباعد} إلى $\vect\Pi$ لم تكن
لتغير شيئًا يُرصد. والاختيار أعلاه هو الأبسط، وهو
الذي يتوافق مع كمية الحركة التي يحملها الضوء (أدناه)، وهو
المستعمل عالميًّا. والكثافتان $\varepsilon_0E^2/2$ و $B^2/2\mu_0$ هما
طاقتا المكثف والمحرّض في كتاب السنة الأولى،
منسوبتين الآن إلى كل نقطة من الفضاء: فمجال مقداره \qty{1}{MV/m} (وهو
حدّ انهيار الهواء) يخزّن \qty{4.4}{J/m^3}، ومجال مقداره \qty{1}{T}
يخزّن \qty{0.4}{MJ/m^3} --- أي أكثر بمئة ألف مرة، ولهذا يبلغ
تخزين الطاقة في المغانط، لا في المكثفات، الميغاجول.
\end{remark}

\section{أين تجري الطاقة: ثلاث دارات}

\begin{example}[السلك المقاوم]\label{ex:b2:poynting-vector:wire}
سلك أسطواني نصف قطره $a$ يحمل التيار المستقر $I$ تحت
المجال $E = j/\gamma$ على امتداد محوره؛ وخارجه مباشرةً $B = \mu_0I/2\pi a$
(سمتي). وعند السطح تشير $\vect\Pi = \vect E\wedge\vect B/\mu_0$
\emph{قطريًّا نحو الداخل}، وطويلتها $EI/2\pi a$؛ وتدفقها عبر
السطح الجانبي لطول $L$ هو $EI/2\pi a\times2\pi aL = (EL)I = UI =
RI^2$: فحرارة جول تدخل السلك من جوانبه، يسلّمها
المجال المحيط، ولا تحملها الإلكترونات نزولًا في السلك.
وفي الداخل، $\Pi(r) = (E/\mu_0)(\mu_0Ir/2\pi a^2) = EIr/2\pi a^2$: فالجريان الداخل
يتناقص نحو المحور كلما أُودعت الطاقة طبقةً طبقةً.
\end{example}

\begin{example}[المكثف والملف اللولبي]\label{ex:b2:poynting-vector:capsol}
مكثف بلبوسين دائريين يُشحن: $\vect E$ محوري، و $\vect B =
\mu_0\varepsilon_0(r/2)\dot E\,\vect e_\theta$ (\cref{ch:b2:maxwell-equations})، ومنه
$\vect\Pi = -\varepsilon_0(r/2)E\dot E\,\vect e_r$، نحو الداخل؛ وعبر الأسطوانة ذات
نصف القطر $R$ والارتفاع $d$ يكون تدفقها $\varepsilon_0\pi R^2d\,E\dot E = \dd(\tfrac12
\varepsilon_0E^2\,\pi R^2d)/\dd t$: فالطاقة الكهربائية تدخل من الحافة. وملف
لولبي ينمو تياره: $\vect B$ محوري، والمجال المحرَّض $\vect E = -(r/2)
\dot B\,\vect e_\theta$، و $\vect\Pi = -(rB\dot B/2\mu_0)\vect e_r$، نحو الداخل مرة أخرى، وتدفقها
$\dd(B^2/2\mu_0\times\text{الحجم})/\dd t$. وفي كل حالة تبلغ الطاقة
الجهازَ عبر المجال خارجه؛ ولا تفعل الأسلاك إلا أن توجّه
المجال.
\end{example}

\begin{remark}[الطاقة في مجالين متعامدين ساكنين]\label{rem:b2:poynting-vector:static}
مكثف مشحون موضوع في مجال مغنطيس دائم له
$\vect\Pi = \vect E\wedge\vect B/\mu_0 \ne \vect 0$ في كل مكان بين لبوسيه،
مع أن لا شيء يتغير ولا شيء يسخن. ولا تناقض في ذلك:
إذ إن $\vect\Pi$ هناك معدومة \omterm{def:b2:maxwell-equations:operators}{التباعد}، وخطوطها تنغلق على نفسها،
وتدفقها عبر أي سطح مغلق ينعدم --- ولا تقول المبرهنة
شيئًا عن جريان دوّار لا يسلّم شيئًا. فالتدفقات
عبر السطوح المغلقة وحدها فيزيائية.
\end{remark}

\section{الضوء: الشدة وضغط الإشعاع}

\begin{proposition}[طاقة موجة مستوية]\label{prop:b2:poynting-vector:wave}
\index{شدة (كهرمغناطيسية)}\index{ضغط الإشعاع}\index{استضاءة}
من أجل موجة مستوية في الخلاء --- وهي حلّ \cref{ch:b2:plane-waves-polarization}،
مع $\vect B = \vect k\wedge\vect E/\omega$، و $B = E/c$، و $\vect E \perp \vect B$ ---
تتساوى كثافتا الطاقة الكهربائية والمغناطيسية ويكون
\[
u = \varepsilon_0E^2 , \qquad \vect\Pi = \varepsilon_0cE^2\,\vect u = cu\,\vect u ,
\]
حيث $\vect u$ جهة الانتشار: فالطاقة تنتقل بالسرعة $c$.
ومن أجل موجة جيبية سعتها $E_0$ تكون \emph{الشدة}
(الاستضاءة) هي التدفق الوسطي،
\[
I = \langle\Pi\rangle = \frac{\varepsilon_0cE_0^2}2 = \frac{E_0^2}{2\mu_0c} ,
\]
وتحمل الموجة كثافة كمية حركة $\vect\Pi/c^2$ (مسلَّم بها)، ومنه
تؤثر على سطح تصيبه بورود ناظمي بما يسمّى \emph{\omterm{prop:b2:poynting-vector:wave}{ضغط
الإشعاع}} $P = I/c$ إذا امتُصّت و $2I/c$ إذا انعكست.
\end{proposition}

\begin{proof}
$B^2/2\mu_0 = E^2/2\mu_0c^2 = \varepsilon_0E^2/2$؛ و $\vect E\wedge\vect B/\mu_0 = E^2/\mu_0c =
\varepsilon_0cE^2$ على امتداد $\vect k$؛ و $\langle\cos^2\rangle = \tfrac12$. وتنتج كثافة
كمية الحركة $\vect\Pi/c^2$ من قوة لورنتس على شحن
السطح الماصّ (أو من النسبية: $p = E/c$ في الضوء): إذ إن
كمية الحركة $\Pi S\,\dd t/c^2\times c$ الواصلة في واحدة الزمن على سطح $S$ هي
قوة $\Pi S/c$.
\end{proof}

\begin{example}[ضوء الشمس والليزر والموجات الصغرية]\label{ex:b2:poynting-vector:orders}
ضوء الشمس عند قمة الغلاف الجوي، $I = \qty{1.36}{kW/m^2}$: $E_0 =
\sqrt{2I/\varepsilon_0c} = \qty{1.0}{kV/m}$، و $B_0 = E_0/c = \qty{3.4}{\micro T}$، وكثافة
الطاقة \qty{4.5}{\micro J/m^3}، والضغط \qty{4.5}{\micro Pa} --- أي قوة قدرها ثقل \qty{0.7}{kg}
على كيلومتر مربع، ومع ذلك تكفي لتوجيه شراع شمسي ول
نفخ غبار مذنَّب في ذيله. ومؤشر ليزر مقداره \qty{1}{mW} على
بقعة \qty{1}{mm^2}: \qty{1}{kW/m^2}، أي كالشمس؛ ومركَّزًا على \qty{1}{\micro m^2}:
\qty{1}{GW/m^2}، و $E_0 = \qty{0.9}{MV/m}$، أي قرب انهيار الهواء. وفرن
موجات صغرية مقداره \qty{1}{kW} \omterm{prop:b2:waves-on-strings:modes}{بموجة مستقرة} تملأ $\qty{30}{L}$: كثافة
الطاقة $\sim 10^{-7}$ J/m$^3$ و $E_0$ بضعة كيلوفولتات في المتر.
\end{example}

\begin{method}[محاسبة الطاقة]\label{met:b2:poynting-vector:method}
(1) أوجد $\vect E$ و $\vect B$ (سكونًا، أو شبه سكون، أو أمواجًا). (2)
كوّن $\vect\Pi = \vect E\wedge\vect B/\mu_0$ و $u$. (3) اختر سطحًا مغلقًا
حول الجهاز واحسب تدفق $\vect\Pi$: فيجب أن
يساوي استطاعة جول مضافًا إليها معدل تغيّر الطاقة المخزَّنة
في الداخل --- وهو تحقق من المجالات. (4) وفي موجة، يكون متوسط $\Pi$
هو الشدة؛ فاقسمه على $c$ للحصول على الضغط، وعلى طاقة الفوتون
$hf$ للحصول على تدفق الفوتونات.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:1}
ضوء الشمس عند الأرض، $I = \qty{1.0}{kW/m^2}$: أوجد السعتين $E_0$ و $B_0$،
وكثافة الطاقة، \omterm{prop:b2:poynting-vector:wave}{وضغط الإشعاع} على سطح أسود وعلى مرآة؛
والقوة على سطح مساحته \qty{100}{m^2}؛ وتدفق الفوتونات في المتر المربع من أجل
طاقة فوتون وسطية \qty{2}{eV}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:2}
مؤشر ليزر مقداره \qty{5}{mW} وقطر حزمته \qty{1.5}{mm}: أوجد الشدة و $E_0$
و $B_0$؛ والحزمة نفسها مركَّزةً على بقعة قطرها \qty{5}{\micro m}؛ وليزر نبضي
مقداره \qty{1}{PW} (أي \qty{1e15}{W}) مركَّز على \qty{10}{\micro m^2}: أوجد $E_0$، وقارنه
بالمجال الذي يربط الإلكترون في ذرة الهيدروجين (\qty{5e11}{V/m}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:3}
كثافات الطاقة: مجال كهربائي مقداره \qty{3}{MV/m} (انهيار الهواء)؛
و \qty{100}{MV/m} (عازل رقيق)؛ ومجال مغناطيسي مقداره \qty{0.1}{T}،
و \qty{1.5}{T} (تصوير بالرنين المغناطيسي)، و \qty{45}{T} (مغنطيس قياسي). وأوجد الطاقة في \qty{1}{m^3} من
فجوة جهاز الرنين؛ وما يعادلها بلترات البنزين (\qty{35}{MJ/L}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:4}
شراع شمسي \qty{100}{m} $\times$ \qty{100}{m}، عاكس عكسًا تامًّا، على
بعد الأرض ($I = \qty{1.36}{kW/m^2}$): أوجد القوة؛ وتسارع مركبة
كتلتها \qty{200}{kg}؛ والسرعة المكتسبة في شهر؛ وقارنها بثقالة الشمس
على المركبة على ذلك البعد (\qty{5.9e-3}{N/kg}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:5}
سلك نحاسي نصف قطره $a = \qty{1}{mm}$ يحمل \qty{10}{A} ($\gamma =
\qty{6e7}{S/m}$). (أ) أوجد $E$ في الداخل، و $B$ عند السطح، و $\vect\Pi$ عند
السطح (جهةً وطويلةً). (ب) وتدفق $\vect\Pi$ إلى \qty{1}{m}
من السلك؛ وقارنه بالمقدار $RI^2$. (ج) و $\Pi(r)$ داخل السلك والاستطاعة
المودَعة بين $r$ و $r + \dd r$؛ وتحقق من أنها $j^2/\gamma\cdot2\pi r\,\dd r$.
(د) ومن أين تأتي الطاقة --- صف المجال بين
البطارية والسلك.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:6}
مكثف بلبوسين دائريين ($R$، و $d$) يُشحن بالتيار $I$. (أ) أوجد $\vect E$ و $\vect B$
و $\vect\Pi$ بين اللبوسين. (ب) وتدفق $\vect\Pi$ عبر الأسطوانة
الجانبية؛ وبيّن أنه يساوي $\dd W_e/\dd t$. (ج) ومن أجل $R = \qty{5}{cm}$ و $d =
\qty{1}{mm}$ و $I = \qty{1}{A}$ في اللحظة التي فيها $E = \qty{1}{MV/m}$: أوجد $\Pi$ عند
الحافة والاستطاعة الداخلة. (د) وهل الطاقة المغناطيسية مهمَلة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:7}
ملف لولبي طويل ($n$، ونصف قطره $R$) ينمو تياره $I(t)$. (أ)
أوجد $\vect E$ المحرَّض في الداخل؛ و $\vect\Pi$؛ وجهتها. (ب) والتدفق عبر
السطح الجانبي في واحدة الطول و $\dd(B^2/2\mu_0\cdot\pi R^2)/\dd t$. (ج)
وخارج الملف مباشرةً (حيث $\vect B = \vect 0$)، ما $\vect\Pi$؟
فكيف تدخل الطاقة إذًا؟ (فكّر أين يجري التيار:
فاللفّ صفيحة رقيقة؛ وخذ السطح داخلها مباشرةً.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:8}
\emph{أين تنتقل استطاعة كابل.} \omterm{prop:b2:dispersion-wave-packets:coax}{كابل محوري} (نصفا قطريه $a$
و $b$) يحمل تيارًا مستقرًّا $I$ في اللبّ، عائدًا في الضفيرة، تحت
التوتر $U$ بينهما (والناقلان مثاليان). (أ) أوجد $\vect E$ و $\vect B$
في العازل ($a < r < b$) من غاوس وأمبير. (ب) و $\vect\Pi$:
جهةً وطويلةً. (ج) وكامِل $\vect\Pi$ على الحلقة $a < r
< b$: بيّن أن النتيجة $UI$ --- فالاستطاعة كلها تنتقل في
العازل. (د) ومع لبّ مقاوم قليلًا، في أي جهة تميل
$\vect\Pi$، وما الذي يساويه تدفقها إلى اللبّ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:9}
\emph{غبار المذنَّبات.} حبة غبار كروية نصف قطرها $a$ وكثافتها
$\rho = \qty{2000}{kg/m^3}$ على البعد $D$ من الشمس تمتصّ
ضوء الشمس ($I = \qty{1.36}{kW/m^2}$ عند $D_0 = \qty{1.5e11}{m}$، وكتلة الشمس
\qty{2.0e30}{kg}). (أ) أوجد قوة الإشعاع وقوة الثقالة؛ وبيّن أن
نسبتهما لا تتوقف على $D$. (ب) ونصف القطر الذي تُنفَخ الحبة دونه
بعيدًا. (ج) ولماذا تشير ذيول المذنَّبات مبتعدةً عن الشمس، ولماذا
تنحني؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:poynting-vector:10}
\emph{مكثف يُفرَّغ.} يُفرَّغ مكثف
\cref{exo:b2:poynting-vector:6} عبر سلك
\cref{exo:b2:poynting-vector:5} ($R = \qty{5.3}{m\ohm}$ في المتر، وطوله
$L$). (أ) أوجد جهة $\vect\Pi$ عند حافة المكثف الآن. (ب) والاستطاعة
المغادرة للمكثف، والاستطاعة الداخلة إلى السلك؛ وماذا يحدث
للفرق (إن وُجد)؟ (ج) واكتب ميزان الطاقة $\dd W_e/\dd t = -RI^2$
وتحقق منه باستعمال $Q = CU$ و $I = -\dd Q/\dd t$. (د) ولماذا لا تظهر
الطاقة المغناطيسية للدارة في الميزان في
البداية وفي النهاية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:poynting-vector:11}
\emph{المغنطيس المشحون.} بين لبوسي مكثف مشحون
($\vect E = E\vect e_z$) يوجد مجال منتظم $\vect B = B\vect e_x$ (أي مغنطيس).
(أ) أوجد $\vect\Pi$؛ وتباعدها. (ب) وخذ السطح المغلق لصندوق
داخل المنطقة: أوجد تدفق $\vect\Pi$. (ج) وأين تذهب خطوط $\vect\Pi$،
وما الذي يغلقها؟ (د) واقتراح: نقيس جريان الطاقة
بماصّ صغير يوضع في الفجوة --- أيسخن؟ ولماذا لا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:poynting-vector:12}
\emph{\omterm{prop:b2:waves-on-strings:modes}{الموجة المستقرة}.} موجتان مستويتان متعاكستا الانتشار متساويتا
السعة تؤلفان $\vect E = 2E_0\cos kx\cos\omega t\,\vect e_y$ و $\vect B = -2(E_0/c)
\sin kx\sin\omega t\,\vect e_z$ (وتحقق من ذلك بماكسويل--فاراداي). (أ) أوجد
$\vect\Pi(x, t)$؛ ومتوسطها الزمني. (ب) وكثافة الطاقة؛ وأين تكون
كهربائية، وأين مغناطيسية، وكيف تتأرجح؟ (ج) وتعكس مرآة ناقلة
عكسًا تامًّا عند $x = 0$ الموجةَ الواردة: أوجد القوة في واحدة
المساحة عليها من حجة كمية الحركة، ومن قوة لابلاس على
تيارها السطحي (باستعمال علاقة العبور من أجل $\vect B$). (د) وبيّن
أن كليهما يعطي $2I/c$ وسطيًّا.
\end{exercise}

\section{مسألة: طاقة خط ولوح وفرن}

\begin{problem}[{مسألة نهاية الأسبوع --- تتبّع الطاقة
الكهرمغناطيسية من خط كهرباء إلى بيت، ومن الشمس إلى لوح شمسي،
ومن مغنطرون إلى كوب ماء}]
\label{pb:b2:poynting-vector:1}

\medskip
\noindent\textbf{الجزء I --- الخط.}
كابل طاقة محوري (نصف قطر لبّه $a = \qty{1.0}{cm}$، ونصف قطر ضفيرته $b =
\qty{3.0}{cm}$، وعازله $\varepsilon_r = 2.3$) يحمل تيارًا مستمرًّا $I =
\qty{500}{A}$ عند $U = \qty{20}{kV}$؛ وناقلية اللبّ $\gamma =
\qty{6e7}{S/m}$.
\begin{enumerate}
  \item المجالان $\vect E(r)$ و $\vect B(r)$ في العازل (بناقلين
        مثاليين أولًا)؛ والقيمتان عند $r = a$.
  \item \omterm{thm:b2:poynting-vector:poynting}{متجهة بوينتنغ} في العازل؛ وتدفقها عبر
        المقطع الحلقي: تحقق من أنه يساوي $UI$.
  \item كثافتا الطاقة $\varepsilon_0\varepsilon_rE^2/2$ و $B^2/2\mu_0$ عند $r = a$
        (وفي عازل تحمل كثافة الطاقة الكهربائية $\varepsilon_r$)؛
        وأيهما تهيمن، وبكم؟
  \item وبأي نسبة من سرعة الضوء «تتحرك» الطاقة
        (بأخذ $\Pi/u$ عند $r = a$)؟ وقارنها بالسرعة $c/\sqrt{\varepsilon_r}$.
  \item والآن صار اللبّ مقاومًا: أوجد المجال $E_z$ داخله؛ وتكتسب
        \omterm{thm:b2:poynting-vector:poynting}{متجهة بوينتنغ} خارج سطحه مباشرةً مركّبةً
        قطرية: أوجد طويلتها، والاستطاعة التي تسلّمها في المتر إلى
        اللبّ؛ وتحقق منها بالمقدار $RI^2$ في المتر.
  \item نسبة المركّبة القطرية إلى المحورية من $\vect\Pi$ عند $r =
        a$: فماذا تقول عن الموضع الذي تذهب إليه الطاقة؟
  \item الطاقة المغناطيسية في المتر من الكابل، $\int B^2/2\mu_0\,\dd\tau$، و
        الطاقة الكهربائية في المتر، $\int\varepsilon_0\varepsilon_rE^2/2\,\dd\tau$؛
        واستعدهما في صورة $\tfrac12\Lambda I^2$ و $\tfrac12\Gamma U^2$ بمحارضة
        وسعة المتر في \cref{ch:b2:dispersion-wave-packets}.
  \item وخط طوله \qty{10}{km}: أوجد الاستطاعة المسلَّمة والاستطاعة الضائعة والنسبة.
        وأعد ذلك مع $U = \qty{200}{kV}$ و $I = \qty{50}{A}$ (والاستطاعة نفسها).
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- اللوح.}
ضوء الشمس ظهرًا في يوم صافٍ: $I = \qty{1.0}{kW/m^2}$؛ ولوح
كهرضوئي مساحته \qty{1.6}{m^2} ومردوده $20\%$؛ وطاقة الفوتون \qty{2.0}{eV}
وسطيًّا.
\begin{enumerate}[resume]
  \item أوجد $E_0$ و $B_0$ وكثافة الطاقة وتدفق الفوتونات في الحزمة.
  \item الاستطاعة الكهربائية المسلَّمة؛ والحرارة الواجب تصريفها؛ وارتفاع
        درجة الحرارة إذا فقد اللوح الحرارة من وجهيه عند $h = \qty{10}{W/(m^2.K)}$.
  \item \omterm{prop:b2:poynting-vector:wave}{ضغط الإشعاع} على اللوح (وهو ماصّ)؛ والقوة؛ وقارنها
        بثقله (\qty{18}{kg}).
  \item الطاقة المتلقاة في اليوم (\qty{6}{h} من الشمس الكاملة المكافئة) وفي
        السنة؛ والمساحة اللازمة لتغطية استهلاك بيت مقداره
        \qty{3500}{kWh/a}.
  \item ويعطي اللوح \qty{30}{V} عند \qty{10.7}{A}: تحقق من الاستطاعة؛
        وأوجد تدفق بوينتنغ على امتداد \qty{5}{m} من الكابل ذي السلكين إلى
        المموِّج --- فأين تنتقل تلك الاستطاعة، في صورة
        المجال؟
  \item وعند مدار الأرض تشعّ الشمس \qty{1.36}{kW/m^2}: أوجد استطاعة
        الشمس الكلية؛ والكتلة التي تحوّلها في الثانية ($E = mc^2$،
        و $c = \qty{3e8}{m/s}$).
  \item عدد الفوتونات التي تصيب اللوح في الثانية.
  \item تدفق كمية حركة الفوتونات عند الأرض: أوجد قوة إشعاع
        الشمس الكلية على الأرض (وقرصها نصف قطره \qty{6400}{km}، وهي ماصّة)؛
        وقارنها بالثقالة (\qty{3.5e22}{N}).
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الفرن.}
يقدّم فرن موجات صغرية \qty{800}{W} عند \qty{2.45}{GHz} إلى جوف حجمه
\qty{30}{L}؛ ويمتصّ كوب فيه \qty{250}{g} من الماء $80\%$ منها.
\begin{enumerate}[resume]
  \item طول الموجة؛ وهل الجوف كبير بالقياس إليه؟ ولماذا تكون
        الجدران معدنية؟
  \item الزمن اللازم لتسخين الماء من \qty{20}{\degreeCelsius} إلى \qty{80}{\degreeCelsius}
        ($c = \qty{4.18}{kJ/(kg.K)}$)؛ والاستطاعة الممتصّة في واحدة الحجم.
  \item ولو جرت الاستطاعة موجةً مستوية عبر مقطع الكوب
        \qty{50}{cm^2}، أوجد الشدة و $E_0$؛ والمجالات الحقيقية
        أمواج مستقرة بسعة مماثلة --- فكم تتباعد البقع الساخنة،
        ولماذا يدور الطبق؟
  \item الطاقة المخزَّنة في مجال الجوف إذا كان عامل جودته
        $Q = 2\pi\times\text{(الطاقة المخزَّنة)}/\text{(الطاقة الضائعة في الدور)}
        \approx 10^3$ وهو فارغ: أوجد الطاقة المخزَّنة، وكثافة الطاقة الوسطى، و $E_0$
        --- وقارنه بانهيار الهواء (\qty{3}{MV/m}).
  \item ويمتصّ الماء لأن جزيئاته ثنائيات قطب مقودة عند
        \qty{2.45}{GHz}: والاستطاعة الموضعية الشبيهة بجول هي $\tfrac12\omega\varepsilon_0
        \varepsilon''E_0^2$ مع $\varepsilon'' \approx 10$ للماء: فأوجد $E_0$ في الماء
        اللازم لكثافة الاستطاعة في السؤال 16؛ وهل هو متوافق؟
  \item طاقة الفوتون عند \qty{2.45}{GHz} بالإلكترون فولت، وعدد الفوتونات
        التي يمتصّها الكوب في الثانية؛ ولماذا تستطيع الموجات الصغرية أن تسخّن ولا
        تستطيع أن تؤيّن (فالتأيين يحتاج نحو \qty{10}{eV}).
  \item ضغط إشعاع مجال الجوف على الجدران (وهو من رتبة
        كثافة الطاقة $u$): أيستحقّ القلق؟
  \item ولماذا يخاطر فرن فارغ (بلا حمل) بالعطب، من حيث
        بوينتنغ؟
  \item لخّص في جدول، من أجل الخط واللوح والفرن: حامل
        الطاقة، وجهة $\vect\Pi$، والاستطاعة، وما الذي
        يحوّلها.
\end{enumerate}
\end{problem}
