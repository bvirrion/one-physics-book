\chapter{ميكانيك الجسم الصلب}\label{ch:b2:rigid-body-mechanics}

انظر إلى عجلة دراجة من حافة الرصيف: تكاد الأسلاك القريبة من
الطريق تسكن فتبدو واضحة المعالم، وأما التي في الأعلى فتنطمس
لأنها تمضي بضعف سرعة الدرّاج. فلكل نقطة من العجلة سرعة تخصّها،
والعجلة مع ذلك جسم واحد --- إذ تحفظ نقاطه المسافات بينها، وهذا
القيد وحده ينظّم حركاتها جميعًا في متجهتين: انسحاب ودوران. وقد
عالج كتاب السنة الأولى الحالة الخاصة لجسم يدور حول محور ثابت؛
أما هذا الفصل فيعطي التحريك العام \omterm{def:b2:rigid-body-mechanics:solid}{لجسم صلب}، وطاقته الحركية
وعزمه الحركي، وقوانين حركته، والقانون الذي يحكم كل عجلة وإطار
ومكبح وسلّم: قانون كولوم في الاحتكاك الجافّ.

\section{تحريك الجسم الصلب}

\begin{definition}[الجسم الصلب]\label{def:b2:rigid-body-mechanics:solid}
\index{جسم صلب}
\emph{الجسم الصلب} (أو الصلب) جملة من النقاط تبقى المسافات
المتبادلة بينها ثابتة: أي إن $\|\vect{AB}\|$ لا يتوقف على الزمن
مهما يكن الزوجان $A$ و $B$ من نقاطه. والمرجع الذي تسكن فيه كل
نقطة من الصلب هو \emph{مرجع مرتبط بالصلب}.
\end{definition}

\begin{theorem}[مجال السرعات في جسم صلب]\label{thm:b2:rigid-body-mechanics:field}
\index{متجهة الدوران}\index{مجال السرعات (جسم صلب)}
في كل لحظة توجد متجهة $\vect\Omega$، هي \emph{\omterm{thm:b2:rigid-body-mechanics:field}{متجهة دوران}} الصلب
(وحدتها \unit{rad/s})، تحقق سرعتا أيّ نقطتين $A$ و $M$ منه في مرجع
$\mathcal R$ العلاقة
\[
\vect v_M = \vect v_A + \vect\Omega\wedge\vect{AM} .
\]
ولا تتوقف $\vect\Omega$ على اختيار $A$؛ وكل متجهة $\vect u$ ثابتة في
الصلب تتطور وفق $\dd\vect u/\dd t = \vect\Omega\wedge\vect u$.
\end{theorem}

\begin{proof}
ليكن $(\vect e_1, \vect e_2, \vect e_3)$ أساسًا متعامدًا نظاميًّا مرتبطًا
بالصلب. من $\vect e_i\cdot\vect e_j = \delta_{ij}$ تحقق المعاملات
$a_{ij} = \dot{\vect e}_i\cdot\vect e_j$ العلاقة $a_{ij} + a_{ji} = 0$: أي إن
مصفوفة التطبيق $\vect u \mapsto \dot{\vect u}$ (من أجل $\vect u$ ثابتة في
الصلب، $\vect u = \sum u_i\vect e_i$ بمعاملات $u_i$ ثابتة) متخالفة
التناظر، وكل مصفوفة $3\times3$ متخالفة التناظر هي مصفوفة التطبيق
$\vect u \mapsto \vect\Omega\wedge\vect u$ حيث $\Omega_1 = a_{23}$ و
$\Omega_2 = a_{31}$ و $\Omega_3 = a_{12}$. ونطبقها على $\vect u =
\vect{AM}$ فنجد $\vect v_M - \vect v_A = \vect\Omega\wedge\vect{AM}$. ولو
قامت $\vect\Omega'$ بالدور نفسه لكان $(\vect\Omega - \vect\Omega')\wedge
\vect{AM} = \vect 0$ من أجل كل $M$، ومنه $\vect\Omega' = \vect\Omega$.
\end{proof}

\begin{remark}[الحركات الثلاث]\label{rem:b2:rigid-body-mechanics:motions}
\index{انسحاب}\index{دوران حول محور ثابت}
\emph{الانسحاب}: $\vect\Omega = \vect 0$، فتشترك النقاط كلها في سرعة
واحدة (وليست الحركة بالضرورة مستقيمة: فمقصورة عجلة الملاهي
تنسحب على دائرة). \emph{\omterm{rem:b2:rigid-body-mechanics:motions}{الدوران حول محور ثابت}} $\Delta$: تسكن نقاط
$\Delta$، ويكون $\vect\Omega = \omega\,\vect e_\Delta$، وتتحرك نقطة تبعد $r$
عن المحور على دائرة بالسرعة $r|\omega|$ --- وهي حالة كتاب السنة
الأولى. أما الحركة العامة فتجمع بينهما:
$\vect v_M = \vect v_G + \vect\Omega\wedge\vect{GM}$، أي انسحاب بسرعة
مركز الكتلة ودوران حول
$G$.
\end{remark}

\begin{definition}[التدحرج دون انزلاق]\label{def:b2:rigid-body-mechanics:rolling}
\index{تدحرج دون انزلاق}\index{سرعة الانزلاق}
يتماسّ صلبان $S_1$ و $S_2$ عند نقطة $I$. و\emph{سرعة انزلاق} $S_1$
على $S_2$ هي $\vect v_{\text{انزلاق}} = \vect v_{I \in S_1}
- \vect v_{I \in S_2}$، أي فرق سرعتي النقطتين الماديتين المنطبقتين
على $I$ في تلك اللحظة. ويقال إن $S_1$ \emph{يتدحرج دون انزلاق} على
$S_2$ إذا كانت $\vect v_{\text{انزلاق}} = \vect 0$. وفي عجلة نصف قطرها
$R$ تتدحرج دون انزلاق على أرض ثابتة، سرعة مركزها $v_G$ وسرعتها
الزاوية $\omega$، يُكتب الشرط $v_G = R\omega$ (وقد اختيرت الإشارات
بحيث تدور العجلة المتدحرجة نحو اليمين في جهة عقارب الساعة).
\end{definition}

\begin{proof}
نطبق \cref{thm:b2:rigid-body-mechanics:field} على العجلة بين $G$
ونقطة التماس $I$: فيكون $\vect v_{I\in S_1} = \vect v_G + \vect\Omega
\wedge\vect{GI}$؛ ومع $\vect v_G = v_G\vect e_x$ و $\vect\Omega = -\omega
\vect e_z$ و $\vect{GI} = -R\vect e_y$ يصير ذلك $(v_G - R\omega)\vect e_x$،
وهو ما يجب أن ينعدم.
\end{proof}

\begin{example}[مجال السرعات في عجلة متدحرجة]\label{ex:b2:rigid-body-mechanics:wheel}
في العجلة المتدحرجة $\vect v_M = \vect\Omega\wedge\vect{IM}$: أي إن
العجلة \emph{تدور في كل لحظة حول نقطة تماسها}. فنقطة التماس
ساكنة، والمركز يمضي بالسرعة $v = R\omega$، والقمة بالسرعة $2v$،
ونقطة من الإطار على ارتفاع $h$ تمضي بالسرعة $\omega\sqrt{2Rh}$
عموديًّا على $\vect{IM}$. ويرسم صمام العجلة منحنى دويريًّا؛ وحجر
ينفلت من المداس يغادر بسرعة نقطة الإطار، حتى $2v$ --- ولهذا كانت
واقيات الطين في الشاحنات ذات شأن.
\end{example}

\section{المقادير الحركية للجسم الصلب}

\begin{proposition}[كمية الحركة والعزم الحركي والطاقة الحركية]\label{prop:b2:rigid-body-mechanics:kinetic}
\index{عزم العطالة}\index{عزم حركي (جسم صلب)}\index{طاقة حركية (جسم صلب)}
من أجل صلب كتلته $M$ ومركز كتلته $G$، في مرجع $\mathcal R$:
\begin{itemize}
  \item كمية حركته $\vect p = M\vect v_G$؛
  \item وإذا دار حول محور $\Delta$ (ثابت في $\mathcal R$، أو مارّ من
        $G$ وثابت الاتجاه) بسرعة زاوية $\omega$، فإن عزمه الحركي
        \emph{على امتداد} ذلك المحور هو
        $L_\Delta = J_\Delta\,\omega$، حيث
        \[
        J_\Delta = \sum_i m_i r_i^2 = \int r^2\,\dd m
        \]
        هو \emph{عزم عطالته} بالنسبة إلى $\Delta$ ($r$ البعد عن
        المحور؛ ووحدته \unit{kg.m^2})؛
  \item وطاقته الحركية هي (كونيغ) $E_k = \tfrac12Mv_G^2 +
        E_k^*$، حيث الطاقة الحركية في المرجع المركزي هي
        $E_k^* = \tfrac12J_{G}\omega^2$ من أجل دوران بالسرعة $\omega$ حول
        محور مارّ من $G$ --- ويكون $E_k = \tfrac12J_\Delta\omega^2$ من أجل
        \omterm{rem:b2:rigid-body-mechanics:motions}{دوران حول محور ثابت} $\Delta$.
\end{itemize}
\end{proposition}

\begin{proof}
بُرهنت مبرهنتا كمية الحركة وكونيغ في كتاب السنة الأولى من أجل
كل جملة. وفي الدوران حول $\Delta$ المارّ من $O$، تمضي نقطة تبعد
$r_i$ عن المحور بالسرعة $r_i\omega$ على دائرة حول المحور؛ وعزمها
الحركي بالنسبة إلى $O$ مسقطًا على $\vect e_\Delta$ هو
$m_ir_i^2\omega$ (وأما المركّبات الأخرى، وهي تتلاشى في صلب متناظر،
فلا حاجة إليها هنا)؛ ثم نجمع. والطاقة الحركية: $\sum\tfrac12m_ir_i^2
\omega^2$.
\end{proof}

\begin{remark}[ما نسلّم به]\label{rem:b2:rigid-body-mechanics:tensor}
من أجل دوران كيفي $\vect\Omega$ لا تكون المتجهة الكاملة $\vect L_G$
موازية $\vect\Omega$ عمومًا: بل $\vect L_G = \mathbf J\,\vect\Omega$
حيث $\mathbf J$ هو \emph{موتر العطالة}، وهو مصفوفة متناظرة متجهاتها
الذاتية هي المحاور الرئيسية. وكل صلب في هذا الفصل يدور حول
محور ثابت أو حول محور تناظر مارّ من $G$، وعنده تكون
$\vect L = J\vect\Omega$ بالضبط؛ وأما الحالة العامة فمن نصيب كتاب
السنة الثالثة.
\end{remark}

\begin{proposition}[عزوم العطالة]\label{prop:b2:rigid-body-mechanics:inertia}
\index{مبرهنة هيغنز}\index{مبرهنة المحاور المتوازية}
من أجل صلب متجانس كتلته $M$، وبالنسبة إلى محور مارّ من $G$: ساق
رفيعة طولها $\ell$ (والمحور عمودي عليها)، $J = \tfrac1{12}M\ell^2$؛ طوق
رفيع أو قشرة أسطوانية نصف قطرها $R$ (حول محورها)، $J = MR^2$؛ قرص
أو أسطوانة ممتلئة، $J = \tfrac12MR^2$؛ كرة ممتلئة، $J = \tfrac25MR^2$؛
قشرة كروية، $J = \tfrac23MR^2$. و\emph{مبرهنة هيغنز} (المحاور
المتوازية): بالنسبة إلى محور $\Delta$ يوازي محورًا $\Delta_G$ مارًّا
من $G$ ويبعد عنه
$d$،
\[
J_\Delta = J_{\Delta_G} + Md^2 .
\]
\end{proposition}

\begin{proof}
الساق: $\int_{-\ell/2}^{\ell/2}x^2\,(M/\ell)\dd x = M\ell^2/12$. والقرص:
حلقات كتلتها $(2M/R^2)r\,\dd r$، فيكون $\int_0^R r^2(2M/R^2)r\,\dd r$ $= MR^2/2$.
والكرة: بالتناظر $J = \tfrac23\int r^2\dd m$ (حيث $r$ البعد عن المركز،
لأن $x^2 + y^2 = \tfrac23(x^2 + y^2 + z^2)$ وسطيًّا على المحاور)،
و $\int r^2\dd m = \int_0^R r^2(3M/R^3)r^2\dd r = \tfrac35MR^2$. وهيغنز: إذا
كانت $\vect r_i$ المتجهة من $\Delta$ إلى $m_i$ عموديًّا على المحور،
و $\vect r_i = \vect d + \vect r_i^*$، فإن $\sum m_ir_i^2 = Md^2
+ 2\vect d\cdot\sum m_i\vect r_i^* + \sum m_ir_i^{*2}$، والمجموع الأوسط
ينعدم بتعريف $G$.
\end{proof}

\begin{example}[دولاب التدوير]\label{ex:b2:rigid-body-mechanics:flywheel}
قرص فولاذي كتلته \qty{50}{kg} ونصف قطره \qty{30}{cm}: $J = \tfrac12 \times
50 \times 0.09 = \qty{2.25}{kg.m^2}$؛ وعند \qty{3000}{rpm} (أي $\omega =
\qty{314}{rad/s}$) يخزّن $\tfrac12J\omega^2 = \qty{111}{kJ}$، وهي الطاقة
الحركية لسيارة صغيرة عند \qty{50}{km/h}. وتُدير دواليب التدوير
المخزِّنة للطاقة دوّارات من ألياف الكربون عند \qty{50000}{rpm} في
الخلاء على محامل مغناطيسية: إذ تنمو الطاقة تناسبًا مع $\omega^2$،
والحدّ هو مقاومة الشدّ في الإطار.
\end{example}

\section{ديناميك الجسم الصلب}

\begin{theorem}[قوانين حركة الجسم الصلب]\label{thm:b2:rigid-body-mechanics:laws}
\index{مبرهنة العزم الحركي (جسم صلب)}\index{استطاعة القوى على جسم صلب}
في مرجع عطالي، ومن أجل صلب تؤثر فيه قوى خارجية $\vect F_i$
مطبَّقة عند النقاط $A_i$:
\begin{enumerate}
  \item (كمية الحركة) $M\,\dd\vect v_G/\dd t = \sum\vect F_i$؛
  \item (العزم الحركي بالنسبة إلى $G$، أو إلى نقطة ثابتة $O$)
        $\dd\vect L_G/\dd t = \sum\vect{GA_i}\wedge\vect F_i$؛ ومسقطًا على
        محور ثابت $\Delta$ أو على محور مارّ من $G$ ثابت الاتجاه
        يدور الصلب حوله بالسرعة $\omega$:
        $J_\Delta\,\dd\omega/\dd t = \mathcal M_\Delta$، وهو مجموع عزوم
        القوى الخارجية بالنسبة إلى المحور؛
  \item (الاستطاعة) استطاعة جملة قوى على صلب هي
        $\mathcal P = \vect R\cdot\vect v_A + \vect{\mathcal M}_A\cdot
        \vect\Omega$ من أجل أي نقطة $A$ من الصلب، حيث $\vect R$ مجموع
        القوى و $\vect{\mathcal M}_A$ عزمها الكلي بالنسبة إلى
        $A$؛ و\emph{القوى الداخلية في صلب معدومة
        الاستطاعة}، ويكون $\dd E_k/\dd t = \mathcal P_{\text{خارجية}}$.
\end{enumerate}
\end{theorem}

\begin{proof}
(1) و (2) هما مبرهنتا كتاب السنة الأولى من أجل جمل النقاط
(ومبرهنة العزم الحركي بالنسبة إلى $G$ صحيحة حتى حين يتسارع $G$،
لأن قوى العطالة في المرجع المركزي معدومة العزم بالنسبة إلى
$G$). و (3): مع $\vect v_{A_i} = \vect v_A + \vect\Omega
\wedge\vect{AA_i}$ نجد $\sum\vect F_i\cdot\vect v_{A_i} = \vect R\cdot\vect v_A
+ \sum\vect F_i\cdot(\vect\Omega\wedge\vect{AA_i}) = \vect R\cdot\vect v_A +
\vect\Omega\cdot\sum\vect{AA_i}\wedge\vect F_i$ (جداء مختلط). وأما القوى
الداخلية فمجموعها وعزمها الكلي معدومان (المبدأ الثالث، وعلى
حامل واحد)، ومنه انعدام استطاعتها على صلب --- \emph{لا} على جملة
قابلة للتشوّه. وعندئذ لا تُبقي مبرهنة الطاقة الحركية إلا الاستطاعة الخارجية.
\end{proof}

\begin{definition}[المفصل المحوري المثالي]\label{def:b2:rigid-body-mechanics:pivot}
\index{مفصل محوري مثالي}\index{محمل}
يدور صلب حول محور ثابت $\Delta$ في \emph{مفصل محوري مثالي} (أي
محمل مثالي) إذا كانت قوى تماس المحمل معدومة العزم بالنسبة إلى
$\Delta$؛ وعندئذ لا تبذل أي استطاعة ($\vect v_A = \vect 0$ على
المحور، و $\mathcal M_\Delta = 0$). وأما المحمل الحقيقي فيضيف عزم
احتكاك $-\Gamma_f\operatorname{sgn}\omega$، أو عزمًا لزجًا $-h\omega$.
\end{definition}

\begin{proposition}[النواس الفيزيائي]\label{prop:b2:rigid-body-mechanics:pendulum}
\index{نواس فيزيائي}\index{نواس مركّب}
يتأرجح صلب كتلته $M$ حول محور أفقي ثابت $\Delta$ في \omterm{def:b2:rigid-body-mechanics:pivot}{مفصل محوري
مثالي}، ومركز كتلته يبعد $d$ عن المحور. وإذا كانت $\theta$ زاوية
$\vect{OG}$ عن الشاقول النازل، فإن
\[
J_\Delta\ddot\theta = -Mgd\sin\theta ,
\]
فيكون دور الاهتزازات الصغيرة $T = 2\pi\sqrt{J_\Delta/Mgd}$ --- وهو
دور نواس بسيط طوله $\ell_{\text{مكافئ}} = J_\Delta/Md$.
\end{proposition}

\begin{proof}
مبرهنة العزم الحركي بالنسبة إلى $\Delta$: عزم الثقل هو
$-Mgd\sin\theta$، وعزم المفصل معدوم. ثم نلجأ إلى التقريب الخطي.
\end{proof}

\begin{example}[المسطرة المترية]\label{ex:b2:rigid-body-mechanics:rule}
مسطرة مترية مثبّتة بمفصل عند أحد طرفيها: $J = \tfrac13M\ell^2$ (بهيغنز من
$\tfrac1{12}M\ell^2$)، و $d = \ell/2$، و $\ell_{\text{مكافئ}} = \tfrac23\ell =
\qty{0.667}{m}$، و $T = \qty{1.64}{s}$ --- فالمسطرة كلها تنوس أبطأ من
كتلة نقطية معلّقة في منتصفها (\qty{1.42}{s}) وأسرع من كتلة معلّقة
عند طرفها (\qty{2.0}{s}).
\end{example}

\section{قوى التماس واحتكاك كولوم}

\begin{definition}[قوة التماس: الناظمية والمماسية]\label{def:b2:rigid-body-mechanics:contact}
\index{رد فعل ناظمي}\index{قوة الاحتكاك}
تنقسم القوة التي يؤثر بها مسند في صلب عند نقطة تماس $I$ إلى
\emph{رد فعل ناظمي} $\vect N$، عمودي على المستوي المماس المشترك
وموجَّه نحو الصلب ($N \ge 0$: فالمسند لا يستطيع إلا الدفع)، وإلى
\emph{مركّبة مماسية} $\vect T$ في المستوي المماس، هي \emph{قوة
الاحتكاك}.
\end{definition}

\begin{theorem}[قانونا كولوم في الاحتكاك الجافّ]\label{thm:b2:rigid-body-mechanics:coulomb}
\index{قانونا كولوم في الاحتكاك}\index{معامل الاحتكاك}\index{احتكاك ساكن}\index{احتكاك حركي}
من أجل صلبين جافّين متماسّين، ومع \emph{معامل احتكاك} $f$ يتوقف
على المادتين وعلى حالة سطحيهما، ولا يتوقف على مساحة التماس ولا
(بتقريب حسن) على السرعة:
\begin{itemize}
  \item إذا انزلق الصلبان أحدهما على الآخر ($\vect v_{\text{انزلاق}} \ne
        \vect 0$)، كانت \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك} معاكسة \omterm{def:b2:rigid-body-mechanics:rolling}{لسرعة الانزلاق}
        وطويلتها $\|\vect T\| = f\|\vect N\|$ (\omterm{thm:b2:rigid-body-mechanics:coulomb}{احتكاك حركي})؛
  \item وإذا لم ينزلقا، فإن $\|\vect T\| \le f\|\vect N\|$ (\omterm{thm:b2:rigid-body-mechanics:coulomb}{احتكاك
        ساكن}): وتكون \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك} عندئذ ما تقتضيه قوانين الحركة
        الأخرى ضمن ذلك الحدّ، وجهتها غير معلومة مسبقًا.
\end{itemize}
و\emph{التماس المثالي} حالة مثالية عندها $f = 0$: أي $\vect T = \vect 0$.
\end{theorem}
\admitted

\begin{remark}[قراءة قانوني كولوم]\label{rem:b2:rigid-body-mechanics:reading}
\index{مخروط الاحتكاك}
القانون الأول معادلة، والثاني متراجحة --- وفي ذلك كل صعوبة
مسائل الاحتكاك. و\emph{الطريقة} أن \emph{نفترض} ألا انزلاق، ونحسب
$\vect T$ و $\vect N$، ثم \emph{نتحقق} من $\|\vect T\| \le f\|\vect N\|$؛
فإن سقط التحقق انزلق الصلب، وكان $\|\vect T\| = f\|\vect N\|$
معاكسًا للانزلاق، وحُلّت المسألة من جديد. وهندسيًّا، على قوة
التماس أن تبقى داخل \emph{\omterm{rem:b2:rigid-body-mechanics:reading}{مخروط الاحتكاك}} ذي نصف الزاوية
$\varphi$ حيث $\tan\varphi
= f$. وقيم نموذجية: المطاط على الإسفلت الجافّ $f \approx 0.8$--$1$،
وعلى الإسفلت المبتلّ $0.4$--$0.6$، وعلى الجليد $0.1$؛ والفولاذ على
الفولاذ $0.15$ جافًّا و $0.05$ مشحَّمًا؛ والمعامل الساكن أكبر
قليلًا في الواقع من الحركي، ولهذا يصعب إيقاف الانزلاق متى بدأ.
\end{remark}

\begin{proposition}[استطاعة قوى التماس؛ التدحرج دون انزلاق]\label{prop:b2:rigid-body-mechanics:power}
\index{احتكاك التدحرج}
استطاعة قوى التماس على صلب يتحرك على مسند ثابت هي
$\mathcal P = \vect T\cdot\vect v_{I\in S}$ (فالقوة الناظمية عمودية
على سرعة نقطة التماس، وهذه تقع في المستوي المماس). وفي
\emph{\omterm{def:b2:rigid-body-mechanics:rolling}{التدحرج دون انزلاق}} تنعدم هذه الاستطاعة: فلا يبذل الاحتكاك
عندئذ أي شغل، وإن كان قد يكون ضروريًّا للحركة. وفي الانزلاق
تكون $-f N v_{\text{انزلاق}} < 0$: أي حرارة.
\end{proposition}

\begin{proof}
$\mathcal P = (\vect N + \vect T)\cdot\vect v_{I\in S}$، و $\vect
v_{I\in S} = \vect v_{\text{انزلاق}}$ تقع في المستوي المماس، فلا تساهم
إلا $\vect T$؛ وهي معدومة حين تنعدم \omterm{def:b2:rigid-body-mechanics:rolling}{سرعة الانزلاق}، وتساوي
$-fNv_{\text{انزلاق}}$ حين ينزلق الصلبان (إذ تكون $\vect T$ معاكسة للمتجهة $\vect
v_{\text{انزلاق}}$). والتدحرج الحقيقي يبدّد قليلًا بتشوّه
الإطار والأرض (\emph{\omterm{prop:b2:rigid-body-mechanics:power}{احتكاك التدحرج}}، ويُنمذَج بعزم صغير
$\Gamma_r = \mu_rNR$ أو بقوة مكافئة $\mu_rN$، مع
$\mu_r \approx 0.01$ لإطار على طريق): وهو أثر آخر أصغر بكثير.
\end{proof}

\begin{method}[صلب يتدحرج على مستو مائل]\label{met:b2:rigid-body-mechanics:incline}
\index{تدحرج على مستو مائل}
صلب متجانس نصف قطره $R$ وكتلته $M$ وعزم عطالته $J =
kMR^2$ حول محوره ($k = 1$ للطوق، و $\tfrac12$ للقرص، و $\tfrac25$ للكرة)
يُترك على مستو مائل بزاوية $\alpha$ ومعامل احتكاك
$f$. ونفترض \omterm{def:b2:rigid-body-mechanics:rolling}{التدحرج دون انزلاق}:
\begin{enumerate}
  \item كمية الحركة على امتداد الميل: $M\dot v = Mg\sin\alpha - T$؛ وناظميًّا:
        $N = Mg\cos\alpha$؛
  \item العزم الحركي بالنسبة إلى $G$: $kMR^2\dot\omega = TR$؛
  \item وشرط اللاانزلاق: $v = R\omega$، ومنه $T = kM\dot v$ و
        \[
        \dot v = \frac{g\sin\alpha}{1 + k} , \qquad T = \frac{k}{1 + k}\,
        Mg\sin\alpha ;
        \]
  \item التحقق: $T \le fN \iff \tan\alpha \le f\,\dfrac{1 + k}{k}$.
        وفيما وراء ذلك الميل ينزلق الصلب، فيكون $T = fMg\cos\alpha$، و $\dot v =
        g(\sin\alpha - f\cos\alpha)$، و $\dot\omega = fg\cos\alpha/kR$.
\end{enumerate}
والتحقق بالطاقة (لا انزلاق، والاحتكاك معدوم الاستطاعة): $\tfrac12(1 + k)Mv^2 =
Mgh$ يعطي $v = \sqrt{2gh/(1 + k)}$ --- فالكرة تسبق القرص، والقرص
يسبق الطوق، مهما تكن الكتلة ونصف القطر.
\end{method}

\begin{example}[لماذا تحتاج السيارة إلى الاحتكاك كي تتسارع]\label{ex:b2:rigid-body-mechanics:car}
سيارة بدفع أمامي: يطبّق المحرك عزمًا على العجلتين الأماميتين؛
والقوة الأفقية الخارجية الوحيدة المؤثرة في السيارة هي احتكاك
الطريق بالإطارات، وهي التي تسارعها. وإذا كان الثقل كله
على العجلتين الدافعتين كان التسارع الأعظمي
$fg \approx \qty{8}{m/s^2}$ على الإسفلت الجافّ --- ونصف ذلك إن لم يكن
على المحور الدافع إلا نصف الثقل؛ وعلى الجليد $\qty{1}{m/s^2}$.
وليس شغل المحرك من فعل الطريق (فنقطة تماسه ساكنة): بل هو شغل
داخلي يحوّله نقل الحركة إلى طاقة حركية --- كما تمدّ عضلات
الماشي الداخلية، لا الأرض، الماشيَ بطاقته.
\end{example}

\begin{example}[الانتقال من الانزلاق إلى التدحرج]\label{ex:b2:rigid-body-mechanics:bowling}
تُطلق كرة بولينغ منزلقة بالسرعة $v_0$ دون دوران. فيبطّئ \omterm{thm:b2:rigid-body-mechanics:coulomb}{الاحتكاك
الحركي} $fMg$ المركز ($\dot v = -fg$) ويدير الكرة
($\tfrac25MR^2\dot\omega = fMgR$، و $\dot\omega = 5fg/2R$) حتى يصير $v = R\omega$:
أي عند $t_1 = 2v_0/7fg$، بالسرعة $v_1 = \tfrac57v_0$. ومن ثَمّ تتدحرج
دون انزلاق بسرعة ثابتة. ويضيع سُبعان من السرعة، و
$1 - (\tfrac57)^2\tfrac75 = \tfrac27$ من الطاقة الحركية، حرارةً في
الانزلاق --- مهما يكن $f$؛ فليس $f$ إلا ما يحدّد سرعة ذلك.
\end{example}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:1}
عزوم العطالة: (أ) ساق طولها \qty{1.0}{m} وكتلتها \qty{0.50}{kg} حول
محور عمودي مارّ من مركزها، ثم من أحد طرفيها؛ (ب) قرص كتلته
\qty{2.0}{kg} ونصف قطره \qty{20}{cm} حول محوره، ثم حول محور مواز
مارّ من إطاره؛ (ج) كرة بلياردو كتلتها \qty{0.16}{kg} وقطرها
\qty{57}{mm}. (د) طوق رفيع وقرص ممتلئ لهما الكتلة ونصف القطر
نفسهما: أيهما يقاوم الإدارة أكثر، وبأي معامل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:2}
دراجة تسير بسرعة \qty{18}{km/h} على عجلتين قطرهما \qty{70}{cm}.
أوجد السرعة الزاوية للعجلتين؛ وسرعة الصمام في المرجع الأرضي
حين يكون في الأعلى، وفي الأسفل، وفي مقدمة العجلة (على ارتفاع
المحور)؛ وسرعة المداس بالنسبة إلى هيكل الدراجة. وينفلت حجر عالق
في المداس عند القمة: بأي سرعة يغادر؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:3}
دولاب تدوير هو قرص فولاذي كتلته \qty{40}{kg} ونصف قطره \qty{25}{cm}.
(أ) الطاقة المخزَّنة عند \qty{6000}{rpm}. (ب) العزم الثابت اللازم
لإدارته من السكون في \qty{2.0}{min}؛ والاستطاعة في نهاية الإدارة.
(ج) وفي حافلة، عليه أن يقدّم \qty{20}{kW} مدة \qty{10}{s}: إلى أي
سرعة يتباطأ؟ (د) ولماذا تُدار دواليب التدوير السريعة في الخلاء؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:4}
قرص متجانس نصف قطره $R$ يتأرجح في مستو شاقولي حول محور أفقي
مارّ من نقطة من إطاره. أوجد \omterm{prop:b2:rigid-body-mechanics:kinetic}{عزم العطالة} بالنسبة إلى المحور؛
ودور الاهتزازات الصغيرة من أجل $R = \qty{20}{cm}$؛ وطول النواس
البسيط المكافئ. والأسئلة نفسها إذا كان المحور على بعد $R/2$ من
المركز. وأين يجب أن يكون المحور ليكون الدور أصغر ما يمكن؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:5}
يُترك طوق وقرص وكرة لها نصف القطر نفسه معًا عند قمة مستو مائل
ارتفاعه \qty{2.0}{m} وميله \qty{20}{\degree}، مع $f =
0.30$. (أ) تحقق من أن الثلاثة تتدحرج دون انزلاق. (ب) سرعة كل منها
في الأسفل؛ وترتيب الوصول. (ج) الزمن الذي يستغرقه كل منها.
(د) بيّن أن \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك} على الكرة هي $\tfrac27Mg\sin\alpha$ وأنها
لا تبذل شغلًا. (ه) عند أي ميل يبدأ الطوق بالانزلاق؟ والكرة؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:6}
\emph{المكبح القرصي.} عجلة وقرص كبحها لهما $J = \qty{1.2}{kg.m^2}$
ويدوران عند \qty{900}{rpm}؛ وتضغط بطانتان على القرص عند نصف قطر
وسطي \qty{12}{cm}، كل واحدة بقوة ناظمية \qty{800}{N}، مع $f = 0.40$.
أوجد عزم الكبح؛ والتباطؤ الزاوي؛ والزمن وعدد الدورات حتى
التوقف؛ والحرارة المحرَّرة، وارتفاع درجة حرارة القرص الفولاذي
ذي الكتلة \qty{1.5}{kg} ($c = \qty{470}{J/(kg.K)}$) إذا احتفظ بها كلها.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:7}
\emph{آلة أتوود ببكرة حقيقية.} كتلتان $m_1 = \qty{2.0}{kg}$
و $m_2 = \qty{1.5}{kg}$ معلّقتان بحبل على بكرة نصف قطرها $R =
\qty{10}{cm}$ وعزم عطالتها $J = \qty{5.0e-3}{kg.m^2}$ على \omterm{def:b2:rigid-body-mechanics:pivot}{مفصل
محوري مثالي}؛ ولا ينزلق الحبل على البكرة. (أ) لماذا يختلف توترا
الحبل؟ (ب) معادلات الكتلتين والبكرة؛ والتسارع. (ج) التوتران.
(د) قارن بالبكرة المعدومة الكتلة؛ ثم استعد التسارع بالطاقة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:8}
\emph{السلّم.} سلّم متجانس طوله $\ell$ وكتلته $m$ يستند بزاوية
$\theta$ عن الأفق إلى جدار شاقولي أملس؛ ومعامل احتكاك الأرض $f$.
(أ) القوى وعزومها بالنسبة إلى القدم؛ ومعادلات التوازن. (ب) بيّن
أن التوازن يقتضي $\tan\theta \ge 1/2f$. (ج) الزاوية الصغرى من أجل
$f = 0.40$. (د) يصعد شخص كتلته $3m$ السلّم الموضوع عند
\qty{60}{\degree}: إلى أي ارتفاع (بنسبة من $\ell$) يستطيع أن يبلغ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:9}
\emph{الدوران الخلفي.} تُضرب كرة بلياردو فتغادر بسرعة مركز
$v_0$ ودوران خلفي $\omega_0$ (أي دوران معاكس للتدحرج). \omterm{thm:b2:rigid-body-mechanics:coulomb}{ومعامل
الاحتكاك} $f$. (أ) معادلتا $v$ و $\omega$ أثناء الانزلاق. (ب) اللحظة
التي تبدأ عندها \omterm{def:b2:rigid-body-mechanics:rolling}{التدحرج دون انزلاق} وسرعتها عندئذ، بدلالة
$v_0$ و $R\omega_0$. (ج) الشرط على
$R\omega_0$ كي تعود الكرة نحو اللاعب. (د) القيم:
$v_0 = \qty{2.0}{m/s}$، و $R\omega_0 = \qty{6.0}{m/s}$، و $f = 0.2$،
و $R = \qty{2.9}{cm}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:10}
\emph{البكرة اللافّة.} بكرة لافّة (كتلتها $M$، وعزم عطالتها $J = kMR^2$
حول محورها، ونصف قطرها الخارجي $R$) موضوعة على طاولة أفقية؛
ويُشدّ أفقيًّا خيط ملفوف على محورها الداخلي ذي نصف القطر $r < R$
بقوة $F$، والخيط يخرج من \emph{أسفل} المحور. (أ) بافتراض
\omterm{def:b2:rigid-body-mechanics:rolling}{التدحرج دون انزلاق}، اكتب معادلتي كمية الحركة والعزم الحركي
وأوجد تسارع المركز؛ وفي أي جهة تتدحرج البكرة؟ (ب) \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك}
اللازمة، والشرط على $F$ كي لا يقع انزلاق. (ج) الأسئلة نفسها إذا
شُدّ الخيط بزاوية $\beta$ فوق الأفق؛ بيّن أن البكرة تبقى ساكنة حين
$\cos\beta = r/R$، وفسّر ذلك هندسيًّا (المحور اللحظي المارّ من
نقطة التماس). (د) وماذا يحدث حين $\cos\beta < r/R$؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:11}
\emph{أين تُضرب كرة البلياردو.} تمنح عصا اللعب دفعًا أفقيًّا
$P$ (أي قوة قصيرة جدًّا) كرةً نصف قطرها $R$ ساكنةً، على ارتفاع
$h$ فوق الطاولة. (أ) كمية الحركة والعزم الحركي بُعيد الضربة
(والاحتكاك مهمَل أثناءها). (ب) بيّن أن الكرة تتدحرج دون انزلاق
فورًا إذا كان $h = \tfrac75R$، وأن لها دورانًا أماميًّا فوق ذلك
ودورانًا خلفيًّا دونه. (ج) من أجل $h = R$ (أي ضربة في المركز)، كم
تنزلق الكرة قبل أن تتدحرج، مع $f = 0.2$ و $v_0 = \qty{3}{m/s}$؟
(د) ولماذا تُبنى حافة طاولة البلياردو على الارتفاع $\tfrac75R$؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:12}
\emph{أسطوانة في تجويف.} أسطوانة ممتلئة نصف قطرها $r$ تتدحرج دون
انزلاق داخل سطح أسطواني ثابت نصف قطره $R > r$، والمحوران أفقيان
متوازيان. ولتكن $\theta$ زاوية مستقيم المركزين عن الشاقول.
(أ) عبّر عن السرعة الزاوية للأسطوانة حول محورها بدلالة
$\dot\theta$ (إشارة: نقطة التماس ساكنة). (ب) الطاقة الحركية
والطاقة الكامنة؛ وبيّن أن دور الاهتزازات الصغيرة هو $T = 2\pi\sqrt{3(R - r)/2g}$.
(ج) قارن بنقطة تنزلق دون احتكاك في التجويف. (د) \omterm{thm:b2:rigid-body-mechanics:coulomb}{معامل الاحتكاك}
الأصغري \omterm{def:b2:rigid-body-mechanics:rolling}{للتدحرج دون انزلاق} عند السعة $\theta_0$
(في الزوايا الصغيرة).
\end{exercise}

\section{مسألة: مركبة على عجلات}

\begin{problem}[{مسألة نهاية الأسبوع --- ما تطلبه السيارة من
إطاراتها: التدحرج والتسارع والكبح، والقوة الواحدة التي تفعل ذلك كله}]
\label{pb:b2:rigid-body-mechanics:1}

سيارة كتلتها $M = \qty{1200}{kg}$ وقاعدة عجلاتها $L = \qty{2.6}{m}$؛
ومركز كتلتها على ارتفاع $h = \qty{0.55}{m}$ وعلى بعدين أفقيين
$a = \qty{1.1}{m}$ عن المحور الأمامي و $b = \qty{1.5}{m}$ عن المحور
الخلفي. ولكل واحدة من العجلات الأربع نصف قطر $R = \qty{0.31}{m}$ وكتلة $m =
\qty{15}{kg}$ وعزم عطالة $J = \qty{1.0}{kg.m^2}$ حول محورها.
\omterm{thm:b2:rigid-body-mechanics:coulomb}{ومعامل الاحتكاك} بين الإطار والطريق $f = 0.80$ (جافًّا)؛ ومعامل
\omterm{prop:b2:rigid-body-mechanics:power}{احتكاك التدحرج} $\mu_r = 0.012$. و $g = \qty{9.81}{m/s^2}$.

\medskip
\noindent\textbf{الجزء I --- التدحرج.}
\begin{enumerate}
  \item تسير السيارة بالسرعة $v = \qty{90}{km/h}$. أوجد السرعة الزاوية
        للعجلات؛ وسرعة نقاط الإطار العليا والسفلى والأمامية في
        مرجع الطريق.
  \item الطاقة الحركية الكلية للسيارة، مع فصل انسحاب الكل عن
        دوران العجلات الأربع؛ وما النسبة التي تمثلها دورانات
        العجلات؟
  \item وهي ساكنة على أرض مستوية، أوجد القوتين الناظميتين $N_f$
        (للعجلتين الأماميتين معًا) و $N_r$ (للخلفيتين) بمعادلتي
        كمية الحركة والعزم الحركي --- وأي محور يحمل أكثر؟
  \item عزم \omterm{prop:b2:rigid-body-mechanics:power}{احتكاك التدحرج} على كل عجلة هو $\mu_rNR$. أوجد
        الاستطاعة المبدَّدة \omterm{prop:b2:rigid-body-mechanics:power}{باحتكاك التدحرج} عند \qty{90}{km/h}؛
        وقارنها بقوة الإعاقة الهوائية $\tfrac12\rho C_xSv^2$ مع $C_xS =
        \qty{0.70}{m^2}$ و $\rho = \qty{1.2}{kg/m^3}$.
  \item يُقطع المحرك فتنساب السيارة على أرض مستوية: اكتب معادلة
        $v(t)$ (مع إدراج عطالة العجلات ككتلة فعّالة) وقدّر المسافة
        اللازمة للتباطؤ من \qty{90}{km/h} إلى
        \qty{45}{km/h}، مع إهمال الإعاقة؛ ثم مع الإعاقة وحدها.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- التسارع.}
السيارة ذات دفع أمامي؛ ويطبّق المحرك عزمًا $\Gamma$ على
العجلتين الأماميتين معًا. وأهمل هنا الإعاقة \omterm{prop:b2:rigid-body-mechanics:power}{واحتكاك التدحرج}.
\begin{enumerate}[resume]
  \item ارسم القوى الخارجية المؤثرة في السيارة كلها. وأي قوة
        تسارعها؟ وما استطاعة الطريق على السيارة، ومن أين تأتي
        الطاقة الحركية؟
  \item اكتب معادلة كمية الحركة للسيارة، ومعادلة العزم الحركي
        للعجلتين الأماميتين حول محورهما، وشرط اللاانزلاق؛ وبيّن
        أن التسارع هو $\dot v =
        \Gamma/R\,/\,(M + 4J/R^2)$ ثم احسب الكتلة الفعّالة.
  \item العزم الحركي بالنسبة إلى مركز الكتلة للسيارة كلها
        (باعتبار عزوم العجلات الحركية مهمَلة): بيّن أن انتقال
        الحمولة عند التسارع $\dot v$ هو $N_r - N_r^0 =
        Mh\dot v/L$ --- فيخفّ المحور الأمامي ويثقل الخلفي.
  \item التسارع الأعظمي دون أن تفلت العجلتان الدافعتان
        (الأماميتان): بيّن أن $\dot v_{\max} = fgb/(L + fh)$ ثم
        احسبه؛ والأمر نفسه لسيارة بدفع خلفي، $\dot v_{\max} =
        fga/(L - fh)$. فأي تصميم يتسارع أشدّ، ولماذا تضع سيارات
        السباق السريع الحمولة في الخلف؟
  \item عزم المحرك عند العجلات اللازم من أجل $\dot v_{\max}$؛
        والاستطاعة الموافقة عند \qty{50}{km/h}.
  \item السيارة نفسها على الجليد ($f = 0.10$): التسارع الأعظمي
        والزمن اللازم لبلوغ \qty{50}{km/h}.
  \item ماذا تفعل \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك} على الإطارين الأماميين بالإطار
        --- أينزلق أم لا --- ولماذا تسارع العجلة المفلتة على
        الجليد السيارة \emph{أقلّ} من العجلة الممسكة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الكبح.}
\begin{enumerate}[resume]
  \item تطبّق المكابح عزمًا على كل عجلة. بيّن أن قوة الكبح على
        السيارة، من أجل عجلة تبقى متدحرجة، هي احتكاك الطريق
        بالإطار، وأن عزم المكبح محدود
        بالمقدار $fN_iR$ لكل عجلة.
  \item انتقال الحمولة عند التباطؤ $\gamma$: حمولة المحور الأمامي
        $N_f = Mg\,b/L + Mh\gamma/L$؛ والقيم عند $\gamma = fg$.
  \item التباطؤ الأعظمي والعجلات الأربع كلها عند حدّ الانزلاق؛
        ومسافة التوقف من \qty{90}{km/h} على طريق جافّة، ثم على
        طريق مبتلّة ($f = 0.45$).
  \item إذا ضُبطت المكابح بحيث يتساوى العزمان الأمامي والخلفي،
        فأي محور ينقفل أولًا عند الكبح الشديد؟ ولماذا يكون
        انقفال المحور الخلفي خطرًا (فكّر في جهة \omterm{def:b2:rigid-body-mechanics:contact}{قوة الاحتكاك}
        على إطار منزلق)؟
  \item تنقفل عجلة فتنزلق: أوجد التباطؤ الآتي من الطريق، وما
        النسبة الضائعة من الكبح الأعظمي مع $f_{\text{حركي}} =
        0.7$ مقابل $f_{\text{ساكن}} = 0.8$. واشرح ما يفعله نظام
        منع الانقفال ولماذا ينبض بالمكابح.
  \item الحرارة المحرَّرة في المكابح من \qty{90}{km/h}؛ وارتفاع
        درجة حرارة أربعة أقراص فولاذية كتلة كل منها \qty{6}{kg} ($c = \qty{470}{J/(kg.K)}$)؛
        ولماذا تُبنى في طرق الجبال مهابط نجاة للشاحنات؟
  \item تستعيد سيارة كهربائية طاقة الكبح عبر محركها بمردود
        \qty{70}{\%}: أوجد الطاقة المستعادة في كل توقف من
        \qty{90}{km/h}؛ وعدد ما يعادله ذلك من توقفات في بطارية
        سعتها \qty{50}{kWh}.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- المنعطف وعجلة مرفوعة عن الأرض.}
\begin{enumerate}[resume]
  \item تنعطف السيارة على طريق مستوية على دائرة نصف قطرها
        $\rho = \qty{50}{m}$ بالسرعة $v$: على قوى الاحتكاك أن تقدّم
        $Mv^2/\rho$. أوجد سرعة الانعطاف العظمى على طريق جافّة ثم مبتلّة.
  \item العزم الحركي بالنسبة إلى مركز الكتلة على امتداد جهة
        الحركة: بيّن أن العجلتين الخارجيتين تتحملان زيادة $\Delta N =
        Mv^2h/\rho w$ حيث $w = \qty{1.5}{m}$ هو عرض المسار؛ وأوجد
        السرعة التي ترتفع عندها العجلتان الداخليتان ($\Delta N = Mg/2$)،
        وقارنها \omterm{def:b2:rigid-body-mechanics:rolling}{بسرعة الانزلاق}: أتنقلب هذه السيارة أم تنزلق أولًا؟
  \item المنعطف مائل بزاوية $\beta$. بيّن أنه عند السرعة
        $v = \sqrt{g\rho\tan\beta}$ لا حاجة إلى أي احتكاك؛ واحسب
        القيمة من أجل $\beta = \qty{10}{\degree}$.
  \item تُرفع عجلة عن الأرض بالرافعة وتُدار باليد إلى
        \qty{60}{rpm}؛ ويؤثر محملها بعزم احتكاك مقداره
        \qty{0.05}{N.m}. فكم تدور من الزمن؟ وما الطاقة الحركية الضائعة؟
  \item تُسقَط العجلة الدائرة على الطريق (والسيارة ساكنة، والعجلة
        عند \qty{60}{rpm}، و $N = \qty{3}{kN}$): أوجد زمن الانزلاق قبل
        توقفها، والمسافة التي تُدفع إليها السيارة لو كانت حرة
        التدحرج (عاملها ككتلة $M$ على عجلات حرة، وأهمل عطالة
        العجلات الأخرى).
  \item لخّص في جدول واحد: الأنظمة الأربعة (التدحرج، والتسارع،
        والكبح، والانعطاف)، والقوة التي تعمل في كل منها، وحدّ
        كولوم الذي عليها أن تحترمه.
\end{enumerate}
\end{problem}
