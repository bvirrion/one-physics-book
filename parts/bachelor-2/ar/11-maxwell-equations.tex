\chapter{معادلات ماكسويل}\label{ch:b2:maxwell-equations}

كتب جيمس كلارك ماكسويل سنة 1865 القوانين التي وجدها كولوم
وأمبير وفاراداي واحدًا واحدًا، وأضاف إلى قانون أمبير حدًّا لم
تطلبه أي تجربة بعد، فوجد أن معادلاته تقبل
أمواجًا تنتقل بسرعة استطاع أن يحسبها من ثوابت
الكهرباء والمغناطيسية --- أي سرعة الضوء. فصارت الكهرباء
والمغناطيسية والبصريات موضوعًا واحدًا. وقد صاغ كتاب السنة الأولى
القوانين تكاملات على سطوح وحلقات؛ وهذا الفصل يكتبها
أربع معادلات \emph{موضعية} صحيحة عند كل نقطة وفي كل لحظة،
بالمؤثرات المتجهية التي تجعل مثل هذه الكتابة ممكنة، ويستخرج
النتائج الأولى: \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة}، والكمونات،
وسلوك المجالات عند سطح فاصل، والتقريبات التي
تكون فيها دارات كتاب السنة الأولى تامة الصحة.

\section{المؤثرات المتجهية}

\begin{definition}[التدرج والتباعد والدوّار واللابلاسي]\label{def:b2:maxwell-equations:operators}
\index{تدرج}\index{تباعد}\index{دوّار}\index{لابلاسي}\index{نابلا}
من أجل مجال سلّمي $f$ ومجال متجهي $\vect A$، بالإحداثيات
الديكارتية مع المتجهة الرمزية $\vect\nabla = (\partial_x, \partial_y,
\partial_z)$:
\begin{align*}
\operatorname{\vect{grad}}f &= \vect\nabla f = (\partial_xf,\ \partial_yf,\ \partial_zf) ,\\
\operatorname{div}\vect A &= \vect\nabla\cdot\vect A = \partial_xA_x + \partial_yA_y + \partial_zA_z ,\\
\operatorname{\vect{curl}}\vect A &= \vect\nabla\wedge\vect A = (\partial_yA_z - \partial_zA_y,\ \partial_zA_x - \partial_xA_z,\ \partial_xA_y - \partial_yA_x) ,\\
\Delta f &= \operatorname{div}\operatorname{\vect{grad}}f = \partial_x^2f + \partial_y^2f + \partial_z^2f ,
\end{align*}
و $\Delta\vect A$ هو اللابلاسي مأخوذًا على كل مركّبة ديكارتية.
ولا يتوقف معناها على الإحداثيات: إذ إن $\operatorname{\vect{grad}}f
\cdot\dd\vect l = \dd f$ هو تغيّر $f$ على امتداد $\dd\vect l$؛ و $\operatorname{div}
\vect A$ هو تدفق $\vect A$ الخارج عبر سطح حجم
صغير، في واحدة الحجم؛ و $\operatorname{\vect{curl}}\vect A\cdot\vect n$ هو
التكامل الخطي للمجال $\vect A$ حول حلقة صغيرة ناظمتها $\vect n$، في واحدة
المساحة.
\end{definition}

\begin{theorem}[مبرهنتا التباعد وستوكس]\label{thm:b2:maxwell-equations:theorems}
\index{مبرهنة التباعد}\index{مبرهنة أوستروغرادسكي}\index{مبرهنة ستوكس}
من أجل سطح مغلق $\Sigma$ يحدّ الحجم $V$، ومنحنٍ مغلق موجَّه
$C$ يحدّ السطح $S$ (والناظمة $\vect n$ تعطيها قاعدة
اليد اليمنى)،
\[
\iint_\Sigma\vect A\cdot\vect n\,\dd S = \iiint_V\operatorname{div}\vect A\,\dd\tau ,
\qquad
\oint_C\vect A\cdot\dd\vect l = \iint_S\operatorname{\vect{curl}}\vect A\cdot\vect n\,\dd S .
\]
\end{theorem}
\admitted

\begin{remark}[ما نسلّم به ولماذا هو معقول]\label{rem:b2:maxwell-equations:whyadmitted}
تُبرهن المبرهنتان في كتاب الرياضيات للسنة الثالثة؛ ويقف
كتاب الرياضيات لهذه السنة عند صيغتهما المستوية
(غرين--ريمان) وعند التكاملات المتعددة. غير أن مضمونهما
هو ما استعملناه في \cref{ch:b2:fluid-kinematics}: بلّط الحجم
بصناديق صغيرة (وسطح الحلقة بحلقات صغيرة) --- فتتلاشى التدفقات
عبر الوجوه الداخلية (والتكاملات الخطية على الحواف
الداخلية) مثنى مثنى، ولا يبقى إلا الحدّ. وتنتج متطابقتان
من التعاريف وتُستعملان باستمرار:
$\operatorname{\vect{curl}}\operatorname{\vect{grad}}f = \vect 0$، و
$\operatorname{div}\operatorname{\vect{curl}}\vect A = 0$، و
$\operatorname{\vect{curl}}\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}
\operatorname{div}\vect A - \Delta\vect A$. ومن أجل المجالات ذات التناظر، تكون
الصيغ المفيدة: من أجل مجال قطري $A_r(r)\vect e_r$، $\operatorname{div}
= \frac1{r^2}\partial_r(r^2A_r)$ (كرويًّا) أو $\frac1r\partial_r(rA_r)$
(أسطوانيًّا)؛ ومن أجل مجال سمتي $A_\theta(r)\vect e_\theta$ حول
محور، $(\operatorname{\vect{curl}})_z = \frac1r\partial_r(rA_\theta)$؛ ومن أجل $f(r)$،
$\Delta f = \frac1{r^2}\partial_r(r^2\partial_rf) = \frac1r\partial_r^2(rf)$ (كرويًّا)،
و $\frac1r\partial_r(r\partial_rf)$ (أسطوانيًّا).
\end{remark}

\section{المعادلات الأربع}

\begin{theorem}[معادلات ماكسويل]\label{thm:b2:maxwell-equations:maxwell}
\index{معادلات ماكسويل}\index{معادلة ماكسويل--غاوس}\index{معادلة ماكسويل--فاراداي}\index{معادلة ماكسويل--أمبير}\index{تيار الإزاحة}
في الخلاء (وفي المادة الموصوفة بكثافتي شحنتها وتيارها
الكليتين $\rho$ و $\vect j$)، يحقق المجالان الكهربائي والمغناطيسي، عند
كل نقطة وفي كل لحظة،
\begin{align*}
\operatorname{div}\vect E &= \frac\rho{\varepsilon_0} &&\text{(ماكسويل--غاوس)} , &
\operatorname{\vect{curl}}\vect E &= -\frac{\partial\vect B}{\partial t} &&\text{(ماكسويل--فاراداي)} ,\\
\operatorname{div}\vect B &= 0 &&\text{(ماكسويل--التدفق)} , &
\operatorname{\vect{curl}}\vect B &= \mu_0\vect j + \mu_0\varepsilon_0\frac{\partial\vect E}{\partial t} &&\text{(ماكسويل--أمبير)} ,
\end{align*}
مع $\varepsilon_0 = \qty{8.85e-12}{F/m}$، و $\mu_0 = 4\pi \times \qty{1e-7}{H/m}$ (و
$\varepsilon_0\mu_0 = 1/c^2$). ويؤثر المجالان في الشحن بقوة
لورنتس $\vect F = q(\vect E + \vect v\wedge\vect B)$. والحدّ $\varepsilon_0\partial_t\vect E$
هو كثافة \emph{\omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة}} عند ماكسويل.
\end{theorem}

\begin{proof}[التكافؤ مع القوانين المكاملة في كتاب السنة الأولى]
طبّق مبرهنة \omterm{def:b2:maxwell-equations:operators}{التباعد} على الأوليين: $\iint_\Sigma\vect E\cdot
\vect n\,\dd S$ يساوي $Q_{\text{داخلية}}/\varepsilon_0$ (غاوس)، وتدفق $\vect B$
عبر أي سطح مغلق معدوم. وطبّق ستوكس على الأخيرتين:
$\oint_C\vect E\cdot\dd\vect l = -\dd\Phi_B/\dd t$ (فاراداي، من أجل حلقة ثابتة)
و $\oint_C\vect B\cdot\dd\vect l = \mu_0I_{\text{محاطة}} + \mu_0\varepsilon_0\,\dd\Phi_E/\dd t$
(أمبير، مكمَّلًا). وبالعكس، القوانين المكاملة، إذ تصحّ لكل
حجم وحلقة، تستلزم الموضعية. وهكذا تكون المعادلات الموضعية
قوانين السنة الأولى مع حدّ جديد واحد، مبرَّر أدناه؛ وتُؤخذ
مسلّمات للكهرمغناطيسية.
\end{proof}

\begin{proposition}[لماذا يلزم تيار الإزاحة]\label{prop:b2:maxwell-equations:displacement}
أخذُ \omterm{def:b2:maxwell-equations:operators}{تباعد} \omterm{thm:b2:maxwell-equations:maxwell}{معادلة ماكسويل--أمبير} واستعمال
$\operatorname{div}\operatorname{\vect{curl}} = 0$ وماكسويل--غاوس يعطي
\[
0 = \mu_0\operatorname{div}\vect j + \mu_0\varepsilon_0\,\partial_t\operatorname{div}\vect E
= \mu_0\Bigl(\operatorname{div}\vect j + \frac{\partial\rho}{\partial t}\Bigr) :
\]
أي إن المعادلات تحوي انحفاظ الشحنة. ودون حدّ
الإزاحة كانت لتقتضي $\operatorname{div}\vect j = 0$ دائمًا
--- وهو باطل في مكثف يُشحن، حيث يتوقف تيار الناقلية
عند اللبوس. وهناك، بين اللبوسين، يحمل $\varepsilon_0\partial_t\vect E$
بالضبط التيار الواصل في السلك، ويدور المجال المغناطيسي
حول الفجوة كما يدور حول سلك.
\end{proposition}

\begin{proof}
\omterm{def:b2:maxwell-equations:operators}{تباعد} الطرفين، كما ذُكر؛ وفي المكثف، يكون تدفق
$\varepsilon_0\partial_t\vect E$ عبر سطح بين اللبوسين
$\varepsilon_0S\,\dd E/\dd t = \dd Q/\dd t = I$.
\end{proof}

\begin{example}[المجال داخل مكثف يُشحن]\label{ex:b2:maxwell-equations:capacitor}
لبوسان دائريان نصف قطرهما $R$، والمسافة بينهما $d \ll R$، يُشحنان بتيار
$I$: فبينهما $E = Q/\varepsilon_0\pi R^2$ منتظم وينمو بالمعدل $\dd E/
\dd t = I/\varepsilon_0\pi R^2$. وبالتناظر $\vect B = B(r)\vect e_\theta$؛ وأمبير--ماكسويل
على دائرة نصف قطرها $r < R$ (ولا يعبرها تيار ناقلية): $2\pi rB
= \mu_0\varepsilon_0\pi r^2\,\dd E/\dd t$، ومنه $B = \mu_0Ir/2\pi R^2$ --- أي مجال
تيار منتظم $I$ موزَّع على القرص، متصل عند $r = R$ مع
مجال السلك $\mu_0I/2\pi r$ في الخارج. ومن أجل $I = \qty{1}{A}$ و $R = \qty{5}{cm}$: $B(R) =
\qty{4}{\micro T}$، وهو قابل للقياس؛ فحدّ ماكسويل حقيقي.
\end{example}

\begin{remark}[الخطية؛ ما تقوله المعادلات]\label{rem:b2:maxwell-equations:linearity}
المعادلات خطية في المجالين ومنابعهما: فالمجالات
تتراكب. واثنتان منها (غاوس، وفاراداي--التدفق) لا تحملان منابع بل
تعبّران عن البنية --- إذ تبدأ خطوط المجال الكهربائي وتنتهي على الشحن،
ولا تنتهي خطوط المجال المغناطيسي أبدًا؛ وأما الأخريان فتقولان كيف تنشئ الشحن
والتيارات، وتغيّر كل مجال، المجالات. ومقروءتين
معًا، ينشئ $\vect B$ متغيّر مجالًا $\vect E$ دوّارًا، وينشئ $\vect E$
متغيّر مجالًا $\vect B$ دوّارًا: فيستطيع الاثنان أن يستديم أحدهما الآخر
بلا أي شحنة --- وتلك هي موجة
\cref{ch:b2:plane-waves-polarization}، \omterm{thm:b2:maxwell-equations:maxwell}{وتيار الإزاحة} هو
الذي يجعلها ممكنة.
\end{remark}

\section{الكمونات}

\begin{proposition}[الكمونان السلّمي والمتجهي]\label{prop:b2:maxwell-equations:potentials}
\index{كمون متجهي}\index{كمون سلّمي}\index{معايرة}\index{معايرة كولوم}\index{معادلة بواسون}
لأن $\operatorname{div}\vect B = 0$ و $\operatorname{\vect{curl}}\vect E = -\partial_t
\vect B$، يشتقّ المجالان من \emph{\omterm{prop:b2:maxwell-equations:potentials}{كمون متجهي}} $\vect A$ ومن
\emph{\omterm{prop:b2:maxwell-equations:potentials}{كمون سلّمي}} $V$:
\[
\vect B = \operatorname{\vect{curl}}\vect A , \qquad
\vect E = -\operatorname{\vect{grad}}V - \frac{\partial\vect A}{\partial t} ,
\]
معيَّنين حتى \emph{تحويل معايرة} $\vect A \to \vect A +
\operatorname{\vect{grad}}\chi$ و $V \to V - \partial_t\chi$، وهو يترك المجالين
دون تغيير. وفي \emph{\omterm{prop:b2:maxwell-equations:potentials}{معايرة كولوم}} $\operatorname{div}\vect A = 0$، يصير قانون
غاوس \emph{\omterm{prop:b2:maxwell-equations:potentials}{معادلة بواسون}}
\[
\Delta V = -\frac\rho{\varepsilon_0} ,
\]
وفي النظام المستقر يصير قانون أمبير $\Delta\vect A = -\mu_0\vect j$.
وفي السكون يكون $V$ الكمون الكهرسكوني الوارد في كتاب السنة الأولى.
\end{proposition}

\begin{proof}
المجال المعدوم \omterm{def:b2:maxwell-equations:operators}{التباعد} دوّارٌ (مسلَّم به، مثل العبارة
العكسية القائلة إن المجال المعدوم \omterm{def:b2:maxwell-equations:operators}{الدوّار} \omterm{def:b2:maxwell-equations:operators}{تدرج}، وكلتاهما من كتاب
الرياضيات للسنة الثانية من أجل المناطق البسيطة الترابط)؛ وعندئذ $\operatorname{\vect{curl}}
(\vect E + \partial_t\vect A) = \vect 0$، ومنه يكون $\vect E + \partial_t\vect A$ تدرجًا.
والمعايرة: $\operatorname{\vect{curl}}\operatorname{\vect{grad}}\chi = \vect 0$ ويتلاشى
حدّا $\partial_t\operatorname{\vect{grad}}\chi$ في $\vect E$. وبواسون: $\operatorname{div}
\vect E = -\Delta V - \partial_t\operatorname{div}\vect A$. وفي السكون: $\operatorname{\vect{curl}}
\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}\operatorname{div}\vect A - \Delta\vect A
= \mu_0\vect j$.
\end{proof}

\begin{example}[كمونان متجهيان]\label{ex:b2:maxwell-equations:vectorpot}
المجال المنتظم $\vect B_0$ يشتقّ من $\vect A = \tfrac12\vect B_0\wedge\vect r$
(وتحقق: $\operatorname{\vect{curl}}(\tfrac12\vect B_0\wedge\vect r) = \vect B_0$، و
$\operatorname{div}\vect A = 0$)؛ وداخل ملف لولبي لانهائي يكون ذلك $A_\theta =
Br/2$، وخارجه $A_\theta = BR^2/2r$ --- أي كمونًا لا ينعدم
حيث ينعدم المجال، وتكامله الخطي على حلقة هو تدفق
$\vect B$ عبرها: $\oint\vect A\cdot\dd\vect l = \Phi_B$ (بستوكس). ويُقرأ قانون فاراداي
عندئذ $\vect E = -\partial_t\vect A$ من أجل المجال المحرَّض.
\end{example}

\section{المجالات عند سطح فاصل}

\begin{proposition}[علاقات العبور]\label{prop:b2:maxwell-equations:boundary}
\index{علاقات العبور (المجالات)}\index{تيار سطحي}
عبر سطح يحمل \omterm{def:b2:charges-currents-conduction:densities}{كثافة الشحنة} السطحية $\sigma$ \omterm{def:b2:charges-currents-conduction:densities}{وكثافة
التيار} السطحية $\vect j_s$ (\unit{A/m})، مع $\vect n_{12}$ الناظمة
الواحدية من الوسط 1 إلى الوسط 2،
\[
\vect E_2 - \vect E_1 = \frac\sigma{\varepsilon_0}\,\vect n_{12} , \qquad
\vect B_2 - \vect B_1 = \mu_0\,\vect j_s\wedge\vect n_{12} :
\]
تتصل المركّبة المماسية للمجال $\vect E$ والمركّبة الناظمية للمجال $\vect B$؛
وتقفز المركّبة الناظمية للمجال $\vect E$ بالمقدار $\sigma/\varepsilon_0$،
والمركّبة المماسية للمجال $\vect B$ بالمقدار $\mu_0j_s$. وعند سطح ناقل
مثالي ($\vect E = \vect B = \vect 0$ في الداخل، في حالات
\cref{ch:b2:wave-interfaces})، يكون $\vect E$ ناظميًّا ومساويًا $\sigma\vect n/
\varepsilon_0$، و $\vect B$ مماسيًّا ومساويًا $\mu_0\vect j_s\wedge\vect n$.
\end{proposition}

\begin{proof}
المركّبات الناظمية: قانونا غاوس والتدفق على علبة مسطّحة
تعبر السطح، مساحتها $\dd S$ وارتفاعها متلاشٍ: $(E_{2n}
- E_{1n})\dd S = \sigma\dd S/\varepsilon_0$، و $(B_{2n} - B_{1n})\dd S = 0$. والمركّبات
المماسية: قانونا فاراداي وأمبير على مستطيل صغير
يعبر السطح، طوله $\dd l$ وارتفاعه متلاشٍ: فتتلاشى
تدفقا $\partial_t\vect B$ و $\partial_t\vect E$ عبر المستطيل
مع مساحته، لكن تيارًا سطحيًّا $j_s\,\dd l$ يعبره:
$(E_{2t} - E_{1t})\dd l = 0$، و $(B_{2t} - B_{1t})\dd l = \mu_0j_s\dd l$ من أجل
الجهة المماسية العمودية على $\vect j_s$.
\end{proof}

\section{الأنظمة شبه المستقرة}

\begin{definition}[التقريبان شبه المستقرين]\label{def:b2:maxwell-equations:arqs}
\index{تقريب شبه مستقر}\index{النظام شبه المستقر}
تكون جملة حجمها $L$ مقودة عند التواتر $f$ \emph{شبه مستقرة}
حين $L \ll \lambda = c/f$: أي إن الزمن $L/c$ الذي تحتاجه المجالات لعبورها
مهمَل بالقياس إلى الدور، و«ترى» كل نقاطها
المنابع في اللحظة نفسها. وتُستعمل نهايتان:
\begin{itemize}
  \item النظام شبه المستقر \emph{المغناطيسي} (أي الدارات ذات
        التيارات، والوشائع، والمحوّلات، والتحريض): إذ يُسقَط \omterm{thm:b2:maxwell-equations:maxwell}{تيار
        الإزاحة} من قانون أمبير، $\operatorname{\vect{curl}}\vect B
        = \mu_0\vect j$ (ومنه $\operatorname{div}\vect j = 0$ وتجري التيارات في
        دارات مغلقة)، بينما يُحفَظ قانون فاراداي كاملًا؛
  \item النظام شبه المستقر \emph{الكهربائي} (أي مكثف
        يُشحن): إذ يُسقَط حدّ فاراداي، ويكون $\vect E$ المجالَ
        الكهرسكوني للشحن اللحظية، بينما يُحفَظ
        \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة} لإيجاد $\vect B$.
\end{itemize}
\end{definition}

\begin{proof}[التبرير]
في النظام المغناطيسي، قارن حدّي قانون أمبير في
ناقل: $\mu_0\varepsilon_0\partial_t E \sim \mu_0\varepsilon_0\omega E$ مقابل $\mu_0\gamma E$،
أي نسبة $\varepsilon_0\omega/\gamma = \omega\tau_r \sim 10^{-17}$ عند \qty{50}{Hz} في النحاس؛
وخارج الناقل يكون \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة} من رتبة
$\varepsilon_0\omega E$ مع $E \sim \omega AB$، أي أصغر من أثر تيار
الناقلية بالمعامل $(\omega L/c)^2 = (L/\lambda)^2 \times 4\pi^2$. وفي النظام
الكهربائي يكون المجال المحرَّض $\omega BL \sim \omega^2\varepsilon_0\mu_0L^2E$ أصغر من $E$
بالمعامل $(L/\lambda)^2$ نفسه.
\end{proof}

\begin{example}[الدارات والهوائيات]\label{ex:b2:maxwell-equations:sizes}
دارة الشبكة عند \qty{50}{Hz} ($\lambda = \qty{6000}{km}$) شبه مستقرة
حتى حجم بلد --- ولهذا تعمل قوانين كيرشهوف
والمحوّل. ودارة حجمها \qty{10}{cm} شبه مستقرة حتى نحو
\qty{100}{MHz}؛ وعند \qty{1}{GHz} ($\lambda = \qty{30}{cm}$) تصير أسلاكها هوائيات،
وتختلف التيارات على امتداد السلك، وتشعّ الدارة: وهو ميدان
\cref{ch:b2:dipole-radiation} وتصميم الموجات الصغرية، حيث يكون مسار
على لوحة خطَّ نقل.
\end{example}

\begin{method}[استعمال المعادلات الموضعية]\label{met:b2:maxwell-equations:method}
(1) استعمل تناظرات المنابع وثوابتها لتحديد
جهة المجالات ومتغيراتها (كما في كتاب السنة الأولى).
(2) اختر الصيغة: المكاملة (غاوس، وأمبير على سطح أو حلقة
محسنَي الاختيار) حين يكفي التناظر؛ والموضعية حين تحتاج إلى
معادلة تفاضلية (بواسون من أجل $V$، أو معادلة الانتشار أو الأمواج
من أجل المجالات). (3) وفي مسألة تتعلق بالزمن، قرّر النظام:
سكوني، أو شبه مستقر مغناطيسي أو كهربائي، أو ماكسويل كاملًا. (4)
وعند السطوح الفاصلة، استعمل علاقات العبور --- فهي الموضع الوحيد
الذي تدخل فيه الشحن والتيارات السطحية. (5) وتحقق من
$\operatorname{div}\vect B = 0$ ومن الأبعاد.
\end{method}

احسب \omterm{def:b2:maxwell-equations:operators}{تباعد} \omterm{def:b2:maxwell-equations:operators}{ودوّار} (أ) $\vect A = (x, y, z)$؛ (ب)

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:2}
(أ) تحقق من أبعاد كل حدّ في \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} الأربع.

«ناقل» في كل حالة؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:3}
مكثف بلبوسين دائريين ($R = \qty{10}{cm}$، و $d = \qty{2}{mm}$) يُشحن
بتيار $I = \qty{0.5}{A}$. أوجد معدل تغيّر $E$؛ وكثافة \omterm{thm:b2:maxwell-equations:maxwell}{تيار

السلك المغذّي على البعد نفسه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:4}
(أ) $\vect E = kx\,\vect e_x$ يملأ منطقة: أوجد \omterm{def:b2:charges-currents-conduction:densities}{كثافة الشحنة} فيها. (ب)
وداخل كرة نصف قطرها $R$، $\vect E = (\rho_0r/3\varepsilon_0)\vect e_r$: تحقق \omterm{def:b2:maxwell-equations:operators}{بالتباعد}
القطري من أن $\rho = \rho_0$، واستعد المجال في الخارج
من قانون غاوس المكامل. (ج) $\vect B = B_0(y/a)\vect e_x$: أهو مجال مغناطيسي
ممكن؟ وأي كثافة تيار تستديمه؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:5}
\emph{الكمونات المتجهية.} (أ) بيّن أن $\vect A = \tfrac12\vect B_0\wedge\vect r$
يعطي $\operatorname{\vect{curl}}\vect A = \vect B_0$ و $\operatorname{div}\vect A = 0$.
(ب) وكذلك يفعل $\vect A' = B_0x\,\vect e_y$: أوجد دالة المعايرة $\chi$ التي تحقق
$\vect A' = \vect A + \operatorname{\vect{grad}}\chi$. (ج) وملف لولبي نصف قطره $R$،
ومجاله $B$ في الداخل: تحقق من $A_\theta = Br/2$ ($r < R$) و $BR^2/2r$ ($r > R$)
باستعمال $(\operatorname{\vect{curl}}\vect A)_z = \frac1r\partial_r(rA_\theta)$؛ والاتصال عند $R$.
(د) والتكامل الخطي للكمون $\vect A$ على دائرة نصف قطرها $r > R$ وتدفق
$\vect B$: فماذا يقول ذلك عن القوة المحركة المحرَّضة في حلقة خارج
ملف لولبي يتغير تياره؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:6}
\emph{علاقات العبور.} (أ) ناقل في توازن كهرسكوني
يحمل $\sigma$ على سطحه: أوجد المجال خارجه مباشرةً (من
القفزة، مع $\vect E = \vect 0$ في الداخل)؛ من أجل $\sigma = \qty{1}{\micro C/m^2}$. (ب) ومستويان
متوازيان عليهما $+\sigma$ و $-\sigma$: أوجد المجال في كل مكان بالقفزات.
(ج) وملف لولبي طويل ($n = \qty{2000}{m^{-1}}$، و $I = \qty{5}{A}$) بوصفه صفيحة
تيار: أوجد $j_s$ وقفزة $B$ و $B$ في الداخل. (د) ووشيعة حلقية: لماذا يكون $B$
معدومًا في الخارج و $\mu_0NI/2\pi r$ في الداخل، من القفزة نفسها؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:7}
\emph{بواسون في بعد واحد.} بين مسرَيين مستويين عند $x = 0$
($V = 0$) و $x = d$ ($V = U$)، تملأ كثافة شحنة منتظمة $\rho_0$
الفجوة (وهي شحنة فراغية). (أ) حُلّ \omterm{prop:b2:maxwell-equations:potentials}{معادلة بواسون} من أجل $V(x)$. (ب) وأوجد المجال
$E(x)$؛ وأين ينعدم إذا كان $\rho_0 > 0$ وكبيرًا؟ (ج) والشحن السطحية على
المسرَيين، من علاقة العبور. (د) ومع $\rho_0 = 0$
استعد المكثف؛ ومع $U = 0$، ارسم $V$ تخطيطيًّا.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:8}
\emph{شبه مستقر أم لا.} (أ) أوجد $\lambda$ عند \qty{50}{Hz} و \qty{1}{MHz}
و \qty{100}{MHz} و \qty{2.4}{GHz}. (ب) وحتى أي تواتر تكون دارة حجمها \qty{10}{cm}
شبه مستقرة (بالمحك $L < \lambda/10$)؟ ورقاقة حجمها \qty{1}{cm}؟ وشبكة
كهربائية طولها \qty{1000}{km}؟ (ج) وفي مكثف نصف قطره $R = \qty{5}{cm}$ عند
\qty{1}{MHz}، قدّر نسبة المجال الكهربائي المحرَّض (الآتي من
$\vect B$ المتغير في التمارين السابقة) إلى المجال الرئيسي: $(\omega R/c)^2
/4$. (د) ولماذا يكون تواتر الشبكة في شبكة تغطي قارة مسألةً
شبه مستقرة مع أن عرض الشبكة \qty{1000}{km}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:9}
بيّن أن قانون أمبير دون \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة}، $\operatorname{\vect{curl}}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:10}
\emph{البيتاترون.} مجال مغناطيسي منتظم $B(t)\vect e_z$ يملأ
أسطوانة نصف قطرها $R$، وينمو بالمعدل $\dd B/\dd t = \dot B$. (أ) بالتناظر
$\vect E = E_\theta(r)\vect e_\theta$: استعمل قانون فاراداي المكامل لإيجاد $E_\theta$
من أجل $r < R$ و $r > R$. (ب) ولا شحنة في أي مكان: علّق على
مجال كهربائي خطوطه مغلقة. (ج) وكم يكسب إلكترون على دائرة
نصف قطرها $r_0 < R$ من الطاقة في كل دورة؟ ومن أجل $\dot B = \qty{10}{T/s}$
و $r_0 = \qty{0.5}{m}$: أوجد الطاقة في الدورة بالإلكترون فولت؛ وعدد الدورات اللازم من أجل \qty{20}{MeV}.
(د) وكي يبقى نصف قطر المدار ثابتًا، يجب أن تنمو كمية الحركة $p = eBr_0$ (الآتية
من القوة المغناطيسية) كما تقتضي الطاقة المكتسبة من المجال المحرَّض:
بيّن الشرط $B(r_0) = \tfrac12\langle B\rangle_{\text{القرص}}$
--- وهو شرط البيتاترون.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:11}

\operatorname{\vect{curl}} = \operatorname{\vect{grad}}\operatorname{div} - \Delta$ والمعادلات
الأخرى لتحصل على $\Delta\vect E = \mu_0\varepsilon_0\,\partial_t^2\vect E$. (ب) والأمر نفسه من أجل
$\vect B$. (ج) وهذه \omterm{thm:b2:waves-on-strings:equation}{معادلة دالمبير} في ثلاثة أبعاد: فما السرعة؟
احسبها من $\varepsilon_0$ و $\mu_0$. (د) ولماذا أقنع هذا العدد
ماكسويل بأن الضوء كهرمغناطيسي، وأي قياس
لفيبر وكولراوش استعمل؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:12}
\emph{المجالات في ناقل عند التواتر المنخفض.} في ناقل أومي
($\vect j = \gamma\vect E$) في \omterm{def:b2:maxwell-equations:arqs}{النظام شبه المستقر} المغناطيسي: (أ) خذ
\omterm{def:b2:maxwell-equations:operators}{دوّار} قانون أمبير واستعمل فاراداي لتبيّن أن $\Delta\vect B = \mu_0\gamma\,
\partial_t\vect B$؛ ولماذا هي معادلة انتشار؟ (ب) والزمن المميز
لنفوذ $\vect B$ إلى عمق $d$: $\tau \sim \mu_0\gamma d^2$؛ وأعطِ القيم من أجل
صفيحة نحاسية سماكتها \qty{1}{cm} ومن أجل لبّ الأرض ($\gamma \approx \qty{5e5}{S/m}$،
و $d \approx \qty{3000}{km}$). (ج) وعند التواتر $f$، أوجد العمق المبلوغ في
دور واحد: بيّن أنه من رتبة $\delta = \sqrt{2/\mu_0\gamma\omega}$؛ وأعطِ القيمة للنحاس
عند \qty{50}{Hz} و \qty{1}{MHz} (وهو \omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة} في
\cref{ch:b2:waves-in-media}). (د) وماذا يستلزم (ب) لتغيّرات
المجال المغناطيسي الأرضي مع الزمن؟
\end{exercise}

\section{مسألة: المكثف والبيتاترون والكمونات}

\begin{problem}[{مسألة نهاية الأسبوع --- معادلات ماكسويل في العمل في
مكثف واحد يُشحن، وفي المسرّع الذي يعمل بقانون فاراداي
وحده، وفي الكمونات التي تحلّها}]
\label{pb:b2:maxwell-equations:1}

\medskip
\noindent\textbf{الجزء I --- المكثف الذي يُشحن.}
لبوسان دائريان نصف قطرهما $R = \qty{10}{cm}$، والمسافة بينهما $d = \qty{1.0}{mm}$،
وبينهما هواء؛ ويجري تيار $I(t) = I_0\cos\omega t$ مع $I_0 = \qty{1.0}{A}$ و $f = \qty{1.0}{MHz}$
في السلك المغذّي على امتداد المحور. وتُهمَل آثار الحواف.
\begin{enumerate}
  \item الشحنة $Q(t)$ والمجال $E(t)$ بين اللبوسين؛ وسعة
        $E$.
  \item كثافة \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة} بين اللبوسين؛ وتحقق من أن
        تدفقه عبر لبوس يساوي $I(t)$.
  \item المجال المغناطيسي $B(r, t)$ بين اللبوسين من أجل $r < R$، بقانون
        أمبير--ماكسويل المكامل؛ والسعة عند $r = R$.
  \item والمجال من أجل $r > R$ (وما زلنا بين مستويي اللبوسين):
        قارنه بمجال السلك عند $r$ نفسه؛ وهل هناك
        انقطاع عند $r = R$؟
  \item ويحرّض $\vect B$ المتغير، بقانون فاراداي، تصحيحًا
        $\vect E_1$ للمجال الكهربائي: باستعمال $\operatorname{\vect{curl}}\vect E_1
        = -\partial_t\vect B$ مع $\vect E_1 = E_1(r)\vect e_z$ (محوري)، بيّن أن
        $E_1(r) - E_1(0) = \dfrac{\mu_0\varepsilon_0r^2}4\,\dfrac{\dd^2E}{\dd t^2}$، ثم
        قدّر $|E_1(R) - E_1(0)|/|E|$.
  \item استنتج صحة المعالجة شبه المستقرة الكهربائية
        عند \qty{1}{MHz}؛ وعند أي تواتر يبلغ التصحيح
        $10\%$؟ (والحلّ التام دالة بسل: فقد صار
        المكثف جوفًا.)
  \item الطاقة: أوجد الطاقة الكهربائية $\tfrac12\varepsilon_0E^2$ المخزَّنة بين
        اللبوسين، وقيمتها العظمى؛ والطاقة المغناطيسية $\int B^2/2\mu_0\,\dd\tau$
        في اللحظة التي فيها $E = 0$؛ ونسبتهما، وصلتها
        بتقدير السؤال 5.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- البيتاترون.}
يُنتج مغنطيس كهربائي مجالًا $\vect B = B(r, t)\vect e_z$ متناظرًا
حول المحور $z$؛ وتدور إلكترونات في حلقة مفرَّغة نصف قطرها $r_0$.
\begin{enumerate}[resume]
  \item بيّن، من ماكسويل--فاراداي في الصيغة المكاملة، أن
        المجال الكهربائي على المدار هو $E_\theta(r_0) = -\dfrac1{2\pi r_0}
        \dfrac{\dd\Phi}{\dd t}$، حيث $\Phi$ التدفق عبر المدار.
  \item معادلة الحركة المماسية لإلكترون (شحنته $-e$):
        $\dd p/\dd t = eE_\theta$ بالطويلة؛ والطاقة المكتسبة في الدورة هي
        $e\,|\dd\Phi/\dd t|$.
  \item والتوازن القطري على الدائرة: $p = eB(r_0)r_0$. وكي يبقى $r_0$
        ثابتًا، بيّن أن $B(r_0, t) = \Phi(t)/2\pi r_0^2$، أي إن
        المجال على المدار يجب أن يكون نصف المجال الوسطي داخله.
  \item ولماذا لا يمكن تسريع الإلكترونات بمجال منتظم؟
        وكيف يُحقَّق الشرط عمليًّا (شكل قطعتي
        القطبين)؟
  \item $r_0 = \qty{0.50}{m}$، و $B(r_0)$ يرتفع من $0$ إلى \qty{0.40}{T} في
        \qty{5}{ms}: أوجد كمية الحركة النهائية؛ والطاقة النهائية (باستعمال $E = pc$ من أجل هذه
        الإلكترونات الفائقة النسبوية، و $\qty{1}{MeV} = \qty{1.6e-13}{J}$)؛
        والطاقة في الدورة وعدد الدورات؛ والمسافة المقطوعة.
  \item وتشعّ الإلكترونات حين تتسارع على دائرة
        (\cref{ch:b2:dipole-radiation}): وهذا يحدّ البيتاترونات ببضع
        مئات من الميغاإلكترون فولت. فمن أين تأتي طاقة الإلكترونات في نهاية المطاف،
        وعبر أي حدّ من \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل}؟
  \item وبعد التسريع تُرمى الإلكترونات على هدف
        فتنتج أشعة سينية: فلماذا كان البيتاترون آلة العلاج
        الإشعاعي في خمسينيات القرن العشرين، وما الذي حلّ محلّه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الكمونات وبواسون.}
\begin{enumerate}[resume]
  \item اكتب $\vect E$ و $\vect B$ بدلالة $V$ و $\vect A$، وتحقق
        من أن ماكسويل--فاراداي وماكسويل--التدفق تتحققان عندئذ
        تلقائيًّا.
  \item وكرة نصف قطرها $a$ تحمل الكثافة المنتظمة $\rho_0$: اكتب
        \omterm{prop:b2:maxwell-equations:potentials}{معادلة بواسون} في التناظر الكروي، وحُلّها في الداخل وفي
        الخارج ($V \to 0$ في اللانهاية، و $V$ و $V'$ متصلان عند $a$)،
        واستعد مجالات كتاب السنة الأولى.
  \item الطاقة الكامنة للكرة، $\tfrac12\int\rho V\,\dd\tau$: أوجد القيمة
        $3Q^2/20\pi\varepsilon_0a$؛ ومن أجل نواة يورانيوم ($Q = 92e$، و $a =
        \qty{7.4}{fm}$)، بالميغاإلكترون فولت --- وهي الطاقة المحرَّرة حين تنشطر.
  \item ومن أجل مكثف الجزء الأول في \omterm{def:b2:maxwell-equations:arqs}{النظام شبه المستقر}
        الكهربائي، أعطِ $V$ بين اللبوسين. وبيّن أن كمونًا متجهيًّا
        \emph{محوريًّا} $A_z(r, t)\,\vect e_z$، وله $(\operatorname{\vect{curl}}
        \vect A)_\theta = -\partial_rA_z$، يعيد إنتاج المجال $B_\theta$ في
        السؤال 3، ثم أوجد $A_z$ من أجل $r < R$ (بأخذ $A_z(0) = 0$).
  \item بيّن أن التصحيح $\vect E_1$ في السؤال 5 هو بالضبط
        $-\partial_t\vect A$ من أجل ذلك $A_z$، حتى ثابت.
  \item المعايرة: يتحقق شرط كولوم $\operatorname{div}\vect A = 0$ من أجل
        $A_z(r)$ عندك --- فلماذا؟ وأي $\vect A$ آخر يعطي $\vect B$ نفسه؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- السطوح الفاصلة والأنظمة.}
\begin{enumerate}[resume]
  \item الشحنة السطحية على لبوسي الجزء الأول في لحظة
        $E$ الأعظمي؛ وتحقق من علاقة العبور على وجه اللبوس
        الداخلي (والمجال معدوم داخل المعدن).
  \item وينتشر التيار قطريًّا في اللبوس من المحور إلى
        الحافة: أوجد \omterm{def:b2:charges-currents-conduction:densities}{كثافة التيار} السطحية $j_s(r)$ في اللبوس؛ والمجال
        المغناطيسي خارج وجه اللبوس الخارجي مباشرةً، من
        علاقة العبور، مقارنًا بالمجال الموجود في السؤال 3
        على الجانب الداخلي.
  \item صنّف: اللبوسان عند \qty{1}{MHz} (أشبه مستقر كهربائي؟)، وحلقة
        السلك المغذّي ذات \qty{20}{cm} (أشبه مستقرة مغناطيسية؟)، والمكثف نفسه عند
        \qty{1}{GHz}؛ وبرّر ذلك بالنسبة $L/\lambda$.
  \item وفي حلقة البيتاترون، لماذا يكون \omterm{def:b2:maxwell-equations:arqs}{التقريب شبه المستقر}
        المغناطيسي ممتازًا (بالأحجام، وبارتفاع عند \qty{100}{Hz}) مع أن
        المجال الكهربائي هناك جوهري؟
  \item لخّص، في جدول، أي معادلة من \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} (موضعية أو
        مكاملة) أجابت كل جزء، وأي حدّ --- \omterm{thm:b2:maxwell-equations:maxwell}{تيار
        الإزاحة}، أو فاراداي، أو غاوس، أو قفزة العبور --- كان حاسمًا.
\end{enumerate}
\end{problem}
