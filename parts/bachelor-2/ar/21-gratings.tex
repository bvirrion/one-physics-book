\chapter{تداخل الأمواج المتعددة والشبكات}\label{ch:b2:gratings}

أمِل قرصًا مدمجًا تحت مصباح فيقذف قوس قزح: فحلزون حفره،
المتباعدة ميكرومترًا ونصفًا، شبكة --- أي آلاف
الشرائط العاكسة الدقيقة، كل منها يعيد نسخة من الموجة، وكلها
تتداخل. وموجتان تعطيان هدب $\cos^2$ الوادعة في الفصلين
السابقين؛ وألف موجة تعطي هدبًا أضيق بألف مرة، صافية
كفاية لفصل طولي موجة يختلفان بجزء من مئة
ألف. وتلك هي الأداة التي قُرئ بها تركيب
الشمس وسرعة المجرات، والتي تقوم في كل
مطياف. وهذا الفصل يجمع $N$ موجة، ويستنتج \omterm{prop:b2:gratings:equation}{معادلة
الشبكة}، وتبديدها وقدرة تمييزها، ويلقى الأداة الأخرى
المتعددة الأمواج، وهي جوف فابري--بيرو الذي سيصير
رنّان الليزر.

\section{تداخل $N$ موجة}

\begin{theorem}[$N$ موجة متساوية بتدرج طوري حسابي]\label{thm:b2:gratings:nwaves}
\index{تداخل الأمواج المتعددة}\index{عظم رئيسي}\index{عظم ثانوي}\index{شعاع طوري}
$N$ موجة مترابطة متساوية السعة $a$، تتأخر كل منها عن سابقتها
بالطور $\varphi$، تتراكب في موجة سعتها $A(\varphi)$ وشدتها
\[
A = a\,\frac{\sin(N\varphi/2)}{\sin(\varphi/2)} , \qquad
I(\varphi) = I_0\,\frac{\sin^2(N\varphi/2)}{\sin^2(\varphi/2)} :
\]
ومواضع العظم \emph{الرئيسية} شدتها $N^2I_0$ حيث $\varphi = 2\pi p$ (أي
الأمواج كلها على الطور نفسه)، ونصف عرضها الزاوي $2\pi/N$ بدلالة $\varphi$ (إذ أول صفرين
عند $\varphi = 2\pi p \pm 2\pi/N$)، يفصل بينها $N - 2$ من مواضع العظم
\emph{الثانوية} لا تعلو على نحو $4.5\%$ من الرئيسية من أجل $N$
الكبيرة. وكلما كثرت الأمواج صفت القمم وسطعت: فالعرض
يتقلص وفق $1/N$، والارتفاع ينمو وفق $N^2$، والطاقة (وهي جداؤهما)
وفق $N$.
\end{theorem}

\begin{proof}
السعات العقدية $a\eu^{\iu k\varphi}$ مع $k = 0, \dots, N - 1$: متتالية
هندسية، ومجموعها $a(1 - \eu^{\iu N\varphi})/(1 - \eu^{\iu\varphi})$، أي $a\eu^{\iu(N - 1)\varphi/2}\sin(N\varphi/2)
/\sin(\varphi/2)$. والأصفار حيث $\sin(N\varphi/2) = 0$ مع $\sin(\varphi/2) \ne 0$: $\varphi = 2\pi
m/N$، و $m$ ليس من مضاعفات $N$. وفي صورة الأشعة الطورية تنغلق سهام $N$ في مضلع
منتظم عند الأصفار، وتصطفّ عند مواضع العظم الرئيسية.
\end{proof}

\section{شبكة الحيود}

\begin{proposition}[معادلة الشبكة]\label{prop:b2:gratings:equation}
\index{شبكة حيود}\index{معادلة الشبكة}\index{رتبة (شبكة)}\index{تبديد زاوي}
\emph{الشبكة} مجموعة من $N$ شقًّا متطابقة متساوية \omterm{def:b2:maxwell-equations:operators}{التباعد} متوازية
(أو أخاديد) دورها $a$. وموجة مستوية واردة تحت الزاوية $\theta_0$
عن الناظم يعيد كل شقّ إصدارها؛ وفي الجهة $\theta$ يكون
للموجتين الآتيتين من شقّين متجاورين \omterm{thm:b2:two-wave-interference:formula}{فرق المسير} $\delta = a(\sin\theta
- \sin\theta_0)$، وتتعاضدان في الجهات
\[
a\,(\sin\theta_p - \sin\theta_0) = p\lambda , \qquad p = 0, \pm1, \pm2, \dots
\]
--- وهي \emph{رتب} الشبكة. والرتبة الصفر هي الحزمة
غير المنحرفة من أجل كل طول موجة؛ وكل رتبة أخرى تنشر أطوال الموجة،
مع \emph{\omterm{prop:b2:gratings:equation}{تبديد زاوي}}
\[
\frac{\dd\theta}{\dd\lambda} = \frac{p}{a\cos\theta} ,
\]
وهو أكبر من أجل شبكة أدقّ ورتبة أعلى؛ وأعظم رتبة هي
$|p| \le a(1 + |\sin\theta_0|)/\lambda$. و\emph{شبكة الانعكاس} (أي أخاديد محفورة
على مرآة) تحقق المعادلة نفسها بالزوايا المنعكسة.
\end{proposition}

\begin{proof}
الطور بين شقّين متتاليين هو $\varphi = 2\pi\delta/\lambda$؛ ومواضع العظم
الرئيسية في \cref{thm:b2:gratings:nwaves} عند $\delta = p\lambda$.
واشتقّ $a\sin\theta = p\lambda + a\sin\theta_0$ عند ثبات $p$ و $\theta_0$.
\end{proof}

\begin{example}[شبكة صفّ وقرص مدمج]\label{ex:b2:gratings:examples}
شبكة فيها $600$ خط في المليمتر ($a = \qty{1.67}{\micro m}$) عند الورود
الناظمي ترسل ضوء \qty{500}{nm} إلى $\sin\theta = 0.3$، أي $\theta = \qty{17.5}{\degree}$ في
الرتبة الأولى، و $\qty{36.9}{\degree}$ في الثانية، وليست لها رتبة ثالثة
($3 \times 0.3 > 1$)؛ ويمتد طيفها في الرتبة الأولى من \qty{13.9}{\degree}
(\qty{400}{nm}) إلى \qty{24.8}{\degree} (\qty{700}{nm}). والقرص المدمج، $a =
\qty{1.6}{\micro m}$، شبكة كهذه في الانعكاس --- ومن هنا قوس القزح،
وكون قرص الفيديو الرقمي ($a = \qty{0.74}{\micro m}$) ينشره أوسع. وأحمر
الرتبة الثانية (\qty{700}{nm} عند $\sin\theta = 0.84$) يتراكب مع بنفسجي الرتبة الثالثة:
فالرتب تتراكب بعد الأولى، وهو إزعاج يُداوى بمرشّحات
أو بموشور.
\end{example}

\section{قدرة التمييز}

\begin{proposition}[قدرة تمييز الشبكة]\label{prop:b2:gratings:resolution}
\index{قدرة التمييز (شبكة)}\index{معيار رايلي (طيفي)}
يُفصَل طولا موجة متقاربان $\lambda$ و $\lambda + \Delta\lambda$ في الرتبة $p$ بالكاد
حين يقع موضع \omterm{thm:b2:gratings:nwaves}{العظم الرئيسي} لأحدهما على أول صفر للآخر
(وهو معيار رايلي)، وذلك يقع عند
\[
\frac\lambda{\Delta\lambda} = pN :
\]
و\emph{قدرة التمييز} هي الرتبة مضروبة في عدد الخطوط
المضاءة. ويمكن كتابتها أيضًا $W(\sin\theta - \sin\theta_0)/\lambda$ حيث $W = Na$
عرض الشبكة: أي أكبر فرق مسير عبر الشبكة،
بأطوال الموجة --- ولهذا تستعمل المطيافات الكبيرة شبكات بعرض عشرات
السنتيمترات وتعمل عند زوايا مماسّة.
\end{proposition}

\begin{proof}
موضع العظم من الرتبة $p$ من أجل $\lambda + \Delta\lambda$ يقع عند $\varphi = 2\pi p$ لذلك
الطول، أي عند $\delta = p(\lambda + \Delta\lambda)$؛ وأول صفر لموضع العظم
من أجل $\lambda$ عند $\delta = p\lambda + \lambda/N$. وبالمساواة: $p\Delta\lambda = \lambda/N$. ثم $pN =
Na(\sin\theta - \sin\theta_0)/\lambda$.
\end{proof}

\begin{example}[فصل خطي الصوديوم]\label{ex:b2:gratings:sodium}
يحتاج الثنائي، $\Delta\lambda = \qty{0.6}{nm}$ عند \qty{589}{nm}، إلى $R = 1000$: أي
ألف خط في الرتبة الأولى، أي \qty{2}{mm} من شبكة \qty{600}{mm^{-1}}.
ويحتاج عرض خطه، \qty{0.02}{nm}، إلى $R = 30\,000$: أي شبكة \qty{5}{cm}
في الرتبة الأولى، أو \qty{2.5}{cm} في الثانية --- وشبكة
\qty{10}{cm} فيها \qty{1200}{mm^{-1}} في الرتبة الثانية تبلغ
$240\,000$، فتميّز \qty{0.0025}{nm}: أي عرض خط عرّضته
حركة الذرات الحرارية، وانزياح دوبلر لنجم يتحرك
بسرعة \qty{1}{km/s}.
\end{example}

\section{جوف فابري--بيرو}

\begin{proposition}[تداخل الأمواج المتعددة بتقسيم السعة]\label{prop:b2:gratings:fabryperot}
\index{مقياس تداخل فابري--بيرو}\index{تابع إيري}\index{النقاوة}\index{المجال الطيفي الحر}
مرآتان متوازيتان عاكستان جزئيًّا (وانعكاسيتهما بالشدة $R$)
تبعد إحداهما عن الأخرى $L$ لا تنفذان موجة عند الورود الناظمي إلا بقدر
ما تتجمع على الطور نفسه الأمواج التي ارتدّت $0, 1, 2, \dots$ مرة بينهما:
فمع $\varphi = 4\pi L/\lambda$ طورَ الذهاب والإياب، تكون الشدة
النافذة (\omterm{prop:b2:gratings:fabryperot}{بتابع إيري})
\[
\frac{I_t}{I_0} = \frac1{1 + F\sin^2(\varphi/2)} , \qquad F = \frac{4R}{(1 - R)^2} :
\]
أي قمم صافية من النفاذ التامّ عند $2L = p\lambda$ (وهي \emph{رنينات}
الجوف، متباعدة في التواتر بمقدار \emph{\omterm{prop:b2:gratings:fabryperot}{المجال الطيفي الحر}}
$c/2L$)، وعرضها النسبي $1/\mathcal F$ حيث \emph{النقاوة} $\mathcal F = \pi
\sqrt R/(1 - R)$ --- أي $30$ من أجل $R = 0.9$، و $300$ من أجل $R = 0.99$. وهو ندّ الشبكة
في قدرة التمييز ($p\mathcal F$ مع $p = 2L/\lambda \sim 10^5$، ومن ثَمّ $R \sim
10^7$)، وهو، مغلقًا على نفسه، رنّان الليزر
(\cref{ch:b2:laser}).
\end{proposition}

\begin{proof}
كل ذهاب وإياب يضرب السعة في $R\eu^{\iu\varphi}$ (إلى طور
ثابت)؛ والسعة النافذة هي $t^2a\sum(R\eu^{\iu\varphi})^k =
t^2a/(1 - R\eu^{\iu\varphi})$ مع $t^2 = 1 - R$؛ ومربع طويلتها $(1 - R)^2/
(1 - 2R\cos\varphi + R^2) = (1 - R)^2/[(1 - R)^2 + 4R\sin^2(\varphi/2)]$. والقمم عند
$\varphi = 2\pi p$؛ ونصف العظم حيث $F\sin^2(\varphi/2) = 1$، أي $\Delta\varphi \approx
4/\sqrt F = 2(1 - R)/\sqrt R$، في وجه $2\pi$ بين القمم.
\end{proof}

\begin{method}[حسابات الشبكة]\label{met:b2:gratings:method}
(1) الدور $a = 1/(\text{عدد الخطوط في المليمتر})$؛ \omterm{prop:b2:gratings:equation}{ومعادلة الشبكة} مع
إشارات الزوايا. (2) واذكر الرتب الموجودة ($|\sin\theta| \le 1$)
وتحقق من تراكبها ($p\lambda_{\max}$ في وجه $(p + 1)\lambda_{\min}$). (3)
والتبديد $p/a\cos\theta$، محوَّلًا إلى موضع على الكاشف
بالبعد البؤري: $\dd x = f\,\dd\theta$. (4) وقدرة التمييز $pN$ حيث $N$
عدد الخطوط المضاءة فعلًا؛ وقارن $\lambda/R$ بالبنية المراد
تمييزها. (5) وفي الأدوات الحقيقية أضف عرض شقّ الدخول (فهو
يشوّش الخطوط) وبكسلات الكاشف.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:gratings:1}
شبكة فيها $500$ خط في المليمتر عند الورود الناظمي، و $\lambda =
\qty{550}{nm}$: أوجد زوايا الرتب؛ وأعلى رتبة؛ والعرض الزاوي
للطيف المرئي في الرتبة الأولى ($400$--\qty{700}{nm})؛ والشبكة نفسها عند
ورود \qty{30}{\degree}: أوجد الرتب على كل جانب.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:2}
ثنائي الصوديوم مع شبكة فيها $600$ خط في المليمتر وعدسة $f =
\qty{50}{cm}$: أوجد الفصل الزاوي والخطي بين الخطين على
الكاشف في الرتبتين الأولى والثانية؛ وحجم البكسل اللازم لرؤيتهما
منفصلين.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:3}
أوجد قدرة تمييز شبكة \qty{2}{cm} فيها $600$ خط في المليمتر في الرتب $1$ و $2$
و $3$؛ وأصغر $\Delta\lambda$ مميَّز عند \qty{550}{nm}؛ أتفصل ثنائي
الصوديوم؟ وخط الهدروجين H$\alpha$ عن توأمه الديوتيريومي (وبينهما \qty{0.18}{nm}
من الفرق)؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:4}
قرص مدمج ($a = \qty{1.6}{\micro m}$) وقرص فيديو رقمي ($a = \qty{0.74}{\micro m}$) مضاءان عند
الورود الناظمي: أوجد زاويتي البنفسجي والأحمر في الرتبة الأولى؛ وأيّ الرتب
موجودة لكل منهما؛ ولمَ يُظهر القرص المدمج قوسي قزح ويُظهر الآخر واحدًا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:5}
\emph{نقش $N$ شقًّا.} (أ) بيّن أن بين موضعي عظم رئيسيين
$N - 1$ صفرًا و $N - 2$ موضع عظم ثانويًّا. (ب) ومن أجل $N$ الكبيرة،
أوجد شدة أول موضع \omterm{thm:b2:gratings:nwaves}{عظم ثانوي} بالنسبة إلى الرئيسي
(عند $N\varphi/2 \approx 3\pi/2$): أي نحو $1/22$. (ج) وارسم النقش من أجل
$N = 3$ و $N = 6$ وأعطِ ارتفاعات مواضع العظم الثانوية من أجل
$N = 3$. (د) والنسبة من الطاقة الكلية في مواضع العظم الرئيسية من أجل
$N$ الكبيرة (قارن تكاملات القمة، وعرضها $\propto 1/N$ وارتفاعها
$\propto N^2$، بالباقي).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:6}
\emph{تراكب الرتب.} (أ) بيّن أن الرتبة $p$ من $\lambda_{\max}$ تتراكب مع
الرتبة $p + 1$ من $\lambda_{\min}$ حين $p\lambda_{\max} > (p + 1)\lambda_{\min}$؛ ومن أجل $400$--\qty{700}{nm}، من أي رتبة؟ (ب) \omterm{prop:b2:gratings:fabryperot}{والمجال الطيفي الحر} في الرتبة $p$: $\Delta\lambda_{\text{حر}}
= \lambda/p$. (ج) ويعمل مطياف في الرتبة $20$ قرب \qty{500}{nm} (أي
شبكة إشيل): أوجد \omterm{prop:b2:gratings:fabryperot}{المجال الطيفي الحر}؛ وكيف يُزال التراكب
(بمبدِّد متقاطع)؟ (د) \omterm{prop:b2:gratings:fabryperot}{والمجال الطيفي الحر} لفابري--بيرو هو $c/2L$:
اكتبه بدلالة طول الموجة وقارنه بمجال الشبكة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:7}
\emph{شبكة موجَّهة.} لشبكة انعكاس أخاديد مائلة بزاوية
التوجيه $\beta$ بحيث يعكس كل وجيه انعكاسًا مرآويًّا في
الجهة التي تُطلَب فيها الرتبة $p$. (أ) ومن أجل تركيب ليترو
($\theta = \theta_0 = \beta$): بيّن أن $2a\sin\beta = p\lambda$؛ وأوجد زاوية التوجيه من أجل
\qty{500}{nm} في الرتبة الأولى مع $1200$ خط في المليمتر. (ب) ولمَ يرسل التوجيه معظم
الضوء إلى رتبة واحدة (فكّر في غلاف حيود وجيه واحد،
\cref{ch:b2:diffraction})؟ (ج) وتُوجَّه الشبكة نفسها
من أجل \qty{1}{\micro m}: ففي أي رتبة تكون فعّالة عند \qty{500}{nm}؟ (د)
وأوجد قدرة تمييز شبكة \qty{10}{cm} في تركيب ليترو عند $\beta = \qty{64}{\degree}$،
و $\lambda = \qty{500}{nm}$ (استعمل $R = 2W\sin\beta/\lambda$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:8}
\emph{الشقّ.} يُصوَّر شقّ دخول مطياف عرضه $s$
على الكاشف بتكبير $1$؛ وتبدّد الشبكة ($600$ خط في المليمتر،
و \qty{5}{cm}، والرتبة $1$، و $f = \qty{50}{cm}$). (أ) أوجد عرض خط على
الكاشف من الشقّ وحده من أجل $s = \qty{50}{\micro m}$، بالنانومتر. (ب)
والعرض الآتي من قدرة تمييز الشبكة. (ج) وأيّهما يغلب؛ وعرض الشقّ
الذي يتساويان عنده. (د) ولمَ لا يُضيَّق الشقّ أكثر؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:9}
\emph{فابري--بيرو.} مرآتان $R = 0.95$، و $L = \qty{5}{mm}$، و $\lambda = \qty{600}{nm}$.
(أ) أوجد الرتبة $p$، \omterm{prop:b2:gratings:fabryperot}{والمجال الطيفي الحر} بالتواتر وبطول الموجة. (ب)
والنقاوة؛ وعرض قمة نفاذ بالتواتر وبطول الموجة. (ج)
وقدرة التمييز $p\mathcal F$؛ وقارنها بشبكة \qty{10}{cm}. (د) ولمَ
يُستعمل فابري--بيرو دائمًا مع مرشّح سابق أو شبكة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:10}
\emph{حدّ الشبكة.} (أ) بيّن أن $R = pN = W(\sin\theta - \sin\theta_0)
/\lambda \le 2W/\lambda$: أي إن قدرة التمييز لا تتجاوز عرض
الشبكة بأطوال الموجة، مضروبًا في اثنين. (ب) وفسّر: أي أكبر فرق
مسير بين الشعاعين الطرفيين. (ج) وشبكة \qty{30}{cm} عند
\qty{500}{nm}: أوجد $R$ الأقصى؛ وأصغر سرعة يمكن تمييزها بانزياح دوبلر
($\Delta\lambda/\lambda = v/c$). (د) ولمَ يبلغ الفلكيون مع ذلك \qty{1}{m/s}
(فكّر في قياس موضع خط بكسر من عرضه،
مع خطوط كثيرة)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:11}
\emph{براغ.} البلورة شبكة ثلاثية الأبعاد: فالأشعة السينية
المنعكسة على المستويات الذرية المتتالية المتباعدة $d$، عند
الزاوية المماسّة $\theta$، تتداخل تداخلًا بنّاءً حين $2d\sin\theta = p\lambda$
(براغ). (أ) استنتج ذلك من \omterm{thm:b2:two-wave-interference:formula}{فرق المسير} بين مستويين.
(ب) وإشعاع النحاس K$\alpha$ (\qty{0.154}{nm}) على ملح الطعام ($d = \qty{0.282}{nm}$):
أوجد زوايا براغ. (ج) ولمَ لا تعطي أطوال الموجة المرئية انعكاس براغ
من البلورات، وماذا يقول العكس (أي غياب حيود الأشعة
السينية من الزجاج) عن الزجاج؟ (د) ولمَ يجب أن يكون $\lambda < 2d$؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:12}
\emph{رصّة طورية.} $N$ هوائيًّا متطابقًا على استقامة، متباعدة $a$، مغذّاة
بالسعة نفسها وبدرجة طور $\psi$ بين المتجاورين. (أ)
بيّن أن الحقل المشعّ في الجهة $\theta$ هو حقل مجموع
$N$ موجة مع $\varphi = 2\pi a\sin\theta/\lambda - \psi$: فالحزمة الرئيسية تشير
حيث $\sin\theta = \lambda\psi/2\pi a$. (ب) وأوجد نصف عرض الحزمة الرئيسية، $\Delta\theta
\approx \lambda/Na\cos\theta$. (ج) ولمَ يجب أن يكون $a \le \lambda/2$ لتفادي حزمة رئيسية ثانية
(أي فصّ شبكي) عند التوجيه؟ (د) ورادار فيه $64$ عنصرًا على
$\lambda/2$، و $\lambda = \qty{3}{cm}$: أوجد عرض الحزمة؛ وزمن مسح $90^\circ$ بخطوات عرض
حزمة واحد إذا تبدّلت الأطوار في \qty{1}{\micro s}.
\end{exercise}

\section{مسألة: مطياف لنجم}

\begin{problem}[{مسألة نهاية الأسبوع --- تصميم المطياف الذي
يقرأ ضوء نجم: الشبكة، والكاشف، وخطوط
الهدروجين، وترنّح كوكب}]
\label{pb:b2:gratings:1}

مطياف: شقّ دخول، ومجمِّع بعده البؤري $f_1 =
\qty{1.0}{m}$، وشبكة انعكاس فيها $1200$ خط في المليمتر وعرضها $W =
\qty{12}{cm}$، وآلة تصوير $f_2 = \qty{1.0}{m}$، وكاشف ببكسلات \qty{15}{\micro m}.
وهو يعمل في الرتبة الأولى، حول \qty{550}{nm}، والشبكة
عند ورود \qty{20}{\degree}.

\medskip
\noindent\textbf{الجزء الأول --- الشبكة.}
\begin{enumerate}
  \item أوجد دور الشبكة؛ وعدد الخطوط المضاءة إذا غطّت الحزمة
        العرض كله.
  \item وزاوية حيود \qty{550}{nm} في الرتبة الأولى؛ وزاويتي
        \qty{400}{nm} و \qty{700}{nm}؛ والعرض الزاوي للطيف
        المرئي.
  \item \omterm{prop:b2:gratings:equation}{والتبديد الزاوي} عند \qty{550}{nm}؛ والتبديد الخطي على
        الكاشف بالنانومتر في المليمتر وبالنانومتر في البكسل.
  \item وقدرة التمييز؛ وأصغر $\Delta\lambda$ مميَّز عند \qty{550}{nm}؛
        وبالبكسلات.
  \item أيقع بنفسجي الرتبة الثانية على مرئي الرتبة
        الأولى؟ وأي مرشّح يزيله؟
  \item وطول الكاشف اللازم للطيف المرئي كله في
        الرتبة الأولى.
  \item والشبكة موجَّهة عند \qty{17}{\degree}: فمن أجل أي طول موجة
        تكون أشدّ فعالية في الرتبة الأولى عند هذا الورود (استعمل
        تقدير ليترو $2a\sin\beta = \lambda$)؟
  \item وتعيد الشبكة $80\%$ من الضوء عند طول موجة التوجيه
        ويعيد المقراب مع البصريات $40\%$: أوجد النسبة من فوتونات النجم
        التي تبلغ الكاشف.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الخطوط.}
\begin{enumerate}[resume]
  \item خطوط بالمر الهدروجينية في النجم: H$\alpha$ \qty{656.3}{nm}،
        و H$\beta$ \qty{486.1}{nm}، و H$\gamma$ \qty{434.0}{nm}: أوجد مواضعها على
        الكاشف بالنسبة إلى \qty{550}{nm}.
  \item وخط H$\alpha$ الآتي من ذرة ديوتيريوم أقصر بمقدار \qty{0.18}{nm}: أيُفصَل؟
        وبكم بكسلًا؟
  \item وثنائي الصوديوم في طيف النجم: أوجد الفصل
        بالبكسلات؛ وعرض كل خط إذا عرّضه جوّ النجم
        إلى \qty{0.05}{nm}.
  \item وشقّ الدخول عرضه \qty{100}{\micro m}، مصوَّرًا بتكبير
        $f_2/f_1$: أوجد عرضه على الكاشف بالبكسلات
        وبالنانومتر؛ أيكون المطياف محدودًا بالشقّ أم بالشبكة؟
  \item والضوء الآتي من المقراب قرص قطره \qty{1}{''} على
        السماء، يجعله المقراب ($f = \qty{20}{m}$) بعرض \qty{100}{\micro m}
        عند الشقّ: أوجد النسبة الضائعة من الضوء إذا ضُيّق الشقّ
        إلى \qty{50}{\micro m} (خذ القرص منتظمًا).
  \item ولمَ يعطي شقّ أضيق خطوطًا أصفى وأطيافًا أكثر
        ضجيجًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- النجم يتحرك.}
\begin{enumerate}[resume]
  \item نجم يبتعد بسرعة \qty{30}{km/s}: أوجد انزياح دوبلر لخط H$\alpha$ بالنانومتر
        وبالبكسلات.
  \item وتضيف حركة الأرض المدارية $\pm\qty{30}{km/s}$ على مدار السنة:
        فلمَ يجب طرحها، وبأي دقة إذا أُريدت
        \qty{100}{m/s}؟
  \item وكوكب يجعل نجمه يترنّح بسرعة \qty{10}{m/s}: أوجد الانزياح بالنانومتر
        وبالبكسلات؛ أهو دون قدرة التمييز؟ وفسّر كيف يستعيده قياس
        مركز الخط بجزء من مئة من عرضه، على
        ألف خط (بربح إحصائي $\sqrt{N}$).
  \item وتتغير حرارة الأداة \qty{1}{K}، فتتمدد
        الشبكة (وهي زجاج، تمدده \qty{1e-5}{K^{-1}}): أوجد انزياح
        الخطوط بالبكسلات؛ وأي استقرار يلزم من أجل \qty{10}{m/s}؟
  \item ومصباح معايرة (ثوريوم--أرغون) يضع مئات الخطوط
        المعلومة على الكاشف نفسه: فسّر دوره.
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- البدائل.}
\begin{enumerate}[resume]
  \item الشبكة نفسها عرضًا محفورة بواقع $300$ خط في المليمتر ومستعملة في
        الرتبة الرابعة (مع مبدِّد متقاطع لفرز الرتب):
        أوجد زاوية الحيود، وقدرة التمييز، والتبديد، \omterm{prop:b2:gratings:fabryperot}{والمجال
        الطيفي الحر} عند \qty{550}{nm}؛ وما الذي تغير، وما الذي لم يتغير؟
  \item ومعيار فابري--بيرو ($L = \qty{1}{mm}$، $R = 0.9$) موضوع قبل
        الشقّ: أوجد \omterm{prop:b2:gratings:fabryperot}{المجال الطيفي الحر} بالنانومتر، والنقاوة، وعرض القمة؛ وكيف
        يبدو مشطُ قمم نفاذه على الكاشف،
        وفيمَ يُستعمل؟
  \item وموشور \qty{60}{\degree} من زجاج فيه $\dd n/\dd\lambda = \qty{-1e-4}{nm^{-1}}$
        له \omterm{prop:b2:gratings:equation}{التبديد الزاوي} $\dd D/\dd\lambda \approx 2\,\dd n/\dd\lambda$ قرب
        أصغر انحراف: قارنه بتبديد الشبكة؛ ولمَ حلّت الشبكات
        محلّ المواشير؟
  \item ولمَ تُبنى أكبر المطيافات الفلكية بشبكات
        بعرض متر وتُستعمل عند زوايا كبيرة (أي الحدّ
        $2W/\lambda$)؟
  \item ونجم قدره ثمانية يعطي \qty{1e4}{} فوتون في الثانية
        وفي النانومتر عند المقراب: أوجد الفوتونات في البكسل في الثانية
        على الكاشف، ونسبة الإشارة إلى الضجيج في تعريض \qty{100}{s}
        (بضجيج الفوتونات وحده).
  \item ولخّص: الأعداد الثلاثة التي تصف المطياف
        (التبديد، وقدرة التمييز، \omterm{prop:b2:gratings:fabryperot}{والمجال الطيفي الحر}) وما الذي يحدد كلًّا منها.
\end{enumerate}
\end{problem}
