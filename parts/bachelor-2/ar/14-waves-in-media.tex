\chapter{الأمواج الكهرمغناطيسية في البلازما والنواقل والعوازل}\label{ch:b2:waves-in-media}

يعبر بثّ على الموجة القصيرة محيطًا بارتداده عن أعالي
الجو، فيعكسه كالمرآة --- ومع ذلك تمرّر الطبقة نفسها
إشارات التلفزة والسواتل. وملعقة معدنية لامعة،
وشبكة معدنية في باب فرن الموجات الصغرية تحبس الأمواج بينما
تنظر إلى طعامك عبرها، وصفيحة زجاج شفافة
للضوء وعاتمة على فوق البنفسجي. وفي كل حالة تدخل موجة
مادةً، فتستجيب شحن المادة، وتغير استجابتها
الموجة: سرعتها ووهنها، بل حتى أتنتشر
أصلًا. وهذا الفصل يعالج النماذج الثلاثة المعيارية للوسط --- أي
غاز إلكترونات حرة، وناقل أومي، وإلكترونات مرتبطة --- بالطريقة
الواحدة التي تعالجها جميعًا: أضف التيار المحرَّض إلى
\omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} واقرأ \omterm{def:b2:dispersion-wave-packets:relation}{علاقة التبدد}.

\section{الأمواج في بلازما}

\begin{definition}[البلازما؛ نموذج الإلكترونات الحرة]\label{def:b2:waves-in-media:plasma}
\index{بلازما}\index{تواتر البلازما}
\emph{البلازما} غاز متعادل إجمالًا من إلكترونات حرة (كثافتها
$n$، وكتلتها $m$، وشحنتها $-e$) وشوارد موجبة، وهي، لكونها أثقل
بآلاف المرات، تُؤخذ ثابتة. وأعالي الجو (أي الأيونوسفير،
و $n \sim 10^{11}$--$10^{12}$ m$^{-3}$)، وأنبوب تفريغ، وإكليل الشمس،
و --- عند التواترات الضوئية --- إلكترونات ناقلية معدن،
كلها بلازما. وتُهمَل التصادمات (فدور الموجة قصير
بالقياس إلى الزمن بين تصادمين). وتواترها المميز
هو \emph{تواتر البلازما}
\[
\omega_p = \sqrt{\frac{ne^2}{\varepsilon_0m}} , \qquad
f_p = \frac{\omega_p}{2\pi} \approx 9\,\text{Hz}\times\sqrt{n\,[\text{m}^{-3}]} :
\]
أي \qty{3}{MHz} إلى \qty{9}{MHz} للأيونوسفير، و \qty{2e15}{Hz} (فوق بنفسجي) في
المعدن.
\end{definition}

\begin{theorem}[علاقة تبدد بلازما]\label{thm:b2:waves-in-media:plasmadispersion}
\index{تواتر القطع (بلازما)}\index{موجة متلاشية (بلازما)}
من أجل موجة مستوية متقدمة توافقية مستعرضة $\underline{\vect E} = \underline{\vect E}_0\eu^{\iu(\omega t - kx)}$ في
\omterm{def:b2:waves-in-media:plasma}{بلازما}، تتذبذب الإلكترونات بالسرعة $\underline{\vect v} = -e\underline{\vect E}/\iu m\omega$
وتحمل \omterm{def:b2:charges-currents-conduction:densities}{كثافة التيار} $\underline{\vect j} = \underline\gamma\underline{\vect E}$ مع
$\underline\gamma = ne^2/\iu m\omega$؛ وتعطي \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} عندئذ
\[
k^2 = \frac{\omega^2 - \omega_p^2}{c^2} .
\]
وفوق $\omega_p$ تنتشر الموجة، مع $v_\varphi = c/\sqrt{1 - \omega_p^2/\omega^2}
> c$ و $v_g = c\sqrt{1 - \omega_p^2/\omega^2} < c$، و $v_\varphi v_g = c^2$؛ ودون
$\omega_p$ يكون $k = -\iu\kappa$ مع $\kappa = \sqrt{\omega_p^2 - \omega^2}/c$: فالموجة
\emph{متلاشية}، تنفذ على مدى $1/\kappa$ وتنعكس انعكاسًا تامًّا
(\cref{ch:b2:wave-interfaces})؛ وتكون \omterm{def:b2:waves-in-media:plasma}{البلازما} عندئذ مرآة. وعند $\omega
\gg \omega_p$ لا تستطيع الإلكترونات المتابعة وتكون \omterm{def:b2:waves-in-media:plasma}{البلازما} شفافة.
\end{theorem}

\begin{proof}
معادلة الحركة $m\,\dd\vect v/\dd t = -e\vect E$ (فالقوة المغناطيسية
$ev B \sim evE/c$ مهمَلة من أجل $v \ll c$، وحركة الشوارد
أصغر بالمعامل $m/M$): ومنه $\iu m\omega\underline{\vect v} = -e\underline{\vect E}$. ثم
$\underline{\vect j} = -ne\underline{\vect v}$. وتبقى \omterm{def:b2:waves-in-media:plasma}{البلازما} متعادلة من أجل موجة
مستعرضة ($\vect k\cdot\vect E = 0$ يعطي $\rho = 0$ بغاوس)، ومنه
يُقرأ ماكسويل--أمبير $-\iu\vect k\wedge\underline{\vect B} = \mu_0\underline\gamma\underline{\vect E}
+ \iu\omega\mu_0\varepsilon_0\underline{\vect E}$ ويُقرأ ماكسويل--فاراداي $\underline{\vect B} = \vect k
\wedge\underline{\vect E}/\omega$؛ وبالجمع، $k^2\underline{\vect E} = \mu_0(\iu\omega\underline\gamma -
\omega^2\varepsilon_0)(-1)\underline{\vect E}$، أي $k^2 = \omega^2\mu_0\varepsilon_0 - \iu\omega\mu_0
\underline\gamma = (\omega^2 - ne^2/\varepsilon_0m)/c^2$. وتنتج السرعات كما في
\cref{ch:b2:dispersion-wave-packets}.
\end{proof}

\begin{remark}[أين تذهب الطاقة]\label{rem:b2:waves-in-media:plasmaenergy}
الناقلية تخيلية: ومنه $\langle\vect j\cdot\vect E\rangle = 0$، فتأخذ
الإلكترونات الطاقة من الموجة نصف دورة وتعيدها
في النصف التالي --- أي وسطًا بلا ضياع، تتوزع فيه طاقة الموجة
بين المجالات والطاقة الحركية للإلكترونات، وتنتقل
بالسرعة $v_g$. وتضيف التصادمات جزءًا حقيقيًّا صغيرًا إلى $\underline\gamma$ و
امتصاصًا صغيرًا: فطبقة D في الأيونوسفير تمتصّ الأمواج المتوسطة
نهارًا، ولهذا تُلتقط محطات التضمين السعوي البعيدة ليلًا.
\end{remark}

\begin{example}[الأيونوسفير والراديو]\label{ex:b2:waves-in-media:ionosphere}
مع $n = \qty{1e12}{m^{-3}}$ في الطبقة F يكون $f_p = \qty{9}{MHz}$: فبثوث الموجة
القصيرة دونه تنعكس (ويمكن أن تقفز حول الأرض
بين الأيونوسفير والأرض)، وأما التضمين الترددي (\qty{100}{MHz}) والتلفزة
ووصلات السواتل فتمرّ. وموجة عند \qty{3}{MHz} لا تنفذ إلا
$1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{6}{m}$ في الطبقة قبل أن
تعود. ومركبة فضائية تعود إلى الجو تُغلَّف
\omterm{def:b2:waves-in-media:plasma}{ببلازما} $n \sim 10^{18}$ m$^{-3}$ و $f_p \sim \qty{10}{GHz}$: فيُفقد الاتصال الراديوي
دقائق.
\end{example}

\section{الأمواج في ناقل أومي: الأثر القشري}

\begin{theorem}[الأثر القشري]\label{thm:b2:waves-in-media:skin}
\index{أثر قشري}\index{عمق القشرة}
في ناقل أومي ($\vect j = \gamma\vect E$، و $\gamma$ حقيقية، عند تواترات
يكون فيها $\omega\tau \ll 1$ و $\varepsilon_0\omega \ll \gamma$، أي دون $\sim\qty{1e16}{Hz}$ في
النحاس) يكون \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة} مهمَلًا وتحقق المجالات
معادلة الانتشار $\Delta\vect E = \mu_0\gamma\,\partial_t\vect E$. ولموجة مستوية متقدمة توافقية تدخل
الناقل عند $x = 0$
\[
\underline k^2 = -\iu\mu_0\gamma\omega , \qquad \underline k = \frac{1 - \iu}\delta , \qquad
\delta = \sqrt{\frac2{\mu_0\gamma\omega}} ,
\]
ومنه $\underline E = E_0\,\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$: فهي تخمد
على \emph{\omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة}} $\delta$، وطورها يدور كذلك بمقدار راديان
واحد في كل $\delta$ --- أي طول موجة $2\pi\delta$، وسرعة طور $\omega\delta \ll
c$. وينحصر التيار $\vect j = \gamma\vect E$ في طبقة سماكتها
$\delta$ تحت السطح، ويتأخر $\vect B$ عن $\vect E$ بمقدار \qty{45}{\degree} مع
$|B| = |E|\sqrt2/\omega\delta$. وفي النحاس، $\delta = \qty{9.2}{mm}$ عند \qty{50}{Hz}،
و \qty{0.21}{mm} عند \qty{100}{kHz}، و \qty{2.1}{\micro m} عند \qty{1}{GHz}.
\end{theorem}

\begin{proof}
$\operatorname{\vect{curl}}\vect B = \mu_0\gamma\vect E$ (بإسقاط \omterm{thm:b2:maxwell-equations:maxwell}{تيار الإزاحة}) و
$\operatorname{\vect{curl}}\vect E = -\partial_t\vect B$: وبأخذ دوّار الثانية،
$-\Delta\vect E = -\mu_0\gamma\partial_t\vect E$ (فالناقل متعادل، و $\operatorname{div}
\vect E = 0$). ومن أجل $\eu^{\iu(\omega t - kx)}$: $k^2 = -\iu\mu_0\gamma\omega$، و
$\sqrt{-\iu} = (1 - \iu)/\sqrt2$ يعطي $\underline k = (1 - \iu)/\delta$ (وهو الجذر الذي
يخمد داخل الناقل). وبفاراداي: $\underline B = \underline k\underline E/\omega = (1 - \iu)
\underline E/\omega\delta$، ومقياسه $\sqrt2E/\omega\delta$ وطوره $-\pi/4$.
\end{proof}

\begin{example}[نتائج الأثر القشري]\label{ex:b2:waves-in-media:skinuses}
\index{مقاومة متناوبة}\index{قفص فاراداي}
(1) سلك نصف قطره $a \gg \delta$ يحمل تياره في حلقة
سماكتها $\delta$: فترتفع مقاومته من $1/\gamma\pi a^2$ إلى نحو $1/\gamma
2\pi a\delta$ في المتر --- ومن أجل سلك نحاسي قطره \qty{1}{mm}، أربعة أضعاف القيمة
المستمرة عند \qty{1}{MHz}، ومئة ضعف عند \qty{1}{GHz}. وتستعمل وشائع التواتر العالي
سلك «ليتز» (وهو خيوط كثيرة معزولة) أو أنابيب مطلية بالفضة؛
وتُجدَل نواقل شبكة الكهرباء بغلاف من الألومنيوم. (2) و
صندوق معدني يوهن موجة بالعامل $\eu^{-d/\delta}$ لكل سماكة $d$: فحاوية
ألومنيوم سماكتها \qty{1}{mm} تحجب \qty{1}{MHz} بمقدار \qty{100}{dB} لكنها لا تحجب \qty{50}{Hz}
إلا بالكاد --- فالمجالات المغناطيسية المنخفضة التواتر تحتاج إلى الحديد. (3)
ولا تتلقى الغواصات إلا الراديو الطويل الموجة جدًّا ($\delta = \qty{2.3}{m}$ عند
\qty{10}{kHz} في ماء البحر)؛ وتُكشف الألغام بالتيارات الدوّامية التي
تحرّضها وشيعة فيها (\cref{ch:b2:maxwell-equations}، وزمن الانتشار
$\mu_0\gamma d^2$).
\end{example}

\section{الأمواج في عازل: الإلكترون المرتبط}

\begin{proposition}[الإلكترون المرتبط ارتباطًا مرنًا؛ القرينة العقدية]\label{prop:b2:waves-in-media:lorentz}
\index{عازل}\index{استقطاب (وسط)}\index{قابلية}\index{قرينة عقدية}\index{نموذج لورنتس}
في عازل يرتبط كل إلكترون بذرّته: فانتقال
$\vect r$ عن التوازن يستدعي قوة إرجاع $-m\omega_0^2\vect r$ و
تخامدًا $-m\Gamma\,\dd\vect r/\dd t$ (وهو الطاقة المشعّة أو المسلَّمة إلى
الشبكة). ومقودًا بالموجة، $m\ddot{\vect r} = -m\omega_0^2\vect r - m\Gamma\dot{\vect r}
- e\vect E$؛ وتعطي الإلكترونات المنتقلة الوسطَ \emph{استقطابًا}
(أي عزم ثنائي قطب في واحدة الحجم) $\vect P = -ne\vect r = \varepsilon_0\underline\chi(\omega)
\underline{\vect E}$ مع \emph{القابلية}
\[
\underline\chi(\omega) = \frac{\omega_p^2}{\omega_0^2 - \omega^2 + \iu\Gamma\omega} ,
\qquad \omega_p^2 = \frac{ne^2}{\varepsilon_0m} ,
\]
والتيار المرتبط $\vect j = \partial_t\vect P$. وتعطي \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} عندئذ
\[
\underline k^2 = \frac{\omega^2}{c^2}\,(1 + \underline\chi) = \frac{\omega^2}{c^2}\,\underline n^2 ,
\qquad \underline n = n' - \iu n'' ,
\]
حيث $n'$ قرينة الانكسار (\omterm{def:b2:dispersion-wave-packets:relation}{وسرعة الطور} $c/n'$) و $n''$
الإخماد (وتخمد الشدة مثل $\eu^{-2n''\omega x/c}$). وبعيدًا دون
رنين ($\omega \ll \omega_0$، و $\Gamma$ مهمَل)، $n'^2 \approx 1 + \omega_p^2/\omega_0^2 +
\omega_p^2\omega^2/\omega_0^4$: فالقرينة \emph{ترتفع} مع التواتر --- وهو التبدد
العادي، وقانون كوشي $n \approx A + B/\lambda^2$ --- والامتصاص
مهمَل؛ وبجوار $\omega_0$ يمتصّ الوسط، وفوقه مباشرةً تهبط $n'$
مع $\omega$ (وهو التبدد الشاذّ).
\end{proposition}

\begin{proof}
بالسعات العقدية: $(\omega_0^2 - \omega^2 + \iu\Gamma\omega)\underline{\vect r} = -e\underline{\vect E}
/m$؛ و $\underline{\vect P} = -ne\underline{\vect r}$. ثم، كما في \omterm{def:b2:waves-in-media:plasma}{البلازما}، $k^2 = \omega^2\mu_0
\varepsilon_0 - \iu\omega\mu_0\underline\gamma$ مع $\underline{\vect j} = \iu\omega\underline{\vect P} = \iu\omega\varepsilon_0
\underline\chi\underline{\vect E}$، أي $\underline\gamma = \iu\omega\varepsilon_0\underline\chi$: ومنه $k^2 = (\omega^2/c^2)
(1 + \underline\chi)$. (\omterm{def:b2:waves-in-media:plasma}{والبلازما} هي الحالة $\omega_0 = 0$ و $\Gamma = 0$.)
والنشر: $1/(\omega_0^2 - \omega^2) \approx (1 + \omega^2/\omega_0^2)/\omega_0^2$.
\end{proof}

أبطأ من الأحمر في الزجاج)، ويبلغ الجزء التخيلي ذروته عند
الرنين (أي الامتصاص)، وتهبط القرينة عبر الرنين
(وهو التبدد الشاذّ).}
\end{omfigure}

\begin{example}[الزجاج والماء والهواء]\label{ex:b2:waves-in-media:dielectrics}
للزجاج رنيناته الإلكترونية في فوق البنفسجي ($\omega_0 \sim 10^{16}$
راديان في الثانية): فهو في المرئي شفاف ومتبدّد، و $n = 1.51$ عند
\qty{700}{nm}، و $1.53$ عند \qty{400}{nm}، وهو ما يكفي لموشور كي ينشر
قوس قزح ولنبضات ليف كي تتمدد (\cref{ch:b2:dispersion-wave-packets})؛
وفي فوق البنفسجي يمتصّ --- فلا تسمرّ خلف نافذة.
وللماء، فوق ذلك، رنين دوراني لجزيئاته القطبية بجوار
\qty{20}{GHz} (وهو استرخاء ديباي لا خط حادّ): فيمتصّ
الموجات الصغرية بشدة، وذلك يسخّن الطعام ويعمي الرادار في المطر،
وله $n'' \approx 0$ في نافذة مرئية ضيّقة --- وهي النافذة التي
تطوّرت العيون عبرها. وأما الهواء، و $n - 1 = 2.9 \times 10^{-4}$، فما زال يحني
الشمس الغاربة بنصف درجة.
\end{example}

\begin{remark}[معادلة واحدة وثلاثة أوساط]\label{rem:b2:waves-in-media:summary}
يدخل كل وسط خطي \omterm{thm:b2:maxwell-equations:maxwell}{معادلات ماكسويل} عبر تياره
المحرَّض $\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$، وتكون \omterm{def:b2:dispersion-wave-packets:relation}{علاقة التبدد}
دائمًا $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma(\omega)$:
\begin{center}\small
\begin{tabular}{lll}
الوسط & $\underline\gamma(\omega)$ & $\underline k^2$\\ \hline
\omterm{def:b2:waves-in-media:plasma}{بلازما} & $ne^2/\iu m\omega$ (تخيلية) & $(\omega^2 - \omega_p^2)/c^2$: تواتر قطع\\
ناقل أومي & $\gamma$ (حقيقية) & $-\iu\mu_0\gamma\omega$: \omterm{thm:b2:waves-in-media:skin}{أثر قشري}\\
عازل & $\iu\omega\varepsilon_0\underline\chi(\omega)$ & $(\omega^2/c^2)(1 + \underline\chi)$: قرينة
\end{tabular}
\end{center}
والمعدن هو الثلاثة تباعًا: أومي دون \qty{1e13}{Hz}، \omterm{def:b2:waves-in-media:plasma}{وبلازما}
(عاكسة) في المرئي، وشفاف في فوق البنفسجي البعيد.
\end{remark}

\begin{method}[الأمواج في وسط]\label{met:b2:waves-in-media:method}
(1) اكتب معادلة حركة الشحن واحصل على $\underline{\vect j} =
\underline\gamma\underline{\vect E}$. (2) وأدخلها في ماكسويل--أمبير مع افتراض الموجة
المستوية المتقدمة التوافقية: $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma$. (3) وخذ الجذر
الذي $\operatorname{Im}k \le 0$ (أي الخمود في جهة السير)، وافصل
الجزء الحقيقي (الانتشار، و $v_\varphi = \omega/k'$) عن التخيلي (\omterm{def:b2:dispersion-wave-packets:complex}{الوهن}).
(4) وميّز الأنظمة في التواتر --- منتشر، أو
متلاشٍ، أو ماصّ --- والتواتر المميز ($\omega_p$،
أو $1/\mu_0\gamma\delta^2$، أو $\omega_0$). (5) وأوجد $\vect B$ من فاراداي، وعند الحاجة
تدفق بوينتنغ والاستطاعة المبدَّدة.
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:3}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:4}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:5}
\emph{\omterm{ex:b2:waves-in-media:skinuses}{المقاومة المتناوبة}.} سلك نحاسي نصف قطره $a = \qty{1.0}{mm}$. (أ) أوجد المقاومة
المستمرة في المتر. (ب) وعند \qty{1}{MHz}: أوجد $\delta$؛ والمقاومة التقريبية
في المتر مع انحصار التيار في حلقة سماكتها $\delta$؛
والنسبة إلى المستمرة. (ج) وعند \qty{100}{MHz}. (د) ولماذا يكون لوشيعة من $N$ لفة من
هذا السلك عامل جودة يكفّ عن التحسن مع $N$ عند التواتر
العالي، وما سلك ليتز؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:6}
\emph{التدريع.} صندوق من الألومنيوم ($\gamma = \qty{3.6e7}{S/m}$) سماكته \qty{1.0}{mm}.
أوجد \omterm{def:b2:dispersion-wave-packets:complex}{وهن} سعة المجال عبر الجدار، \omterm{def:b2:sound-waves:decibel}{بالديسيبل}، عند
\qty{50}{Hz} و \qty{10}{kHz} و \qty{1}{MHz} (بالعامل $\eu^{-d/\delta}$، وبإهمال الانعكاس عند
السطحين). ولماذا يُدرَّع مجال الشبكة المغناطيسي
بالحديد (العالي النفوذية) لا بالنحاس؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:7}
\emph{الطاقة في \omterm{def:b2:waves-in-media:plasma}{بلازما}.} من أجل الموجة المستعرضة في
\cref{thm:b2:waves-in-media:plasmadispersion}: (أ) أوجد سعة سرعة
الإلكترونات وكثافة الطاقة الحركية $\tfrac12nm\langle v^2\rangle$. (ب)
وكثافتي الطاقة الكهربائية والمغناطيسية الوسطيتين. (ج) وبيّن أن كثافة
الطاقة الكلية هي $\tfrac12\varepsilon_0E_0^2(1 + \dots)$ --- واحسب القوس ---
وأن $\langle\Pi\rangle$ مقسومًا عليها يساوي $v_g$. (د) وأي نسبة من
الطاقة تكون في الإلكترونات عند $\omega = 2\omega_p$؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:8}
\emph{تبدد النوابض.} تعبر نبضات راديوية من نابض مسافة
$L$ من \omterm{def:b2:waves-in-media:plasma}{بلازما} بين نجمية ($n \ll$ أي شيء، و $\omega \gg \omega_p$). (أ) بيّن
أن تأخّر المجموعة هو $t \approx (L/c)(1 + \omega_p^2/2\omega^2)$. (ب) والتأخّر
بين الوصول عند $f_1$ و $f_2$ هو $\Delta t = \dfrac{e^2}{8\pi^2
\varepsilon_0mc}\,nL\Bigl(\dfrac1{f_1^2} - \dfrac1{f_2^2}\Bigr)$: احسب الثابت.
(ج) ونابض على $L = \qty{1}{kpc} = \qty{3.1e19}{m}$ عبر $n = \qty{3e4}{m^{-3}}$:
أوجد التأخّر بين \qty{400}{MHz} و \qty{1.4}{GHz}. (د) ويقيس الفلكيون
$\Delta t$ ويعرفون $n$: فماذا يحصلون عليه؟ (وهو مقياس التبدد، وهو
مقياس البعد المعياري للنوابض.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:9}
\emph{خط طيفي.} في غاز مخلخل ($\underline n \approx 1 + \underline\chi/2$) يعطي
\omterm{prop:b2:waves-in-media:lorentz}{نموذج لورنتس} $n'' = \tfrac12\operatorname{Im}\underline\chi$. (أ) بيّن أنه بجوار
$\omega_0$ يكون $n'' \approx \dfrac{\omega_p^2}{4\omega_0}\,\dfrac{\Gamma/2}{(\omega_0 - \omega)^2 + \Gamma^2/4}$، وهو
منحني لورنتس عرضه الكامل $\Gamma$. (ب) وأوجد معامل الامتصاص $\alpha = 2n''\omega/c$
عند المركز. (ج) وبخار الصوديوم، $n = \qty{1e17}{m^{-3}}$، و $\lambda_0 = \qty{589}{nm}$،
و $\Gamma = \qty{6e7}{s^{-1}}$: أوجد $\alpha$ عند المركز والسماكة التي تمتصّ
$90\%$ (بإهمال التعريض الدوبلري، وهو في الواقع يوسّع الخط
مئة ضعف). (د) وارسم تخطيطيًّا $n' - 1$ عبر الخط؛ وأين يكون
التبدد شاذًّا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:10}
\emph{تسخين جول في القشرة.} موجة $E_0\eu^{\iu\omega t}$ (على امتداد $y$)
تدخل ناقلًا يملأ $x > 0$. (أ) اكتب $\underline E(x, t)$ واستنتج
$\underline B(x, t)$ (على امتداد $z$) من فاراداي. (ب) \omterm{thm:b2:poynting-vector:poynting}{ومتجهة بوينتنغ} الوسطية عند
السطح: بيّن أن $\langle\Pi_x(0)\rangle = E_0^2/2\mu_0\omega\delta$. (ج) واستطاعة جول
الوسطى في واحدة المساحة، $\int_0^\infty\tfrac12\gamma|\underline E|^2\dd x$: بيّن أنها تساوي
(ب). (د) وعبّر عن النتيجة في صورة $\tfrac12R_sH_0^2$ مع $H_0 = B_0/\mu_0$ المجال
المغناطيسي السطحي و $R_s = 1/\gamma\delta$ \emph{المقاومة
السطحية}؛ وأعطِ القيمة للنحاس عند \qty{1}{GHz}، وضياع
جوف موجات صغرية جدرانه تحمل $H_0 = \qty{100}{A/m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:11}
\emph{لماذا تلمع المعادن.} عامِل إلكترونات ناقلية الفضة ($n =
\qty{5.9e28}{m^{-3}}$) \omterm{def:b2:waves-in-media:plasma}{بلازما} بلا تصادمات عند التواترات الضوئية.
(أ) أوجد $\omega_p$ وطول الموجة الموافق. (ب) وعند \qty{500}{nm}: يكون $k$
تخيليًّا --- فأوجد عمق النفوذ؛ وتنعكس الموجة انعكاسًا تامًّا (بلا
امتصاص في هذا النموذج): فالفضة مرآة. (ج) وعند \qty{200}{nm}؟ ومن هنا
«شفافية فوق البنفسجي» في الفلزات القلوية. (د) والفضة الحقيقية
تعكس $95\%$، والذهب يبدو أصفر: فما الذي يغفله النموذج (فكّر في
التصادمات، وفي الإلكترونات المرتبطة التي تقع رنيناتها في
المرئي في الذهب)؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:12}
\emph{الماء في فرن الموجات الصغرية.} يعطي توجّه جزيئات الماء
القطبية قابلية $\underline\chi = \chi_s/(1 + \iu\omega\tau)$ (وهو استرخاء
ديباي)، مع $\chi_s \approx 80$ و $\tau = \qty{8}{ps}$، فوق ثابت
$\chi_\infty \approx 4$. (أ) أوجد الجزأين الحقيقي والتخيلي من $\underline\varepsilon_r = 1 + \chi_\infty
+ \underline\chi$ عند \qty{2.45}{GHz} وعند \qty{20}{GHz}. (ب) \omterm{prop:b2:waves-in-media:lorentz}{والقرينة العقدية} و
طول امتصاص الشدة $1/(2n''\omega/c)$ عند \qty{2.45}{GHz}: فكم
عمقًا يطبخ الفرن؟ (ج) والاستطاعة الممتصّة في واحدة الحجم من أجل
سعة مجال $E_0$ في الماء: $\tfrac12\omega\varepsilon_0\varepsilon''E_0^2$؛ وأعطِ القيمة من أجل
$E_0 = \qty{2}{kV/m}$. (د) ولماذا يُستعمل \qty{2.45}{GHz} لا
\qty{20}{GHz} حيث الامتصاص أعظمي؟
\end{exercise}

\section{مسألة: الأيونوسفير وباب الفرن والموشور}

\begin{problem}[{مسألة نهاية الأسبوع --- ثلاث مواد كما تراها موجة:
البلازما التي تحني إشارات تحديد المواقع، والمعدن الذي يحبس الموجات الصغرية،
والزجاج الذي ينشر الضوء الأبيض}]
\label{pb:b2:waves-in-media:1}

\medskip
\noindent\textbf{الجزء I --- الأيونوسفير وتحديد المواقع.}
كثافة الإلكترونات في الطبقة F هي $n = \qty{1e12}{m^{-3}}$؛ والعمود الكلي من
الإلكترونات على امتداد مسار شاقولي هو $N_T = \int n\,\dd z = \qty{2e17}{m^{-2}}$
نهارًا. ونظام تحديد المواقع: $f_1 = \qty{1575}{MHz}$، و $f_2 = \qty{1228}{MHz}$.
\begin{enumerate}
  \item تواتر \omterm{def:b2:waves-in-media:plasma}{بلازما} الطبقة F؛ وهل يمرّ نظام تحديد المواقع؟ وهل تمرّ
        موجة قصيرة عند \qty{5}{MHz}؟
  \item بيّن أنه من أجل $\omega \gg \omega_p$ تكون \omterm{prop:b2:dispersion-wave-packets:group}{سرعة المجموعة} $v_g \approx c(1 -
        \omega_p^2/2\omega^2)$ \omterm{def:b2:dispersion-wave-packets:relation}{وسرعة الطور} $c(1 + \omega_p^2/2\omega^2)$.
  \item وتأخّر المجموعة الزائد لإشارة تحديد المواقع وهي تعبر الطبقة: بيّن
        أن $\Delta t = \dfrac{e^2}{8\pi^2\varepsilon_0mc}\,\dfrac{N_T}{f^2}$ واحسبه من أجل
        $f_1$؛ وخطأ المدى $c\,\Delta t$.
  \item ويستعمل المستقبل التواترين: فيزيل الخطأ من التأخّرين.
        عبّر عن $N_T$ بدلالة $\Delta t_1 - \Delta t_2$؛
        وأي دقة على فرق التأخّرين تعطي $N_T$ حتى $1\%$؟
  \item وكم دورة حامل زائدة يمثّل تأخّر السؤال 3
        عند $f_1$؟
  \item ويكون طور الحامل \emph{متقدمًا} بالمقدار نفسه الذي
        تتأخر به المجموعة: فلماذا (إشارتا التصحيحين)؟ وأيهما
        يهمّ مستقبلًا يعدّ دورات الحامل؟
  \item وليلًا يهبط $N_T$ عشرة أضعاف: أوجد الخطأ في مستقبل
        وحيد التواتر نهارًا وليلًا.
  \item ويرفع توهج شمسي $n$ في الطبقة D (على \qty{80}{km}) إلى
        \qty{1e10}{m^{-3}} مع تصادمات كثيرة: فأي الخدمات الراديوية
        تتضرر، ولماذا لا يتدهور تحديد المواقع إلا قليلًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- باب الفرن.}
جدران الجوف وشبكة الباب من الفولاذ ($\gamma = \qty{1.0e6}{S/m}$)؛
وثقوب الشبكة عرضها \qty{1.0}{mm} في صفيحة سماكتها \qty{0.5}{mm}؛
و $f = \qty{2.45}{GHz}$، و $P = \qty{800}{W}$.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة} في الفولاذ عند \qty{2.45}{GHz}؛ وهل الجدار «سميك»؟
  \item والمقاومة السطحية $R_s = 1/\gamma\delta$؛ ومع مجال مغناطيسي
        سطحي سعته $H_0 = \qty{30}{A/m}$ على الجدران (\qty{1}{m^2})،
        أوجد الاستطاعة الضائعة في الجدران، $\tfrac12R_sH_0^2$ في واحدة المساحة؛ والنسبة
        من $P$.
  \item وفي ثقب عرضه $a \ll \lambda$ لا يستطيع المجال الانتشار (فالثقب
        دليل موجة دون القطع، \cref{ch:b2:guided-waves}):
        بل يخمد مثل $\eu^{-\pi z/a}$ عبر السماكة $z$. أوجد \omterm{def:b2:dispersion-wave-packets:complex}{وهن}
        السعة عبر الصفيحة ذات \qty{0.5}{mm}، \omterm{def:b2:sound-waves:decibel}{بالديسيبل}.
  \item ولماذا ترى عبر الشبكة (طول موجة الضوء
        مقابل حجم الثقب) بينما لا تستطيع الموجات الصغرية الخروج؟
  \item ونافذة الزجاج عازل $\underline\varepsilon_r = 4 - 0.02\iu$:
        أوجد القرينة، وطول \omterm{def:b2:dispersion-wave-packets:complex}{وهن} الشدة فيه؛ وهل يسخن
        الزجاج؟
  \item وشوكة معدنية في الفرن: يتضخم المجال عند أطرافها؛
        وبعمق القشرة في السؤال 8، قدّر \omterm{def:b2:charges-currents-conduction:densities}{كثافة التيار}
        عند السطح من أجل $H_0 = \qty{300}{A/m}$ (باستعمال $j \approx H_0/\delta$
        عند السطح) والتسخين الموضعي $j^2/\gamma$ --- فلماذا الشرر؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- السلك عند التواتر الراديوي.}
سلك نحاسي نصف قطره $a = \qty{0.5}{mm}$ ($\gamma = \qty{6e7}{S/m}$) في وشيعة
مسخّن تحريضي عند \qty{13.56}{MHz} يحمل \qty{20}{A} فعّالة.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة}؛ والمقطع الناقل الفعّال؛ \omterm{ex:b2:waves-in-media:skinuses}{والمقاومة المتناوبة} في
        المتر مقابل المستمرة.
  \item الاستطاعة المبدَّدة في المتر من سلك الوشيعة؛ ودرجة الحرارة التي
        كان السلك ليبلغها بتبريد حملي $h = \qty{20}{W/(m^2.K)}$.
  \item وتحرّض الوشيعة تيارات في قطعة عمل فولاذية ($\gamma =
        \qty{1e6}{S/m}$، و $\mu_r \approx 1$ وهي ساخنة): أوجد \omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة} هناك؛ ولماذا
        يسخّن التسخين التحريضي طبقة سطحية فقط، وكيف يُستعمل
        ذلك في تقسية التروس؟
  \item ولخفض ضياع الوشيعة يُستبدَل بالسلك أنبوب نحاسي قطره \qty{3}{mm}
        فيه ماء: أوجد المقاومة الجديدة في المتر؛ وهل يحمل
        داخل الأنبوب تيارًا؟
  \item وعند أي تواتر يساوي \omterm{def:b2:dispersion-wave-packets:complex}{عمق القشرة} في النحاس نصف قطر
        السلك؟ وأي تقريب يحلّ دونه محلّ صيغة
        «السلك السميك»؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الموشور.}
لزجاج $n(\lambda) = 1.500 + \qty{5.0e3}{nm^2}/\lambda^2$، ورنينه
عند $\lambda_0 = \qty{100}{nm}$.
\begin{enumerate}[resume]
  \item تحقق من أن صيغة كوشي تنتج من \omterm{prop:b2:waves-in-media:lorentz}{نموذج لورنتس} بعيدًا
        دون الرنين، ومن أن $\omega_p^2/\omega_0^2 = n^2(\infty) - 1$: وأوجد قيمة
        $\omega_p^2/\omega_0^2$ وقيمة $n$ الفعّالة لهذا الزجاج.
  \item القرينة \omterm{def:b2:dispersion-wave-packets:relation}{وسرعة الطور} عند \qty{400}{nm} و \qty{700}{nm}؛ وقرينة
        المجموعة عند \qty{550}{nm}؛ والتأخّر بين اللونين على
        \qty{1}{km} من هذا الزجاج في ليف.
  \item وموشور زاويته \qty{60}{\degree} عند أصغر انحراف من أجل \qty{550}{nm}:
        أوجد الانحراف (باستعمال $n\sin(A/2) = \sin((A + D)/2)$)؛ والبعد الزاوي
        بين \qty{400}{nm} و \qty{700}{nm} (بالاشتقاق)؛ وعلى شاشة
        على بعد \qty{2}{m}، عرض الطيف.
  \item ولماذا يكون الزجاج عاتمًا عند \qty{200}{nm}، وما
        لونه إذا أضاف أثر من الحديد رنينًا ضعيفًا عند \qty{700}{nm}؟
  \item وفي صورة لورنتس، لماذا ينحرف الأزرق أكثر من الأحمر في
        الموشور؟
  \item أعطِ في جدول واحد، من أجل الطبقة F والفولاذ والنحاس و
        الزجاج عند تواترات عملها، طبيعةَ
        $\underline\gamma$ ومصيرَ الموجة.
\end{enumerate}
\end{problem}
