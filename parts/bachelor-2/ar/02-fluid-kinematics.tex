\chapter{تحريك الموائع}\label{ch:b2:fluid-kinematics}

قف على جسر وانظر إلى النهر. تمرّ ورقة منجرفة، فتتسارع بين
الدعامات، وتتباطأ في الحوض أسفلها، وتدور دورة في دوّامة، ثم
تمضي. ولك أن تصف النهر بأن تتبع تلك الورقة --- موضعها وسرعتها
في كل لحظة --- أو بأن تقف ساكنًا فتسجّل، عند كل نقطة من النهر،
سرعة مرور الماء. والوصف الثاني هو وصف فيزيائي الموائع: أي
\emph{مجال السرعات}. وهذا الفصل يقيم الوصفين، ويتعلّم حساب
تسارع جسيم انطلاقًا من المجال، ويكتب انحفاظ الكتلة قانونًا
موضعيًّا، ويصنّف الجريانات بحسب دواميتها --- وهي المفردات التي
ستحتاج إليها معادلتا أويلر ونافييه.

\section{وصف الوسط المتصل}

\begin{definition}[الجسيم المائع؛ فرضية الوسط المتصل]\label{def:b2:fluid-kinematics:particle}
\index{جسيم مائع}\index{فرضية الوسط المتصل}\index{مقياس متوسطي}
\emph{الجسيم المائع} حجم من المائع صغير كفايةً ليُعامَل نقطةً
على مقياس الجريان، وكبير كفايةً في الوقت نفسه ليحوي عددًا
هائلًا من الجزيئات، بحيث تكون كثافته $\rho$ وضغطه $P$ ودرجة
حرارته $T$ وسرعته المتوسطة $\vect v$ متوسطات معرَّفة تعريفًا
حسنًا (أي المقياس \emph{المتوسطي}، وهو نحو الميكرومتر عادةً).
و\emph{فرضية الوسط المتصل} تفترض وجود مثل هذا المقياس: إذ يجب أن
يكون المسار الحرّ الوسطي $\ell$ للجزيئات أصغر بكثير من مقياس
الجريان $L$، أي $\ell/L \ll 1$. وفي الهواء عند الضغط الجوي
$\ell \approx \qty{70}{nm}$؛ وفي سائل يكون $\ell$ بحجم الجزيء. ويوصف
المائع عندئذ بمجموعة \emph{مجالات} $\rho(M, t)$ و $P(M, t)$ و $\vect v(M, t)$ معرَّفة عند كل نقطة.
\end{definition}

\begin{definition}[الوصفان اللاغرانجي والأويلري]\label{def:b2:fluid-kinematics:euler}
\index{وصف لاغرانجي}\index{وصف أويلري}\index{خط تيار}\index{خط مسار}\index{جريان مستقر}
يتتبع الوصف \emph{اللاغرانجي} كل \omterm{def:b2:fluid-kinematics:particle}{جسيم مائع}: موضعه
$\vect r(t)$ وسرعته $\dd\vect r/\dd t$ بدلالة الزمن وموضعه
الابتدائي. وأما الوصف \emph{الأويلري} فيعطي، عند كل نقطة ثابتة
$M$ ولحظة $t$، سرعة $\vect v(M, t)$ الجسيم المارّ من $M$ في تلك
اللحظة. و\emph{خط المسار} هو مسار جسيم؛ و\emph{خط التيار} في
اللحظة $t$ منحنٍ يماسّ في كل نقطة $\vect v(M, t)$؛ و\emph{أنبوب
التيار} هو السطح المؤلَّف من خطوط التيار المارّة من منحنٍ مغلق.
ويكون الجريان \emph{مستقرًّا} حين لا تتوقف المجالات الأويلرية
على الزمن، $\partial\vect v/\partial t = \vect 0$؛ وعندئذ تنطبق خطوط
التيار وخطوط المسار وتكون ثابتة.
\end{definition}

\begin{example}[وصفان لجريان واحد]\label{ex:b2:fluid-kinematics:hose}
ماء في خرطوم حديقة مقطعه $S$ وتدفقه $Q$ يمضي بالسرعة $v = Q/S$
في كل مكان: فالمجال الأويلري منتظم، وحركة كل جسيم \omterm{def:b2:fluid-kinematics:euler}{بالوصف
اللاغرانجي} منتظمة. وأما في الفوهة المتضيّقة فيبقى المجال
الأويلري مستقرًّا لكن $v$ يتزايد على امتداد المحور: فكل جسيم
\emph{يتسارع} وهو يمرّ فيها، مع أن لا شيء يتغير مع الزمن عند أي
نقطة ثابتة. وذلك أول ما على \omterm{def:b2:fluid-kinematics:euler}{الوصف الأويلري} أن يتعلّم التعبير
عنه.
\end{example}

\section{المشتق الجسيمي}

\begin{theorem}[المشتق الجسيمي (المشتق المادي)]\label{thm:b2:fluid-kinematics:material}
\index{مشتق جسيمي}\index{تسارع حملي}
معدّل تغيّر مقدار $G(M, t)$ (سلّمي أو متجهي) \emph{بالنسبة إلى
\omterm{def:b2:fluid-kinematics:particle}{الجسيم المائع}} المارّ من $M$ في اللحظة $t$ هو
\[
\frac{\mathrm DG}{\mathrm Dt} = \frac{\partial G}{\partial t} + (\vect v\cdot
\operatorname{\vect{grad}})\,G ,
\qquad\text{وعلى الخصوص}\qquad
\vect a = \frac{\mathrm D\vect v}{\mathrm Dt} = \frac{\partial\vect v}{\partial t} +
(\vect v\cdot\operatorname{\vect{grad}})\,\vect v ,
\]
حيث $(\vect v\cdot\operatorname{\vect{grad}}) = v_x\partial_x + v_y\partial_y +
v_z\partial_z$. والحدّ الأول هو معدّل التغيّر \emph{الموضعي} (عند
نقطة ثابتة)، والثاني هو المعدّل \emph{الحملي} (لأن الجسيم ينتقل
إلى حيث يكون $G$ مختلفًا). ومن أجل السرعة، $(\vect v\cdot
\operatorname{\vect{grad}})\vect v = \operatorname{\vect{grad}}(v^2/2) +
(\operatorname{\vect{curl}}\vect v)\wedge\vect v$.
\end{theorem}

\begin{proof}
في $\dd t$ ينتقل الجسيم من $M$ إلى $M + \vect v\,\dd t$، ومنه $\dd G =
G(M + \vect v\dd t, t + \dd t) - G(M, t) = \partial_tG\,\dd t + \dd t\,(v_x\partial_x
+ v_y\partial_y + v_z\partial_z)G$ حتى المرتبة الأولى (بقاعدة اشتقاق
دالة لأربعة متغيرات). وتُتحقَّق المتطابقة المتجهية على المركّبة $x$:
$[\operatorname{\vect{grad}}(v^2/2)]_x = v_x\partial_xv_x + v_y\partial_x
v_y + v_z\partial_xv_z$ و $[(\operatorname{\vect{curl}}\vect v)\wedge\vect v]_x =
v_y(\partial_yv_x - \partial_xv_y) + v_z(\partial_zv_x - \partial_xv_z)$ (ومركّبات
$\operatorname{\vect{curl}}\vect v$ معطاة أدناه في
\cref{def:b2:fluid-kinematics:vorticity})؛ ومجموعهما $v_x\partial_xv_x +
v_y\partial_yv_x + v_z\partial_zv_x$.
\end{proof}

\begin{example}[التسارع في فوهة مستقرة]\label{ex:b2:fluid-kinematics:nozzle}
\omterm{def:b2:fluid-kinematics:euler}{جريان مستقر} ذو بعد واحد $v(x) = v_0(1 + x/L)$ على امتداد فوهة:
ينعدم الحدّ الموضعي، ويكون $a = v\,\dd v/\dd x = v_0^2(1 + x/L)/L$.
وجسيم يدخل بالسرعة $v_0 = \qty{2}{m/s}$ ويخرج بالسرعة $2v_0$ على طول $L
= \qty{5}{cm}$ يحسّ تسارعًا $a = 80$ حتى \qty{160}{m/s^2}: أي من ثمانية
إلى ستة عشر $g$، في جريان لا يتوقف فيه شيء على الزمن.
\end{example}

\begin{example}[الموضعي مقابل الحملي]\label{ex:b2:fluid-kinematics:weather}
تنخفض درجة الحرارة في محطة أرصاد أثناء الليل ($\partial T/
\partial t < 0$: وهذا موضعي)، وتنخفض كذلك لطرد هوائي تحمله الريح
نحو الشمال ($\vect v\cdot\operatorname{\vect{grad}}T < 0$: وهذا حملي، أو
\emph{انسياقي}). ومقياس حرارة على منطاد ينجرف مع الريح يسجّل
المجموع، $\mathrm DT/\mathrm Dt$؛ وأما محطة الأرصاد فلا تسجّل إلا
الحدّ الأول.
\end{example}

\section{انحفاظ الكتلة}

\begin{definition}[التدفق الكتلي والتدفق الحجمي]\label{def:b2:fluid-kinematics:flowrate}
\index{تدفق كتلي}\index{تدفق حجمي}\index{كثافة تيار الكتلة}
\emph{التدفق الكتلي} عبر سطح موجَّه $S$ هو الكتلة العابرة له في
واحدة الزمن، $D_m = \sum\rho\,\vect v\cdot\vect n\,\dd S$ --- وهو تدفق
\emph{كثافة تيار الكتلة} $\vect j = \rho\vect v$ (ووحدتها
\unit{kg.m^{-2}.s^{-1}})؛ و\emph{التدفق الحجمي} هو $D_V = \sum\vect v
\cdot\vect n\,\dd S$ (\unit{m^3/s}). ومن أجل سرعة منتظمة ناظمية على
مقطع مستو مساحته $S$ يكون $D_m = \rho vS$ و $D_V = vS$.
\end{definition}

\begin{proof}
في $\dd t$ يكون المائع العابر للعنصر $\dd S$ هو ما تحويه
الأسطوانة المائلة ذات القاعدة $\dd S$ والراسم $\vect v\,\dd t$، وحجمها
$\vect v\cdot\vect n\,\dd S\,\dd t$ وكتلتها $\rho$ مضروبًا في ذلك.
\end{proof}

\begin{theorem}[الانحفاظ الموضعي للكتلة]\label{thm:b2:fluid-kinematics:continuity}
\index{معادلة الاستمرار}\index{تباعد}
عند كل نقطة من مائع،
\[
\frac{\partial\rho}{\partial t} + \operatorname{div}(\rho\vect v) = 0 ,
\qquad\text{حيث}\qquad
\operatorname{div}\vect A = \frac{\partial A_x}{\partial x} + \frac{\partial A_y}{\partial y} +
\frac{\partial A_z}{\partial z}
\]
هو \emph{تباعد} مجال متجهي: أي التدفق الصافي الخارج
للمجال $\vect A$ عبر سطح حجم صغير، في واحدة الحجم. ويكافئ ذلك
$\mathrm D\rho/\mathrm Dt + \rho\operatorname{div}\vect v = 0$.
\end{theorem}

\begin{proof}
لنأخذ الصندوق الثابت $[x, x + \dd x]\times[y, y + \dd y]\times[z, z + \dd z]$.
تتغير كتلته $\rho\,\dd x\dd y\dd z$ بالمعدل $\partial_t\rho\,\dd x\dd y
\dd z$. وتدخل الكتلة عبر الوجه عند $x$ بالمعدل $j_x(x)\dd y\dd z$
وتخرج عبر الوجه عند $x + \dd x$ بالمعدل $j_x(x + \dd x)\dd y\dd z$:
فالخروج الصافي $\partial_xj_x\,\dd x\dd y\dd z$، وكذلك من أجل $y$ و $z$.
وانحفاظ الكتلة --- إذ لا تُخلق كتلة --- يعطي $\partial_t\rho = -
(\partial_xj_x + \partial_yj_y + \partial_zj_z)$. وتنتج الصيغة الثانية من
$\operatorname{div}(\rho\vect v) = \vect v\cdot\operatorname{\vect{grad}}\rho +
\rho\operatorname{div}\vect v$.
\end{proof}

\begin{remark}[التباعد، لقاء أول]\label{rem:b2:fluid-kinematics:divergence}
يبيّن الاشتقاق ما يقيسه $\operatorname{div}$: أي التدفق الخارج من
صندوق صغير في واحدة الحجم. وبالجمع على الصناديق التي تبلّط
حجمًا منتهيًا $V$، تتلاشى الوجوه الداخلية مثنى مثنى ولا يبقى
إلا السطح الخارجي: $\iint_{\partial V}\vect A\cdot\vect n\,\dd S = \iiint_V
\operatorname{div}\vect A\,\dd\tau$ --- وهي مبرهنة التباعد
(أوستروغرادسكي)، وسنصوغها بدقة في \cref{ch:b2:maxwell-equations}
ونستعملها في الكهرمغناطيسية كلها؛ وهي مبرهنة في كتاب
الرياضيات للسنة الثالثة. ومكاملًا على حجم ثابت، يُقرأ القانون
الموضعي $\dd m_V/\dd t = -D_m^{\text{خارج}}$: أي إن الكتلة في الداخل
تتغير بما يعبر الحدّ.
\end{remark}

\begin{corollary}[الجريان اللاانضغاطي؛ أنابيب التيار]\label{cor:b2:fluid-kinematics:incompressible}
\index{جريان لاانضغاطي}
يكون الجريان \emph{لاانضغاطيًّا} حين تبقى كثافة كل جسيم
ثابتة، $\mathrm D\rho/\mathrm Dt = 0$، وهو ما يكافئ
\[
\operatorname{div}\vect v = 0 .
\]
ويجري السائل المتجانس جريانًا لاانضغاطيًّا؛ وكذلك الغاز عند
سرعات أدنى بكثير من سرعة الصوت (انظر \cref{ch:b2:sound-waves}). وفي
\omterm{def:b2:fluid-kinematics:euler}{جريان مستقر} يكون \omterm{def:b2:fluid-kinematics:flowrate}{التدفق الكتلي} $\rho vS$ واحدًا عبر كل مقطع من
أنبوب التيار؛ فإن كان الجريان لاانضغاطيًّا و $\rho$ منتظمة فوق
ذلك، كان $vS$ ثابتًا: أي إن المائع يتسارع حيث يضيق الأنبوب.
\end{corollary}

\begin{proof}
$\mathrm D\rho/\mathrm Dt = -\rho\operatorname{div}\vect v$. وأما الأنبوب،
فنطبق الموازنة المكاملة على الحجم المحصور بين مقطعين: فلا تدفق
عبر السطح الجانبي (إذ السرعة مماسية)، والحالة مستقرة، ومنه
يساوي الداخل الخارج.
\end{proof}

\begin{example}[الدم]\label{ex:b2:fluid-kinematics:blood}
يحمل الأبهر (ومقطعه \qty{2.5}{cm^2}) \qty{5}{L/min} $= \qty{83}{cm^3/s}$:
أي $v = \qty{33}{cm/s}$. ويمرّ التدفق نفسه عبر نحو $10^{10}$
شعيرة مقطعها الكلي زهاء \qty{2500}{cm^2}: أي $v = \qty{0.3}{mm/s}$ ---
وهو بطء يكفي للتبادلات عبر الجدران، وتلك غاية التفرّع.
\end{example}

\section{الدوامية والجريانات اللادورانية}

\begin{definition}[الدوامية]\label{def:b2:fluid-kinematics:vorticity}
\index{دوامية}\index{دوّار}\index{جريان لادوراني}
\emph{دوامية} جريان هي المجال المتجهي
\[
\vect\omega = \operatorname{\vect{curl}}\vect v , \qquad
\operatorname{\vect{curl}}\vect A = \Bigl(\frac{\partial A_z}{\partial y} -
\frac{\partial A_y}{\partial z},\; \frac{\partial A_x}{\partial z} - \frac{\partial A_z}
{\partial x},\; \frac{\partial A_y}{\partial x} - \frac{\partial A_x}{\partial y}\Bigr) .
\]
ويكون الجريان \emph{لادورانيًّا} حيث $\vect\omega = \vect 0$.
\end{definition}

\begin{proposition}[الدوامية تقيس الدوران الموضعي]\label{prop:b2:fluid-kinematics:rotation}
يدور عنصر مائع صغير مركزه $M$، ككل، بالسرعة الزاوية
$\vect\Omega = \tfrac12\vect\omega(M)$. وفي دوران صلب
$\vect v = \vect\Omega\wedge\vect{OM}$ تكون \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} منتظمة،
$\vect\omega = 2\vect\Omega$؛ وفي جريان القصّ المستوي $\vect v = ky\,\vect e_x$
تكون $\vect\omega = -k\vect e_z$: أي إن عجلة ريشية تُلقى في جريان
قصّ تدور مع أن \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار} مستقيمة.
\end{proposition}

\begin{proof}
بجوار $M$، $\vect v(M + \vect\epsilon) = \vect v(M) + \mathbf G\vect\epsilon$ حيث
$\mathbf G$ مصفوفة $\partial_jv_i$؛ ونفصلها إلى جزئها المتناظر
(وهو تشوّه صرف يمطّ العنصر دون أن يديره ككل) وجزئها المتخالف
التناظر، ومصفوفته $\tfrac12(\partial_jv_i -
\partial_iv_j)$ هي مصفوفة $\vect\epsilon \mapsto \tfrac12\vect\omega\wedge\vect\epsilon$
(\cref{thm:b2:rigid-body-mechanics:field}). وفي الدوران الصلب حول $\vect e_z$:
$\vect v = \Omega(-y, x, 0)$، و $\omega_z = \partial_xv_y - \partial_yv_x = 2\Omega$.
وفي القصّ: $\omega_z = -\partial_y(ky) = -k$.
\end{proof}

\begin{definition}[كمون السرعات؛ التكامل الخطي]\label{def:b2:fluid-kinematics:potential}
\index{كمون السرعات}\index{تكامل خطي للسرعة}\index{جريان كموني}
في \omterm{def:b2:fluid-kinematics:vorticity}{جريان لادوراني} تشتقّ السرعة من \emph{كمون سرعات}:
$\vect v = \operatorname{\vect{grad}}\varphi$ (وعندئذ لا يتوقف التكامل
الخطي للمتجهة $\vect v$ على امتداد مسار إلا على طرفيه، $\int_A^B\vect v
\cdot\dd\vect l = \varphi(B) - \varphi(A)$). فإن كان الجريان
لاانضغاطيًّا كذلك، كان $\Delta\varphi = \operatorname{div}\operatorname{\vect{grad}}
\varphi = 0$: أي إن الكمون يحقق معادلة لابلاس، تمامًا كما يفعل
الكمون الكهرسكوني في منطقة خالية من الشحنة --- وهو \emph{الجريان
الكموني}. و\emph{التكامل الخطي} للسرعة على امتداد منحنٍ مغلق $C$
هو $\Gamma = \oint_C\vect v\cdot\dd\vect l$.
\end{definition}

\begin{proof}
كون مجال معدوم \omterm{def:b2:fluid-kinematics:vorticity}{الدوّار} على منطقة بسيطة الترابط تدرجًا، وكون
التدرج معدوم \omterm{def:b2:fluid-kinematics:vorticity}{الدوّار} ($\partial_y\partial_z\varphi = \partial_z\partial_y\varphi$)،
هما من كتاب الرياضيات للسنة الثانية؛ وتنتج معادلة لابلاس من
$\operatorname{div}\vect v = 0$.
\end{proof}

\begin{example}[ثلاثة جريانات مستوية]\label{ex:b2:fluid-kinematics:planeflows}
\index{جريان الركود}\index{دوامة نقطية}\index{دوامة رانكين}
(1) \emph{\omterm{ex:b2:fluid-kinematics:planeflows}{جريان الركود}} $\vect v = k(x, -y)$: فيكون $\operatorname{div}\vect v =
0$، و $\vect\omega = \vect 0$، و $\varphi = \tfrac12k(x^2 - y^2)$؛ \omterm{def:b2:fluid-kinematics:euler}{وخطوط التيار}
$xy = \text{ثابت}$، وهي قطوع زائدة --- كنفث يصطدم بجدار، بجوار المحور. (2)
\emph{\omterm{ex:b2:fluid-kinematics:planeflows}{الدوامة النقطية}} $\vect v = \dfrac{\Gamma}{2\pi r}\vect e_\theta$ (بالإحداثيات
القطبية المستوية): وهي لاانضغاطية و\emph{لادورانية في كل مكان
إلا عند $r = 0$}، و $\varphi = \Gamma\theta/2\pi$ (متعدد القيم)، ويكون
التكامل الخطي على أي دائرة حول المركز $\Gamma$ --- أي إن
\omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} مركَّزة على المحور. (3) \emph{\omterm{ex:b2:fluid-kinematics:planeflows}{دوامة رانكين}}: لبّ
$r < a$ في دوران صلب $v_\theta = \Omega r$ (ودواميته $2\Omega$)
موصول \omterm{ex:b2:fluid-kinematics:planeflows}{بالدوامة النقطية} $v_\theta = \Omega a^2/r$ في الخارج: وهو نموذج
الإعصار القمعي، ودوّامة حوض الاستحمام، والدوّامة خلف دعامة
الجسر --- وتبلغ السرعة ذروتها عند حافة اللبّ.
\end{example}

\begin{remark}[لماذا يهمّ التكامل الخطي]\label{rem:b2:fluid-kinematics:stokes}
من أجل حلقة مستوية صغيرة مساحتها $\dd S$ وناظمتها $\vect n$، يكون
التكامل الخطي $\oint\vect v\cdot\dd\vect l = \vect\omega\cdot\vect n\,\dd S$ ---
أي إن \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} هي التكامل الخطي في واحدة المساحة، تمامًا كما أن
التباعد هو التدفق في واحدة الحجم. (وتحقق من ذلك على الدوران
الصلب: $2\pi r\cdot\Omega r = 2\Omega\cdot\pi r^2$.) وبالجمع على سطح
نحصل على مبرهنة ستوكس، $\oint_C\vect v\cdot\dd\vect l = \iint_S
\operatorname{\vect{curl}}\vect v\cdot\vect n\,\dd S$، وستُصاغ في
\cref{ch:b2:maxwell-equations}. ويولّد الجناح قوة الرفع بحمله
تكاملًا خطيًّا حوله؛ ويدوم خط الدوامة في مائع مثالي (مبرهنة
كلفن) --- ولهذا تبقى الدوّامة خلف الدعامة زمنًا طويلًا.
\end{remark}

\begin{method}[قراءة مجال سرعات]\label{met:b2:fluid-kinematics:reading}
إذا أُعطي $\vect v(M, t)$: (1) أهو مستقر؟ ($\partial_t\vect v = \vect 0$)؛ (2)
أهو لاانضغاطي؟ ($\operatorname{div}\vect v = 0$)؛ (3) أهو
لادوراني؟ ($\operatorname{\vect{curl}}\vect v = \vect 0$؛ فإن كان كذلك
أوجد $\varphi$)؛ (4) أوجد \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار} من $\dd x/v_x = \dd y/v_y = \dd z/
v_z$ عند $t$ ثابتة، \omterm{def:b2:fluid-kinematics:euler}{وخطوط المسار} من $\dd\vect r/\dd t = \vect v(\vect r,
t)$؛ (5) احسب التسارع $\partial_t\vect v + (\vect v\cdot
\operatorname{\vect{grad}})\vect v$. وبالإحداثيات القطبية المستوية $(r, \theta)$،
$\operatorname{div}\vect v = \dfrac1r\dfrac{\partial(rv_r)}{\partial r} + \dfrac1r
\dfrac{\partial v_\theta}{\partial\theta}$ و $\omega_z = \dfrac1r\dfrac{\partial(rv_\theta)}
{\partial r} - \dfrac1r\dfrac{\partial v_r}{\partial\theta}$ (مسلَّم بهما هنا؛
ويُشتقّان في \cref{ch:b2:maxwell-equations}).
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:fluid-kinematics:1}
المسار الحرّ الوسطي في الهواء \qty{70}{nm} عند \qty{1}{bar} ويتغير
مثل $1/P$. هل يصلح وصف الوسط المتصل من أجل (أ) الهواء حول سيارة،
(ب) الهواء في قناة مجهرية عرضها \qty{1}{\micro m} عند \qty{1}{bar}؛
(ج) الهواء على ارتفاع \qty{100}{km} ($P \approx \qty{3e-7}{bar}$) حول
ساتل طوله \qty{1}{m}؟ أعطِ $\ell/L$ في كل حالة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fluid-kinematics:2}
\omterm{def:b2:fluid-kinematics:euler}{جريان مستقر} ذو بعد واحد $v(x) = v_0(1 + x/L)$ من أجل $0 \le x \le L$.
أوجد تسارع جسيم عند $x$؛ والزمن الذي يستغرقه في العبور من $0$ إلى
$L$؛ وقارنه بالمقدارين $L/v_0$ و $L/2v_0$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fluid-kinematics:3}
في أنبوب نصف قطره $R$ يكون منحني السرعات $v(r) = v_{\max}(1 -
r^2/R^2)$ على امتداد المحور. أوجد \omterm{def:b2:fluid-kinematics:flowrate}{التدفق الحجمي}؛ والسرعة الوسطى $\langle v\rangle
= D_V/\pi R^2$؛ والقيم من أجل $R = \qty{5}{mm}$ و $v_{\max} = \qty{0.4}{m/s}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:fluid-kinematics:4}
من أجل كل مجال، قل أيكون الجريان لاانضغاطيًّا وأيكون
لادورانيًّا؛ وأعطِ \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية}: (أ) $\vect v = (ax, -ay, 0)$؛ (ب)
$\vect v = (-\Omega y, \Omega x, 0)$؛ (ج) $\vect v = (ax, ay, 0)$؛ (د) $\vect v =
(ky, 0, 0)$؛ (ه) $\vect v = (ax, -ay, bz)$ --- ومن أجل أي $b$ يكون
لاانضغاطيًّا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fluid-kinematics:5}
خرطوم حديقة قطره الداخلي \qty{12}{mm} يملأ دلوًا سعته \qty{10}{L} في
\qty{50}{s}. أوجد السرعة في الخرطوم؛ والسرعة في فوهة قطرها
\qty{3}{mm}؛ وإذا تضيّقت الفوهة على طول \qty{4}{cm}، قدّر التسارع
الوسطي لجسيم يعبرها (بواحدة $g$)؛ والزمن الذي يمضيه في الفوهة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fluid-kinematics:6}
جريان مستو غير مستقر $\vect v = (U, V\cos\omega t)$ حيث $U$ و $V$ و $\omega$
ثوابت. (أ) \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار} في اللحظة $t$. (ب) \omterm{def:b2:fluid-kinematics:euler}{خط مسار} الجسيم الذي
يكون في المبدأ عند $t = 0$. (ج) ارسم الاثنين من أجل $V = U$ عند $t = 0$
وعند $t = \pi/2\omega$؛ ولماذا يختلفان؟ (د) تسارع جسيم.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fluid-kinematics:7}
نهر عرضه \qty{50}{m} وعمقه \qty{2.0}{m} يجري بالسرعة \qty{1.2}{m/s}. (أ)
التدفق. (ب) ثم يدخل مضيقًا عرضه \qty{15}{m} وعمقه \qty{4.0}{m}: فما
السرعة؟ (ج) ويضيف رافد \qty{30}{m^3/s}؛ وفي المصبّ يصير عرض النهر
\qty{60}{m} وعمقه \qty{2.5}{m}: فما السرعة؟ (د) ويجري غاز ساخن
جريانًا مستقرًّا في أنبوب ثابت المقطع؛ وبين نقطتين تتضاعف درجة
حرارته عند ثبات الضغط: فماذا يحدث لسرعته؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fluid-kinematics:8}
\emph{\omterm{ex:b2:fluid-kinematics:planeflows}{دوامة رانكين}.} $v_\theta = \Omega r$ من أجل $r < a$، و $v_\theta = \Omega a^2/r$
من أجل $r > a$. (أ) \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} في كل منطقة (استعمل الصيغة
القطبية). (ب) التكامل الخطي على دائرة نصف قطرها $r$ مركزها على
المحور، في المنطقتين؛ وانسبه إلى تدفق \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية}. (ج) إعصار قمعي: $a = \qty{50}{m}$،
و $v_{\max} = \qty{70}{m/s}$: أوجد $\Omega$ و $\Gamma$ والسرعة عند \qty{500}{m}.
(د) أجريان اللبّ لادوراني؟ والخارج؟ وأين يقع «الدوران»
فيزيائيًّا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:fluid-kinematics:9}
\emph{\omterm{ex:b2:fluid-kinematics:planeflows}{جريان الركود}} $\vect v = k(x, -y)$. (أ) أوجد الكمون $\varphi$
وتحقق من $\Delta\varphi = 0$. (ب) \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار}؛ والدالة $\psi = kxy$
ثابتة عليها --- بيّن أن $v_x = \partial\psi/\partial y$ و $v_y = -\partial\psi/
\partial x$ (وهي \emph{دالة التيار}). (ج) مجال التسارع؛ بيّن أنه
$\operatorname{\vect{grad}}(v^2/2)$ وفسّره. (د) \omterm{def:b2:fluid-kinematics:euler}{خط مسار} جسيم ينطلق من
$(x_0, y_0)$: $x = x_0\eu^{kt}$ و $y = y_0\eu^{-kt}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fluid-kinematics:10}
\emph{جريان متمدد.} جريان ذو بعد واحد $v(x, t) = x/(t + \tau)$
من أجل $t > 0$ و $\tau > 0$ ثابت. (أ) \omterm{def:b2:fluid-kinematics:euler}{خطوط المسار}: بيّن أن $x(t) = x_0(1 +
t/\tau)$؛ وأوجد تسارع جسيم، من \omterm{def:b2:fluid-kinematics:euler}{خط المسار} ومن \omterm{thm:b2:fluid-kinematics:material}{المشتق الجسيمي} معًا.
(ب) الكثافة منتظمة في كل لحظة، $\rho(t)$: استعمل انحفاظ الكتلة
لإيجاد $\rho(t)$. (ج) تحقق بمتابعة الكتلة المحصورة بين جسيمين.
(د) ولماذا يكون هذا المجال نموذجًا أحادي البعد لكون يتمدد
تمددًا منتظمًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fluid-kinematics:11}
\emph{منبع في تيار.} راكِب جريانًا منتظمًا $\vect v = U\vect e_x$
ومنبعًا مستويًا $\vect v = \dfrac{q}{2\pi r}\vect e_r$ (و $q$ \omterm{def:b2:fluid-kinematics:flowrate}{التدفق
الحجمي} في واحدة الطول على امتداد $z$). (أ) الكمون $\varphi = Ux +
(q/2\pi)\ln r$؛ وتحقق من أن الجريانين لاانضغاطيان ولادورانيان
(إلا عند المبدأ). (ب) نقطة الركود. (ج) دالة التيار $\psi
= Uy + q\theta/2\pi$؛ \omterm{def:b2:fluid-kinematics:euler}{وخط التيار} المارّ من نقطة الركود هو
$r\sin\theta = (q/2\pi U)(\pi - \theta)$: ارسمه. (د) بيّن أن هذا الخط
يقع بعيدًا في المصبّ عند $y = \pm q/2U$: فالجريان خارجه هو الجريان
حول نصف جسم كليل عرضه $q/U$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:fluid-kinematics:12}
\emph{دوامة ومصبّ.} جريان مستو $\vect v = -\dfrac{q}{2\pi r}\vect e_r +
\dfrac{\Gamma}{2\pi r}\vect e_\theta$ من أجل $r > a$. (أ) بيّن أنه
لاانضغاطي ولادوراني. (ب) \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار}: حلزونات لوغاريتمية
$r = r_0\eu^{-q\theta/\Gamma}$. (ج) تسارع جسيم: بيّن أنه قطري صرف،
$-v^2/r\,\vect e_r$. (د) الزمن اللازم لجسيم كي يحلزن من $r_0$ إلى $a$؛
وعدد الدورات التي يقطعها. (ه) القيم: $q =
\qty{1.5e-3}{m^2/s}$، و $\Gamma = \qty{3e-2}{m^2/s}$، و $r_0 = \qty{0.5}{m}$،
و $a = \qty{2}{cm}$.
\end{exercise}

\section{مسألة: من ثقب المغسلة إلى الإعصار}

\begin{problem}[{مسألة نهاية الأسبوع --- التحريك نفسه على ثلاثة
مقاييس: الدوّامة فوق مصرف، والإعصار القمعي، والإعصار المداري من ساتل}]
\label{pb:b2:fluid-kinematics:1}

\medskip
\noindent\textbf{الجزء I --- حوض الاستحمام.}
يُفرَّغ حوض عبر ثقب نصف قطره $a = \qty{2}{cm}$؛ والماء، وعمقه
$h = \qty{20}{cm}$، يغادر \omterm{def:b2:fluid-kinematics:flowrate}{بالتدفق الحجمي} $D_V =
\qty{0.30}{L/s}$. وبعيدًا عن الثقب يُنمذَج الجريان مستويًا
(لا يتوقف على العمق): $\vect v = v_r(r)\vect e_r + v_\theta(r)\vect e_\theta$.
\begin{enumerate}
  \item باستعمال انحفاظ الكتلة عبر أسطوانة نصف قطرها $r$
        وارتفاعها $h$، بيّن أن $v_r = -q/2\pi r$ حيث $q = D_V/h$؛
        وأعطِ قيمة $q$.
  \item وقبل نزع السدادة كان الماء قد حُرِّك فصار يدور بالسرعة
        \qty{1.0}{cm/s} عند $r = \qty{50}{cm}$. وبتسليم أن كل حلقة من
        الماء تحفظ عزمها الحركي وهي تتحرك نحو الداخل ($rv_\theta$
        ثابت)، أوجد $v_\theta(r)$ والتكامل الخطي $\Gamma$.
  \item تحقق من أن \omterm{cor:b2:fluid-kinematics:incompressible}{الجريان لاانضغاطي} ولادوراني من أجل
        $r > a$.
  \item \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار}: بيّن أنها حلزونات لوغاريتمية، وأوجد كم
        دورة يقطع الجسيم بين $r = \qty{50}{cm}$ و
        $r = a$.
  \item الزمن الذي يستغرقه جسيم من \qty{50}{cm} حتى الثقب،
        وسرعته السمتية عند الوصول.
  \item تسارع جسيم عند $r$؛ وقارنه بالتسارع $g$ عند $r = a$.
        (وينخفض السطح الحرّ حيث يجب أن يوازنه الضغط ---
        أي القمع؛ وذلك في الفصل التالي.)
  \item يمنح دوران الأرض كل مائع ساكن في الحوض دورانًا
        «خلفيًّا» بالمعدل الشاقولي المحلي
        $\Omega_\oplus\sin\lambda \approx \qty{5e-5}{rad/s}$. قارن السرعة
        السمتية التي يمنحها عند \qty{50}{cm} بسرعة التحريك
        \qty{1}{cm/s}، وقل أيتحدد اتجاه دوران حوض يُفرَّغ بنصف
        الكرة الذي هو فيه.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- الإعصار القمعي دوامةَ رانكين.}
نصف قطر اللبّ $a = \qty{60}{m}$، والريح العظمى $v_{\max} = \qty{80}{m/s}$،
وكثافة الهواء $\rho = \qty{1.2}{kg/m^3}$.
\begin{enumerate}[resume]
  \item اكتب $v_\theta(r)$ في المنطقتين وأعطِ $\Omega$ و $\Gamma$.
  \item \omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} في اللبّ وخارجه؛ وارسم $v_\theta$ و $\omega_z$
        بدلالة $r$.
  \item تسارع جسيم هوائي عند حافة اللبّ؛ وقارنه بالتسارع $g$.
  \item بيّن أن عجلة ريشية صغيرة توضع خارج اللبّ تنجرف حول
        الإعصار دون أن تدور حول محورها، بينما تدور الموضوعة في
        اللبّ دورة واحدة في كل دورة حول المحور.
  \item الطاقة الحركية للبّ في المتر الواحد من الارتفاع.
  \item الطاقة الحركية للجريان الخارجي في المتر الواحد من
        الارتفاع، بين $a$ و $R = \qty{5}{km}$؛ ولماذا لا تتوقف على
        $R$ إلا لوغاريتميًّا؟
  \item مع ارتفاع \qty{1}{km}، أوجد الطاقة الحركية الكلية؛
        وقارنها بالطاقة التي يحرّرها \qty{1}{t} من مادة TNT (\qty{4.2}{GJ}).
  \item ويجري الهواء أيضًا نحو الداخل ونحو الأعلى في الإعصار
        القمعي. فإذا غُذّي اللبّ بجريان قطري داخل بالسرعة
        \qty{5}{m/s} عبر سطحه الجانبي على أول \qty{300}{m} من
        الارتفاع، فما السرعة الشاقولية التي على الهواء أن
        يبلغها في أعلى تلك الطبقة (منتظمةً على مقطع اللبّ)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الإعصار المداري.}
يُنمذَج الإعصار المداري دوامةَ رانكين نصف قطر لبّها
$a = \qty{40}{km}$ و $v_{\max} = \qty{50}{m/s}$. ويُحسّ دوران الأرض
عبر \emph{\omterm{def:b2:fluid-kinematics:vorticity}{الدوامية} الكوكبية} $f = 2\Omega_\oplus\sin\lambda
\approx \qty{5e-5}{s^{-1}}$ عند خط العرض \qty{20}{\degree}: فحلقة هواء
ساكنة بالنسبة إلى الأرض تحمل أصلًا، في المرجع العطالي،
السرعة السمتية $fr/2$ حول أي محور شاقولي.
\begin{enumerate}[resume]
  \item \omterm{def:b2:fluid-kinematics:vorticity}{دوامية} اللبّ والتكامل الخطي $\Gamma$ للإعصار؛
        وقارن \omterm{def:b2:fluid-kinematics:vorticity}{دوامية} اللبّ بالمقدار $f$.
  \item حلقة هواء نصف قطرها $r_0 = \qty{500}{km}$، ساكنة ابتداءً
        بالنسبة إلى الأرض، تُجذب نحو المركز بالضغط المنخفض.
        وبتسليم أن عزمها الحركي في واحدة الكتلة حول المحور،
        $r(v_\theta + fr/2)$، محفوظ، أوجد سرعتها $v_\theta$
        بالنسبة إلى الأرض عند $r = \qty{100}{km}$
        وعند $r = \qty{40}{km}$.
  \item في أي جهة يدور الإعصار في نصف الكرة الشمالي، ولماذا؟
        ولماذا تكون الجهة محددة في إعصار وغير محددة في حوض؟
  \item ولماذا لا يمكن أن يُجذب الهواء إلى المحور تمامًا بهذه
        الطريقة --- وما الذي يحدّ الريح ويترك عينًا؟
  \item يتقارب الهواء نحو الإعصار بالسرعة \qty{5}{m/s} عبر
        أسطوانة نصف قطرها \qty{500}{km} وارتفاعها \qty{1}{km}: أوجد
        \omterm{def:b2:fluid-kinematics:flowrate}{التدفق الكتلي} المجذوب ($\rho = \qty{1.2}{kg/m^3}$).
  \item ينتقل الإعصار ككل نحو الغرب بالسرعة \qty{6}{m/s}. اكتب
        مجال السرعات الأويلري كما يُرى من الأرض بدلالة المجال
        $\vect v_0$ للدوامة في مرجعها؛ وهل \omterm{def:b2:fluid-kinematics:euler}{الجريان مستقر} في
        المرجع الأرضي؟ وأي جانب من العاصفة أشدّ رياحًا؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- التدفقات وخلاصة.}
\begin{enumerate}[resume]
  \item يهطل المطر بمعدل \qty{50}{mm/h} على حوض تجميع مساحته
        \qty{10}{km^2} ويبلغ كله نهرًا مقطعه \qty{40}{m^2}: أوجد
        السرعة الوسطى للنهر في الفيضان.
  \item الدم: يحمل الأبهر (\qty{2.5}{cm^2}) \qty{5.0}{L/min}؛
        وللشعيرات مقطع كلي \qty{0.25}{m^2}: أوجد السرعة في كل
        منهما؛ والزمن الذي يستغرقه الدم في عبور شعيرة طولها \qty{1}{mm}.
  \item يُفرَّغ خزان مقطعه $S$ عبر ثقب مقطعه $s$ في أسفله؛
        وسرعة الخروج $\sqrt{2gh}$ (حيث $h$ مستوى الماء --- مسلَّم
        بها، وتُشتقّ في الفصل التالي). اكتب انحفاظ الكتلة وأوجد
        $h(t)$ وزمن التفريغ من أجل
        $S = \qty{1}{m^2}$ و $s = \qty{1}{cm^2}$ و $h_0 = \qty{1}{m}$.
  \item أعطِ الجريانات الثلاثة في الأجزاء الأول والثاني والثالث في جدول واحد:
        نصف القطر، والسرعة، \omterm{def:b2:fluid-kinematics:vorticity}{ودوامية} اللبّ، والتكامل الخطي ---
        والمقدار الذي ينحفظ أثناء الجريان الداخل فيجعلها تدور
        دورانًا متسارعًا.
\end{enumerate}
\end{problem}
