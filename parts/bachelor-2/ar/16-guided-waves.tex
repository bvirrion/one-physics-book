\chapter{الأمواج الموجَّهة والأجواف}\label{ch:b2:guided-waves}

نبضةُ الرادار تنتقل من المرسل إلى الطبق داخل أنبوب نحاسي مستطيل
المقطع؛ وضوء وصلة الإنترنت يقطع عشرة آلاف كيلومتر داخل خيط زجاجي
أدقّ من الشعرة؛ وأمواج فرن الميكروويف ترتدّ بين جدرانه فتستقر على
نقش من البقع الحارة والباردة. والموجة المحصورة بجدران ناقلة أو
كاسرة لم تعد موجة مستوية حرة تذهب حيث تشاء: إذ عليها أن تحقق
الشروط الحدية على الجدران، وذلك ينتقي الأشكال التي يجوز لها أن
تتخذها --- أي \emph{الأنماط} --- ويمنعها دون \emph{تواتر قطع}. وهذا
الفصل يبني أبسط دليل، وهو صفيحتان ناقلتان متوازيتان، انطلاقًا من
انعكاسات \cref{ch:b2:wave-interfaces}؛ ثم \omterm{prop:b2:guided-waves:rectangular}{دليل الموجة المستطيل}،
والجوف المغلق، \omterm{prop:b2:guided-waves:fibre}{والليف البصري} في صورة الأشعة.

\section{الأمواج بين صفيحتين ناقلتين}

\begin{proposition}[أنماط الدليل ذي الصفيحتين المتوازيتين]\label{prop:b2:guided-waves:plates}
\index{دليل موجة}\index{موجة موجَّهة}\index{نمط (دليل موجة)}\index{تواتر القطع (دليل موجة)}
يحدّ مستويان ناقلان تمامًا عند $y = 0$ و $y = a$ خلاءً. ابحث عن موجة
تنتشر وفق $z$ بالشكل $\vect E = E(y)\,\eu^{\iu(\omega t - k_gz)}
\vect e_x$ (أي الحقل موازٍ للصفيحتين، عمودي على جهة
الانتشار). \omterm{thm:b2:maxwell-equations:maxwell}{ومعادلات ماكسويل} في الخلاء وشرط $E = 0$
على الصفيحتين تفرض
\[
E(y) = E_0\sin\frac{n\pi y}a , \qquad
k_g^2 = \frac{\omega^2}{c^2} - \Bigl(\frac{n\pi}a\Bigr)^2 , \qquad n = 1, 2, \dots
\]
ولا ينتشر النمط $n$ إلا فوق \emph{تواتر قطعه} $f_n = nc/2a$؛ ودونه
تكون $k_g$ تخيلية فيتلاشى الحقل على طول الدليل. وسرعتاه
الطورية والزمرية هما
\[
v_\varphi = \frac c{\sqrt{1 - f_n^2/f^2}} > c , \qquad
v_g = c\sqrt{1 - f_n^2/f^2} < c , \qquad v_\varphi v_g = c^2 ,
\]
وحقله المغناطيسي له مركّبتان وفق $y$ ووفق $z$: \omterm{prop:b2:guided-waves:plates}{فالموجة
الموجَّهة} ليست مستعرضة في جهة الانتشار.
\end{proposition}

\begin{proof}
عوّض في $\Delta\vect E = \partial_t^2\vect E/c^2$: $E'' - k_g^2E = -(\omega^2/c^2)E$، أي
$E'' + (\omega^2/c^2 - k_g^2)E = 0$، وحلولها المنعدمة عند $y = 0$ وعند
$y = a$ هي الجيوب مع $\omega^2/c^2 - k_g^2 = (n\pi/a)^2$؛ وتتحقق $\operatorname{div}
\vect E = 0$ لأن $E_x$ لا تتعلق بالمتغير $x$. وتعطي فاراداي
$\underline B_y = (k_g/\omega)\underline E$ و $\underline B_z = (\iu/\omega)\partial_y\underline E$، والأخيرة
في التربيع. وتنتج السرعتان من صيغة كلاين--غوردون
\omterm{def:b2:dispersion-wave-packets:relation}{لعلاقة التبدد} (\cref{ch:b2:dispersion-wave-packets}).
\end{proof}

\begin{remark}[النمط بوصفه موجتين مستويتين]\label{rem:b2:guided-waves:zigzag}
$\sin(n\pi y/a)\eu^{-\iu k_gz}$ هو مجموع موجتين مستويتين شعاعا
موجتيهما $(0, \pm n\pi/a, k_g)$، وطويلتهما $\omega/c$، ترتدّان بين
الصفيحتين تحت الزاوية $\theta$ عن المحور حيث $\cos\theta = k_gc/\omega$:
فالنمط الموجَّه موجة مستوية تتعرّج بين الجدارين، وتنعكس
عند كل منهما مع $r = -1$، \omterm{prop:b2:waves-on-strings:modes}{والموجة المستقرة} المستعرضة هي
شرط تداخلها البنّاء مع نفسها. والطاقة تسير على
التعرّج بالسرعة $c$، ومن ثَمّ على المحور بالسرعة $c\cos\theta = v_g$؛
أما تقاطعات القمم مع المحور فتجري بالسرعة $c/\cos\theta = v_\varphi$. وعند
القطع ترتدّ الموجة جيئةً وذهابًا عبر الدليل ولا تبلغ
مكانًا.
\end{remark}

\begin{example}[أرقام]\label{ex:b2:guided-waves:numbers}
صفيحتان تتباعدان \qty{3}{cm}: $f_1 = \qty{5}{GHz}$، و $f_2 = \qty{10}{GHz}$؛ وبين
هذين التواترين لا ينتشر إلا نمط واحد (وهو التشغيل
\emph{الأحادي النمط}، الذي يمنع النبضة من الانقسام إلى نبضات ذات سرعات
زمرية $v_g$ مختلفة). وعند \qty{8}{GHz} يكون للنمط $n = 1$: $v_g = 0.78c$، و $v_\varphi =
1.28c$، وطول موجة الدليل $\lambda_g = 2\pi/k_g = \qty{4.8}{cm}$ بدل
\qty{3.75}{cm} في الفضاء الحر. وعند \qty{4}{GHz} لا ينفذ شيء: إذ يتلاشى
الحقل وفق $\eu^{-z/\ell}$ مع $\ell = c/2\pi\sqrt{f_1^2 - f^2} = \qty{1.6}{cm}$ --- وهو
مبدأ الشبكة في باب الفرن، \omterm{prop:b2:guided-waves:plates}{ودليل الموجة} دون القطع
المستعمل موهنًا معايرًا، والقنوات المعدنية في
التهوية التي تمرّر الهواء ولا تمرّر الراديو.
\end{example}

\section{دليل الموجة المستطيل والجوف}

\begin{proposition}[الدليل المستطيل؛ النمط TE$_{10}$]\label{prop:b2:guided-waves:rectangular}
\index{دليل موجة مستطيل}\index{نمط TE}
أنبوب معدني أجوف مقطعه المستطيل $a \times b$ ($a > b$) يحمل
أنماطًا تواترات قطعها هي
\[
f_{mn} = \frac c2\sqrt{\Bigl(\frac ma\Bigr)^2 + \Bigl(\frac nb\Bigr)^2} ;
\]
وأدناها، وهو \emph{\omterm{prop:b2:guided-waves:rectangular}{النمط TE}$_{10}$} ($f_{10} = c/2a$)، له $\vect E = E_0\sin
(\pi x/a)\,\eu^{\iu(\omega t - k_gz)}\,\vect e_y$: أي نصف جيب عبر الضلع
العريض، ومنتظم عبر الضلع الضيق، منعدم على الجدارين $x = 0$
و $x = a$ --- وهو بالضبط نمط الصفيحتين المتوازيتين، إذ الجداران
الجانبيان $y = 0$ و $b$ عموديان على $\vect E$ فلا يفرضان عليه شيئًا. ويُستعمل
الدليل بين $f_{10}$ وتواتر القطع التالي؛ وخطوط حقله المغناطيسي
تنغلق في المستوي $xz$، والتيارات التي تحدثها في الجدران
هي ما لا يجوز لشقٍّ أن يقطعه. والاستطاعة المحمولة هي $\tfrac14\varepsilon_0cE_0^2\,ab
\sqrt{1 - f_{10}^2/f^2}$.
\end{proposition}

\begin{proof}
النمط العام جداءُ $\cos$/$\sin$ في $x$ و $y$ مع
الشروط الحدية على الجدران الأربعة، فيعطي $k_g^2 = \omega^2/c^2 - (m\pi/a)^2
- (n\pi/b)^2$ (وهو مقبول في العام؛ أما حالة TE$_{10}$ فهي القضية
أعلاه). وأما الاستطاعة: فمتوسط شعاع بوينتنغ وفق $z$ هو $E_0^2\sin^2(\pi x/a)
\,k_g/2\mu_0\omega$ مكاملًا على المقطع.
\end{proof}

\begin{proposition}[جوف رنّان]\label{prop:b2:guided-waves:cavity}
\index{جوف رنّان}\index{نمط (جوف)}
إغلاق دليل بجدارين ناقلين يبعد أحدهما عن الآخر $d$ يحوّله
إلى \emph{جوف}، تُجبَر فيه \omterm{prop:b2:guided-waves:plates}{الموجة الموجَّهة} على أن تؤلف \omterm{prop:b2:waves-on-strings:modes}{موجة
مستقرة} وفق $z$ أيضًا: $k_g = p\pi/d$، وتواترات الرنين في
صندوق مستطيل $a \times b \times d$ هي
\[
f_{mnp} = \frac c2\sqrt{\Bigl(\frac ma\Bigr)^2 + \Bigl(\frac nb\Bigr)^2 + \Bigl(\frac pd\Bigr)^2} .
\]
وعدد الأنماط التي تواتراتها دون $f$ ينمو وفق $8\pi V f^3/3c^3$ من أجل
صندوق حجمه $V$ (وهو مقبول: عُدّ نقاط الشبكة
$(m/2a, n/2b, p/2d)$ في ثُمن كرة، مع استقطابين)
--- وهي كثافة الأنماط التي ستحتاج إليها نظرية الإشعاع
الحراري (\cref{ch:b2:thermal-radiation}). ولكل نمط عامل جودة
$Q$ تحدده ضياعات الجدران (أي المقاومة السطحية في
\cref{ch:b2:waves-in-media})، وهو $10^3$--$10^4$ عادةً للنحاس عند
تواترات الميكروويف، و $10^{10}$ لجوف فائق الناقلية.
\end{proposition}

\begin{proof}
أمواج مستقرة وفق الجهات الثلاث بعقد على الجدران الستة؛
وشعاع الموجة $(m\pi/a, n\pi/b, p\pi/d)$ طويلته $\omega/c$.
\end{proof}

\begin{example}[أفران ورادارات وساعات]\label{ex:b2:guided-waves:cavities}
فرن ميكروويف أبعاده $30 \times 30 \times 20$ سم$^3$ قرب \qty{2.45}{GHz} له
عشرات الأنماط ضمن نسبة قليلة من تواتر المغنيترون:
فالحقل تراكب أمواج مستقرة، مواضع عظمها متباعدة
\qty{6}{cm}، تسوّيها الصينية الدوّارة (و"مقلّب الأنماط").
ومغنيترون الرادار نفسه \omterm{prop:b2:guided-waves:cavity}{جوف رنّان}؛ والمسرّع
الجسيمي سلسلة أجواف فائقة الناقلية يدفع نمطها TM
الرزم؛ وذرات ساعة السيزيوم تعبر جوفًا ميكرويًّا
موائمًا عند \qty{9.19}{GHz}.
\end{example}

\section{الليف البصري في صورة الأشعة}

\begin{proposition}[ليف ذو قرينة سُلّمية]\label{prop:b2:guided-waves:fibre}
\index{ليف بصري}\index{فتحة عددية}\index{تبدد نمطي}
لبٌّ زجاجي قرينته $n_1$ يحيط به غلاف قرينته $n_2$ أقلّ قليلًا
يوجّه الضوء \omterm{thm:b2:wave-interfaces:snell}{بالانعكاس الكلي الداخلي} عند حدّ اللب.
والشعاع الداخل من الوجه الطرفي من الهواء تحت الزاوية $\alpha$ عن
المحور يُحتبَس إذا كان $\sin\alpha < \sqrt{n_1^2 - n_2^2}$، وهي \emph{\omterm{prop:b2:guided-waves:fibre}{الفتحة
العددية}}؛ وأشعة الزوايا المختلفة تقطع أطوالًا مختلفة، فتتمدد
النبضة الداخلة في ليف طوله $L$ بالمقدار
\[
\Delta t = \frac{n_1L}c\Bigl(\frac{n_1}{n_2} - 1\Bigr) ,
\]
وهو \emph{\omterm{prop:b2:guided-waves:fibre}{التبدد النمطي}} --- نحو \qty{50}{ns} في الكيلومتر من أجل $n_1
- n_2 = 0.01$، وهو ما يحدّ الليف المتعدد الأنماط ببضعة ميغابتات في
الثانية على عشرة كيلومترات. أما الليف \emph{الأحادي النمط} فلبّه
من الدقة (\qty{9}{\micro m}) بحيث لا ينتشر فيه إلا نمط واحد، مثل الدليل
بين أول تواتري قطع له؛ ولا يبقى إلا التبدد
اللوني للزجاج (\cref{ch:b2:dispersion-wave-packets}).
\end{proposition}

\begin{proof}
يقتضي الانعكاس الكلي عند فاصل اللب والغلاف أن يكون $\sin\theta >
n_2/n_1$ من أجل الزاوية $\theta$ عن ناظم الحدّ، أي $\cos
\theta' > n_2/n_1$ من أجل الزاوية $\theta'$ عن المحور في الداخل؛ والانكسار
عند المدخل يعطي $\sin\alpha = n_1\sin\theta'$، ومن ثَمّ $\sin\alpha < n_1\sqrt{1
- n_2^2/n_1^2}$. وأشدّ الأشعة المحتبَسة ميلًا يقطع $L/\cos\theta' = Ln_1/n_2$
مقابل $L$ للشعاع المحوري، بالسرعة $c/n_1$.
\end{proof}

\begin{method}[مسك دفتر الأمواج الموجَّهة]\label{met:b2:guided-waves:method}
(1) عيّن الجدران وشرطها (المركّبة المماسّية $\vect E = 0$ على
الناقل؛ والانعكاس الكلي من أجل العازل). (2) اكتب الحقل
موجةً مستقرة مستعرضة مضروبة في عامل تقدّمي على طول
الدليل؛ فالشروط الحدية تكمّم عدد الموجة المستعرض.
(3) وتعطي \omterm{def:b2:dispersion-wave-packets:relation}{علاقة التبدد} $k_g^2 = \omega^2/c^2 - k_\perp^2$ تواتر
القطع، و $v_\varphi$، و $v_g$، و $\lambda_g$. (4) وعُدّ الأنماط المنتشرة عند
تواتر العمل؛ واقصد أن يكون واحدًا. (5) وفي الجوف أضف الشرط
الطولي وأوجد تواترات الرنين؛ وفي الضياعات استعمل
المقاومة السطحية.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:guided-waves:1}
صفيحتان تتباعدان \qty{2.0}{cm}: أوجد تواترات قطع أول ثلاثة أنماط؛
وعند \qty{12}{GHz}، أيّ الأنماط ينتشر، ومن أجل الأول أوجد $k_g$،
و $\lambda_g$، و $v_\varphi$، و $v_g$؛ وعند \qty{6}{GHz} أوجد طول تلاشي الحقل.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:guided-waves:2}
\omterm{prop:b2:guided-waves:plates}{لدليل موجة} قياسي في الحزمة X: $a = \qty{22.9}{mm}$، و $b = \qty{10.2}{mm}$.
أوجد قطع TE$_{10}$، و TE$_{20}$، و TE$_{01}$؛ والنطاق الأحادي النمط؛ وعند
\qty{10}{GHz}: $\lambda_g$، و $v_g$، وزاوية التعرّج. ولمَ يكون $b \approx a/2$
اختيارًا جيدًا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:guided-waves:3}
جوف أبعاده $30 \times 30 \times 20$ سم$^3$: أوجد تواترات الأنماط $(1,1,0)$،
و $(1,0,1)$، و $(2,1,1)$، و $(3,2,1)$؛ وعدد الأنماط دون \qty{2.5}{GHz} من
صيغة العدّ؛ وأدنى نمط لجوف مكعب تواتره \qty{9.19}{GHz}
--- أوجد ضلعه.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:guided-waves:4}
ليف له $n_1 = 1.48$، و $n_2 = 1.46$. أوجد \omterm{prop:b2:guided-waves:fibre}{الفتحة العددية} ونصف زاوية
القبول؛ والتمدد النمطي في الكيلومتر؛ وأعظم معدل بتات على
\qty{2}{km} إذا وجب أن يدوم البتّ أطول من التمدد؛ وأعد ذلك مع
$n_1 - n_2 = 0.003$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:5}
من أجل نمط الصفيحتين المتوازيتين $n = 1$: (أ) أوجد $\vect B$ من قانون
فاراداي وبيّن أن له مركّبة وفق $z$ في التربيع مع $E$. (ب) وأوجد \omterm{prop:b2:maxwell-equations:boundary}{التيار
السطحي} على كل صفيحة (بالعلاقة الحدية). (ج) ومتوسط شعاع بوينتنغ
وفق $z$ والاستطاعة في وحدة العرض؛ وتحقق من أن متوسط مركّبته
المستعرضة معدوم. (د) وبيّن أن سرعة الطاقة، أي الاستطاعة
على متوسط الطاقة في وحدة الطول، تساوي $v_g$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:6}
\emph{دون القطع.} (أ) ثقب قطره \qty{1}{mm} في شبكة \qty{0.5}{mm} في باب
فرن عند \qty{2.45}{GHz}: عامله معاملة دليل عرضه $a = \qty{1}{mm}$؛
وأوجد طول التلاشي \omterm{def:b2:dispersion-wave-packets:complex}{والوهن} عبر الصفيحة \omterm{def:b2:sound-waves:decibel}{بالديسيبل} (قارن
\cref{ch:b2:waves-in-media}). (ب) وقناة تهوية مقطعها \qty{20}{cm}
في غرفة مدرَّعة: إلى أي تواتر تحجب الراديو،
وكم توهن \qty{100}{MHz} على \qty{1}{m}؟ (ج) و"موهن \omterm{prop:b2:guided-waves:plates}{دليل الموجة}
دون القطع" يستعمل أنبوبًا قطره \qty{1}{cm} عند
\qty{1}{GHz}: أوجد \omterm{def:b2:dispersion-wave-packets:complex}{الوهن} في السنتيمتر (استعمل صيغة الصفيحتين مع $a =
\qty{1}{cm}$ تقديرًا). (د) ولمَ لا يتعلق \omterm{def:b2:dispersion-wave-packets:complex}{الوهن}
بناقلية الجدران؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:7}
\emph{الاستطاعة والانهيار.} يحمل دليل الحزمة X في \cref{exo:b2:guided-waves:2}
\omterm{prop:b2:guided-waves:rectangular}{النمط TE}$_{10}$ عند \qty{10}{GHz}. (أ) أوجد الاستطاعة من أجل $E_0 = \qty{1}{MV/m}$. (ب) وينهار
الهواء عند \qty{3}{MV/m}: أوجد أعظم استطاعة؛ وكيف تحمل الرادارات
الميغاواطات (بالضغط، وبالغازات)؟ (ج) وأما ضياعات الجدران: \omterm{prop:b2:maxwell-equations:boundary}{فالتيار السطحي}
على الجدار العريض من رتبة $E_0/\mu_0c$؛ ومع $R_s = \qty{0.026}{\ohm}$ من أجل
النحاس عند \qty{10}{GHz}، قدّر الضياع في المتر \omterm{def:b2:dispersion-wave-packets:complex}{والوهن}
\omterm{def:b2:sound-waves:decibel}{بالديسيبل} في المتر. (د) وقارن ذلك \omterm{prop:b2:dispersion-wave-packets:coax}{بكابل محوري} وهنه \qty{0.5}{dB/m}: لمَ تُستعمل
الأدلة عند هذه التواترات؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:8}
\emph{جودة الجوف.} جوف نحاسي مكعب ضلعه \qty{10}{cm} في
أدنى أنماطه. (أ) أوجد التواتر. (ب) والطاقة المخزونة هي $\tfrac12\varepsilon_0E_0^2V/4$
(وهو مقبول)، وضياع الجدران $\tfrac12R_sH_0^2\times$ المساحة مع $H_0 \approx E_0/
\mu_0c$ و $R_s = 1/\gamma\delta$: أوجد عامل الجودة $Q = \omega W/P$ --- واحسبه.
(ج) وعرض نطاق الرنين $f/Q$؛ وزمن خفوته $Q/\omega$. (د) وللجوف
الفائق الناقلية $R_s \sim \qty{1e-8}{\ohm}$: أوجد $Q$، ولمَ
تستعمله المسرّعات.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:9}
\emph{عدّ الأنماط.} (أ) بيّن أن عدد أنماط صندوق
حجمه $V$ تواتراتها دون $f$ هو $N(f) = 8\pi Vf^3/3c^3$ (أي نقاط
$(m, n, p)$ في ثُمن كرة نصف قطرها $2fL/c$ من أجل مكعب، مع
استقطابين). (ب) وعدد الأنماط في وحدة الحجم ووحدة التواتر،
$\dd N/V\dd f = 8\pi f^2/c^3$. (ج) وغرفة حجمها \qty{1}{m^3}: أوجد الأنماط في الهرتز عند
\qty{1}{GHz}؛ وعند \qty{500}{THz} (أي الضوء المرئي). (د) ولمَ تكون أنماط الجوف
الصغير متفرقة بينما تؤلف أنماط الغرفة الكبيرة مستمرًّا --- وما
حال الفرن عند \qty{2.45}{GHz}؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:10}
\emph{\omterm{prop:b2:guided-waves:rectangular}{النمط TE}$_{10}$ كاملًا.} في الدليل $0 < x < a$، و $0 < y < b$: $\vect E =
E_0\sin(\pi x/a)\eu^{\iu(\omega t - k_gz)}\vect e_y$. (أ) أوجد $\underline B_x$
و $\underline B_z$ من فاراداي؛ وتحقق من $\operatorname{div}\vect B = 0$. (ب) وتحقق من ماكسويل--أمبير
في الخلاء، واستعد $k_g^2 = \omega^2/c^2 - (\pi/a)^2$. (ج) وأوجد التيارات السطحية
على الجدران الأربعة (العريضان $y = 0$ و $b$ والضيقان $x = 0$
و $a$)؛ وأيّ التيارات يجري وفق $z$، وأيّها يجري عرضًا؟ (د) وشقٌّ مقطوع وفق
$z$ في وسط جدار عريض لا يشعّ، وشقٌّ عرضيّ يشعّ:
فسّر ذلك.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:11}
\emph{دليل عازل.} صفيحة قرينتها $n_1$ وسماكتها $a$ في
وسط $n_2 < n_1$ توجّه الضوء بالانعكاس الكلي. (أ) من أجل شعاع تحت
الزاوية $\theta$ عن ناظم الوجهين، شرط التوجيه هو أن يكون
طور الذهاب والإياب عبر الصفيحة، $2k_0n_1a\cos\theta - 2\varphi_r$ (حيث $\varphi_r$
انزياح طور الانعكاس الكلي، ويؤخذ $\approx 0$ هنا)، من مضاعفات
$2\pi$: أوجد عدد الأنماط من أجل $a = \qty{50}{\micro m}$، و $\lambda = \qty{1.5}{\micro m}$،
و $n_1 = 1.48$، و $n_2 = 1.46$ (واقبل أن عدد الأنماط $\approx 2a\sqrt{n_1^2 - n_2^2}/\lambda$
في كل استقطاب). (ب) وأوجد السماكة من أجل نمط واحد. (ج) وقارن ذلك بلبّ
ليف أحادي النمط قطره \qty{9}{\micro m}. (د) ولمَ لا يكون للدليل العازل،
خلافًا للمعدني، تواتر قطع لأدنى أنماطه؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:12}
\emph{ليف متدرج القرينة.} للتقليل من \omterm{prop:b2:guided-waves:fibre}{التبدد النمطي} تتناقص
قرينة اللب من المحور نحو الخارج، $n(r) = n_1\sqrt{1 - 2\Delta(r/a)^2}$ مع
$\Delta \approx 0.01$. (أ) فسّر نوعيًّا لمَ يستطيع شعاع خارج المحور، وهو
أطول، أن يستغرق زمن الشعاع المحوري نفسه. (ب) وباستعمال معادلة
الأشعة في وسط طبقي، $n(r)\cos\theta(r) = \text{ثابت}$، بيّن
أن شعاعًا يُطلق من المحور بزاوية صغيرة يتذبذب جيبيًّا
حوله (انشر $n$ حتى المرتبة الثانية). (ج) وأوجد دور التذبذب
من أجل $a = \qty{25}{\micro m}$. (د) والتمدد النمطي المتبقي من رتبة
$n_1L\Delta^2/2c$: أوجد قيمته في الكيلومتر، مقارنةً بالسُّلّمي
$n_1L\Delta/c$.
\end{exercise}

\section{مسألة: جوف الفرن والليف البصري}

\begin{problem}[{مسألة نهاية الأسبوع --- نظامان من الأمواج الموجَّهة في كل
مطبخ وكل مدينة: الفرن الذي يطهو بالأمواج المستقرة
والليف الذي يحمل الإنترنت}]
\label{pb:b2:guided-waves:1}

\medskip
\noindent\textbf{الجزء الأول --- من المغنيترون إلى الجوف.}
يغذّي مغنيترون \qty{800}{W} عند \qty{2.45}{GHz} دليلَ موجة مستطيلًا
($a = \qty{8.6}{cm}$، و $b = \qty{4.3}{cm}$) ينفتح على جوف
الفرن ($30 \times 30 \times 20$ سم$^3$)؛ والجدران فولاذ، $\gamma = \qty{1e6}{S/m}$.
\begin{enumerate}
  \item أوجد تواترات قطع أنماط TE$_{10}$، و TE$_{20}$، و TE$_{01}$ في الدليل؛
        وتحقق من أن TE$_{10}$ وحده ينتشر عند \qty{2.45}{GHz}.
  \item وطول موجة الدليل والسرعتين الطورية والزمرية عند \qty{2.45}{GHz}.
  \item وسعة الحقل $E_0$ في الدليل من أجل \qty{800}{W} (بصيغة استطاعة
        TE$_{10}$)؛ وقارن ذلك بحقل انهيار الهواء.
  \item وسعة \omterm{prop:b2:maxwell-equations:boundary}{التيار السطحي} على الجدار العريض (من رتبة
        $E_0/\mu_0c$)؛ وعمق الجلد والمقاومة السطحية للفولاذ؛
        والضياع في المتر من الدليل.
  \item وأنماط الجوف: اذكر ما تقع تواتراته ضمن $3\%$
        من \qty{2.45}{GHz} (جرّب $(m, n, p)$ حتى $4$)؛ وكم عددها؟
  \item والمسافة بين البقع الحارة لأحد تلك الأنماط على كل
        محور؛ ولمَ تدور الصينية، وما "مقلّب الأنماط"؟
  \item وعامل جودة الجوف محمّلًا بالطعام نحو
        $200$: أوجد عرض نطاق استجابته؛ ولمَ لا يهمّ كثيرًا عدم التوافق
        بين تواتر المغنيترون والأنماط.
  \item وللجوف الفارغ $Q \approx 3000$: أوجد الطاقة المخزونة عند دخل
        \qty{800}{W}، ومتوسط كثافة الطاقة، وسعة الحقل؛ وقارن ذلك
        بالسؤال 3 وبالانهيار. وما الذي يحمي المغنيترون
        حين يعمل الفرن فارغًا؟
  \item والزمن الذي تقطع فيه الطاقة \qty{20}{cm} من الدليل من
        المغنيترون إلى الجوف.
  \item ويُغذّى المغنيترون بمنبع مقوَّم نصف موجة فلا يشعّ
        إلا في نصف كل دور من دورات الشبكة: أوجد الاستطاعة العظمى في أثناء
        الرشقات من أجل متوسط \qty{800}{W}؛ وماذا يفعل الجوف
        بين الرشقات (استعمل $Q \approx 200$)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الباب والقطع.}
\begin{enumerate}[resume]
  \item وشبكة الباب ثقوبها \qty{1.5}{mm} في صفيحة \qty{1}{mm}: أوجد طول
        التلاشي في ثقب \omterm{def:b2:dispersion-wave-packets:complex}{والوهن} \omterm{def:b2:sound-waves:decibel}{بالديسيبل} عبر الصفيحة.
  \item ويُختم حرف الباب بخانق ربع موجة: أي بشقٍّ
        عمقه $\lambda/4$ حول إطار الباب، يقدّم دارة مفتوحة
        عند الفجوة (تذكّر \cref{ch:b2:wave-interfaces}): أوجد عمق
        الشقّ عند \qty{2.45}{GHz}.
  \item والتسرب القانوني \qty{5}{mW/cm^2} على \qty{5}{cm}: فأيّ كسر من
        \qty{800}{W} عبر باب مساحته \qty{100}{cm^2} يقابل ذلك؟
  \item وقناة طولها \qty{30}{cm} وقطرها \qty{4}{cm} تهوّي الجوف: أهي
        دون القطع عند \qty{2.45}{GHz}؟ وأوجد \omterm{def:b2:dispersion-wave-packets:complex}{الوهن} على طولها.
  \item ولمَ يجب أن تكون الشبكة موصولة كهربائيًّا بإطار الباب من كل
        جانب (وبمَ تتصرف فجوة في الوصل)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الليف.}
ليف ذو قرينة سُلّمية: اللب $n_1 = 1.465$، والغلاف $n_2 = 1.460$، وقطر اللب
\qty{50}{\micro m} (متعدد الأنماط) أو \qty{9}{\micro m} (أحادي النمط)؛
و $\lambda = \qty{1.55}{\micro m}$؛ \omterm{def:b2:dispersion-wave-packets:complex}{والوهن} \qty{0.2}{dB/km}.
\begin{enumerate}[resume]
  \item أوجد \omterm{prop:b2:guided-waves:fibre}{الفتحة العددية} وزاوية القبول؛ أيسهل إدخال
        الضوء؟
  \item والتمدد النمطي في الكيلومتر من أجل الليف المتعدد الأنماط؛ وأعظم
        معدل بتات على \qty{10}{km} (بتٌّ واحد في زمن التمدد).
  \item وعدد أنماط الليف المتعدد الأنماط (اقبل $N \approx
        \tfrac12(\pi d\,\mathrm{NA}/\lambda)^2$)؛ وعدد أنماط الأحادي ($d =
        \qty{9}{\micro m}$: بيّن أن الصيغة نفسها تعطي نحو $1$).
  \item وفي الليف الأحادي النمط يكون التمدد الباقي لونيًّا:
        مع $\omega'' \approx \qty{-0.2}{m^2/s}$ (\cref{ch:b2:dispersion-wave-packets})
        ونبضات مدتها \qty{50}{ps}، أوجد المسافة التي تتضاعف عليها النبضة؛
        ومعدل البتات على \qty{100}{km}.
  \item \omterm{def:b2:dispersion-wave-packets:complex}{والوهن} على \qty{100}{km} \omterm{def:b2:sound-waves:decibel}{بالديسيبل} وبوصفه نسبة استطاعات؛ ومن أجل دخل
        \qty{1}{mW}: أوجد استطاعة الخرج؛ ومع مضخّمات كل \qty{80}{km}، كم
        يلزم منها لوصلة عابرة للأطلسي طولها \qty{6000}{km}؟
  \item والتواتر وطاقة الفوتون عند \qty{1.55}{\micro m}؛ وعدد الفوتونات في
        الثانية في \qty{1}{mW}، وفي البتّ عند \qty{10}{Gbit/s}، عند الدخل
        وبعد \qty{100}{km}.
  \item ولمَ \qty{1.55}{\micro m} (فكّر في آليتي الضياع في
        السيليكا: تبعثر رايلي، وهو يتناقص وفق $1/\lambda^4$، والامتصاص تحت
        الأحمر الذي يتصاعد بعد \qty{1.6}{\micro m})؟
  \item وانعكاس فرينل عند طرف ليف مشقوق يواجه الهواء: أوجد الضياع
        \omterm{def:b2:sound-waves:decibel}{بالديسيبل}؛ ولمَ تستعمل الوصلات هلامًا موائمًا للقرينة أو تماسًّا
        فيزيائيًّا مصقولًا.
  \item ولمَ لا يشعّ الليف عند انحناء لطيف، ولمَ يفقد الضوء
        عند انحناء حادّ (فكّر في زاوية الشعاع
        على حدّ اللب)؟
  \item وقارن دليلي هذه المسألة: ما الذي يحصر الموجة،
        وما الذي يحدّ عرض النطاق، وما الذي يحدّ الضياعات.
\end{enumerate}
