\chapter{النموذج السُّلَّمي للضوء}\label{ch:b2:scalar-light-model}

سلّط مصباحين يدويين على الجدار نفسه فتكون البقعة أسطع مرتين
لا غير؛ وسلّط عليه حزمتين من ليزر واحد فتنكسر البقعة إلى
هدب مضيئة ومظلمة. فالمصباحان يجمعان شدتيهما، وحزمتا الليزر
تجمعان سعتيهما --- وكل مقياس تداخل وصورة مجسّمة
وشبكة حيود في الفصول الآتية يقوم على معرفة متى يفعل الضوء
هذا ومتى ذاك. وهذا الفصل يهيّئ الأدوات: الضوء موجة \emph{سُلَّمية}
ذات طور، و\emph{\omterm{def:b2:scalar-light-model:path}{المسير الضوئي}} الذي يقيس ذلك الطور
على امتداد شعاع، وكيفية إصدار المنبع --- أي في \omterm{prop:b2:scalar-light-model:coherence}{قطارات موجية} قصيرة
\emph{طول ترابطها} هو الذي يقرر أتستطيع موجتان أن تتداخلا --- ثم
كيفية تسجيل الكاشف له: أي متوسط على مليارات الأدوار
لا يبقي إلا الشدة.

\section{الضوء موجةً سُلَّمية}

\begin{definition}[النموذج السُّلَّمي؛ الموجة أحادية اللون]\label{def:b2:scalar-light-model:scalar}
\index{النموذج السُّلَّمي للضوء}\index{اهتزاز (ضوئي)}\index{موجة أحادية اللون}
في معظم بصريات الأمواج لا تؤدي طبيعة الضوء الشعاعية دورًا (إذ
تشترك الأمواج المتراكبة في استقطاب واحد، أو يكون الضوء غير مستقطب
فتتوسط الاستقطابات): فيحلّ محلّ الحقل \emph{اهتزاز سُلَّمي}
حقيقي هو
\[
s(M, t) = a(M)\cos\bigl(\omega t - \varphi(M)\bigr) ,
\]
بسعة $a(M)$ وطور $\varphi(M)$ في كل نقطة، وكلاهما
بطيء التغير على سُلَّم طول الموجة. وللموجة \emph{أحادية اللون}
تواتر $\omega$ واحد: وهو تمثيل مثالي، إذ يصدر المنبع الحقيقي
في نطاق من التواترات. وأما الضوء المرئي: فيكون $\lambda_0$ من $400$ إلى
\qty{750}{nm} في الخلاء، و $\nu = c/\lambda_0$ من $7.5 \times 10^{14}$ إلى $4 \times 10^{14}$ هرتز،
وأدواره نحو \qty{2}{fs}.
\end{definition}

\begin{definition}[المسير الضوئي والطور]\label{def:b2:scalar-light-model:path}
\index{مسير ضوئي}\index{طور (ضوئي)}\index{جبهة موجة}\index{مبرهنة مالوس--دوبان}
في وسط قرينته $n$ تنتقل الموجة بالسرعة $c/n$ ويكون طول موجتها
$\lambda_0/n$. و\emph{المسير الضوئي} على منحنٍ من $A$ إلى $B$ هو
\[
(AB) = \int_A^Bn\,\dd s ,
\]
والطور الذي تراكمه \omterm{def:b2:scalar-light-model:scalar}{موجة أحادية اللون} منتقلةً على
شعاع من $A$ إلى $B$ هو
\[
\varphi(B) - \varphi(A) = \frac{2\pi}{\lambda_0}\,(AB) = \omega\,\tau_{AB} ,
\]
حيث $\tau_{AB}$ زمن الانتقال. و\emph{جبهة الموجة} سطح
متساوي الطور؛ وفي وسط متماثل المناحي تكون أشعة البصريات الهندسية
عمودية على جبهات الموجة (مبرهنة مالوس--دوبان)، ويكون المسير
الضوئي بين جبهتي موجة واحدًا على كل شعاع.
\end{definition}

\begin{proof}[تبرير]
يتقدم الطور بمقدار $2\pi$ في كل طول موجة محلي $\lambda_0/n$، أي بمقدار
$2\pi n\,\dd s/\lambda_0$ في كل عنصر $\dd s$؛ ويُجمع على امتداد الشعاع. والموجة
$a\cos(\omega t - \varphi)$ حلّ \omterm{thm:b2:waves-on-strings:equation}{لمعادلة الأمواج} تنتقل فيه
الطاقة وفق $\operatorname{\vect{grad}}\varphi$، عموديًّا على
السطوح $\varphi = \text{ثابت}$: وذلك هو الشعاع، وأما قولنا إن \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} واحد
بين \omterm{def:b2:scalar-light-model:path}{جبهتي موجة} فهو تعريف \omterm{def:b2:scalar-light-model:path}{جبهة الموجة}
مقروءًا بالمقلوب. وأما المبرهنة الكاملة (وهي تصمد للانعكاسات
والانكسارات) فمقبولة.
\end{proof}

\begin{example}[التطابق البؤري؛ العدسة صفيحةَ طور]\label{ex:b2:scalar-light-model:lens}
\index{تطابق بؤري}\index{عدسة رقيقة (طور)}
منبع نقطي $A$ تصوّره أداة مثالية (متطابقة بؤريًّا) في $A'$:
فلكل الأشعة من $A$ إلى $A'$ \emph{\omterm{def:b2:scalar-light-model:path}{المسير الضوئي} نفسه} --- أي إن
الأمواج الواصلة على كل شعاع متوافقة الطور في $A'$، ولذلك
تتراكم هناك في نقطة مضيئة. وفي عدسة رقيقة بعدها البؤري $f$
يكون هذا قولًا في الطور: إذ يجب أن تخرج موجة مستوية واردة على المحور
موجةً كروية متقاربة نحو البؤرة $F'$؛ وبما أن \omterm{prop:b2:sound-waves:spherical}{الموجة الكروية}
المركّزة على البعد $f$ لها، على البعد $r$ عن المحور،
طور $2\pi r^2/2\lambda f$ متقدم على قيمته المحورية (بالتقريب المحوري
القريب، $\sqrt{f^2 + r^2} \approx f + r^2/2f$)، وجب أن \emph{تؤخّر}
العدسة الموجة بالمقدار $\varphi_L(r) = -\pi r^2/\lambda f$: فهي أثخن في
الوسط بالمقدار الذي يجعل المسيرات الضوئية كلها إلى $F'$
متساوية. فالعدسة صفيحة طور؛ وكل صورة مجسّمة أو شاشة بلورية سائلة
تفرض خريطة الطور نفسها هي عدسة.
\end{example}

\section{كيف يُصدَر الضوء: الترابط}

\begin{proposition}[قطارات الموجة؛ زمن الترابط وطوله]\label{prop:b2:scalar-light-model:coherence}
\index{قطار موجة}\index{زمن الترابط}\index{طول الترابط}\index{عرض طيفي}
المنبع الكلاسيكي (لهب، أو مصباح، أو نجم) مجموعة ذرات
يصدر كل منها، في لحظات عشوائية وبأطوار عشوائية،
\emph{\omterm{prop:b2:scalar-light-model:coherence}{قطارات موجية}} قصيرة مدتها $\tau_c$: فالاهتزاز الذي يحدثه عند
نقطة لا يحفظ طورًا ثابتًا إلا خلال $\tau_c$ ثم يقفز. والقطار
الذي مدته $\tau_c$ ليس أحادي اللون: إذ يمتد طيفه على نطاق
تواتري $\Delta\nu \approx 1/\tau_c$ (وهو تبادل فورييه، مقبول: فالجيب
المنتهي لا يمكن أن يكون ذا تواتر واحد). و\emph{\omterm{prop:b2:scalar-light-model:coherence}{زمن الترابط}}
$\tau_c$ و\emph{\omterm{prop:b2:scalar-light-model:coherence}{طول الترابط}}
\[
\ell_c = c\,\tau_c \approx \frac c{\Delta\nu} = \frac{\lambda^2}{\Delta\lambda}
\]
يميّزان المنبع: فالضوء الأبيض $\Delta\lambda \approx \qty{300}{nm}$، و $\ell_c
\approx \qty{1}{\micro m}$؛ والمصباح الطيفي ($\Delta\lambda \sim \qty{0.01}{nm}$)، $\ell_c \sim
\qty{3}{cm}$؛ والليزر، من الأمتار إلى الكيلومترات. ولا تستطيع موجتان مشتقتان من
المنبع نفسه أن تتداخلا إلا إذا كان فرق مسيرهما أصغر
من $\ell_c$ --- وفي ما وراءه تنتميان إلى قطارين مختلفين، بأطوار
لا صلة بينها.
\end{proposition}

\begin{proof}
$\ell_c = c/\Delta\nu$ و $|\Delta\nu/\nu| = |\Delta\lambda/\lambda|$ مع $\nu = c/\lambda$ يعطيان $\ell_c =
\lambda^2/\Delta\lambda$. وللعرض الطيفي نفسه أسباب فيزيائية: العرض
الطبيعي (التخامد الإشعاعي، \cref{ch:b2:dipole-radiation})، والتعريض
الدوبلري بحركة الذرات الحرارية، والتصادمات --- وكل منها
يقصّر القطار.
\end{proof}

\begin{remark}[منبعان، ومنبع واحد]\label{rem:b2:scalar-light-model:twosources}
\index{منابع غير مترابطة}
مصباحان متمايزان (أو ذرتان متمايزتان) يصدران قطارات بأطوار
عشوائية مستقلة: فيتغير طورهما النسبي كل $\tau_c
\sim \qty{1e-9}{s}$، أي أسرع بمليون مرة من استجابة أي عين أو آلة
تصوير --- فيتوسط حدّ التداخل إلى الصفر وتتجمع
الشدتان. والتداخل يحتاج إلى موجتين \emph{تتذكران قفزات
الطور نفسها}: أي نسختين من موجة واحدة، تُنالان بشطرها
(شقّان، أو مرآتان، أو وجها صفيحة) وتُجمعان بتأخير
أقصر من $\tau_c$. والليزر، وقطاراته تدوم ميكروثواني أو
أكثر، يخفف هذا القيد تخفيفًا هائلًا؛ لكنه لا يزيله أبدًا.
\end{remark}

\section{الكواشف والشدة}

\begin{proposition}[ما يقيسه الكاشف]\label{prop:b2:scalar-light-model:intensity}
\index{شدة (بصريات)}\index{كاشف (بصريات)}\index{تدفق فوتوني}
كل كاشف ضوئي --- العين (وزمن استجابتها $\sim\qty{0.1}{s}$)، والفيلم
الفوتوغرافي، وبكسل CCD (\qty{1}{ms})، والثنائي الضوئي (\qty{1}{ns}) ---
يستجيب في زمن طويل بالقياس إلى الدور ($10^{-15}$ ثانية)
ويعطي \emph{الشدة}، أي متوسط مربع الاهتزاز
في الزمن،
\[
I = K\,\langle s^2\rangle = \tfrac12K\,a^2 ,
\]
(حيث $K$ ثابت يُهمَل غالبًا؛ و $I$ متناسبة مع الاستضاءة
بالوحدة \unit{W/m^2}). والمستوى المطلق لا يهمّ إلا في وجه واحد: فالضوء
يصل في فوتونات طاقتها $h\nu$، ويعدّ الكاشف في المتوسط
$I/h\nu$ فوتونًا في وحدة المساحة والزمن.
\end{proposition}

\begin{proof}
$\langle\cos^2(\omega t - \varphi)\rangle = \tfrac12$ على أي مجال أطول بكثير
من $2\pi/\omega$. وأما الفوتونات: ففي الفصل الأخير من مجلد السنة الأولى؛ وهنا
تلتقي الموجة السُّلَّمية بالكمّ.
\end{proof}

\begin{example}[رتب المقادير]\label{ex:b2:scalar-light-model:orders}
مصباح \qty{60}{W} يشعّ \qty{3}{W} من الضوء المرئي، على بعد \qty{2}{m}:
$I = 3/4\pi\cdot4 = \qty{60}{mW/m^2}$، و $I/h\nu = 1.6 \times 10^{17}$ فوتونًا في المتر
المربع والثانية، أي $3 \times 10^{12}$ في الثانية عبر بؤبؤ قطره \qty{5}{mm} ---
فلا أثر للحبيبية في \qty{0.1}{s} من زمن العين. وأما
أخفت نجم تراه العين فيرسل $\sim 10^3$ فوتونًا في الثانية إلى
البؤبؤ؛ وتستثار العصيّة بحفنة منها. وأما ضوء الشمس: \qty{1}{kW/m^2}، أي $3 \times 10^{21}$
فوتونًا في المتر المربع والثانية.
\end{example}

\begin{remark}[التراكب المترابط وغير المترابط]\label{rem:b2:scalar-light-model:superposition}
\index{تراكب مترابط}\index{تراكب غير مترابط}
اهتزازان $a_1\cos(\omega t - \varphi_1)$ و $a_2\cos(\omega t - \varphi_2)$ عند
نقطة يعطيان $s = s_1 + s_2$ و
\[
I = \tfrac12K\bigl[a_1^2 + a_2^2 + 2a_1a_2\langle\cos(\varphi_1 - \varphi_2)\rangle\bigr] .
\]
فإذا كان فرق الطور ثابتًا في أثناء الكشف
(أي موجتان \emph{مترابطتان}): $I = I_1 + I_2 + 2\sqrt{I_1I_2}\cos(\varphi_1 - \varphi_2)$،
وهو حدّ التداخل الذي يدرسه الفصل التالي. وإذا هام
عشوائيًّا (أي موجتان \emph{غير مترابطتين}): فإن $\langle\cos\rangle = 0$ و $I = I_1 +
I_2$ --- وهما المصباحان اليدويان.
\end{remark}

\begin{method}[تهيئة مسألة في البصريات]\label{met:b2:scalar-light-model:method}
(1) اكتب كل موجة بالشكل $a\cos(\omega t - \varphi)$ مع $\varphi = 2\pi(\text{المسير
الضوئي})/\lambda_0$ ابتداءً من المنبع (أو من \omterm{def:b2:scalar-light-model:path}{جبهة موجة} مشتركة). (2) وقرّر
الترابط: أهو المنبع نفسه، وفرق المسير $< \ell_c$؟ فإن كان كذلك فاجمع السعات
(والترميز العقدي مريح: $\underline s = a\eu^{\iu(\omega t - \varphi)}$، و $I =
\tfrac12K|\underline s|^2$)؛ وإلا فاجمع الشدات. (3) وفي العدسات
والمرايا، استعمل تساوي المسيرات الضوئية بين النقطتين المترافقتين، أو
صورة صفيحة الطور. (4) وحوّل إلى ما يُقاس: أي الشدة،
والتباين، وعدّات الفوتونات، وتحقق من تمييز الكاشف في الزمن
والفضاء.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:scalar-light-model:1}
حزمة عند \qty{600}{nm} تعبر \qty{1.0}{cm} من الزجاج ($n = 1.5$). أوجد \omterm{def:b2:scalar-light-model:path}{المسير
الضوئي}؛ وعدد أطوال الموجة في داخله؛ والطور الزائد والتأخير الزائد
بالقياس إلى \qty{1}{cm} من الهواء؛ وسماكة الزجاج التي تؤخّر
الموجة طول موجة واحدة بالضبط بالنسبة إلى الهواء.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:scalar-light-model:2}
أوجد \omterm{prop:b2:scalar-light-model:coherence}{زمن الترابط} وطوله من أجل: الضوء الأبيض ($400$--\qty{750}{nm})؛ وخط مصباح
صوديوم عرضه \qty{0.02}{nm} عند \qty{589}{nm}؛ وثنائي باعث للضوء أحمر (\qty{630}{nm}،
$\Delta\lambda = \qty{25}{nm}$)؛ وثنائي ليزري متعدد الأنماط ($\Delta\nu = \qty{100}{GHz}$)؛ وليزر
مستقر ($\Delta\nu = \qty{1}{kHz}$). فأيّها يعطي هدبًا مع فرق
مسير \qty{1}{mm}؟ ومع \qty{10}{m}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:scalar-light-model:3}
مؤشر ليزر \qty{1}{mW} عند \qty{650}{nm}: أوجد الفوتونات في الثانية؛ ومصباح
\qty{100}{W} (منه \qty{5}{W} مرئية، بمتوسط \qty{550}{nm}) على \qty{3}{m}: أوجد الفوتونات في
الثانية عبر بؤبؤ قطره \qty{6}{mm}؛ وعددها في \qty{0.1}{s} من زمن العين؛
والتقلب النسبي $1/\sqrt N$ لذلك العدد --- أيكون الضوء حبيبيًّا؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:scalar-light-model:4}
منبع نقطي على \qty{1.0}{m} من شاشة، و $\lambda = \qty{500}{nm}$. أوجد فرق \omterm{def:b2:scalar-light-model:path}{المسير
الضوئي} بين مركز الشاشة ونقطة على \qty{1}{cm}
عن المحور (بالضبط، ثم بالصيغة المحورية القريبة $r^2/2d$)؛ وعدد
أطوال الموجة؛ ونصف قطر \omterm{def:b2:scalar-light-model:path}{جبهة الموجة} عند الشاشة؛ وعلى أي بعد عن المحور
تخطئ الصيغة المحورية القريبة بطول موجة واحد؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:scalar-light-model:5}
\emph{العدسة صفيحةَ طور.} (أ) لموجة مستوية على المحور
طور منتظم على مستوي عدسة رقيقة؛ وبعد العدسة يكون الطور
$\varphi_L(r) = -\pi r^2/\lambda f$ بالنسبة إلى المركز. بيّن أن \omterm{def:b2:scalar-light-model:path}{جبهة الموجة}
الخارجة، بالتقريب المحوري القريب، كرة نصف قطرها $f$
مركزها $F'$. (ب) ومنبع نقطي على البعد $d$ قبل العدسة
يرسل موجة طورها على مستوي العدسة $+\pi r^2/\lambda d$ (بالنسبة إلى
المركز): بيّن أن الموجة تتقارب بعد العدسة عند $d'$ حيث
$1/d' = 1/f - 1/d$، وهي صيغة العدسة. (ج) وأوجد مقطع سماكة عدسة
زجاجية مستوية محدّبة ($n = 1.5$) بعدها البؤري \qty{10}{cm}: $e(r) =
e_0 - r^2/2(n - 1)f$؛ وأوجد السهم عند $r = \qty{1}{cm}$. (د) و"عدسة فرينل" تطوي
المقطع كلما تجاوز $\lambda/(n - 1)$: فلمَ تظل
تجمّع الضوء، ولمَ لا يكون ذلك تامًّا إلا للون واحد؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:scalar-light-model:6}
\emph{\omterm{ex:b2:scalar-light-model:lens}{التطابق البؤري} لمرآة.} على المرآة أن ترسل موجة مستوية واردة
على محورها إلى نقطة $F$. (أ) اكتب شرط "تساوي المسيرات
الضوئية من \omterm{def:b2:scalar-light-model:path}{جبهة موجة} مستوية إلى $F$ من أجل كل نقطة من المرآة"
وبيّن أنه يعرّف قطعًا مكافئًا $y^2 = 4fx$ بؤرته $F$. (ب) ولمَ لا تكون المرآة
الكروية متطابقة بؤريًّا إلا تقريبًا (قارن $y^2 = 4fx$ بالدائرة
التي نصف قطرها $2f$ قرب الرأس)؟ (ج) ومن أجل مرآة قطرها \qty{20}{cm}
و $f = \qty{1}{m}$، أوجد خطأ مسير الكرة عند الحافة، بأطوال
موجة \qty{500}{nm}. (د) ولمَ تحتمل المقاريب الراديوية من الكروية ما لا
تحتمله المقاريب الضوئية؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:scalar-light-model:7}
\emph{فيرما.} \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} من $A$ (في الوسط $n_1$) إلى $B$ (في
$n_2$) عبر نقطة $P$ من السطح الفاصل المستوي هو $n_1AP + n_2PB$. (أ)
بيّن أنه مستقر (أصغري) من أجل النقطة $P$ التي تحقق
$n_1\sin\theta_1 = n_2\sin\theta_2$. (ب) وفسّر: فالمسارات المجاورة لها
الطور نفسه، فتتجمع الأمواج عليها --- ولهذا "يختار" الضوء
شعاع ديكارت. (ج) واستنتج قانون الانعكاس بالطريقة
نفسها. (د) ولمَ يكون المسير عبر عدسة من $A$ إلى صورتها $A'$
واحدًا على كل شعاع، لا مستقرًّا فحسب؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:scalar-light-model:8}
\emph{لمَ لا يتداخل مصباحان.} مصباحا صوديوم يضيئان
شاشة؛ وعند نقطة تكون سعتا موجتيهما $a$ وطورهما النسبي
$\Delta\varphi$ يقفز عشوائيًّا كل $\tau_c = \qty{1e-10}{s}$. (أ)
أوجد الشدة عند تلك النقطة بدلالة $\Delta\varphi$؛ وعظمها وصغرها.
(ب) ويتوسط الكاشف على $\tau_d = \qty{1}{ms}$: فكم قيمة
مستقلة من $\Delta\varphi$؟ وأوجد متوسط الشدة؛ والحجم النسبي
للتقلب الباقي في حدّ التداخل ($\sim 1/\sqrt N$). (ج) وثنائي
ضوئي زمنه $\tau_d = \qty{1e-11}{s}$: فماذا يرى؟ (د) وكيف نجحت
أول تجربة أظهرت التداخل بين ليزرين مستقلين (1963)؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:scalar-light-model:9}
\emph{العمق الظاهري.} قطعة نقود على العمق $h$ في الماء ($n = 1.33$).
(أ) باستعمال \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} (أو قانون ديكارت عند الزوايا الصغيرة)، بيّن
أن العين فوق الماء تراها على العمق $h/n$. (ب) ومن أجل $h = \qty{1.2}{m}$: أوجد العمق
الظاهري؛ وبكم يتجاوز \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} من القطعة إلى العين
طوله الهندسي؟ (ج) وسمكة على العمق $h$ ترى ذبابة على \qty{1}{m}
فوق السطح على أي ارتفاع ظاهري؟ (د) ولمَ يبدو الحوض
أضحل أيضًا حين يُنظر إليه مائلًا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:scalar-light-model:10}
\emph{التباين والترابط.} نسختان من موجة بينهما فرق
مسير $\delta$ تتراكبان؛ وطيف المنبع منتشر
انتشارًا منتظمًا على $[\nu_0 - \Delta\nu/2, \nu_0 + \Delta\nu/2]$، وكل تواتر
يتداخل مع نفسه فقط. (أ) أوجد الشدة الآتية من تواتر واحد: $2I_0
[1 + \cos(2\pi\nu\delta/c)]$. (ب) وكامِل على النطاق وبيّن أن
المجموع هو $2I_0[1 + \operatorname{sinc}(\pi\Delta\nu\,\delta/c)\cos(2\pi\nu_0\delta/c)]$ مع
$\operatorname{sinc}x = \sin x/x$. (ج) وعرّف تباين الهدب $V = (I_{\max} -
I_{\min})/(I_{\max} + I_{\min})$ بجوار $\delta$ وبيّن أن $V =
|\operatorname{sinc}(\pi\Delta\nu\,\delta/c)|$: وهو ينعدم عند $\delta = c/\Delta\nu = \ell_c$. (د)
وأعطِ الأرقام من أجل ثنائي باعث للضوء ($\Delta\lambda = \qty{25}{nm}$ عند \qty{630}{nm}): أوجد $\delta$ الذي عنده
ينعدم التباين أول مرة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:scalar-light-model:11}
\emph{رؤية النجوم.} تكشف العين ومضة حين تبلغ نحو $10$ فوتونات
عصيّةً خلال \qty{0.1}{s}؛ والبؤبؤ المتكيّف مع الظلام \qty{7}{mm}؛ وخذ
\qty{500}{nm}. (أ) أوجد أصغر \omterm{prop:b2:scalar-light-model:intensity}{تدفق فوتوني} وأصغر استضاءة عند العين. (ب) وتعطي
الشمس \qty{1}{kW/m^2}؛ ونجم قدره $m$ يعطي $10^{-0.4(m + 26.7)}$
منها: أوجد استضاءة نجم قدره ستة (وهو حدّ العين المجردة)
ومعدل الفوتونات إلى البؤبؤ؛ أيتوافق ذلك مع (أ)؟ (ج) ومقراب
فتحته \qty{200}{mm}: أوجد الكسب في الفوتونات، والقدر الذي يبلغه.
(د) ولمَ يتحدث الفلكيون عن "ضجيج الفوتونات" في صورة خافتة،
وكيف يتغير مع زمن التعريض؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:scalar-light-model:12}
\emph{ما وراء التقريب المحوري القريب.} \omterm{prop:b2:sound-waves:spherical}{لموجة كروية} من نقطة على البعد $d$
عند البعد المستعرض $r$ مسير زائد مضبوط قدره $\sqrt{d^2 +
r^2} - d$. (أ) انشره حتى المرتبة الرابعة: $r^2/2d - r^4/8d^3$. (ب) ويُحتفظ
بالحدّ التربيعي (المحوري القريب) ويُهمَل الرباعي حين يكون
الأخير دون $\lambda/4$: بيّن أن ذلك يقتضي $r^4 < 2\lambda d^3$. (ج) والأرقام:
$d = \qty{1}{m}$، و $\lambda = \qty{500}{nm}$: أوجد أعظم $r$؛ و $d = \qty{1}{cm}$ (أي مجهر):
أوجد أعظم $r$، والزاوية الموافقة. (د) وعلّق: إن "زيغانات"
البصريات الهندسية هي هذه الحدود المهمَلة.
\end{exercise}

\section{مسألة: مصباح صوديوم ومؤشر ليزر}

\begin{problem}[{مسألة نهاية الأسبوع --- منبعا ضوء مفكَّكان: مصباح
الشارع القديم الأصفر، والمؤشر الأحمر في قاعة الدرس، والعدد الوحيد
الذي يقرر أيستطيع كلٌّ منهما أن يصنع هدبًا}]
\label{pb:b2:scalar-light-model:1}

\medskip
\noindent\textbf{الجزء الأول --- مصباح الصوديوم.}
يصدر الصوديوم ثنائيه الأصفر عند $\lambda_1 = \qty{589.0}{nm}$ و $\lambda_2 =
\qty{589.6}{nm}$. والعرض الطبيعي لكل خط $\Delta\nu_{\text{طبيعي}} =
\qty{10}{MHz}$؛ وفي المصباح تكون الذرات عند \qty{500}{K} (وكتلة الصوديوم
\qty{23}{u}، و $k_B = \qty{1.38e-23}{J/K}$، و $u = \qty{1.66e-27}{kg}$).
\begin{enumerate}
  \item أوجد تواتري الخطين وفرقهما.
  \item وطول ترابط خط واحد لو لم يُحسب إلا عرضه
        الطبيعي.
  \item والتعريض الدوبلري: ذرة تتحرك بالسرعة $v$ على خط
        النظر تصدر عند $\nu(1 + v/c)$؛ ومع السرعة الحرارية التربيعية الوسطى $\sqrt{k_BT
        /m}$ على محور واحد، قدّر $\Delta\nu_D$ وقارنه بالعرض
        الطبيعي.
  \item وطول ترابط خط واحد معرَّض دوبلريًّا؛ وفي مصباح
        حقيقي عالي الضغط تعرّضه التصادمات عشر مرات
        أخرى: فما \omterm{prop:b2:scalar-light-model:coherence}{طول الترابط} عندئذ؟
  \item والخطان معًا: على أي فرق مسير ينتقل نظاما هدبهما
        من التوافق إلى التعاكس ثم يعودان
        (أي "طول الضربات" $\lambda^2/\Delta\lambda$)؟ وكم هدبة ذلك؟
  \item وطالب يصنع مقياس تداخل بحزمتين من المصباح
        ويزيد فرق المسير من الصفر: صِف
        التباين المرئي --- أي الضربات والانطفاء الأخير --- مع
        الأرقام الموجودة.
  \item ويشعّ المصباح \qty{1}{W} من الضوء الأصفر: أوجد الفوتونات في الثانية؛
        وطاقة الفوتون الواحد بالإلكترون فولت، ولمَ يكون الضوء أصفر.
  \item ومصباحا صوديوم متجاوران: أيستطيع ضوءاهما أن يتداخلا؟
        ولمَ يختلف السؤال في خطي مصباح واحد؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- مؤشر الليزر.}
ليزر ثنائي أحمر \qty{1}{mW} عند \qty{650}{nm} يصدر، إذا كان رخيصًا، عدة
أنماط طولية منتشرة على $\Delta\lambda = \qty{0.5}{nm}$؛ وأما المستقر
الأحادي النمط فله $\Delta\nu = \qty{1}{MHz}$.
\begin{enumerate}[resume]
  \item أوجد طولي ترابط المؤشرين.
  \item والفوتونات المصدَرة في الثانية؛ ومتوسط \omterm{def:b2:maxwell-equations:operators}{التباعد} بين الفوتونات على
        امتداد الحزمة (بالأمتار)؛ وقارنه \omterm{prop:b2:scalar-light-model:coherence}{بطول الترابط} ---
        فماذا يقول ذلك عن "تداخل الفوتونات"؟
  \item وحوّل الانتشار \qty{0.5}{nm} إلى عرض تواتري؛ وجوف
        الثنائي طوله \qty{1}{mm} مع $n = 3.5$: أوجد \omterm{def:b2:maxwell-equations:operators}{تباعد} أنماطه
        الطولية ($c/2nL$) وعدد الأنماط في الانتشار.
  \item وتقع الحزمة (وقطرها \qty{1}{mm}) على ثنائي ضوئي يعطي
        كل فوتون فيه إلكترونًا: أوجد التيار.
  \item ويُستعمل المؤشر الرخيص في مقياس تداخل بحزمتين مع
        فرق مسير \qty{2}{mm}: أتظهر الهدب أم لا؟ وماذا عن المستقر
        مع \qty{10}{m}؟
  \item وخرج الثنائي في الحقيقة يُفتح ويُغلق في
        نبضات مدتها \qty{1}{ns}: أتستطيع نبضتان من دورين متعاقبين
        أن تتداخلا؟ وما الذي يحدد الترابط هنا، أطول النبضة
        أم عرض الخط؟
  \item ولمَ يبدو ضوء الليزر "مرقّطًا" على الجدار، ولمَ
        لا يبدو ضوء المصباح كذلك؟
  \item وثنائي باعث للضوء أحمر عند \qty{650}{nm} له $\Delta\lambda = \qty{25}{nm}$: أوجد \omterm{prop:b2:scalar-light-model:coherence}{طول
        الترابط}؛ ولمَ تفضّل بعض أجهزة العرض هذه الثنائيات على الليزرات؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- المسيرات الضوئية والكواشف.}
\begin{enumerate}[resume]
  \item تُدخَل شريحة زجاجية \qty{2}{mm} ($n = 1.5$) في أحد ذراعي
        مقياس التداخل: أوجد \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} الزائد، وعدد
        الهدب التي تمرّ أمام علامة.
  \item وتُمال الشريحة بزاوية \qty{10}{\degree}: أوجد المسير الزائد (استعمل $e/\cos r$
        للطول في الزجاج، ناقصًا الهواء الذي يحلّ محلّه، وبتقريب
        من المرتبة الأولى $e(n/\cos r - \cos(i - r)/\cos r)$ --- أو قدّره
        ببساطة بالمسار الهندسي الأطول)؛ وأوجد رتبة مقدار
        انزياح الهدب.
  \item والكاشف آلة تصوير ببكسلات \qty{10}{\micro m}؛ والهدب
        متباعدة \qty{50}{\micro m}: فكم بكسلًا في الهدبة، وماذا
        يحدث للتباين المقيس إذا هبط \omterm{def:b2:maxwell-equations:operators}{التباعد} إلى
        \qty{10}{\micro m}؟
  \item وتعريض \qty{10}{ms} عند \qty{1}{\micro W/cm^2} على بكسل: أوجد الفوتونات في
        البكسل؛ وضجيج الفوتونات النسبي.
  \item وموجتان مترابطتان متساويتان شدتهما $I_0$ تلتقيان بفرق
        مسير $\delta$: اكتب الشدة $I(\delta)$ ودور
        الهدب بدلالة $\delta$.
  \item وتمرّ الهدب أمام نقطة بمعدل \qty{1}{kHz} (بمرآة
        مهتزّة): فماذا ترى العين، وماذا يرى الثنائي الضوئي؟
  \item وتمرّ ذراعا مقياس التداخل في \qty{10}{m} من الهواء ($n - 1 =
        \qty{2.9e-4}{}$): أوجد \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} الزائد على الخلاء، بالهدب؛
        وما الذي يفعله تغيّر ضغط الهواء بنسبة \qty{1}{\%} في النقش.
  \item ويعطي المصباح والمؤشر \qty{1}{\micro W/cm^2} نفسها: فأيّهما
        يعطي فوتونات أكثر في الثانية، ولمَ لا يهمّ ذلك في
        الهدب؟
  \item ولخّص: من أجل كل منبع، \omterm{prop:b2:scalar-light-model:coherence}{طول الترابط}، وما الذي يحدّه،
        وفرق مسير مقياس التداخل الذي يسمح به.
\end{enumerate}
\end{problem}
