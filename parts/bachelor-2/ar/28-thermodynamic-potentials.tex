\section{المتطابقة الأساسية ومتغيراتها}

\begin{proposition}[المتطابقة الأساسية]\label{prop:b2:thermodynamic-potentials:identity}
من أجل نظام مغلق ثابت التركيب، تحقق الطاقة الداخلية،
منظورًا إليها تابعًا للإنتروبيا والحجم،
\[
\dd U = T\,\dd S - P\,\dd V, \qquad
T = \Big(\frac{\partial U}{\partial S}\Big)_V, \quad
P = -\Big(\frac{\partial U}{\partial V}\Big)_S ,
\]
\omterm{thm:b2:open-systems:firstlaw}{والإنتالبي} $H = U + PV$ يحقق $\dd H = T\,\dd S + V\,\dd P$.
وكل تابع، في \emph{متغيراته الطبيعية}\index{متغيرات
طبيعية} --- $(S,V)$ من أجل $U$، و $(S,P)$ من أجل $H$ --- يحوي كل
ترموديناميك النظام: فمشتقاته الجزئية تعطي سائر
متغيرات الحالة. والمبدأ الثاني ينتقي توازن نظام
معزول (ثابت $U$ و $V$) بوصفه حالة $S$ العظمى.
\end{proposition}

\begin{proof}
المبدأ الأول من أجل تحول عكوس، $\dd U = \delta Q_{\text{عكوس}}
+ \delta W_{\text{عكوس}} = T\,\dd S - P\,\dd V$، لا يحوي إلا توابع
حالة ومن ثَمّ يصحّ من أجل أي تغير متناهٍ في الصغر بين
حالتي توازن. وأضف $\dd(PV)$ من أجل $H$.
\end{proof}

\begin{remark}[لمَ توابع جديدة]\label{rem:b2:thermodynamic-potentials:why}
لا تُجرى التجارب عند إنتروبيا ثابتة. فدورق الكيميائي يقوم عند
حرارة الغرفة وضغط الجوّ؛ والمكبس في منظِّم حراري يعمل عند $T$ ثابتة.
ونريد توابع متغيراتها الطبيعية $(T,V)$ و $(T,P)$، وتؤدي، في تلك
الشروط، الدور الذي تؤديه $S$ من أجل نظام معزول: أي أن تقول
أي طريق يسلك النظام وكم عملًا يستطيع أن يعطي. و
\emph{تحويل لوجاندر} --- أي طرح $TS$ لمبادلة $S$ بالمتغير $T$
متغيرًا مستقلًّا --- يبنيها.
\end{remark}

\section{الطاقة الحرة والإنتالبي الحرّ}

\begin{definition}[الطاقة الحرة لهلمهولتز، والإنتالبي الحرّ لغيبس]\label{def:b2:thermodynamic-potentials:FG}
\[
F = U - TS \quad\text{(\emph{الطاقة الحرة}\index{طاقة حرة})}, \qquad
G = H - TS = U + PV - TS \quad\text{(\emph{الإنتالبي الحرّ}\index{إنتالبي حرّ})}.
\]
وكلاهما تابعا حالة شاملان، يقاسان بالجول.
\end{definition}

\begin{theorem}[التفاضليات والمقادير المشتقة]\label{thm:b2:thermodynamic-potentials:dFdG}
\[
\dd F = -S\,\dd T - P\,\dd V, \qquad
\dd G = -S\,\dd T + V\,\dd P ;
\]
ومن ثَمّ، في \omterm{prop:b2:thermodynamic-potentials:identity}{المتغيرات الطبيعية} $(T,V)$ للتابع $F$ و $(T,P)$ للتابع $G$،
\[
S = -\Big(\frac{\partial F}{\partial T}\Big)_V, \quad
P = -\Big(\frac{\partial F}{\partial V}\Big)_T, \qquad
S = -\Big(\frac{\partial G}{\partial T}\Big)_P, \quad
V = \Big(\frac{\partial G}{\partial P}\Big)_T ,
\]
و(غيبس--هلمهولتز) $U = F - T(\partial F/\partial T)_V =
-T^2\,\partial(F/T)/\partial T$، و $H = -T^2\,\partial(G/T)/\partial T$.
ومعرفة $F(T,V)$ أو $G(T,P)$ معرفةٌ لكل شيء: معادلة
الحالة، والإنتروبيا، والطاقة، والسعات الحرارية.
\end{theorem}

\begin{proof}
$\dd F = \dd U - T\dd S - S\dd T = -S\dd T - P\dd V$؛ وكذلك من أجل $G$
من $\dd H$. ثم اقرأ المشتقات الجزئية؛ وأما من أجل غيبس--هلمهولتز، فإن
$F - T\,\partial_TF = F + TS = U$ و $\partial_T(F/T) = (T\partial_TF -
F)/T^2 = -U/T^2$.
\end{proof}

\begin{example}[الغاز المثالي]\label{ex:b2:thermodynamic-potentials:perfect}
من أجل $n$ مولًا، $U = nc_{V}T + U_0$ و $S = nc_V\ln T + nR\ln V + S_0$
يعطيان $F(T,V) = -nRT\ln V + f(T)$: فيكون $P = -\partial_VF = nRT/V$
--- أي إن معادلة الحالة تخرج من $F$ --- ويعطي $S = -\partial_TF$
الإنتروبيا من جديد. وفي المتغيرات $(T,P)$، $G(T,P) = nRT\ln P +
g(T)$، ومن ثَمّ $V = \partial_PG = nRT/P$، ويكون \omterm{def:b2:thermodynamic-potentials:FG}{الإنتالبي الحرّ} \emph{المولي}
$\mu(T,P) = \mu^\circ(T) + RT\ln(P/P^\circ)$ --- وهو \emph{الكمون
الكيميائي}\index{كمون كيميائي} للغاز، وهو ينمو مع
ضغطه: فالغاز يجري من $\mu$ العالي إلى المنخفض، أي هنا من الضغط العالي
إلى المنخفض.
\end{example}

\section{العمل الأعظمي وجهة التطور}

\begin{theorem}[التحولات الأحادية الحرارة]\label{thm:b2:thermodynamic-potentials:monothermal}
نظام يتبادل الحرارة مع منظِّم حراري واحد عند $T_0$، وتساوي
حرارتاه الابتدائية والنهائية $T_0$، يتلقى في أي
تحول العمل
\[
W \ge \Delta F , \qquad\text{أي}\qquad W_{\text{مستعاد}} = -W \le -\Delta F :
\]
فنقصان $F$ هو \emph{أعظم عمل مستعاد} من
النظام عند حرارة ثابتة (ويُبلَغ في تحول عكوس) ---
ومن هنا الطاقة ``الحرة'': أي الجزء من $U$ الذي يمكن تحويله إلى عمل،
والباقي، $TS$، مقيَّد. وإذا كان التحول فوق ذلك أحادي الضغط
(أي الضغط $P_0$ في الخارج، والضغطان الابتدائي والنهائي $P_0$)، فإن العمل
\emph{غير عمل الجوّ} (أي \omterm{def:b2:open-systems:cv}{العمل النافع}، ككهربائي مثلًا)
يحقق $W_{\text{u}} \ge \Delta G$: والمقدار $-\Delta G$ هو أعظم \omterm{def:b2:open-systems:cv}{عمل نافع}
مستعاد عند $T$ و $P$ ثابتين (كعمل خلية وقود، أو
بطارية).
\end{theorem}

\begin{proof}
المبدأ الأول: $\Delta U = W + Q$. والمبدأ الثاني مع منظِّم واحد: $\Delta S
= Q/T_0 + S_{\text{c}}$، و $S_{\text{c}} \ge 0$، ومن ثَمّ $Q \le T_0\Delta S$. ومن ثَمّ
$W = \Delta U - Q \ge \Delta U - T_0\Delta S = \Delta(U - T_0S) = \Delta F$، باستعمال
$T = T_0$ عند الطرفين. وبفصل عمل الجوّ $-P_0\Delta V$،
يكون $W = W_{\text{u}} - P_0\Delta V$ و $W_{\text{u}} \ge \Delta U + P_0
\Delta V - T_0\Delta S = \Delta G$.
\end{proof}

\begin{theorem}[معايير التطور والتوازن]\label{thm:b2:thermodynamic-potentials:criteria}
النظام المحفوظ عند $T$ و $V$ ثابتين والذي لا يتلقى عملًا لا يمكن
أن يتطور إلا بحيث \emph{يتناقص $F$}؛ وتوازنه حالة
$F$ الصغرى. والنظام المحفوظ عند $T$ و $P$ ثابتين والذي لا يتلقى عملًا
نافعًا لا يمكن أن يتطور إلا بحيث \emph{يتناقص $G$}؛ وتوازنه
حالة $G$ الصغرى. (وعلى العموم، من أجل نظام في تماس
مع محيط عند $T_0$ و $P_0$، يتناقص المقدار $U - T_0S + P_0V$.)
\end{theorem}

\begin{proof}
خذ $W = 0$ (أو $W_{\text{u}} = 0$) في المبرهنة السابقة: $\Delta F
\le 0$ (أو $\Delta G \le 0$)، مع المساواة من أجل تطور عكوس، أي
توازني.
\end{proof}

\begin{example}[التمدد متساوي الحرارة، حقيقيًّا ومثاليًّا]\label{ex:b2:thermodynamic-potentials:expansion}
مول واحد من غاز عند \qty{300}{K} يتمدد من \qty{1}{L} إلى \qty{10}{L} في
تماس مع المنظِّم: $\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$. والتمدد
العكوس يستعيد \qty{5.7}{kJ}؛ والتمدد في وجه الجوّ
عند \qty{1}{bar} لا يستعيد إلا $P_0\Delta V = \qty{0.9}{kJ}$؛ والتمدد
الحرّ في الخلاء لا شيء. و \qty{5.7}{kJ} هي أعظم ما
يستطيع الغاز ومنظِّمه أن يعطياه من أجل تغير الحالة هذا، مهما
كانت الآلة بارعة.
\end{example}

\section{علاقات ماكسويل}

\begin{theorem}[علاقات ماكسويل]\label{thm:b2:thermodynamic-potentials:maxwell}
لأن $\dd F$ و $\dd G$ تفاضليان تامّان (بمبرهنة شوارتز
في المشتقات الثانية المختلطة)،
\[
\Big(\frac{\partial S}{\partial V}\Big)_T = \Big(\frac{\partial P}{\partial T}\Big)_V,
\qquad
\Big(\frac{\partial S}{\partial P}\Big)_T = -\Big(\frac{\partial V}{\partial T}\Big)_P .
\]
وهي تعبّر عمّا لا يُقاس (أي كيف تتغير الإنتروبيا مع الحجم أو
الضغط عند حرارة ثابتة) بمعادلة الحالة. ونتيجتان:
\[
\Big(\frac{\partial U}{\partial V}\Big)_T = T\Big(\frac{\partial P}{\partial T}\Big)_V - P,
\qquad
C_P - C_V = T\Big(\frac{\partial P}{\partial T}\Big)_V\Big(\frac{\partial V}{\partial T}\Big)_P
= \frac{TV\alpha^2}{\kappa_T} ,
\]
حيث $\alpha = V^{-1}(\partial V/\partial T)_P$ معامل التمدد و
$\kappa_T = -V^{-1}(\partial V/\partial P)_T$ الانضغاطية متساوية الحرارة.
\end{theorem}

\begin{proof}
$\partial^2F/\partial T\partial V$ محسوبًا بالترتيبين: $-\partial_VS =
-\partial_TP$. والأمر نفسه من أجل $G$. ثم $\dd U = T\dd S - P\dd V$ عند $T$ ثابتة
يعطي $(\partial_VU)_T = T(\partial_VS)_T - P$؛ وأما $C_P - C_V$ فينتج
من كتابة $S(T,V(T,P))$: $(\partial_TS)_P = (\partial_TS)_V +
(\partial_VS)_T(\partial_TV)_P$، مضروبًا في $T$.
\end{proof}

\begin{example}[ما تكشفه العلاقات]\label{ex:b2:thermodynamic-potentials:reveal}
من أجل غاز مثالي، $T(\partial_TP)_V - P = 0$: فالطاقة لا تتعلق
بالحجم (وهو قانون جول من جديد) و $C_P - C_V = nR$. ومن أجل غاز فان
دير فالس، $(\partial_VU)_T = a/V_{\text{m}}^2$: أي الضغط الداخلي
للتجاذبات. ومن أجل الماء السائل، $(\partial_TP)_V = \alpha/\kappa_T = 2.1
\times 10^{-4}/4.6 \times 10^{-10} = \qty{4.6e5}{Pa/K}$: فتسخين الماء في
وعاء صلب ممتلئ كلفنًا واحدًا يرفع الضغط \qty{4.6}{bar}
(فلا تُحكِم غلق قنينة ممتلئة)، و $(\partial_VU)_T \approx \qty{1400}{bar}$ ---
أي تماسك السائل؛ ومع ذلك $c_P - c_V = Tv\alpha^2/\kappa_T \approx
\qty{30}{J/kg/K}$، أي أقلّ من $1\%$ من $c_P$: فالسعتان الحراريتان في
المادة المكثفة تكادان تتطابقان.
\end{example}

\begin{proposition}[المرونة الإنتروبية للمطاط]\label{prop:b2:thermodynamic-potentials:rubber}
من أجل شريط طوله $L$ تحت التوتر $f$، $\dd U = T\dd S + f\dd L$
و $\dd F = -S\dd T + f\dd L$، بحيث $(\partial S/\partial L)_T =
-(\partial f/\partial T)_L$ و $(\partial U/\partial L)_T = f - T(\partial f/
\partial T)_L$. والمطاط يحقق، بتقريب جيد، $f = T\,\varphi(L)$
مع $\varphi > 0$ من أجل $L > L_0$: فيكون $(\partial U/\partial L)_T = 0$ --- أي إن
الطاقة لا تتغير بالمدّ --- ويكون $(\partial S/\partial L)_T =
-\varphi(L) < 0$: فالتوتر \emph{إنتروبي}، $f = -T(\partial S/\partial
L)_T$، والشريط يشدّ راجعًا لأن لسلاسله الممدودة تشكيلات
أقلّ. والنتائج: فإذا مُدّ متساوي الحرارة أطلق حرارة
($Q = T\Delta S < 0$)؛ وإذا مُدّ كظومًا دفئ؛ وإذا سُخِّن تحت
حمل ثابت انكمش (فثبات $f/T$ يعني ثبات $\varphi(L)$، ومن ثَمّ عند
$f$ ثابتة يجب أن تهبط $\varphi$ كلما ارتفعت $T$).
\end{proposition}

\begin{proof}
علاقة ماكسويل من $\dd F$؛ والباقي تعويض. والأصل
المجهري هو المسير العشوائي في \cref{ch:b2:particle-diffusion}:
فسلسلة من $N$ وصلة طول كلٍّ منها $a$ توزَّع المسافة بين طرفيها
توزيعًا غاوسيًّا تباينه $Na^2$، ومن ثَمّ يكون عدد
التشكيلات عند الامتداد $L$ هو $\Omega \propto \eu^{-L^2/2Na^2}$، و $S = S_0 -
k_BL^2/2Na^2$ و $f = k_BTL/Na^2$: أي نابض صلابته
\emph{متناسبة مع الحرارة}.
\end{proof}

\section{التوازن بين طورين}

\begin{theorem}[شرط التعايش]\label{thm:b2:thermodynamic-potentials:coexistence}
مادة نقية عند $T$ و $P$ ثابتين، موزَّعة بين طورين كتلتاهما
$m_1$ و $m_2$ وإنتالبياهما الحرّان في وحدة الكتلة $g_1(T,P)$ و
$g_2(T,P)$، لها $G = m_1g_1 + m_2g_2$؛ وعند التوازن يكون $G$ أصغريًّا، ومن ثَمّ
\begin{itemize}
\item إذا كان $g_1 < g_2$ ذهبت المادة كلها إلى الطور 1 (فالطور
      المستقر هو ذو $g$ \emph{الأدنى})؛
\item ولا يتعايش الطوران إلا حيث
      \[
      g_1(T,P) = g_2(T,P) ,
      \]
      وهي علاقة بين $T$ و $P$: أي منحني التعايش في مخطط
      الأطوار.
\end{itemize}
وعلى ذلك المنحني (كلابيرون)،
\[
\frac{\dd P}{\dd T} = \frac{s_2 - s_1}{v_2 - v_1} = \frac{L_{1\to2}}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
$\dd G = (g_1 - g_2)\dd m_1$ عند $T$ و $P$ ثابتين (مع ثبات $m_1 + m_2$):
فيتناقص $G$ بنقل الكتلة إلى الطور ذي $g$ الأدنى، حتى ينفرد ذلك
الطور أو حتى يصير $g_1 = g_2$. ثم اشتقّ $g_1 = g_2$ على
المنحني مع $\dd g = -s\,\dd T + v\,\dd P$ لكل طور: $-s_1\dd T +
v_1\dd P = -s_2\dd T + v_2\dd P$؛ و $s_2 - s_1 = L/T$ لأن الانتقال
عند $T$ و $P$ ثابتين عكوس. وقد استخرج مجلد السنة الأولى هذه الصيغة
من دورة كارنو؛ وهي هنا تخرج من $G$.
\end{proof}

\begin{remark}[فان دير فالس وإنشاء ماكسويل]\label{rem:b2:thermodynamic-potentials:maxwellconstruction}
تعطي معادلة فان دير فالس $(P + a/v^2)(v - b) = RT$، دون
حرارتها الحرجة $T_{\text{c}} = 8a/27Rb$، ميزوترمات فيها حلقة. و
المائع الحقيقي لا يتبع الحلقة: فعند $P_{\text{إشباع}}(T)$ يقفز من
فرع السائل إلى فرع البخار، و $P_{\text{إشباع}}$ يثبّته $g_{\text{سائل}}
= g_{\text{بخار}}$، أي $\int_{\text{سائل}}^{\text{بخار}}v\,\dd P = 0$ على
الحلقة: فالخط الأفقي يقتطع \emph{مساحتين متساويتين}. وأجزاء
الحلقة التي فيها $(\partial P/\partial v)_T > 0$ غير مستقرة ميكانيكيًّا؛ و
الأجزاء بينها وبين الخط شبه مستقرة --- وهي السائل المفرط التسخين
الذي ``يقفز'' في الميكروويف، والبخار المفرط الإشباع في
غرفة الغيوم.
\end{remark}

\begin{method}[استعمال الكمونات]\label{meth:b2:thermodynamic-potentials:method}
(1) عيّن القيود: معزول $\to$ $S$ عظمى؛ و $T,V$ ثابتان $\to$
$F$ صغرى؛ و $T,P$ ثابتان $\to$ $G$ صغرى. (2) والعمل الأعظمي: $-\Delta F$ عند $T$
ثابتة، و $-\Delta G$ من \omterm{def:b2:open-systems:cv}{العمل النافع} عند $T,P$ ثابتين. (3) وأتحتاج إلى مشتق
$S$ عند $T$ ثابتة؟ فاستعمل علاقة ماكسويل ومعادلة الحالة.
(4) والطوران: قارن $g$؛ والتعايش $g_1 = g_2$؛ وكلابيرون من أجل
الميل؛ وإشارة $\Delta v$ من أجل إشارة الميل. (5) \omterm{ex:b2:thermodynamic-potentials:perfect}{والكمون الكيميائي}
$\mu = g$ في المول: فالمادة تجري نحو $\mu$ الأدنى.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:1}
من أجل $n$ مولًا من غاز مثالي، اكتب $F(T,V)$ و $G(T,P)$ حتى
توابع في $T$ وحدها؛ واستعد معادلة الحالة من كلٍّ منهما؛ وبيّن
أن $S = -\partial_TF$ يعطي الإنتروبيا المعلومة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:2}
مول واحد من غاز، عند \qty{300}{K}، من \qty{1}{L} إلى \qty{10}{L} في
منظِّم حراري: أوجد $\Delta F$، و $\Delta U$، و $\Delta S$؛ والعمل المستعاد إذا كان عكوسًا؛
وإذا كان في وجه \qty{1}{bar}؛ وإذا كان في الخلاء؛ والإنتروبيا المحدَثة في كل
حالة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:3}
الماء والجليد عند \qty{1}{bar}: بيّن أن $g_{\text{سائل}} - g_{\text{جليد}} \approx
(L/T_{\text{انصهار}})(T_{\text{انصهار}} - T)$ قرب \qty{0}{\celsius} واحسبه
عند \qty{-5}{} و $+\qty{5}{\celsius}$ ($L = \qty{334}{kJ/kg}$). فأي طور
مستقر في كل حالة؛ وهل الآخر مستحيل؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:4}
ميول كلابيرون للماء: الانصهار ($\Delta v = -\qty{9.1e-5}{m^3/kg}$) و
التبخر عند \qty{100}{\celsius} ($L = \qty{2.26}{MJ/kg}$، و $\Delta v =
\qty{1.67}{m^3/kg}$). أوجد نقطة الانصهار تحت \qty{70}{bar}؛ ونقطة الغليان عند
\qty{0.7}{bar} وعند \qty{2}{bar}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:5}
\emph{ماكسويل.} (أ) استنتج علاقتي ماكسويل في الفصل
والعلاقتين من $\dd U$ و $\dd H$. (ب) وبيّن $(\partial U/\partial V)_T =
T(\partial P/\partial T)_V - P$؛ وقدّرها من أجل غاز مثالي وغاز فان
دير فالس. (ج) والماء السائل: $\alpha = \qty{2.1e-4}{K^{-1}}$، و $\kappa_T =
\qty{4.6e-10}{Pa^{-1}}$: أوجد ارتفاع الضغط في الكلفن في وعاء صلب ممتلئ،
و $(\partial U/\partial V)_T$. (د) وقنينة زجاجية ممتلئة محكمة تُسخَّن
\qty{10}{K}: علّق.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:6}
\emph{$C_P - C_V$.} (أ) استنتج $C_P - C_V = TV\alpha^2/\kappa_T$. (ب)
والغاز المثالي. (ج) والماء عند \qty{300}{K}؛ والنحاس ($\alpha = \qty{5e-5}{K^{-1}}$،
و $\kappa_T = \qty{7e-12}{Pa^{-1}}$، و $\rho = \qty{8900}{kg/m^3}$)؛ وقارنهما بالمقدار
$c_P$ (\qty{4180}{} و \qty{385}{J/kg/K}). (د) ولمَ يمكن الكلام على
``السعة الحرارية'' لجسم صلب؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:7}
\emph{المطاط.} شريط يحقق $f = CT(L - L_0)$ مع $C = \qty{1}{N/m/K}$.
(أ) أوجد $(\partial S/\partial L)_T$ و $(\partial U/\partial L)_T$. (ب) وتغيّر
الإنتروبيا والحرارة والعمل عند المدّ متساوي الحرارة عند \qty{300}{K} من
$L_0$ إلى $L_0 + \qty{0.1}{m}$. (ج) وإذا مُدّ كظومًا بدل ذلك (والسعة
الحرارية \qty{4}{J/K}): أوجد تغيّر الحرارة. (د) وعُلّق بثقل ثابت
وسُخِّن \qty{30}{K}: أوجد التغير النسبي في $L - L_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:8}
\emph{تبخر بركة.} \omterm{ex:b2:thermodynamic-potentials:perfect}{الكمون الكيميائي} لبخار الماء
عند الضغط الجزئي $p$ هو $\mu_{\text{v}} = \mu_{\text{v}}^\circ(T) +
RT\ln(p/p^\circ)$، وهو يساوي كمون السائل حين $p = p_{\text{إشباع}}(T)$.
(أ) بيّن أن السائل يتبخر حين $p < p_{\text{إشباع}}$ وأن البخار
يتكاثف حين $p > p_{\text{إشباع}}$؛ وعرّف الرطوبة النسبية. (ب)
وأوجد $\mu_{\text{سائل}} - \mu_{\text{v}}$ عند رطوبة $50\%$ و \qty{293}{K}. (ج) ولمَ
يجفّ الغسيل حتى عند \qty{20}{\celsius}، ولا يجفّ قط عند رطوبة $100\%$؟
(د) والندى: على أي سطوح يتكوّن أولًا، ولماذا؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:9}
\emph{فان دير فالس.} $(P + a/v^2)(v - b) = RT$ (مولية). (أ) أوجد
النقطة الحرجة ($\partial_vP = \partial_v^2P = 0$): $v_{\text{c}} = 3b$،
و $T_{\text{c}} = 8a/27Rb$، و $P_{\text{c}} = a/27b^2$. (ب) وثنائي أكسيد الكربون،
$T_{\text{c}} = \qty{304}{K}$، و $P_{\text{c}} = \qty{73.8}{bar}$: أوجد $a$ و $b$. (ج)
وبرّر قاعدة ماكسويل للمساحتين المتساويتين من $g$. (د) وأي أجزاء الحلقة
غير مستقرة، وأيّها شبه مستقرة، وأي ظواهر تُظهر شبه المستقرة؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:10}
\emph{كلاوزيوس--كلابيرون مكاملةً.} من أجل التبخر، أهمل $v_{\text{سائل}}$
وعامل البخار غازًا مثاليًّا. (أ) بيّن أن $\dd\ln P/\dd T = L_{\text{m}}/RT^2$
($L_{\text{m}}$ مولية) وكامِل من أجل $L_{\text{m}}$ ثابتة. (ب) وللماء،
$L_{\text{m}} = \qty{40.7}{kJ/mol}$: أوجد $P_{\text{إشباع}}$ عند \qty{20}{\celsius} من
\qty{1}{bar} عند \qty{100}{\celsius}؛ وقارنها بالمقيسة \qty{23}{mbar}.
(ج) وتنبّأ بضغط النقطة الثلاثية (\qty{0.01}{\celsius}) وقارنه
بمقدار \qty{611}{Pa}. (د) وبيّن أن القانون نفسه ينتج من غيبس--هلمهولتز
مطبَّقًا على $\Delta g$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:11}
\emph{التناضح.} يخفض مذابٌ مخلخل \omterm{ex:b2:thermodynamic-potentials:perfect}{الكمون الكيميائي} للمذيب:
$\mu = \mu^\circ - RTx_{\text{s}}$ (حيث $x_{\text{s}}$ الكسر المولي
للمذاب). وغشاء لا يمرّر إلا المذيب؛ ومذيب نقي على
جهة، ومحلول على الأخرى، وعند $T$ نفسها. (أ) بيّن أن التوازن
يقتضي ضغطًا زائدًا $\Pi$ على المحلول مع $\Pi v_{\text{m}} =
RTx_{\text{s}}$، أي $\Pi = cRT$ (فانت هوف). (ب) وماء البحر، $c \approx
\qty{1.1}{mol/L}$ من الجسيمات المنحلّة، عند \qty{293}{K}: أوجد $\Pi$. (ج) وأصغر
عمل لاستخراج متر مكعب من الماء العذب؛ وقارنه بالمنشآت الحقيقية
(\qty{3}{kWh/m^3}). (د) ولمَ تنفجر الخلايا النباتية في الماء النقي و
تذوي في الماء المالح؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:12}
\emph{السلسلة.} سلسلة بوليمر من $N$ وصلة طول كلٍّ منها $a$: وعدد
التشكيلات التي شعاع الطرف إلى الطرف فيها طوله $L$ على امتداد
الشريط هو $\Omega \propto \eu^{-L^2/2Na^2}$ (بالمسير العشوائي). (أ) أوجد $S(L)$
و، عند $U$ ثابتة، التوتر $f = -T\,\partial S/\partial L$. (ب) وصلابة
السلسلة وتعلّقها بالحرارة. (ج) ومن أجل $N = 1000$، و $a =
\qty{0.5}{nm}$، و \qty{300}{K}، و $L = \qty{10}{nm}$: أوجد $f$؛ وكم سلسلة على
التفرّع تعطي \qty{30}{N}؟ (د) وقدّر \omterm{prop:b2:waves-on-strings:chain}{معامل يونغ} لمطاط
فيه $10^{26}$ سلسلة في المتر المكعب، كلٌّ منها $N = 100$.
\end{exercise}

\section{مسألة: الشريط المطاطي ومخطط أطوار الماء}

\begin{problem}[{مسألة نهاية الأسبوع --- إنتروبيا تشدّ، وأطوار
تتنافس}]\label{pb:b2:thermodynamic-potentials:1}
\textbf{الجزء الأول --- الشريط المطاطي.}
شريط كتلته \qty{2}{g} (وسعته الحرارية \qty{4}{J/K}) له التوتر
$f = CT(L - L_0)$ مع $C = \qty{1}{N/m/K}$؛ وعند \qty{300}{K} و $L - L_0 =
\qty{0.1}{m}$، يكون $f = \qty{30}{N}$.
\begin{enumerate}
  \item اكتب $\dd U$ و $\dd F$ للشريط ($\dd U = T\dd S + f\dd L$).
  \item واستنتج علاقة ماكسويل $(\partial S/\partial L)_T = -(\partial
        f/\partial T)_L$.
  \item وبيّن أن $(\partial U/\partial L)_T = 0$ من أجل هذا الشريط: فماذا
        يعني ذلك؟
  \item والمدّ متساوي الحرارة من $L_0$ إلى $L_0 + \qty{0.1}{m}$ عند
        \qty{300}{K}: أوجد $\Delta S$، والحرارة $Q$، والعمل $W$؛ وتحقق من $\Delta U = 0$.
  \item والمدّ نفسه كظومًا: أوجد تغيّر الحرارة.
  \item ويحمل الشريط ثقلًا ثابتًا؛ ويُسخَّن \qty{30}{K}: أوجد
        $L - L_0$ الجديد. وصِف ``محرك المطاط'' (أي دولاب أضلاعه
        مطاطية يُسخَّن من جهة).
  \item وسلسلة من $N$ وصلة طول كلٍّ منها $a$ لها $\Omega \propto \eu^{-L^2/2Na^2}$:
        أوجد $S(L)$ و $f(L,T)$؛ ولمَ تنمو الصلابة مع $T$؟
  \item ومن أجل $N = 1000$، و $a = \qty{0.5}{nm}$، و $L = \qty{10}{nm}$، و \qty{300}{K}: أوجد $f$
        لسلسلة واحدة وعدد السلاسل من أجل \qty{30}{N}.
  \item ولمَ يتصرف نابض فولاذي عكس ذلك (فيهبط توتره
        مع $T$ عند طول ثابت)؟
\end{enumerate}

\textbf{الجزء الثاني --- مخطط أطوار الماء.}
المعطيات: $L_{\text{انصهار}} = \qty{334}{kJ/kg}$، و $v_{\text{جليد}} = \qty{1.091e-3}{m^3/kg}$،
و $v_{\text{سائل}} = \qty{1.000e-3}{m^3/kg}$؛ و $L_{\text{تبخر}} = \qty{2.26}{MJ/kg}$ عند
\qty{100}{\celsius}، و $v_{\text{بخار}} = \qty{1.67}{m^3/kg}$؛ و $L_{\text{m}} = \qty{40.7}{kJ/mol}$؛
والنقطة الثلاثية \qty{0.01}{\celsius} و \qty{611}{Pa}؛ والنقطة الحرجة \qty{647}{K}
و \qty{221}{bar}.
\begin{enumerate}[resume]
  \item ارسم $g(T)$ للجليد والسائل والبخار عند \qty{1}{bar}؛ وبرّر
        ترتيب الميول وعيّن الانتقالات.
  \item واستنتج صيغة كلابيرون من $g_1 = g_2$.
  \item وميل منحني الانصهار؛ وإشارته، ولمَ هي غير معتادة.
  \item ومتزلج: \qty{700}{N} على \qty{1}{cm^2}: أوجد تغيّر نقطة
        الانصهار؛ واستنتج ما يخصّ التفسير الشائع للتزلج.
  \item وتحت \qty{3}{km} من الجليد ($\rho = \qty{917}{kg/m^3}$): أوجد الضغط و
        نقطة الانصهار؛ والنتيجة على صفيحة الجليد.
  \item وميل منحني التبخر عند \qty{100}{\celsius}؛ ونقطة الغليان
        عند \qty{0.7}{bar} وعند \qty{2}{bar}.
  \item وكامِل كلاوزيوس--كلابيرون ببخار مثالي و
        $L_{\text{m}}$ ثابتة؛ وأوجد $P_{\text{إشباع}}$ عند \qty{20}{\celsius} وعند
        النقطة الثلاثية؛ وعلّق على الدقة.
  \item وعند النقطة الثلاثية، قارن ميلي منحنيي التسامي و
        التبخر ($L_{\text{تسام}} = L_{\text{انصهار}} + L_{\text{تبخر}}$).
  \item والماء المفرط التبريد عند \qty{-10}{\celsius}: أوجد $g_{\text{سائل}} - g_{\text{جليد}}$؛
        ولمَ تستطيع قطيرات الغيوم البقاء سائلة حتى \qty{-40}{\celsius}؟
  \item وماذا يحلّ بالمقادير $\Delta v$ و $L$ والميل عند النقطة
        الحرجة؛ وما المائع فوق الحرج؟
\end{enumerate}

\textbf{الجزء الثالث --- تطبيقات.}
\begin{enumerate}[resume]
  \item فسّر إنشاء ماكسويل للمساحتين المتساويتين على ميزوترمة
        فان دير فالس من استمرار $g$.
  \item والماء المفرط التسخين في الميكروويف ``يقفز'': فأي فرع من
        الميزوترمة، ولمَ الخطر؟
  \item والتجفيف بالتجميد: صِف المسار في مخطط $(P,T)$ و
        لمَ يغادر الماء من دون أن ينصهر.
  \item ولمَ يتبخر صحن ماء عند \qty{20}{\celsius} في هواء جافّ
        ويتوقف في هواء مشبع --- بدلالة الكمونات الكيميائية؟
  \item وقدر ضغط عند \qty{2}{bar}: لمَ ينضج الطعام أسرع بكثير
        (إذ يتضاعف معدل الكيمياء تقريبًا كل
        \qty{10}{K})؟
  \item ولخّص: أي كمون لأي قيد، وماذا تشتري
        علاقات ماكسويل.
\end{enumerate}
\end{problem}
