\chapter{انتشار الجسيمات}\label{ch:b2:particle-diffusion}

ألقِ بلورة صباغ في ماء ساكن وراقب: تتكوّن حولها
غيمة ملونة، وتكبر، وتلين حروفها، وتنتشر --- في
دقيقة على مليمتر، وفي ساعة على سنتيمتر، وفي أسبوع على
الكأس كلها. ولا شيء يدفع الصباغ؛ بل يحمله التزاحم
الدائم للجزيئات، الذي يرسل كل جسيم في مسير عشوائي
ويحمله، في المتوسط، من حيث الكثرة إلى حيث القلّة.
وهذا هو \emph{الانتشار}، وهو أبطأ النقل وأعمّه:
فهو يغذّي كل خلية بالأكسجين، ويشوب كل ترانزستور بالبور،
ويقسّي الفولاذ بالكربون، ويحمل عطرًا عبر غرفة --- وإن كان،
كما سنرى، لا في الزمن الذي قد يُظنّ. وهذا الفصل يعطي
الانتشار قانونه (فيك)، ومعادلته (من موازنة جسيمية)،
وحلوله المميّزة وسلالمه الزمنية ($L \sim \sqrt{Dt}$)، وأصله
المجهري في المسير العشوائي --- وهو يفسّر أيضًا لمَ تعرف \omterm{thm:b2:particle-diffusion:equation}{معادلة
الانتشار}، خلافًا لكل معادلات الميكانيك، جهةَ الزمن.
وسيعيد الفصل التالي استعمال ذلك كله من أجل الحرارة.

\section{كثافة الجسيمات والتدفق وقانون فيك}

\begin{definition}[الكثافة وتيار الجسيمات]\label{def:b2:particle-diffusion:flux}
من أجل نوع من الجسيمات (جزيئات، أو شوارد، أو ذرات في جسم صلب) تكون
\emph{الكثافة العددية}\index{كثافة عددية} $n(M,t)$ هي عدد
الجسيمات في وحدة الحجم حول $M$ (بالوحدة \unit{m^{-3}}؛ والتركيز
المولي هو $c = n/N_A$). و\emph{كثافة تيار
الجسيمات}\index{كثافة تيار الجسيمات} $\vect{j}_N$ هي الشعاع
الذي يجعل عدد الجسيمات العابرة عنصر سطح موجَّهًا
$\dd\vect S$ في $\dd t$ مساويًا $\vect{j}_N\cdot\dd\vect S\,\dd t$ (بالوحدة
\unit{m^{-2}.s^{-1}}): و\emph{تدفق الجسيمات} عبر سطح $S$ هو
$\Phi_N = \iint_S \vect{j}_N\cdot\dd\vect S$، أي عدد الجسيمات في
الثانية عبر $S$. ومن أجل جسيمات يحملها مائع يتحرك بالسرعة $\vect v$،
يكون $\vect{j}_N = n\vect v$ (وهو الحمل)؛ والانتشار هو النقل الذي
يبقى في مائع ساكن.
\end{definition}

\begin{theorem}[قانون فيك]\label{thm:b2:particle-diffusion:fick}
في وسط ساكن، حيث الكثافة غير منتظمة، يظهر تيار
جسيمات، متناسب مع متدرج الكثافة ومعاكس
له:
\[
\vect{j}_N = -D\,\vect{\operatorname{grad}}\,n ,
\]
حيث \emph{معامل الانتشار}\index{معامل الانتشار}
$D > 0$ (بالوحدة \unit{m^2/s}) يتعلق بالنوع المنتشر، وبالوسط
وبالحرارة. والجسيمات تنزل \emph{مع} انحدار متدرج الكثافة،
من المناطق الأكثف إلى الأرقّ، بمعدل يحدده $D$.
\end{theorem}

\begin{proof}
ظاهراتي (أي قانون تجربة، كقانون أوم): خطي في
المتدرج من أجل المتدرجات الصغيرة، ومتماثل المناحي في وسط متماثل المناحي، وذو
الإشارة التي تفرضها التجربة. ونموذج المسير العشوائي في
\cref{sec:b2:particle-diffusion:microscopic} يستنتجه، ويستنتج $D$ معه،
من الحركة الجزيئية.
\end{proof}

\begin{example}[رتب مقادير $D$]\label{ex:b2:particle-diffusion:orders}
الغازات: $D \sim \qty{1e-5}{m^2/s}$ (بخار الماء في الهواء
\qty{2.5e-5}{m^2/s}، وجزيء عطر \qty{5e-6}{m^2/s}). والسوائل:
$D \sim \qty{1e-9}{m^2/s}$ (الأكسجين في الماء \qty{2e-9}{m^2/s}، والسكر
\qty{5e-10}{m^2/s}، وبروتين \qty{1e-10}{m^2/s}). والأجسام الصلبة: ضئيل و
يتصاعد بشدة مع الحرارة، $D = D_0\,\eu^{-E_{\text{a}}/k_BT}$
(البور في السيليسيوم: \qty{1.5e-17}{m^2/s} عند \qty{1100}{\celsius}،
وصغير صغرًا لا يُقاس في حرارة الغرفة --- ولهذا يحفظ الترانزستور،
متى صُنع، مقطع تشويبه عقودًا). أي أربع رتب من
المقدار من الغاز إلى السائل، وثماني أو أكثر من السائل إلى الصلب.
\end{example}

\section{الموازنة الجسيمية ومعادلة الانتشار}

\begin{theorem}[الموازنة الجسيمية المحلية]\label{thm:b2:particle-diffusion:balance}
إذا لم تُحدَث الجسيمات ولم تُتلَف،
\[
\frac{\partial n}{\partial t} + \operatorname{div}\vect{j}_N = 0 ;
\]
ومع منبع يحدث $\sigma$ جسيمًا في وحدة الحجم والزمن
(تفاعل كيميائي، أو امتصاص)، يكون $\partial_t n +
\operatorname{div}\vect{j}_N = \sigma$. وفي بُعد واحد (حيث تتعلق \omterm{def:b2:charges-currents-conduction:densities}{الكثافة
والتيار} بالمتغير $x$ وحده): $\partial_t n + \partial_x j_N = 0$.
\end{theorem}

\begin{proof}
خذ الشريحة بين $x$ و $x + \dd x$، ومقطعها $S$: فهي تحوي
$n\,S\,\dd x$ جسيمًا؛ وفي $\dd t$ يدخل $j_N(x)S\,\dd t$ عبر
الوجه الأيسر ويخرج $j_N(x + \dd x)S\,\dd t$ عبر الأيمن، ومن ثَمّ
$\partial_t n\,S\,\dd x\,\dd t = -(j_N(x+\dd x) - j_N(x))S\,\dd t =
-\partial_x j_N\,\dd x\,S\,\dd t$. وفي ثلاثة أبعاد يعطي العدّ نفسه على
صندوق صغير \omterm{def:b2:maxwell-equations:operators}{التباعد} (أي التدفق الخارج من سطح مغلق في
وحدة الحجم، \cref{ch:b2:maxwell-equations})، أو مباشرةً: من أجل أي حجم
ثابت $V$، $\dd/\dd t \iiint_V n\,\dd\tau = -\iint_{\partial V}
\vect{j}_N\cdot\dd\vect S = -\iiint_V \operatorname{div}\vect{j}_N\,\dd\tau$
بمبرهنة أوستروغرادسكي.
\end{proof}

\begin{theorem}[معادلة الانتشار]\label{thm:b2:particle-diffusion:equation}
بجمع قانون فيك والموازنة، من أجل $D$ منتظم،
\[
\frac{\partial n}{\partial t} = D\,\Delta n + \sigma ,
\qquad\text{وفي بُعد واحد}\quad
\frac{\partial n}{\partial t} = D\,\frac{\partial^2 n}{\partial x^2} + \sigma .
\]
و\emph{معادلة الانتشار}\index{معادلة الانتشار} هذه خطية
(فالحلول تتراكب)، من المرتبة الأولى في الزمن والثانية في الفضاء
--- وليست معادلة أمواج: فليس لها سرعة انتشار، بل
علاقة مميّزة بين الطول والزمن،
\[
L \sim \sqrt{D t}, \qquad t \sim \frac{L^2}{D} :
\]
فالانتشار على ضعف المسافة يستغرق أربعة أضعاف الزمن.
\end{theorem}

\begin{proof}
$\partial_t n = -\operatorname{div}(-D\,\vect{\operatorname{grad}}\,n) +
\sigma = D\operatorname{div}\vect{\operatorname{grad}}\,n + \sigma = D\Delta
n + \sigma$. وأما السلالم: فقارن $n/t$ بالمقدار $Dn/L^2$.
\end{proof}

\begin{remark}[الانتشار غير عكوس]\label{rem:b2:particle-diffusion:irreversible}
بدّل $t$ بالمقدار $-t$ في \omterm{thm:b2:waves-on-strings:equation}{معادلة الأمواج}، $\partial_t^2 u = c^2
\partial_x^2 u$، فلا تتغير: ففيلم موجة يُدار بالمقلوب
موجة ممكنة. وافعل ذلك في \omterm{thm:b2:particle-diffusion:equation}{معادلة الانتشار} فتنقلب إشارة
الطرف الأيسر: فإن $n(x,-t)$ \emph{ليس} حلًّا. والغيمة التي
تنتشر طبيعية؛ أما التي تتجمع تلقائيًّا في بلورة
فلا --- فللانتشار سهم زمن، وهو سهم المبدأ الثاني
(و\cref{ch:b2:heat-conduction} يحسب الإنتروبيا التي يحدثها). وتبيّن
المعادلة أيضًا أن الانتشار يملّس: فحيث تكون $n$ محليًّا
عظمى ($\partial_x^2 n < 0$) تتناقص، وحيث تكون صغرى
تتزايد.
\end{remark}

\begin{example}[كم يستغرق ذلك؟]\label{ex:b2:particle-diffusion:times}
$t \sim L^2/D$. والأكسجين عبر خلية، $L = \qty{10}{\micro m}$ في الماء:
$10^{-10}/2 \times 10^{-9} = \qty{0.05}{s}$ --- فالانتشار يغذّي الخلية
بيسر؛ وعبر \qty{1}{mm} من نسيج: \qty{500}{s} --- أي بطيء جدًّا، ولهذا
لا يكون شيء حيّ أثخن من كسر من المليمتر بلا أوعية دموية. والسكر عبر
فنجان شاي غير مقلّب، $L = \qty{5}{cm}$:
$2.5 \times 10^{-3}/5 \times 10^{-10} = 5 \times 10^6\,\unit{s}$، أي شهران
--- فقلّبه. وعطر عبر غرفة بالانتشار وحده، \qty{5}{m} في
الهواء: $25/10^{-5} = \qty{3e6}{s}$، أي شهر؛ وأنت تشمّه في دقيقة
لأن الهواء يتحرك: و\emph{الحمل} ينقل، والانتشار لا يفعل إلا
إتمام المليمترات الأخيرة.
\end{example}

\section{الحلول المميّزة}

\begin{proposition}[النظام المستقر: الغشاء]\label{prop:b2:particle-diffusion:membrane}
بين خزّانين محفوظين عند $n_1$ و $n_2$، عبر غشاء
سماكته $e$ ومساحته $S$، تكون الكثافة المستقرة خطية،
\[
n(x) = n_1 + (n_2 - n_1)\frac{x}{e}, \qquad
j_N = D\,\frac{n_1 - n_2}{e}, \qquad
\Phi_N = \frac{n_1 - n_2}{R_{\text{d}}}, \quad R_{\text{d}} = \frac{e}{DS}:
\]
فللغشاء \emph{مقاومة انتشارية} $e/DS$، تمامًا كما للسلك
$\ell/\gamma S$ --- والمقاومات على التسلسل تتجمع، وغشاء رقيق
صغير $D$ قد يغلب مع ذلك.
\end{proposition}

\begin{proof}
مستقر، وفي بُعد واحد، وبلا منبع: $\partial_x^2 n = 0$، فتكون $n$
تآلفية؛ و $j_N = -D\,\dd n/\dd x$ منتظم.
\end{proof}

\begin{proposition}[النظام المستقر: الهندسة الكروية]\label{prop:b2:particle-diffusion:sphere}
حول كرة نصف قطرها $a$ يُحفَظ سطحها عند $n_0$، في
وسط تؤول فيه $n \to 0$ بعيدًا، تكون الكثافة المستقرة
\[
n(r) = n_0\,\frac{a}{r}, \qquad
\Phi_N = 4\pi D a\,n_0 :
\]
فالتيار الكلي المطلَق (أو الممتصّ، بقلب الإشارات) ينمو
مع \emph{نصف قطر} الكرة لا مع مساحتها --- فهندسة
المصبّ الانتشاري هي هندسة سعة كهربائية ساكنة.
\end{proposition}

\begin{proof}
$\Delta n = \frac{1}{r^2}\frac{\dd}{\dd r}(r^2\,\dd n/\dd r) = 0$ يعطي
$n = A + B/r$؛ والشروط الحدية تثبّت $A = 0$ و $B = n_0 a$؛ ثم
$\Phi_N = -4\pi r^2 D\,\dd n/\dd r = 4\pi D n_0 a$ عند كل $r$ (فلا
تراكم في النظام المستقر).
\end{proof}

\begin{proposition}[الغاوسي المنتشر]\label{prop:b2:particle-diffusion:gaussian}
$N$ جسيمًا في وحدة المساحة تُطلَق عند $t = 0$ على المستوي $x = 0$ من
وسط لانهائي تنتشر وفق
\[
n(x,t) = \frac{N}{\sqrt{4\pi D t}}\,\exp\Big(-\frac{x^2}{4Dt}\Big) :
\]
أي غاوسي انحرافه المعياري $\sigma(t) = \sqrt{2Dt}$، ومساحته ثابتة
$N$ (فالجسيمات محفوظة) وقمته متناقصة $\propto
1/\sqrt t$. وفي ثلاثة أبعاد، تنتشر $N$ جسيمًا تُطلَق عند نقطة
وفق $n = N\,(4\pi Dt)^{-3/2}\exp(-r^2/4Dt)$، مع $\langle r^2
\rangle = 6Dt$. ونبضة جسيمات تُطلَق عند \emph{سطح}
نصف فضاء (لا تستطيع مغادرته) تعطي ضعف العبارة أعلاه
من أجل $x > 0$.
\end{proposition}

\begin{proof}
عوّض: مع $u = x^2/4Dt$، يكون $\partial_t n = n\,(-1/2t + u/t)$ و
$D\,\partial_x^2 n = D\,n\,(-1/2Dt + x^2/4D^2t^2) = n\,(-1/2t + u/t)$؛
فهما متساويان. والتنظيم $\int n\,\dd x = N$ ينتج من $\int
\eu^{-x^2/4Dt}\dd x = \sqrt{4\pi Dt}$، و $\int x^2 n\,\dd x/N = 2Dt$.
وأما الوحدانية (بالنبضة الابتدائية المعطاة) فمقبولة.
\end{proof}

\begin{proposition}[تركيز سطحي ثابت]\label{prop:b2:particle-diffusion:erfc}
نصف فضاء $x > 0$ فارغ ابتداءً، ويُحفَظ سطحه عند
الكثافة $n_0$ منذ $t = 0$ (كغاز في تماس مع جسم صلب
يحلّه)، يمتلئ وفق
\[
n(x,t) = n_0\,\operatorname{erfc}\Big(\frac{x}{2\sqrt{Dt}}\Big),
\qquad
\operatorname{erfc}(u) = \frac{2}{\sqrt\pi}\int_u^{\infty}\eu^{-s^2}\dd s,
\]
وهو \emph{تابع الخطأ المتمّم}، الذي يهبط من $1$ عند $u =
0$ إلى $0.16$ عند $u = 1$ و $0.005$ عند $u = 2$: فعمق النفوذ
هو $\sim 2\sqrt{Dt}$ من جديد، والعدد الكلي للجسيمات الممتصّة في
وحدة المساحة هو $2n_0\sqrt{Dt/\pi}$.
\end{proposition}

\begin{proof}
ابحث عن $n = f(u)$ مع $u = x/2\sqrt{Dt}$: فتصير المعادلة
$f'' + 2uf' = 0$، ومن ثَمّ $f' \propto \eu^{-u^2}$ وتكون $f$ تابع
خطأ؛ والشرطان $f(0) = n_0$ و $f(\infty) = 0$ ينتقيان
$\operatorname{erfc}$. والعدد الممتصّ هو $\int_0^\infty n\,\dd x =
n_0\,2\sqrt{Dt}\int_0^\infty\operatorname{erfc}(u)\dd u = 2n_0\sqrt{Dt/\pi}$.
\end{proof}

\begin{example}[تقسية الفولاذ]\label{ex:b2:particle-diffusion:carburising}
قطعة فولاذية تُحفَظ عند \qty{900}{\celsius} في غاز غني بالكربون تمتصّ
الكربون عند سطحها؛ ومع $D = \qty{5e-12}{m^2/s}$ للكربون في الحديد

\end{example}

\section{الصورة المجهرية: المسير العشوائي}\label{sec:b2:particle-diffusion:microscopic}

\begin{proposition}[المسير العشوائي ومعامل الانتشار]\label{prop:b2:particle-diffusion:walk}
جسيم يخطو خطوة طولها $\ell$ في جهة عشوائية
كل $\tau$ (كجزيء بين تصادمين) له، بعد $N =
t/\tau$ خطوة، متوسط مربع انتقال
\[
\langle x^2\rangle = N\ell^2 = \frac{\ell^2}{\tau}\,t
\quad\text{(بُعد واحد)}, \qquad
\langle r^2\rangle = 3\langle x^2\rangle \quad\text{(ثلاثة أبعاد)}.
\]
وبالمقارنة مع الحلّ الغاوسي، $\langle x^2\rangle = 2Dt$:
\[
D = \frac{\ell^2}{2\tau} \quad\text{(بُعد واحد)}, \qquad
D = \frac{\ell^2}{6\tau} = \frac{\ell\,v^*}{6}\ \sim\ \frac{\ell\,v^*}{3}
\quad\text{(ثلاثة أبعاد، مع } v^* = \ell/\tau\text{؛ والنظرية الحركية تعطي } \tfrac13) .
\]
ومسافة الماشي تنمو وفق $\sqrt t$ لا وفق $t$: فليذهب ضعف البعد
يحتاج إلى أربعة أضعاف عدد الخطوات.
\end{proposition}

\begin{proof}
$x = \sum_i \epsilon_i\ell$ مع $\epsilon_i = \pm1$ مستقلة (في بُعد واحد):
$\langle x^2\rangle = \sum_{i,j}\langle\epsilon_i\epsilon_j\rangle\ell^2 =
N\ell^2$، إذ تتوسط الحدود المتقاطعة إلى الصفر. وتوزيع $x$
من أجل $N$ الكبيرة يؤول إلى غاوسي (وهي مبرهنة النهاية المركزية، وبرهانها
من درس الاحتمالات) --- ولهذا يكون القانون
العياني \omterm{thm:b2:particle-diffusion:equation}{معادلة الانتشار}؛ وأما قانون فيك نفسه فينتج
من عدّ الماشين العابرين مستويًا من الجهتين: $j_N \approx
-\tfrac12\ell v^*\,\partial_x n$ (ومتوسط أدقّ يعطي $1/3$ في
ثلاثة أبعاد).
\end{proof}

\begin{example}[من الجزيئات إلى $D$]\label{ex:b2:particle-diffusion:kinetic}
الهواء في حرارة الغرفة: المسار الحر الوسطي $\ell \approx \qty{70}{nm}$،
والسرعة الوسطى $v^* \approx \qty{500}{m/s}$: $D \approx \ell v^*/3 =
\qty{1.2e-5}{m^2/s}$، كما يُقاس. ولينتشر جزيء \qty{1}{m}
يحتاج إلى $N = (L/\ell)^2 = 2 \times 10^{14}$ تصادم، أي $N\ell/v^*
\approx \qty{3e4}{s}$ --- أي ثماني ساعات لرحلة كان يقطعها في
\qty{2}{ms} لو طار مستقيمًا. وفي سائل، تحقق كرة نصف قطرها $a$
تتقاذفها الجزيئات علاقة ستوكس--أينشتاين $D =
k_BT/6\pi\eta a$ (أينشتاين 1905): فمن أجل السكر، $a \approx \qty{0.4}{nm}$
في الماء، $D = \qty{5e-10}{m^2/s}$؛ ومن أجل حبيبة \qty{1}{\micro m}،
$\qty{4e-13}{m^2/s}$، أي انتقال ميكرومتر في الثانية ---
وهي الحركة البراونية التي قاسها بيران ليعدّ عدد أفوغادرو. وفي
جسم صلب لا تقفز الذرة إلى موقع مجاور إلا حين يزوّدها تقلّب
حراري بطاقة التنشيط: $D = D_0\,\eu^{-E_{\text{a}}/k_BT}$،
ويتضاعف كل بضع عشرات من الكلفنات قرب \qty{1000}{\celsius}.
\end{example}

\begin{method}[تقديرات الانتشار]\label{meth:b2:particle-diffusion:method}
(1) عيّن $D$ (الغاز $10^{-5}$، والسائل $10^{-9}$، والصلب بأرينيوس).
(2) والزمن أو المسافة: $t \sim L^2/D$، و $L \sim \sqrt{Dt}$ (مع المضبوط
$\sqrt{2Dt}$ لعرض الغاوسي، و $2\sqrt{Dt}$ لعمق تابع الخطأ المتمّم).
(3) والمستقر: خطي في شريحة ($R_{\text{d}} = e/DS$)، و $1/r$ حول
كرة ($\Phi = 4\pi D a\,\Delta n$)، و $\ln r$ حول أسطوانة. (4)
والعابر: غاوسي من أجل نبضة، وتابع خطأ متمّم من أجل سطح محفوظ؛ وتراكَب.
(5) واسأل ألا يغلب الحمل. (6) والتحقق المجهري:
$D \sim \ell v^*/3$.
\end{method}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:7}
\emph{الحفظ واللاعكوسية.} (أ) من المعادلة، بيّن
أن $\int n\,\dd x$ ثابت وأن $\dd\langle x^2\rangle/\dd t =
2D$ (كامِل بالتجزئة، و $n \to 0$ في اللانهاية). (ب) وبيّن أن
$n(x,-t)$ لا تحقق المعادلة. (ج) وماذا يصير الحلّ
الغاوسي من أجل $t < 0$؟ (د) وبيّن أن $\int n^2\,\dd x$ لا يمكن إلا أن
يتناقص: فالانتشار يسطّح.
\end{exercise}

\qty{900}{\celsius}؛ والسطح محفوظ عند $n_0$. (أ) تحقق من أن $n_0
\operatorname{erfc}(x/2\sqrt{Dt})$ يحلّ المعادلة. (ب) وأوجد العمق الذي
تكون فيه $n = 0.1\,n_0$ بعد \qty{4}{h} (و $\operatorname{erfc}(1.16) = 0.1$).
(ج) والزمن اللازم لضعف العمق. (د) وعند \qty{950}{\celsius} يكون $D$ أكبر
بمقدار $2.5$ مرة: أوجد الزمن الموفَّر.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:9}
\emph{ستوكس--أينشتاين.} $D = k_BT/6\pi\eta a$، وللماء $\eta =
\qty{1e-3}{Pa.s}$، وعند \qty{300}{K}. (أ) أوجد $D$ من أجل $a = \qty{1}{\micro m}$ و
الانتقال التربيعي الوسطي في \qty{1}{s} و \qty{1}{min}. (ب) و $D$
من أجل بروتين، $a = \qty{3}{nm}$. (ج) وللسكر $D = \qty{5e-10}{m^2/s}$:
أوجد نصف القطر الفعّال. (د) وكيف يعطي قياس $\langle x^2\rangle$ لحبيبة
تحت المجهر عددَ أفوغادرو؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:10}
\emph{تأخّر الغشاء.} غشاء سماكته $e$، فارغ ابتداءً،
يُعرَّض على أحد وجهيه للكثافة $n_1$ عند $t = 0$، والوجه الآخر محفوظ
عند $0$. (أ) ما التدفق المستقر النهائي؟ (ب) وبرهن أن التدفق
على الوجه البعيد يرتفع خلال زمن $\sim e^2/D$ (والتأخّر المضبوط
$e^2/6D$). (ج) ولصاقة دواء مع $e = \qty{20}{\micro m}$، و $D =
\qty{1e-13}{m^2/s}$ عبر الطبقة الخارجية للجلد: أوجد زمن التأخّر. (د) ولمَ
يفيد التأخّر في قياس $D$ ويفيد التدفق المستقر في قياس
$D \times$ الانحلالية؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:11}
\emph{الماصّ التامّ.} كرة نصف قطرها $a$ تمتصّ كل
جسيم يلامسها، في وسط عند $n_\infty$ بعيدًا. (أ)
أوجد المقطع المستقر والتيار الكلي الملتقَط. (ب) وجرثومة، $a =
\qty{1}{\micro m}$، في محلول سكر عند $n_\infty = \qty{6e20}{m^{-3}}$،
و $D = \qty{5e-10}{m^2/s}$: أوجد الجزيئات الملتقَطة في الثانية. (ج) ويغطّي
سطحها $N_{\text{r}}$ مستقبلًا ماصًّا صغيرًا نصف قطره
$s$، والباقي عاكس: وكل مستقبل وحده يلتقط $\approx
4Ds n_\infty$ (كقرص)؛ بيّن أن الخلية كلها تلتقط قرابة
العظمى متى صار $N_{\text{r}} s \gg a$، أي بكسر ضئيل من
سطحها مغطًّى. (د) وعلّق: لمَ تستطيع الخلايا تحمّل آلاف
المستقبلات المختلفة.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:12}
\emph{حلّها على شبكة.} قسّم الفضاء إلى خلايا حجمها $\Delta x$
والزمن إلى خطوات $\Delta t$؛ واكتب $n_i^{k+1} = n_i^k + \alpha\,(n_{i+1}^k
- 2n_i^k + n_{i-1}^k)$ مع $\alpha = D\Delta t/\Delta x^2$. (أ) برّر
ذلك من المعادلة. (ب) وفسّره مسيرًا عشوائيًّا حين $\alpha =
1/2$. (ج) وبيّن أنه من أجل $\alpha > 1/2$ تنمو كثافة تتناوب $+,-,+,-$
من خلية إلى خلية: أي عدم استقرار. (د) ومن أجل مقطع البور في
المسألة أدناه ($D = \qty{1.5e-17}{m^2/s}$، و $\Delta x = \qty{10}{nm}$)،
أوجد أكبر $\Delta t$ مستقر وعدد الخطوات في ساعة.
\end{exercise}

\section{مسألة: رقاقة مشوبة وغرفة معطَّرة}

\begin{problem}[{مسألة نهاية الأسبوع --- الانتشار في جسم صلب وفي غاز}]\label{pb:b2:particle-diffusion:1}
\textbf{الجزء الأول --- دفع البور في السيليسيوم.}
ينتشر البور في السيليسيوم وفق $D = D_0\,\eu^{-E_{\text{a}}/k_BT}$، مع $D_0 =
\qty{7.6e-5}{m^2/s}$، و $E_{\text{a}} = \qty{3.46}{eV}$؛ و $k_B =
\qty{8.62e-5}{eV/K}$. وقد رُسّبت جرعة $Q = \qty{1e18}{m^{-2}}$ من البور
في طبقة رقيقة جدًّا على سطح رقاقة تشويبها
الخلفي $n_B = \qty{1e21}{m^{-3}}$؛ ثم تُسخَّن
(``تُدفَع إلى الداخل'') عند \qty{1100}{\celsius} ساعةً واحدة. والسطح
يعكس البور (فلا هروب).
\begin{enumerate}
  \item احسب $D$ عند \qty{1100}{\celsius} وعند \qty{1000}{\celsius}.
        وبأي عامل يتغير على هذه \qty{100}{K}؟
  \item ولمَ يكون المقطع بعد الدفع $n(x,t) = (Q/\sqrt{\pi
        Dt})\exp(-x^2/4Dt)$ لا الغاوسي في
        \cref{prop:b2:particle-diffusion:gaussian}؟ وتحقق من أن
        تكامله على $x > 0$ هو $Q$.
  \item والتركيز السطحي بعد ساعة.
  \item و\emph{عمق الوصلة} $x_j$ هو حيث $n = n_B$: احسبه.
  \item وكيف يتغير $x_j$ إذا دام الدفع أربع ساعات؟
        (وانتبه: فالتركيز السطحي يتغير أيضًا.)
  \item والتغير النسبي في $D$ من أجل خطأ \qty{10}{K} في الحرارة؛
        وضبط الحرارة اللازم ليكون $x_j$ بدقة \qty{1}{\%}.
  \item وعند حرارة الغرفة، $D$: قدّر الزمن اللازم لينتقل المقطع
        \omterm{def:b2:maxwell-equations:operators}{بتباعد} ذري واحد، واستنتج.
\end{enumerate}

\textbf{الجزء الثاني --- الترسيب المسبق.}
وقد أُدخلت الجرعة نفسها عند \qty{950}{\celsius} من غاز يحفظ
السطح عند حدّ الانحلالية $n_0 = \qty{2e26}{m^{-3}}$ مدة
\qty{30}{min}.
\begin{enumerate}[resume]
  \item أوجد $D$ عند \qty{950}{\celsius} و $\sqrt{Dt}$ من أجل \qty{30}{min}.
  \item والمقطع في نهاية الترسيب المسبق؛ والعمق الذي تكون عنده $n =
        n_B$ (و $\operatorname{erfc}(3.2) \approx 6 \times 10^{-6}$).
  \item والجرعة المدخَلة؛ وقارنها بجرعة $Q$ في الجزء الأول.
  \item ولمَ يُرسَّب مسبقًا عند حرارة أدنى ويُدفَع عند
        حرارة أعلى؟
  \item وبرّر أنه يمكن، في أثناء الدفع، معاملة الطبقة المرسَّبة
        مسبقًا كأنها رقيقة لانهائيًّا.
\end{enumerate}

\textbf{الجزء الثالث --- عطر في غرفة.}
قطرة عطر (\qty{1}{mg}، وكتلتها المولية \qty{150}{g/mol}) تتبخر
دفعة واحدة في ركن غرفة ساكنة عند \qty{300}{K} و \qty{1}{bar}.
وخذ قطر تصادم جزيء العطر مع الهواء $d =
\qty{0.5}{nm}$ \omterm{def:b2:particle-diffusion:flux}{والكثافة العددية} للهواء $n_{\text{a}} = P/k_BT$.
\begin{enumerate}[resume]
  \item أوجد عدد الجزيئات المطلَقة؛ \omterm{def:b2:particle-diffusion:flux}{والكثافة العددية} للهواء.
  \item والمسار الحر الوسطي $\ell = 1/(\sqrt2\,\pi d^2 n_{\text{a}})$ لجزيء
        العطر، وسرعته الوسطى $v^* = \sqrt{8RT/\pi M}$.
  \item وقدّر $D = \ell v^*/3$.
  \item وزمن الانتشار على \qty{5}{m}؛ وإذا انساق الهواء بسرعة
        \qty{0.1}{m/s}؟
  \item واكتب الغاوسي الثلاثي الأبعاد للغيمة (وهي في ركن:
        فالجدران تعكس، فاضرب في $8$). وعند نقطة على بعد \qty{1}{m}،
        في أي لحظة يكون التركيز أعظم ما يكون؟
  \item وأعظم \omterm{def:b2:particle-diffusion:flux}{كثافة عددية} هناك؛ وقارنها بعتبة إدراك
        قدرها \qty{1e13}{m^{-3}}.
  \item وحرف الغيمة: على أي بعد تكون الكثافة $10^{-6}$ من
        قيمتها المركزية بعد يوم؟
  \item وعدد التصادمات التي يعانيها جزيء عطر في يوم.
\end{enumerate}

\textbf{الجزء الرابع --- الماشي وسهم الزمن.}
\begin{enumerate}[resume]
  \item ماشٍ يخطو $N$ خطوة $\pm\ell$ على مستقيم: أوجد متوسط الانتقال
        ومتوسط مربعه؛ وبيّن أن $\langle x^2\rangle = 2Dt$
        يعيّن $D = \ell^2/2\tau$.
  \item واحتمال أن يعود الماشي إلى بدايته بالضبط بعد
        $N = 2$ و $4$ و $6$ خطوات؛ وعلّق على المنحى.
  \item وصوّر غيمة منتشرة وأدر الفيلم بالمقلوب: فماذا ترى،
        وأي معادلة تُنتهَك؟
  \item وفسّر في فقرة كيف تنتج الحركة العكوسة للجزيئات
        معادلةَ الانتشار غير العكوسة.
  \item ولخّص: سلّم الطول، وسلّم الزمن، والمقدار الوحيد
        الذي يميّز غازًا وسائلًا وجسمًا صلبًا ساخنًا.
\end{enumerate}
\end{problem}
