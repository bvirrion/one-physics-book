\section{الإشعاع الحراري والجسم الأسود}

\begin{definition}[الإشعاع الحراري والامتصاصية والجسم الأسود]\label{def:b2:thermal-radiation:blackbody}
كل جسم عند الحرارة $T$ يصدر إشعاعًا كهرمغناطيسيًّا من
الاضطراب الحراري لشحناته: وهو \emph{إشعاعه
الحراري}\index{إشعاع حراري}. ويتلقى سطحه أيضًا
إشعاعًا من محيطه ويمتصّ منه الكسر $\alpha$
(وهي \emph{الامتصاصية}، بين $0$ و $1$، وهي في العام تتعلق
بطول الموجة)، فيعكس الباقي أو ينفذه. و\emph{الجسم
الأسود}\index{جسم أسود} يمتصّ كل شيء ($\alpha = 1$ عند كل
طول موجة). وثقب صغير في جدار جوف مغلق هو الجسم الأسود
العملي: فالإشعاع الداخل فيه يُمتصّ عند الجدران
قبل أن يجد سبيل الخروج.
\end{definition}

\begin{proposition}[إشعاع التوازن وقانون كيرشوف]\label{prop:b2:thermal-radiation:kirchhoff}
داخل جوف مغلق جدرانه عند $T$، يكون الإشعاع في
توازن مع الجدران ويكون \emph{عامًّا}: متماثل المناحي،
غير مستقطب، وله كثافة طاقة طيفية $u_\nu(T)$ (أي الطاقة في وحدة
الحجم ووحدة التواتر) تتعلق بالمتغيرين $T$ و $\nu$ وحدهما، لا بمادة
الجوف ولا بشكله. والإشعاع الخارج من ثقب صغير في
جوف كهذا هو \emph{إشعاع \omterm{def:b2:thermal-radiation:blackbody}{الجسم الأسود}} عند $T$. والسطح الذي
يمتصّ الكسر $\alpha_\nu$ من الإشعاع الذي يتلقاه عند
التواتر $\nu$ يصدر، عند ذلك التواتر، الكسر
$\varepsilon_\nu = \alpha_\nu$ مما كان يصدره \omterm{def:b2:thermal-radiation:blackbody}{جسم أسود} عند الحرارة
نفسها (وهو \emph{إصداريته}\index{إصدارية}):
\emph{فالماصّ الجيد مصدِر جيد}.
\end{proposition}

\begin{proof}
أما العمومية: فاصِل جوفين عند $T$ نفسها بثقب فيه
مرشّح يمرّر تواترًا واحدًا؛ فلو اختلفت الكثافتان لجرت الطاقة
من أحدهما إلى الآخر عند حرارتين متساويتين، وذلك ينتهك المبدأ
الثاني. وأما كيرشوف: فضع السطح في الجوف عند $T$ نفسه؛ ففي
التوازن يجب أن يصدر عند كل تواتر ما يمتصّه بالضبط،
$\varepsilon_\nu u_\nu = \alpha_\nu u_\nu$.
\end{proof}

\section{قانون بلانك}

\begin{theorem}[قانون بلانك]\label{thm:b2:thermal-radiation:planck}
كثافة الطاقة الطيفية لإشعاع \omterm{def:b2:thermal-radiation:blackbody}{الجسم الأسود} عند الحرارة
$T$ هي
\[
u_\nu(T) = \frac{8\pi h\nu^3}{c^3}\,\frac{1}{\eu^{h\nu/k_BT} - 1},
\]
والاستطاعة التي يشعّها سطح أسود في وحدة المساحة ووحدة
طول الموجة (وهي \emph{خروجيته الطيفية}\index{خروجية طيفية}) هي
\[
M_\lambda(T) = \frac{2\pi hc^2}{\lambda^5}\,\frac{1}{\eu^{hc/\lambda k_BT} - 1}.
\]
وثمة حدّان: من أجل $h\nu \ll k_BT$، $u_\nu \approx 8\pi\nu^2 k_BT/c^3$
(وهو رايلي--جينز: إذ يحمل كل نمط من أنماط الجوف الطاقة الكلاسيكية
$k_BT$ --- وهو قانون تكامله لانهائي، أي ``كارثة
فوق البنفسجي'')؛ ومن أجل $h\nu \gg k_BT$، $u_\nu \approx (8\pi h\nu^3/c^3)
\eu^{-h\nu/k_BT}$ (وهو فين: أي عامل بولتزمان لفوتون طاقته
$h\nu$، \cref{ch:b2:boltzmann-factor}).
\end{theorem}

\begin{proof}
مقبول: فعدّ أنماط الجوف ($8\pi\nu^2/c^3$ في وحدة
الحجم والتواتر، مع استقطابين) من مجلد السنة الثالثة،
ومتوسط طاقة النمط، $h\nu/(\eu^{h\nu/k_BT} - 1)$ بدل
الكلاسيكي $k_BT$، هو الفرضية الكمومية: إذ تُتبادَل الطاقة مع
النمط في كمّات $h\nu$، فتتجمد الأنماط التي $h\nu \gg k_BT$
فيها. والعلاقة $M_\lambda\,\dd\lambda = (c/4)\,u_\nu\,\dd\nu$ هي
العامل الهندسي المحسوب في \cref{thm:b2:thermal-radiation:stefan}.
\end{proof}

\begin{proposition}[قانون فين للانزياح]\label{prop:b2:thermal-radiation:wien}
تبلغ \omterm{thm:b2:thermal-radiation:planck}{الخروجية الطيفية} $M_\lambda$ قمتها عند
\[
\lambda_{\max}\,T = \frac{hc}{4.965\,k_B} = \qty{2.90e-3}{m.K} .
\]
فالجسم عند \qty{300}{K} يشعّ في معظمه قرب \qty{10}{\micro m}؛ والشمس
(\qty{5800}{K}) قرب \qty{0.5}{\micro m}، في الأخضر، حيث تكون العين
أشدّ حساسية --- لا مصادفةً.
\end{proposition}

\begin{proof}
مع $x = hc/\lambda k_BT$، يكون $M_\lambda \propto x^5/(\eu^x - 1)$؛ و $\dd/\dd x = 0$
يعطي $5(\eu^x - 1) = x\eu^x$، أي $x = 5(1 - \eu^{-x})$، وجذره
غير المعدوم $x = 4.965$.
\end{proof}

\begin{theorem}[قانون ستيفان--بولتزمان]\label{thm:b2:thermal-radiation:stefan}
سطح أسود عند $T$ يشعّ، في وحدة المساحة، الاستطاعة الكلية
\[
M = \int_0^\infty M_\lambda\,\dd\lambda = \sigma T^4, \qquad
\sigma = \frac{2\pi^5k_B^4}{15h^3c^2} = \qty{5.67e-8}{W/m^2/K^4} ;
\]
وكثافة طاقة إشعاع التوازن هي $u = (4\sigma/c)T^4$
وضغطه $P = u/3$. والسطح الرمادي الذي \omterm{prop:b2:thermal-radiation:kirchhoff}{إصداريته} $\varepsilon$
يشعّ $\varepsilon\sigma T^4$.
\end{theorem}

\begin{proof}
كامِل كثافة بلانك: $u = \int u_\nu\,\dd\nu = (8\pi h/c^3)(k_BT/h)^4
\int_0^\infty x^3\,\dd x/(\eu^x - 1)$ والتكامل هو $\pi^4/15$
(وهو مقبول من التحليل): $u = (8\pi^5k_B^4/15h^3c^3)\,T^4$. والاستطاعة
الخارجة من ثقب مساحته الواحدة في جدار الجوف هي $cu/4$: إذ
الإشعاع متماثل المناحي، والفوتونات المتجهة نحو الثقب ضمن
الزاوية المجسمة $\dd\Omega$ عند الزاوية $\theta$ عن ناظمه تحمل
التدفق $c\,u\,(\dd\Omega/4\pi)\cos\theta$، و $\int\cos\theta\,\dd\Omega/4\pi$
على نصف الكرة الخارجي يساوي $1/4$. ومن ثَمّ $M = cu/4 = \sigma T^4$.
وأما الضغط، كما في أي غاز متماثل المناحي من جسيمات تتحرك بالسرعة $c$، فهو
$u/3$.
\end{proof}

\begin{example}[رتب المقادير]\label{ex:b2:thermal-radiation:orders}
عند \qty{300}{K}: $\sigma T^4 = \qty{460}{W/m^2}$ --- فالشخص الذي مساحته
\qty{1.7}{m^2} يشعّ \qty{800}{W}، لكنه يتلقى ما يقاربها من
الجدران عند \qty{293}{K}؛ والضياع الصافي، $\sigma A(T^4 - T_0^4) \approx
\qty{100}{W}$، هو كل خرج الجسم الاستقلابي، ولهذا
يحتاج المرء إلى ثياب في غرفة تبدو معتدلة. وأما سطح الشمس: فيكون $\sigma
(5800)^4 = \qty{6.4e7}{W/m^2}$، والمجموع $3.9 \times 10^{26}\,\unit{W}$. وشعيرة
مصباح عند \qty{2800}{K}: \qty{3.5e6}{W/m^2}، أي \qty{60}{W} من نصف
سنتيمتر مربع من التنغستن. والخلفية الكونية، \qty{2.7}{K}:
قمتها عند \qty{1}{mm}، و $\qty{3e-6}{W/m^2}$، و $400$ فوتون في السنتيمتر
المكعب من الكون.
\end{example}

\section{التبادلات الإشعاعية}

\begin{proposition}[التبادل الصافي والمعامل الإشعاعي]\label{prop:b2:thermal-radiation:exchange}
جسم رمادي صغير (\omterm{prop:b2:thermal-radiation:kirchhoff}{إصداريته} $\varepsilon$، ومساحته $S$، وحرارته $T$)
داخل محيط كبير عند $T_0$ يفقد الاستطاعة الصافية
\[
P = \varepsilon\sigma S\,(T^4 - T_0^4)
\ \approx\ 4\varepsilon\sigma T_0^3\,S\,(T - T_0) = h_{\text{r}}S(T - T_0)
\quad\text{من أجل } |T - T_0| \ll T_0 ,
\]
حيث $h_{\text{r}} = 4\varepsilon\sigma T_0^3 \approx \qty{6}{W/m^2/K}$ في
حرارة الغرفة --- أي مقارِن للحمل الطبيعي: ففي غرفة،
يحمل الإشعاع نحو نصف الحرارة الخارجة من جسم أو مشعّ.
وبين صفيحتين سوداوين كبيرتين متوازيتين تكون كثافة التدفق الصافي
$\sigma(T_1^4 - T_2^4)$؛ ومن أجل صفيحتين رماديتين إصداريتاهما $\varepsilon_1$
و $\varepsilon_2$ تكون $\sigma(T_1^4 - T_2^4)/(1/\varepsilon_1 + 1/\varepsilon_2
- 1)$، وكل \emph{درع إشعاعي} رقيق يوضع بينهما يقسمها
أكثر --- وهو مبدأ ترمس الخلاء ورقاقة الذهب على
الأقمار الاصطناعية.
\end{proposition}

\begin{proof}
يصدر الجسم $\varepsilon\sigma T^4$ ويمتصّ الكسر $\alpha =
\varepsilon$ من إشعاع \omterm{def:b2:thermal-radiation:blackbody}{الجسم الأسود} $\sigma T_0^4$ الذي يملأ الحيّز
(بكيرشوف). ثم خطِّ $T^4 - T_0^4 \approx 4T_0^3(T - T_0)$. وأما
الصفيحتان الرماديتان: فاجمع الانعكاسات المتعددة (وهي متتالية هندسية).
\end{proof}

\begin{example}[الصقيع تحت سماء صافية، وآلة التصوير الحرارية]\label{ex:b2:thermal-radiation:frost}
في ليلة صافية تشعّ السماء كجسم عند نحو \qty{250}{K}
(أي الجوّ الأعلى الجافّ)، بينما الهواء قرب الأرض عند
\qty{278}{K}. وورقة أو سقف سيارة ($\varepsilon = 0.95$) يوازن
ضياعه الإشعاعي إلى السماء بالحمل من الهواء ($h =
\qty{5}{W/m^2/K}$): فإن $\varepsilon\sigma(T^4 - 250^4) = h(278 - T)$ تعطي $T
\approx \qty{266}{K}$ --- أي صقيع عند \qty{-7}{\celsius} على العشب بينما
يقرأ مقياس الحرارة $+5$. وغيمة (تشعّ عند \qty{275}{K}) ترفع $T$
إلى \qty{277}{K}: فلا صقيع. وآلة التصوير الحرارية تقيس الإشعاعية قرب
\qty{10}{\micro m} وتحوّلها إلى حرارة بافتراض
$\varepsilon \approx 0.95$؛ والمعدن المصقول ($\varepsilon = 0.1$) يعكس
الغرفة في معظمه فيبدو عند حرارة الغرفة مهما كانت حرارته
--- ومن هنا الشريط الأسود الذي يلصقه المصورون الحراريون على الأجزاء اللامعة.
\end{example}

\section{الشمس والأرض والدفيئة}

\begin{proposition}[الثابت الشمسي والحرارة الفعّالة]\label{prop:b2:thermal-radiation:earth}
تسلّم الشمس (ونصف قطرها $R_\odot$، وحرارة سطحها $T_\odot$) عند
البعد $d$ كثافة التدفق $S = \sigma T_\odot^4 (R_\odot/d)^2 =
\qty{1361}{W/m^2}$ فوق جوّ الأرض (وهو \emph{الثابت
الشمسي}\index{الثابت الشمسي}). وكوكب بياضه $A$ (أي الكسر
المنعكس) يمتصّ $(1 - A)S\pi R^2$ ويشعّه، في الحالة المستقرة،
من سطحه كله $4\pi R^2$ \omterm{def:b2:thermal-radiation:blackbody}{كجسم أسود} عند \emph{حرارته
الفعّالة}\index{حرارة فعّالة}
\[
T_{\text{e}} = \Big(\frac{(1 - A)S}{4\sigma}\Big)^{1/4} = \qty{255}{K}
\quad\text{من أجل الأرض } (A = 0.30) .
\]
\end{proposition}

\begin{proof}
حفظ الطاقة: الممتصّ $=$ المصدَر، \omterm{prop:b2:laser:gain}{والمقطع العرضي} $\pi R^2$
من أجل الحزمة المعترَضة، والكرة كاملةً من أجل الإصدار (ليلًا
ونهارًا، موزَّعًا بالدوران والدوران الجوي).
\end{proof}

\begin{proposition}[نموذج الدفيئة ذي الطبقة الواحدة]\label{prop:b2:thermal-radiation:greenhouse}
أحِط الكوكب بطبقة جوّ رقيقة شفافة
لضوء الشمس لكنها تمتصّ كل ما يصدره السطح من تحت الأحمر، وتشعّ
\omterm{def:b2:thermal-radiation:blackbody}{كجسم أسود} عند $T_{\text{a}}$ صاعدًا ونازلًا. وفي الحالة
المستقرة،
\[
T_{\text{a}} = T_{\text{e}}, \qquad
T_{\text{s}} = 2^{1/4}\,T_{\text{e}} = \qty{303}{K} .
\]
وإذا امتصّت الطبقة الكسر $\varepsilon$ من تحت الأحمر وحده،
كان $T_{\text{s}}^4 = T_{\text{e}}^4/(1 - \varepsilon/2)$؛ والقيمة المرصودة
\qty{288}{K} تقابل $\varepsilon \approx 0.78$. والسطح
أدفأ من \omterm{prop:b2:thermal-radiation:earth}{الحرارة الفعّالة} لأنه يتلقى، زيادةً
على الشمس، تحت الأحمر الذي يعيده الجوّ نازلًا --- وهو
\emph{أثر الدفيئة}\index{أثر الدفيئة}، ومقداره \qty{33}{K} على الأرض،
ولولاه لتجمدت المحيطات.
\end{proposition}

\begin{proof}
موازنة الطبقة: تمتصّ $\sigma T_{\text{s}}^4$، وتصدر $2\sigma
T_{\text{a}}^4$. وموازنة الكوكب منظورًا إليه من الفضاء: يصدر $\sigma
T_{\text{a}}^4 = (1 - A)S/4 = \sigma T_{\text{e}}^4$. وموازنة
السطح: يتلقى $(1 - A)S/4 + \sigma T_{\text{a}}^4 = 2\sigma T_{\text{e}}^4$،
ويصدر $\sigma T_{\text{s}}^4$. وأما الطبقة الجزئية: فاستبدل بتحت الأحمر
الممتصّ والمصدَر $\varepsilon\sigma T_{\text{s}}^4$ و
$\varepsilon\sigma T_{\text{a}}^4$ ودع الكسر $1 - \varepsilon$
من إشعاع السطح يفلت مباشرة.
\end{proof}

\begin{remark}[لمَ تكون الدفيئة انتقائية]\label{rem:b2:thermal-radiation:selective}
يبلغ ضوء الشمس قمته عند \qty{0.5}{\micro m}، وإشعاع الأرض عند
\qty{10}{\micro m}: فالطيفان لا يكادان يتراكبان، فيمكن لغاز أن يكون
شفافًا للأول ماصًّا للثاني. والآزوت
والأكسجين لا يمتصّان أيًّا منهما؛ وأما بخار الماء وثنائي أكسيد الكربون والميثان والأوزون
فلها عصابات اهتزاز في تحت الأحمر وهي غازات الدفيئة. والنافذة
بين \qty{8}{} و \qty{13}{\micro m} هي ما تنظر عبره آلة التصوير الحرارية،
وما تشير إليه \qty{250}{K} لسماء الليل، وما
تستغله دهانات التبريد الإشعاعي. وزجاج الدفيئة الحقيقية يفعل
الشيء نفسه (شفاف عند \qty{0.5}{\micro m}، كتيم عند \qty{10}{\micro m})،
لكنه يحجب الحمل في معظمه --- والاسم مجحف قليلًا.
\end{remark}

\begin{method}[تقديرات الإشعاع]\label{meth:b2:thermal-radiation:method}
(1) $\lambda_{\max} = \qty{2900}{\micro m.K}/T$ يقول أين يقع
الطيف؛ (2) و $\varepsilon\sigma T^4$ في وحدة المساحة؛ (3) وأما التبادلات: فالمصدَر
ناقصًا الممتصّ، مخطَّطًا $h_{\text{r}} = 4\varepsilon\sigma T^3$ قرب
حرارة الغرفة؛ (4) وأما الكوكب: $(1 - A)S/4 = \sigma T_{\text{e}}^4$، ثم
الطبقات؛ (5) وقارن بالتوصيل والحمل --- فالإشعاع
يغلب عند الحرارة العالية ($T^4$) وفي الخلاء؛ (6) وانتبه للإصداريات:
فالمعادن اللامعة \qty{0.05}{}، والدهانات والجلد والماء والتربة $\approx 0.9$--$0.98$
في تحت الأحمر مهما كان لونها المرئي.
\end{method}

\section{مسألة: ميزانية طاقة الأرض والمصباح المتوهج}

\begin{problem}[{مسألة نهاية الأسبوع --- الأجسام السوداء من الشمس إلى
الشعيرة}]\label{pb:b2:thermal-radiation:1}
المعطيات: $\sigma = \qty{5.67e-8}{W/m^2/K^4}$، و $\lambda_{\max}T = \qty{2.90e-3}{m.K}$،
و $h = \qty{6.63e-34}{J.s}$، و $k_B = \qty{1.38e-23}{J/K}$، و $c = \qty{3.00e8}{m/s}$؛
والشمس: $R_\odot = \qty{6.96e8}{m}$، و $d = \qty{1.50e11}{m}$؛ والأرض: $R =
\qty{6.37e6}{m}$، وبياضها $0.30$.

\textbf{الجزء الأول --- الشمس جسمًا أسود.}
\begin{enumerate}
  \item يبلغ الطيف الشمسي قمته عند \qty{0.50}{\micro m}: أوجد حرارة
        السطح.
  \item وخروجية السطح واللمعان الكلي.
  \item \omterm{prop:b2:thermal-radiation:earth}{والثابت الشمسي} عند الأرض؛ وكثافة طاقة ضوء الشمس هناك
        وضغطه.
  \item ومتوسط طاقة الفوتون ($\approx 2.7\,k_BT$) وعدد الفوتونات
        الشمسية في الثانية في المتر المربع عند الأرض.
  \item وقد أشعّت الشمس بهذا المعدل $4.5 \times 10^9$ سنة:
        أوجد الطاقة المصدَرة؛ وقارنها بالمقدار $Mc^2$ من أجل $M = \qty{2e30}{kg}$.
\end{enumerate}

\textbf{الجزء الثاني --- الأرض بلا جوّ.}
\begin{enumerate}[resume]
  \item أوجد الاستطاعة الممتصّة من الأرض؛ \omterm{prop:b2:thermal-radiation:earth}{والحرارة الفعّالة}.
  \item ولمَ تقسم الموازنة على $4$؟ وأي حرارة كانت تكون
        عند النقطة تحت الشمس من أرض لا تدور ولا جوّ لها
        ($A = 0.30$)؟
  \item والقمر ($A = 0.12$): أوجد الحرارة تحت الشمس؛ ولمَ تهبط جهته
        الليلية إلى \qty{100}{K} (فكّر في الحرارة المخزونة في
        الثرى في أثناء النهار الذي يدوم أسبوعين، و \qty{1e6}{J/m^2/K} فعّالة).
  \item وطول موجة قمة إصدار الأرض؛ وقارنه بقمة
        الشمس.
  \item ولو رفع مزيد من الجليد أو الغيوم البياضَ إلى $0.35$، فبكم
        كانت تهبط \omterm{prop:b2:thermal-radiation:earth}{الحرارة الفعّالة}؟
\end{enumerate}

\textbf{الجزء الثالث --- الدفيئة.}
\begin{enumerate}[resume]
  \item اكتب الموازنات الثلاث (الفضاء، والطبقة، والسطح) في
        نموذج الطبقة الواحدة واحصل على $T_{\text{s}} = 2^{1/4}T_{\text{e}}$.
  \item ومع طبقة تمتصّ الكسر $\varepsilon$ من
        تحت الأحمر: أوجد $T_{\text{s}}(\varepsilon)$؛ وقيمة $\varepsilon$
        التي تعطي \qty{288}{K}.
  \item وأي الغازات يمتصّ وأيّها لا يمتصّ، وفي أي عصابة
        من أطوال الموجة؛ وما النافذة الجوية؟
  \item وتحت الأحمر الذي يعيده الجوّ إلى الأرض،
        بالوحدة \unit{W/m^2}، في نموذج $\varepsilon = 0.78$؛ وقارنه
        بضوء الشمس الممتصّ.
  \item وقسر مقداره \qty{3.7}{W/m^2} (بتضعيف ثنائي أكسيد الكربون):
        أوجد تغيّر الحرارة $\Delta T = \Delta F/4\sigma T_{\text{e}}^3$
        بلا \omterm{def:b2:feedback-oscillators:loop}{تغذيات راجعة}.
  \item والسعة الحرارية لطبقة محيط ممزوجة \qty{50}{m} في المتر
        المربع وثابت زمن الاستجابة.
  \item ولمَ تبرّد الغيوم (بالبياض) وتدفّئ (بتحت الأحمر)
        السطح؛ وأي الأثرين يغلب ليلًا؟
\end{enumerate}

\textbf{الجزء الرابع --- المصباح المتوهج.}
لمصباح \qty{60}{W} شعيرة تنغستن عند \qty{2800}{K}،
و $\varepsilon = 0.35$؛ والكسر من استطاعة \omterm{def:b2:thermal-radiation:blackbody}{الجسم الأسود} الذي يقع في
المرئي ($0.4$--\qty{0.7}{\micro m}) هو $0.08$ عند \qty{2800}{K} و $0.14$
عند \qty{3200}{K}.
\begin{enumerate}[resume]
  \item أوجد المساحة المشعّة للشعيرة؛ وطول سلك \qty{50}{\micro m}
        له تلك المساحة؛ ولمَ يكون ملفوفًا؟
  \item وطول موجة القمة؛ والاستطاعة المرئية؛ والمردود الضوئي.
  \item ومقاومة الشعيرة في التشغيل عند \qty{230}{V}؛ ومقاومية
        التنغستن أصغر نحو عشر مرات في حرارة الغرفة: أوجد التيار عند التشغيل، ونتيجةً لذلك.
  \item وعند \qty{3200}{K}: أوجد المساحة اللازمة للاستطاعة \qty{60}{W} نفسها
        والاستطاعة المرئية؛ ولمَ لا يُشغَّل كل مصباح عند \qty{3200}{K}
        (إذ يتضاعف معدل تبخر التنغستن تقريبًا كل
        \qty{50}{K})؟
  \item وماذا تفعل دورة الهالوجين، ولمَ تسمح بشعيرة
        أسخن؟
  \item وأين يذهب باقي \qty{60}{W}، ولمَ يبلغ زجاج
        المصباح \qty{150}{\celsius}؟
  \item وزمن تبريد الشعيرة بعد الإطفاء: قدّره من
        كتلتها (\qty{20}{mg}، و $c = \qty{140}{J/kg/K}$) ومن استطاعتها
        المشعّة.
  \item ولخّص: ما الذي تثبّته حرارة \omterm{def:b2:thermal-radiation:blackbody}{جسم أسود}، وما الذي
        يشترك فيه الشعيرة والشمس والأرض.
\end{enumerate}
\end{problem}
