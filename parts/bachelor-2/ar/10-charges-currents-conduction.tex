\section{كثافتا الشحنة والتيار}

\begin{definition}[كثافة الشحنة؛ كثافة التيار]\label{def:b2:charges-currents-conduction:densities}
\index{كثافة الشحنة}\index{كثافة التيار}\index{شدة التيار (تدفق j)}
على \omterm{def:b2:fluid-kinematics:particle}{المقياس المتوسطي}، تحمل المادة \emph{كثافة شحنة حجمية}
$\rho(M, t)$ (\unit{C/m^3}): أي الشحنة $\rho\,\dd\tau$ في $\dd\tau$. وقد يحمل سطح
\emph{كثافة سطحية} $\sigma$ (\unit{C/m^2})، ويحمل سلك كثافةً
خطية $\lambda$ (\unit{C/m}). وإذا تحركت حوامل النوع $i$ (وشحنتها
$q_i$، وكثافتها العددية $n_i$) بالسرعة المتوسطة $\vect v_i$، كانت
\emph{كثافة التيار}
\[
\vect j = \sum_in_iq_i\,\vect v_i \qquad (\unit{A/m^2}) ,
\]
و\emph{شدة التيار} عبر سطح موجَّه $S$ هي التدفق
$I = \iint_S\vect j\cdot\vect n\,\dd S$: أي الشحنة العابرة للسطح $S$ في واحدة
الزمن. ومن أجل نوع واحد، $\vect j = \rho_m\vect v$ مع $\rho_m = nq$ كثافة
الشحنة المتحركة.
\end{definition}

\begin{proof}
في $\dd t$ تكون حوامل النوع $i$ العابرة للعنصر $\dd S$ هي الموجودة في
الأسطوانة المائلة ذات القاعدة $\dd S$ والراسم $\vect v_i\dd t$: وعددها
$n_i\vect v_i\cdot\vect n\,\dd S\,\dd t$، وشحنتها $q_i$ مضروبًا في ذلك.
\end{proof}

\begin{example}[كم تنساق الإلكترونات ببطء]\label{ex:b2:charges-currents-conduction:drift}
\index{سرعة الانسياق}
في النحاس إلكترون ناقلية واحد لكل ذرة: $n = \rho_{\text{Cu}}N_A/M =
8900 \times 6.02 \times 10^{23}/0.0635 = \qty{8.5e28}{m^{-3}}$. وتيار مقداره
\qty{10}{A} في سلك مقطعه \qty{1.5}{mm^2} يعطي $j = \qty{6.7e6}{A/m^2}$، وتكون
\emph{\omterm{ex:b2:charges-currents-conduction:drift}{سرعة الانسياق}} $v = j/ne = 6.7 \times 10^6/(8.5 \times 10^{28} \times 1.6 \times
10^{-19}) = \qty{0.5}{mm/s}$ --- أي ساعتين في المتر. ويضيء المصباح
في الحال لأن \emph{المجال} الذي يدفع الإلكترونات يُقام
على امتداد السلك كله في نانوثوانٍ (وينتقل بما يقارب
سرعة الضوء، كما بيّن كابل \cref{ch:b2:dispersion-wave-packets}):
فتنطلق الإلكترونات كلها معًا، كالماء في خرطوم مملوء.
\end{example}

\section{انحفاظ الشحنة}

\begin{theorem}[الانحفاظ الموضعي للشحنة]\label{thm:b2:charges-currents-conduction:conservation}
\index{انحفاظ الشحنة (موضعي)}\index{معادلة الاستمرار (شحنة)}
عند كل نقطة،
\[
\frac{\partial\rho}{\partial t} + \operatorname{div}\vect j = 0 ;
\]
ومكاملةً على حجم ثابت $V$ يحدّه السطح المغلق $\Sigma$، يكون
$\dd Q_V/\dd t = -I_{\text{خارج}}$: أي إن الشحنة في الداخل لا تتغير إلا بما
يعبر الحدّ. وفي \emph{النظام المستقر} ($\partial_t\rho = 0$)،
$\operatorname{div}\vect j = 0$: فينحفظ تدفق $\vect j$ على امتداد أنبوب
تيار، وتعبر الشدّة نفسها كل مقطع من سلك، وتساوي
التيارات الداخلة إلى عقدة التياراتِ الخارجة منها (وهو قانون العقد لكيرشهوف).
\end{theorem}

\begin{proof}
كلمةً كلمةً موازنة الكتلة في \cref{thm:b2:fluid-kinematics:continuity}
مع $\rho$ \omterm{def:b2:charges-currents-conduction:densities}{كثافة الشحنة} و $\vect j$ تيارها: فالخروج الصافي
للشحنة عبر وجوه صندوق ثابت الستة، في واحدة الحجم، هو
$\operatorname{div}\vect j$، ولا تُخلق شحنة. والصيغة المكاملة تجمع
الصناديق (فتتلاشى الوجوه الداخلية). ومن أجل أنبوب بين مقطعين
$S_1$ و $S_2$ بلا تدفق عبر جانبه، $I_1 = I_2$؛ والعقدة
حجم صغير تدخله عدة أسلاك: $\sum I_{\text{داخل}} = \sum I_{\text{خارج}}$.
\end{proof}

\begin{remark}[حيث يتوقف التيار: المكثف]\label{rem:b2:charges-currents-conduction:capacitor}
السلك الذي يغذّي لبوس مكثف أنبوب تيار ينتهي: إذ يكون
تدفق $\vect j$ إلى اللبوس $I = \dd Q/\dd t$، وتتراكم الشحنة
--- ومنه $\operatorname{div}\vect j \ne 0$ هناك، والنظام غير مستقر.
ومع ذلك يجري التيار $I$ نفسه في السلك على الجانب الآخر: فلا بدّ من شيء
يحمل الموازنة عبر الفجوة. و\cref{ch:b2:maxwell-equations}
يسمّيه --- أي تيار الإزاحة --- ويجعل انحفاظ
الشحنة حجر الزاوية في معادلات ماكسويل.
\end{remark}

\section{نموذج درودة وقانون أوم}

\begin{proposition}[نموذج درودة؛ قانون أوم الموضعي]\label{prop:b2:charges-currents-conduction:drude}
\index{نموذج درودة}\index{قانون أوم (موضعي)}\index{ناقلية}\index{مقاومية}\index{حركية}
نمذج إلكترونات ناقلية معدن جسيماتٍ حرة (كتلتها $m$،
وشحنتها $-e$، وكثافتها $n$) تحسّ المجال $\vect E$، وتفقد وسطيًّا
كمية حركة انسياقها بالتصادمات مع الشبكة كل $\tau$
ثانية --- أي بقوة احتكاك $-m\vect v/\tau$ على السرعة المتوسطة:
\[
m\frac{\dd\vect v}{\dd t} = -e\vect E - \frac m\tau\vect v .
\]
وفي مجال مستقر يستقر الانسياق، في بضعة $\tau$، عند $\vect v = -e\tau
\vect E/m$، وتكون \omterm{def:b2:charges-currents-conduction:densities}{كثافة التيار}
\[
\vect j = \gamma\,\vect E , \qquad \gamma = \frac{ne^2\tau}{m} :
\]
وهو \emph{قانون أوم الموضعي}، و $\gamma$ هي \emph{الناقلية}
(\unit{S/m}؛ ومقلوبها $\rho_e = 1/\gamma$ هو \emph{المقاومية}،
\unit{\ohm.m}). والنسبة $\mu = e\tau/m$ بين \omterm{ex:b2:charges-currents-conduction:drift}{سرعة الانسياق} والمجال هي
\emph{الحركية} (\unit{m^2/(V.s)}). ومن أجل ناقل طوله $L$
ومقطعه المنتظم $S$ يحمل $\vect j$ منتظمة، تعطي مكاملة $E = j/\gamma$
على امتداد الطول $U = EL = (L/\gamma S)I$: أي المقاومة $R = L/\gamma S
= \rho_eL/S$ الواردة في كتاب السنة الأولى.
\end{proposition}

\begin{proof}
مع $\vect v = -e\tau\vect E/m$ يكون $\vect j = -ne\vect v = (ne^2\tau/m)\vect E$. ويدوم
الطور العابر $\vect v(t) = \vect v_\infty(1 - \eu^{-t/\tau})$ مدة $\tau \sim
\qty{1e-14}{s}$، وهي لحظية على كل مقياس زمني في الدارات. وتنتج المقاومة
من $\int\vect E\cdot\dd\vect l$ بين الطرفين.
\end{proof}

\begin{example}[النحاس، انطلاقًا من الناقلية المقيسة]\label{ex:b2:charges-currents-conduction:copper}
النحاس: $\gamma = \qty{6.0e7}{S/m}$، و $n = \qty{8.5e28}{m^{-3}}$: ومنه $\tau = m\gamma/ne^2 =
9.1 \times 10^{-31} \times 6 \times 10^7/(8.5 \times 10^{28} \times 2.56 \times 10^{-38}) = \qty{2.5e-14}{s}$
--- أي خمس وعشرون فمتوثانية بين تصادمين. وسرعة الإلكترونات
العشوائية في معدن نحو \qty{1.6e6}{m/s} (وهو أثر كمومي، لا
السرعة الحرارية $\sqrt{3k_BT/m} \approx \qty{1e5}{m/s}$)، ومنه يكون المسار الحرّ الوسطي $\ell
= v\tau \approx \qty{40}{nm}$، أي مئة مسافة ذرية: فالإلكترونات لا
تصطدم بالذرات نفسها (فالبلورة التامة شفافة
لها) بل بعيوبها وباهتزازاتها الحرارية --- ولهذا
تهبط $\gamma$ حين يُسخَّن المعدن أو يُخلَط. والحركية $e\tau/m =
\qty{4.4e-3}{m^2/(V.s)}$: أي \qty{4}{mm/s} لكل فولت في المتر.
\end{example}

\begin{proposition}[قانون أوم عند التواترات العالية]\label{prop:b2:charges-currents-conduction:acdrude}
\index{ناقلية عقدية}
من أجل مجال جيبي $\underline{\vect E}\,\eu^{\iu\omega t}$ تعطي معادلة درودة
$\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$ مع
\[
\underline\gamma(\omega) = \frac{\gamma_0}{1 + \iu\omega\tau} , \qquad \gamma_0 = \frac{ne^2\tau}{m} :
\]
فتكون الناقلية حقيقية ومساوية لقيمتها المستمرة ما دام $\omega\tau
\ll 1$، أي حتى نحو \qty{1e13}{Hz} في النحاس --- عبر مجال
الراديو والموجات الصغرية كله؛ وفي تحت الأحمر وفي المرئي ($\omega\tau \gg 1$)
تصير تخيلية، $\underline\gamma \approx ne^2/\iu m\omega$: فتتذبذب الإلكترونات
بحرية، خارج طور المجال، ويسلك المعدن
سلوك بلازما \cref{ch:b2:waves-in-media}.
\end{proposition}

\begin{proof}
$\iu\omega m\underline{\vect v} = -e\underline{\vect E} - m\underline{\vect v}/\tau$، ومنه
$\underline{\vect v} = -e\tau\underline{\vect E}/m(1 + \iu\omega\tau)$ و $\underline{\vect j}
= -ne\underline{\vect v}$.
\end{proof}

\begin{remark}[النواقل وأنصاف النواقل والمحاليل الشاردية]\label{rem:b2:charges-currents-conduction:materials}

وماء الصنبور $10^{-2}$، والماء النقي $10^{-5}$. والعوازل (الزجاج والبوليمرات):

$q$: ففي الألومنيوم أو الزنك تكون الحوامل موجبة ---
أي ثقوبًا --- ولا يفسّر ذلك أي نموذج كلاسيكي.
\end{remark}

\section{الطاقة: قانون جول الموضعي}

\begin{proposition}[قانون جول الموضعي]\label{prop:b2:charges-currents-conduction:joule}
\index{قانون جول (موضعي)}
يقدّم المجال الكهربائي إلى الحوامل، في واحدة الحجم والزمن،
الاستطاعة $\vect j\cdot\vect E$؛ وفي ناقل أومي تكون
\[
p = \vect j\cdot\vect E = \gamma E^2 = \frac{j^2}\gamma \ \ (\unit{W/m^3}) ,
\]
وتتحول كلها حرارةً بالتصادمات (فالطاقة الحركية للانسياق
مهمَلة وثابتة). ومكاملةً على سلك، $\int j^2/\gamma\,\dd\tau
= (L/\gamma S)I^2 = RI^2$.
\end{proposition}

\begin{proof}
القوة $q\vect E$ على كل حامل تبذل الشغل بالمعدل $q\vect E\cdot\vect v$؛
ومجموعةً على $n\,\dd\tau$ حاملًا: $nq\vect v\cdot\vect E\,\dd\tau = \vect j\cdot
\vect E\,\dd\tau$ (والقوة المغناطيسية لا تبذل شغلًا). وفي حالة درودة
المستقرة تساوي هذه الاستطاعةُ ما يبدّده حدّ الاحتكاك، أي
ما يُنقل إلى الشبكة.
\end{proof}

\begin{example}[سلك يسخن]\label{ex:b2:charges-currents-conduction:heating}
السلك ذو \qty{1.5}{mm^2} عند \qty{10}{A}: $p = j^2/\gamma = (6.7 \times 10^6)^2/6 \times 10^7 =
\qty{7.4e5}{W/m^3}$، أي \qty{1.1}{W} في المتر --- ويصرّفها الهواء
حوله بسهولة. ولو تُرك لنفسه (بلا تبريد) لسخن النحاس ($c = \qty{385}{J/(kg.K)}$،
و $\rho = \qty{8900}{kg/m^3}$) بالمعدل $p/\rho c = \qty{0.2}{K/s}$. وعند \qty{100}{A}
في السلك نفسه يكون المعدل أكبر مئة مرة، أي \qty{20}{K/s}: فينصهر
في دقيقة --- وهو مبدأ المصهر، الذي صُمّم سلكه الرفيع
لينصهر أولًا.
\end{example}

\begin{proposition}[استرخاء الشحنة في ناقل]\label{prop:b2:charges-currents-conduction:relaxation}
\index{زمن الاسترخاء (ناقل)}
في ناقل أومي تخمد كثافة شحنة حجمية وفق $\rho(t) =
\rho_0\eu^{-t/\tau_r}$ حيث
\[
\tau_r = \frac{\varepsilon_0}{\gamma} :
\]
أي \qty{1.5e-19}{s} للنحاس، و \qty{2e-12}{s} لماء البحر، و \qty{10}{s} للزجاج.
وحتى التواترات من رتبة $1/\tau_r$ لا يحمل الناقل \emph{أي
شحنة حجمية}: بل تجري كل شحنة فائضة إلى سطحه. وهكذا تتوقف كلمة
«ناقل» على المقياس الزمني: فماء البحر ناقل
للأمواج الراديوية وعازل للضوء المرئي.
\end{proposition}

\begin{proof}
انحفاظ الشحنة مع $\vect j = \gamma\vect E$ وقانون غاوس في صيغته
الموضعية، $\operatorname{div}\vect E = \rho/\varepsilon_0$ (\cref{ch:b2:maxwell-equations}):
$\partial_t\rho = -\gamma\operatorname{div}\vect E = -(\gamma/\varepsilon_0)\rho$.
\end{proof}

\begin{method}[من القانون الموضعي إلى المكامل]\label{met:b2:charges-currents-conduction:method}
لإيجاد مقاومة ناقل كيفي الشكل: (1) استعمل التناظر لكتابة
$\vect j$ (وفي النظام المستقر ينحفظ تدفقها، ومنه في
أنبوب $j \times$ المقطع $= I$)؛ (2) $\vect E = \vect j/\gamma$؛ (3) كامِل
$\vect E$ على امتداد خط من مسرى إلى الآخر: $U = \int\vect E\cdot
\dd\vect l$، و $R = U/I$؛ (4) تحقق باستطاعة جول $\int j^2/\gamma\,
\dd\tau = RI^2$. والخطوات نفسها تعطي تسرّب كابل، ومقاومة
الأرض حول مانعة صواعق، أو مقاومة تماس مخروطي.
\end{method}

\section{مسألة: السلك النحاسي، من الذرة إلى الشبكة}

\begin{problem}[{مسألة نهاية الأسبوع --- متر واحد من سلك منزلي مفحوصًا
على أربعة مقاييس: الإلكترون، والقوانين الموضعية، والدارة،
وخط النقل}]
\label{pb:b2:charges-currents-conduction:1}

النحاس: كتلته المولية \qty{63.5}{g/mol}، وكثافته \qty{8900}{kg/m^3}، وفيه
إلكترون ناقلية واحد لكل ذرة، وناقليته $\gamma = \qty{6.0e7}{S/m}$ عند
\qty{20}{\degreeCelsius}، ومقاوميته المرتفعة $\rho_e(T) = \rho_e(20)[1 +
\alpha(T - 20)]$ مع $\alpha = \qty{4.0e-3}{K^{-1}}$؛ و $c = \qty{385}{J/(kg.K)}$. والسلك:
مقطعه $S = \qty{2.5}{mm^2}$، وطوله \qty{1}{m}، وتياره \qty{16}{A}.

\medskip
\noindent\textbf{الجزء I --- الإلكترونات.}
\begin{enumerate}
  \item الكثافة العددية $n$ لإلكترونات الناقلية.
  \item \omterm{def:b2:charges-currents-conduction:densities}{كثافة التيار} \omterm{ex:b2:charges-currents-conduction:drift}{وسرعة الانسياق}؛ وكم يحتاج إلكترون من
        الزمن لقطع المتر.
  \item $\tau$ درودة والحركية؛ والمسار الحرّ الوسطي بسرعة عشوائية
        \qty{1.6e6}{m/s}؛ وكم مسافة ذرية (\qty{0.26}{nm}) يعادل ذلك،
        وماذا يقول عن الشيء الذي تصطدم به
        الإلكترونات؟
  \item وتنمو الاهتزازات الحرارية مع $T$، ومنه $\tau \propto 1/T$ تقريبًا:
        تحقق من ذلك بالمقارنة مع $\alpha$ بجوار حرارة الغرفة ($\dd\rho_e/\rho_e =
        \dd T/T$ كانت لتعطي $\alpha = 1/293$).
  \item المجال الكهربائي في السلك؛ والتوتر الكلي على المتر؛
        والطاقة الحركية للانسياق لكل إلكترون مقارنةً بالمقدار $k_BT$ ---
        فهل يزعج التيارُ غازَ الإلكترونات؟
  \item عدد الإلكترونات التي تعبر مقطعًا من السلك في الثانية.
  \item شحنة إلكترونات الناقلية في المتر من السلك؛ والقوة
        التي يؤثر بها المجال فيها كلها؛ وقارنها بثقل
        السلك. وأين تنتهي تلك القوة (فكّر في
        التصادمات)؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- القوانين الموضعية.}
\begin{enumerate}[resume]
  \item استطاعة جول في واحدة الحجم وفي المتر.
  \item ويفقد السلك، في الهواء، حرارةً بالمعدل $h(T - T_a)$ في واحدة
        المساحة مع $h = \qty{10}{W/(m^2.K)}$ (بالحمل): أوجد ارتفاع
        الحرارة عند التوازن؛ وتحقق مما إذا كان تصحيح المقاومية
        مهمًّا.
  \item والسلك نفسه مدفونًا في عزل لا يمرّر إلا $h =
        \qty{2}{W/(m^2.K)}$: أوجد ارتفاع الحرارة الجديد، مع إدراج
        تصحيح المقاومية (حُلّ الموازنة).
  \item ومتروكًا بلا أي تبريد، أوجد معدل ارتفاع الحرارة عند \qty{16}{A}
        وعند \qty{160}{A} (أي قصر دارة)؛ والزمن اللازم لبلوغ درجة
        الانصهار (\qty{1085}{\degreeCelsius}) في الحالة الثانية؛ وما الذي يحمي
        التمديدات؟
  \item وعند \qty{50}{Hz}: $\omega\tau = ?$ وهل يتأثر قانون أوم؟ (والأثر
        القشري في \cref{ch:b2:waves-in-media} أمر آخر.)
  \item اكتب زمن استرخاء النحاس $\varepsilon_0/\gamma$؛ وماذا يعني
        \omterm{def:b2:charges-currents-conduction:densities}{لكثافة الشحنة} داخل السلك، وأين تقع
        الشحنة التي تنتج المجال $E$ داخل السلك؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- انحفاظ الشحنة في الدارة.}
يغذّي السلك مكثفًا سعته $C = \qty{100}{\micro F}$ عبر مفتاح؛ ويكون
التيار $I(t) = I_0\eu^{-t/\tau_c}$ مع $I_0 = \qty{16}{A}$ و $\tau_c = \qty{1}{ms}$.
\begin{enumerate}[resume]
  \item الشحنة التي تبلغ اللبوس؛ والتوتر النهائي.
  \item طبّق $\dd Q/\dd t = -I_{\text{خارج}}$ على سطح مغلق حول
        اللبوس: فأي جانب يحمل تيار ناقلية، وأيهما لا يحمله؟
        وأي مقدار يجب أن يتغير إذًا في الفجوة؟
  \item وعند $t = \qty{0.5}{ms}$، أوجد معدل تغيّر شحنة اللبوس؛ وإذا كان
        اللبوسان \qty{1}{cm^2} ويبعدان \qty{0.1}{mm} في الهواء، أوجد معدل
        تغيّر المجال بينهما.
  \item بيّن أن $\varepsilon_0\,\dd E/\dd t \times$ مساحة اللبوس يساوي التيار
        في السلك --- وهو المقدار الذي يضيفه \cref{ch:b2:maxwell-equations}
        إلى قانون أمبير.
  \item ويُقطع السلك ويُترك طرفاه متباعدين \qty{1}{mm} في الهواء:
        فأي تيار يجري (وناقلية الهواء $\gamma \approx \qty{1e-14}{S/m}$)؟ وعند أي
        مجال ينهار الهواء (\qty{3}{MV/m}): فأي توتر على
        الفجوة يصنع شرارة؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- الخط.}
خط عند \qty{230}{V} من النحاس مقطعه \qty{10}{mm^2}، وطوله \qty{500}{m} (ذهابًا وإيابًا:
أي \qty{1}{km} من السلك)، يغذّي مزرعة تسحب \qty{40}{A}.
\begin{enumerate}[resume]
  \item المقاومة وهبوط التوتر والاستطاعة الضائعة في الخط؛ والنسبة
        من الاستطاعة المسلَّمة.
  \item والاستطاعة نفسها عند \qty{20}{kV} بمحوّل عند كل طرف: أوجد
        التيار والهبوط والضياع؛ واستنتج لماذا يجري النقل عند
        توتر عال.
  \item ويُستعمل الألومنيوم ($\gamma = \qty{3.6e7}{S/m}$، و \qty{2700}{kg/m^3}) على
        الأبراج: أوجد المقطع اللازم للمقاومة نفسها التي لمقطع \qty{10}{mm^2} من
        النحاس؛ والكتلة في الكيلومتر لكليهما؛ ولماذا يفوز الألومنيوم
        في الأعالي والنحاس في الجدار.
  \item ومِجَسّ هول مشبوك حول الخط يقرأ مجاله: أوجد توتر هول
        في مقياس مشبكي فيه $n = \qty{1e22}{m^{-3}}$ و $t = \qty{0.1}{mm}$، مغذًّى بتيار
        \qty{5}{mA} في مجال \qty{0.1}{T} مركَّز بلبّ حديدي.
  \item وبعد حمل يوم حارّ يستقر السلك عند \qty{80}{\degreeCelsius}:
        أوجد مقاومته عندئذ والضياع الزائد.
  \item وتصيب صاعقة الخط: \qty{20}{kA} مدة \qty{100}{\micro s}. أوجد الطاقة
        المبدَّدة في \qty{1}{km} من السلك؛ وارتفاع الحرارة إن لم يخرج
        شيء؛ \omterm{def:b2:charges-currents-conduction:densities}{وكثافة التيار} --- وقارنها بالجزء الأول.
  \item لخّص: المقادير الخمسة ($n$ و $\tau$ و $\gamma$ و $j$ و $E$) التي
        تصف السلك على كل مقياس، والقانون الذي يربط كل زوج.
\end{enumerate}
\end{problem}
