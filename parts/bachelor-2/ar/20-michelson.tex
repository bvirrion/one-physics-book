\chapter{مقياس تداخل ميكلسون}\label{ch:b2:michelson}

في عام 1887 شطر ميكلسون ومورلي حزمة ضوء إلى نصفين، وأرسلا
النصفين على ذراعين متعامدين، ثم جمعاهما، وراقبا
الهدب بحثًا عن الإزاحة التي كان يجب أن تُحدثها حركة الأرض في الأثير:
فلم يتحرك شيء، وتغيرت الفيزياء. وفي عام 2015 سجّلت الأداة
نفسها --- ذراعان، صارا الآن بطول أربعة كيلومترات ومطويّان إلى
ألف --- موجة ثقالية غيّرت طوليهما
بجزء من ألف من قطر البروتون. وبين هذين التاريخين قاس \omterm{def:b2:michelson:instrument}{مقياس
تداخل ميكلسون} المتر بأطوال الموجة، وفصل
الخطوط الطيفية، وصار الأداة القياسية في البصريات الدقيقة.
وهذا الفصل يصفه في تشكيلتيه --- الإسفين الهوائي
والصفيحة المتوازية --- ويبيّن كيف تقيس هدبه الأطوال
والقرائن والأطياف.

\section{الأداة}

\begin{definition}[مقياس تداخل ميكلسون]\label{def:b2:michelson:instrument}
\index{مقياس تداخل ميكلسون}\index{شاطر الحزمة}\index{الصفيحة المعوِّضة}
\emph{شاطر حزمة} $Sp$ --- وهو صفيحة زجاجية ذات وجه نصف عاكس
--- على \qty{45}{\degree} يقسم حزمة واردة إلى حزمتين: تذهب إحداهما إلى
المرآة $M_1$، والأخرى إلى $M_2$، وكلتاهما عمودية على حزمتها؛ وبعد
الانعكاس تعود كل منهما إلى $Sp$، حيث يُرسَل نصفها نحو
الراصد، فتتداخل الحزمتان هناك. وتوضع \emph{صفيحة معوِّضة}
$C$، مطابقة للشاطر وموازية له، في الذراع
الذي كان يعبر زجاج الشاطر مرة واحدة، حتى تعبر
الحزمتان سماكة الزجاج نفسها: فيكون \omterm{thm:b2:two-wave-interference:formula}{فرق المسير} عندئذ
فرق مسيرات هوائية محضًا، واحدًا من أجل كل طول موجة --- وهو ما
تقتضيه هدب الضوء الأبيض. وبعدا المرآتين عن $Sp$ هما $d_1$
و $d_2$؛ وإحدى المرآتين مركّبة على لولب ميكرومتري وعلى
لوالب ميل.
\end{definition}

\begin{proposition}[الترتيب المكافئ]\label{prop:b2:michelson:equivalent}
\index{إسفين هوائي (ميكلسون)}\index{صفيحة هوائية (ميكلسون)}
يكافئ مقياس التداخل، من موضع الراصد، المرآةَ
$M_2$ والصورةَ $M_1'$ للمرآة $M_1$ في الشاطر: أي طبقة هواء رقيقة
بين سطحين عاكسين، سماكتها $e = |d_1 - d_2|$، ويتداخل
انعكاساها (\omterm{prop:b2:two-wave-interference:film}{تقسيم السعة}):
\begin{itemize}
  \item $M_1' \parallel M_2$: أي \emph{صفيحة هوائية} سماكتها $e$ --- \omterm{prop:b2:michelson:rings}{فحلقات
        تساوي الميل}، موضَّعة في اللانهاية، $\delta = 2e\cos i$؛
  \item $M_1'$ مائلة قليلًا بالزاوية $\alpha$ عن $M_2$: أي \emph{إسفين هوائي}
        --- فهدب مستقيمة متساوية السماكة، موضَّعة على
        المرآتين (قرب الإسفين)، $\delta = 2\alpha x$ عند الورود الناظمي،
        وتباعدها $\lambda/2\alpha$.
\end{itemize}
وعند \emph{التماس} ($e = 0$، $\alpha = 0$) يكون الحقل منتظمًا، وفي الضوء
الأبيض يكون الموضع الوحيد الذي تتفق فيه الألوان: أي "الهدبة البيضاء".
\end{proposition}

\begin{proof}
طيّ ذراع $M_1$ على ذراع $M_2$ بالانعكاس في $Sp$ يحمل
الحزمتين على مستقيم واحد بمرآتين تتباعدان $e$؛ والموجتان
العائدتان هما المنعكستان على وجهي طبقة الهواء (بلا
$\lambda/2$ زائدة: إذ تنعكس كلتاهما على معدن، ويعاملهما الشاطر
معاملة متناظرة، إلا من طور ثابت يزيح النقش كله).
ثم تنطبق نتائج الأغشية الرقيقة في \cref{ch:b2:two-wave-interference}
مع $n = 1$.
\end{proof}

\section{نظاما الهدب}

\begin{proposition}[حلقات الصفيحة الهوائية]\label{prop:b2:michelson:rings}
\index{حلقات تساوي الميل}\index{هدب هايدنغر}
مع $M_1' \parallel M_2$ \omterm{prop:b2:two-wave-interference:spatial}{ومنبع ممتد}، يرى الراصد (أو عدسة
بعدها البؤري $f$ وشاشة في مستويها البؤري) حلقات
متحدة المركز؛ وللحلقة عند الزاوية $i$ الرتبة $p = 2e\cos i/\lambda$؛ والحلقات
مضيئة حيث يكون $p$ عددًا صحيحًا. وللمركز أعلى رتبة، $p_0 =
2e/\lambda$؛ وبجواره، $\cos i \approx 1 - i^2/2$، فيكون للحلقة المضيئة $k$ من
المركز (معدودةً من الأولى، عند $\varepsilon = p_0 - \lfloor p_0\rfloor$
دون رتبة المركز) نصف القطر
\[
r_k = f\,i_k = f\sqrt{\frac{(k - 1 + \varepsilon)\lambda}e} :
\]
فالحلقات أضيق كلما كبر $e$ (أي حلقات أكثر وأرقّ)، وتتباعد كلما $e \to 0$
حتى يصير الحقل، عند التماس، منتظمًا واحدًا. وتحريك مرآة بمقدار $\lambda/2$ يغيّر
$p_0$ بواحد: فتُولَد حلقة في المركز (أو يبتلعها) في كل
نصف طول موجة من الإزاحة.
\end{proposition}

\begin{proof}
$\delta = 2e\cos i$ من أجل الصفيحة المتوازية (\cref{ch:b2:two-wave-interference}،
$n = 1$، بلا $\lambda/2$ زائدة من أجل انعكاسين معدنيين)؛ و $2e(1 - i^2/2) =
(p_0 - k + 1 - \varepsilon)\lambda$ مع $2e = p_0\lambda$ يعطي $ei^2 = (k - 1 + \varepsilon)\lambda$.
\end{proof}

\begin{proposition}[هدب الإسفين الهوائي]\label{prop:b2:michelson:wedge}
\index{هدب تساوي السماكة (ميكلسون)}\index{هدب فيزو}
مع $M_1'$ مائلةً بزاوية صغيرة $\alpha$ وورود قريب من الناظمي، يكون
\omterm{thm:b2:two-wave-interference:formula}{فرق المسير} عند نقطة المرآتين الواقعة على البعد $x$ من
مستقيم التماس هو $\delta = 2\alpha x$: أي هدب مستقيمة موازية لذلك المستقيم،
متباعدة $\lambda/2\alpha$، موضَّعة على المرآتين (وتُرى بالتجميع
على $M_2$، أو تُسقَط على شاشة بعدسة ترافق $M_2$ مع
الشاشة). وتقع هدبة الرتبة الصفر على مستقيم التماس، وهي نفسها من أجل
كل طول موجة: فتكون بيضاء في الضوء الأبيض (أو سوداء، بحسب
أطوار الشاطر)، تكتنفها بضع هدب ملونة --- وهي
العلامة التي يُعرَف بها التماس.
\end{proposition}

\begin{proof}
\omterm{prop:b2:two-wave-interference:film}{هدب تساوي السماكة} لإسفين من الهواء، $e(x) = \alpha x$. وأما التوضيع:
فمن أجل \omterm{prop:b2:two-wave-interference:spatial}{منبع ممتد} تتجمع الأشعة المارّة بنقطة معينة من الإسفين
هناك، بالمقدار $\delta$ نفسه عند الورود القريب من الناظمي.
\end{proof}

\section{القياس بالميكلسون}

\begin{proposition}[الأطوال والقرائن والأطياف]\label{prop:b2:michelson:measure}
\index{مطيافية تحويل فورييه}\index{طيف مخدَّد}
عدّ $N$ هدبة تمرّ بالمركز بينما تتحرك مرآة بمقدار $\Delta e$
يعطي $\Delta e = N\lambda/2$: أي الأطوال بأطوال الموجة (بلولب ميكرومتري
معاير حتى \qty{0.1}{\micro m}؛ وقد قيس المتر نفسه هكذا). وصفيحة
شفافة سماكتها $\ell$ وقرينتها $n$ تُدخَل في أحد الذراعين
تضيف $2(n - 1)\ell$ إلى $\delta$ وتزيح النقش بمقدار $2(n - 1)\ell/\lambda$
هدبة: فتُقاس القرائن (وتغيراتها مع الضغط أو الحرارة،
في خلية غاز). ومع ثنائي طولا موجته $\lambda$ و $\lambda + \Delta\lambda$، يخرج
نظاما الهدب عن التوافق ثم يعودان إليه كلما كبر $e$، فينعدم
التباين كل
\[
\Delta e = \frac{\lambda^2}{2\Delta\lambda} :
\]
أي فصل الخطوط المتقاربة. وعلى العموم يرسم التباين بدلالة
$e$ طيفَ المنبع (إذ يخبو التباين على $e \sim \ell_c/2$): وتسجيل الشدة عند المركز بينما
يمسح $e$ هو \emph{\omterm{prop:b2:michelson:measure}{مطيافية تحويل فورييه}}، وهي طريقة
المطيافات تحت الحمراء.
\end{proposition}

\begin{proof}
الرتبة عند المركز هي $2e/\lambda$؛ ولكل خط رتبته. ويتطابق
النظامان من جديد حين يكون $2e/\lambda - 2e/(\lambda + \Delta\lambda) = 1$، أي $2e\Delta\lambda
/\lambda^2 = 1$؛ ويكونان على تعاكس في منتصف الطريق. وأما القول العام فهو
صيغة التباين في \cref{exo:b2:scalar-light-model:10}.
\end{proof}

\begin{example}[الصوديوم، والمتر]\label{ex:b2:michelson:sodium}
ثنائي الصوديوم ($\Delta\lambda = \qty{0.6}{nm}$ عند \qty{589}{nm}): تتشوّش الحلقات وتصفو
كل $\Delta e = 589^2/(2 \times 0.6)$ نانومتر $= \qty{0.29}{mm}$ --- وهو قياس كلاسيكي
للمقدار $\Delta\lambda$ بدقة بضعة في المئة بلولب. وقد عدّ ميكلسون (1892)
هدب خط الكادميوم الأحمر على طول المتر
المعياري: $1\,553\,163.5$ طول موجة، وهو أول ربط بين
المتر والذرة؛ وأما اليوم فيُعرَّف المتر بالسرعة $c$ وبالثانية،
وتتحقق مقاييس التداخل من قوالب المعايرة في الإنتاج بدقة
النانومترات.
\end{example}

\begin{remark}[الذراعان اللذان لم يقيسا شيئًا، واللذان قاسا موجة]\label{rem:b2:michelson:history}
\index{تجربة ميكلسون--مورلي}\index{كاشف الأمواج الثقالية}
توقّع ميكلسون ومورلي أن تغيّر سرعة الأرض $v$ في
الأثير زمن الذهاب والإياب على الذراع الموازي لها بمقدار
$(L/c)(v/c)^2$ بالقياس إلى الذراع العمودي، أي إزاحة هدب مقدارها
$2L(v/c)^2/\lambda \approx 0.4$ هدبة من أجل $L = \qty{11}{m}$ و $v = \qty{30}{km/s}$؛
وكان يجب أن يؤرجح تدوير الجهاز النقش بذلك المقدار.
وكانا يريان $0.01$ هدبة؛ فلم يريا شيئًا. وأما كواشف الأمواج
الثقالية اليوم فميكلسونات ذراعاها \qty{4}{km}، والضوء يرتدّ
فيهما نحو ثلاثمئة مرة، وليزرها مستقر كفاية لقراءة
تغيّر في طول الذراع $\Delta L/L \sim 10^{-21}$ --- أي $10^{-18}$ متر، وهو جزء من ألف
من البروتون --- في وجه ضجيج الحركة الزلزالية والفوتونات
نفسها.
\end{remark}

\begin{method}[استعمال الميكلسون]\label{met:b2:michelson:method}
(1) عيّن التشكيلة: صفيحة متوازية (فحلقات في اللانهاية؛
تُرصَد بعدسة أو بالعين على اللانهاية) أو إسفين (فهدب على
المرآتين؛ فجمّع على $M_2$). (2) واكتب $\delta = 2e\cos i$ أو $2\alpha x$ ثم
الرتبة. (3) وترجم كل قياس إلى هدب: $\Delta e = N\lambda/2$،
و $2(n - 1)\ell$ من أجل صفيحة، و $\lambda^2/2\Delta\lambda$ من أجل ثنائي. (4) واستعمل
موضع التماس ($e = 0$، والهدبة البيضاء) مرجعًا مطلقًا؛
ولاحظ أن أنصاف أقطار الحلقات \emph{تصغر} كلما كبر $e$. (5) وأما حدود
الترابط: فالحلقات مرئية ما دام $2e < \ell_c$؛ والإسفين يحتاج إلى
ورود قريب من الناظمي.
\end{method}

\section{تمارين}

\begin{exercise}[$\star$]\label{exo:b2:michelson:1}
ميكلسون في ضوء الصوديوم ($\lambda = \qty{589}{nm}$): تُحرَّك المرآة بمقدار
\qty{0.10}{mm}: أوجد عدد الحلقات المارّة بالمركز؛ والإزاحة
الموافقة لعدد $250$ هدبة؛ والدقة على $e$ إذا أمكن قراءة عُشر
هدبة.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:2}
صفيحة متوازية سماكتها $e = \qty{0.50}{mm}$، وعدسة $f = \qty{1.0}{m}$، و $\lambda =
\qty{589}{nm}$: أوجد الرتبة عند المركز؛ أيكون المركز مضيئًا؟ وأنصاف أقطار
أول ثلاث حلقات مضيئة؛ وماذا يحدث للحلقات إذا نقص $e$
إلى \qty{0.05}{mm}؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:3}
إسفين هوائي مع $\alpha = \qty{1.0e-4}{rad}$، و $\lambda = \qty{633}{nm}$: أوجد \omterm{def:b2:maxwell-equations:operators}{تباعد} الهدب؛
وعدد الهدب عبر \qty{2}{cm} من المرآة؛ وتتضاعف زاوية الإسفين:
فماذا يحدث؟ وكيف يُوجَد التماس بالضوء الأبيض؟
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:4}
تُدخَل صفيحة زجاجية سماكتها \qty{0.10}{mm} في أحد الذراعين: أوجد إزاحة الهدب من أجل
$n = 1.52$ عند \qty{589}{nm}؛ وتُقاس الإزاحة بدقة $0.2$ هدبة: أوجد الدقة
على $n$. ولمَ لا يمكن إجراء القياس نفسه بالضوء الأبيض
لو كانت الصفيحة بسماكة \qty{1}{cm}؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:5}
\emph{ثنائي الصوديوم.} (أ) أوجد السماكة $e$ بين اختفاءين
متتاليين للحلقات، من أجل $\Delta\lambda = \qty{0.597}{nm}$. (ب) ويقيس طالب
$17$ اختفاءً على \qty{5.00}{mm}: أوجد $\Delta\lambda$ ودقتها.
(ج) وأوجد رتبتي الخطين عند $e = \qty{0.29}{mm}$: وتحقق من أنهما تختلفان بمقدار
$\tfrac12$. (د) ولمَ لا يكون التباين معدومًا تمامًا أبدًا (إذ شدتا الخطين
بالنسبة $2 : 1$): أوجد أصغر تباين.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:6}
\emph{قرينة الهواء.} تُفرَّغ خلية طولها \qty{10.0}{cm} في أحد الذراعين، ثم
تُملأ ببطء بالهواء عند \qty{1}{bar}؛ فتمرّ $94$ هدبة ($\lambda = \qty{589}{nm}$).
(أ) أوجد $n - 1$ للهواء. (ب) وعدد الهدب المارّة من أجل تغير \qty{1}{\%} في
الضغط؛ والمعامل الحراري إذا كان $n - 1 \propto \rho$. (ج) ولمَ يهمّ ذلك
في استقرار أي مقياس تداخل في الهواء؟ (د) وأوجد الدقة على
$n - 1$ من عُشر هدبة.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:7}
\emph{\omterm{prop:b2:scalar-light-model:coherence}{طول الترابط}.} مع مصباح زئبق منخفض الضغط (الخط الأخضر،
\qty{546}{nm}، وعرضه \qty{0.01}{nm}) تخبو الحلقات كلما كبر $e$. (أ) أوجد \omterm{prop:b2:scalar-light-model:coherence}{طول
الترابط}؛ والسماكة $e$ التي يصير التباين عندها مهمَلًا. (ب) وأعد ذلك
مع مصباح عالي الضغط ($\Delta\lambda = \qty{1}{nm}$). (ج) ومع ليزر هليوم--نيون
($\Delta\nu = \qty{1}{MHz}$): أيمكن رؤية الحلقات مع $e = \qty{10}{m}$؟ (د) وما الذي
يحدّ هدب الضوء الأبيض بنحو $\pm2$ رتبة حول التماس؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:8}
\emph{\omterm{def:b2:michelson:instrument}{الصفيحة المعوِّضة}.} من دون $C$، تعبر إحدى الحزمتين زجاج
الشاطر ($e_g = \qty{5}{mm}$، $n = 1.52$) ثلاث مرات وتعبره الأخرى مرة. (أ)
أوجد \omterm{def:b2:scalar-light-model:path}{المسير الضوئي} الزائد؛ أيمكن تعويضه بتحريك مرآة من أجل
ضوء أحادي اللون؟ (ب) ولمَ لا يمكن ذلك في الضوء الأبيض (بسبب التبدد: إذ يتغير $n$
بمقدار $0.01$ عبر المرئي)؟ (ج) وأي تسامح على
توازي $C$ و $Sp$ يحفظ الهدبة البيضاء (خذ فرق سماكة
باقيًا $< \qty{0.3}{\micro m}$)؟ (د) والشواطر الحديثة
أغشية رقيقة أو مكعبات: فلمَ لا تلزم $C$ عندئذ؟
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:9}
\emph{ميكلسون--مورلي.} ذراعان $L = \qty{11}{m}$ (مطويّان)، و $\lambda = \qty{550}{nm}$،
وسرعة الأرض المدارية $v = \qty{30}{km/s}$. (أ) في نظرية الأثير يستغرق
الذهاب والإياب على الذراع الموازي للشعاع $\vect v$ زمنًا $2L/c(1 - v^2/c^2)$
ويستغرق العمودي $2L/c\sqrt{1 - v^2/c^2}$: أوجد الفرق من أدنى
مرتبة. (ب) وإزاحة الهدب المتوقعة عند تدوير الجهاز \qty{90}{\degree}
(إذ يتبادل الذراعان الدورين). (ج) وكانا يكشفان $0.01$ هدبة: أوجد الحدّ
الأعلى على $v$. (د) وماذا تقول نتيجة صفرية، وماذا استنتج أينشتاين؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:10}
\emph{مطيافية فورييه.} الشدة عند مركز الحلقات
هي $I(e) = \int S(\nu)[1 + \cos(4\pi\nu e/c)]\dd\nu$ من أجل منبع طيفه
$S(\nu)$. (أ) من أجل خط واحد: أوجد $I(e)$؛ والدور بدلالة $e$. (ب) ومن أجل خط
غاوسي عرضه $\Delta\nu$: بيّن أن للهدب غلافًا غاوسيًّا عرضه
$\sim c/2\Delta\nu$ بدلالة $e$ (واقبل أن تحويل فورييه لغاوسي
غاوسيٌّ). (ج) وأما التمييز: فخطان يبعد أحدهما عن الآخر $\Delta\nu$ ينفصلان إذا
بلغ المسح $e_{\max} \approx c/2\Delta\nu$؛ ومن أجل $e_{\max} = \qty{10}{cm}$، أوجد
قدرة التمييز $\lambda/\Delta\lambda$ عند \qty{1}{\micro m}. (د) ولمَ تُفضَّل هذه الطريقة
في تحت الأحمر (بسبب الإشارة، وميزة "تعدد الإرسال")؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:11}
\emph{\omterm{rem:b2:michelson:history}{كاشف الأمواج الثقالية}.} ذراعان $L = \qty{4}{km}$، والضوء معاد التدوير
إلى $N = 300$ ذهابًا وإيابًا، و $\lambda = \qty{1064}{nm}$، وتغيّر الموجة الذراعين بمقدار
$\Delta L/L = 10^{-21}$ في جهتين متعاكستين. (أ) أوجد تغيّر \omterm{thm:b2:two-wave-interference:formula}{فرق المسير}؛
وانزياح الطور. (ب) وتُثبَّت الأداة على هدبة مظلمة وتقيس
الشدة الصغيرة $I = I_0\sin^2(\Delta\varphi/2)$: أوجد تغيّر الشدة من أجل
الإشارة. (ج) ومع \qty{100}{W} من الضوء الدائر عند \qty{1064}{nm}، أوجد الفوتونات
في الثانية؛ وضجيج الفوتونات في \qty{1}{ms} ($\sqrt N$) في وجه الإشارة:
أتُكشَف؟ وماذا تشتري استطاعة أكبر؟ (د) ولمَ تكون الذراعان
بطول كيلومترات، ولمَ تُعلَّق المرايا؟
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:12}
\emph{التوضيع.} (أ) بيّن أن للشعاعين الخارجين من
نقطة $P$ من $M_2$ نحو راصد في اللانهاية (بالورود $i$) في الإسفين فرق
مسير $2e(P)\cos i$ زائدًا حدًّا $\propto\alpha i$ ينعدم عند
$i = 0$: فالهدب على المرآتين عند الورود الناظمي وحده. (ب)
ويضيء \omterm{prop:b2:two-wave-interference:spatial}{منبع ممتد} الإسفين بورودات حتى
$i_{\max}$: فما الذي يجب أن يحققه $i_{\max}$ لتبقى الهدب صافية
($2e\,i_{\max}^2/2 < \lambda/4$ عند حرف الحقل، $e = \qty{20}{\micro m}$)؟ (ج)
وفي الصفيحة المتوازية لا تتحرك الحلقات حين تتحرك العين:
فلماذا؟ (د) وترى طالبة هدبًا تهيم حين تحرّك رأسها:
ففي أي تشكيلة هي، وماذا عليها أن تضبط؟
\end{exercise}

\section{مسألة: ثنائي الصوديوم وقرينة الهواء}

\begin{problem}[{مسألة نهاية الأسبوع --- جلسة قياس كاملة على
مقياس تداخل ميكلسون: إيجاد التماس، وقراءة الأطوال، وفصل
الخط الأصفر، ووزن الهواء}]
\label{pb:b2:michelson:1}

لمقياس التداخل لولب ميكرومتري يقرأ \qty{1}{\micro m}؛ ويعطي
مصباح الصوديوم $\lambda_1 = \qty{589.0}{nm}$ و $\lambda_2 = \qty{589.6}{nm}$ بشدتين
نسبتهما $2 : 1$؛ ويتوفر أيضًا مصباح أبيض وليزر هليوم--نيون (\qty{632.8}{nm}).

\medskip
\noindent\textbf{الجزء الأول --- التهيئة.}
\begin{enumerate}
  \item بالليزر ترى الطالبة بقعتي الذراعين على
        بطاقة؛ فتطابقهما بلوالب الميل ثم ترى
        هدبًا مستقيمة: فما هذه التشكيلة، وماذا يقول
        تباعدها عن الميل الباقي (\omterm{def:b2:maxwell-equations:operators}{والتباعد} \qty{2}{mm})؟
  \item وبإدارة لوالب الميل تنحني الهدب ثم تصير
        حلقات: فلمَ تظهر الحلقات، وأين تقع $M_1'$ الآن؟
  \item وتزيح المرآة فتعدّ $158$ حلقة ابتلعها
        المركز: أوجد الإزاحة؛ وفي أي جهة يتغير $e$؟
  \item وباستمرارها تتباعد الحلقات وتكبر البقعة المركزية
        حتى تملأ الحقل: فماذا بلغت؟ ويحلّ مصباح الصوديوم
        محلّ الليزر فلا ترى شيئًا مميزًا: فلمَ ذلك،
        ولمَ يُظهر الضوء الأبيض هدبة مركزية مضيئة تكتنفها
        هدب ملونة عند تلك النقطة؟
  \item وتميل $M_1$ ميلًا طفيفًا جدًّا عند التماس فترى، في الضوء
        الأبيض، هدبًا مستقيمة ملونة: فسّر ذلك، وقل كم هدبة
        تتوقعها على كل جانب من البيضاء.
  \item وبقعتا الليزر بعرض \qty{10}{mm} وتريد عشر حلقات
        في حقل عدسة $f = \qty{20}{cm}$ ($\lambda = \qty{632.8}{nm}$):
        أوجد السماكة $e$ اللازمة (استعمل نصف قطر الحلقة $k$).
  \item ولمَ يجب أن توازي \omterm{def:b2:michelson:instrument}{الصفيحة المعوِّضة} الشاطر،
        وماذا كانت ترى في الضوء الأبيض لو نُزعت؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثاني --- الثنائي.}
\begin{enumerate}[resume]
  \item انطلاقًا من التماس بمصباح الصوديوم، تزيد $e$:
        فيهبط تباين الحلقات، ثم ينعدم، ثم يعود. أوجد سماكة
        الاختفاء الأول؛ والاختفاء $n$.
  \item وتسجّل $12$ اختفاءً بين $e = \qty{0.15}{mm}$ و
        $e = \qty{3.35}{mm}$ (وهما الأول والأخير مقروءين على اللولب):
        أوجد $\Delta\lambda$؛ والدقة إذا كانت كل قراءة بدقة \qty{5}{\micro m}.
  \item وأوجد رتبتي الخطين $p_1$ و $p_2$ عند الاختفاء
        الأول؛ وعند عودة التباين التامّ أول مرة.
  \item وأصغر تباين عند اختفاء، مع نسبة الشدتين
        $2 : 1$ (إذ تبقى هدب الخط الأضعف).
  \item وكم اختفاءً تستطيع أن تعدّ قبل أن يمحو العرض الطبيعي
        للخطين (\qty{0.02}{nm} لكل منهما) كل شيء؟
  \item وماذا كان يُظهر مصباح له ثلاثة خطوط متساوية \omterm{def:b2:maxwell-equations:operators}{التباعد}؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الثالث --- الهواء.}
توضع خلية طولها $\ell = \qty{5.00}{cm}$ لها نافذتان زجاجيتان في أحد
الذراعين وتُفرَّغ؛ ثم يُدخَل الهواء ببطء بينما تُعدّ هدب
الليزر.
\begin{enumerate}[resume]
  \item أوجد \omterm{thm:b2:two-wave-interference:formula}{فرق المسير} الضوئي الناتج عن ملء الخلية إلى
        \qty{1}{atm} إذا كان $n - 1 = \qty{2.9e-4}{}$؛ وعدد الهدب.
  \item وتعدّ $46$ هدبة: أوجد قيمة $n - 1$ عندها.
  \item وتُدخِل \qty{0.20}{bar} فقط: أوجد الهدب المتوقعة ($n - 1 \propto$
        الضغط).
  \item وتسخن الغرفة \qty{3}{K} في أثناء القياس، وتتغير قرينة
        \qty{40}{cm} من الهواء المكشوف في كل ذراع بمقدار
        $-\qty{1e-6}{}$ في الكلفن: أوجد هيام الهدب؛ أهو مشكلة في
        الجزء الثاني، وفي الجزء الثالث؟
  \item ولمَ لا تزيح نافذتا الخلية القياس (فهما موجودتان
        طوال العدّ)؟
  \item وتُعاد التجربة بغاز CO$_2$ ($n - 1 = \qty{4.5e-4}{}$):
        أوجد الهدب من أجل خلية ممتلئة؛ وكيف يمكن تعرّف الغاز من
        العدّ وحده؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء الرابع --- الحدود.}
\begin{enumerate}[resume]
  \item يُقرأ اللولب حتى \qty{1}{\micro m}: أوجد دقة طول
        مقيس بعدّ الهدب على \qty{1}{cm}، مقابل قياسه باللولب
        وحده.
  \item واهتزاز سعته \qty{50}{nm} عند \qty{100}{Hz} يهزّ إحدى
        المرآتين: أوجد حركة الهدب؛ وماذا ترى العين، وماذا ترى
        آلة تصوير على $25$ إطارًا في الثانية؟
  \item ويهيم تواتر الليزر بمقدار \qty{1e-8}{} نسبيًّا
        في أثناء عدّ على $e = \qty{1}{cm}$: أوجد الخطأ بالهدب.
  \item وقالب معايرة \qty{10}{mm} مصدَّق بدقة $\lambda/20$
        بمقياس التداخل: أوجد الدقة المطلقة والنسبية.
  \item ولمَ يكون الميكلسون أداة الاختيار من أجل الأطوال،
        والشبكة (في الفصل التالي) من أجل الأطياف؟
  \item ولخّص المقادير الأربعة المقيسة (طول، وفصل ثنائي،
        وقرينة، وطول ترابط) وملاحظة الهدب
        التي أعطت كلًّا منها.
\end{enumerate}
\end{problem}
