\chapter{الموائع المثالية: أويلر وبرنولي}\label{ch:b2:euler-bernoulli}

أمسك ورقة من حافتها وانفخ فوقها: فترتفع. وافتح صنبور الدُّشّ
فيميل الستار نحو الداخل. ويحمل الجناح ثلاثمئة طن لأن الهواء
يجري فوق سطحه العلوي أسرع مما يجري تحته. وهذه الوقائع الثلاث
عبارة واحدة: \emph{حيث يمضي المائع أسرع، يكون ضغطه أدنى}. وهذا
الفصل يكتب قانون نيوتن الثاني \omterm{def:b2:fluid-kinematics:particle}{لجسيم مائع} في أبسط نموذج ---
وهو المائع \emph{المثالي}، الذي يحسّ الضغط ولا يحسّ الاحتكاك ---
ويستخرج منه مبرهنة برنولي، أي موازنة الطاقة على \omterm{def:b2:fluid-kinematics:euler}{خط تيار}،
وتمتد تطبيقاتها من مقياس التدفق في أنبوب إلى مؤشر سرعة
الطائرة.

\section{معادلة أويلر}

\begin{proposition}[قوة الضغط على جسيم مائع]\label{prop:b2:euler-bernoulli:pressure}
\index{قوة الضغط (كثافة حجمية)}
محصّلة قوى الضغط التي يؤثر بها المائع المحيط في جسيم حجمه
$\dd\tau$ هي
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
أي إن الضغط يؤثر كقوة حجمية كثافتها $-\operatorname{\vect{grad}}P$،
تدفع من الضغط المرتفع نحو المنخفض.
\end{proposition}

\begin{proof}
على الصندوق الوارد في \cref{thm:b2:fluid-kinematics:continuity}، يتلقى
الوجهان عند $x$ و $x + \dd x$ القوتين $+P(x)\dd y\dd z$ و $-P(x + \dd x)\dd y\dd z$
على امتداد $x$: فالصافي $-\partial_xP\,\dd x\dd y\dd z$؛ وكذلك من أجل $y$ و $z$.
\end{proof}

\begin{definition}[المائع المثالي]\label{def:b2:euler-bernoulli:perfect}
\index{مائع مثالي}
\emph{المائع المثالي} هو الذي لا تكون فيه قوة التماس بين
الجسيمات المتجاورة إلا الضغط --- ناظمية على كل سطح، بلا إجهاد
مماسي (لزج). وهو حالة مثالية صالحة، كما يشرح الفصل التالي،
بعيدًا عن الجدران وعند عدد رينولدز الكبير: كالماء في أنبوب
بعيدًا عن الجدار، والهواء حول جناح خارج الطبقة الحدّية
الرقيقة، والموجة على البحر.
\end{definition}

\begin{theorem}[معادلة أويلر]\label{thm:b2:euler-bernoulli:euler}
\index{معادلة أويلر}
في مرجع عطالي، يحقق \omterm{def:b2:euler-bernoulli:perfect}{مائع مثالي} كثافته $\rho$ خاضع للثقالة، عند
كل نقطة،
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
وتُضاف إلى الطرف الأيمن أي كثافة قوة حجمية أخرى $\vect f_v$
(كهربائية، أو عطالية في مرجع غير عطالي). وعند السكون تؤول إلى
العلاقة الهيدروستاتيكية $\operatorname{\vect{grad}}P = \rho\vect g$ الواردة
في كتاب السنة الأولى.
\end{theorem}

\begin{proof}
قانون نيوتن الثاني للجسيم ذي الكتلة $\rho\,\dd\tau$: فتسارعه هو
\omterm{thm:b2:fluid-kinematics:material}{المشتق الجسيمي} (\cref{thm:b2:fluid-kinematics:material})، والقوى هي
محصّلة الضغط والثقل $\rho\vect g\,\dd\tau$؛ ثم نقسم على $\dd\tau$.
\end{proof}

\begin{example}[خزان يتسارع]\label{ex:b2:euler-bernoulli:tank}
الماء في خزان تحمله شاحنة تتسارع بالتسارع $\vect a$ ساكن في
مرجع الشاحنة؛ وفيه تعطي \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} مع قوة العطالة $-\rho\vect a$
العلاقة $\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$: أي إن الضغط
يتزايد على امتداد «الثقالة الفعّالة» $\vect g - \vect a$، ويكون
السطح الحرّ (وهو سطح متساوي الضغط) عموديًّا عليها ويميل إلى
الخلف بالزاوية $\arctan(a/g)$ --- أي \qty{11}{\degree} من أجل $a = \qty{2}{m/s^2}$.
وفي دلو يدور بالسرعة $\Omega$، تعطي القوة الطاردة المركزية $\rho\Omega^2r\,\vect e_r$
العلاقة $P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$ ويكون السطح الحرّ
المجسّم المكافئ $z = \Omega^2r^2/2g$.
\end{example}

\section{مبرهنة برنولي}

\begin{theorem}[برنولي]\label{thm:b2:euler-bernoulli:bernoulli}
\index{مبرهنة برنولي}\index{ضغط الركود}\index{ضغط ديناميكي}
من أجل مائع \emph{مثالي} \emph{لاانضغاطي} منتظم الكثافة في
جريان \emph{مستقر}، في مرجع عطالي وثقالة منتظمة، يكون المقدار
\[
P + \tfrac12\rho v^2 + \rho gz
\]
ثابتًا \emph{على امتداد كل \omterm{def:b2:fluid-kinematics:euler}{خط تيار}}. وإذا كان الجريان
\emph{لادورانيًّا} فوق ذلك، كان الثابت واحدًا في المائع كله.
والحدّ $\tfrac12\rho v^2$ هو \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{الضغط الديناميكي}}؛ و $P + \tfrac12\rho v^2$
هو \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{ضغط الركود}}، أي الضغط الذي يبلغه المائع لو أُسكن على
امتداد خط تياره.
\end{theorem}

\begin{proof}
\omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} المستقرة مع $(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$ و $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$، و $\rho$ منتظمة:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
والطرف الأيمن عمودي على $\vect v$، فليس لتدرج مقدار برنولي
مركّبة على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط التيار}: ومنه يبقى المقدار ثابتًا عليه.
وإذا كانت $\vect\omega = \vect 0$ انعدم التدرج في كل مكان.
\end{proof}

\begin{remark}[ما تقوله مبرهنة برنولي وما لا تقوله]\label{rem:b2:euler-bernoulli:meaning}
مقسومةً على $\rho$، تُقرأ المبرهنة هكذا: الطاقة الميكانيكية في
واحدة الكتلة، $v^2/2 + gz$، مضافًا إليها $P/\rho$، محفوظة على امتداد
مسار الجسيم. وحدّ الضغط هو الشغل المبذول \emph{على} الجسيم من
ضغط المائع خلفه، ناقصًا الشغل الذي يبذله على المائع أمامه: أي
مبرهنة الطاقة لجسيم يُدفع في مجال ضغط بلا احتكاك. وشروطها
أربعة: \omterm{def:b2:euler-bernoulli:perfect}{مائع مثالي} (بلا ضياع لزج)، ومستقر، ولاانضغاطي،
و\emph{على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط تيار}}. ولا تجوز مقارنة نقطتين على خطي
تيار مختلفين إلا في \omterm{def:b2:fluid-kinematics:vorticity}{جريان لادوراني}. وهي لا تقول «إن الهواء
السريع يمصّ»: بل تقول إن جسيمًا \emph{تسارع} (بانخفاض ضغط) صار
الآن عند ضغط أدنى --- فالسبب هو مجال الضغط، والسرعة هي الأثر.
\end{remark}

\begin{remark}[الموائع الانضغاطية]\label{rem:b2:euler-bernoulli:compressible}
تتغير الكثافة في الغاز، ولكن إذا كان الجريان مستقرًّا ومثاليًّا
وثابت الإنتروبيا أعطى البرهان نفسه انحفاظ $h + v^2/2 + gz$
على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط تيار}، حيث $h$ الإنثالبي في واحدة الكتلة ($h = c_PT$
في غاز مثالي): وهي الصيغة المستعملة في الفوهات والعنفات في
\cref{ch:b2:open-systems}. وعند سرعات أدنى بكثير من سرعة الصوت
($v \lesssim \qty{100}{m/s}$ في الهواء) تتغير الكثافة بأقل من $5\%$
وتكون المبرهنة اللاانضغاطية دقيقة.
\end{remark}

\section{تطبيقات}

\begin{proposition}[مقياس فنتوري]\label{prop:b2:euler-bernoulli:venturi}
\index{أثر فنتوري}\index{مقياس فنتوري}
في أنبوب أفقي يضيق من المقطع $S_1$ إلى $S_2 < S_1$، ترتفع
السرعة ($v_2 = v_1S_1/S_2$، بانحفاظ الكتلة) وينخفض الضغط:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
وقياس فرق الضغط بمانومتر يعطي التدفق: وذلك \emph{\omterm{prop:b2:euler-bernoulli:venturi}{مقياس فنتوري}}.
والأثر نفسه يشغّل المُمزِج في المحرك، ومرذاذ الطلاء، ومدخل
الهواء في موقد بنزن، و«مصّ» قطار عابر.
\end{proposition}

\begin{proof}
مبرهنة برنولي على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط تيار} مركزي عند ارتفاع ثابت، مع $v_2 =
v_1S_1/S_2$؛ ثم نحلّ بالنسبة إلى $v_1$.
\end{proof}

\begin{proposition}[أنبوب بيتو]\label{prop:b2:euler-bernoulli:pitot}
\index{أنبوب بيتو}
أنبوب يواجه الجريان، مغلق من طرفه البعيد، يُسكن المائع عند
فمه (أي \emph{نقطة ركود})؛ وفتحة ثانية على جانبه، موازية
للجريان، تقرأ الضغط السكوني $P$. والفرق هو \omterm{thm:b2:euler-bernoulli:bernoulli}{الضغط الديناميكي}:
\[
P_{\text{ركود}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{ركود}} - P)/\rho} .
\]
وكل طائرة تقيس سرعتها الجوية بهذه الطريقة.
\end{proposition}

\begin{proof}
مبرهنة برنولي على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط التيار} المنتهي عند نقطة الركود،
حيث $v = 0$؛ ولا تزعج الفتحات الجانبية الجريان، الذي يمرّ عند
$v$ و $P$ غير المضطربين.
\end{proof}

\begin{proposition}[صيغة توريتشيلي]\label{prop:b2:euler-bernoulli:torricelli}
\index{صيغة توريتشيلي}\index{تدفق الخروج}
خزان كبير مفتوح يُفرَّغ عبر ثقب صغير على عمق $h$ تحت السطح
الحرّ؛ ويغادر النفث بالسرعة
\[
v = \sqrt{2gh} ,
\]
وهي سرعة السقوط الحرّ من السطح. والتدفق هو $D_V \approx
\alpha s\sqrt{2gh}$ حيث $s$ مساحة الثقب و $\alpha \approx 0.6$
\emph{معامل التقلّص} لفتحة حادّة الحافة (إذ تستمر \omterm{def:b2:fluid-kinematics:euler}{خطوط التيار}
في التقارب بعد الثقب).
\end{proposition}

\begin{proof}
مبرهنة برنولي من السطح الحرّ (وهو ساكن، لأن الخزان كبير:
$v_{\text{سطح}} = v\,s/S \ll v$؛ وضغطه $P_0$) إلى النفث (وضغطه $P_0$
خارج الثقب مباشرةً): $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$. والجريان
شبه مستقر: إذ يتغير $h$ ببطء.
\end{proof}

\begin{proposition}[قوة الرفع على الجناح]\label{prop:b2:euler-bernoulli:lift}
\index{قوة الرفع}\index{مبرهنة كوتا--جوكوفسكي}\index{أثر ماغنوس}
في الجريان حول جناح يمرّ الهواء فوق السطح العلوي أسرع مما
يمرّ تحت السفلي؛ ومنه يكون الضغط أدنى فوق منه تحت، وتكون
المحصّلة \emph{\omterm{prop:b2:euler-bernoulli:lift}{قوة الرفع}} $F_L =
\tfrac12\rho v^2SC_L$ (و $S$ مساحة الجناح، و $C_L \approx 0.3$--$1.5$ معامل
الرفع، وهو يتوقف على المقطع الجانبي وعلى زاوية الهجوم). ويكافئ
فرق السرعتين \emph{تكاملًا خطيًّا} $\Gamma$ للسرعة حول المقطع،
وتكون \omterm{prop:b2:euler-bernoulli:lift}{قوة الرفع} في واحدة الباع $F_L' =
\rho v\Gamma$ (كوتا--جوكوفسكي، مسلَّم بها). وكرة تدور تجرّ الهواء
معها فتُرفع جانبيًّا بالطريقة نفسها: وهو \emph{\omterm{prop:b2:euler-bernoulli:lift}{أثر ماغنوس}} في
كرة المضرب المشذَّبة أو الركلة الحرة المنحنية.
\end{proposition}

\begin{proof}
مبرهنة برنولي على خطي التيار فوق الجناح وتحته مباشرةً، وهما
ينطلقان من الشروط نفسها في المنبع: $P_{\text{تحت}} - P_{\text{فوق}} = \tfrac12\rho
(v_{\text{فوق}}^2 - v_{\text{تحت}}^2)$؛ ثم نكامل على الوتر. وأما لماذا
يكون الجريان العلوي أسرع --- إذ تفرض الحافة الخلفية الحادّة
نقطة الركود الخلفية هناك فتثبّت التكامل الخطي (شرط كوتا) ---
فذلك خارج نموذج \omterm{def:b2:euler-bernoulli:perfect}{المائع المثالي} وحده؛ إذ يحتاج إلى الطبقة
الحدّية في الفصل التالي.
\end{proof}

\begin{proposition}[جريان غير مستقر: الأنبوب على شكل U]\label{prop:b2:euler-bernoulli:utube}
\index{اهتزاز في أنبوب U}\index{برنولي غير المستقر}
عمود سائل طوله الكلي $L$ يملأ أنبوبًا على شكل U منتظم المقطع؛
فإذا أُزيح بالمقدار $z$ عن التوازن اهتزّ وفق
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
كنواس طوله $L/2$. وعمومًا، من أجل جريان غير مستقر لكنه
لادوراني ذي كمون $\varphi$، تُكامَل \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} فتعطي
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$، وهو الثابت
نفسه في المائع كله (\omterm{prop:b2:euler-bernoulli:utube}{برنولي غير المستقر}).
\end{proposition}

\begin{proof}
السرعة $v = \dot z$ منتظمة على امتداد الأنبوب (فهو لاانضغاطي
ومنتظم المقطع). ونسقط \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على محور الأنبوب ونكاملها
على امتداد العمود من سطح حرّ إلى الآخر: فتتلاشى حدود الضغط
($P_0$ عند الطرفين)، ويكون $\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$، وينعدم
الحدّ الحملي $\rho\,\Delta(v^2/2)$ (إذ السرعة واحدة عند الطرفين)،
ويعطي حدّ الثقالة $\rho g\cdot2z$. ومنه $\rho L\ddot z = -2\rho gz$.
وينتج التعميم من $\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ من أجل $\vect v = \operatorname{\vect{grad}}\varphi$.
\end{proof}

\begin{example}[إقلاع أنبوب]\label{ex:b2:euler-bernoulli:start}
أنبوب أفقي طوله $\ell$ يصل خزانًا ذا علو ثابت $h$ بطرف
مفتوح؛ ويُفتح الصمام عند $t = 0$. ومع $v$ منتظمة
على امتداد الأنبوب، تعطي \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} مكاملةً من سطح الخزان إلى
المخرج $\ell\,\dd v/\dd t = gh - \tfrac12v^2$، وحلّها
$v = V\tanh(Vt/2\ell)$ حيث $V = \sqrt{2gh}$: أي إن الجريان يبلغ سرعة
توريتشيلي بثابت الزمن $2\ell/V$ --- وهو \qty{13}{s} من أجل أنبوب جرّ
طوله \qty{400}{m} تحت علو \qty{200}{m}.
\end{example}

\begin{method}[استعمال مبرهنة برنولي]\label{met:b2:euler-bernoulli:method}
(1) تحقق من الشروط الأربعة؛ وقرّر أيكون الجريان لادورانيًّا
(جريان منتظم في المنبع حول عائق: نعم؛ جريان في أنبوب ذي منحني
سرعات: لا --- فابقَ على \omterm{def:b2:fluid-kinematics:euler}{خط تيار} واحد). (2) اختر نقطتين
على \omterm{def:b2:fluid-kinematics:euler}{خط تيار} يُعرف عندهما أكبر عدد ممكن من المقادير: سطح حرّ
($P = P_0$، و $v \approx 0$ إذا كان المقطع كبيرًا)، أو نفث في الهواء
الطلق ($P = P_0$)، أو نقطة ركود ($v = 0$). (3) أضف انحفاظ
الكتلة $vS = \text{ثابت}$ لربط السرعات. (4) وفي الجريانات غير
المستقرة، كامِل \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على امتداد الخط بدلًا من ذلك.
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:2}

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:3}
خزان فيه ماء عمقه \qty{3.0}{m}؛ ويُثقب ثقب مساحته \qty{1.0}{cm^2}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:4}
تهبّ ريح بسرعة \qty{30}{m/s} فوق سقف مستو مساحته \qty{100}{m^2}؛
والهواء تحت السقف ساكن عند الضغط الجوي ($\rho = \qty{1.2}{kg/m^3}$).
أوجد فرق الضغط والقوة الصافية على السقف؛ وقارنها بثقل

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:5}
خزان مستطيل مغلق طوله \qty{2.0}{m}، نصفه ماء، موضوع على
شاحنة. (أ) وهي تتسارع بالتسارع \qty{3.0}{m/s^2}: أوجد زاوية السطح
الحرّ، وفرق ارتفاع الماء بين الجدارين الخلفي والأمامي.
(ب) فرق الضغط بين ركني القاع. (ج) وهي تكبح
بالتباطؤ \qty{6}{m/s^2} --- أيبلغ الماء الغطاء الذي يعلو \qty{0.8}{m}
عن القاع؟ (د) والشاحنة نفسها تنعطف في منحنٍ نصف قطره \qty{40}{m}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
دلو نصف قطره \qty{15}{cm} فيه ماء عمقه \qty{20}{cm} يُدار حول
محوره بمعدل دورتين في الثانية. (أ) شكل السطح الحرّ ومعادلته.
(ب) فرق الارتفاع بين الحافة والمركز. (ج) وبانحفاظ الحجم، أوجد

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{التضيّق.} يجري الدم ($\rho = \qty{1060}{kg/m^3}$) بالسرعة \qty{0.50}{m/s}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{درجة حرارة الركود.} من أجل غاز مثالي في \omterm{def:b2:fluid-kinematics:euler}{جريان مستقر} ثابت
الإنتروبيا، ينحفظ $c_PT + v^2/2$ على امتداد \omterm{def:b2:fluid-kinematics:euler}{خط تيار}. (أ) بيّن أن
الهواء المُسكَن عند مقدمة طائرة يبلغ $T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$، حيث $M = v/c$ عدد ماخ و $c =
\sqrt{\gamma RT/M_{\text{مول}}}$ سرعة الصوت. (ب) أوجد درجة حرارة
المقدمة في طائرة ركاب عند $M = 0.85$ في هواء عند \qty{220}{K}؛ وفي
طائرة أسرع من الصوت عند $M = 2$؛ وفي كبسولة عائدة عند $M = 25$ (أتبقى
الصيغة ذات معنى؟). (ج) بيّن أن صيغة بيتو اللاانضغاطية تفرط في
تقدير السرعة عند $M = 0.85$، وبكم تقريبًا (انشر
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ حتى المرتبة الثانية في $M^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{الطور العابر عند الإقلاع.} أنبوب طوله $\ell = \qty{50}{m}$ منتظم
المقطع يصل خزانًا كبيرًا (علوّه $h = \qty{5.0}{m}$ فوق
المخرج) بصمام عند الطرف المفتوح. ويُفتح الصمام عند $t = 0$. (أ)
كامِل \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على امتداد الأنبوب لتحصل على $\ell\,\dd v/\dd t = gh - v^2/2$.
(ب) حُلّها: $v = V\tanh(Vt/2\ell)$ حيث $V = \sqrt{2gh}$. (ج) الزمن اللازم
لبلوغ $90\%$ من السرعة النهائية. (د) ثم يُغلق الصمام فجأةً في
\qty{0.5}{s} من التدفق الكامل: قدّر التباطؤ الوسطي وفرط
الضغط عند الصمام من $\rho\ell\,\dd v/\dd t$ --- ولماذا يولّد الإغلاق
الحقيقي أكثر من ذلك بكثير (في الفصول التالية).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{السطح الحرّ لدوامة.} يدور الماء دوامةً نقطية كما في
\cref{ch:b2:fluid-kinematics}، $v_\theta = \Gamma/2\pi r$، لادورانية،
وسطحها الحرّ عند $z = 0$ بعيدًا. (أ) لماذا يجوز تطبيق مبرهنة برنولي
بين أي نقطتين؟ (ب) شكل السطح الحرّ $z(r)$. (ج)
وداخل لبّ رانكين ($r < a$، و $v_\theta = \Omega r$، ودوراني) استعمل
بدلًا من ذلك \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} في المرجع الدوّار أو مباشرةً: بيّن أن $z = z(a) +
\Omega^2(r^2 - a^2)/2g$. (د) العمق الكلي للقمع من أجل دوامة حوض
استحمام قيمته $\Gamma = \qty{0.03}{m^2/s}$ و $a = \qty{5}{mm}$؛ ومن أجل إعصار قمعي
($\Gamma = \qty{3e4}{m^2/s}$ و $a = \qty{60}{m}$، في الهواء: عبّر عن
النتيجة بانخفاض ضغط عند المركز بدلًا من ذلك).
\end{exercise}

\section{مسألة: السدّ وأنبوب الجرّ والعنفة}

\begin{problem}[{مسألة نهاية الأسبوع --- محطة كهرمائية من الخزان
إلى النفث: الضغوط والسرعات والاستطاعات والضياعات والصدمة
المائية}]
\label{pb:b2:euler-bernoulli:1}

خزان فيه ماء عمقه \qty{150}{m} خلف سدّ؛ وسطحه الحرّ
يعلو $H = \qty{200}{m}$ فوق فوهات عنفة. وينقل أنبوب جرّ (وهو
أنبوب فولاذي) قطره $D = \qty{3.0}{m}$ وطوله $\ell = \qty{400}{m}$
التدفق التصميمي $D_V = \qty{60}{m^3/s}$ نزولًا إلى بيت العنفات،
حيث تحوّله فوهة إلى نفث حرّ عند الضغط الجوي $P_0 =
\qty{1.0}{bar}$ يصطدم بأكواب عجلة بلتون. و $\rho =
\qty{1000}{kg/m^3}$، و $g = \qty{9.81}{m/s^2}$؛ \omterm{def:b2:euler-bernoulli:perfect}{والمائع مثالي} ما لم
يُذكر خلاف ذلك.

\medskip
\noindent\textbf{الجزء I --- عند السكون وعند التدفق التصميمي.}
\begin{enumerate}
  \item الضغط عند قاعدة السدّ، والقوة على شريحة شاقولية من
        السدّ عرضها \qty{1}{m} من أعلاه إلى أسفله.
  \item وصمامات العنفة مغلقة، أوجد الضغط في أنبوب الجرّ عند
        بيت العنفات.
  \item السرعة في أنبوب الجرّ عند التدفق التصميمي؛ \omterm{thm:b2:euler-bernoulli:bernoulli}{والضغط
        الديناميكي} هناك؛ والضغط عند طرف أنبوب الجرّ عند بيت
        العنفات (قبل الفوهة)، بمبرهنة برنولي من سطح الخزان.
  \item سرعة النفث الحرّ؛ ومقطع الفوهة وقطرها.
  \item الاستطاعة الحركية التي يحملها النفث؛ بيّن أنها تساوي
        $\rho gD_VH$ ثم احسبها.
  \item كتلة الماء الموجود في أنبوب الجرّ عند التدفق التصميمي
        وطاقته الحركية؛ وزمن عبور جسيم نزولًا في الأنبوب.
  \item بيّن أن العلو $H$ يتوزع، عند التدفق التصميمي، بين الضغط
        في أسفل أنبوب الجرّ والطاقة الحركية للماء، وأن الفوهة
        تحوّل الأول إلى الثانية: فما الضغط بُعيد الفوهة مباشرةً؟
  \item لا ضياع علو في \omterm{def:b2:euler-bernoulli:perfect}{مائع مثالي}، وأما الحقيقي فله ضياع: إذ
        الضياع في أنبوب هو $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ حيث $\lambda
        \approx 0.015$ للفولاذ الأملس. أوجد ضياع العلو بأمتار الماء،
        والنسبة الضائعة من الاستطاعة.
  \item يقترح مهندس أنبوب جرّ قطره \qty{2}{m} توفيرًا للفولاذ.
        أعد السؤال 7؛ وعلّق.
\end{enumerate}

\medskip
\noindent\textbf{الجزء II --- قياس التدفق.}
يُركَّب \omterm{prop:b2:euler-bernoulli:venturi}{مقياس فنتوري} في أنبوب الجرّ، حلقه \qty{2.0}{m}.
\begin{enumerate}[resume]
  \item السرعة عند الحلق وانخفاض الضغط بين الأنبوب والحلق عند
        التدفق التصميمي.
  \item ويُقرأ الانخفاض على مانومتر زئبقي ($\rho_{\text{زئبق}} =
        \qty{13600}{kg/m^3}$): فما فرق الارتفاع؟
  \item بيّن أن التدفق يتناسب مع الجذر التربيعي لقراءة
        المانومتر، وأعطِ القراءة عند نصف التدفق التصميمي.
  \item ولماذا يجب أن تكون مآخذ الضغط مستوية مع الجدار لا
        بارزة في الجريان؟
  \item ويمرّ جزء من الأنبوب، لأسباب طبوغرافية، فوق قمة ترتفع
        \qty{6}{m} \emph{فوق} سطح الخزان. أوجد الضغط هناك عند
        التدفق التصميمي؛ وماذا يحدث، وما الذي على المصمم فعله؟
\end{enumerate}

\medskip
\noindent\textbf{الجزء III --- الأنظمة غير المستقرة.}
مساحة سطح الخزان $A = \qty{2.0}{km^2}$.
\begin{enumerate}[resume]
  \item المعدل الذي ينخفض به مستوى الخزان عند التدفق
        التصميمي، بالسنتيمتر في الساعة.
  \item وتُوقَف العنفة وتُترك الفوهة (بمقطعها من السؤال 4)
        مفتوحة على الهواء: فيُفرَّغ الخزان بالثقالة. اكتب موازنة
        الكتلة وأوجد المستوى $h(t)$ (بجريان توريتشيلي شبه
        المستقر).
  \item الزمن اللازم لهبوط المستوى \qty{10}{m}؛ وقارنه بالزمن
        عند التدفق التصميمي الثابت.
  \item وعند إقلاع المحطة: أنبوب الجرّ مملوء، وتُفتح الفوهة
        عند $t = 0$. بأخذ السرعة منتظمة على امتداد أنبوب الجرّ
        وإهمال هندسة الفوهة، كامِل \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على امتداد
        الأنبوب لتحصل على $\ell\,\dd v/\dd t = gH - v^2/2$، ثم حُلّها وأعطِ
        ثابت الزمن.
  \item وأثناء هذا الإقلاع، في اللحظة التي تبلغ فيها $v$ نصف
        قيمتها النهائية، ما الضغط في أسفل أنبوب الجرّ؟
        واشرح إشارة الفرق عن القيمة المستقرة.
\end{enumerate}

\medskip
\noindent\textbf{الجزء IV --- إغلاق الصمام.}
\begin{enumerate}[resume]
  \item يُغلق الصمام عند العنفة في $\tau = \qty{10}{s}$؛ وافترض أن
        الماء في أنبوب الجرّ يتباطأ تباطؤًا منتظمًا. وباستعمال
        \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على امتداد الأنبوب، قدّر فرط الضغط عند
        الصمام، بالبار.
  \item وفي الإغلاق السريع لا يكون الماء لاانضغاطيًّا: بل تسري
        موجة ضغط صاعدةً في الأنبوب بالسرعة $c = \qty{1400}{m/s}$ (وهي
        سرعة الصوت في الماء داخل أنبوب فولاذي --- \cref{ch:b2:sound-waves})،
        ويكون فرط الضغط $\Delta P = \rho c\,\Delta v$ (جوكوفسكي).
        أوجد القيمة من أجل توقف مفاجئ من التدفق التصميمي؛
        وقارنها بالضغط السكوني. وكم تستغرق الموجة كي تبلغ
        الخزان وتعود، وماذا يقول ذلك عن معنى «سريع»؟
  \item ولحماية أنبوب الجرّ يُوصَل \emph{بئر توازن} --- وهو بئر
        شاقولي مفتوح مقطعه $A_s = \qty{100}{m^2}$ --- عند ملتقى
        أنبوب الجرّ بنفق أفقي طوله $L = \qty{2000}{m}$ ومقطعه $A =
        \qty{7.0}{m^2}$ آتٍ من الخزان. وبعد إغلاق مفاجئ يبقى
        الماء في النفق متحركًا فيرتفع المستوى $z$ في البئر. اكتب
        انحفاظ الكتلة بين النفق والبئر.
  \item كامِل \omterm{thm:b2:euler-bernoulli:euler}{معادلة أويلر} على امتداد النفق (\omterm{def:b2:euler-bernoulli:perfect}{مائع مثالي}) لتبيّن أن
        $z$ يحقق $\ddot z + \dfrac{gA}{LA_s}z = 0$؛ وأوجد دور
        الاهتزاز.
  \item الارتفاع الأعظمي للمستوى في البئر بعد إغلاق من التدفق
        التصميمي (بحجة الطاقة: إذ تصير الطاقة الحركية لماء
        النفق طاقة كامنة في البئر).
  \item لخّص: الضغوط الخمسة التي التقيناها في هذه المحطة
        (السكوني عند السدّ، والسكوني عند العنفة، والديناميكي في
        الأنبوب، والصدمة المائية، وارتفاع البئر) والقانون الذي
        يعطي كلًّا منها.
\end{enumerate}
\end{problem}
