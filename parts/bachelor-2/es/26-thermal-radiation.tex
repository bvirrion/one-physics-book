\chapter{Radiación térmica}\label{ch:b2:thermal-radiation}

Todo lo caliente brilla. Una mano, una pared, un vaso de agua a
temperatura ambiente radian de forma invisible, en longitudes de onda
cercanas a \qty{10}{\micro m}, unos cientos de vatios por metro cuadrado;
caliéntese un atizador y a \qty{600}{\celsius} brilla en rojo apagado, a
\qty{1000}{\celsius} en naranja y a \qty{2500}{\celsius} --- el filamento de
una bombilla --- en blanco; la superficie del Sol, a \qty{5800}{K}, radia
en blanco amarillento sesenta megavatios por metro cuadrado, y esa luz,
tras ocho minutos de viaje, es la que mantiene la Tierra a una temperatura
en la que el agua es líquida. La radiación es el tercer modo de
transferencia de calor y el único que cruza el espacio vacío; es además el
lugar donde falló la física clásica y Planck, en 1900, tuvo que introducir
el cuanto. Este capítulo enuncia la ley que encontró, saca de ella los dos
resultados que se usan constantemente --- el desplazamiento de Wien y la
ley de Stefan--Boltzmann --- y los aplica a los intercambios entre cuerpos,
al Sol y a la Tierra, y al modelo de una capa del \omterm{prop:b2:thermal-radiation:greenhouse}{efecto invernadero} que fija nuestro clima.

\section{La radiación térmica y el cuerpo negro}

\begin{definition}[Radiación térmica, absortividad, cuerpo negro]\label{def:b2:thermal-radiation:blackbody}
Todo cuerpo a la temperatura $T$ emite radiación electromagnética por la
agitación térmica de sus cargas: su \emph{radiación
térmica}\index{radiación térmica}. Su superficie recibe además
radiación de su entorno y absorbe una fracción $\alpha$ de ella
(la \emph{absortividad}, entre $0$ y $1$, que en general depende de
la longitud de onda), reflejando o transmitiendo el resto. Un \emph{cuerpo
negro}\index{cuerpo negro} lo absorbe todo ($\alpha = 1$ a cualquier
longitud de onda). Un agujerito en la pared de una cavidad cerrada es el
cuerpo negro práctico: la radiación que entra la absorben las paredes
antes de que encuentre la salida.
\end{definition}

\begin{proposition}[Radiación de equilibrio y ley de Kirchhoff]\label{prop:b2:thermal-radiation:kirchhoff}
Dentro de una cavidad cerrada cuyas paredes están a $T$, la radiación está en
equilibrio con las paredes y es \emph{universal}: isótropa,
no polarizada y con una densidad espectral de energía $u_\nu(T)$ (energía por unidad de
volumen y de frecuencia) que depende solo de $T$ y $\nu$, no del
material ni de la forma de la cavidad. La radiación que sale de un agujerito de
una cavidad así es la \emph{radiación del \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro}} a $T$. Una superficie que
absorbe la fracción $\alpha_\nu$ de la radiación que recibe a la
frecuencia $\nu$ emite, a esa frecuencia, la fracción
$\varepsilon_\nu = \alpha_\nu$ de lo que emitiría un \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} a la misma
temperatura (su \emph{emisividad}\index{emisividad}):
\emph{los buenos absorbentes son buenos emisores}.
\end{proposition}

\begin{proof}
Universalidad: conéctense dos cavidades a la misma $T$ por un agujero con un
filtro que deje pasar una frecuencia; si las densidades difirieran, la energía
fluiría de una a otra a temperaturas iguales, violando el segundo
principio. Kirchhoff: colóquese la superficie en la cavidad a su propia $T$; en
equilibrio debe emitir a cada frecuencia exactamente lo que absorbe,
$\varepsilon_\nu u_\nu = \alpha_\nu u_\nu$.
\end{proof}

\section{La ley de Planck}

\begin{theorem}[Ley de Planck]\label{thm:b2:thermal-radiation:planck}
La densidad espectral de energía de la radiación del \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} a la temperatura
$T$ vale
\[
u_\nu(T) = \frac{8\pi h\nu^3}{c^3}\,\frac{1}{\eu^{h\nu/k_BT} - 1},
\]
y la potencia radiada por una superficie negra por unidad de área y de
longitud de onda (su \emph{exitancia espectral}\index{exitancia espectral}) es
\[
M_\lambda(T) = \frac{2\pi hc^2}{\lambda^5}\,\frac{1}{\eu^{hc/\lambda k_BT} - 1}.
\]
Dos límites: para $h\nu \ll k_BT$, $u_\nu \approx 8\pi\nu^2 k_BT/c^3$
(Rayleigh--Jeans: cada modo de la cavidad lleva la energía clásica
$k_BT$ --- una ley cuya integral es infinita, la ``catástrofe
ultravioleta''); para $h\nu \gg k_BT$, $u_\nu \approx (8\pi h\nu^3/c^3)
\eu^{-h\nu/k_BT}$ (Wien: el factor de Boltzmann de un fotón de energía
$h\nu$, \cref{ch:b2:boltzmann-factor}).
\end{theorem}

\begin{proof}
Admitido: el recuento de los modos de la cavidad ($8\pi\nu^2/c^3$ por unidad de
volumen y de frecuencia, con dos polarizaciones) corresponde al volumen del tercer año,
y la energía media de un modo, $h\nu/(\eu^{h\nu/k_BT} - 1)$ en vez de la
clásica $k_BT$, es la hipótesis cuántica: la energía se intercambia con
el modo en cuantos $h\nu$, de modo que los modos con $h\nu \gg k_BT$ quedan
congelados. La relación $M_\lambda\,\dd\lambda = (c/4)\,u_\nu\,\dd\nu$ es el
factor geométrico que se calcula en \cref{thm:b2:thermal-radiation:stefan}.
\end{proof}

\begin{proposition}[Ley del desplazamiento de Wien]\label{prop:b2:thermal-radiation:wien}
La \omterm{thm:b2:thermal-radiation:planck}{exitancia espectral} $M_\lambda$ tiene su máximo en
\[
\lambda_{\max}\,T = \frac{hc}{4.965\,k_B} = \qty{2.90e-3}{m.K} .
\]
Un cuerpo a \qty{300}{K} radia sobre todo cerca de \qty{10}{\micro m}; el Sol
(\qty{5800}{K}), cerca de \qty{0.5}{\micro m}, en el verde, donde el ojo es
más sensible --- y no por casualidad.
\end{proposition}

\begin{proof}
Con $x = hc/\lambda k_BT$, $M_\lambda \propto x^5/(\eu^x - 1)$; $\dd/\dd x = 0$
da $5(\eu^x - 1) = x\eu^x$, es decir, $x = 5(1 - \eu^{-x})$, cuya raíz no
nula es $x = 4.965$.
\end{proof}

\begin{theorem}[Ley de Stefan--Boltzmann]\label{thm:b2:thermal-radiation:stefan}
Una superficie negra a $T$ radia, por unidad de área, la potencia total
\[
M = \int_0^\infty M_\lambda\,\dd\lambda = \sigma T^4, \qquad
\sigma = \frac{2\pi^5k_B^4}{15h^3c^2} = \qty{5.67e-8}{W/m^2/K^4} ;
\]
la densidad de energía de la radiación de equilibrio vale $u = (4\sigma/c)T^4$
y su presión $P = u/3$. Una superficie gris de \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividad} $\varepsilon$
radia $\varepsilon\sigma T^4$.
\end{theorem}

\begin{proof}
Intégrese la densidad de Planck: $u = \int u_\nu\,\dd\nu = (8\pi h/c^3)(k_BT/h)^4
\int_0^\infty x^3\,\dd x/(\eu^x - 1)$ y la integral vale $\pi^4/15$
(admitido del análisis): $u = (8\pi^5k_B^4/15h^3c^3)\,T^4$. La potencia
que sale de un agujero de área unidad en la pared de la cavidad es $cu/4$: la
radiación es isótropa, los fotones que van hacia el agujero dentro del
ángulo sólido $\dd\Omega$ con el ángulo $\theta$ respecto de su normal llevan el
flujo $c\,u\,(\dd\Omega/4\pi)\cos\theta$, y $\int\cos\theta\,\dd\Omega/4\pi$
sobre el hemisferio exterior vale $1/4$. De ahí $M = cu/4 = \sigma T^4$.
La presión, como para cualquier gas isótropo de partículas que se mueven a $c$, vale
$u/3$.
\end{proof}

\begin{example}[Órdenes de magnitud]\label{ex:b2:thermal-radiation:orders}
A \qty{300}{K}: $\sigma T^4 = \qty{460}{W/m^2}$ --- una persona de
\qty{1.7}{m^2} radia \qty{800}{W}, pero recibe casi otro tanto de
las paredes a \qty{293}{K}; la pérdida neta, $\sigma A(T^4 - T_0^4) \approx
\qty{100}{W}$, es toda la producción metabólica del cuerpo, y por eso hace falta
ropa en una habitación que parece templada. La superficie del Sol: $\sigma
(5800)^4 = \qty{6.4e7}{W/m^2}$, en total $3.9 \times 10^{26}\,\unit{W}$. El
filamento de una bombilla a \qty{2800}{K}: \qty{3.5e6}{W/m^2}, \qty{60}{W} de medio
centímetro cuadrado de wolframio. El fondo cósmico, a \qty{2.7}{K}:
máximo en \qty{1}{mm}, $\qty{3e-6}{W/m^2}$ y $400$ fotones por centímetro
cúbico del universo.
\end{example}

\section{Intercambios radiativos}

\begin{proposition}[Intercambio neto y coeficiente radiativo]\label{prop:b2:thermal-radiation:exchange}
Un cuerpo gris pequeño (de \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividad} $\varepsilon$, área $S$ y temperatura $T$)
dentro de un entorno amplio a $T_0$ pierde la potencia neta
\[
P = \varepsilon\sigma S\,(T^4 - T_0^4)
\ \approx\ 4\varepsilon\sigma T_0^3\,S\,(T - T_0) = h_{\text{r}}S(T - T_0)
\quad\text{para } |T - T_0| \ll T_0 ,
\]
donde $h_{\text{r}} = 4\varepsilon\sigma T_0^3 \approx \qty{6}{W/m^2/K}$ a
temperatura ambiente --- comparable a la convección natural: en una habitación,
la radiación se lleva cerca de la mitad del calor que sale de un cuerpo o de un radiador.
Entre dos placas negras paralelas y grandes, la densidad de flujo neta vale
$\sigma(T_1^4 - T_2^4)$; para placas grises de \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividades} $\varepsilon_1$ y
$\varepsilon_2$ vale $\sigma(T_1^4 - T_2^4)/(1/\varepsilon_1 + 1/\varepsilon_2
- 1)$, y cada \emph{pantalla de radiación} fina interpuesta la divide
aún más --- el principio del termo y de la lámina dorada de los
satélites.
\end{proposition}

\begin{proof}
El cuerpo emite $\varepsilon\sigma T^4$ y absorbe la fracción $\alpha =
\varepsilon$ de la radiación negra $\sigma T_0^4$ que llena el recinto
(Kirchhoff). Linealícese $T^4 - T_0^4 \approx 4T_0^3(T - T_0)$. Para las
placas grises, súmense las reflexiones múltiples (una serie geométrica).
\end{proof}

\begin{example}[Escarcha bajo un cielo despejado y la cámara térmica]\label{ex:b2:thermal-radiation:frost}
En una noche despejada el cielo radia como un cuerpo a unos \qty{250}{K}
(la alta atmósfera seca), mientras que el aire junto al suelo está a
\qty{278}{K}. Una hoja o el techo de un coche ($\varepsilon = 0.95$) equilibra su
pérdida radiativa hacia el cielo con la convección del aire ($h =
\qty{5}{W/m^2/K}$): $\varepsilon\sigma(T^4 - 250^4) = h(278 - T)$ da $T
\approx \qty{266}{K}$ --- escarcha a \qty{-7}{\celsius} en la hierba mientras
el termómetro marca $+5$. Una nube (que radia a \qty{275}{K}) sube $T$
a \qty{277}{K}: sin escarcha. Una cámara térmica mide la radiancia cerca de
\qty{10}{\micro m} y la convierte en temperatura suponiendo
$\varepsilon \approx 0.95$; un metal pulido ($\varepsilon = 0.1$) refleja
sobre todo la habitación y aparece a la temperatura de esta, sea cual sea la
suya --- de ahí la cinta negra que los termógrafos pegan en las piezas brillantes.
\end{example}

\section{El Sol, la Tierra y el invernadero}

\begin{proposition}[Constante solar y temperatura efectiva]\label{prop:b2:thermal-radiation:earth}
El Sol (de radio $R_\odot$ y temperatura superficial $T_\odot$) entrega a la
distancia $d$ la densidad de flujo $S = \sigma T_\odot^4 (R_\odot/d)^2 =
\qty{1361}{W/m^2}$ por encima de la atmósfera terrestre (la \emph{constante
solar}\index{constante solar}). Un planeta de albedo $A$ (la fracción
reflejada) absorbe $(1 - A)S\pi R^2$ y, en régimen estacionario, la radia
desde toda su superficie $4\pi R^2$ como un \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} a su \emph{temperatura
efectiva}\index{temperatura efectiva}
\[
T_{\text{ef}} = \Big(\frac{(1 - A)S}{4\sigma}\Big)^{1/4} = \qty{255}{K}
\quad\text{para la Tierra } (A = 0.30) .
\]
\end{proposition}

\begin{proof}
Conservación de la energía: absorbido $= $ emitido, con la \omterm{prop:b2:laser:gain}{sección eficaz} $\pi R^2$
para el haz interceptado y la esfera completa para la emisión (día y
noche, repartidos por la rotación y la circulación).
\end{proof}

\begin{proposition}[Modelo de invernadero de una capa]\label{prop:b2:thermal-radiation:greenhouse}
Rodéese el planeta con una capa fina de atmósfera transparente
a la luz solar pero que absorbe todo el infrarrojo emitido por la superficie y
radia como un \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} a $T_{\text{atm}}$ tanto hacia arriba como hacia abajo. En
régimen estacionario,
\[
T_{\text{atm}} = T_{\text{ef}}, \qquad
T_{\text{sup}} = 2^{1/4}\,T_{\text{ef}} = \qty{303}{K} .
\]
Si la capa solo absorbe la fracción $\varepsilon$ del infrarrojo,
$T_{\text{sup}}^4 = T_{\text{ef}}^4/(1 - \varepsilon/2)$; los \qty{288}{K}
observados corresponden a $\varepsilon \approx 0.78$. La superficie está
más caliente que la \omterm{prop:b2:thermal-radiation:earth}{temperatura efectiva} porque recibe, además
del Sol, el infrarrojo que la atmósfera devuelve hacia abajo --- el
\emph{efecto invernadero}\index{efecto invernadero}, de \qty{33}{K} en la Tierra,
sin el cual los océanos estarían helados.
\end{proposition}

\begin{proof}
Balance de la capa: absorbe $\sigma T_{\text{sup}}^4$ y emite $2\sigma
T_{\text{atm}}^4$. Balance del planeta visto desde el espacio: emite $\sigma
T_{\text{atm}}^4 = (1 - A)S/4 = \sigma T_{\text{ef}}^4$. Balance de la
superficie: recibe $(1 - A)S/4 + \sigma T_{\text{atm}}^4 = 2\sigma T_{\text{ef}}^4$
y emite $\sigma T_{\text{sup}}^4$. Para la capa parcial, sustitúyanse el
infrarrojo absorbido y el emitido por $\varepsilon\sigma T_{\text{sup}}^4$ y
$\varepsilon\sigma T_{\text{atm}}^4$ y déjese escapar directamente la fracción $1 - \varepsilon$
de la radiación de la superficie.
\end{proof}

\begin{remark}[Por qué es selectivo el invernadero]\label{rem:b2:thermal-radiation:selective}
La luz solar tiene su máximo en \qty{0.5}{\micro m} y la radiación terrestre en
\qty{10}{\micro m}: los dos espectros apenas se solapan, así que un gas puede ser
transparente al primero y absorbente para el segundo. El nitrógeno y el
oxígeno no absorben ninguno; el vapor de agua, el dióxido de carbono, el metano y el ozono
tienen bandas de vibración en el infrarrojo y son los gases de \omterm{prop:b2:thermal-radiation:greenhouse}{efecto invernadero}. La
ventana entre \qty{8}{} y \qty{13}{\micro m} es por donde mira la cámara térmica,
a lo que se refieren los \qty{250}{K} del cielo nocturno y lo que
aprovechan las pinturas de enfriamiento radiativo. El vidrio de un invernadero real hace
lo mismo (transparente a \qty{0.5}{\micro m}, opaco a \qty{10}{\micro m}),
pero sobre todo bloquea la convección --- el nombre es un poco injusto.
\end{remark}

\begin{method}[Estimaciones de radiación]\label{meth:b2:thermal-radiation:method}
(1) $\lambda_{\max} = \qty{2900}{\micro m.K}/T$ dice dónde está el espectro;
(2) $\varepsilon\sigma T^4$ por unidad de área; (3) intercambios: lo emitido
menos lo absorbido, linealizado como $h_{\text{r}} = 4\varepsilon\sigma T^3$ cerca de la
temperatura ambiente; (4) un planeta: $(1 - A)S/4 = \sigma T_{\text{ef}}^4$ y luego
las capas; (5) compárese con la conducción y la convección --- la radiación
gana a alta temperatura ($T^4$) y en el vacío; (6) cuidado con las \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividades}:
metales brillantes \qty{0.05}{}; pinturas, piel, agua y suelo, $\approx 0.9$--$0.98$
en el infrarrojo, sea cual sea su color visible.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:1}
$\lambda_{\max}$ y $\sigma T^4$ para: el Sol (\qty{5800}{K}), un ser humano
(\qty{310}{K}), el filamento de una bombilla (\qty{2800}{K}), el fondo cósmico
(\qty{2.73}{K}), el nitrógeno líquido (\qty{77}{K}) y un atizador al rojo
(\qty{900}{K}). ¿Cuáles son visibles?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:2}
A partir de $T_\odot = \qty{5772}{K}$, $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.496e11}{m}$:
la \omterm{prop:b2:thermal-radiation:earth}{constante solar}, la luminosidad del Sol, la potencia que intercepta la
Tierra ($R = \qty{6.37e6}{m}$) y esa misma repartida entre $8 \times 10^9$
personas. Compárese con el consumo mundial de potencia (\qty{2e13}{W}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:3}
Una persona: piel a \qty{306}{K}, $\varepsilon = 0.98$, \qty{1.7}{m^2}, en una
habitación a \qty{293}{K}. Potencia emitida, potencia absorbida y pérdida neta; el
coeficiente radiativo $h_{\text{r}}$; compárese con los \qty{100}{W} del
metabolismo y conclúyase sobre la ropa y sobre una habitación a \qty{20}{\celsius}
con paredes frías.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:4}
Una bombilla de \qty{60}{W}: filamento de wolframio a \qty{2800}{K}, $\varepsilon = 0.35$.
Área radiante; longitud de un hilo de \qty{50}{\micro m} con esa área;
cerca del $8\%$ de la potencia es visible: potencia luminosa; y un LED que dé
la misma luz con un rendimiento del $30\%$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:5}
\emph{Los dos límites.} (a) Desarróllese la $u_\nu$ de Planck para $h\nu \ll k_BT$ e
identifíquese la energía por modo. (b) Muéstrese que la ley de Rayleigh--Jeans
da una energía total infinita. (c) A \qty{300}{K}, la frecuencia en que
$h\nu = k_BT$; conclúyase que la \omterm{def:b2:thermal-radiation:blackbody}{radiación térmica} de radio y microondas es
clásica. (d) En el límite de Wien, interprétese $\eu^{-h\nu/k_BT}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:6}
\emph{Wien.} (a) Dedúzcase $x = 5(1 - \eu^{-x})$ y resuélvase numéricamente
con tres cifras. (b) Valor de $\lambda_{\max}T$. (c) Muéstrese que $u_\nu$
tiene su máximo en $h\nu = 2.82\,k_BT$ y que eso no corresponde a
$c/\lambda_{\max}$: ¿por qué no? (d) Temperatura de una estrella cuyo espectro
tiene el máximo en \qty{290}{nm}; su color a la vista.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:7}
\emph{Stefan.} (a) Realícese la integración de la ley de Planck, dado
$\int_0^\infty x^3\dd x/(\eu^x - 1) = \pi^4/15$, y el valor de $\sigma$.
(b) Justifíquese el factor $c/4$ entre la densidad y la exitancia. (c)
Dado $\int_0^\infty x^2\dd x/(\eu^x - 1) = 2.404$, el número de fotones
por unidad de volumen a $T$ y la energía media del fotón en unidades de $k_BT$.
(d) Fotones por centímetro cúbico a \qty{300}{K} y a \qty{2.73}{K}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:8}
\emph{Pantallas.} Dos placas negras grandes a $T_1$ y $T_2$. (a) Flujo neto.
(b) Se coloca entre ellas una lámina negra fina: su temperatura y el
nuevo flujo. (c) Con $n$ láminas. (d) Placas grises de \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividad} $\varepsilon$:
muéstrese que el flujo vale $\sigma(T_1^4 - T_2^4)/(2/\varepsilon - 1)$ y calcúlese
para superficies pulidas ($\varepsilon = 0.05$) a \qty{300}{K} y
\qty{77}{K} --- el termo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:9}
\emph{Escarcha nocturna.} Superficie con $\varepsilon = 0.95$, cielo a \qty{250}{K}, aire a
\qty{278}{K} con $h = \qty{5}{W/m^2/K}$. (a) Escríbase el balance y despéjese
la temperatura de la superficie. (b) Bajo una nube a \qty{275}{K}. (c) ¿Por qué
impide el viento la escarcha? (d) ¿Por qué se libra de ella un coche aparcado bajo un árbol?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:10}
\emph{Planetas.} (a) \omterm{prop:b2:thermal-radiation:earth}{Temperaturas efectivas} de Venus ($S = \qty{2600}{W/m^2}$,
$A = 0.75$), la Tierra y Marte ($S = \qty{590}{W/m^2}$, $A = 0.25$). (b) Sus
temperaturas superficiales son \qty{737}{K}, \qty{288}{K} y \qty{215}{K}:
calentamiento por invernadero de cada uno. (c) En el modelo de $n$ capas opacas,
muéstrese que $T_{\text{sup}}^4 = (n + 1)T_{\text{ef}}^4$; ¿cuántas capas para Venus?
(d) La Luna: $A = 0.12$, sin atmósfera y de rotación lenta --- temperatura
del punto subsolar, y por qué la cara nocturna cae a \qty{100}{K}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:11}
\emph{Sensibilidad climática sin realimentaciones.} (a) Linealícese el
balance planetario: un pequeño forzamiento adicional $\Delta F$ (en \unit{W/m^2}) en
lo alto de la atmósfera eleva la \omterm{prop:b2:thermal-radiation:earth}{temperatura efectiva} en $\Delta T
= \Delta F/4\sigma T_{\text{ef}}^3$. Calcúlese $4\sigma T_{\text{ef}}^3$. (b)
Doblar el dióxido de carbono es un forzamiento de unos \qty{3.7}{W/m^2}:
$\Delta T$ en este modelo. (c) La capa de mezcla del océano (\qty{50}{m} de
agua) almacena el calor: constante de tiempo de la respuesta. (d) ¿Por qué hacen
las realimentaciones del vapor de agua y del albedo del hielo que la sensibilidad real sea mayor
que la de (b)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:12}
\emph{Pirometría.} (a) En el límite de Wien, muéstrese que el cociente de las
radiancias en dos longitudes de onda, $\lambda_1$ y $\lambda_2$, da $T$
independientemente de una \omterm{prop:b2:thermal-radiation:kirchhoff}{emisividad} gris. (b) Una barra de acero: $M_{0.65}/M_{0.9}
= 0.10$: temperatura. (c) Una cámara térmica a \qty{10}{\micro m} supone
$\varepsilon = 0.95$; mira aluminio pulido ($\varepsilon = 0.1$)
a \qty{350}{K} en una habitación a \qty{293}{K}: en régimen linealizado,
¿qué temperatura indica? (d) ¿Por qué una superficie pintada junto
al aluminio, a los mismos \qty{350}{K}, se lee correctamente?
\end{exercise}

\section{Problema: el balance energético de la Tierra y la bombilla incandescente}

\begin{problem}[{Problema de fin de semana --- cuerpos negros del Sol al
filamento}]\label{pb:b2:thermal-radiation:1}
Datos: $\sigma = \qty{5.67e-8}{W/m^2/K^4}$, $\lambda_{\max}T = \qty{2.90e-3}{m.K}$,
$h = \qty{6.63e-34}{J.s}$, $k_B = \qty{1.38e-23}{J/K}$, $c = \qty{3.00e8}{m/s}$;
Sol: $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.50e11}{m}$; Tierra: $R =
\qty{6.37e6}{m}$, albedo $0.30$.

\textbf{Parte I --- El Sol como \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro}.}
\begin{enumerate}
  \item El espectro solar tiene su máximo en \qty{0.50}{\micro m}: temperatura
        superficial.
  \item Exitancia de la superficie y luminosidad total.
  \item \omterm{prop:b2:thermal-radiation:earth}{Constante solar} en la Tierra; densidad de energía y presión de
        la luz solar allí.
  \item Energía media del fotón ($\approx 2.7\,k_BT$) y número de fotones
        solares por segundo y por metro cuadrado en la Tierra.
  \item El Sol lleva radiando a este ritmo $4.5 \times 10^9$ años:
        energía emitida; compárese con $Mc^2$ para $M = \qty{2e30}{kg}$.
\end{enumerate}

\textbf{Parte II --- La Tierra sin atmósfera.}
\begin{enumerate}[resume]
  \item Potencia absorbida por la Tierra; \omterm{prop:b2:thermal-radiation:earth}{temperatura efectiva}.
  \item ¿Por qué divide el balance entre $4$? ¿Cuál sería la
        temperatura en el punto subsolar de una Tierra sin rotación ni
        aire ($A = 0.30$)?
  \item La Luna ($A = 0.12$): temperatura subsolar; y por qué su cara
        nocturna baja a \qty{100}{K} (piénsese en el calor almacenado en el
        regolito durante el día de dos semanas, \qty{1e6}{J/m^2/K} efectivos).
  \item Longitud de onda del máximo de emisión de la Tierra; compárese con la
        del Sol.
  \item Si más hielo o más nubes elevaran el albedo a $0.35$, ¿cuánto
        bajaría la \omterm{prop:b2:thermal-radiation:earth}{temperatura efectiva}?
\end{enumerate}

\textbf{Parte III --- El invernadero.}
\begin{enumerate}[resume]
  \item Escríbanse los tres balances (espacio, capa, superficie) del
        modelo de una capa y obténgase $T_{\text{sup}} = 2^{1/4}T_{\text{ef}}$.
  \item Con una capa que absorbe la fracción $\varepsilon$ del
        infrarrojo: $T_{\text{sup}}(\varepsilon)$; el valor de $\varepsilon$
        que da \qty{288}{K}.
  \item Qué gases absorben, cuáles no y en qué banda de longitudes
        de onda; ¿qué es la ventana atmosférica?
  \item El infrarrojo que la atmósfera devuelve al suelo,
        en \unit{W/m^2}, en el modelo con $\varepsilon = 0.78$; compárese con
        la luz solar absorbida.
  \item Un forzamiento de \qty{3.7}{W/m^2} (dióxido de carbono duplicado):
        cambio de temperatura $\Delta T = \Delta F/4\sigma T_{\text{ef}}^3$
        sin realimentaciones.
  \item Capacidad calorífica de una capa de mezcla oceánica de \qty{50}{m} por metro
        cuadrado y constante de tiempo de la respuesta.
  \item ¿Por qué enfrían las nubes (albedo) y calientan a la vez (infrarrojo) la
        superficie? ¿Qué efecto gana de noche?
\end{enumerate}

\textbf{Parte IV --- La bombilla incandescente.}
Una bombilla de \qty{60}{W} tiene un filamento de wolframio a \qty{2800}{K},
$\varepsilon = 0.35$; la fracción de la potencia de un \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} que cae en
el visible ($0.4$--\qty{0.7}{\micro m}) vale $0.08$ a \qty{2800}{K} y $0.14$
a \qty{3200}{K}.
\begin{enumerate}[resume]
  \item Área radiante del filamento; longitud de un hilo de \qty{50}{\micro m}
        que la tenga; ¿por qué está arrollado?
  \item Longitud de onda del máximo; potencia visible; rendimiento luminoso.
  \item Resistencia del filamento en funcionamiento a \qty{230}{V}; la
        resistividad del wolframio es unas diez veces menor a temperatura
        ambiente: corriente al encender y una consecuencia.
  \item A \qty{3200}{K}: área necesaria para los mismos \qty{60}{W} y
        potencia visible; ¿por qué no hacer funcionar toda bombilla a \qty{3200}{K}
        (el ritmo de evaporación del wolframio se duplica aproximadamente cada
        \qty{50}{K})?
  \item ¿Qué hace el ciclo del halógeno y por qué permite un filamento más
        caliente?
  \item ¿A dónde va el resto de los \qty{60}{W} y por qué alcanza el
        vidrio de la bombilla \qty{150}{\celsius}?
  \item Tiempo de enfriamiento del filamento tras apagarlo: estímese a partir de
        su masa (\qty{20}{mg}, $c = \qty{140}{J/kg/K}$) y de su potencia
        radiada.
  \item Resúmase: qué fija la temperatura de un \omterm{def:b2:thermal-radiation:blackbody}{cuerpo negro} y qué
        tienen en común el filamento, el Sol y la Tierra.
\end{enumerate}
\end{problem}
