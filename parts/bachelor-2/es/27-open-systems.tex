\chapter{Balances termodinámicos y sistemas abiertos}\label{ch:b2:open-systems}

La termodinámica del volumen del primer año trataba sistemas cerrados ---
una masa fija de gas en un cilindro, comprimida, calentada, expandida. Pero
las máquinas que de verdad calientan nuestras casas, enfrían nuestra comida y
producen nuestra electricidad están atravesadas por un \emph{flujo}: el vapor
recorre la turbina de una central a cientos de kilogramos por segundo, un
refrigerante circula por el compresor, el condensador, la válvula y el
evaporador de una bomba de calor, y el aire pasa por el compresor y la
cámara de combustión de un reactor. Cada componente es un \emph{\omterm{def:b2:open-systems:cv}{sistema
abierto}}, una región del espacio con materia que entra y sale, y las leyes
primera y segunda han de reescribirse para él. La reescritura es breve
--- la clave es que el fluido empujado hacia dentro del sistema lleva, además de
su energía interna, el trabajo hecho para empujarlo, y por eso es la
\emph{\omterm{thm:b2:open-systems:firstlaw}{entalpía}}, y no la energía interna, la energía natural de un flujo.
Este capítulo establece los balances de masa, energía y entropía para
\omterm{def:b2:open-systems:cv}{sistemas abiertos} en \omterm{def:b2:open-systems:cv}{régimen estacionario}, los aplica a los dispositivos elementales ---
\omterm{prop:b2:open-systems:devices}{tobera}, \omterm{prop:b2:open-systems:devices}{estrangulamiento}, compresor, turbina, \omterm{prop:b2:open-systems:devices}{intercambiador de calor} --- y monta
esos dispositivos en los dos ciclos que dimensiona el problema de fin de semana: una
bomba de calor y una central térmica de vapor.

\section{Sistemas abiertos y balance de masa}

\begin{definition}[Sistema abierto, volumen de control, régimen estacionario]\label{def:b2:open-systems:cv}
Un \emph{sistema abierto}\index{sistema abierto} es una región fija del espacio,
el \emph{volumen de control}\index{volumen de control} $\Sigma$, limitado por una
superficie por la que el fluido entra y sale (secciones de entrada y de salida)
y a través de la cual pueden intercambiarse calor y trabajo con el
exterior; el trabajo recibido por partes móviles (un eje, un pistón)
distintas del propio flujo es el \emph{trabajo útil}\index{trabajo útil}
$W_{\text{u}}$ (trabajo en el eje). El flujo es \emph{estacionario}\index{régimen
estacionario} cuando todos los campos dentro de $\Sigma$ son independientes del tiempo: la masa,
la energía y la entropía contenidas en $\Sigma$ son entonces constantes.
\end{definition}

\begin{proposition}[Balance de masa]\label{prop:b2:open-systems:mass}
Por una sección de área $S$ donde el fluido de densidad $\rho$ se mueve
a la celeridad media $c$ normal a ella pasa el \emph{caudal
másico}\index{caudal másico} $\dot m = \rho c S$ (en \unit{kg/s}). En
\omterm{def:b2:open-systems:cv}{régimen estacionario}, lo que entra sale: $\sum_{\text{ent}}\dot m = \sum_{\text{sal}}
\dot m$; para una sola corriente, $\dot m$ es el mismo a la entrada y a la salida,
$\rho_1c_1S_1 = \rho_2c_2S_2$ --- la \omterm{thm:b2:fluid-kinematics:continuity}{ecuación de continuidad} de
\cref{ch:b2:fluid-kinematics} integrada sobre la sección.
\end{proposition}

\begin{proof}
La masa contenida en $\Sigma$ es constante, así que los flujos se compensan.
\end{proof}

\section{El primer principio para un sistema abierto}

\begin{theorem}[Balance de energía de un flujo estacionario de una corriente]\label{thm:b2:open-systems:firstlaw}
Para un flujo \omterm{def:b2:open-systems:cv}{estacionario} de \omterm{def:b2:fluid-kinematics:flowrate}{caudal másico} $\dot m$ que entra en el estado 1
y sale en el estado 2, recibiendo la potencia útil (en el eje)
$P_{\text{u}}$ y la potencia térmica $P_{\text{term}}$,
\[
\dot m\,\Big[(h_2 - h_1) + \tfrac12(c_2^2 - c_1^2) + g(z_2 - z_1)\Big]
= P_{\text{u}} + P_{\text{term}} ,
\]
o bien, por unidad de masa de fluido que cruza el sistema,
\[
\Delta h + \Delta\big(\tfrac12c^2\big) + g\,\Delta z = w_{\text{u}} + q .
\]
La \emph{entalpía}\index{entalpía (en un flujo)} $h = u + Pv$ sustituye a
la energía interna: $Pv$ es el \emph{trabajo de flujo}, el trabajo que hace el
fluido de aguas arriba para meter una unidad de masa, menos el que hace para
sacarla.
\end{theorem}

\begin{proof}
Sígase el sistema cerrado formado por el fluido que hay dentro de $\Sigma$ en $t$ más
la porción $\dd m = \dot m\,\dd t$ a punto de entrar; en $t + \dd t$ consta
del fluido de dentro de $\Sigma$ (en el mismo estado, por ser \omterm{def:b2:open-systems:cv}{estacionario}) más la
porción $\dd m$ que ha salido. Su energía cambió en
$\dd m\,[(u_2 + \tfrac12c_2^2 + gz_2) - (u_1 + \tfrac12c_1^2 + gz_1)]$. Recibió
$P_{\text{u}}\dd t + P_{\text{term}}\dd t$, más el trabajo de las
fuerzas de presión en la entrada, $P_1S_1 \times c_1\dd t = P_1v_1\,\dd m$
(que mete la porción), menos el de la salida, $P_2v_2\,\dd m$. El
primer principio para este sistema cerrado, con $u + Pv = h$, da el resultado.
\end{proof}

\begin{remark}[Cómo usar el primer principio]\label{rem:b2:open-systems:using}
Para un gas ideal, $\Delta h = c_p\Delta T$; para un líquido, $\Delta h \approx
c\,\Delta T + v\,\Delta P$ (el segundo término es el trabajo de la bomba); para un vapor
se lee $h$ en tablas o diagramas. El término cinético suele ser
despreciable: a \qty{10}{m/s}, $c^2/2 = \qty{50}{J/kg}$, frente a
\qty{1}{kJ/kg} por cada kelvin de aire --- solo cuenta en \omterm{prop:b2:open-systems:devices}{toberas} y
álabes de turbina, donde las celeridades llegan a cientos de metros por segundo. El
término potencial, $g\Delta z \approx \qty{10}{J/kg}$ por metro, cuenta para el
agua de una presa, nunca en un ciclo de gas. Con varias entradas y salidas
el balance se escribe $\sum_{\text{sal}}\dot m\,h - \sum_{\text{ent}}\dot m\,h =
P_{\text{u}} + P_{\text{term}}$ (despreciados los términos cinético y potencial).
\end{remark}

\section{El segundo principio para un sistema abierto}

\begin{theorem}[Balance de entropía de un flujo estacionario]\label{thm:b2:open-systems:secondlaw}
Para un flujo \omterm{def:b2:open-systems:cv}{estacionario} de una sola corriente que intercambia las potencias térmicas
$P_{\text{term},i}$ con focos a las temperaturas $T_i$,
\[
\dot m\,(s_2 - s_1) = \sum_i\frac{P_{\text{term},i}}{T_i} + \dot S_{\text{c}},
\qquad \dot S_{\text{c}} \ge 0 ;
\]
por unidad de masa, $\Delta s = s_{\text{int}} + s_{\text{c}}$. Un flujo adiabático
y reversible es \emph{isentrópico}; un flujo adiabático real tiene $s_2 >
s_1$. El \emph{rendimiento isentrópico}\index{rendimiento isentrópico} de
una turbina vale $\eta_{\text{T}} = w_{\text{real}}/w_{\text{s}} = (h_1 - h_2)/(h_1
- h_{2s})$ y el de un compresor $\eta_{\text{C}} = (h_{2s} - h_1)/(h_2 -
h_1)$, donde $2s$ es el estado alcanzado isentrópicamente a la misma
presión de salida; ambos valen típicamente de $0.8$ a $0.9$. El trabajo perdido por
irreversibilidad, referido a la temperatura ambiente $T_0$, es
$T_0\,s_{\text{c}}$ por unidad de masa.
\end{theorem}

\begin{proof}
El mismo sistema cerrado que antes: su entropía cambia en $\dd m\,(s_2 -
s_1)$, recibe $\sum P_{\text{term},i}\dd t/T_i$ de los focos y el
resto se crea.
\end{proof}

\section{Dispositivos elementales}

\begin{proposition}[Tobera, estrangulamiento, compresor, turbina, intercambiador]\label{prop:b2:open-systems:devices}
Todos adiabáticos salvo indicación en contra, \omterm{def:b2:open-systems:cv}{estacionarios} y con los términos cinético y potencial
despreciados salvo cuando son lo esencial.
\begin{itemize}
\item \emph{Tobera}\index{tobera} (sin eje, sin calor): $\tfrac12c_2^2 -
      \tfrac12c_1^2 = h_1 - h_2$ --- la \omterm{thm:b2:open-systems:firstlaw}{entalpía} se vuelve celeridad; un
      \emph{difusor} hace lo contrario.
\item \emph{Estrangulamiento}\index{estrangulamiento} (válvula, tapón poroso; sin eje,
      sin calor, con celeridades pequeñas): $h_2 = h_1$, \emph{isentálpico}. Para un
      gas ideal, $T_2 = T_1$; para un gas real la temperatura suele
      bajar (efecto Joule--Thomson); para un líquido cerca de la saturación una
      fracción se evapora de golpe --- la válvula de expansión de todo
      frigorífico.
\item \emph{Compresor, bomba, turbina}\index{turbina} (máquinas
      adiabáticas): $w_{\text{u}} = h_2 - h_1$ --- positivo para un compresor y
      negativo para una turbina; para una bomba de líquido, $w_{\text{u}} \approx
      v(P_2 - P_1)$, minúsculo porque $v$ es pequeño.
\item \emph{Intercambiador de calor}\index{intercambiador de calor} (dos corrientes, sin
      eje, adiabático en conjunto): $\dot m_{\text{a}}(h_{\text{a},2} -
      h_{\text{a},1}) = -\dot m_{\text{b}}(h_{\text{b},2} - h_{\text{b},1})$
      --- lo que una pierde lo gana la otra; el intercambio crea entropía
      salvo que las corrientes estén a la misma temperatura en todas partes
      (la contracorriente la reduce).
\item \emph{Cámara de mezcla}: $\sum\dot m_{\text{ent}}h_{\text{ent}} = \dot
      m_{\text{sal}}h_{\text{sal}}$.
\end{itemize}
\end{proposition}

\begin{proof}
Cada uno es el primer principio con los términos oportunos anulados.
\end{proof}

\begin{example}[Números para los dispositivos]\label{ex:b2:open-systems:numbers}
Vapor que entra en la \omterm{prop:b2:open-systems:devices}{tobera} de una turbina con $h = \qty{3000}{kJ/kg}$ y sale
con \qty{2800}{kJ/kg} alcanza $c = \sqrt{2 \times 200 \times 10^3} =
\qty{630}{m/s}$. Aire comprimido adiabáticamente de \qty{1}{bar} y
\qty{293}{K} a \qty{8}{bar}: isentrópicamente, $T_2 = 293 \times 8^{0.286} =
\qty{531}{K}$ y $w = c_p\Delta T = \qty{239}{kJ/kg}$; con $\eta_{\text{C}} =
0.8$, \qty{299}{kJ/kg} y \qty{590}{K} --- por eso los compresores se
refrigeran entre etapas. Un radiador de coche que evacúa \qty{42}{kW} de agua
enfriada \qty{10}{K} (\qty{1}{kg/s}) calienta \qty{2.1}{kg/s} de aire
\qty{20}{K}. Un refrigerante líquido a \qty{40}{\celsius} estrangulado hasta
\qty{0}{\celsius} llega como una niebla con un $28\%$ de vapor: esa evaporación súbita
es la que enfría el líquido hasta la temperatura del evaporador.
\end{example}

\section{Ciclos}

\begin{proposition}[Refrigeración por compresión de vapor y bomba de calor]\label{prop:b2:open-systems:fridge}
Un refrigerante circula por: (1$\to$2) un compresor adiabático
que eleva el vapor de la presión del evaporador a la presión del
condensador, $w = h_2 - h_1$; (2$\to$3) un condensador donde cede
$q_{\text{cond}} = h_2 - h_3$ al lado caliente (la habitación para una bomba de calor, la
cocina para un frigorífico) y sale como líquido; (3$\to$4) un \omterm{prop:b2:open-systems:devices}{estrangulamiento}, $h_4
= h_3$; (4$\to$1) un evaporador donde absorbe $q_{\text{evap}} = h_1 - h_4$
del lado frío. Los coeficientes de eficiencia valen
\[
\mathrm{COP}_{\text{fr}} = \frac{h_1 - h_4}{h_2 - h_1}, \qquad
\mathrm{COP}_{\text{cal}} = \frac{h_2 - h_3}{h_2 - h_1} = \mathrm{COP}_{\text{fr}} + 1 ,
\]
acotados por los de Carnot, $T_{\text{fr}}/(T_{\text{cal}} - T_{\text{fr}})$ y
$T_{\text{cal}}/(T_{\text{cal}} - T_{\text{fr}})$; el ciclo se lee con más facilidad en
el diagrama $(\log P, h)$, donde los dos intercambios y el trabajo son
segmentos horizontales.
\end{proposition}

\begin{proof}
Los cuatro balances del primer principio; el balance de energía de todo el ciclo
da $q_{\text{cond}} = q_{\text{evap}} + w$, de donde la relación entre
los dos COP.
\end{proof}

\begin{proposition}[El ciclo de vapor (de Rankine)]\label{prop:b2:open-systems:rankine}
El agua circula por: (1$\to$2) una bomba, $w_{\text{b}} \approx v(P_2 -
P_1)$, unos pocos \unit{kJ/kg}; (2$\to$3) una caldera a alta presión donde se
calienta, se vaporiza y se sobrecalienta, $q_{\text{ent}} = h_3 - h_2$;
(3$\to$4) una turbina que expande el vapor hasta la presión del condensador,
$w_{\text{t}} = h_3 - h_4$ (un millar de \unit{kJ/kg}); (4$\to$1) un condensador
a baja presión donde el vapor húmedo vuelve a ser líquido y cede
$q_{\text{sal}} = h_4 - h_1$ al agua de refrigeración. El rendimiento vale
\[
\eta = \frac{w_{\text{t}} - w_{\text{b}}}{q_{\text{ent}}} = 1 - \frac{q_{\text{sal}}}{q_{\text{ent}}} ,
\]
de $0.35$ a $0.45$ en la práctica; el calor se aporta a una temperatura
media $\bar T = q_{\text{ent}}/(s_3 - s_2)$ muy por debajo del máximo de la
caldera, y por eso el sobrecalentamiento, el recalentamiento intermedio y las presiones altas suben
$\eta$, y por eso el escape debe quedar bastante seco ($x_4 \gtrsim 0.88$) para
librar a los últimos álabes de la erosión por gotas.
\end{proposition}

\begin{proof}
Primer principio en cada componente y luego en el ciclo: $w_{\text{t}} - w_{\text{b}}
= q_{\text{ent}} - q_{\text{sal}}$.
\end{proof}

\begin{remark}[Otros ciclos, y un transitorio]\label{rem:b2:open-systems:other}
La turbina de gas (compresor, cámara de combustión, turbina: el ciclo de
Brayton) tiene el rendimiento ideal $1 - r^{-(\gamma-1)/\gamma}$ para la
relación de presiones $r$ y expulsa sus gases a varios cientos de grados;
llevar esos gases a un ciclo de vapor (ciclo combinado) alcanza el $60\%$.
Los \omterm{def:b2:open-systems:cv}{sistemas abiertos} tienen también transitorios: llenar un depósito rígido vacío desde
una línea a $T_0$ termina en $T = \gamma T_0$ (el trabajo de flujo $Pv$ se convierte en
energía interna), \qty{120}{K} más caliente para el aire --- la razón de que una
botella de buceo se caliente al llenarse.
\end{remark}

\begin{method}[Balances de sistemas abiertos]\label{meth:b2:open-systems:method}
(1) Dibújese el \omterm{def:b2:open-systems:cv}{volumen de control}; enumérense entradas, salidas, eje y calor.
(2) Masa: $\sum\dot m_{\text{ent}} = \sum\dot m_{\text{sal}}$. (3) Energía
por unidad de masa: $\Delta h + \Delta c^2/2 + g\Delta z = w_{\text{u}} + q$, con
$\Delta h = c_p\Delta T$ para gases, tablas para vapores y $v\Delta P$ para
bombas. (4) Entropía: $\Delta s = \sum q_i/T_i + s_{\text{c}}$, estados
isentrópicos de referencia, rendimientos y $T_0s_{\text{c}}$ perdido. (5) Ciclos: léase
el diagrama, súmese a lo largo del lazo y compárese con Carnot.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:open-systems:1}
(a) Una manguera de jardín entrega \qty{12}{L/min} por una \omterm{prop:b2:open-systems:devices}{tobera} de \qty{1}{cm}:
celeridad de salida. (b) Agua caliente a \qty{60}{\celsius}, \qty{0.1}{kg/s},
se mezcla con agua fría a \qty{10}{\celsius}, \qty{0.2}{kg/s}: temperatura
de salida. (c) ¿Por qué es irreversible la mezcla? Calcúlese la
entropía creada por segundo.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:open-systems:2}
El vapor entra en una \omterm{prop:b2:open-systems:devices}{tobera} a \qty{50}{m/s} con $h = \qty{3000}{kJ/kg}$ y
sale con \qty{2800}{kJ/kg}: celeridad de salida. Aire a \qty{300}{K} acelerado
desde el reposo hasta \qty{300}{m/s} en una \omterm{prop:b2:open-systems:devices}{tobera} ($c_p = \qty{1005}{J/kg/K}$):
caída de temperatura. Un difusor frena aire de \qty{250}{m/s} al reposo: subida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:open-systems:3}
\omterm{prop:b2:open-systems:devices}{Estrangulamiento}. (a) Un gas ideal: ¿qué le ocurre a $T$ y a $s$? (b) Vapor húmedo
a \qty{10}{bar} ($h_{\text{f}} = 763$, $h_{\text{fg}} = \qty{2015}{kJ/kg}$),
de título $0.9$, estrangulado a \qty{1}{bar} ($h_{\text{f}} = 417$, $h_{\text{fg}}
= \qty{2258}{kJ/kg}$): título final --- el calorímetro de \omterm{prop:b2:open-systems:devices}{estrangulamiento}. (c)
Refrigerante líquido a \qty{40}{\celsius} ($h = \qty{256}{kJ/kg}$) estrangulado
a \qty{0}{\celsius} ($h_{\text{f}} = 200$, $h_{\text{g}} = \qty{399}{kJ/kg}$):
fracción de vapor.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:open-systems:4}
(a) Un radiador enfría \qty{1}{kg/s} de agua de \qty{90}{} a \qty{80}{\celsius};
el aire ($c_p = \qty{1005}{J/kg/K}$) se calienta de \qty{20}{} a \qty{40}{\celsius}:
\omterm{def:b2:fluid-kinematics:flowrate}{caudal másico} y \omterm{def:b2:fluid-kinematics:flowrate}{caudal volumétrico} de aire. (b) Una turbina de vapor, \qty{10}{kg/s},
con $h$ de \qty{3400}{} a \qty{2400}{kJ/kg}: potencia. (c) Una bomba eleva el
agua de \qty{1}{} a \qty{60}{bar}: trabajo por kilogramo y potencia para
\qty{500}{kg/s}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:5}
\emph{El trabajo de flujo.} (a) Dedúzcase de nuevo el primer principio para \omterm{def:b2:open-systems:cv}{sistemas abiertos} y
explíquese de dónde sale $Pv$. (b) ¿Por qué aparece la energía interna $u$
en un sistema cerrado y $h$ en un flujo? (c) Compárense $c^2/2$ a
\qty{10}{m/s} y a \qty{300}{m/s} con $c_p\Delta T$ para \qty{1}{K} de aire.
(d) Agua que cae \qty{100}{m} por una tubería forzada: $g\Delta z$ frente a
$c\,\Delta T$ para \qty{1}{K}; conclúyase cuánto se calienta el agua si se
quita la turbina.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:6}
\emph{Compresores.} Aire, \qty{1}{bar}, \qty{293}{K}, hasta \qty{8}{bar}; $\gamma =
1.4$, $c_p = \qty{1005}{J/kg/K}$, $r = \qty{287}{J/kg/K}$. (a) Temperatura y
trabajo de salida isentrópicos. (b) Con $\eta_{\text{C}} = 0.8$: trabajo, temperatura
de salida y entropía creada. (c) Dos etapas de relación $\sqrt8$ con
enfriamiento de vuelta a \qty{293}{K} entre ellas: trabajo ahorrado. (d) Límite
isotermo $rT\ln 8$; por qué los compresores reales se le acercan con muchas etapas.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:7}
\emph{Turbina.} Vapor a \qty{60}{bar} y \qty{500}{\celsius}: $h_3 = \qty{3422}{kJ/kg}$,
$s_3 = \qty{6.88}{kJ/kg/K}$; condensador a \qty{0.05}{bar} (\qty{33}{\celsius}):
$s_{\text{f}} = 0.476$, $s_{\text{g}} = \qty{8.394}{kJ/kg/K}$, $h_{\text{f}} = 138$,
$h_{\text{fg}} = \qty{2423}{kJ/kg}$. (a) Título y \omterm{thm:b2:open-systems:firstlaw}{entalpía} de salida isentrópicos, y
trabajo. (b) Con $\eta_{\text{T}} = 0.85$: trabajo, \omterm{thm:b2:open-systems:firstlaw}{entalpía} y título
de salida. (c) Entropía creada por kilogramo y trabajo perdido a $T_0 =
\qty{306}{K}$. (d) Por qué es un problema un escape demasiado húmedo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:8}
\emph{Bomba de calor.} Evaporador a \qty{0}{\celsius}, condensador a \qty{40}{\celsius};
$h_1 = 399$ (vapor saturado a \qty{0}{\celsius}), $h_{2s} = 426$, $h_3 = 256$,
$h_4 = h_3$ (\unit{kJ/kg}); $\eta_{\text{C}} = 0.8$. (a) $h_2$ y el trabajo.
(b) Calores intercambiados y ambos COP. (c) COP de Carnot; cociente. (d) \omterm{def:b2:fluid-kinematics:flowrate}{Caudal másico}
y potencia eléctrica para \qty{8}{kW} de calefacción; por qué cae el COP cuando
el aire exterior se enfría.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:9}
\emph{Entropía en un intercambiador.} Agua caliente, \qty{1}{kg/s}, de \qty{90}{} a
\qty{50}{\celsius}, calienta agua fría, \qty{2}{kg/s}, desde \qty{10}{\celsius}.
(a) Temperatura de salida de la corriente fría. (b) Entropía creada por
segundo ($c = \qty{4180}{J/kg/K}$). (c) Potencia perdida a $T_0 = \qty{293}{K}$,
comparada con los \qty{167}{kW} intercambiados. (d) ¿Por qué crearía más entropía una
disposición en equicorriente que en contracorriente para el mismo servicio?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:open-systems:10}
\emph{Llenar un depósito.} Un depósito rígido, vacío y aislado se conecta
a una línea que lleva un gas ideal a $T_0$, $P_0$. (a) Escríbase el balance de energía
del depósito durante el llenado (no \omterm{def:b2:open-systems:cv}{estacionario}: su energía crece) y
muéstrese que el gas del interior acaba a $T = \gamma T_0$. (b) Aire a \qty{293}{K}:
temperatura final. (c) Si el depósito contenía inicialmente gas a $T_0$ y
$P_1 < P_0$, arguméntese que la temperatura final queda entre $T_0$ y
$\gamma T_0$. (d) ¿Por qué se calienta una botella de buceo al llenarse, y por qué
se llena despacio dentro del agua?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:open-systems:11}
\emph{Recalentamiento intermedio.} La turbina de \cref{exo:b2:open-systems:7} se divide:
expansión hasta \qty{10}{bar} (final isentrópico: $h = \qty{2850}{kJ/kg}$), recalentamiento
hasta \qty{500}{\celsius} ($h = 3478$, $s = \qty{7.76}{kJ/kg/K}$) y expansión hasta
\qty{0.05}{bar}, con ambas etapas isentrópicas. Trabajo de la bomba \qty{6}{kJ/kg}, $h_2 =
\qty{144}{kJ/kg}$. (a) Título y \omterm{thm:b2:open-systems:firstlaw}{entalpía} de salida de la segunda etapa. (b)
Trabajo y calor totales; rendimiento; compárese con la expansión única
($\eta = 0.40$ isentrópico). (c) ¿Por qué ayuda el recalentamiento, en términos de la
temperatura media de aporte de calor? (d) ¿Qué más arregla?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:open-systems:12}
\emph{Turbina de gas y ciclo combinado.} Aire comprimido isentrópicamente
de \qty{1}{bar} y \qty{293}{K} a \qty{12}{bar}, calentado a presión constante
hasta \qty{1500}{K} y expandido isentrópicamente hasta \qty{1}{bar}; $\gamma = 1.4$,
$c_p = \qty{1005}{J/kg/K}$. (a) Las cuatro temperaturas. (b) Trabajos,
calor y rendimiento; muéstrese que $\eta = 1 - r^{-(\gamma - 1)/\gamma}$. (c) Los
gases de escape a $T_4$ producen vapor para un ciclo de Rankine de rendimiento $0.35$
que recupera el $70\%$ del calor de los gases por encima de \qty{373}{K}: rendimiento
combinado. (d) ¿Por qué es el ciclo combinado la central térmica más eficiente
que existe?
\end{exercise}

\section{Problema: una bomba de calor doméstica y una central de vapor}

\begin{problem}[{Problema de fin de semana --- dos máquinas dimensionadas con los
balances de sistemas abiertos}]\label{pb:b2:open-systems:1}
\textbf{Parte I --- La bomba de calor.}
Datos del refrigerante (\unit{kJ/kg}, \unit{kJ/kg/K}): vapor saturado a
\qty{0}{\celsius} ($P = \qty{2.9}{bar}$): $h_1 = 399$, $s_1 = 1.727$;
líquido saturado a \qty{0}{\celsius}: $h_{\text{f}} = 200$, $s_{\text{f}} = 1.00$;
a la presión del condensador, \qty{10.2}{bar} ($T_{\text{sat}} = \qty{40}{\celsius}$):
final de compresión isentrópica $h_{2s} = 426$, líquido saturado $h_3 = 256$,
$s_3 = 1.19$; el vapor sobrecalentado tiene allí $c_p \approx \qty{1.0}{kJ/kg/K}$.
Compresor con $\eta_{\text{C}} = 0.8$; demanda de la casa \qty{8}{kW}.
\begin{enumerate}
  \item Dibújese el ciclo en un diagrama $(\log P, h)$ y nómbrense los cuatro
        componentes con el proceso que realiza cada uno.
  \item Estado 4 tras el \omterm{prop:b2:open-systems:devices}{estrangulamiento}: \omterm{thm:b2:open-systems:firstlaw}{entalpía} y fracción de vapor.
  \item Calor absorbido por kilogramo en el evaporador.
  \item Trabajo del compresor, isentrópico y real; $h_2$ y una estimación de
        $T_2$.
  \item Calor cedido en el condensador; ambos COP; compruébese su
        relación.
  \item COP de Carnot entre \qty{0}{} y \qty{40}{\celsius}; ¿qué
        fracción alcanza la máquina?
  \item \omterm{def:b2:fluid-kinematics:flowrate}{Caudal másico} de refrigerante y potencia eléctrica (rendimiento
        del motor $0.9$).
  \item Entropía creada por kilogramo en el compresor y en el
        \omterm{prop:b2:open-systems:devices}{estrangulamiento} ($s_4 = s_{\text{f}} + x\,(s_1 - s_{\text{f}})$); ¿cuál es
        peor?
\end{enumerate}

\textbf{Parte II --- El tiempo frío y las opciones de diseño.}
\begin{enumerate}[resume]
  \item Con \qty{-10}{\celsius} en el exterior, el evaporador trabaja a
        \qty{-15}{\celsius}: COP de Carnot para calefacción; ¿por qué cae más deprisa el
        COP real (piénsese en la relación de presiones y en la
        densidad del vapor)?
  \item ¿Por qué se forma escarcha en la batería exterior y qué cuesta el
        ciclo de desescarche?
  \item Suelo radiante a \qty{35}{\celsius} frente a radiadores a
        \qty{55}{\celsius}: COP de Carnot con el evaporador a
        \qty{0}{\celsius}; conclúyase.
  \item Compárense, por cada kilovatio hora de combustible quemado en una central de
        rendimiento $0.40$, el calor que entrega esta bomba de calor ($\mathrm{COP}
        = 4$) y el de una caldera de gas de rendimiento $0.9$ que quema el combustible
        directamente.
\end{enumerate}

\textbf{Parte III --- La central de vapor.}
Caldera a \qty{60}{bar} y \qty{500}{\celsius}: $h_3 = 3422$, $s_3 = 6.88$; a
\qty{60}{bar}, $T_{\text{sat}} = \qty{276}{\celsius}$, $h_{\text{f}} = 1213$,
$h_{\text{g}} = 2784$; condensador a \qty{0.05}{bar} y \qty{33}{\celsius}:
$h_{\text{f}} = 138$, $h_{\text{fg}} = 2423$, $s_{\text{f}} = 0.476$, $s_{\text{g}} =
8.394$; líquido con $v = \qty{1e-3}{m^3/kg}$; turbina con $\eta_{\text{T}} = 0.85$;
potencia eléctrica neta \qty{600}{MW}.
\begin{enumerate}[resume]
  \item Dibújese el ciclo en un diagrama $(T,s)$ y nómbrense los procesos.
  \item Trabajo de la bomba y $h_2$.
  \item Calor aportado en la caldera, repartido en precalentamiento, vaporización
        y sobrecalentamiento.
  \item Expansión isentrópica: título, \omterm{thm:b2:open-systems:firstlaw}{entalpía} y trabajo de salida.
  \item Expansión real: trabajo, \omterm{thm:b2:open-systems:firstlaw}{entalpía} y título de salida; por qué importa
        el título.
  \item Rendimiento del ciclo; Carnot entre \qty{773}{} y \qty{306}{K};
        la temperatura media de aporte de calor $\bar T = q_{\text{ent}}/(s_3
        - s_2)$ y el rendimiento de Carnot que daría.
  \item \omterm{def:b2:fluid-kinematics:flowrate}{Caudal másico} de vapor; calor cedido en el condensador; caudal de agua
        de refrigeración para una subida de \qty{10}{K}, o masa de agua que evapora
        por segundo una torre de refrigeración (\qty{2.4}{MJ/kg}).
  \item Entropía creada en la turbina por kilogramo y por segundo;
        el trabajo perdido a \qty{306}{K}, como fracción del trabajo de la
        turbina.
  \item Recalentar a \qty{10}{bar} hasta \qty{500}{\celsius} sube el
        rendimiento hasta cerca de $0.43$ y el título de salida hasta $0.92$:
        explíquense ambos efectos cualitativamente.
\end{enumerate}

\textbf{Parte IV --- La cadena completa.}
\begin{enumerate}[resume]
  \item El hogar transfiere su calor al vapor desde los gases de combustión a
        unos \qty{1500}{K}: entropía creada por segundo en esa transferencia
        (calor a \qty{1500}{K} hacia vapor a $\bar T$); compárese con la
        de la turbina.
  \item El condensador cede su calor al agua a \qty{306}{K} ---
        que acaba llegando al ambiente a \qty{293}{K}: entropía
        creada por segundo ahí.
  \item Ordénense las tres fuentes de irreversibilidad (hogar, turbina,
        condensador) y dígase a dónde va el esfuerzo de los ingenieros.
  \item Resúmase el método de los \omterm{def:b2:open-systems:cv}{sistemas abiertos} en cinco líneas.
\end{enumerate}
\end{problem}
