\chapter{La ecuación de Schrödinger y las funciones de onda}\label{ch:b2:schrodinger-wave-functions}

Dispárense electrones de uno en uno contra un par de rendijas y regístrese dónde cae
cada uno: un punto aquí, un punto allá, al parecer al azar --- y tras diez
mil puntos, las franjas del experimento de Young, como si cada electrón
hubiera pasado por las dos rendijas e interferido consigo mismo. El último
capítulo del volumen del primer año se encontró con esta rareza: la luz viene en
fotones, la materia tiene una longitud de onda $\lambda = h/p$, y la posición y el
momento lineal no pueden ser ambos nítidos. Lo que no dio es la
\emph{ecuación} --- la regla que dice cómo evoluciona la onda asociada a una
partícula, como la ley de Newton dice cómo evoluciona una trayectoria. Esa
ecuación es la de Schrödinger (1926), y con ella la descripción cuántica
se vuelve una teoría calculable: se escribe una \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{función de onda}}
$\psi(x,t)$, cuyo módulo al cuadrado es la probabilidad de encontrar la
partícula, y se resuelve una ecuación en derivadas parciales lineal. Este
capítulo enuncia la ecuación, explica qué significa $\psi$ y qué no,
halla sus \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios} y los \omterm{prop:b2:dispersion-wave-packets:group}{paquetes de ondas} de la partícula
libre, y muestra dónde reaparece la mecánica de Newton --- para
todo lo que pese más que una molécula.

\section{Lo que nos dejó el volumen del primer año}

\begin{proposition}[Las partículas son ondas]\label{prop:b2:schrodinger-wave-functions:recap}
Un fotón de frecuencia $\nu$ tiene la energía $E = h\nu = \hbar\omega$ y el
momento lineal $p = h/\lambda = \hbar k$; una partícula material de momento $p$
se comporta como una onda de longitud de onda $\lambda = h/p$ (de Broglie), con $E =
\hbar\omega$ y $p = \hbar k$ también --- la \omterm{def:b2:diffraction:huygens}{difracción} de electrones por
cristales y la interferometría de neutrones e incluso de moléculas lo confirman; y ningún
estado puede tener una posición y un momento más nítidos que $\Delta x\,\Delta p
\ge \hbar/2$ (Heisenberg). Con $\hbar = h/2\pi = \qty{1.05e-34}{J.s}$: un
electrón de \qty{100}{eV} tiene $\lambda = \qty{0.12}{nm}$, el tamaño de un
átomo; un grano de polvo de \qty{1e-15}{kg} a \qty{1}{mm/s} tiene $\lambda =
\qty{7e-16}{m}$, menor que un núcleo --- se comporta clásicamente.
\end{proposition}

\begin{proof}
Recordado del volumen del primer año; la tarea ahora es la dinámica de la
onda.
\end{proof}

\section{La función de onda}

\begin{definition}[Función de onda y regla de Born]\label{def:b2:schrodinger-wave-functions:psi}
El estado de una partícula que se mueve según $x$ se describe con una
\emph{función de onda}\index{función de onda} compleja $\psi(x,t)$ tal que
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
es la probabilidad de encontrar la partícula entre $x$ y $x + \dd x$
en el instante $t$ (\emph{regla de Born}\index{regla de Born}); de ahí la
\emph{normalización} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, y los
valores medios $\langle x\rangle = \int x|\psi|^2\dd x$, $\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$. Las funciones de onda obedecen al
\emph{principio de superposición}\index{principio de superposición}: si
$\psi_1$ y $\psi_2$ son estados posibles, también lo es $\alpha\psi_1 +
\beta\psi_2$ (normalizado), y las probabilidades contienen entonces el
término de interferencia $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$. Un
factor de fase global $\eu^{\iu\theta}$ no cambia nada; una fase relativa
entre dos componentes superpuestas cambia la interferencia.
\end{definition}

\begin{remark}[Lo que $\psi$ no es]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ no es un campo clásico repartido por el espacio como $\vect E$, ni una
nube de materia: el electrón se encuentra siempre entero, en un punto.
$\psi$ es una \emph{amplitud de probabilidad}, el objeto que
interfiere; solo se observa $|\psi|^2$, y solo de forma estadística,
repitiendo el experimento con partículas preparadas de forma idéntica. Los puntos
de la doble rendija son electrones sueltos; las franjas son $|\psi_1 +
\psi_2|^2$.
\end{remark}

\section{La ecuación de Schrödinger}

\begin{theorem}[Ecuación de Schrödinger]\label{thm:b2:schrodinger-wave-functions:equation}
Una partícula de masa $m$ en la energía potencial $V(x)$ tiene una \omterm{def:b2:schrodinger-wave-functions:psi}{función
de onda} que obedece a
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
Es lineal (superposición), de primer orden en el tiempo (la \omterm{def:b2:schrodinger-wave-functions:psi}{función de onda}
en un instante la determina en todos los posteriores) y compleja ($\psi$
no puede ser real: la $\iu$ es esencial). Es un \emph{postulado},
justificado por sus consecuencias.
\end{theorem}

\begin{proof}
Más una motivación que una demostración. Una partícula libre de momento nítido
$p = \hbar k$ y energía $E = \hbar\omega$ debería ser una onda plana $\psi =
A\eu^{\iu(kx - \omega t)}$, para la que $\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ y $-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$.
La relación clásica $E = p^2/2m$ queda así reproducida por la ecuación
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$, y la generalización
natural para $E = p^2/2m + V$ añade $V\psi$. Nótese que la
ecuación debe ser lineal para valer con superposiciones de ondas planas, y
de primer orden en el tiempo con una $\iu$ para que la dependencia temporal sea
$\eu^{-\iu\omega t}$ y no $\cos\omega t$ --- una \omterm{thm:b2:waves-on-strings:equation}{ecuación de ondas} real
daría $E^2 \propto p^4$.
\end{proof}

\begin{proposition}[Conservación de la probabilidad]\label{prop:b2:schrodinger-wave-functions:current}
La densidad de probabilidad $\rho = |\psi|^2$ y la \emph{corriente de
probabilidad}\index{corriente de probabilidad}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
obedecen a $\partial_t\rho + \partial_xj = 0$: la probabilidad ni se crea ni se
destruye, y $\int|\psi|^2\dd x$ sigue valiendo $1$. Para una onda plana
$A\eu^{\iu kx}$, $j = |A|^2\hbar k/m = \rho v$: densidad por velocidad, como para
cualquier flujo.
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$; a partir de la
ecuación y de su conjugada, $\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$, $\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$; los términos en $V$ se cancelan y el resto vale $(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$.
\end{proof}

\begin{proposition}[Momento lineal, valores medios y límite clásico]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
El momento lineal de un estado se obtiene con el operador $\hat p =
-\iu\hbar\,\partial_x$ (que da $\hbar k$ sobre una onda plana): $\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, y la energía con $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$. Los valores medios obedecen
(Ehrenfest) a
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
el centro del \omterm{prop:b2:dispersion-wave-packets:group}{paquete de ondas} sigue la ley de Newton exactamente cuando $V$ es
a lo sumo cuadrático, y de forma aproximada siempre que el paquete sea estrecho
frente a la escala en que varía $V$ --- el límite clásico,
que vale para cualquier objeto macroscópico porque $\hbar$ es tan pequeño.
\end{proposition}

\begin{proof}
Derívese $\langle x\rangle = \int x|\psi|^2$, úsese $\partial_t\rho =
-\partial_xj$ e intégrese por partes: $\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$. La segunda relación es el mismo cálculo un
orden más arriba (admitido aquí).
\end{proof}

\section{Estados estacionarios}

\begin{theorem}[Estados estacionarios]\label{thm:b2:schrodinger-wave-functions:stationary}
Si $V$ no depende del tiempo, la ecuación tiene soluciones de la
forma separada
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(la \emph{ecuación de Schrödinger independiente del tiempo}\index{ecuación
de Schrödinger!independiente del tiempo}): los \emph{estados
estacionarios}\index{estado estacionario}, de energía $E$ bien definida, cuya densidad
de probabilidad $|\varphi|^2$ no se mueve. Las energías $E$ para las que $\varphi$ es
aceptable (acotada y normalizable para un estado ligado) son los
\emph{niveles de energía}; la solución general es una superposición $\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$, y una superposición de dos niveles
tiene una densidad que oscila a la \emph{frecuencia de Bohr}\index{frecuencia
de Bohr} $\nu_{21} = (E_2 - E_1)/h$ --- la frecuencia de la luz que el
átomo emite o absorbe.
\end{theorem}

\begin{proof}
Introdúzcase $\varphi(x)f(t)$: $\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
una función de $t$ igual a una función de $x$, luego una constante $E$; $f
= \eu^{-\iu Et/\hbar}$. Para dos niveles, $|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$.
La completitud de los \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios} se admite.
\end{proof}

\section{La partícula libre: ondas planas y paquetes de ondas}

\begin{proposition}[Partícula libre]\label{prop:b2:schrodinger-wave-functions:free}
Para $V = 0$ los \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios} son las ondas planas $\eu^{\iu(kx -
\omega t)}$ con
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{g}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
las ondas de materia son \emph{dispersivas}; la \omterm{def:b2:dispersion-wave-packets:relation}{velocidad de fase} no tiene
sentido físico, y la \emph{\omterm{prop:b2:dispersion-wave-packets:group}{velocidad de grupo}} es la velocidad de la partícula. Una onda
plana no es normalizable --- una partícula de momento perfectamente nítido está
repartida por todo el espacio ---, así que una partícula real es un \emph{paquete
de ondas}\index{paquete de ondas}, una superposición de ondas planas en una banda
$\Delta k$, localizada en $\Delta x \sim 1/\Delta k$, que se mueve a $v_{\text{g}}$ y
\emph{se ensancha}: un paquete gaussiano de anchura inicial $\sigma_0$ tiene en el
instante $t$ la anchura
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
y cumple $\Delta x\,\Delta p = \hbar/2$ en $t = 0$ (el mínimo permitido)
y más después.
\end{proposition}

\begin{proof}
La \omterm{def:b2:dispersion-wave-packets:relation}{relación de dispersión} se sigue de la ecuación; $v_{\text{g}}$ como en
\cref{ch:b2:dispersion-wave-packets}. La fórmula del ensanchamiento sale de
la síntesis de Fourier del paquete --- cada componente $\eu^{\iu kx}$ recoge
la fase $\eu^{-\iu\hbar k^2t/2m}$, y la integral gaussiana con un
coeficiente complejo da una gaussiana de anchura creciente; la integral de Fourier
es la herramienta del volumen de matemáticas del tercer año y admitimos el
resultado. Físicamente: el paquete contiene momentos repartidos en $\Delta p
= \hbar/2\sigma_0$, luego velocidades repartidas en $\Delta v = \hbar/2m\sigma_0$,
y tras un tiempo $t$ las componentes más rápidas han adelantado a las más lentas en $\Delta v\,t$
--- la fórmula dice exactamente esto.
\end{proof}

\begin{example}[Quién se ensancha]\label{ex:b2:schrodinger-wave-functions:spreading}
Un electrón localizado en $\sigma_0 = \qty{1}{nm}$: $\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- tras un nanosegundo su paquete mide \qty{60}{\micro m}
de ancho: un electrón dejado a solas no se queda quieto; ligado en un átomo
no se ensancha porque el potencial lo sujeta (un \omterm{thm:b2:schrodinger-wave-functions:stationary}{estado estacionario}).
Un grano de polvo de \qty{1e-15}{kg} localizado en \qty{1}{\micro m}: $\tau =
\qty{2e7}{s}$, ocho meses; una bala: más que la edad del universo.
El ensanchamiento cuántico es real, e invisible para todo lo que puede verse.
\end{example}

\begin{remark}[Energía y tiempo]\label{rem:b2:schrodinger-wave-functions:energytime}
Un estado que solo existe durante $\Delta t$ --- un átomo excitado de vida media
$\tau$, un \omterm{prop:b2:scalar-light-model:coherence}{tren de ondas} de longitud $c\tau$ --- es una superposición de energías
repartidas en $\Delta E \sim \hbar/\Delta t$: la relación de Fourier de
\cref{ch:b2:scalar-light-model} entre la duración de un tren y su
\omterm{prop:b2:scalar-light-model:coherence}{anchura espectral}, aplicada ahora a $E = \hbar\omega$. Un nivel atómico de
vida media \qty{10}{ns} tiene una anchura $\Delta E = \qty{7e-8}{eV}$, una anchura
natural de raya de \qty{16}{MHz}; la anchura Doppler de \cref{ch:b2:laser}
(\qty{1.5}{GHz}) la esconde en un gas, pero no en un átomo frío o un ion en
reposo.
\end{remark}

\begin{method}[Cómo trabajar con funciones de onda]\label{meth:b2:schrodinger-wave-functions:method}
(1) Normalícese; las probabilidades son $\int|\psi|^2$ sobre la región. (2)
Con $V$ independiente del tiempo: búsquese $\varphi(x)\eu^{-\iu Et/\hbar}$, resuélvase la
ecuación estacionaria e impónganse las condiciones de contorno --- los niveles.
(3) Superpónganse \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios} para la evolución; dos niveles baten a
$(E_2 - E_1)/h$. (4) Movimiento libre: $v_{\text{g}} = p/m$, tiempo de ensanchamiento
$2m\sigma_0^2/\hbar$. (5) Corrientes: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$;
para $A\eu^{\iu kx} + B\eu^{-\iu kx}$, $j = (|A|^2 - |B|^2)\hbar k/m$ (sin término cruzado).
(6) Límite clásico: Ehrenfest, y $\lambda$ frente al tamaño del
aparato.
\end{method}

\section{Ejercicios}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:1}
Longitudes de onda de De Broglie: un electrón de \qty{100}{eV}, de \qty{10}{keV} y de
\qty{100}{keV} (úsese $p = \sqrt{2mE}$ y dígase después por qué el último necesita una
corrección); un neutrón térmico (\qty{25}{meV}, $m = \qty{1.67e-27}{kg}$); una
molécula de C$_{60}$ ($m = \qty{1.2e-24}{kg}$) a \qty{200}{m/s}; y una pelota de
tenis. ¿Cuáles de ellos muestran interferencias?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:2}
$\psi(x) = A\eu^{-x^2/2a^2}$: $A$ (úsese $\int\eu^{-u^2}\dd u = \sqrt\pi$);
$\langle x\rangle$, $\Delta x$; probabilidad de encontrar la partícula en
$|x| < a$ ($\operatorname{erf}(1) = 0.84$) y en $|x| < 2a$ ($0.9995$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:3}
Compruébese que $\varphi(x)\eu^{-\iu Et/\hbar}$ resuelve la ecuación cuando $\varphi$
resuelve la estacionaria. Frecuencia del fasor para $E = \qty{1}{eV}$;
para una superposición de dos niveles separados \qty{2}{eV}, frecuencia y
longitud de onda de la oscilación de $|\psi|^2$ --- y de la luz
emitida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:4}
Tiempo de ensanchamiento $\tau = 2m\sigma_0^2/\hbar$ y anchura tras \qty{1}{ns} y
\qty{1}{s}: un electrón con $\sigma_0 = \qty{1}{nm}$ y con $\sigma_0 = \qty{1}{mm}$;
un protón con $\sigma_0 = \qty{1}{nm}$; un grano de \qty{1e-12}{kg} con
$\sigma_0 = \qty{1}{\micro m}$. ¿Por qué se quedan las cosas donde se dejan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:5}
\emph{Corriente.} (a) Dedúzcase la \omterm{thm:b2:fluid-kinematics:continuity}{ecuación de continuidad}. (b) $j$ para
$A\eu^{\iu kx}$, para $A\sin kx$ y para $A\eu^{\iu kx} + B\eu^{-\iu kx}$. (c) ¿Por qué es
la ausencia de término cruzado en el último caso la afirmación de que los flujos
reflejado e incidente simplemente se restan? (d) Muéstrese que para una \omterm{def:b2:schrodinger-wave-functions:psi}{función
de onda} real $j = 0$: un \omterm{thm:b2:schrodinger-wave-functions:stationary}{estado estacionario} ligado no lleva corriente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:6}
\emph{Un paquete gaussiano.} $\psi(x,0) = A\eu^{\iu k_0x}\eu^{-x^2/4\sigma^2}$. (a)
$A$, $\langle x\rangle$, $\Delta x$. (b) $\langle p\rangle$ con $\hat p =
-\iu\hbar\partial_x$. (c) Admitiendo $\Delta p = \hbar/2\sigma$, compruébese
$\Delta x\Delta p = \hbar/2$. (d) Demuéstrese $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$ a partir de la \omterm{thm:b2:fluid-kinematics:continuity}{ecuación de continuidad}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:7}
\emph{Electrones en un tubo.} Unos electrones acelerados con \qty{100}{V}
recorren \qty{30}{cm}. (a) Celeridad, longitud de onda y tiempo de vuelo. (b) \omterm{def:b2:dispersion-wave-packets:relation}{Velocidades de fase} y
de grupo; ¿cuál es la del electrón? (c) Un paquete inicialmente de
\qty{10}{\micro m} de ancho: tiempo de ensanchamiento y anchura a la llegada. (d) \omterm{def:b2:diffraction:huygens}{Difracción}
por una abertura de \qty{0.3}{mm} por el camino: apertura angular y mancha en la
pantalla. ¿Vale la imagen de los rayos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:8}
\emph{Fases.} (a) Muéstrese que $\psi$ y $\eu^{\iu\theta}\psi$ dan las mismas
predicciones. (b) Para $\psi = (\psi_1 + \eu^{\iu\alpha}\psi_2)/\sqrt2$, escríbase
$|\psi|^2$ y muéstrese que $\alpha$ importa. (c) En la doble rendija, ¿qué
son $\psi_1$ y $\psi_2$, y qué fija su fase relativa en un punto
de la pantalla? (d) ¿Por qué destruye las franjas detectar por qué rendija pasó
el electrón (cualitativamente)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:9}
\emph{Ehrenfest.} (a) Para $V = \tfrac12m\omega^2x^2$, muéstrese que $\langle
x\rangle$ obedece exactamente a $\ddot{\langle x\rangle} = -\omega^2\langle x\rangle$.
(b) Para $V = mgx$, $\langle x\rangle$ cae como una piedra. (c) Para un $V$
general, ¿cuándo vale $\langle V'(x)\rangle \approx V'(\langle x\rangle)$? (d) Una
bolita de \qty{1}{g} en un cuenco: estímese el tamaño de paquete que haría falta para
que fuese ``cuántica'' y compárese con un átomo.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:10}
\emph{El bamboleo.} En una caja $0 < x < L$ de paredes impenetrables, los
\omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios} son $\varphi_n = \sqrt{2/L}\sin(n\pi x/L)$, $E_n =
n^2h^2/8mL^2$ (\cref{ch:b2:potential-wells-tunneling}). Tómese $\psi =
(\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/\sqrt2$. (a)
$|\psi|^2$ en función del tiempo. (b) Muéstrese que $\langle x\rangle = L/2 -
(16L/9\pi^2)\cos\omega_{21}t$ (úsese $\int_0^L x\sin(\pi x/L)\sin(2\pi x/L)\dd x
= -8L^2/9\pi^2$). (c) Electrón con $L = \qty{1}{nm}$: $E_2 - E_1$, la
frecuencia y la longitud de onda de la oscilación. (d) ¿Qué hace una carga
que oscila (\cref{ch:b2:dipole-radiation})? Conclúyase sobre las
rayas espectrales.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{El ensanchamiento, deducido.} El paquete $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
es la superposición $\int g(k)\eu^{\iu kx}\dd k$ con $g(k) \propto
\eu^{-\sigma_0^2k^2}$ (admitido: la transformada de Fourier de una gaussiana es una
gaussiana, de anchura $1/2\sigma_0$ en $k$). (a) Escríbase $\psi(x,t)$ dando a cada
componente su fase $\eu^{-\iu\hbar k^2t/2m}$. (b) Complétese el cuadrado en
el exponente y admítase $\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$ para $\alpha$ complejo de parte real positiva: muéstrese que
$|\psi|^2$ es una gaussiana de anchura $\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$. (c) Interprétese $\tau$ mediante $\Delta v = \Delta p/m$. (d) ¿Por qué no
se ensancha en el vacío un paquete de fotones?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{Energía y tiempo.} (a) Un estado que decae como $\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$
(con probabilidad $\eu^{-t/\tau}$) es una superposición de energías: admitiendo
que su distribución de energías es una lorentziana de anchura total $\Gamma =
\hbar/\tau$, dese $\Gamma$ para $\tau = \qty{10}{ns}$ (un átomo), \qty{1e-8}{s} y
\qty{1e-23}{s} (una resonancia hadrónica, en MeV). (b) Anchura natural de raya del
nivel de \qty{10}{ns} en frecuencia; compárese con la anchura Doppler de
\qty{1.5}{GHz}. (c) ¿Cómo se ve la anchura natural (átomos fríos,
iones atrapados)? (d) Relaciónese con la \omterm{prop:b2:scalar-light-model:coherence}{longitud de coherencia} del \omterm{prop:b2:scalar-light-model:coherence}{tren de ondas}
emitido.
\end{exercise}

\section{Problema: el electrón en un tubo y el ensanchamiento de un paquete de ondas}

\begin{problem}[{Problema de fin de semana --- cuándo es partícula un electrón
y cuándo onda}]\label{pb:b2:schrodinger-wave-functions:1}
Datos: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $m_{\text{p}} = \qty{1.67e-27}{kg}$.

\textbf{Parte I --- El tubo de rayos catódicos.}
Los electrones salen de un cátodo caliente con energía despreciable, se aceleran
con $U = \qty{20}{kV}$, pasan por una abertura de \qty{0.3}{mm} y golpean una
pantalla a \qty{40}{cm}.
\begin{enumerate}
  \item Celeridad y momento lineal (no relativistas); longitud de onda de De Broglie.
  \item La energía cinética es el $4\%$ de $mc^2$: ¿es aceptable el tratamiento
        no relativista para este problema?
  \item Tiempo de vuelo hasta la pantalla.
  \item \omterm{def:b2:diffraction:huygens}{Difracción} por la abertura: apertura angular $\sim\lambda/a$ y
        la mancha que produce en la pantalla; compárese con un píxel
        (\qty{0.3}{mm}).
  \item Heisenberg en la abertura: $\Delta p_y \gtrsim \hbar/2a$ da la
        misma apertura --- compruébese.
  \item Un paquete de \qty{0.3}{mm} de ancho a lo largo del haz: tiempo de ensanchamiento
        $2m\sigma_0^2/\hbar$ y anchura en la pantalla.
  \item Unas placas deflectoras aplican una fuerza transversal: por Ehrenfest, ¿cómo
        se mueve $\langle y\rangle$? Conclúyase que el tubo es un
        aparato clásico.
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Velocidad de fase} de la onda del electrón; ¿por qué no es paradójico que
        difiera de la celeridad del electrón?
\end{enumerate}

\textbf{Parte II --- \omterm{prop:b2:dispersion-wave-packets:group}{Paquetes de ondas}.}
\begin{enumerate}[resume]
  \item Para un paquete gaussiano de anchura $\sigma_0$: $\Delta p$ y la
        dispersión de velocidades $\Delta v$.
  \item Tiempo de ensanchamiento de un electrón con $\sigma_0 = \qty{0.1}{nm}$,
        \qty{1}{nm} y \qty{1}{\micro m}; de un protón con \qty{1}{nm}; y de un
        virus (\qty{1e-17}{kg}) con \qty{10}{nm}.
  \item Un electrón de un átomo de hidrógeno está localizado en \qty{0.1}{nm}
        y no se ensancha: ¿por qué?
  \item Un electrón libre de \qty{1}{eV} con $\sigma_0 = \qty{1}{nm}$: ¿cuánto
        recorre antes de duplicar su anchura? Compárese con la
        longitud de onda.
  \item En un metal, los electrones de conducción son ondas de $\lambda \approx
        \qty{0.5}{nm}$ por todo el cristal: ¿en qué sentido están
        ``deslocalizados''?
  \item Un microscopio electrónico a \qty{100}{keV} tiene $\lambda \approx
        \qty{4}{pm}$ y sin embargo solo resuelve \qty{0.1}{nm}: ¿por qué (sus lentes
        magnéticas admiten un semiángulo de unos \qty{10}{mrad})?
  \item Dese el criterio que decide si un objeto debe tratarse
        con una \omterm{def:b2:schrodinger-wave-functions:psi}{función de onda} o con Newton.
\end{enumerate}

\textbf{Parte III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{Estados estacionarios} y transiciones.}
Un electrón está confinado en $0 < x < L = \qty{1}{nm}$; $\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$.
\begin{enumerate}[resume]
  \item $E_1$, $E_2$ y $E_3$ en electronvoltios; las \omterm{thm:b2:schrodinger-wave-functions:stationary}{frecuencias de Bohr}
        $\nu_{21}$ y $\nu_{31}$ y las longitudes de onda.
  \item Compruébese que $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ no depende del tiempo;
        ¿dónde es más probable encontrar el electrón en el estado fundamental?
  \item Para $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$: $|\psi|^2$ en $t = 0$ y en $t = T_{21}/2$; dibújese.
  \item $\langle x\rangle(t)$ (dado $\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$):
        amplitud del bamboleo.
  \item El \omterm{thm:b2:dipole-radiation:field}{dipolo oscilante} $e\langle x\rangle$ radia: ¿a qué
        frecuencia, y qué dice eso sobre las rayas espectrales y
        sobre la estabilidad de los \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios}?
  \item Energía del estado $\psi$: ¿es $E_1$, $E_2$ u otra
        cosa? ¿Qué da una medida?
\end{enumerate}

\textbf{Parte IV --- Interpretación.}
\begin{enumerate}[resume]
  \item En el experimento de la doble rendija con un electrón cada vez, ¿qué es aleatorio
        y qué no?
  \item Escríbase la \omterm{prop:b2:schrodinger-wave-functions:current}{corriente de probabilidad} de $A\eu^{\iu kx}$ y relaciónese con
        un haz de $n$ electrones por unidad de longitud a la celeridad $v$.
  \item ¿Por qué debe ser compleja y de primer orden en el tiempo la ecuación de
        Schrödinger (dos razones)?
  \item Resúmase en cinco líneas: la \omterm{def:b2:schrodinger-wave-functions:psi}{función de onda}, la ecuación, los
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionarios}, el paquete y el límite clásico.
\end{enumerate}
\end{problem}
