\chapter{बहु-तरंग व्यतिकरण और जालक}\label{ch:b2:gratings}

किसी दीपक के नीचे कोई संहत चक्रिका तिरछी कीजिए और वह कोई इंद्रधनुष फेंक
देती है: उसका गड्ढों का सर्पिल, जिनके बीच डेढ़ माइक्रोमीटर है, कोई जालक है
--- हज़ारों नन्ही परावर्ती पट्टियाँ, हर एक तरंग की कोई प्रति लौटाती हुई, और
सब व्यतिकरण करती हुईं। दो तरंगें पिछले अध्यायों की कोमल $\cos^2$ फ्रिंजें
देती हैं; हज़ार तरंगें हज़ार गुना सँकरी फ्रिंजें देती हैं, इतनी तीखी कि दो ऐसे
तरंगदैर्घ्य अलग कर दें जो लाख में एक भाग से भिन्न हों। यही वह यंत्र है जिससे
सूर्य की संरचना और मंदाकिनियों की चाल पढ़ी गई, और जो हर
वर्णक्रममापी में बैठा है। यह अध्याय $N$ तरंगें जोड़ता है, \omterm{prop:b2:gratings:equation}{जालक का
समीकरण}, उसका परिक्षेपण और उसकी विभेदन-शक्ति निकालता है, और उस दूसरे
बहु-तरंग यंत्र से मिलाता है, फ़ाब्री--पेरो कोटर, जो लेज़र का
अनुनादक बनेगा।

\section{$N$ तरंगों का व्यतिकरण}

\begin{theorem}[समांतर कला-श्रेढ़ी में $N$ बराबर तरंगें]\label{thm:b2:gratings:nwaves}
\index{बहु-तरंग व्यतिकरण}\index{मुख्य अधिकतम}\index{गौण अधिकतम}\index{कला-सदिश}
$N$ संसक्त तरंगें, जिनका आयाम $a$ बराबर है और जिनमें हर एक पिछली से
कला $\varphi$ पीछे है, आयाम $A(\varphi)$ और इस तीव्रता वाली किसी तरंग में
अध्यारोपित होती हैं
\[
A = a\,\frac{\sin(N\varphi/2)}{\sin(\varphi/2)} , \qquad
I(\varphi) = I_0\,\frac{\sin^2(N\varphi/2)}{\sin^2(\varphi/2)} :
\]
अर्थात् तीव्रता $N^2I_0$ के \emph{\omterm{thm:b2:gratings:nwaves}{मुख्य अधिकतम}} वहाँ जहाँ $\varphi = 2\pi p$ (सारी
तरंगें एक ही कला में), जिनकी कोणीय अर्ध-चौड़ाई $2\pi/N$ है, $\varphi$ में (पहले शून्य
$\varphi = 2\pi p \pm 2\pi/N$ पर हैं), और जिनके बीच $N - 2$ \emph{\omterm{thm:b2:gratings:nwaves}{गौण
अधिकतम}} हैं, जो मुख्य वालों के लगभग $4.5\%$ से ऊँचे नहीं होते, बड़े $N$ के
लिए। जितनी अधिक तरंगें, उतनी ही तीखी और चमकीली चोटियाँ: चौड़ाई
$1/N$ की तरह सिकुड़ती है, ऊँचाई $N^2$ की तरह बढ़ती है, और ऊर्जा (उनका गुणनफल)
$N$ की तरह।
\end{theorem}

\begin{proof}
सम्मिश्र आयाम $a\eu^{\iu k\varphi}$, $k = 0, \dots, N - 1$: यह एक
गुणोत्तर श्रेढ़ी है,
\[
a(1 - \eu^{\iu N\varphi})/(1 - \eu^{\iu\varphi}) = a\eu^{\iu(N - 1)\varphi/2}\sin(N\varphi/2)
/\sin(\varphi/2) .
\]
शून्य वहाँ जहाँ $\sin(N\varphi/2) = 0$ पर $\sin(\varphi/2) \ne 0$: $\varphi = 2\pi
m/N$, $m$ $N$ का गुणज नहीं। कला-सदिश चित्र में $N$ तीर
शून्यों पर किसी सम बहुभुज में बंद हो जाते हैं और \omterm{thm:b2:gratings:nwaves}{मुख्य अधिकतम} पर
एक पंक्ति में आ जाते हैं।
\end{proof}

\section{विवर्तन जालक}

\begin{proposition}[जालक का समीकरण]\label{prop:b2:gratings:equation}
\index{विवर्तन जालक}\index{जालक का समीकरण}\index{कोटि (जालक)}\index{कोणीय परिक्षेपण}
कोई \emph{जालक} $N$ एक-सी, बराबर दूरी वाली, समांतर झिर्रियों
(या खाँचों) का कोई समुच्चय है, जिनका आवर्त $a$ है। अभिलंब से कोण $\theta_0$ पर आपाती कोई समतल तरंग
हर झिर्री से फिर उत्सर्जित होती है; दिशा $\theta$ में
पड़ोसी झिर्रियों की तरंगों का \omterm{thm:b2:two-wave-interference:formula}{पथांतर} $\delta = a(\sin\theta
- \sin\theta_0)$ है, और वे इन दिशाओं में पुष्ट होती हैं
\[
a\,(\sin\theta_p - \sin\theta_0) = p\lambda , \qquad p = 0, \pm1, \pm2, \dots
\]
--- अर्थात् जालक की \emph{कोटियाँ}। शून्य कोटि हर तरंगदैर्घ्य के लिए बिना
मुड़ी पुंज है; और हर दूसरी कोटि तरंगदैर्घ्यों को फैला देती है,
\emph{\omterm{prop:b2:gratings:equation}{कोणीय परिक्षेपण}} के साथ
\[
\frac{\dd\theta}{\dd\lambda} = \frac{p}{a\cos\theta} ,
\]
जो किसी महीन जालक और किसी ऊँची कोटि के लिए बड़ा है; और अधिकतम कोटि
$|p| \le a(1 + |\sin\theta_0|)/\lambda$ है। कोई \emph{परावर्तन जालक} (किसी दर्पण पर खींचे
खाँचे) परावर्तित कोणों के साथ वही समीकरण मानता है।
\end{proposition}

\begin{proof}
क्रमिक झिर्रियों के बीच कला $\varphi = 2\pi\delta/\lambda$ है; और
\cref{thm:b2:gratings:nwaves} के \omterm{thm:b2:gratings:nwaves}{मुख्य अधिकतम} $\delta = p\lambda$ पर हैं।
$a\sin\theta = p\lambda + a\sin\theta_0$ का अवकलन कीजिए, $p$, $\theta_0$ स्थिर रखकर।
\end{proof}

भेजता है, दूसरी में $\qty{36.9}{\degree}$ पर, और उसकी कोई तीसरी कोटि नहीं है
($3 \times 0.3 > 1$); उसका पहली-कोटि वर्णक्रम \qty{13.9}{\degree}
(\qty{400}{nm}) से \qty{24.8}{\degree} (\qty{700}{nm}) तक चलता है। कोई संहत चक्रिका, $a =
\qty{1.6}{\micro m}$, परावर्तन में ऐसा ही कोई जालक है --- इसीलिए इंद्रधनुष,
और इसीलिए कोई DVD ($a = \qty{0.74}{\micro m}$) उसे और चौड़ा फैलाती है।
दूसरी कोटि का लाल ($\sin\theta = 0.84$ पर \qty{700}{nm}) तीसरी कोटि के
बैंगनी पर चढ़ जाता है: पहली के परे कोटियाँ चढ़ जाती हैं, और यह झंझट छन्नों
या किसी प्रिज़्म से दूर होती है।
\end{example}

\section{विभेदन-शक्ति}

\begin{proposition}[किसी जालक की विभेदन-शक्ति]\label{prop:b2:gratings:resolution}
\index{विभेदन-शक्ति (जालक)}\index{रैले कसौटी (वर्णक्रमी)}
पास-पास के दो तरंगदैर्घ्य $\lambda$ और $\lambda + \Delta\lambda$ कोटि $p$ में ठीक तभी सुलझते हैं
जब एक का \omterm{thm:b2:gratings:nwaves}{मुख्य अधिकतम} दूसरे के पहले शून्य पर पड़े
(रैले की कसौटी), और यह तब होता है जब
\[
\frac\lambda{\Delta\lambda} = pN :
\]
अर्थात् \emph{विभेदन-शक्ति} कोटि गुणा जगमगाई रेखाओं की
संख्या है। उसे $W(\sin\theta - \sin\theta_0)/\lambda$ भी लिखा जा सकता है, जहाँ $W = Na$ जालक की
चौड़ाई है: अर्थात् जालक भर का सबसे बड़ा \omterm{thm:b2:two-wave-interference:formula}{पथांतर},
तरंगदैर्घ्यों में --- और इसीलिए बड़े वर्णक्रमलेखी दसियों सेंटीमीटर चौड़े
जालक लेते हैं और सर्पण कोणों पर काम करते हैं।
\end{proposition}

\begin{proof}
कोटि $p$ का अधिकतम, $\lambda + \Delta\lambda$ के लिए, उस तरंगदैर्घ्य के लिए $\varphi = 2\pi p$ पर है,
अर्थात् $\delta = p(\lambda + \Delta\lambda)$ पर; और $\lambda$ के लिए उस अधिकतम का पहला शून्य
$\delta = p\lambda + \lambda/N$ पर है। बराबर रखने पर: $p\Delta\lambda = \lambda/N$। तब $pN =
Na(\sin\theta - \sin\theta_0)/\lambda$।
\end{proof}

\begin{example}[सोडियम रेखाओं को अलग करना]\label{ex:b2:gratings:sodium}
युग्मक, \qty{589}{nm} पर $\Delta\lambda = \qty{0.6}{nm}$, को $R = 1000$ चाहिए: अर्थात्
पहली कोटि में हज़ार रेखाएँ, यानी \num{600} रेखा प्रति मिलीमीटर वाले किसी
जालक का \qty{2}{mm}। उसकी अपनी रेखा-चौड़ाई, \qty{0.02}{nm}, को $R = 30\,000$ चाहिए: अर्थात् पहली कोटि में \qty{5}{cm} का कोई
जालक, या दूसरी में \qty{2.5}{cm} --- और
\num{1200} रेखा प्रति मिलीमीटर वाला \qty{10}{cm} का कोई जालक दूसरी कोटि में
$240\,000$ तक पहुँचता है, और \qty{0.0025}{nm} सुलझा देता है: अर्थात् परमाणुओं की तापीय गति से
चौड़ी हुई किसी रेखा की चौड़ाई, और \qty{1}{km/s} से चलते किसी तारे का
डॉप्लर खिसकाव।
\end{example}

\section{फ़ाब्री--पेरो कोटर}

\begin{proposition}[आयाम-विभाजन से बहु-तरंग व्यतिकरण]\label{prop:b2:gratings:fabryperot}
\index{फ़ाब्री--पेरो व्यतिकरणमापी}\index{एरी फलन}\index{सूक्ष्मता}\index{मुक्त वर्णक्रमी परास}
दो समांतर आंशिक परावर्ती दर्पण (तीव्रता-परावर्तकता $R$), जो $L$ दूरी पर हैं,
अभिलंब आपतन में किसी तरंग को केवल उतना ही पार जाने देते हैं
जितना उनके बीच $0, 1, 2, \dots$ बार उछल चुकी तरंगें एक ही
कला में जुड़ें: चक्कर-कला $\varphi = 4\pi L/\lambda$ के साथ पारगत
तीव्रता है (एरी का फलन)
\[
\frac{I_t}{I_0} = \frac1{1 + F\sin^2(\varphi/2)} , \qquad F = \frac{4R}{(1 - R)^2} :
\]
अर्थात् $2L = p\lambda$ पर पूरे पारगमन की तीखी चोटियाँ (कोटर के
\emph{अनुनाद}, जिनकी आवृत्ति में दूरी \emph{\omterm{prop:b2:gratings:fabryperot}{मुक्त वर्णक्रमी परास}}
$c/2L$ है), जिनकी सापेक्ष चौड़ाई $1/\mathcal F$ है, जहाँ \emph{सूक्ष्मता} $\mathcal F = \pi
\sqrt R/(1 - R)$ --- $30$, $R = 0.9$ के लिए; $300$, $R = 0.99$ के लिए। वह विभेदन-शक्ति में जालक की
बराबरी है ($p\mathcal F$, जहाँ $p = 2L/\lambda \sim 10^5$, अतः $R \sim
10^7$) और, अपने ऊपर बंद कर देने पर, लेज़र का अनुनादक
(\cref{ch:b2:laser})।
\end{proposition}

\begin{proof}
हर चक्कर आयाम को $R\eu^{\iu\varphi}$ से गुणा कर देता है (किसी
अचर कला तक); और पारगत आयाम $t^2a\sum(R\eu^{\iu\varphi})^k =
t^2a/(1 - R\eu^{\iu\varphi})$ है, जहाँ $t^2 = 1 - R$; उसका वर्ग-मापांक $(1 - R)^2/
(1 - 2R\cos\varphi + R^2) = (1 - R)^2/[(1 - R)^2 + 4R\sin^2(\varphi/2)]$ है। चोटियाँ
$\varphi = 2\pi p$ पर; और अर्ध-अधिकतम वहाँ जहाँ $F\sin^2(\varphi/2) = 1$, अर्थात् $\Delta\varphi \approx
4/\sqrt F = 2(1 - R)/\sqrt R$, चोटियों के बीच के $2\pi$ के सामने।
\end{proof}

\begin{method}[जालक की गणनाएँ]\label{met:b2:gratings:method}
(1) आवर्त $a = 1/(\text{प्रति मिमी रेखाएँ})$; और कोणों के चिह्नों के साथ \omterm{prop:b2:gratings:equation}{जालक का
समीकरण}। (2) जो कोटियाँ होती हैं उन्हें सूचीबद्ध कीजिए ($|\sin\theta| \le 1$)
और उनका चढ़ाव जाँचिए ($p\lambda_{\max}$ के सामने $(p + 1)\lambda_{\min}$)। (3)
परिक्षेपण $p/a\cos\theta$, जिसे फ़ोकस दूरी के साथ संसूचक पर किसी जगह में बदल
लीजिए: $\dd x = f\,\dd\theta$। (4) विभेदन-शक्ति $pN$, जहाँ $N$ असल में
जगमगाई रेखाएँ हैं; और सुलझाई जाने वाली संरचना से $\lambda/R$ की तुलना कीजिए। (5) असली यंत्रों के लिए
प्रवेश-झिर्री की चौड़ाई (जो रेखाओं को धुँधला करती है) और संसूचक के पिक्सल भी जोड़िए।
\end{method}

\section{समस्या: किसी तारे के लिए कोई वर्णक्रमलेखी}

\begin{problem}[{सप्ताहांत समस्या --- वह वर्णक्रमलेखी गढ़ना जो

और किसी ग्रह का डगमगाना}]
\label{pb:b2:gratings:1}

कोई वर्णक्रमलेखी: प्रवेश-झिर्री, फ़ोकस दूरी $f_1 =
\qty{1.0}{m}$ का संरेखक, प्रति मिलीमीटर $1200$ रेखाओं और चौड़ाई $W =
\qty{12}{cm}$ का कोई परावर्तन जालक, $f_2 = \qty{1.0}{m}$ का कोई कैमरा, और \qty{15}{\micro m}
पिक्सलों वाला कोई संसूचक। वह पहली कोटि में, \qty{550}{nm} के आसपास,
जालक को \qty{20}{\degree} आपतन पर रखकर काम करता है।

\medskip
\noindent\textbf{भाग I --- जालक।}
\begin{enumerate}
  \item जालक का आवर्त; और यदि पुंज पूरी चौड़ाई ढके तो जगमगाई
        रेखाओं की संख्या।
  \item पहली कोटि में \qty{550}{nm} का विवर्तन कोण; और \qty{400}{nm} तथा
        \qty{700}{nm} का; और दृश्य वर्णक्रम की कोणीय
        चौड़ाई।
  \item \qty{550}{nm} पर \omterm{prop:b2:gratings:equation}{कोणीय परिक्षेपण}; और संसूचक पर रैखिक परिक्षेपण
        nm प्रति mm तथा nm प्रति पिक्सल में।
  \item विभेदन-शक्ति; \qty{550}{nm} पर सुलझाया गया सबसे छोटा $\Delta\lambda$; और
        पिक्सलों में।
  \item क्या दूसरी कोटि का बैंगनी पहली कोटि के दृश्य पर पड़ता है?
        उसे कौन-सा छन्ना हटाता है?
  \item पहली कोटि में पूरे दृश्य वर्णक्रम के लिए ज़रूरी संसूचक की
        लंबाई।
  \item जालक \qty{17}{\degree} पर ब्लेज़ किया गया है: इस आपतन पर पहली कोटि में
        वह किस तरंगदैर्घ्य के लिए सबसे दक्ष है (लिट्रो आकलन
        $2a\sin\beta = \lambda$ लीजिए)?
  \item जालक ब्लेज़ तरंगदैर्घ्य पर $80\%$ प्रकाश लौटाता है और
        दूरदर्शी तथा प्रकाशिकी $40\%$: तारे के कितने अंश फ़ोटॉन
        संसूचक तक पहुँचते हैं।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- रेखाएँ।}
\begin{enumerate}[resume]
  \item तारे की हाइड्रोजन बामर रेखाएँ: H$\alpha$ \qty{656.3}{nm},
        H$\beta$ \qty{486.1}{nm}, H$\gamma$ \qty{434.0}{nm}: \qty{550}{nm} के सापेक्ष
        संसूचक पर उनकी जगहें।
  \item किसी ड्यूटीरियम परमाणु की H$\alpha$ \qty{0.18}{nm} छोटी है: अलग हुई?
        कितने पिक्सलों से?
  \item तारे के वर्णक्रम में सोडियम युग्मक: पिक्सलों में
        दूरी; और हर रेखा की चौड़ाई यदि तारे का वायुमंडल
        उन्हें \qty{0.05}{nm} तक चौड़ा कर दे।
  \item प्रवेश-झिर्री \qty{100}{\micro m} चौड़ी है, और उसका प्रतिबिंब
        आवर्धन $f_2/f_1$ पर बनता है: संसूचक पर उसकी चौड़ाई पिक्सलों में और
        nm में; क्या वर्णक्रमलेखी झिर्री-सीमित है या जालक-सीमित?
  \item दूरदर्शी से आता प्रकाश आकाश पर \qty{1}{''} की किसी चकती की तरह
        आता है, जिसे दूरदर्शी ($f = \qty{20}{m}$) झिर्री पर \qty{100}{\micro m}
        चौड़ा बना देता है: यदि झिर्री सँकरी करके \qty{50}{\micro m} कर दी जाए तो खोए
        प्रकाश का अंश (चकती को एकसमान लीजिए)।
  \item कोई सँकरी झिर्री तीखी रेखाएँ पर अधिक रव वाले वर्णक्रम क्यों
        देती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- तारा चलता है।}
\begin{enumerate}[resume]
  \item कोई तारा \qty{30}{km/s} से दूर जा रहा है: H$\alpha$ का डॉप्लर खिसकाव nm
        और पिक्सलों में।
  \item पृथ्वी की कक्षीय गति वर्ष भर $\pm\qty{30}{km/s}$ जोड़ती है:
        उसे घटाना क्यों पड़ता है, और यदि \qty{100}{m/s} चाहिए तो
        किस परिशुद्धता तक?
  \item कोई ग्रह अपने तारे को \qty{10}{m/s} पर डगमगाता है: nm और
        पिक्सलों में खिसकाव; क्या वह विभेदन से नीचे है? समझाइए कि किसी रेखा के
        केंद्र को उसकी चौड़ाई के सौवें भाग तक, हज़ार रेखाओं भर
        नापना उसे कैसे लौटा लाता है (सांख्यिकीय लाभ $\sqrt{N}$)।
  \item यंत्र का ताप \qty{1}{K} बदलता है, और जालक (काँच, प्रसार
        \qty{1e-5}{K^{-1}}) फैलता है: पिक्सलों में रेखाओं का खिसकाव;
        और \qty{10}{m/s} के लिए कितनी स्थिरता चाहिए?
  \item कोई अंशांकन दीपक (थोरियम--आर्गन) उसी संसूचक पर सैकड़ों ज्ञात
        रेखाएँ डाल देता है: उसकी भूमिका समझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- विकल्प।}
\begin{enumerate}[resume]
  \item उतनी ही चौड़ाई का जालक प्रति मिलीमीटर $300$ रेखाओं से खींचा जाए और
        चौथी कोटि में काम में लिया जाए (कोटियाँ छाँटने को किसी प्रति-परिक्षेपक के साथ):
        \qty{550}{nm} पर विवर्तन कोण, विभेदन-शक्ति, परिक्षेपण और \omterm{prop:b2:gratings:fabryperot}{मुक्त
        वर्णक्रमी परास}; क्या बदला, क्या नहीं?
  \item झिर्री से पहले रखा कोई फ़ाब्री--पेरो एटेलॉन ($L = \qty{1}{mm}$, $R = 0.9$):
        nm में \omterm{prop:b2:gratings:fabryperot}{मुक्त वर्णक्रमी परास}, सूक्ष्मता, चोटी की चौड़ाई; उसकी
        पारगमन-चोटियों की कंघी संसूचक पर कैसी दिखती है,
        और वह किस काम आती है?
  \item $\dd n/\dd\lambda = \qty{-1e-4}{nm^{-1}}$ वाले काँच का कोई \qty{60}{\degree} प्रिज़्म
        न्यूनतम विचलन के पास \omterm{prop:b2:gratings:equation}{कोणीय परिक्षेपण} $\dd D/\dd\lambda \approx 2\,\dd n/\dd\lambda$
        रखता है: जालक वाले से तुलना कीजिए; और जालकों ने प्रिज़्मों की जगह
        क्यों ले ली?
  \item सबसे बड़े खगोलीय वर्णक्रमलेखी मीटर भर चौड़े जालकों से और बड़े
        कोणों पर काम करते हुए क्यों बनाए जाते हैं (सीमा
        $2W/\lambda$)?
  \item कांतिमान-8 का कोई तारा दूरदर्शी पर प्रति सेकंड और प्रति
        नैनोमीटर \qty{1e4}{} फ़ोटॉन देता है: संसूचक पर प्रति पिक्सल प्रति सेकंड
        फ़ोटॉन, और \qty{100}{s} के किसी उद्भासन का संकेत-रव अनुपात
        (केवल फ़ोटॉन रव)।
  \item समेटिए: वे तीन संख्याएँ जो किसी वर्णक्रमलेखी का वर्णन करती हैं
        (परिक्षेपण, विभेदन-शक्ति, \omterm{prop:b2:gratings:fabryperot}{मुक्त वर्णक्रमी परास}) और हर एक को क्या
        तय करता है।
\end{enumerate}
\end{problem}
