\chapter{प्रवाहों में संवेग और ऊर्जा के संतुलन}\label{ch:b2:flow-balances}

कोई अग्निशामक नली के सहारे टिका खड़ा है: तीस मीटर प्रति सेकंड से निकलता
पानी सैकड़ों न्यूटन से पीछे धकेलता है, यद्यपि टोंटी को पानी के सिवा कुछ
छूता तक नहीं। कोई जेट इंजन पंख से लटका रहता है और हवा को पीछे फेंककर
वायुयान को आकाश में खींच ले जाता है। बगीचे का फुहारा बिना किसी मोटर के
घूमता है। हर दशा में कोई तरल किसी प्रदेश में घुसता है, अलग संवेग के साथ
निकलता है, और यही अंतर एक बल है।
यह अध्याय यांत्रिकी के नियम --- संवेग, कोणीय
संवेग, ऊर्जा --- किसी नियत द्रव्यमान के लिए नहीं, बल्कि किसी स्थिर
प्रदेश को पार करते तरल के लिए लिखता है, अर्थात् किसी \emph{\omterm{def:b2:flow-balances:cv}{विवृत निकाय}}
के लिए: यही वह रूप है जिसमें अभियंता टरबाइन, राकेट, पंप और नलियाँ नापते हैं।

\section{किसी विवृत निकाय के संतुलन}

\begin{definition}[नियंत्रण आयतन; विवृत निकाय]\label{def:b2:flow-balances:cv}
\index{नियंत्रण आयतन}\index{विवृत निकाय}
\emph{नियंत्रण आयतन} (या नियंत्रण पृष्ठ) आकाश का कोई स्थिर प्रदेश
$\Sigma$ है जिससे होकर तरल बहता है; और किसी भी क्षण उसके भीतर का तरल
एक \emph{विवृत निकाय} है, जो $\Sigma$ के प्रवेशों और निर्गमों से होकर
बाहर के साथ द्रव्यमान का विनिमय करता है। जिस बंद निकाय पर न्यूटन के
नियम लागू होते हैं वह वह तरल है जो $\Sigma$ के भीतर समय $t$ पर है और
जिसका $t + \dd t$ तक पीछा किया जाए: तब वह $\Sigma$ में से निकल चुके भाग को
घटाकर और घुस चुके भाग को जोड़कर बनता है।
\end{definition}

\begin{theorem}[स्थायी प्रवाह में संवेग-संतुलन]\label{thm:b2:flow-balances:momentum}
\index{संवेग-संतुलन (स्थायी प्रवाह)}\index{संवेग अभिवाह}
किसी \omterm{def:b2:flow-balances:cv}{नियंत्रण आयतन} से होकर \omterm{def:b2:fluid-kinematics:euler}{स्थायी प्रवाह} के लिए, जिसमें एक प्रवेश (काट
$S_1$, एकसमान वेग $\vect v_1$) और एक निर्गम ($S_2$, $\vect v_2$) है, और
\omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर} $D_m$ दोनों पर एक ही है,
\[
\sum\vect F_{\text{बाह्य}} = D_m\,(\vect v_2 - \vect v_1) ,
\]
होता है, जहाँ $\sum\vect F_{\text{बाह्य}}$ $\Sigma$ के भीतर के तरल पर लगे
सारे बाह्य बलों का योग है: भार, उसके संपर्क में आई दीवारों और पिंडों के
बल, और प्रवेश तथा निर्गम काटों पर दाब-बल
($P_1S_1\vect n_1$ भीतर की ओर धकेलता, और निर्गम पर $-P_2S_2\vect n_2$)। कई
प्रवेशों और निर्गमों की दशा में दायाँ पक्ष बाहर जाते
संवेग-अभिवाहों $D_{m,i}\vect v_i$ का योग घटा भीतर आते अभिवाहों का योग है।
\end{theorem}

\begin{proof}
वह बंद निकाय $\mathcal S$ लीजिए जो $\Sigma$ के उस तरल से बना है जो समय
$t$ पर वहाँ है, और उस परत $\delta m = D_m\dd t$ से जो
$S_1$ से होकर घुसने वाली है। $t + \dd t$ पर वह $\Sigma$ के तरल और उस परत $\delta m$
से बनता है जो $S_2$ से होकर निकल चुकी है। उसका संवेग $\vect p_\Sigma(t) +
\delta m\,\vect v_1$ से $\vect p_\Sigma(t + \dd t) + \delta m\,\vect v_2$ हो गया; और \omterm{def:b2:fluid-kinematics:euler}{स्थायी
प्रवाह} में $\vect p_\Sigma$ अचर है, अतः $\dd\vect p_{\mathcal S} = D_m\dd t\,(\vect v_2
- \vect v_1)$। $\mathcal S$ के लिए न्यूटन का दूसरा नियम, जिसके बाह्य बल
(प्रथम कोटि तक) $\Sigma$ के तरल पर लगे बल ही हैं, यह परिणाम देता है।
\end{proof}

\begin{example}[दीवार पर धार; अग्निशामक की नली]\label{ex:b2:flow-balances:jet}
$s$ काट और $v$ चाल की कोई धार किसी दीवार पर लंबवत टकराकर
उस पर फैल जाती है: धार के अनुदिश बाहर जाता संवेग शून्य है, अतः
दीवार को $F = D_mv = \rho sv^2$ मिलता है --- \qty{24}{m/s} पर \qty{5}{cm} की
धार के लिए \qty{1.1}{kN}। नली और टोंटी के तरल पर वही संतुलन लगाने पर,
टोंटी अग्निशामक को $D_mv$ से पीछे धकेलती है:
\qty{30}{m/s} पर \qty{20}{L/s} \qty{600}{N} देते हैं, अर्थात् एक व्यक्ति का भार।
यदि दीवार $u$ से पीछे हटे, तो प्रति एकांक समय उस तक केवल $D_m' = \rho s(v - u)$
पहुँचता है और वह सापेक्ष चाल $v - u$ से आता है: $F = \rho s(v - u)^2$।
\end{example}

\begin{proposition}[पेल्टन बाल्टी; राकेट का प्रणोद]\label{prop:b2:flow-balances:pelton}
\index{पेल्टन टरबाइन}\index{प्रणोद}\index{राकेट समीकरण}
(क) $v$ चाल और $D_m$ \omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर} की कोई धार, जिसे कोई बाल्टी
$180^\circ$ मोड़कर लौटा देती है और जो धार की दिशा में $u$ से चलती है: बाल्टी पर
बल $F = 2D_m(v - u)$ होता है (जहाँ $D_m$ वह प्रवाह दर है जो चक्र की चलती
बाल्टियाँ वास्तव में रोकती हैं), शक्ति $Fu$
$u = v/2$ पर अधिकतम होती है और तब $\tfrac12D_mv^2$ के बराबर: अर्थात् धार की
पूरी गतिज शक्ति। (ख) $m(t)$ द्रव्यमान का कोई राकेट, जो गैस को पीछे
सापेक्ष चाल $v_e$ और \omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर} $D_m = -\dd m/\dd t$ से फेंकता है,
$F = D_mv_e$ प्रणोद अनुभव करता है; मुक्त आकाश में उसकी चाल
$\Delta v = v_e\ln(m_0/m_1)$ बढ़ती है (त्सिओल्कोव्स्की का \omterm{prop:b2:flow-balances:pelton}{राकेट समीकरण}); और गुरुत्व के अधीन
$m\,\dd v/\dd t = D_mv_e - mg$।
\end{proposition}

\begin{proof}
(क) बाल्टी के तंत्र में धार $v - u$ से आती है और
$-(v - u)$ से निकलती है; संवेग-संतुलन $2D_m(v - u)$ देता है (बाल्टी
आदर्श है: सापेक्ष चाल की कोई हानि नहीं)। शक्ति $2D_mu(v - u)$, जो
$u = v/2$ पर अधिकतम है। (ख) $\dd t$ में राकेट और फेंकी गई गैस का संवेग: मुक्त आकाश में $(m + \dd m)
(v + \dd v) + (-\dd m)(v - v_e) = mv$, अतः $m\,\dd v = -\dd m\,v_e$
और $v_1 - v_0 = v_e\ln(m_0/m_1)$; और गुरुत्व के साथ कुल संवेग
$-mg\,\dd t$ से बदलता है।
\end{proof}

\begin{example}[एक प्रक्षेपक]\label{ex:b2:flow-balances:rocket}
पहला चरण: $D_m = \qty{2500}{kg/s}$, $v_e = \qty{3000}{m/s}$ पर: प्रणोद
\qty{7.5}{MN}, जो \qty{600}{t} के राकेट को मंच से $7.5 \times 10^6/6 \times 10^5 - g
= \qty{2.7}{m/s^2}$ पर उठाता है। \qty{200}{s} में \qty{500}{t} जलाने पर
$\Delta v = 3000\ln6 - 9.81 \times 200 = 5400 - 2000 = \qty{3.4}{km/s}$ मिलता है ---
गुरुत्व-हानि ही कारण है कि प्रक्षेपक तेज़ी से चढ़ते हैं, और लघुगणक ही
कारण कि वे चरणों में बनाए जाते हैं।
\end{example}

\begin{example}[नली के मोड़ पर बल]\label{ex:b2:flow-balances:bend}
किसी नली में $P$ और $v$ पर बहता पानी, जिसकी काट $S$ है, $90^\circ$ मुड़ता है।
कोहनी के तरल पर संतुलन: $\vect F_{\text{दीवार}} + PS\vect e_x - PS
\vect e_y = D_m(v\vect e_y - v\vect e_x)$, अतः दीवार तरल को
$\vect F_{\text{दीवार}} = -(PS + \rho Sv^2)(\vect e_x - \vect e_y)$ से धकेलती है और तरल
कोहनी को बाहर की ओर $(PS + \rho Sv^2)\sqrt2$ से। $D = \qty{20}{cm}$,
$P = \qty{3}{bar}$, $v = \qty{3}{m/s}$ के लिए: $PS = \qty{9.4}{kN}$, $\rho Sv^2 = \qty{0.28}{kN}$,
और समद्विभाजक के अनुदिश बल \qty{13.7}{kN} --- यहाँ दाब वाला पद भारी पड़ता
है, और यही वह बल है जिसे पाइपलाइन के लंगर-गुटके थामते हैं।
\end{example}

\section{कोणीय संवेग: टरबाइन और फुहारे}

\begin{theorem}[कोणीय संवेग का संतुलन; ऑयलर का टरबाइन समीकरण]\label{thm:b2:flow-balances:angular}
\index{कोणीय संवेग का संतुलन (प्रवाह)}\index{ऑयलर का टरबाइन समीकरण}
\omterm{def:b2:fluid-kinematics:euler}{स्थायी प्रवाह} में, किसी बिंदु $O$ के परितः $\Sigma$ के तरल पर लगे बाह्य
बलों का कुल आघूर्ण बाहर जाते कोणीय संवेग के अभिवाह में से भीतर आते अभिवाह को
घटाने के बराबर होता है: $\sum\vect{\mathcal M}_O = D_m(\vect r_2\wedge\vect v_2 - \vect r_1
\wedge\vect v_1)$। $Oz$ अक्ष की किसी घूर्णी मशीन (टरबाइन या पंप) के लिए,
जिसमें तरल त्रिज्या $r_1$ पर दिगंशी वेग
$v_{\theta1}$ के साथ घुसता है और $r_2$ पर $v_{\theta2}$ के साथ निकलता है, तरल
द्वारा घूर्णक पर लगाया गया बलाघूर्ण और उससे मिलती शक्ति ये हैं:
\[
\Gamma = D_m(r_1v_{\theta1} - r_2v_{\theta2}) , \qquad
\mathcal P = \Gamma\omega = D_m\omega(r_1v_{\theta1} - r_2v_{\theta2}) .
\]
कोई टरबाइन तरल से भँवर निकाल लेती है; कोई पंप उसमें भँवर भर देता है।
\end{theorem}

\begin{proof}
वही बंद-निकाय वाला तर्क, जिसमें $\vect r\wedge\vect v$ $\vect v$ की जगह
ले लेता है। घूर्णक पर बलाघूर्ण तरल पर घूर्णक के
बलाघूर्ण का ऋणात्मक है; और प्रवेश तथा निर्गम पृष्ठों
(जो घूर्णन-पृष्ठ हैं) पर दाब-बलों का अक्ष के परितः कोई आघूर्ण
नहीं होता।
\end{proof}

\begin{example}[घास का फुहारा]\label{ex:b2:flow-balances:sprinkler}

$\Gamma_f$ हो तो वह $\omega = (w - \Gamma_f/D_mR)/R$ पर ठहर जाता है।
\end{example}

\section{ऊर्जा-संतुलन: पंप, टरबाइन और हानियाँ}

\begin{theorem}[किसी स्थायी प्रवाह का यांत्रिक ऊर्जा-संतुलन]\label{thm:b2:flow-balances:energy}
\index{ऊर्जा-संतुलन (प्रवाह)}\index{द्रवचालित शक्ति}\index{शीर्ष}
किसी असंपीड्य तरल के \omterm{def:b2:fluid-kinematics:euler}{स्थायी प्रवाह} के लिए, जो प्रवेश 1 और निर्गम 2 के
बीच किसी मशीन (पंप या टरबाइन) से होकर जाता है, जहाँ $\mathcal P_u$ चलते
भागों द्वारा तरल को दी गई यांत्रिक शक्ति है (पंप के लिए धनात्मक,
टरबाइन के लिए ऋणात्मक) और $\mathcal P_{\text{क्षय}} \ge 0$ \omterm{def:b2:viscous-flows:viscosity}{श्यानता} से
क्षयित शक्ति,
\[
D_V\Bigl[\bigl(P_2 + \tfrac12\rho v_2^2 + \rho gz_2\bigr) - \bigl(P_1 + \tfrac12\rho v_1^2
+ \rho gz_1\bigr)\Bigr] = \mathcal P_u - \mathcal P_{\text{क्षय}} .
\]
$\rho gD_V$ से भाग देने पर संतुलन \emph{शीर्षों} में (तरल के मीटरों में)
मिलता है: $H_2 - H_1 = H_{\text{पंप}} - H_{\text{हानि}}$, जहाँ $H = P/\rho g +
v^2/2g + z$। बिना मशीन वाले \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल} के लिए यही बर्नूली है; और शीर्ष को
$H_{\text{पंप}}$ से चढ़ाते किसी पंप की \emph{\omterm{thm:b2:flow-balances:energy}{द्रवचालित शक्ति}}
$\rho gD_VH_{\text{पंप}}$ है।
\end{theorem}

\begin{proof}
\cref{thm:b2:flow-balances:momentum} की उपपत्ति वाले बंद निकाय के लिए
गतिज ऊर्जा प्रमेय: $\dd t$ में उसकी गतिज ऊर्जा
$D_m\dd t\,(v_2^2 - v_1^2)/2$ से बदलती है; और उस पर किया गया कार्य
गुरुत्व का, $-D_m\dd t\,g(z_2 - z_1)$, चलते सिरों के फलकों पर दाब-बलों का,
$(P_1S_1v_1 - P_2S_2v_2)\dd t = (P_1 - P_2)D_V\dd t$, चलते
भागों का, $\mathcal P_u\dd t$, और आंतरिक श्यान बलों का,
$-\mathcal P_{\text{क्षय}}\dd t$, है (विराम में पड़ी दीवारें कोई कार्य नहीं करतीं)।
अब $\dd t$ से भाग दीजिए।
\end{proof}

\begin{example}[पंप नापना]\label{ex:b2:flow-balances:pump}
पानी को \qty{10}{cm} की \qty{100}{m} नली ($\lambda = 0.02$) से होकर
\qty{20}{L/s} पर \qty{30}{m} ऊपर चढ़ाना है: $v = \qty{2.5}{m/s}$, शीर्ष-हानि $\lambda(L/D)
v^2/2g = 0.02 \times 1000 \times 0.32 = \qty{6.4}{m}$, निर्गम का गतिज शीर्ष
\qty{0.32}{m}: अतः $H_{\text{पंप}} = \qty{36.7}{m}$, \omterm{thm:b2:flow-balances:energy}{द्रवचालित शक्ति} $\rho gD_VH =
\qty{7.2}{kW}$, और $70\%$ दक्षता पर \qty{10}{kW} बिजली। पंप के
निर्गम पर दाब प्रवेश से $\rho gH_{\text{पंप}} \approx \qty{3.6}{bar}$ ऊपर होता है।
\end{example}

\begin{proposition}[एकाएक चौड़ाई: बोर्दा--कार्नो हानि]\label{prop:b2:flow-balances:borda}
\index{बोर्दा--कार्नो हानि}\index{शीर्ष-हानि (एकल)}
जब $S_1$ काट की कोई नली एकाएक $S_2
> S_1$ काट की नली में खुलती है, तो धार किसी विक्षुब्ध प्रदेश में फैल जाती
है और आगे दाब \emph{चढ़ता} तो है, पर बर्नूली के कहे से कम:
प्रति एकांक आयतन खोई यांत्रिक ऊर्जा
\[
\Delta P_{\text{हानि}} = \tfrac12\rho(v_1 - v_2)^2 ,
\]
होती है, अर्थात् ``खोए'' सापेक्ष वेग की गतिज ऊर्जा। किसी बड़ी टंकी में
खुलती नली का प्रवाह ($v_2 \to 0$) अपनी सारी गतिज ऊर्जा खो देता है।
\end{proposition}

\begin{proof}
चौड़ाई और आगे की उस काट के बीच के तरल पर संवेग-संतुलन, जहाँ प्रवाह फिर
एकसमान हो जाए; प्रयोग दिखाता है कि चौड़ाई पर छल्लेदार दीवार पर दाब
$P_1$ ही होता है (धार अभी फैली नहीं है)।
अतः $(P_1 - P_2)S_2 = D_m(v_2 - v_1) = \rho S_2v_2(v_2 -
v_1)$, अर्थात् $P_2 - P_1 = \rho v_2(v_1 - v_2)$। बर्नूली
$\tfrac12\rho(v_1^2 - v_2^2)$ देता; और अंतर $\tfrac12\rho(v_1 - v_2)^2$ है।
\end{proof}

\begin{method}[कोई संतुलन खड़ा करना]\label{met:b2:flow-balances:method}
(1) \omterm{def:b2:flow-balances:cv}{नियंत्रण आयतन} खींचिए: उसकी सीमा प्रवाह को केवल वहीं काटे जहाँ वेग और
दाब ज्ञात हों या चाहिए हों (एकसमान काटें,
$P_0$ पर मुक्त धारें, मुक्त सतहें)। (2) \emph{भीतर के तरल} पर लगे बाह्य बल
गिनाइए: भार, दीवार के बल (अज्ञात, और प्रायः यही लक्ष्य),
कटी काटों पर दाब। (3) बाहर जाता संवेग-अभिवाह घटा भीतर आता लिखिए;
और चलते भागों के लिए, यदि प्रवाह उस भाग के तंत्र में स्थायी हो, तो
उसी तंत्र में काम कीजिए। (4) बलाघूर्णों के लिए कोणीय संवेग का संतुलन; और
शक्तियों के लिए शीर्षों में ऊर्जा-संतुलन। (5) दीवार पर बल
तरल पर दीवार के बल का ऋणात्मक है; और यदि पूरे उपकरण पर कुल बल चाहिए
हो तो उपकरण के बाहर लगते वायुमंडलीय दाब को भी जोड़िए।
\end{method}

घूर्णक से मँडराता है। घूर्णक को ऐसी चक्रिका मानिए जो स्थिर हवा लेकर
उसे चक्रिका पर एकसमान चाल $v_i$ से नीचे धकेलती है, और बहुत नीचे $2v_i$ से
(गुणक 2 स्वीकार कीजिए)। (क) $\rho$, $A$, $v_i$ के पदों में प्रणोद;
$v_i$ का मान। (ख) हवा को दी गई शक्ति (बहुत नीचे उसकी गतिज
शक्ति)। (ग) बड़े घूर्णक को कम शक्ति क्यों चाहिए? (घ) \qty{25}{cm} के चार
परिनोदकों वाले \qty{1}{kg} के ड्रोन के लिए वही गणना;
शक्ति-भार अनुपात की तुलना।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:flow-balances:8}
\emph{एकाएक चौड़ाई।} \qty{5}{cm} की नली में \qty{4.0}{m/s} पर पानी
\qty{10}{cm} की नली में घुसता है। (क) आगे की चाल; बर्नूली के अनुसार और
बोर्दा--कार्नो के अनुसार दाब-वृद्धि; हानि पास्कल में और शीर्ष के
मीटरों में। (ख) वही चौड़ाई क्रमिक बनाने पर (कोई विसारक) बर्नूली की वृद्धि का $80\%$
वापस मिलता है: हानि। (ग) जब \qty{5}{cm} की नली किसी बड़ी टंकी में
गिरे तो हानि। (घ) दिए गए प्रवाह पर (क) और (ग) में खोई शक्ति।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:flow-balances:9}
कोई पंप अपने से \qty{5.0}{m} नीचे के कुएँ से पानी खींचता है और उसे
\qty{6}{cm} की \qty{60}{m} नली से होकर ($\lambda = 0.025$, और जोड़ों के लिए $2\,v^2/2g$ की एकल हानि) अपने से \qty{25}{m} ऊपर की टंकी में
\qty{10}{L/s} पहुँचाता है।
(क) चाल और हानियाँ। (ख) पंप-शीर्ष और \omterm{thm:b2:flow-balances:energy}{द्रवचालित शक्ति}; $65\%$ दक्षता
पर धुरी-शक्ति। (ग) पंप के प्रवेश (चूषण की ओर) और निर्गम पर दाब।
(घ) पानी का वाष्प दाब \qty{2.3}{kPa} है: कुएँ से ऊपर पंप की अधिकतम
ऊँचाई।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:flow-balances:10}
\emph{जल-राकेट।} \qty{2.0}{L} की किसी बोतल में \qty{1.0}{L} पानी और
\qty{5.0}{bar} (निरपेक्ष) पर हवा है; टोंटी का व्यास \qty{2.2}{cm} है;
और खाली बोतल \qty{0.10}{kg}। (क) आरंभिक निर्गम चाल (बर्नूली, पानी की
सतह विराम में) और प्रणोद; आरंभिक त्वरण। (ख) पानी निकलते-निकलते
हवा समतापी रूप से फैलती है: आधा पानी निकल जाने पर दाब और निर्गम चाल।
(ग) जलने की अवधि और पाई गई चाल आँकिए
(गुरुत्व और कर्षण छोड़ दीजिए, और प्रणोद को उसका माध्य मान मानिए)। (घ)
पानी ख़त्म होने पर खाली बोतल क्या करती है, और थोड़ा-सा पानी (बहुत नहीं)
सबसे अच्छी ऊँचाई क्यों देता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:flow-balances:11}
\emph{फ़्रांसिस टरबाइन।} पानी किसी चक्र में $r_1 = \qty{1.0}{m}$ पर
$v_{\theta1} = \qty{20}{m/s}$ के साथ घुसता है और $r_2 = \qty{0.50}{m}$ पर बिना भँवर के निकलता है;
प्रवाह \qty{30}{m^3/s}; घूर्णन \qty{150}{rpm}। (क) बलाघूर्ण और शक्ति। (ख)
निकाला गया शीर्ष, $H = \mathcal P/\rho gD_V$। (ग) यदि चक्र वही प्रवेश-वेग रखते हुए
\qty{200}{rpm} पर चले और निर्गम का भँवर घूर्णन की दिशा में
$v_{\theta2} = \qty{3}{m/s}$ हो जाए: शक्ति, और शीर्ष का वह
अंश जो व्यर्थ निर्गम-भँवर में बदल जाता है। (घ) पेल्टन चक्र ऊँचे शीर्षों
और छोटे प्रवाहों के लिए, और फ़्रांसिस उलटी दशा के लिए, क्यों काम आता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:flow-balances:12}
\emph{जल-उछाल।} एकांक चौड़ाई की किसी क्षैतिज नहर में
$h_1$ गहराई का पानी $v_1$ से बहता हुआ एकाएक $h_2$ गहराई और
$v_2$ चाल पर उछल जाता है (किसी बंधिका के नीचे या रसोई के सिंक में दिखती झागदार सीढ़ी)। (क) द्रव्यमान-संरक्षण।
(ख) उछाल के दोनों ओर के तरल पर संवेग-संतुलन, जहाँ हर ओर दाब
द्रवस्थैतिक है: दिखाइए कि
$\tfrac12gh_1^2 + v_1^2h_1 = \tfrac12gh_2^2 + v_2^2h_2$। (ग) उससे निकालिए कि $h_2/h_1 =
\tfrac12\bigl(-1 + \sqrt{1 + 8\,\mathrm{Fr}_1^2}\bigr)$, जहाँ $\mathrm{Fr}_1 =
v_1/\sqrt{gh_1}$ (फ़्रूड संख्या); और उछाल के लिए $\mathrm{Fr}_1 > 1$ चाहिए। (घ)
उछाल में खोया शीर्ष, $\Delta H = (h_2 - h_1)^3/4h_1h_2$; $h_1 =
\qty{0.20}{m}$, $v_1 = \qty{4.0}{m/s}$ के लिए अंक, और प्रति मीटर चौड़ाई
क्षयित शक्ति।
\end{exercise}

\section{समस्या: धारें, राकेट और चक्र}

\begin{problem}[{सप्ताहांत समस्या --- जेट वायुयान को क्या धकेलता है, राकेट
को क्या उठाता है, टरबाइन को क्या घुमाता है, और पानी को ऊपर पंप करने में
क्या लगता है}]
\label{pb:b2:flow-balances:1}

\medskip
\noindent\textbf{भाग I --- टर्बोजेट।}
परीक्षण-मंच पर कोई इंजन $D_a = \qty{50}{kg/s}$ स्थिर हवा लेता है,
$D_f = \qty{1.0}{kg/s}$ ईंधन जलाता है, और मिश्रण को वायुमंडलीय दाब पर $v_e =
\qty{600}{m/s}$ से फेंकता है।
\begin{enumerate}
  \item परीक्षण-मंच पर प्रणोद।
  \item $v = \qty{250}{m/s}$ की उड़ान में (इंजन के सापेक्ष वही $v_e$):
        प्रणोद, और नोदन-शक्ति $Fv$।
  \item इंजन-तंत्र में निकाली गई, गैस को दी गई गतिज शक्ति।
        नोदन दक्षता $\eta_p = Fv/\mathcal P_{\text{गतिज}}$; दिखाइए
        कि ईंधन का द्रव्यमान छोड़ने पर $\eta_p = 2v/(v + v_e)$।
  \item ईंधन \qty{43}{MJ/kg} देता है: इंजन की ऊष्मीय दक्षता
        (गतिज शक्ति बटा ऊष्मा-शक्ति) और समग्र दक्षता।
  \item कोई टर्बोफ़ैन उसी गतिज शक्ति से
        $\qty{500}{kg/s}$ हवा त्वरित करता है: \qty{250}{m/s} पर निकास-चाल, प्रणोद और $\eta_p$।
        यात्री-विमानों में विशाल पंखे क्यों होते हैं?
  \item वायुमंडल के ऊपर, राकेट का काम करने के लिए, टर्बोजेट विराम में
        क्यों नहीं चल सकता?
  \item उतरते समय प्रतिप्रणोद पंखे के प्रवाह को अक्ष से
        \qty{45}{\degree} पर आगे की ओर मोड़ देता है: प्रश्न 5 के टर्बोफ़ैन के लिए
        \qty{70}{m/s} पर ब्रेक-बल, जब पंखे का निकास इंजन के सापेक्ष
        \qty{150}{m/s} हो; और उसी प्रवाह से मिलते अग्र
        प्रणोद से तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- राकेट।}
एक-चरण का कोई राकेट: आरंभिक द्रव्यमान $m_0 = \qty{300}{t}$, नोदक
\qty{200}{t}, निकास-चाल $v_e = \qty{3.0}{km/s}$, जलने का समय $\tau =
\qty{150}{s}$ अचर \omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर} पर।
\begin{enumerate}[resume]
  \item राकेट और $\dd t$ में फेंकी गई गैस पर संवेग-संतुलन लगाकर
        ऊर्ध्वाधर उड़ान के लिए $m\,\dd v/\dd t = D_mv_e - mg$
        निकालिए।
  \item \omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर}, प्रणोद, और उड़ान-भरने पर तथा
        जलने के अंत पर त्वरण।
  \item समाकलन करके जलने के अंत की चाल पाइए; ``गुरुत्व-हानि''
        क्या है?
  \item वही राकेट \qty{100}{t} नोदक वाले दो चरणों में बाँटने पर जलने के
        अंत की चाल, जहाँ पहले चरण का खाली द्रव्यमान
        (\qty{20}{t}) दूसरे के जलने से पहले गिरा दिया जाता है (यहाँ
        गुरुत्व छोड़ दीजिए): एक चरण से तुलना कीजिए।
  \item उसी द्रव्यमान-अनुपात के साथ कक्षीय चाल
        (\qty{7.8}{km/s}, गुरुत्व छोड़कर) तक पहुँचने के लिए एक चरण को
        कौन-सी निकास-चाल चाहिए होती? नोदकों के चुनाव पर टिप्पणी दीजिए।
  \item उड़ान भरते समय निकास-धार (\qty{3}{km/s} पर \qty{1333}{kg/s}) मंच के
        ज्वाला-विक्षेपक से टकराकर क्षैतिज कर दी जाती है:
        विक्षेपक पर बल।
  \item राकेट का निकास भी एक तरल-प्रवाह है: किसी वास्तविक टोंटी के लिए
        प्रणोद ऊँचाई के साथ \emph{बढ़ता} क्यों है (टोंटी की निर्गम काट पर
        दाब सोचिए)?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- पेल्टन चक्र।}
पिछले अध्याय के संयंत्र की धार: $D_V = \qty{60}{m^3/s}$, $v =
\qty{62.6}{m/s}$ पर, जो दो चक्रों में बँटी है; हर चक्र की त्रिज्या $R =
\qty{1.5}{m}$ है और उसकी बाल्टियाँ धार को \qty{165}{\degree} मोड़ती हैं।
\begin{enumerate}[resume]
  \item बाल्टी-चाल $u$ पर एक चक्र की बाल्टियों पर बल, और
        शक्ति; उत्तम $u$।
  \item उत्तम बिंदु पर घूर्णन चाल \unit{rpm} में; धुरी पर बलाघूर्ण।
  \item जनित्र को \qty{50}{Hz} के किसी गुणज बटा उसके ध्रुव-युग्मों की
        संख्या पर घूमना है: ध्रुव-युग्मों की संख्या चुनिए।
  \item एक चक्र की शक्ति और धार की गतिज शक्ति के सापेक्ष
        दक्षता; $95\%$ जनित्र-दक्षता पर कुल वैद्युत शक्ति।
  \item उत्तम बिंदु पर किसी बाल्टी पर कोणीय संवेग का संतुलन (\omterm{thm:b2:flow-balances:angular}{ऑयलर का
        टरबाइन समीकरण}) लगाकर: प्रति किलोग्राम भीतर आते और बाहर जाते
        पानी का कोणीय संवेग, और बलाघूर्ण की जाँच।
  \item भार एकाएक हट जाता है और चक्र $u =
        v$ की ओर बेक़ाबू भागने लगता है: बल और बलाघूर्ण का क्या होता है, और
        धार को हटा देने वाला विक्षेपक हर पेल्टन संयंत्र में क्यों होता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- ऊपर पंप करना।}
संयंत्र भंडारण के लिए भी काम आता है: रात में \qty{40}{m^3/s} पानी
जलनाल (\qty{400}{m}, \qty{3}{m}, $\lambda = 0.015$) से होकर \qty{200}{m} ऊपर के
जलाशय में वापस पंप किया जाता है।
\begin{enumerate}[resume]
  \item \qty{40}{m^3/s} पर जलनाल में शीर्ष-हानियाँ; आवश्यक पंप-शीर्ष।
  \item \omterm{thm:b2:flow-balances:energy}{द्रवचालित शक्ति}, और $88\%$ पंप-दक्षता पर खींची गई
        वैद्युत शक्ति।
  \item पंप करते समय जलनाल के तल पर दाब; टरबाइन-दशा (पिछला अध्याय)
        के उसके मान से तुलना कीजिए।
  \item \qty{8}{h} की हर रात में संचित ऊर्जा (चढ़ाए गए पानी की
        स्थितिज ऊर्जा); और भंडारण की चक्रीय दक्षता, यदि
        टरबाइन-दशा उपलब्ध शीर्ष का $90\%$ और जनित्र $95\%$ वापस दे।
  \item इस समस्या में काम आए चारों संतुलनों (संवेग, कोणीय
        संवेग, ऊर्जा, द्रव्यमान) और हर एक से नापे गए उपकरण का सार दीजिए।
\end{enumerate}
\end{problem}
