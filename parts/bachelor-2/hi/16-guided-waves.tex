\chapter{निर्देशित तरंगें और कोटर}\label{ch:b2:guided-waves}

किसी रडार की स्पंद प्रेषक से तश्तरी तक ताँबे की किसी आयताकार नली के भीतर
चलती है; किसी इंटरनेट कड़ी का प्रकाश बाल से भी पतले काँच के किसी धागे के
भीतर दस हज़ार किलोमीटर चलता है; किसी भट्ठी की सूक्ष्मतरंगें उसकी दीवारों
के बीच उछलती हैं और गरम तथा ठंडे धब्बों की किसी आकृति में बैठ जाती हैं।
चालक या अपवर्तक दीवारों में क़ैद कोई तरंग अब कहीं भी जाने को स्वतंत्र समतल
तरंग नहीं रही: उसे दीवारों पर सीमा-शर्तें मानी पड़ती हैं, और वही चुन देती
हैं कि वह कौन-कौन-से रूप ले सकती है --- \emph{विधाएँ} --- तथा किसी
\emph{कटाव} आवृत्ति से नीचे उसे मना कर देती हैं। यह अध्याय सबसे सरल
निर्देशिका, दो समांतर चालक पट्टियाँ, \cref{ch:b2:wave-interfaces} के
परावर्तनों से बनाता है; फिर \omterm{prop:b2:guided-waves:rectangular}{आयताकार तरंग-निर्देशिका}, बंद कोटर, और किरण
चित्र में \omterm{prop:b2:guided-waves:fibre}{प्रकाशिक तंतु}।

\section{दो चालक पट्टियों के बीच तरंगें}

\begin{proposition}[समांतर-पट्टी निर्देशिका की विधाएँ]\label{prop:b2:guided-waves:plates}
\index{तरंग-निर्देशिका}\index{निर्देशित तरंग}\index{विधा (तरंग-निर्देशिका)}\index{कटाव आवृत्ति (तरंग-निर्देशिका)}
दो \omterm{prop:b2:wave-interfaces:conductors}{पूर्ण चालक} तल $y = 0$ और $y = a$ किसी निर्वात को घेरे हैं।
$z$ के अनुदिश चलती कोई तरंग $\vect E = E(y)\,\eu^{\iu(\omega t - k_gz)}
\vect e_x$ खोजिए (क्षेत्र पट्टियों के समांतर, और संचरण के
लंबवत)। निर्वात में \omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल के समीकरण} और पट्टियों पर $E = 0$ की शर्त
थोप देते हैं
\[
E(y) = E_0\sin\frac{n\pi y}a , \qquad
k_g^2 = \frac{\omega^2}{c^2} - \Bigl(\frac{n\pi}a\Bigr)^2 , \qquad n = 1, 2, \dots
\]
विधा $n$ केवल अपनी \emph{कटाव आवृत्ति} $f_n = nc/2a$ के ऊपर चलती है;
नीचे $k_g$ काल्पनिक है और क्षेत्र निर्देशिका के अनुदिश क्षय कर जाता है। उसके
कला और \omterm{prop:b2:dispersion-wave-packets:group}{समूह वेग} हैं
\[
v_\varphi = \frac c{\sqrt{1 - f_n^2/f^2}} > c , \qquad
v_g = c\sqrt{1 - f_n^2/f^2} < c , \qquad v_\varphi v_g = c^2 ,
\]
और उसके चुंबकीय क्षेत्र के घटक $y$ के अनुदिश तथा $z$ के अनुदिश हैं: कोई
\omterm{prop:b2:guided-waves:plates}{निर्देशित तरंग} संचरण की दिशा में अनुप्रस्थ नहीं होती।
\end{proposition}

\begin{proof}
$\Delta\vect E = \partial_t^2\vect E/c^2$ में डालिए: $E'' - k_g^2E = -(\omega^2/c^2)E$, अर्थात्
$E'' + (\omega^2/c^2 - k_g^2)E = 0$, जिसके $y = 0$ और
$y = a$ पर लुप्त होते हल वे ज्या हैं जिनमें $\omega^2/c^2 - k_g^2 = (n\pi/a)^2$; और $\operatorname{div}
\vect E = 0$ इसलिए टिकता है कि $E_x$ $x$ पर निर्भर नहीं करता। फ़ैराडे देते हैं
$\underline B_y = (k_g/\omega)\underline E$ और $\underline B_z = (\iu/\omega)\partial_y\underline E$, जिनमें दूसरा
समपाद में है। वेग परिक्षेपण-संबंध के क्लाइन--गोर्डन रूप से निकल आते हैं
(\cref{ch:b2:dispersion-wave-packets})।
\end{proof}

\begin{remark}[विधा दो समतल तरंगों की तरह]\label{rem:b2:guided-waves:zigzag}
$\sin(n\pi y/a)\eu^{-\iu k_gz}$ दो समतल तरंगों का योग है जिनके
तरंग-सदिश $(0, \pm n\pi/a, k_g)$ हैं, मापांक $\omega/c$, और जो
पट्टियों के बीच अक्ष से $\theta$ कोण पर उछलती रहती हैं, जहाँ $\cos\theta = k_gc/\omega$:
अर्थात् कोई निर्देशित विधा दीवारों के बीच टेढ़ी-मेढ़ी चलती कोई समतल तरंग है, जो हर
दीवार पर $r = -1$ से परावर्तित होती है, और अनुप्रस्थ \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} वही
शर्त है कि वह अपने साथ संपोषी व्यतिकरण करे। ऊर्जा
उस टेढ़ी-मेढ़ी राह पर $c$ से चलती है, अतः अक्ष के अनुदिश $c\cos\theta = v_g$ से;
और शिखरों के अक्ष के साथ प्रतिच्छेद $c/\cos\theta = v_\varphi$ से दौड़ते हैं। कटाव
पर तरंग निर्देशिका के पार आगे-पीछे उछलती रह जाती है और कहीं नहीं
जाती।
\end{remark}

\begin{example}[आँकड़े]\label{ex:b2:guided-waves:numbers}
\qty{3}{cm} की दूरी पर पट्टियाँ: $f_1 = \qty{5}{GHz}$, $f_2 = \qty{10}{GHz}$; इन
आवृत्तियों के बीच केवल एक विधा चलती है (\emph{एक-विधा}
संचालन, जो किसी स्पंद को अलग-अलग $v_g$ वाली कई स्पंदों में
बँटने से रोक देता है)। \qty{8}{GHz} पर विधा $n = 1$ का $v_g = 0.78c$, $v_\varphi =
1.28c$ है, और निर्देशिका-तरंगदैर्घ्य $\lambda_g = 2\pi/k_g = \qty{4.8}{cm}$, मुक्त
आकाश के \qty{3.75}{cm} के बजाय। \qty{4}{GHz} पर कुछ भी पार नहीं जाता: क्षेत्र
$\eu^{-z/\ell}$ की तरह क्षय करता है, जिसमें $\ell = c/2\pi\sqrt{f_1^2 - f^2} = \qty{1.6}{cm}$ ---
यही किसी भट्ठी के दरवाज़े की जाली का, कटाव से नीचे की उस तरंग-निर्देशिका का
जो अंशांकित क्षीणक की तरह काम में आती है, और वायु-संचार की उन धातु
नलियों का सिद्धांत है जो हवा तो जाने दें पर रेडियो नहीं।
\end{example}

\section{आयताकार तरंग-निर्देशिका और कोटर}

\begin{proposition}[आयताकार निर्देशिका; TE$_{10}$ विधा]\label{prop:b2:guided-waves:rectangular}
\index{आयताकार तरंग-निर्देशिका}\index{TE विधा}
$a \times b$ ($a > b$) आयताकार अनुप्रस्थ काट की कोई खोखली धातु नली वे
विधाएँ ले चलती है जिनकी कटाव आवृत्तियाँ हैं
\[
f_{mn} = \frac c2\sqrt{\Bigl(\frac ma\Bigr)^2 + \Bigl(\frac nb\Bigr)^2} ;
\]
सबसे नीची, \emph{TE$_{10}$ विधा} ($f_{10} = c/2a$), का $\vect E = E_0\sin
(\pi x/a)\,\eu^{\iu(\omega t - k_gz)}\,\vect e_y$ है: चौड़ी भुजा के पार आधी ज्या,
सँकरी भुजा के पार एकसमान, और दोनों दीवारों $x = 0$,
$x = a$ पर लुप्त --- ठीक वही समांतर-पट्टी विधा, जहाँ बग़ल की दीवारें $y = 0$, $b$
$\vect E$ के लंबवत हैं और उस पर कुछ नहीं थोपतीं। निर्देशिका
$f_{10}$ और अगले कटाव के बीच काम में आती है; उसके चुंबकीय क्षेत्र की रेखाएँ
$xz$ तल में बंद होती हैं और वे दीवारों में जो धाराएँ प्रेरित करती हैं, उन्हें
कोई झिरी काटनी नहीं चाहिए। ले जाई गई शक्ति $\tfrac14\varepsilon_0cE_0^2\,ab
\sqrt{1 - f_{10}^2/f^2}$ है।
\end{proposition}

\begin{proof}
सामान्य विधा $\cos$/$\sin$ के गुणनफल हैं $x$ और $y$ में, चारों दीवारों पर
सीमा-शर्तों के साथ, जो देते हैं $k_g^2 = \omega^2/c^2 - (m\pi/a)^2
- (n\pi/b)^2$ (सामान्य रूप में स्वीकृत; TE$_{10}$ हाल ऊपर वाला
प्रमेय ही है)। शक्ति: $z$ के अनुदिश माध्य पॉयंटिंग सदिश $E_0^2\sin^2(\pi x/a)
\,k_g/2\mu_0\omega$ है, जिसे अनुप्रस्थ काट पर समाकलित कीजिए।
\end{proof}

\begin{proposition}[कोटर अनुनादक]\label{prop:b2:guided-waves:cavity}
\index{कोटर अनुनादक}\index{विधा (कोटर)}
किसी निर्देशिका को $d$ दूरी पर दो चालक दीवारों से बंद कर देना उसे किसी
\emph{कोटर} में बदल देता है, जिसमें \omterm{prop:b2:guided-waves:plates}{निर्देशित तरंग} को $z$ के अनुदिश भी \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी
तरंग} बनानी पड़ती है: $k_g = p\pi/d$, और किसी आयताकार बक्से
$a \times b \times d$ की अनुनाद आवृत्तियाँ हैं
\[
f_{mnp} = \frac c2\sqrt{\Bigl(\frac ma\Bigr)^2 + \Bigl(\frac nb\Bigr)^2 + \Bigl(\frac pd\Bigr)^2} .
\]
$f$ से नीची आवृत्ति वाली विधाओं की संख्या
$8\pi V f^3/3c^3$ की तरह बढ़ती है, आयतन $V$ के किसी बक्से के लिए (स्वीकृत: किसी गोले के आठवें भाग में जालक
$(m/2a, n/2b, p/2d)$ के बिंदु गिनिए, दो ध्रुवणों के साथ)
--- वही विधा-घनत्व जो तापीय विकिरण के सिद्धांत को
चाहिए होगा (\cref{ch:b2:thermal-radiation})। हर विधा का कोई गुणता गुणक
$Q$ होता है जिसे दीवार की हानियाँ तय करती हैं (\cref{ch:b2:waves-in-media}
का सतही प्रतिरोध), जो सूक्ष्मतरंग आवृत्तियों पर ताँबे के लिए प्रायः
$10^3$--$10^4$ होता है, और किसी अतिचालक कोटर के लिए $10^{10}$।
\end{proposition}

\begin{proof}
तीनों दिशाओं के अनुदिश \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंगें}, छहों दीवारों पर निस्पंदों के साथ;
और तरंग-सदिश $(m\pi/a, n\pi/b, p\pi/d)$ का मापांक $\omega/c$ है।
\end{proof}

\begin{example}[भट्ठियाँ, रडार, घड़ियाँ]\label{ex:b2:guided-waves:cavities}
$30 \times 30 \times 20$ cm$^3$ की किसी सूक्ष्मतरंग भट्ठी में \qty{2.45}{GHz} के पास
मैग्नेट्रॉन की आवृत्ति के कुछ ही प्रतिशत के भीतर दर्जनों विधाएँ होती हैं:
क्षेत्र \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंगों} का कोई अध्यारोपण है, जिसके अधिकतम
\qty{6}{cm} की दूरी पर हैं, और जिन्हें घूमती तश्तरी (तथा कोई "विधा-मथनी")
चिकना कर देती है। किसी रडार का मैग्नेट्रॉन स्वयं कोई \omterm{prop:b2:guided-waves:cavity}{कोटर अनुनादक} है; कोई कण
त्वरक अतिचालक कोटरों की कोई शृंखला है जिसकी TM विधा
गुच्छों को धकेलती है; और किसी सीज़ियम घड़ी के परमाणु \qty{9.19}{GHz} पर सधे
किसी सूक्ष्मतरंग कोटर से होकर गुज़रते हैं।
\end{example}

\section{किरण चित्र में प्रकाशिक तंतु}

\begin{proposition}[सोपान-अपवर्तनांक तंतु]\label{prop:b2:guided-waves:fibre}
\index{प्रकाशिक तंतु}\index{संख्यात्मक द्वारक}\index{विधात्मक परिक्षेपण}
$n_1$ अपवर्तनांक का कोई काँच का क्रोड, जिसके चारों ओर ज़रा कम अपवर्तनांक
$n_2$ का कोई आवरण है, क्रोड की सीमा पर \omterm{thm:b2:wave-interfaces:snell}{पूर्ण आंतरिक परावर्तन} से प्रकाश को
निर्देशित करता है। हवा से सिरे के फलक में अक्ष से $\alpha$ कोण पर घुसती
कोई किरण तभी फँस जाती है जब $\sin\alpha < \sqrt{n_1^2 - n_2^2}$ हो, अर्थात्
\emph{\omterm{prop:b2:guided-waves:fibre}{संख्यात्मक द्वारक}}; अलग-अलग कोणों की किरणें अलग-अलग लंबाइयाँ तय करती हैं, और
लंबाई $L$ के किसी तंतु में घुसती कोई स्पंद इतनी फैल जाती है
\[
\Delta t = \frac{n_1L}c\Bigl(\frac{n_1}{n_2} - 1\Bigr) ,
\]
अर्थात् \emph{\omterm{prop:b2:guided-waves:fibre}{विधात्मक परिक्षेपण}} --- $n_1
- n_2 = 0.01$ के लिए कोई \qty{50}{ns} प्रति किलोमीटर, जो किसी बहुविधा तंतु को
दस किलोमीटर पर कुछ मेगाबिट प्रति सेकंड तक सीमित कर देता है। किसी
\emph{एक-विधा} तंतु का क्रोड इतना पतला (\qty{9}{\micro m}) होता है कि केवल
एक विधा चलती है, ठीक उस निर्देशिका की तरह जो अपने पहले दो कटावों के बीच हो;
और तब जो बचता है वह काँच का वर्णिक परिक्षेपण है
(\cref{ch:b2:dispersion-wave-packets})।
\end{proposition}

\begin{proof}
क्रोड--आवरण अंतरापृष्ठ पर पूर्ण परावर्तन के लिए चाहिए $\sin\theta >
n_2/n_1$, उस कोण $\theta$ के लिए जो सीमा के अभिलंब से बनता है, अर्थात् $\cos
\theta' > n_2/n_1$, उस कोण $\theta'$ के लिए जो भीतर अक्ष से बनता है; और प्रवेश पर अपवर्तन
देता है $\sin\alpha = n_1\sin\theta'$, जिससे $\sin\alpha < n_1\sqrt{1
- n_2^2/n_1^2}$। सबसे तीखी फँसी किरण $L/\cos\theta' = Ln_1/n_2$ चलती है,
जबकि अक्षीय किरण $L$, और दोनों $c/n_1$ से।
\end{proof}

\begin{method}[निर्देशित-तरंग का हिसाब]\label{met:b2:guided-waves:method}
(1) दीवारें और उनकी शर्त पहचानिए (किसी चालक पर स्पर्शीय $\vect E = 0$;
किसी परावैद्युत के लिए पूर्ण परावर्तन)। (2) क्षेत्र को
निर्देशिका के अनुदिश किसी प्रगामी गुणक गुणा किसी अनुप्रस्थ \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग}
के रूप में लिखिए; सीमा-शर्तें अनुप्रस्थ तरंग-संख्या को क्वांटित कर देती हैं।
(3) परिक्षेपण-संबंध $k_g^2 = \omega^2/c^2 - k_\perp^2$ कटाव, $v_\varphi$, $v_g$,
$\lambda_g$ दे देता है। (4) कार्यकारी आवृत्ति पर चलती विधाएँ गिनिए; एक ही
का लक्ष्य रखिए। (5) किसी कोटर के लिए अनुदैर्घ्य शर्त जोड़िए और अनुनाद
आवृत्तियाँ निकालिए; और हानियों के लिए सतही प्रतिरोध काम में लीजिए।
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:guided-waves:4}
किसी तंतु का $n_1 = 1.48$, $n_2 = 1.46$ है। \omterm{prop:b2:guided-waves:fibre}{संख्यात्मक द्वारक} और स्वीकृति
अर्ध-कोण; प्रति किलोमीटर विधात्मक फैलाव; \qty{2}{km} पर अधिकतम बिट दर,

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:5}
समांतर-पट्टी विधा $n = 1$ के लिए: (क) फ़ैराडे के नियम से $\vect B$ निकालिए
और दिखाइए कि उसका कोई $z$ घटक $E$ के समपाद में है। (ख) हर पट्टी पर
सतही धारा (सीमा-संबंध)। (ग) $z$ के अनुदिश माध्य पॉयंटिंग सदिश
और प्रति एकांक चौड़ाई शक्ति; जाँचिए कि उसका अनुप्रस्थ
घटक माध्य में शून्य है। (घ) दिखाइए कि ऊर्जा-वेग, अर्थात् शक्ति
बटा प्रति एकांक लंबाई माध्य ऊर्जा, $v_g$ के बराबर है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:6}
\emph{कटाव से नीचे।} (क) \qty{2.45}{GHz} पर किसी भट्ठी के दरवाज़े की \qty{0.5}{mm} जाली
में \qty{1}{mm} का कोई छेद: उसे $a = \qty{1}{mm}$ चौड़ाई की कोई निर्देशिका मानिए;
शीट के पार क्षय-लंबाई और dB में \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन}
(\cref{ch:b2:waves-in-media} से तुलना कीजिए)। (ख) किसी परिरक्षित कमरे में
\qty{20}{cm} अनुप्रस्थ काट की कोई वायु-संचार नली: वह किस आवृत्ति तक रेडियो
रोकती है, और \qty{1}{m} पर \qty{100}{MHz} को कितना क्षीण करती है? (ग) कोई
"कटाव-से-परे निर्देशिका क्षीणक" \qty{1}{GHz} पर \qty{1}{cm} व्यास की नली काम में लेता है:
प्रति सेंटीमीटर \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} (आकलन के लिए $a =
\qty{1}{cm}$ के साथ पट्टी वाला सूत्र लीजिए)। (घ) \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} दीवार की चालकता से
स्वतंत्र क्यों है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:7}
\emph{शक्ति और भंजन।} \cref{exo:b2:guided-waves:2} की X-पट्टी निर्देशिका
\qty{10}{GHz} पर TE$_{10}$ ले चलती है। (क) $E_0 = \qty{1}{MV/m}$ के लिए शक्ति। (ख) हवा
\qty{3}{MV/m} पर टूट जाती है: अधिकतम शक्ति; रडार मेगावाट कैसे ले चलते
हैं (दाबन, गैसें)? (ग) दीवार की हानियाँ: चौड़ी दीवार पर सतही धारा
$E_0/\mu_0c$ की कोटि की है; \qty{10}{GHz} पर ताँबे के लिए $R_s = \qty{0.026}{\ohm}$
लेकर, प्रति मीटर हानि और dB/m में \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} आँकिए। (घ) किसी \omterm{prop:b2:dispersion-wave-packets:coax}{समाक्ष केबल}
के \qty{0.5}{dB/m} से तुलना कीजिए: इन आवृत्तियों पर निर्देशिकाएँ क्यों काम में ली जाती हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:8}
\emph{कोटर की गुणता।} \qty{10}{cm} भुजा का कोई ताँबे का घनाकार कोटर अपनी
सबसे नीची विधा में। (क) आवृत्ति। (ख) संचित ऊर्जा $\tfrac12\varepsilon_0E_0^2V/4$
है (स्वीकृत) और दीवार की हानि $\tfrac12R_sH_0^2\times$ क्षेत्रफल, जहाँ $H_0 \approx E_0/
\mu_0c$ और $R_s = 1/\gamma\delta$: गुणता गुणक $Q = \omega W/P$ --- उसे निकालिए।
(ग) अनुनाद की पट्ट-चौड़ाई $f/Q$; और गूँज-क्षय काल $Q/\omega$। (घ) किसी
अतिचालक कोटर का $R_s \sim \qty{1e-8}{\ohm}$ है: $Q$, और त्वरक उन्हें
क्यों काम में लेते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:guided-waves:9}
\emph{विधाएँ गिनना।} (क) दिखाइए कि आयतन $V$ के किसी बक्से की उन
विधाओं की संख्या जिनकी आवृत्ति $f$ से नीचे है, $N(f) = 8\pi Vf^3/3c^3$ है (बिंदु
$(m, n, p)$ किसी घन के लिए त्रिज्या $2fL/c$ के गोले के आठवें भाग में, और दो
ध्रुवण)। (ख) प्रति एकांक आयतन और आवृत्ति विधाओं की संख्या,
$\dd N/V\dd f = 8\pi f^2/c^3$। (ग) \qty{1}{m^3} का कोई कमरा: \qty{1}{GHz} पर प्रति हर्ट्ज़
विधाएँ; और \qty{500}{THz} (दृश्य प्रकाश) पर। (घ) किसी छोटे कोटर की विधाएँ
विरल और किसी बड़े कमरे की कोई सांतत्यक क्यों होती हैं --- और \qty{2.45}{GHz} पर
भट्ठी का क्या हाल है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:10}
\emph{TE$_{10}$ पूरा-पूरा।} निर्देशिका $0 < x < a$, $0 < y < b$ में: $\vect E =
E_0\sin(\pi x/a)\eu^{\iu(\omega t - k_gz)}\vect e_y$। (क) फ़ैराडे से $\underline B_x$
और $\underline B_z$ निकालिए; $\operatorname{div}\vect B = 0$ जाँचिए। (ख) निर्वात में
मैक्सवेल--ऐंपियर जाँचिए, और $k_g^2 = \omega^2/c^2 - (\pi/a)^2$ फिर से पाइए। (ग) चारों
दीवारों पर सतही धाराएँ (चौड़ी दीवारें $y = 0$, $b$ और सँकरी दीवारें $x = 0$,
$a$); कौन-सी धाराएँ $z$ के अनुदिश बहती हैं, कौन-सी पार? (घ) किसी चौड़ी
दीवार के बीचोंबीच $z$ के अनुदिश कटी कोई झिरी विकिरित नहीं करती, पर उसके
पार कटी करती है: समझाइए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:11}
\emph{परावैद्युत निर्देशिका।} अपवर्तनांक $n_1$ और मोटाई $a$ का कोई पट्ट किसी
माध्यम $n_2 < n_1$ में पूर्ण परावर्तन से प्रकाश निर्देशित करता है। (क) फलकों के
अभिलंब से $\theta$ कोण पर किसी किरण के लिए निर्देशन की शर्त यह है कि
पट्ट के पार एक चक्कर की कला, $2k_0n_1a\cos\theta - 2\varphi_r$ ($\varphi_r$
पूर्ण परावर्तन का कला-विस्थापन, यहाँ $\approx 0$ लिया गया), $2\pi$ की कोई
गुणज हो: $a = \qty{50}{\micro m}$, $\lambda = \qty{1.5}{\micro m}$,
$n_1 = 1.48$, $n_2 = 1.46$ के लिए विधाओं की संख्या (प्रति ध्रुवण विधा-गिनती
$\approx 2a\sqrt{n_1^2 - n_2^2}/\lambda$ स्वीकार कीजिए)। (ख) एक ही विधा के लिए मोटाई।
(ग) \qty{9}{\micro m} के किसी एक-विधा तंतु के क्रोड से तुलना कीजिए। (घ) परावैद्युत
निर्देशिका की सबसे नीची विधा का, धातु वाली के उलट, कोई कटाव क्यों नहीं हो सकता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:guided-waves:12}
\emph{क्रमिक-अपवर्तनांक तंतु।} \omterm{prop:b2:guided-waves:fibre}{विधात्मक परिक्षेपण} घटाने के लिए क्रोड का
अपवर्तनांक अक्ष से बाहर की ओर घटता जाता है, $n(r) = n_1\sqrt{1 - 2\Delta(r/a)^2}$, जिसमें
$\Delta \approx 0.01$। (क) गुणात्मक रूप से समझाइए कि अक्ष से हटी कोई किरण, लंबी
होकर भी, अक्षीय किरण जितना ही समय क्यों ले सकती है। (ख) किसी स्तरित माध्यम
में किरण-समीकरण, $n(r)\cos\theta(r) = \text{अचर}$, काम में लेकर दिखाइए कि
अक्ष से किसी छोटे कोण पर छोड़ी गई कोई किरण उसके इर्द-गिर्द ज्यावक्रीय
दोलन करती है ($n$ का दूसरी कोटि तक प्रसार कीजिए)। (ग) $a = \qty{25}{\micro m}$
के लिए दोलन का आवर्तकाल। (घ) बचा हुआ विधात्मक फैलाव
$n_1L\Delta^2/2c$ की कोटि का है: प्रति किलोमीटर मान, सोपान-अपवर्तनांक के
$n_1L\Delta/c$ के सामने।
\end{exercise}

\section{समस्या: भट्ठी का कोटर और प्रकाशिक तंतु}

\begin{problem}[{सप्ताहांत समस्या --- हर रसोई और हर शहर में निर्देशित-तरंग
की दो व्यवस्थाएँ: वह भट्ठी जो अप्रगामी तरंगों से पकाती है और
वह तंतु जो इंटरनेट ले चलता है}]
\label{pb:b2:guided-waves:1}

\medskip
\noindent\textbf{भाग I --- मैग्नेट्रॉन से कोटर तक।}
कोई मैग्नेट्रॉन \qty{2.45}{GHz} पर \qty{800}{W} किसी \omterm{prop:b2:guided-waves:rectangular}{आयताकार
तरंग-निर्देशिका} ($a = \qty{8.6}{cm}$, $b = \qty{4.3}{cm}$) में खिलाता है, जो भट्ठी के
कोटर ($30 \times 30 \times 20$ cm$^3$) में खुलती है; दीवारें इस्पात की, $\gamma = \qty{1e6}{S/m}$।
\begin{enumerate}
  \item निर्देशिका की TE$_{10}$, TE$_{20}$, TE$_{01}$ विधाओं की कटाव
        आवृत्तियाँ; जाँचिए कि \qty{2.45}{GHz} पर केवल TE$_{10}$ चलती है।
  \item \qty{2.45}{GHz} पर निर्देशिका-तरंगदैर्घ्य, कला और \omterm{prop:b2:dispersion-wave-packets:group}{समूह वेग}।
  \item \qty{800}{W} के लिए निर्देशिका में क्षेत्र-आयाम $E_0$ (TE$_{10}$ शक्ति
        सूत्र); हवा के भंजन क्षेत्र से तुलना कीजिए।
  \item चौड़ी दीवार पर सतही धारा का आयाम ($E_0/\mu_0c$ की
        कोटि का); इस्पात की \omterm{def:b2:dispersion-wave-packets:complex}{त्वचा गहराई} और सतही प्रतिरोध;
        और निर्देशिका के प्रति मीटर हानि।
  \item कोटर की विधाएँ: उन्हें सूचीबद्ध कीजिए जिनकी आवृत्तियाँ
        \qty{2.45}{GHz} के $3\%$ के भीतर हों ($(m, n, p)$ को $4$ तक आज़माइए); कितनी हुईं?
  \item ऐसी किसी एक विधा के गरम धब्बों की हर अक्ष के अनुदिश दूरी;
        तश्तरी घूमती क्यों है, और "विधा-मथनी" क्या है?
  \item भोजन से भरे कोटर का गुणता गुणक लगभग $200$ होता है: उसके
        उत्तर की पट्ट-चौड़ाई; और मैग्नेट्रॉन की आवृत्ति तथा विधाओं के बीच
        कोई बेमेल बहुत मायने क्यों नहीं रखता।
  \item ख़ाली कोटर का $Q \approx 3000$ है: \qty{800}{W} निवेश पर संचित ऊर्जा,
        माध्य ऊर्जा घनत्व, क्षेत्र-आयाम; प्रश्न 3 से और भंजन से तुलना
        कीजिए। भट्ठी ख़ाली चलने पर मैग्नेट्रॉन को क्या बचाता है?
  \item मैग्नेट्रॉन से कोटर तक निर्देशिका के \qty{20}{cm} पार करने में
        ऊर्जा को लगा समय।
  \item मैग्नेट्रॉन को अर्ध-तरंग दिष्टकृत आपूर्ति खिलाती है और वह हर
        मुख्य चक्र के केवल आधे भाग में उत्सर्जित करता है: \qty{800}{W} के
        माध्य के लिए झोंकों में शिखर शक्ति; और झोंकों के बीच कोटर
        क्या करता है ($Q \approx 200$ लीजिए)?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दरवाज़ा और कटाव।}
\begin{enumerate}[resume]
  \item दरवाज़े की जाली में \qty{1}{mm} की शीट पर \qty{1.5}{mm} के छेद हैं:
        किसी छेद में क्षय-लंबाई और शीट के पार dB में \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन}।
  \item दरवाज़े का किनारा किसी चतुर्थांश-तरंग "अवरोधक" से सील है:
        दरवाज़े की चौखट के चारों ओर $\lambda/4$ गहरी कोई झिरी, जो झिरी के मुँह पर
        कोई खुला परिपथ दिखाती है (\cref{ch:b2:wave-interfaces} याद कीजिए):
        \qty{2.45}{GHz} पर झिरी की गहराई।
  \item क़ानूनी रिसाव \qty{5}{cm} पर \qty{5}{mW/cm^2} है: कोई \qty{100}{cm^2} का
        दरवाज़ा लेने पर वह \qty{800}{W} के किस अंश के बराबर है?
  \item \qty{4}{cm} व्यास की कोई \qty{30}{cm} नली कोटर को हवा देती है: क्या वह
        \qty{2.45}{GHz} पर कटाव से नीचे है? उसकी लंबाई पर \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन}।
  \item जाली को दरवाज़े की चौखट से चारों ओर विद्युत् रूप से जोड़ा होना
        क्यों ज़रूरी है (जोड़ में कोई अंतराल किस तरह बरतता)?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- तंतु।}
कोई सोपान-अपवर्तनांक तंतु: क्रोड $n_1 = 1.465$, आवरण $n_2 = 1.460$, क्रोड
का व्यास \qty{50}{\micro m} (बहुविधा) या \qty{9}{\micro m} (एक-विधा);
$\lambda = \qty{1.55}{\micro m}$; \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} \qty{0.2}{dB/km}।
\begin{enumerate}[resume]
  \item \omterm{prop:b2:guided-waves:fibre}{संख्यात्मक द्वारक} और स्वीकृति कोण; क्या उसमें प्रकाश डालना
        आसान है?
  \item बहुविधा तंतु के लिए प्रति किलोमीटर विधात्मक फैलाव; \qty{10}{km}
        पर अधिकतम बिट दर (प्रति फैलाव-काल एक बिट)।
  \item बहुविधा तंतु की विधाओं की संख्या ($N \approx
        \tfrac12(\pi d\,\mathrm{NA}/\lambda)^2$ स्वीकार कीजिए); और एक-विधा वाले की ($d =
        \qty{9}{\micro m}$: दिखाइए कि वही सूत्र लगभग $1$ देता है)।
  \item एक-विधा तंतु में बचा हुआ फैलाव वर्णिक है:
        $\omega'' \approx \qty{-0.2}{m^2/s}$ (\cref{ch:b2:dispersion-wave-packets})
        और \qty{50}{ps} की स्पंदों के साथ, वह दूरी जिस पर कोई स्पंद दुगुनी हो
        जाए; और \qty{100}{km} पर बिट दर।
  \item \qty{100}{km} पर \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} dB में और किसी शक्ति-अनुपात के रूप में;
        निवेश \qty{1}{mW}: निर्गम शक्ति; और हर \qty{80}{km} पर प्रवर्धक लेने पर
        \qty{6000}{km} की किसी पार-अटलांटिक कड़ी के लिए कितने?
  \item \qty{1.55}{\micro m} पर आवृत्ति और फ़ोटॉन ऊर्जा; \qty{1}{mW} में प्रति
        सेकंड फ़ोटॉन, और \qty{10}{Gbit/s} पर प्रति बिट, निवेश पर
        और \qty{100}{km} के बाद।
  \item \qty{1.55}{\micro m} ही क्यों (सिलिका की दो हानि-क्रियाविधियों के बारे
        में सोचिए: रैले प्रकीर्णन, जो $1/\lambda^4$ की तरह गिरता है, और अवरक्त
        अवशोषण, जो \qty{1.6}{\micro m} के परे बढ़ता है)?
  \item हवा के सामने कटे किसी तंतु-सिरे पर फ्रेनेल परावर्तन: dB में
        हानि; और संयोजक अपवर्तनांक-मिलान जेल या माँजा हुआ भौतिक
        संपर्क क्यों काम में लेते हैं।
  \item कोई तंतु किसी हलके मोड़ पर विकिरित क्यों नहीं करता, और किसी
        तीखे मोड़ पर प्रकाश क्यों खो देता है (क्रोड की सीमा पर किरण
        के कोण के बारे में सोचिए)?
  \item इस समस्या की दोनों निर्देशिकाओं की तुलना कीजिए: तरंग को क्या
        क़ैद करता है, पट्ट-चौड़ाई को क्या सीमित करता है, और हानियाँ क्या।
\end{enumerate}
\end{problem}
