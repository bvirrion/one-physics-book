\chapter{कण विसरण}\label{ch:b2:particle-diffusion}

ठहरे हुए पानी में रंजक का कोई रवा डालिए और देखते रहिए: उसके चारों ओर कोई
रंगीन बादल बनता है, बढ़ता है, अपने किनारों पर नरम पड़ जाता है, और फैलता जाता
है --- किसी मिनट में किसी मिलीमीटर भर, किसी घंटे में किसी सेंटीमीटर भर,
किसी हफ़्ते में पूरे गिलास भर। रंजक को कोई धकेलता नहीं; उसे अणुओं की वह
लगातार धक्कमपेल ले चलती है, जो हर कण को किसी यादृच्छिक चाल पर भेज देती है
और, माध्य में, वहाँ से जहाँ बहुत हैं वहाँ जहाँ कम। यही \emph{विसरण} है, जो
परिवहनों में सबसे धीमा और सबसे सार्वभौम है: वह हर कोशिका को ऑक्सीजन खिलाता
है, हर ट्रांज़िस्टर को बोरॉन से मादित करता है, इस्पात को कार्बन से कठोर
करता है, और किसी इत्र को कमरा पार करने देता है --- हालाँकि, जैसा हम देखेंगे,
उतने समय में नहीं जितना कोई सोचे। यह अध्याय विसरण को उसका नियम (फ़िक), उसका
समीकरण (कणों के किसी संतुलन से), उसके अभिलाक्षणिक हल तथा काल-पैमाने
($L \sim \sqrt{Dt}$), और यादृच्छिक चाल में उसका सूक्ष्म मूल देता है --- जो
यह भी समझा देता है कि \omterm{thm:b2:particle-diffusion:equation}{विसरण समीकरण}, यांत्रिकी के हर समीकरण के उलट, समय की
दिशा क्यों जानता है। अगला अध्याय यह सब ऊष्मा के लिए फिर से काम में लेगा।

\section{कण घनत्व, फ्लक्स और फ़िक का नियम}

\begin{definition}[घनत्व और कण धारा]\label{def:b2:particle-diffusion:flux}
कणों की किसी जाति (अणु, आयन, किसी ठोस में परमाणु) के लिए
\emph{संख्या घनत्व}\index{संख्या घनत्व} $n(M,t)$ $M$ के इर्द-गिर्द प्रति
एकांक आयतन कणों की संख्या है (\unit{m^{-3}} में; और मोलर
सांद्रता $c = n/N_A$)। \emph{कण धारा-घनत्व}\index{कण धारा-घनत्व}
$\vect{j}_N$ वह सदिश है जिसके लिए किसी दिष्ट सतह-अवयव
$\dd\vect S$ को $\dd t$ में पार करते कणों की संख्या $\vect{j}_N\cdot\dd\vect S\,\dd t$ हो (\unit{m^{-2}.s^{-1}} में):
और किसी सतह $S$ से होकर \emph{कण फ्लक्स}
$\Phi_N = \iint_S \vect{j}_N\cdot\dd\vect S$ है, अर्थात् $S$ से होकर प्रति
सेकंड कणों की संख्या। $\vect v$ से चलते किसी तरल से ले जाए गए कणों के लिए
$\vect{j}_N = n\vect v$ (संवहन); और विसरण वह परिवहन है जो विरामावस्था के किसी
तरल में बचा रहता है।
\end{definition}

\begin{theorem}[फ़िक का नियम]\label{thm:b2:particle-diffusion:fick}
विरामावस्था के किसी माध्यम में, जहाँ घनत्व एकसमान न हो, कोई कण
धारा प्रकट होती है, जो घनत्व की \omterm{def:b2:maxwell-equations:operators}{प्रवणता} के समानुपाती और
उलटी होती है:
\[
\vect{j}_N = -D\,\vect{\operatorname{grad}}\,n ,
\]
जहाँ \emph{विसरण गुणांक}\index{विसरण गुणांक}
$D > 0$ (\unit{m^2/s} में) विसरित होती जाति पर, माध्यम पर
और ताप पर निर्भर करता है। कण घनत्व की \omterm{def:b2:maxwell-equations:operators}{प्रवणता} से \emph{नीचे} जाते
हैं, सघन से विरल इलाक़ों की ओर, ऐसी दर से जो $D$ तय करता है।
\end{theorem}

\begin{proof}
परिघटनात्मक (अनुभव का कोई नियम, ओम वाले की तरह): छोटी \omterm{def:b2:maxwell-equations:operators}{प्रवणताओं} के लिए
\omterm{def:b2:maxwell-equations:operators}{प्रवणता} में रैखिक, किसी समदैशिक माध्यम में समदैशिक, और
उस चिह्न के साथ जो अनुभव थोपता है। \cref{sec:b2:particle-diffusion:microscopic}
का यादृच्छिक-चाल प्रतिरूप उसे, और उसके साथ $D$ को, आणविक गति से निकाल देता है।
\end{proof}

\begin{example}[$D$ की कोटियाँ]\label{ex:b2:particle-diffusion:orders}
गैसें: $D \sim \qty{1e-5}{m^2/s}$ (हवा में जलवाष्प
\qty{2.5e-5}{m^2/s}, इत्र का कोई अणु \qty{5e-6}{m^2/s})। द्रव:
$D \sim \qty{1e-9}{m^2/s}$ (जल में ऑक्सीजन \qty{2e-9}{m^2/s}, चीनी
\qty{5e-10}{m^2/s}, कोई प्रोटीन \qty{1e-10}{m^2/s})। ठोस: नन्हा और
ताप के साथ तेज़ी से बढ़ता, $D = D_0\,\eu^{-E_{\text{सक्रियण}}/k_BT}$
(सिलिकॉन में बोरॉन: \qty{1100}{\celsius} पर \qty{1.5e-17}{m^2/s},
और कमरे के ताप पर नापने लायक़ भी नहीं --- और इसीलिए कोई ट्रांज़िस्टर,
एक बार बन जाने के बाद, दशकों तक अपना मादन-परिच्छेद बनाए रखता है)। गैस से द्रव तक चार कोटियाँ,
और द्रव से ठोस तक आठ या उससे अधिक।
\end{example}

\section{कणों का संतुलन और विसरण समीकरण}

\begin{theorem}[स्थानीय कण-संतुलन]\label{thm:b2:particle-diffusion:balance}
यदि कण न बनते हों न नष्ट होते हों,
\[
\frac{\partial n}{\partial t} + \operatorname{div}\vect{j}_N = 0 ;
\]
और प्रति एकांक आयतन तथा समय $\sigma$ कण बनाते किसी स्रोत के साथ
(कोई रासायनिक अभिक्रिया, कोई अवशोषण), $\partial_t n +
\operatorname{div}\vect{j}_N = \sigma$। एक विमा में (घनत्व और
धारा केवल $x$ पर निर्भर): $\partial_t n + \partial_x j_N = 0$।
\end{theorem}

\begin{proof}
$x$ और $x + \dd x$ के बीच का पट्ट लीजिए, जिसका \omterm{prop:b2:laser:gain}{अनुप्रस्थ काट} $S$ हो: उसमें
$n\,S\,\dd x$ कण हैं; और $\dd t$ में $j_N(x)S\,\dd t$ बाईं
सतह से भीतर आते हैं तथा $j_N(x + \dd x)S\,\dd t$ दाईं से बाहर जाते हैं, अतः
$\partial_t n\,S\,\dd x\,\dd t = -(j_N(x+\dd x) - j_N(x))S\,\dd t =
-\partial_x j_N\,\dd x\,S\,\dd t$। तीन विमाओं में किसी छोटे बक्से पर वही गिनती
\omterm{def:b2:maxwell-equations:operators}{अपसरण} दे देती है (प्रति एकांक आयतन किसी बंद सतह से बाहर फ्लक्स,
\cref{ch:b2:maxwell-equations}), या सीधे: किसी भी नियत
आयतन $V$ के लिए, ओस्ट्रोग्राद्स्की प्रमेय से $\dd/\dd t \iiint_V n\,\dd\tau = -\iint_{\partial V}
\vect{j}_N\cdot\dd\vect S = -\iiint_V \operatorname{div}\vect{j}_N\,\dd\tau$।
\end{proof}

\begin{theorem}[विसरण समीकरण]\label{thm:b2:particle-diffusion:equation}
फ़िक के नियम और संतुलन को जोड़ने पर, एकसमान $D$ के लिए,
\[
\frac{\partial n}{\partial t} = D\,\Delta n + \sigma ,
\qquad\text{एक विमा में}\quad
\frac{\partial n}{\partial t} = D\,\frac{\partial^2 n}{\partial x^2} + \sigma .
\]
यह \emph{विसरण समीकरण}\index{विसरण समीकरण} रैखिक है
(हल अध्यारोपित होते हैं), समय में पहली कोटि का और दिक् में दूसरी कोटि का
--- कोई तरंग समीकरण नहीं: उसकी कोई संचरण-चाल नहीं है, पर लंबाई और समय के बीच
कोई अभिलाक्षणिक संबंध है,
\[
L \sim \sqrt{D t}, \qquad t \sim \frac{L^2}{D} :
\]
अर्थात् दुगुनी दूरी भर विसरित होने में चौगुना समय लगता है।
\end{theorem}

\begin{proof}
$\partial_t n = -\operatorname{div}(-D\,\vect{\operatorname{grad}}\,n) +
\sigma = D\operatorname{div}\vect{\operatorname{grad}}\,n + \sigma = D\Delta
n + \sigma$। और पैमानों के लिए $n/t$ की $Dn/L^2$ से तुलना कीजिए।
\end{proof}

\begin{remark}[विसरण अनुत्क्रमणीय है]\label{rem:b2:particle-diffusion:irreversible}
$t$ की जगह $-t$ रख दीजिए, तरंग समीकरण $\partial_t^2 u = c^2
\partial_x^2 u$ में, और वह अपरिवर्तित रहता है: किसी तरंग की
फ़िल्म उलटी चलाई जाए तो भी वह कोई संभव तरंग है। \omterm{thm:b2:particle-diffusion:equation}{विसरण समीकरण} में वही कीजिए और बाईं
ओर का चिह्न पलट जाता है: $n(x,-t)$ कोई हल \emph{नहीं} है। फैलता हुआ कोई बादल
स्वाभाविक है; अपने आप सिमटकर किसी रवे में बदलता कोई बादल
नहीं --- अर्थात् विसरण के पास समय का कोई तीर है, दूसरे नियम वाला तीर
(\cref{ch:b2:heat-conduction} उसकी बनाई एंट्रॉपी निकालता है)। यह
समीकरण यह भी दिखाता है कि विसरण चिकना करता है: जहाँ $n$ स्थानीय रूप से कोई
अधिकतम हो ($\partial_x^2 n < 0$) वहाँ वह घटता है, और जहाँ न्यूनतम हो वहाँ
बढ़ता है।
\end{remark}

\begin{example}[कितना समय लगता है?]\label{ex:b2:particle-diffusion:times}
$t \sim L^2/D$। किसी कोशिका भर ऑक्सीजन, जल में $L = \qty{10}{\micro m}$:
$10^{-10}/2 \times 10^{-9} = \qty{0.05}{s}$ --- अर्थात् विसरण किसी कोशिका को
आसानी से खिला देता है; और \qty{1}{mm} ऊतक भर: \qty{500}{s} --- बहुत धीमा, और इसीलिए
कोई भी जीवित वस्तु बिना रक्त-वाहिकाओं के किसी मिलीमीटर के अंश से मोटी
नहीं होती। बिना हिलाई चाय के किसी प्याले भर चीनी, $L = \qty{5}{cm}$:
$2.5 \times 10^{-3}/5 \times 10^{-10} = 5 \times 10^6\,\unit{s}$, यानी दो महीने
--- हिला दीजिए। और अकेले विसरण से किसी कमरे भर कोई इत्र, हवा में \qty{5}{m}:
$25/10^{-5} = \qty{3e6}{s}$, यानी कोई महीना; पर वह आपको किसी मिनट में सूँघ आता है
क्योंकि हवा चलती है: \emph{संवहन} ले जाता है, और विसरण केवल
आख़िरी मिलीमीटर पूरे करता है।
\end{example}

\section{अभिलाक्षणिक हल}

\begin{proposition}[स्थायी दशा: झिल्ली]\label{prop:b2:particle-diffusion:membrane}
$n_1$ और $n_2$ पर थामे गए दो भंडारों के बीच, मोटाई $e$ और क्षेत्रफल $S$
की किसी झिल्ली के पार, स्थायी घनत्व रैखिक है,
\[
n(x) = n_1 + (n_2 - n_1)\frac{x}{e}, \qquad
j_N = D\,\frac{n_1 - n_2}{e}, \qquad
\Phi_N = \frac{n_1 - n_2}{R_{\text{विस}}}, \quad R_{\text{विस}} = \frac{e}{DS}:
\]
अर्थात् झिल्ली का कोई \emph{विसरणी प्रतिरोध} $e/DS$ है, ठीक वैसे जैसे किसी तार का
$\ell/\gamma S$ --- श्रेणी में प्रतिरोध जुड़ते हैं, और छोटे $D$ की कोई पतली झिल्ली
भी हावी हो सकती है।
\end{proposition}

\begin{proof}
स्थायी, एकविमीय, बिना किसी स्रोत के: $\partial_x^2 n = 0$, अतः $n$
एकघाती है; और $j_N = -D\,\dd n/\dd x$ एकसमान।
\end{proof}

\begin{proposition}[स्थायी दशा: गोलीय ज्यामिति]\label{prop:b2:particle-diffusion:sphere}
त्रिज्या $a$ के किसी गोले के इर्द-गिर्द, जिसकी सतह $n_0$ पर थामी हो, ऐसे किसी
माध्यम में जहाँ दूर $n \to 0$ हो, स्थायी घनत्व है
\[
n(r) = n_0\,\frac{a}{r}, \qquad
\Phi_N = 4\pi D a\,n_0 :
\]
अर्थात् छोड़ी गई (या, चिह्न उलटकर, सोखी गई) कुल धारा गोले की
\emph{त्रिज्या} के साथ बढ़ती है, उसकी सतह के साथ नहीं --- किसी विसरणी गर्त
की ज्यामिति वही है जो किसी स्थिरवैद्युत धारिता की।
\end{proposition}

\begin{proof}
$\Delta n = \frac{1}{r^2}\frac{\dd}{\dd r}(r^2\,\dd n/\dd r) = 0$ देता है
$n = A + B/r$; और सीमा-शर्तें $A = 0$, $B = n_0 a$ तय कर देती हैं; तब
$\Phi_N = -4\pi r^2 D\,\dd n/\dd r = 4\pi D n_0 a$ हर $r$ पर (स्थायी दशा में कोई
संचय नहीं)।
\end{proof}

\begin{proposition}[फैलता गाउसी]\label{prop:b2:particle-diffusion:gaussian}
प्रति एकांक क्षेत्रफल $N$ कण, जो $t = 0$ पर किसी अनंत माध्यम के तल $x = 0$ पर
छोड़े गए हों, ऐसे फैलते हैं
\[
n(x,t) = \frac{N}{\sqrt{4\pi D t}}\,\exp\Big(-\frac{x^2}{4Dt}\Big) :
\]
अर्थात् मानक विचलन $\sigma(t) = \sqrt{2Dt}$ का कोई गाउसी, जिसका
क्षेत्रफल $N$ अचर है (कण संरक्षित हैं) और जिसकी चोटी $\propto
1/\sqrt t$ घटती है। तीन विमाओं में, किसी बिंदु पर छोड़े गए $N$ कण
$n = N\,(4\pi Dt)^{-3/2}\exp(-r^2/4Dt)$ की तरह फैलते हैं, जिसमें $\langle r^2
\rangle = 6Dt$। और किसी अर्ध-आकाश की \emph{सतह} पर छोड़ी गई कणों की कोई स्पंद
(जिसे वे छोड़ नहीं सकते) $x > 0$ के लिए ऊपर वाले व्यंजक का
दुगुना देती है।
\end{proposition}

\begin{proof}
प्रतिस्थापित कीजिए: $u = x^2/4Dt$ के साथ, $\partial_t n = n\,(-1/2t + u/t)$ और
$D\,\partial_x^2 n = D\,n\,(-1/2Dt + x^2/4D^2t^2) = n\,(-1/2t + u/t)$;
बराबर। और सामान्यीकरण $\int n\,\dd x = N$ $\int
\eu^{-x^2/4Dt}\dd x = \sqrt{4\pi Dt}$ से निकलता है, तथा $\int x^2 n\,\dd x/N = 2Dt$।
अनन्यता (दी गई आरंभिक स्पंद के लिए) स्वीकृत है।
\end{proof}

\begin{proposition}[अचर सतह-सांद्रता]\label{prop:b2:particle-diffusion:erfc}
शुरू में ख़ाली कोई अर्ध-आकाश $x > 0$, जिसकी सतह घनत्व $n_0$ पर
$t = 0$ से थामी हो (किसी ठोस के संपर्क में कोई गैस जो उसमें
घुल जाती है), ऐसे भरता है
\[
n(x,t) = n_0\,\operatorname{erfc}\Big(\frac{x}{2\sqrt{Dt}}\Big),
\qquad
\operatorname{erfc}(u) = \frac{2}{\sqrt\pi}\int_u^{\infty}\eu^{-s^2}\dd s,
\]
अर्थात् \emph{पूरक त्रुटि फलन}, जो $1$ ($u =
0$ पर) से गिरकर $0.16$ ($u = 1$ पर) और $0.005$ ($u = 2$ पर) हो जाता है: प्रवेश गहराई
फिर $\sim 2\sqrt{Dt}$ है, और प्रति एकांक क्षेत्रफल सोखे गए कणों की
कुल संख्या $2n_0\sqrt{Dt/\pi}$ है।
\end{proposition}

\begin{proof}
$n = f(u)$ खोजिए, $u = x/2\sqrt{Dt}$ के साथ: समीकरण बन जाता है
$f'' + 2uf' = 0$, अतः $f' \propto \eu^{-u^2}$ और $f$ कोई त्रुटि
फलन है; और शर्तें $f(0) = n_0$, $f(\infty) = 0$
$\operatorname{erfc}$ चुन लेती हैं। सोखी गई संख्या $\int_0^\infty n\,\dd x =
n_0\,2\sqrt{Dt}\int_0^\infty\operatorname{erfc}(u)\dd u = 2n_0\sqrt{Dt/\pi}$ है।
\end{proof}

\begin{example}[इस्पात को कठोर करना]\label{ex:b2:particle-diffusion:carburising}
कार्बन-भरी किसी गैस में \qty{900}{\celsius} पर थामा इस्पात का कोई पुरज़ा
अपनी सतह पर कार्बन सोखता है; और गरम लोहे में कार्बन के लिए $D = \qty{5e-12}{m^2/s}$
के साथ, चार घंटे $2\sqrt{Dt} = \qty{0.5}{mm}$ देते हैं: अर्थात् किसी सख़्तजान क्रोड पर आधे
मिलीमीटर की कोई कठोर खाल --- गियरों और बेयरिंगों का \emph{सतह-कठोरण},
जिसका समय पूरक-त्रुटि-फलन परिच्छेद से तय किया जाता है।
\end{example}

\section{सूक्ष्म चित्र: यादृच्छिक चाल}\label{sec:b2:particle-diffusion:microscopic}

\begin{proposition}[यादृच्छिक चाल और विसरण गुणांक]\label{prop:b2:particle-diffusion:walk}
कोई कण जो लंबाई $\ell$ का कोई क़दम किसी यादृच्छिक दिशा में हर $\tau$ पर
लेता है (दो टक्करों के बीच कोई अणु), $N =
t/\tau$ क़दमों के बाद यह माध्य वर्ग विस्थापन रखता है
\[
\langle x^2\rangle = N\ell^2 = \frac{\ell^2}{\tau}\,t
\quad\text{(एक विमा)}, \qquad
\langle r^2\rangle = 3\langle x^2\rangle \quad\text{(तीन विमाएँ)}.
\]
गाउसी हल से तुलना करने पर, $\langle x^2\rangle = 2Dt$:
\[
D = \frac{\ell^2}{2\tau} \quad\text{(1D)}, \qquad
D = \frac{\ell^2}{6\tau} = \frac{\ell\,v^*}{6}\ \sim\ \frac{\ell\,v^*}{3}
\quad\text{(3D, जहाँ } v^* = \ell/\tau\text{; और गतिज सिद्धांत देता है } \tfrac13) .
\]
चलने वाले की दूरी $\sqrt t$ की तरह बढ़ती है, $t$ की तरह नहीं: दुगुना दूर जाने के लिए
उसे चार गुना क़दम चाहिए।
\end{proposition}

\begin{proof}
$x = \sum_i \epsilon_i\ell$, जिसमें स्वतंत्र $\epsilon_i = \pm1$ (1D):
$\langle x^2\rangle = \sum_{i,j}\langle\epsilon_i\epsilon_j\rangle\ell^2 =
N\ell^2$, क्योंकि तिरछे पद माध्य में शून्य हो जाते हैं। $x$ का बंटन, बड़े $N$ के लिए,
किसी गाउसी की ओर जाता है (केंद्रीय सीमा प्रमेय, जिसका
प्रमाण प्रायिकता के पाठ्यक्रम का है) --- और इसीलिए
स्थूल नियम \omterm{thm:b2:particle-diffusion:equation}{विसरण समीकरण} है; और फ़िक का नियम ख़ुद
किसी तल को दोनों ओर से पार करते चलने वालों की गिनती से निकल आता है: $j_N \approx
-\tfrac12\ell v^*\,\partial_x n$ (और अधिक सावधान माध्य तीन विमाओं में $1/3$ देता है)।
\end{proof}

कमरे के ताप पर हवा: माध्य मुक्त पथ $\ell \approx \qty{70}{nm}$,
माध्य चाल $v^* \approx \qty{500}{m/s}$: $D \approx \ell v^*/3 =
\qty{1.2e-5}{m^2/s}$, जैसा नापा गया है। \qty{1}{m} भर विसरित होने के लिए किसी अणु को
$N = (L/\ell)^2 = 2 \times 10^{14}$ टक्करें चाहिए, अर्थात् $N\ell/v^*
\approx \qty{3e4}{s}$ --- यानी आठ घंटे, उस यात्रा के लिए जो वह सीधे उड़कर
\qty{2}{ms} में कर लेता। किसी द्रव में, अणुओं से धकेला गया त्रिज्या $a$ का कोई
गोला स्टोक्स--आइंस्टाइन संबंध $D =
k_BT/6\pi\eta a$ मानता है (आइंस्टाइन 1905): चीनी के लिए, जल में $a \approx \qty{0.4}{nm}$,
$D = \qty{5e-10}{m^2/s}$; और \qty{1}{\micro m} के किसी कण के लिए,
$\qty{4e-13}{m^2/s}$, अर्थात् प्रति सेकंड किसी माइक्रोमीटर का विस्थापन ---
वही ब्राउनी गति जिसे पेरैं ने आवोगाद्रो संख्या गिनने के लिए नापा। किसी
ठोस में कोई परमाणु किसी पड़ोसी स्थान पर तभी कूदता है जब कोई तापीय
उतार-चढ़ाव सक्रियण ऊर्जा दे दे: $D = D_0\,\eu^{-E_{\text{सक्रियण}}/k_BT}$,
जो \qty{1000}{\celsius} के पास हर कुछ दसियों केल्विन पर दुगुना हो जाता है।
\end{example}

\begin{method}[विसरण के आकलन]\label{meth:b2:particle-diffusion:method}
(1) $D$ पहचानिए (गैस $10^{-5}$, द्रव $10^{-9}$, ठोस आरेनियस)।
(2) समय या दूरी: $t \sim L^2/D$, $L \sim \sqrt{Dt}$ (गाउसी चौड़ाई के लिए यथार्थ
$\sqrt{2Dt}$, और पूरक-त्रुटि-फलन गहराई के लिए $2\sqrt{Dt}$)।
(3) स्थायी: किसी पट्ट में रैखिक ($R_{\text{विस}} = e/DS$), किसी गोले के इर्द-गिर्द
$1/r$ ($\Phi = 4\pi D a\,\Delta n$), और किसी बेलन के इर्द-गिर्द $\ln r$। (4)
क्षणिक: किसी स्पंद के लिए गाउसी, थामी गई सतह के लिए पूरक त्रुटि फलन; अध्यारोपित कीजिए।
(5) पूछिए कि कहीं संवहन हावी तो नहीं। (6) सूक्ष्म जाँच:
$D \sim \ell v^*/3$।
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:2}
\qty{1}{\micro m} मोटी और \qty{1}{cm^2} क्षेत्रफल की कोई झिल्ली, जिसके भीतर $D =

शुद्ध जल से अलग करती है। धारा-घनत्व, मोलर फ्लक्स और प्रति सेकंड अणुओं की
संख्या; विसरणी प्रतिरोध; और ऐसी दो झिल्लियाँ श्रेणी में हों तो।

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:3}
रंजक की कोई पतली परत ($N = \qty{1e20}{m^{-2}}$) जल में छोड़ी जाती है,
$D = \qty{1e-9}{m^2/s}$। \qty{1}{s}, \qty{1}{h} और \num{1} दिन के बाद
चौड़ाई $\sqrt{2Dt}$ और चोटी का घनत्व। चोटी \qty{1}{s} वाले अपने मान का
$1\%$ कब रह जाती है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:4}
हवा के अणु: $\ell = \qty{70}{nm}$, $v^* = \qty{500}{m/s}$।
$D$ आँकिए; \qty{1}{m} भर विसरित होने के लिए टक्करों की संख्या और समय; और
सीधी उड़ान से तुलना कीजिए। जल में किसी अणु के लिए वही ($\ell \approx

ठीक-ठीक क्यों नहीं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:5}
\emph{किसी ऊतक में ऑक्सीजन।} मोटाई $2a$ का ऊतक का कोई पट्ट
एकसमान दर $q$ (प्रति एकांक आयतन) से ऑक्सीजन खपाता है; और उसके दोनों फलक
$n_0$ पर थामे हैं। (क) गर्त के साथ स्थायी समीकरण लिखिए और उसे
हल कीजिए। (ख) ऑक्सीजन के केंद्र तक पहुँचने की शर्त। (ग) $D = \qty{2e-9}{m^2/s}$,
$n_0 = \qty{0.2}{mol/m^3}$, $q = \qty{0.01}{mol/m^3/s}$: अधिकतम $a$। (घ)
केशिकाओं की दूरी पर निष्कर्ष निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:6}
\emph{घुलता हुआ दाना।} ठहरे हुए जल में त्रिज्या $a = \qty{1}{mm}$ का चीनी का कोई
गोला अपनी सतह को संतृप्ति घनत्व $n_{\text{संत}}
= \qty{5800}{mol/m^3}$ पर थामे रखता है; दूर जल शुद्ध है; और $D =
\qty{5e-10}{m^2/s}$। (क) स्थायी परिच्छेद और कुल धारा। (ख)
दाने में मोलों की संख्या (घनत्व \qty{1590}{kg/m^3}, मोलर द्रव्यमान
\qty{342}{g/mol}) और घुलने का समय (धारा को अचर लीजिए)।
(ग) असली समय अधिक क्यों है, और हिलाने से इतनी मदद क्यों मिलती है?
(घ) दिखाइए कि किसी बहुत लंबे बेलन के लिए स्थायी परिच्छेद
लघुगणकीय है और किसी अनंत माध्यम में समस्या का कोई हल नहीं है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:7}
\emph{संरक्षण और अनुत्क्रमणीयता।} (क) समीकरण से दिखाइए
कि $\int n\,\dd x$ अचर है और यह कि $\dd\langle x^2\rangle/\dd t =
2D$ (खंडशः समाकलन कीजिए, अनंत पर $n \to 0$)। (ख) दिखाइए कि
$n(x,-t)$ समीकरण नहीं मानता। (ग) $t < 0$ के लिए
गाउसी हल का क्या होता है? (घ) दिखाइए कि $\int n^2\,\dd x$ केवल
घट ही सकता है: विसरण चपटा करता है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:8}
\emph{कार्बनन।} गरम लोहे में कार्बन, \qty{900}{\celsius} पर $D = \qty{5e-12}{m^2/s}$;
और सतह $n_0$ पर थामी। (क) जाँचिए कि $n_0
\operatorname{erfc}(x/2\sqrt{Dt})$ समीकरण को हल करता है। (ख) \qty{4}{h} के बाद वह गहराई
जिस पर $n = 0.1\,n_0$ ($\operatorname{erfc}(1.16) = 0.1$)।
(ग) दुगुनी गहराई के लिए समय। (घ) \qty{950}{\celsius} पर $D$ $2.5$
गुना बड़ा है: बचा हुआ समय।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:9}
\emph{स्टोक्स--आइंस्टाइन।} $D = k_BT/6\pi\eta a$, जल $\eta =
\qty{1e-3}{Pa.s}$, \qty{300}{K}। (क) $D$, $a = \qty{1}{\micro m}$ के लिए, और
\qty{1}{s}, \qty{1}{min} में वर्ग-माध्य-मूल विस्थापन। (ख) किसी प्रोटीन के लिए $D$,
$a = \qty{3}{nm}$। (ग) चीनी का $D = \qty{5e-10}{m^2/s}$ है:
प्रभावी त्रिज्या। (घ) किसी सूक्ष्मदर्शी के नीचे किसी कण का $\langle x^2\rangle$
नापना आवोगाद्रो संख्या कैसे दे देता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:10}
\emph{किसी झिल्ली का समय-विलंब।} मोटाई $e$ की कोई झिल्ली, जो शुरू में
ख़ाली है, एक फलक पर $n_1$ के सामने रखी जाती है $t = 0$ पर, और दूसरा फलक
$0$ पर थामा जाता है। (क) अंतिम स्थायी फ्लक्स क्या है? (ख) तर्क दीजिए कि दूर वाले फलक पर
फ्लक्स $\sim e^2/D$ के किसी समय भर चढ़ता है (यथार्थ विलंब
$e^2/6D$ है)। (ग) त्वचा की बाहरी परत से होकर $e = \qty{20}{\micro m}$, $D =
\qty{1e-13}{m^2/s}$ वाला कोई औषधि-पट्ट: विलंब-काल। (घ) विलंब
$D$ नापने के लिए और स्थायी फ्लक्स $D \times$ विलेयता नापने के लिए क्यों
काम का है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:11}
\emph{पूर्ण अवशोषक।} त्रिज्या $a$ का कोई गोला हर उस कण को सोख लेता है
जो उसे छू ले, और दूर माध्य $n_\infty$ पर है। (क)
स्थायी परिच्छेद और पकड़ी गई कुल धारा। (ख) कोई जीवाणु, $a =
\qty{1}{\micro m}$, $n_\infty = \qty{6e20}{m^{-3}}$ वाले किसी चीनी-विलयन में,
$D = \qty{5e-10}{m^2/s}$: प्रति सेकंड पकड़े गए अणु। (ग) उसकी
सतह $N_{\text{ग्राही}}$ छोटे अवशोषी ग्राहियों से ढकी है, जिनकी त्रिज्या $s$ है,
और बाक़ी परावर्ती: हर ग्राही अकेला $\approx
4Ds n_\infty$ पकड़ता है (कोई चकती); दिखाइए कि $N_{\text{ग्राही}} s \gg a$ होते ही पूरी कोशिका लगभग
अधिकतम पकड़ लेती है, अर्थात् अपनी सतह के किसी नन्हे अंश के ढके होने पर।
(घ) टिप्पणी: कोशिकाएँ हज़ारों अलग-अलग ग्राही क्यों रख सकती हैं।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:12}
\emph{उसे किसी जाल पर हल करना।} दिक् को आकार $\Delta x$ की कोठरियों में
और समय को क़दमों $\Delta t$ में बाँटिए; और
$n_i^{k+1} = n_i^k + \alpha\,(n_{i+1}^k
- 2n_i^k + n_{i-1}^k)$ लिखिए, $\alpha = D\Delta t/\Delta x^2$ के साथ। (क) उसे समीकरण से न्यायसंगत
ठहराइए। (ख) $\alpha =
1/2$ पर उसकी व्याख्या किसी यादृच्छिक चाल की तरह कीजिए। (ग) दिखाइए कि $\alpha > 1/2$ के लिए
कोठरी-दर-कोठरी $+,-,+,-$ बदलता कोई घनत्व बढ़ता जाता है: अस्थिरता। (घ) नीचे की समस्या
के बोरॉन परिच्छेद के लिए ($D = \qty{1.5e-17}{m^2/s}$, $\Delta x = \qty{10}{nm}$),
सबसे बड़ा स्थायी $\Delta t$ और किसी एक घंटे के लिए क़दमों की संख्या।
\end{exercise}

\section{समस्या: कोई मादित वेफ़र और कोई महकता कमरा}

\begin{problem}[{सप्ताहांत समस्या --- किसी ठोस में और किसी गैस में विसरण}]\label{pb:b2:particle-diffusion:1}
\textbf{भाग I --- सिलिकॉन में बोरॉन को भीतर धकेलना।}
बोरॉन सिलिकॉन में $D = D_0\,\eu^{-E_{\text{सक्रियण}}/k_BT}$ से विसरित होता है, $D_0 =
\qty{7.6e-5}{m^2/s}$, $E_{\text{सक्रियण}} = \qty{3.46}{eV}$; $k_B =
\qty{8.62e-5}{eV/K}$। बोरॉन की कोई मात्रा $Q = \qty{1e18}{m^{-2}}$ किसी वेफ़र
की सतह पर बहुत पतली किसी परत में जमा की गई है, और उस वेफ़र का पृष्ठभूमि मादन
$n_B = \qty{1e21}{m^{-3}}$ है; फिर उसे \qty{1100}{\celsius} पर किसी एक घंटे
के लिए गरम किया जाता है (``भीतर धकेला जाता है'')। सतह
बोरॉन को परावर्तित करती है (कोई निकास नहीं)।
\begin{enumerate}
  \item \qty{1100}{\celsius} पर और \qty{1000}{\celsius} पर $D$ निकालिए।
        इन \qty{100}{K} भर वह किस गुणक से बदलता है?
  \item भीतर धकेलने के बाद परिच्छेद $n(x,t) = (Q/\sqrt{\pi
        Dt})\exp(-x^2/4Dt)$ क्यों है, और
        \cref{prop:b2:particle-diffusion:gaussian} वाला गाउसी क्यों नहीं? जाँचिए कि
        $x > 0$ पर उसका समाकल $Q$ है।
  \item किसी एक घंटे के बाद सतह-सांद्रता।
  \item \emph{संधि गहराई} $x_j$ वहाँ है जहाँ $n = n_B$: उसे निकालिए।
  \item यदि भीतर धकेलना चार घंटे चले तो $x_j$ कैसे बदलता है?
        (सावधान: सतह-सांद्रता भी बदलती है।)
  \item ताप में \qty{10}{K} की किसी त्रुटि के लिए $D$ का सापेक्ष बदलाव;
        और $x_j$ को \qty{1}{\%} तक रखने के लिए ज़रूरी ताप-नियंत्रण।
  \item कमरे के ताप पर $D$: परिच्छेद के किसी एक परमाणविक दूरी भर खिसकने का
        समय आँकिए, और निष्कर्ष निकालिए।
\end{enumerate}

\textbf{भाग II --- पूर्व-निक्षेपण।}
वह मात्रा ख़ुद \qty{950}{\celsius} पर किसी ऐसी गैस से डाली गई थी जो
सतह को \qty{30}{min} तक विलेयता की सीमा $n_0 = \qty{2e26}{m^{-3}}$ पर
थामे रखती है।
\begin{enumerate}[resume]
  \item \qty{950}{\celsius} पर $D$ और \qty{30}{min} के लिए $\sqrt{Dt}$।
  \item पूर्व-निक्षेपण के अंत में परिच्छेद; और वह गहराई जिस पर $n =
        n_B$ ($\operatorname{erfc}(3.2) \approx 6 \times 10^{-6}$)।
  \item डाली गई मात्रा; भाग I के $Q$ से तुलना कीजिए।
  \item पूर्व-निक्षेपण किसी नीचे ताप पर और भीतर धकेलना किसी ऊँचे पर
        क्यों किया जाता है?
  \item न्यायसंगत ठहराइए कि भीतर धकेलने के दौरान पूर्व-निक्षेपित परत को
        अनंत रूप से पतली माना जा सकता है।
\end{enumerate}

\textbf{भाग III --- किसी कमरे में कोई इत्र।}
इत्र की कोई बूँद (\qty{1}{mg}, मोलर द्रव्यमान \qty{150}{g/mol}) \qty{300}{K},
\qty{1}{bar} के किसी ठहरे हुए कमरे के कोने में तुरंत वाष्पित हो जाती है।
इत्र के अणु का हवा के साथ टक्कर-व्यास $d =
\qty{0.5}{nm}$ और हवा का \omterm{def:b2:particle-diffusion:flux}{संख्या घनत्व} $n_{\text{वायु}} = P/k_BT$ लीजिए।
\begin{enumerate}[resume]
  \item छोड़े गए अणुओं की संख्या; और हवा का \omterm{def:b2:particle-diffusion:flux}{संख्या घनत्व}।
  \item इत्र के अणु का माध्य मुक्त पथ $\ell = 1/(\sqrt2\,\pi d^2 n_{\text{वायु}})$,
        और उसकी माध्य चाल $v^* = \sqrt{8RT/\pi M}$।
  \item $D = \ell v^*/3$ आँकिए।
  \item \qty{5}{m} भर विसरित होने का समय; और यदि हवा
        \qty{0.1}{m/s} से बहे तो?
  \item बादल के लिए त्रिविम गाउसी लिखिए (कोई कोना:
        दीवारें परावर्तित करती हैं, $8$ से गुणा कीजिए)। \qty{1}{m} दूर किसी बिंदु पर,
        किस समय सांद्रता अधिकतम है?
  \item वहाँ अधिकतम \omterm{def:b2:particle-diffusion:flux}{संख्या घनत्व}; \qty{1e13}{m^{-3}} की किसी बोध-देहली
        से तुलना कीजिए।
  \item बादल का किनारा: किसी एक दिन के बाद किस दूरी पर घनत्व उसके
        केंद्रीय मान का $10^{-6}$ है?
  \item इत्र का कोई अणु किसी एक दिन में कितनी टक्करें झेलता है।
\end{enumerate}

\textbf{भाग IV --- चलने वाला और समय का तीर।}
\begin{enumerate}[resume]
  \item कोई चलने वाला किसी रेखा पर $N$ क़दम $\pm\ell$ लेता है: माध्य और माध्य
        वर्ग विस्थापन; दिखाइए कि $\langle x^2\rangle = 2Dt$
        $D = \ell^2/2\tau$ पहचान लेता है।
  \item $N = 2$, $4$, $6$ क़दमों के बाद चलने वाले के ठीक अपने आरंभ पर
        लौटने की प्रायिकता; और प्रवृत्ति पर टिप्पणी कीजिए।
  \item किसी विसरित होते बादल की फ़िल्म बनाइए और उसे उलटा चलाइए: आपको क्या
        दिखता है, और कौन-सा समीकरण टूटता है?
  \item किसी एक अनुच्छेद में समझाइए कि अणुओं की उत्क्रमणीय गति
        अनुत्क्रमणीय \omterm{thm:b2:particle-diffusion:equation}{विसरण समीकरण} कैसे बना देती है।
  \item समेटिए: लंबाई का पैमाना, समय का पैमाना, और वह एक
        राशि जो किसी गैस, किसी द्रव और किसी गरम ठोस में फ़र्क़ करती है।
\end{enumerate}
\end{problem}
