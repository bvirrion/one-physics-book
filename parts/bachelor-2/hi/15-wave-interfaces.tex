\chapter{अंतरापृष्ठों पर परावर्तन और पारगमन}\label{ch:b2:wave-interfaces}

प्रतिध्वनि-लेखी करने वाला जाँच-यंत्र और त्वचा के बीच कोई जेल दबा देता है:
उसके बिना ध्वनि हवा की झिल्ली से टकराकर लौट जाती और शरीर में कभी न
घुसती। किसी कैमरे का लेंस तरंगदैर्घ्य के चौथाई भर मोटी किसी झिल्ली से
लेपित होता है, और उसकी सतहें परावर्तित करना छोड़ देती हैं। कोई रेडियो
तरंग आयनमंडल से मिलकर लौट पड़ती है; प्रकाश चाँदी के दर्पण से मिलकर लौट
पड़ता है; कोई स्पंद किसी केबल के सिरे तक पहुँचती है और, वहाँ जो जुड़ा हो
उसके अनुसार, उलटी या सीधी लौट आती है। हर तरंग, माध्यम का बदलना पाकर,
किसी परावर्तित और किसी पारगत भाग में बँट जाती है, और एक ही विचार हिस्से
तय करता है: अंतरापृष्ठ पर \emph{सांतत्य शर्तें}, जो दोनों माध्यमों की
प्रतिबाधाओं के साथ लिखी जाती हैं। यह अध्याय उसे ध्वनि पर, दो
परावैद्युतों के बीच के प्रकाश पर, किसी \omterm{def:b2:waves-in-media:plasma}{प्लाज़्मा} पर, किसी धातु पर और किसी
केबल पर लगाता है, और उससे परावर्तन तथा अपवर्तन के नियम और वह एक कोण
निकालता है जिस पर काँच कुछ भी परावर्तित नहीं करता।

\section{अभिलंब आपतन: प्रतिबाधा-नियम}

\begin{theorem}[अभिलंब आपतन पर परावर्तन और पारगमन]\label{thm:b2:wave-interfaces:normal}
\index{परावर्तन गुणांक}\index{पारगमन गुणांक}\index{प्रतिबाधा-मिलान}
कोई समतल तरंग माध्यम 1 में माध्यम 2 के साथ बने समतल अंतरापृष्ठ $x = 0$ की
ओर चलती है; हर माध्यम किसी प्रतिबाधा $Z_i$ से पहचाना जाता है (अंतरापृष्ठ
पर सतत रहने वाली दो क्षेत्र-राशियों का अनुपात --- ध्वनि के लिए दाब और
वेग, प्रकाश के लिए $E$ और $H = B/\mu_0$, किसी लाइन के लिए वोल्टता और धारा)।
$x = 0$ पर दोनों राशियों का सांतत्य, "बल जैसी" राशि (दाब, $E$, $v$) के लिए, देता है
\[
r = \frac{Z_2 - Z_1}{Z_2 + Z_1} , \qquad t = \frac{2Z_2}{Z_2 + Z_1} ,
\qquad R = r^2 , \quad T = \frac{4Z_1Z_2}{(Z_1 + Z_2)^2} , \quad R + T = 1 .
\]
("बहाव जैसी" राशि --- वेग, $H$, धारा --- के लिए $r$ चिह्न
बदल देता है।) $Z_1 = Z_2$ पर कुछ भी परावर्तित नहीं होता (\emph{मिलान वाले} माध्यम);
और $Z_2 \gg Z_1$ ($r \to +1$) या $Z_2 \ll Z_1$ ($r \to -1$) पर लगभग सब कुछ।
\end{theorem}

\begin{proof}
आपाती $p_i = A\cos(\omega t - k_1x)$, परावर्तित $p_r = rA\cos(\omega t + k_1x)$,
पारगत $p_t = tA\cos(\omega t - k_2x)$, तीनों की आवृत्ति वही। इनसे जुड़ी
बहाव-राशियाँ $p/Z_1$, $-p_r/Z_1$ (पीछे जाती कोई तरंग
$v = -p/Z$ ले चलती है) और $p_t/Z_2$ हैं। $x = 0$ पर सांतत्य: $1 + r = t$ और $(1 - r)/Z_1
= t/Z_2$; हल कीजिए। ऊर्जाएँ: तीव्रताएँ $p^2/Z$ हैं, अतः $R = r^2$ और
$T = t^2Z_1/Z_2$।
\end{proof}

\begin{example}[तीन अंतरापृष्ठों पर ध्वनि]\label{ex:b2:wave-interfaces:sound}
वायु ($Z = \qty{410}{kg.m^{-2}.s^{-1}}$) से जल ($1.5 \times 10^6$): $r = 0.9995$,
$T = 1.1 \times 10^{-3}$ --- अर्थात् $99.9\%$ परावर्तित; कोई तैराक ऊपर की दुनिया का
लगभग कुछ नहीं सुनता, और पराश्रव्य जाँच-यंत्र को अपना जेल चाहिए। ऊतक
($1.6 \times 10^6$) से अस्थि ($7 \times 10^6$): $r = 0.63$, $T = 0.6$। वसा ($1.38 \times
10^6$) से पेशी ($1.70 \times 10^6$): $r = 0.10$, $R = 1\%$ --- वही मंद प्रतिध्वनियाँ जिनसे
कोई चित्र बनता है। दीवार से बँधी कोई डोरी ($Z_2 \to \infty$): बल के लिए $r = +1$,
वेग के लिए $-1$, और स्पंद उलटी होकर लौटती है
(\cref{ch:b2:waves-on-strings})।
\end{example}

\begin{proposition}[अभिलंब आपतन पर प्रकाश; परावर्तन-रोधी लेप]\label{prop:b2:wave-interfaces:optics}
\index{परावर्तकता (अभिलंब आपतन)}\index{परावर्तन-रोधी लेप}\index{चतुर्थांश-तरंग परत}
$n_1$, $n_2$ अपवर्तनांकों के दो पारदर्शी माध्यमों के बीच के प्रकाश के लिए (किसी
परावैद्युत की प्रतिबाधा $Z = \sqrt{\mu_0/\varepsilon_0}/n$ है, जो
$n$ के व्युत्क्रमानुपाती है), विद्युत् क्षेत्र के लिए
\[
r = \frac{n_1 - n_2}{n_1 + n_2} , \qquad R = \Bigl(\frac{n_1 - n_2}{n_1 + n_2}\Bigr)^2 :
\]
हर वायु--काँच सतह पर $4\%$, जल के लिए $2\%$, हीरे के लिए $17\%$। माध्यमों के बीच
$n = \sqrt{n_1n_2}$ अपवर्तनांक और $\lambda/4n$ मोटाई की कोई परत
$\lambda$ पर परावर्तन रद्द कर देती है (\emph{चतुर्थांश-तरंग \omterm{prop:b2:wave-interfaces:optics}{परावर्तन-रोधी
लेप}}): उसकी दोनों सतहों पर परावर्तित तरंगों के आयाम बराबर
और कलाएँ उलटी होती हैं। वही परत ऊँचे अपवर्तनांक के साथ, या
एकांतर \omterm{prop:b2:wave-interfaces:optics}{चतुर्थांश-तरंग परतों} का कोई ढेर, उलटा करता है और कोई ऐसा
दर्पण बना देता है जो $99.99\%$ परावर्तित करे --- लेज़रों के दर्पण।
\end{proposition}

\begin{proof}
स्पर्शीय $\vect E$ और $\vect B$ का सांतत्य (कोई सतही
धारा नहीं): $E_i + E_r = E_t$ और $B_i - B_r = B_t$, जहाँ हर माध्यम में $B = nE/c$,
अर्थात् $Z \propto 1/n$ के साथ वही प्रतिबाधा-नियम। लेप: दोनों
परावर्तित आयाम $(n_1 - n)/(n_1 + n)$ और $(n - n_2)/(n + n_2)$ हैं,
जो $n^2 = n_1n_2$ पर बराबर होते हैं; और परत में एक चक्कर कला
$2\times2\pi n(\lambda/4n)/\lambda = \pi$ जोड़ देता है, अतः वे रद्द हो जाते हैं (बहु-परावर्तन,
छोटे, \cref{exo:b2:wave-interfaces:5} के यथार्थ निर्वाह में शामिल)।
\end{proof}

\section{तिरछा आपतन: देकार्त के नियम और ब्रूस्टर कोण}

\begin{theorem}[परावर्तन और अपवर्तन के नियम]\label{thm:b2:wave-interfaces:snell}
\index{स्नेल--देकार्त नियम}\index{पूर्ण आंतरिक परावर्तन}\index{क्षयी तरंग (पूर्ण परावर्तन)}
तरंग-सदिश $\vect k_i$ की कोई समतल तरंग दो माध्यमों के बीच के समतल
अंतरापृष्ठ से मिलती है, जिनमें तरंगें $v_1$ और $v_2$ से चलती हैं। सांतत्य
शर्तें अंतरापृष्ठ के हर बिंदु पर और हर समय लागू होनी चाहिए:
अतः परावर्तित और पारगत तरंगों की आवृत्ति और
\emph{स्पर्शीय तरंग-सदिश} आपाती तरंग वाले ही होने चाहिए। इससे
तीनों तरंग-सदिश समतलीय होते हैं (आपतन तल), परावर्तन कोण
आपतन कोण के बराबर होता है, और
\[
\frac{\sin\theta_1}{v_1} = \frac{\sin\theta_2}{v_2} \qquad (n_1\sin\theta_1 = n_2\sin\theta_2 \text{ प्रकाश के लिए}) .
\]
जब $v_2 > v_1$ और $\sin\theta_1 > v_1/v_2$ हो तो कोई वास्तविक $\theta_2$ नहीं होता:
\emph{\omterm{thm:b2:wave-interfaces:snell}{पूर्ण आंतरिक परावर्तन}}; तब माध्यम 2 में क्षेत्र कोई
\emph{\omterm{def:b2:dispersion-wave-packets:complex}{क्षयी तरंग}} है, $\propto\eu^{-x/\ell}\eu^{\iu(\omega t - k_ty)}$, जो अंतरापृष्ठ के
अनुदिश चलती है और उससे दूर $\ell = \lambda_2/2\pi
\sqrt{\sin^2\theta_1(v_2/v_1)^2 - 1}$ पर, अर्थात् किसी तरंगदैर्घ्य के अंश भर में, क्षय हो जाती है; वह
ऊर्जा दूर नहीं ले जाती --- पूर्ण परावर्तन।
\end{theorem}

\begin{proof}
$\vect k = (k_x, k_y, 0)$ और अंतरापृष्ठ $x = 0$ लेने पर, तीनों तरंगों की कलाएँ
$\omega t - k_yy$ हर $y$, $t$ के लिए मेल खानी चाहिए: वही
$\omega$, वही $k_y = (\omega/v_1)\sin\theta_1 = (\omega/v_1)\sin\theta_r = (\omega/v_2)\sin
\theta_2$। यदि $(\omega/v_2)^2 - k_y^2 < 0$ हो तो $k_{2x}$ काल्पनिक है, $k_{2x} = -\iu/\ell$:
अर्थात् क्षयन; और $x$ के अनुदिश उसका पॉयंटिंग सदिश माध्य में शून्य होता है।
\end{proof}

\begin{proposition}[ब्रूस्टर कोण; परावर्तन से ध्रुवण]\label{prop:b2:wave-interfaces:brewster}
\index{ब्रूस्टर कोण}\index{परावर्तन से ध्रुवण}\index{फ्रेनेल गुणांक}
किसी परावैद्युत पर पड़ते प्रकाश के लिए परावर्तित आयाम
ध्रुवण पर निर्भर करता है (फ्रेनेल के गुणांक, सामान्य रूप में स्वीकृत):
आपतन तल \emph{में} पड़े ध्रुवण (p) के लिए वह
\emph{\omterm{prop:b2:wave-interfaces:brewster}{ब्रूस्टर कोण}} पर लुप्त हो जाता है
\[
\tan\theta_B = \frac{n_2}{n_1} ,
\]
जिस पर परावर्तित और अपवर्तित किरणें लंबवत होती हैं; तब
$\theta_B$ पर परावर्तित प्रकाश आपतन तल के लंबवत पूरी तरह ध्रुवित
होता है (s), और उसके आसपास आंशिक रूप से --- वही चौंध जो पानी
और सड़कों पर उठती है और ध्रुवक धूप-चश्मे हटा देते हैं। काँच ($n = 1.5$) के लिए
$\theta_B = \qty{56}{\degree}$, जल के लिए \qty{53}{\degree}।
\end{proposition}

\begin{proof}
परावर्तित तरंग माध्यम 2 के द्विध्रुवों से विकिरित होती है, जिन्हें
अपवर्तित तरंग अपने $\vect E$ के अनुदिश दोलित कर देती है, और वह p
ध्रुवण के लिए आपतन तल में, अपवर्तित किरण के लंबवत,
पड़ता है। कोई द्विध्रुव अपनी ही अक्ष के अनुदिश विकिरित नहीं करता
(\cref{ch:b2:dipole-radiation}); जब परावर्तित दिशा
उस अक्ष के समांतर हो --- अर्थात् अपवर्तित किरण के लंबवत, $\theta_1
+ \theta_2 = \pi/2$ --- तब कुछ भी परावर्तित नहीं होता। तब $\sin\theta_2 = \cos\theta_1$ और
देकार्त देते हैं $\tan\theta_1 = n_2/n_1$। s ध्रुवण का $\vect E$
आपतन तल के लंबवत रहता है, कभी परावर्तित किरण के अनुदिश नहीं:
अतः वह सदा कुछ न कुछ परावर्तित होता है।
\end{proof}

\section{प्लाज़्मा, धातुएँ और पूर्ण चालक}

\begin{proposition}[किसी प्लाज़्मा पर और किसी धातु पर परावर्तन]\label{prop:b2:wave-interfaces:conductors}
\index{पूर्ण चालक}\index{सतही धारा (परावर्तन)}\index{हागेन--रूबेंस संबंध}
(क) किसी \omterm{def:b2:waves-in-media:plasma}{प्लाज़्मा} पर अभिलंब आपतन में $\omega < \omega_p$ की कोई तरंग पूरी तरह
परावर्तित होती है: "अपवर्तनांक" $\underline n = -\iu\sqrt{\omega_p^2/\omega^2 - 1}$ काल्पनिक है,
$r = (1 - \underline n)/(1 + \underline n)$ का मापांक $1$ है, और क्षेत्र
\cref{ch:b2:waves-in-media} के क्षयी $\eu^{-\kappa x}$ की तरह भीतर घुसता है। (ख) किसी
अच्छे चालक का $\underline n = (1 - \iu)c/\omega\delta$ है, जिसका मापांक विशाल है: $r \approx
-1$, और परावर्तकता $R \approx 1 - 2\omega\delta/c = 1 - \sqrt{8\varepsilon_0\omega/\gamma}$
(हागेन--रूबेंस): \qty{10}{GHz} पर ताँबे के लिए $0.99999$, दूर अवरक्त में
$0.99$। (ग) \emph{\omterm{prop:b2:wave-interfaces:conductors}{पूर्ण चालक}} ($\gamma \to \infty$, $\delta \to 0$):
ठीक-ठीक $r = -1$ मिलता है $\vect E$ के लिए, सतह पर कुल स्पर्शीय $\vect E$ लुप्त हो जाता है,
और आपाती तथा परावर्तित तरंगें \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} बना देती हैं
\[
\vect E = -2E_0\sin kx\sin\omega t\,\vect e_y , \qquad
\vect B = 2\frac{E_0}c\cos kx\cos\omega t\,\vect e_z ,
\]
जिसमें $\vect E$ की निस्पंदें ($\vect B$ के प्रस्पंद) $x = 0, -\lambda/2, \dots$ पर हैं; क्षेत्र
को कोई \emph{सतही धारा} $\vect j_s = \vect n\wedge\vect B/
\mu_0 = (2E_0/\mu_0c)\cos\omega t\,\vect e_y$ ले चलती है, और उस पर लगा लाप्लास बल
\omterm{prop:b2:poynting-vector:wave}{विकिरण दाब} $2I/c$ है।
\end{proposition}

\begin{proof}
(क) और (ख): $Z \propto 1/\underline n$ के साथ वही प्रतिबाधा-नियम और
\cref{ch:b2:waves-in-media} के \omterm{prop:b2:waves-in-media:lorentz}{सम्मिश्र अपवर्तनांक}; (ख) के लिए, $r = (1 - \underline n)/(1 +
\underline n) \approx -1 + 2/\underline n$ और $R = |r|^2 \approx 1 - 4\operatorname{Re}(1/
\underline n) = 1 - 2\omega\delta/c$। (ग) भीतर $\vect E = \vect 0$ लेने पर,
सीमा-संबंध $\vect E_{2t} = \vect E_{1t}$ देता है $E_{\text{कुल}}(0) = 0$, अतः $r =
-1$; आपाती $E_0\cos(\omega t - kx)$ और परावर्तित $-E_0\cos(\omega t + kx)$ जोड़ दीजिए;
$\vect B$ फ़ैराडे से; सतह पर $\vect B$ की छलाँग $\mu_0\vect j_s
\wedge\vect n$ है; और उस चादर पर प्रति एकांक क्षेत्रफल माध्य बल $\tfrac12\langle j_sB\rangle$
$\varepsilon_0E_0^2 = 2I/c$ है (\cref{ch:b2:poynting-vector})।
\end{proof}

\section{किसी केबल का सिरा}

\begin{proposition}[किसी संचरण लाइन का सिरा]\label{prop:b2:wave-interfaces:coax}
\index{मिलान वाला भार}\index{अप्रगामी तरंग अनुपात}\index{काल-क्षेत्र परावर्तनमिति}
अभिलाक्षणिक प्रतिबाधा $Z_c$ की कोई लाइन $x = \ell$ पर प्रतिबाधा
$\underline Z_L$ के किसी भार पर ख़त्म होती है: वोल्टता तरंग $\underline r =
(\underline Z_L - Z_c)/(\underline Z_L + Z_c)$ से परावर्तित होती है --- किसी \emph{मिलान वाले} भार
$Z_L = Z_c$ के लिए कुछ नहीं, किसी लघुपथ के लिए $r = -1$, किसी खुले सिरे के लिए $r = +1$। कोई
बेमेल लाइन कोई आंशिक \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} ले चलती है, जिसमें अधिकतम और
न्यूनतम वोल्टता का अनुपात \emph{\omterm{prop:b2:wave-interfaces:coax}{अप्रगामी तरंग अनुपात}} $(1 + |r|)/(1 - |r|)$ है।
कोई छोटी स्पंद भेजकर प्रतिध्वनि का समय लेना किसी ख़राबी की जगह बता देता है
(\omterm{prop:b2:wave-interfaces:coax}{काल-क्षेत्र परावर्तनमिति}) और प्रतिध्वनि के चिह्न से उसकी प्रकृति पढ़ लेता है।
\end{proposition}

\begin{proof}
भार पर, $v = Z_Li$ जहाँ $v = v_i + v_r$ और $i = (v_i - v_r)/Z_c$
(\cref{ch:b2:dispersion-wave-packets}); और वही प्रतिबाधा-नियम एक बार फिर।
\omterm{prop:b2:wave-interfaces:coax}{अप्रगामी तरंग अनुपात}: लाइन के अनुदिश $|v|$ $|v_i|(1 + |r|)$ और $|v_i|(1 -
|r|)$ के बीच बदलता रहता है।
\end{proof}

\begin{method}[कोई भी अंतरापृष्ठ]\label{met:b2:wave-interfaces:method}
(1) वे दोनों राशियाँ पहचानिए जो अंतरापृष्ठ के पार सतत रहती हैं, और
हर माध्यम की प्रतिबाधा को किसी प्रगामी तरंग में उनके अनुपात के रूप में।
(2) एक ओर आपाती $+$ परावर्तित, दूसरी ओर पारगत लिखिए,
बराबर आवृत्तियों और बराबर स्पर्शीय तरंग-सदिशों के साथ (देकार्त)।
(3) सांतत्य थोपिए: दो समीकरण, दो अज्ञात $r$, $t$। (4) ऊर्जा:
हानिरहित माध्यमों के लिए $R = |r|^2$, $T = 1 - R$। (5) सीमाएँ जाँचिए: बराबर
प्रतिबाधाएँ (कोई परावर्तन नहीं), अनंत या शून्य प्रतिबाधा ($|r| = 1$, और $r$
का चिह्न), और परतों के लिए क्रमिक परावर्तनों की कलाएँ।
\end{method}

शामिल): $R$ निकालिए। (ग) \qty{400}{nm} और \qty{700}{nm} पर $R$ (वही
परत): किसी लेपित लेंस के बचे परावर्तन का रंग क्या है?
(घ) उत्तम कोटि के लेंस कई परतें क्यों काम में लेते हैं?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:wave-interfaces:6}
\emph{पराश्रव्य जाँच-यंत्र।} पारक्रामक की सिरैमिक का $Z_1 = \qty{30e6}{kg.m^{-2}.s^{-1}}$,
ऊतक का $Z_2 = \qty{1.6e6}{kg.m^{-2}.s^{-1}}$, $f = \qty{5}{MHz}$। (क) सीधे
पारगत ऊर्जा; बाक़ी सिरैमिक के भीतर गूँजती रहती है --- इससे
स्पंद की लंबाई का क्या होता है? (ख) किसी चतुर्थांश-तरंग मिलान परत की
प्रतिबाधा और मोटाई (परत में \omterm{thm:b2:sound-waves:equation}{ध्वनि की चाल} \qty{3000}{m/s})।
(ग) जाँच-यंत्र के आगे हवा की \qty{10}{\micro m} की झिल्ली के साथ: पारगत
ऊर्जा (दो अंतरापृष्ठ, कोई अनुनाद नहीं) --- इसीलिए जेल। (घ) जेल का
$Z \approx Z_2$ है: अब जेल--त्वचा अंतरापृष्ठ से कितना अंश पार जाता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:wave-interfaces:7}
\emph{\omterm{def:b2:waves-in-media:plasma}{प्लाज़्मा} दर्पण।} \qty{3}{MHz} की कोई तरंग आयनमंडल ($n_e =
\qty{1e12}{m^{-3}}$) से अभिलंब आपतन में मिलती है। (क) $\underline n$; $r$; $|r| = 1$ जाँचिए और
$r$ की कला दीजिए। (ख) प्रवेश गहराई। (ग) परत के नीचे \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी
तरंग}: $\vect E$ की पहली निस्पंद की जगह। (घ) असल में तरंग किसी ऐसी परत पर
तिरछी आती है जिसका घनत्व ऊँचाई के साथ बढ़ता है: गुणात्मक रूप से
समझाइए कि वह उस स्तर तक पहुँचने से पहले ही क्यों मुड़ जाती है जहाँ
$\omega = \omega_p$ हो (घटते अपवर्तनांक के साथ देकार्त लीजिए), और ऊँची आवृत्ति
की कोई तरंग क्यों निकल भागती है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:wave-interfaces:8}
\emph{धातु के दर्पण।} ताँबा ($\gamma = \qty{6e7}{S/m}$): परावर्तकता $1 -
\sqrt{8\varepsilon_0\omega/\gamma}$, \qty{10}{GHz} पर, \qty{30}{THz} (CO$_2$ लेज़र) पर और,
औपचारिक रूप से, \qty{5e14}{Hz} पर --- वहाँ सूत्र क्यों चूक जाता है ($\omega\tau$
की तुलना कीजिए)? किसी ताँबे के दर्पण पर \qty{1}{kW} की CO$_2$ लेज़र-पुंज से
सोखी शक्ति; और ऐसे दर्पण जल-शीतित क्यों होते हैं।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:wave-interfaces:9}
कोई \qty{50}{\ohm} लाइन $75$, $25$, $0$, $\infty$ और \qty{50}{\ohm} पर ख़त्म होती है:
\omterm{thm:b2:wave-interfaces:normal}{परावर्तन गुणांक}, \omterm{prop:b2:wave-interfaces:coax}{अप्रगामी तरंग अनुपात}, और परावर्तित शक्ति का
अंश। कोई \qty{100}{W} प्रेषक किसी \qty{50}{\ohm} केबल से होकर \qty{75}{\ohm} के
किसी ऐंटेना को खिलाता है: ऐंटेना तक पहुँचती शक्ति (पहली बार में),
और बाक़ी कहाँ जाती है। कोई प्रतिध्वनि \qty{1.2}{\micro s} बाद सीधी लौटती है ($c_{\text{लाइन}}
= \qty{2e8}{m/s}$): ख़राबी कहाँ है और क्या है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:wave-interfaces:10}
\emph{s ध्रुवण के लिए फ्रेनेल।} आपतन तल के लंबवत $\vect E$ वाली कोई तरंग
अंतरापृष्ठ $n_1 \to n_2$ से $\theta_1$ पर मिलती है।
(क) तीनों क्षेत्र $\vect E$ और
$\vect B = \vect k\wedge\vect E/\omega$ के स्पर्शीय घटक लिखिए। (ख) स्पर्शीय
$\vect E$ और $\vect B$ का सांतत्य थोपिए और $r_s = \dfrac{n_1\cos\theta_1 - n_2\cos\theta_2}
{n_1\cos\theta_1 + n_2\cos\theta_2}$ निकालिए। (ग) अभिलंब आपतन की सीमा और
सर्पण सीमा जाँचिए। (घ) \qty{45}{\degree} पर काँच के लिए: $R_s$; उसकी तुलना $R_p
= \bigl(\frac{n_2\cos\theta_1 - n_1\cos\theta_2}{n_2\cos\theta_1 + n_1\cos\theta_2}\bigr)^2$
(स्वीकृत) से और अभिलंब आपतन के $4\%$ से कीजिए।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:wave-interfaces:11}
\emph{बाधित पूर्ण परावर्तन।} काँच ($n = 1.5$) में प्रकाश किसी
काँच--वायु फलक पर \qty{45}{\degree} पर पूरी तरह परावर्तित होता है; हवा में
$d$ दूरी पर काँच की कोई दूसरी सतह ले आई जाती है। (क) क्षयी क्षय-लंबाई
$\ell$, $\lambda = \qty{600}{nm}$ के लिए। (ख) यह मानकर कि अंतराल से पारगत
तीव्रता मोटे तौर पर $\eu^{-2d/\ell}$ है, $d$ निकालिए $50\%$
और $1\%$ पारगमन के लिए। (ग) काँच पर दबी कोई उँगली परावर्तन का क्या
कर देती है (अंगुलिचिह्न पाठक)? (घ) यह
\cref{ch:b2:potential-wells-tunneling} की सुरंगन का प्रकाशिक
समरूप क्यों है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:wave-interfaces:12}
\emph{दो दर्पण।} $x = 0$ और $x =
L$ पर दो समांतर \omterm{prop:b2:wave-interfaces:conductors}{पूर्ण चालक} कोई \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} घेर लेते हैं। (क) दिखाइए कि केवल $L = m\lambda/2$ संभव है:
कोटर की विधाएँ, $f_m = mc/2L$। (ख) $L = \qty{15}{cm}$ के लिए पहली
तीन विधाएँ; कौन-सी \qty{2.45}{GHz} के पास है? (ग) भीतर क्षेत्र-आयाम
$E_0 = \qty{20}{kV/m}$ के लिए दोनों दर्पणों पर सतही धाराएँ और हर एक पर बल
(भीतर की ओर या बाहर की ओर?)। (घ) अब दर्पण ताँबे के हैं:
\cref{ch:b2:waves-in-media} के सतही प्रतिरोध $R_s = 1/\gamma\delta$ को काम
में लेते हुए, \qty{2.45}{GHz} पर $H_0 = E_0/\mu_0c$ (चुंबकीय प्रस्पंद) के साथ प्रति एकांक क्षेत्रफल खोई शक्ति, और संचित ऊर्जा
(माध्य में प्रति एकांक आयतन $\tfrac12\varepsilon_0E_0^2$, $L$ लंबाई पर) के
क्षय होने का समय --- अर्थात् गुणता गुणक $Q = 2\pi f\,\times$ संचित ऊर्जा / खोई शक्ति।
\end{exercise}

\section{समस्या: लेप, जेल और रडार अवशोषक}

\begin{problem}[{सप्ताहांत समस्या --- अंतरापृष्ठों को ग़ायब करना: लेपित
लेंस, मिलान वाला पारक्रामक, रडार सोखता पर्दा, और वह दर्पण जिसे
ग़ायब नहीं किया जा सकता}]
\label{pb:b2:wave-interfaces:1}

\medskip
\noindent\textbf{भाग I --- लेपित लेंस।}
किसी ज़ूम लेंस में $16$ वायु--काँच सतहें हैं जिनका अपवर्तनांक $1.52$ है;
प्रकाश \qty{550}{nm} पर।
\begin{enumerate}
  \item किसी बिना लेप की सतह की परावर्तकता; पूरे लेंस का पारगमन;
        और भटकते परावर्तनों के रूप में घूमते प्रकाश का अंश।
  \item आदर्श एकल-परत लेप: अपवर्तनांक और मोटाई; उस अपवर्तनांक का
        कोई आम ठोस उपलब्ध क्यों नहीं है, और MgF$_2$ ($1.38$) के साथ
        बची परावर्तकता क्या है?
  \item $16$ सतहें प्रति सतह $1.3\%$ के साथ: पारगमन; और $0.3\%$
        (बहुपरत) पर: पारगमन। पुराने लेंस कम विभेद वाले "चपटे"
        प्रतिबिंब क्यों देते हैं?
  \item लेप \qty{550}{nm} के लिए बनाया गया है: दिखाइए कि
        \qty{440}{nm} पर एक चक्कर की कला $\varphi$ $1.25\pi$ है और
        वहाँ परावर्तकता आँकिए (दो-परावर्तन सूत्र, $R
        \approx r_1^2 + r_2^2 + 2r_1r_2\cos\varphi$, जिसमें $r_1 = -0.16$, $r_2 =
        -0.05$)। परावर्तन का रंग?
  \item कोई परावैद्युत दर्पण $1.38$ और $2.35$ की \omterm{prop:b2:wave-interfaces:optics}{चतुर्थांश-तरंग परतें}
        चुनता जाता है: अब क्रमिक अंतरापृष्ठों से परावर्तित आयाम
        एक ही कला में जुड़ते हैं; किसी एक जोड़े (दो अंतरापृष्ठ) का
        परावर्तन, और $20$ जोड़े $99.9\%$ तक क्यों पहुँच जाते हैं।
  \item लेपित लेंस में परावर्तित तरंग "रद्द" हो जाती है: उसकी
        ऊर्जा कहाँ चली गई?
  \item ऐसा कोई दर्पण तरंगदैर्घ्यों की एक पट्टी परावर्तित करता है और
        बाक़ी पार जाने देता है: क्यों, और यदि वह हरा परावर्तित करे तो
        पारगमन में उसका रंग कैसा दिखता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- जाँच-यंत्र और जेल।}
सिरैमिक $Z_1 = \qty{30e6}{kg.m^{-2}.s^{-1}}$, मिलान परत, जेल, त्वचा $Z =
\qty{1.6e6}{kg.m^{-2}.s^{-1}}$, $f = \qty{5}{MHz}$।
\begin{enumerate}[resume]
  \item ऊर्जा में सीधा सिरैमिक--ऊतक पारगमन; और $f$ पर आदर्श
        प्रतिबाधा (मान) की \omterm{prop:b2:wave-interfaces:optics}{चतुर्थांश-तरंग परत} के साथ।
  \item परत \qty{5}{MHz} पर चतुर्थांश तरंग है: \qty{3}{MHz} और
        \qty{7}{MHz} पर पारगमन (असली चक्कर-कला के साथ दो-परावर्तन
        आकलन लीजिए): इससे जाँच-यंत्र की पट्ट-चौड़ाई का क्या होता
        है, और कोई छोटी स्पंद (गहराई के विभेदन के लिए चाहिए)
        उससे क्यों टकराती है?
  \item जाँच-यंत्र और त्वचा के बीच हवा की \qty{5}{\micro m} की झिल्ली:
        दोनों अंतरापृष्ठों से आयाम-पारगमन ($t$ गुणा कीजिए,
        अनुनाद छोड़ दीजिए); पारगत ऊर्जा; और dB में।
  \item जेल ($Z = \qty{1.5e6}{kg.m^{-2}.s^{-1}}$) के साथ वही।
  \item सिरैमिक की अपनी गूँज: स्पंद सिरैमिक के भीतर उछलती रहती है,
        ऊतक वाले फलक पर प्रति परावर्तन $r \approx 0.9$ के साथ
        (और पीछे $\approx 1$); आयाम के $1\%$ तक गिरने के लिए
        चक्करों की संख्या, और सिरैमिक के पीछे कोई अवशोषक क्यों
        लगाया जाता है।
  \item \qty{5}{cm} गहराई पर \qty{1}{mm} की किसी अस्थि-सतह से प्रतिध्वनि:
        जाँच-यंत्र तक लौटती ऊर्जा का अंश (अस्थि पर परावर्तन, और
        \qty{0.5}{dB/cm/MHz} \omterm{def:b2:dispersion-wave-packets:complex}{क्षीणन} की दो यात्राएँ); अस्थि कोई ध्वनिक
        छाया क्यों डालती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- रडार अवशोषक।}
\qty{10}{GHz} का कोई रडार धातु की किसी पट्टिका पर पड़ता है। उसे छिपाने के लिए
सतही प्रतिरोध $R_s$ (\unit{\ohm} प्रति वर्ग) की कोई पतली प्रतिरोधी चादर
पट्टिका के आगे $d$ दूरी पर रख दी जाती है, और बीच में हवा।
मुक्त आकाश की प्रतिबाधा $Z_0 = \sqrt{\mu_0/\varepsilon_0} = \qty{377}{\ohm}$ है।
\begin{enumerate}[resume]
  \item नंगी धातु की पट्टिका (\omterm{prop:b2:wave-interfaces:conductors}{पूर्ण चालक}) की परावर्तकता, और
        $r$ की कला।
  \item हवा के अंतराल को प्रतिबाधा $Z_0$ की किसी संचरण लाइन की तरह
        लीजिए जिसे पट्टिका ने लघुपथित कर दिया है: दिखाइए कि चादर
        पर, पट्टिका की ओर देखते हुए, दिखती प्रतिबाधा $\underline Z_{\text{प्रवेश}} = \iu Z_0
        \tan(2\pi d/\lambda)$ है (लघुपथ से \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} $v = v_i(\eu^{-\iu kx} -
        \eu^{\iu kx})$ लिखिए और $v/i$ लीजिए $x = -d$ पर); और $d =
        \lambda/4$ पर वह अनंत है (कोई खुला परिपथ)।
  \item उस खुले परिपथ के समांतर $R_s$ प्रतिरोध की चादर आती हुई तरंग
        को $R_s$ दिखाती है: \omterm{thm:b2:wave-interfaces:normal}{परावर्तन
        गुणांक} $(R_s - Z_0)/(R_s + Z_0)$; निष्कर्ष निकालिए कि $\lambda/4$ पर
        \qty{377}{\ohm} प्रति वर्ग की कोई चादर कुछ भी परावर्तित नहीं करती
        (सैलिसबरी पर्दा)। \qty{10}{GHz} पर $d$ का मान।
  \item ऊर्जा कहाँ जाती है? चादर में व्यय शक्ति, प्रति एकांक क्षेत्रफल
        $|v|^2/2R_s$, को आपाती तीव्रता के सामने रखकर जाँचिए।
  \item \qty{8}{GHz} और \qty{12}{GHz} पर वही पर्दा: $\underline Z_{\text{प्रवेश}}$,
        कुल प्रतिबाधा $R_s \parallel \underline Z_{\text{प्रवेश}}$, और
        परावर्तकता; और इस चाल की पट्ट-चौड़ाई।
  \item अब रडार पर्दे पर अभिलंब से \qty{30}{\degree} पर पड़ता है:
        अंतराल से एक चक्कर की कला में क्या बदलता है, और उससे
        मिलान का क्या होता है?
  \item असली छिपाव-लेप एक सैलिसबरी पर्दे के बजाय कई हानिकर परतें
        और गढ़ी हुई सतहें क्यों काम में लेते हैं?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- वह दर्पण जो टिका रहता है।}
\begin{enumerate}[resume]
  \item \qty{500}{nm} पर चाँदी कोई \omterm{def:b2:waves-in-media:plasma}{प्लाज़्मा} जैसी बरतती है ($\omega_p = \qty{1.4e16}{rad/s}$):
        $\underline n$, $r$ (मापांक और कला), प्रवेश गहराई। कोई
        चतुर्थांश-तरंग लेप किसी धातु का परावर्तन क्यों नहीं रोक सकता?
  \item किसी चाँदी के दर्पण ($R = 0.97$) पर \qty{1}{kW} का हरा लेज़र: सोखी
        शक्ति, और यदि कुछ भी ठंडा न करे तो \qty{10}{g} के दर्पण
        ($c = \qty{235}{J/(kg.K)}$) की ताप-वृद्धि की दर।
  \item चाँदी के आगे \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग}: $\vect E$ की निस्पंदों की दूरी
        और पहली की जगह ($r$ की कला लीजिए); किसी निस्पंद पर बैठे
        धूल के किसी कण को क्या मिलता है?
  \item प्रश्न 22 की पुंज \qty{1}{cm^2} की है: तीव्रता, क्षेत्र-आयाम,
        और दर्पण को जो सतही धारा-घनत्व ले चलना पड़ता है
        (पूर्ण-चालक आकलन $j_s = 2E_0/\mu_0c$)।
  \item इस समस्या के तीनों "दर्पणों" --- धातु, \omterm{def:b2:waves-in-media:plasma}{प्लाज़्मा}, परावैद्युत
        ढेर --- की तुलना उस क्रियाविधि से कीजिए जो तरंग को लौटाती
        है, $r$ की कला से, और उससे जो उनकी परावर्तकता को सीमित
        करता है।
\end{enumerate}
\end{problem}
