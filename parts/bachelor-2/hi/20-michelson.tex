\chapter{माइकेल्सन व्यतिकरणमापी}\label{ch:b2:michelson}

1887 में माइकेल्सन और मोर्ली ने प्रकाश की किसी पुंज को दो में चीरा, दोनों
आधों को लंबवत भुजाओं के अनुदिश भेजा, उन्हें फिर जोड़ा, और फ्रिंजों को उस
खिसकाव के लिए ताकते रहे जो ईथर में पृथ्वी की गति से होना चाहिए था: कुछ नहीं
हिला, और भौतिकी बदल गई। 2015 में उसी यंत्र ने --- दो भुजाएँ, अब चार
किलोमीटर लंबी और हज़ार बार मोड़ी हुईं --- कोई ऐसी गुरुत्व तरंग दर्ज की जिसने
उनकी लंबाइयाँ किसी प्रोटॉन के व्यास के हज़ारवें भाग जितनी बदल दीं। इन दोनों
तारीख़ों के बीच \omterm{def:b2:michelson:instrument}{माइकेल्सन व्यतिकरणमापी} ने मीटर को तरंगदैर्घ्यों में नापा,
वर्णक्रमी रेखाएँ सुलझाईं, और परिशुद्ध प्रकाशिकी का मानक औज़ार बन गया।
यह अध्याय उसे उसके दोनों विन्यासों में --- वायु-पच्चर और
समांतर पट्ट --- बताता है, और दिखाता है कि उसकी फ्रिंजें लंबाइयाँ,
अपवर्तनांक और वर्णक्रम कैसे नापती हैं।

\section{यंत्र}

\begin{definition}[माइकेल्सन व्यतिकरणमापी]\label{def:b2:michelson:instrument}
\index{माइकेल्सन व्यतिकरणमापी}\index{पुंज-विभाजक}\index{क्षतिपूरक पट्टिका}
कोई \emph{पुंज-विभाजक} $Sp$ --- अर्ध-परावर्ती फलक वाली कोई काँच की पट्टिका
--- \qty{45}{\degree} पर किसी आपाती पुंज को दो में बाँट देता है: एक दर्पण
$M_1$ को जाती है, दूसरी $M_2$ को, और दोनों अपनी-अपनी पुंजों के लंबवत हैं;
परावर्तन के बाद हर एक $Sp$ पर लौटती है, जहाँ उसका आधा प्रेक्षक की ओर भेज
दिया जाता है, और वहीं दोनों आधे व्यतिकरण करते हैं। विभाजक जैसी ही और उसके
समांतर कोई \emph{क्षतिपूरक पट्टिका} $C$ उस भुजा में रख दी जाती है जो
नहीं तो विभाजक के काँच को केवल एक बार पार करती, ताकि दोनों
पुंजें काँच की वही मोटाई पार करें: तब \omterm{thm:b2:two-wave-interference:formula}{पथांतर} हवा के पथों का
कोई शुद्ध अंतर रह जाता है, जो हर तरंगदैर्घ्य के लिए वही है --- और
श्वेत-प्रकाश की फ्रिंजों को यही चाहिए। दर्पणों की $Sp$ से दूरियाँ $d_1$
और $d_2$ हैं; और एक दर्पण किसी सूक्ष्ममापी पेंच तथा झुकाव के पेंचों पर
लगा है।
\end{definition}

\begin{proposition}[तुल्य व्यवस्था]\label{prop:b2:michelson:equivalent}
\index{वायु-पच्चर (माइकेल्सन)}\index{वायु-पट्ट (माइकेल्सन)}
प्रेक्षक की ओर से देखने पर व्यतिकरणमापी दर्पण
$M_2$ और उस प्रतिबिंब $M_1'$ के तुल्य है जो विभाजक में $M_1$ का बनता है: अर्थात् दो परावर्ती
सतहों के बीच हवा की कोई पतली परत, जिसकी मोटाई $e = |d_1 - d_2|$ है, और जिसके
दोनों परावर्तन व्यतिकरण करते हैं (\omterm{prop:b2:two-wave-interference:film}{आयाम-विभाजन}):
\begin{itemize}
  \item $M_1' \parallel M_2$: मोटाई $e$ का कोई \emph{वायु-पट्ट} --- \omterm{prop:b2:michelson:rings}{समान नति
        के वलय}, अनंत पर स्थानीकृत, $\delta = 2e\cos i$;
  \item $M_1'$ $\alpha$ से ज़रा झुका हुआ, $M_2$ के सापेक्ष: कोई \emph{वायु-पच्चर}
        --- समान मोटाई की सीधी फ्रिंजें, दर्पणों पर स्थानीकृत
        (पच्चर के पास), अभिलंब आपतन में $\delta = 2\alpha x$,
        दूरी $\lambda/2\alpha$।
\end{itemize}
\emph{संपर्क} पर ($e = 0$, $\alpha = 0$) क्षेत्र एकसमान है, और श्वेत
प्रकाश के लिए वही एक जगह है जहाँ रंग सहमत होते हैं: अर्थात् "श्वेत फ्रिंज"।
\end{proposition}

\begin{proof}
$M_1$ वाली भुजा को $M_2$ वाली भुजा पर, $Sp$ में परावर्तन से, मोड़ देने से
दोनों पुंजें एक ही रेखा पर आ जाती हैं, जिसमें दो दर्पण $e$ दूर हैं; और लौटती दोनों
तरंगें वही हैं जो हवा की परत के दोनों फलकों से परावर्तित हैं (कोई अतिरिक्त
$\lambda/2$ नहीं: दोनों धातु पर परावर्तित होती हैं, और विभाजक उन्हें
सममित रूप से बरतता है, किसी अचर कला तक, जो पूरी आकृति को खिसका देती है)।
तब \cref{ch:b2:two-wave-interference} के पतली-झिल्ली वाले परिणाम
$n = 1$ के साथ लागू होते हैं।
\end{proof}

\section{दोनों फ्रिंज-व्यवस्थाएँ}

\begin{proposition}[वायु-पट्ट के वलय]\label{prop:b2:michelson:rings}
\index{समान नति के वलय}\index{हाइडिंगर फ्रिंजें}
$M_1' \parallel M_2$ और किसी \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} के साथ प्रेक्षक (या फ़ोकस दूरी $f$ का
कोई लेंस और उसके फ़ोकस तल में कोई पर्दा) संकेंद्री
वलय देखता है; कोण $i$ वाले वलय की कोटि $p = 2e\cos i/\lambda$ है; और वलय वहाँ
चमकीले हैं जहाँ $p$ पूर्णांक हो। केंद्र की कोटि सबसे ऊँची है, $p_0 =
2e/\lambda$; उसके पास $\cos i \approx 1 - i^2/2$, अतः केंद्र से गिना $k$-वाँ चमकीला वलय
(पहले वाले से गिनते हुए, जो केंद्र की कोटि से $\varepsilon = p_0 - \lfloor p_0\rfloor$
नीचे है) त्रिज्या रखता है
\[
r_k = f\,i_k = f\sqrt{\frac{(k - 1 + \varepsilon)\lambda}e} :
\]
अर्थात् बड़े $e$ के साथ वलय सिकुड़े हुए (अधिक और पतले वलय), और $e \to 0$ पर फैलते
हुए, यहाँ तक कि संपर्क पर कोई एकसमान क्षेत्र। किसी दर्पण को $\lambda/2$ खिसकाने पर
$p_0$ एक से बदल जाता है: अर्थात् हर आधे तरंगदैर्घ्य के विस्थापन पर केंद्र से कोई
वलय जन्म लेता है (या उसमें निगल लिया जाता है)।
\end{proposition}

\begin{proof}
समांतर पट्ट के लिए $\delta = 2e\cos i$ (\cref{ch:b2:two-wave-interference},
$n = 1$, दो धातु-परावर्तनों के लिए कोई अतिरिक्त $\lambda/2$ नहीं); और $2e(1 - i^2/2) =
(p_0 - k + 1 - \varepsilon)\lambda$, $2e = p_0\lambda$ के साथ, देता है $ei^2 = (k - 1 + \varepsilon)\lambda$।
\end{proof}

\begin{proposition}[वायु-पच्चर की फ्रिंजें]\label{prop:b2:michelson:wedge}
\index{समान मोटाई की फ्रिंजें (माइकेल्सन)}\index{फ़िज़ो फ्रिंजें}
$M_1'$ को किसी छोटे कोण $\alpha$ से झुकाने पर और लगभग-अभिलंब आपतन में, दर्पणों के
उस बिंदु पर जो संपर्क-रेखा से दूरी $x$ पर है,
\omterm{thm:b2:two-wave-interference:formula}{पथांतर} $\delta = 2\alpha x$ है: अर्थात् उस रेखा के समांतर सीधी फ्रिंजें,
$\lambda/2\alpha$ की दूरी पर, और दर्पणों पर स्थानीकृत (उन्हें $M_2$ पर फ़ोकस करके
देखा जाता है, या $M_2$ और पर्दे को संयुग्मी बनाते किसी लेंस से पर्दे पर
प्रक्षेपित किया जाता है)। शून्य-कोटि की फ्रिंज संपर्क-रेखा पर पड़ती है, और हर
तरंगदैर्घ्य के लिए वही: अतः श्वेत प्रकाश के साथ वह श्वेत है (या काली, विभाजक
की कलाओं के अनुसार), जिसके दोनों ओर कुछ रंगीन फ्रिंजें हैं --- वही
पहचान जिससे संपर्क ढूँढ़ा जाता है।
\end{proposition}

\begin{proof}
हवा की किसी पच्चर की समान-मोटाई फ्रिंजें, $e(x) = \alpha x$। स्थानीकरण:
किसी \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} के लिए पच्चर के किसी दिए बिंदु से होकर जाती किरणें
वहीं फिर मिलती हैं, लगभग-अभिलंब आपतन में उसी $\delta$ के साथ।
\end{proof}

\section{माइकेल्सन से नापना}

\begin{proposition}[लंबाइयाँ, अपवर्तनांक, वर्णक्रम]\label{prop:b2:michelson:measure}
\index{फ़ूरिये-रूपांतर वर्णक्रममिति}\index{खाँचेदार वर्णक्रम}
केंद्र से गुज़रती $N$ फ्रिंजें गिनना, जबकि कोई दर्पण $\Delta e$ खिसके,
$\Delta e = N\lambda/2$ देता है: अर्थात् तरंगदैर्घ्यों में लंबाइयाँ (कोई सूक्ष्ममापी पेंच
\qty{0.1}{\micro m} तक अंशांकित; मीटर ख़ुद ऐसे ही नापा गया था)। किसी
भुजा में डाली गई मोटाई $\ell$ और अपवर्तनांक $n$ की कोई पारदर्शी पट्टिका
$2(n - 1)\ell$ को $\delta$ में जोड़ देती है और आकृति को $2(n - 1)\ell/\lambda$
फ्रिंजें खिसका देती है: अर्थात् अपवर्तनांक (और किसी गैस-कोठरी में दाब या ताप के
साथ उनके बदलाव)। तरंगदैर्घ्य $\lambda$ और $\lambda + \Delta\lambda$ के किसी युग्मक के साथ, जैसे-जैसे $e$ बढ़ता है
दोनों फ्रिंज-व्यवस्थाएँ क़दम से बाहर और फिर क़दम में आ जाती हैं, और
विभेद हर बार लुप्त होता है
\[
\Delta e = \frac{\lambda^2}{2\Delta\lambda} :
\]
अर्थात् पास-पास की रेखाओं का बँटवारा। और आम तौर पर $e$ के फलन की तरह
विभेद स्रोत का वर्णक्रम खींच देता है (विभेद
$e \sim \ell_c/2$ भर क्षय करता है): $e$ के चलते रहते हुए केंद्र पर तीव्रता दर्ज करना
\emph{\omterm{prop:b2:michelson:measure}{फ़ूरिये-रूपांतर वर्णक्रममिति}} है, जो अवरक्त वर्णक्रममापियों की विधि है।
\end{proposition}

\begin{proof}
केंद्र पर कोटि $2e/\lambda$ है; और हर रेखा की अपनी है। दोनों
व्यवस्थाएँ फिर तब संपाती होती हैं जब $2e/\lambda - 2e/(\lambda + \Delta\lambda) = 1$, अर्थात् $2e\Delta\lambda
/\lambda^2 = 1$; और आधे रास्ते वे विरोध में हैं। सामान्य कथन
\cref{exo:b2:scalar-light-model:10} का विभेद-सूत्र है।
\end{proof}

\begin{example}[सोडियम, और मीटर]\label{ex:b2:michelson:sodium}
सोडियम युग्मक (\qty{589}{nm} पर $\Delta\lambda = \qty{0.6}{nm}$): वलय हर
$\Delta e = 589^2/(2 \times 0.6)$ nm $= \qty{0.29}{mm}$ पर धुँधले और तीखे होते जाते हैं --- अर्थात्
किसी पेंच से कुछ प्रतिशत तक $\Delta\lambda$ नापने का कोई चिरपरिचित तरीक़ा। माइकेल्सन (1892) ने
मानक मीटर की लंबाई भर लाल कैडमियम रेखा की फ्रिंजें गिनीं:
$1\,553\,163.5$ तरंगदैर्घ्य, अर्थात् मीटर और किसी परमाणु के बीच पहली गाँठ;
आज मीटर $c$ और सेकंड से परिभाषित है, और
व्यतिकरणमापी उत्पादन में मानक गुटके नैनोमीटरों तक जाँचते हैं।
\end{example}

\begin{remark}[वे भुजाएँ जिन्होंने कुछ नहीं नापा, और वे जिन्होंने कोई तरंग नापी]\label{rem:b2:michelson:history}
\index{माइकेल्सन--मोर्ली प्रयोग}\index{गुरुत्व-तरंग संसूचक}
माइकेल्सन और मोर्ली को उम्मीद थी कि ईथर में पृथ्वी का वेग $v$
उसके समांतर भुजा के अनुदिश आने-जाने का समय लंबवत भुजा के सापेक्ष
$(L/c)(v/c)^2$ बदल देगा, अर्थात् $2L(v/c)^2/\lambda \approx 0.4$ फ्रिंज का कोई
खिसकाव, $L = \qty{11}{m}$ और $v = \qty{30}{km/s}$ के लिए;
और उपकरण को घुमाने पर आकृति उतनी ही झूल जानी चाहिए थी।
वे $0.01$ फ्रिंज देख सकते थे; उन्होंने कुछ नहीं देखा। आज के \omterm{rem:b2:michelson:history}{गुरुत्व-तरंग
संसूचक} \qty{4}{km} भुजाओं वाले माइकेल्सन हैं, जिनमें प्रकाश हर भुजा में
कोई तीन सौ बार उछलता है, और लेज़र इतना स्थिर है कि भुजा की लंबाई में
$\Delta L/L \sim 10^{-21}$ का बदलाव पढ़ ले --- $10^{-18}$ m, किसी प्रोटॉन का
हज़ारवाँ भाग --- भूकंपी गति और ख़ुद फ़ोटॉनों के रव के सामने।
\end{remark}

\begin{method}[माइकेल्सन काम में लेना]\label{met:b2:michelson:method}
(1) विन्यास पहचानिए: समांतर पट्ट (अनंत पर वलय; किसी लेंस से या
अनंत पर आँख से देखिए) या पच्चर (दर्पणों पर फ्रिंजें;
$M_2$ पर फ़ोकस कीजिए)। (2) $\delta = 2e\cos i$ या $2\alpha x$ और
कोटि लिखिए। (3) हर माप को फ्रिंजों में बदलिए: $\Delta e = N\lambda/2$,
किसी पट्टिका के लिए $2(n - 1)\ell$, किसी युग्मक के लिए $\lambda^2/2\Delta\lambda$। (4) संपर्क
की जगह ($e = 0$, श्वेत फ्रिंज) निरपेक्ष संदर्भ की तरह लीजिए;
और ध्यान रखिए कि $e$ बढ़ने पर वलयों की त्रिज्याएँ \emph{सिकुड़ती} हैं। (5) संसक्ति
की सीमाएँ: वलय तब तक दिखते हैं जब तक $2e < \ell_c$; और पच्चर को
लगभग-अभिलंब आपतन चाहिए।
\end{method}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:10}

(क) किसी अकेली रेखा के लिए: $I(e)$; और $e$ में आवर्त। (ख) चौड़ाई $\Delta\nu$ की

विभेदन-शक्ति $\lambda/\Delta\lambda$। (घ) अवरक्त में यह विधि
क्यों पसंद की जाती है (संकेत, "बहुसंकेत" लाभ)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:12}

फ्रिंजों के तीखे बने रहने के लिए $i_{\max}$ को क्या मानना चाहिए
(क्षेत्र के किनारे पर $2e\,i_{\max}^2/2 < \lambda/4$, $e = \qty{20}{\micro m}$)? (ग)
समांतर पट्ट के लिए आँख हिलने पर वलय क्यों नहीं हिलते? (घ) कोई
विद्यार्थिनी ऐसी फ्रिंजें देखती है जो सिर हिलाने पर बहती हैं:
वह किस विन्यास में है, और उसे क्या सुधारना चाहिए?
\end{exercise}

\section{समस्या: सोडियम युग्मक और हवा का अपवर्तनांक}

\begin{problem}[{सप्ताहांत समस्या --- किसी माइकेल्सन व्यतिकरणमापी पर कोई
पूरा माप-सत्र: संपर्क ढूँढ़ना, लंबाइयाँ पढ़ना, पीली रेखा को
बाँटना, हवा को तौलना}]
\label{pb:b2:michelson:1}

व्यतिकरणमापी में कोई सूक्ष्ममापी पेंच है जो \qty{1}{\micro m} तक पढ़ता है;
सोडियम दीपक $\lambda_1 = \qty{589.0}{nm}$ और $\lambda_2 = \qty{589.6}{nm}$ देता है,
तीव्रताओं $2 : 1$ के साथ; और कोई श्वेत दीपक तथा कोई हीलियम--नियोन लेज़र (\qty{632.8}{nm}) भी
उपलब्ध हैं।

\medskip
\noindent\textbf{भाग I --- सजाना।}
\begin{enumerate}
  \item लेज़र के साथ विद्यार्थिनी को किसी गत्ते पर दोनों भुजाओं के दो
        धब्बे दिखते हैं; वह उन्हें झुकाव के पेंचों से एक पर एक रख देती है और तब
        सीधी फ्रिंजें देखती है: यह कौन-सा विन्यास है, और
        उनकी दूरी (\qty{2}{mm}) बचे झुकाव के बारे में क्या बताती है?
  \item झुकाव के पेंच घुमाने पर फ्रिंजें मुड़ती हैं और फिर वलय बन
        जाती हैं: वलय क्यों प्रकट होते हैं, और $M_1'$ अब कहाँ है?
  \item वह दर्पण खिसकाती है और केंद्र से निगले गए $158$ वलय गिनती
        है: विस्थापन; और $e$ किस दिशा में बदल रहा है?
  \item जैसे-जैसे वह आगे बढ़ती है, वलय फैलते हैं और केंद्रीय धब्बा बढ़कर
        पूरा क्षेत्र भर देता है: वह कहाँ पहुँच गई? लेज़र की जगह सोडियम
        दीपक आता है और उसे कुछ ख़ास नहीं दिखता: क्यों नहीं, और
        उस बिंदु पर श्वेत प्रकाश कोई चमकीली केंद्रीय फ्रिंज, जिसके दोनों ओर रंगीन हों, क्यों दिखाता है?
  \item संपर्क पर वह $M_1$ को बहुत ज़रा झुकाती है और श्वेत
        प्रकाश में सीधी रंगीन फ्रिंजें देखती है: समझाइए, और बताइए कि श्वेत वाली के
        हर ओर वह कितनी की उम्मीद कर सकती है।
  \item लेज़र के धब्बे \qty{10}{mm} चौड़े हैं और वह $f = \qty{20}{cm}$ के किसी लेंस
        के क्षेत्र में दस वलय चाहती है ($\lambda = \qty{632.8}{nm}$):
        ज़रूरी मोटाई $e$ ($k$-वें वलय की त्रिज्या काम में लीजिए)।
  \item \omterm{def:b2:michelson:instrument}{क्षतिपूरक पट्टिका} विभाजक के समांतर क्यों होनी चाहिए,
        और उसे हटा देने पर उसे श्वेत प्रकाश में क्या दिखता?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- युग्मक।}
\begin{enumerate}[resume]
  \item सोडियम दीपक के साथ संपर्क से शुरू करके वह $e$ बढ़ाती है:
        वलयों का विभेद गिरता है, लुप्त होता है, लौट आता है। पहले
        लोप की मोटाई; और $n$-वें की।
  \item वह $12$ लोप दर्ज करती है, $e = \qty{0.15}{mm}$ और $e = \qty{3.35}{mm}$
        के बीच (पहला और अंतिम पेंच पर पढ़े गए):
        $\Delta\lambda$; और यदि हर पाठ \qty{5}{\micro m} तक ठीक हो तो परिशुद्धता।
  \item पहले लोप पर दोनों रेखाओं की कोटियाँ $p_1$ और $p_2$;
        और पूरे विभेद के पहले लौटने पर।
  \item किसी लोप पर न्यूनतम विभेद, $2 : 1$ के तीव्रता-अनुपात
        को देखते हुए (दुर्बल रेखा की फ्रिंजें बची रहती हैं)।
  \item रेखाओं की प्राकृतिक चौड़ाई (हर एक \qty{0.02}{nm}) सब कुछ धो
        डालने से पहले वह कितने लोप गिन सकती है?
  \item बराबर दूरी की तीन रेखाओं वाला कोई दीपक क्या दिखाता?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- हवा।}
काँच की खिड़कियों वाली लंबाई $\ell = \qty{5.00}{cm}$ की कोई कोठरी एक भुजा में
रखकर पंप से ख़ाली की जाती है; फिर लेज़र की फ्रिंजें गिनते हुए धीरे-धीरे हवा
भीतर आने दी जाती है।
\begin{enumerate}[resume]
  \item यदि $n - 1 = \qty{2.9e-4}{}$ हो तो कोठरी को \qty{1}{atm} तक भरने से
        बना प्रकाशिक \omterm{thm:b2:two-wave-interference:formula}{पथांतर}; और फ्रिंजों की संख्या।
  \item वह $46$ फ्रिंजें गिनती है: $n - 1$ का उसका मान।
  \item वह केवल \qty{0.20}{bar} भीतर आने देती है: अपेक्षित फ्रिंजें ($n - 1 \propto$
        दाब)।
  \item माप के दौरान कमरा \qty{3}{K} गरम हो जाता है, और हर भुजा में
        खुली हवा के \qty{40}{cm} का अपवर्तनांक प्रति केल्विन
        $-\qty{1e-6}{}$ बदलता है: फ्रिंज-बहाव; क्या वह भाग II के लिए,
        और भाग III के लिए, कोई समस्या है?
  \item कोठरी की खिड़कियाँ माप को क्यों नहीं खिसकातीं (वे पूरी गिनती
        के दौरान वहीं रहती हैं)?
  \item प्रयोग CO$_2$ ($n - 1 = \qty{4.5e-4}{}$) के साथ दोहराया जाता है:
        भरी कोठरी के लिए फ्रिंजें; और केवल गिनती से गैस कैसे पहचानी जा
        सकती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सीमाएँ।}
\begin{enumerate}[resume]
  \item पेंच \qty{1}{\micro m} तक पढ़ा जाता है: \qty{1}{cm} भर फ्रिंज गिनकर नापी गई
        किसी लंबाई की परिशुद्धता, अकेले पेंच से नापने के
        सामने।
  \item \qty{100}{Hz} पर \qty{50}{nm} आयाम का कोई कंपन एक दर्पण को हिलाता
        है: फ्रिंजों की गति; आँख क्या देखती है, और प्रति सेकंड $25$ फ़्रेम
        वाला कोई कैमरा क्या देखता है?
  \item $e = \qty{1}{cm}$ भर की किसी गिनती के दौरान लेज़र की आवृत्ति सापेक्ष मान में
        \qty{1e-8}{} बहती है: फ्रिंजों में त्रुटि।
  \item \qty{10}{mm} का कोई मानक गुटका व्यतिकरणमिति से $\lambda/20$ तक
        प्रमाणित है: निरपेक्ष और सापेक्ष परिशुद्धता।
  \item लंबाइयों के लिए माइकेल्सन और वर्णक्रम के लिए जालक (अगला अध्याय)
        ही क्यों चुने जाते हैं?
  \item नापी गई चारों राशियाँ (कोई लंबाई, कोई युग्मक-बँटवारा,
        कोई अपवर्तनांक, कोई \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई}) और वह फ्रिंज-प्रेक्षण
        समेटिए जिसने हर एक दी।
\end{enumerate}
\end{problem}
