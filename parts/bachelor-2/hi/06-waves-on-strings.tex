\chapter{तारों और छड़ों पर तरंगें: दालांबेर समीकरण}\label{ch:b2:waves-on-strings}

गिटार का तार छेड़िए और कोई आकृति उस पर दौड़ पड़ती है, सेतु से टकराकर
लौटती है, और उस अप्रगामी रूप में ठहर जाती है जिसकी आवृत्ति आप स्वर के
रूप में सुनते हैं। रेल की पटरी पर कान लगाइए और गाड़ी आने से बहुत पहले
आप उसे सुन लेते हैं: कोई संपीडन इस्पात में पाँच किलोमीटर प्रति सेकंड से
दौड़ता है। वर्ष 1 के खंड ने ऐसी तरंगों का --- प्रगामी, अप्रगामी,
अध्यारोपित --- वर्णन किया था, पर यह नहीं बताया कि वे होती क्यों हैं।
यह अध्याय उन्हें चलाने वाला समीकरण निकालता है, अर्थात्
\emph{दालांबेर \omterm{thm:b2:waves-on-strings:equation}{तरंग-समीकरण}}, पहले किसी तार की अनुप्रस्थ गति के लिए,
फिर परमाणु-दर-परमाणु द्रव्यमानों और स्प्रिंगों से बनी छड़ की अनुदैर्घ्य
गति के लिए; और फिर वह ऊर्जा निकालता है जो कोई तरंग ले जाती है, तथा वह
प्रतिबाधा जो तय करती है कि माध्यम बदलने पर क्या होगा।

\section{कंपित तार}

\begin{theorem}[तार का तरंग-समीकरण]\label{thm:b2:waves-on-strings:equation}
\index{दालांबेर समीकरण}\index{तरंग-समीकरण}\index{तरंग-चाल (तार)}
$\mu$ रैखिक द्रव्यमान घनत्व का कोई तार, जिसे तनाव $T$ से खींचा गया है,
छोटे अनुप्रस्थ विस्थापन $y(x, t)$ करता है (ढाल $|\partial_xy| \ll 1$)। तब
\[
\frac{\partial^2y}{\partial t^2} = c^2\,\frac{\partial^2y}{\partial x^2} ,
\qquad c = \sqrt{\frac{T}{\mu}} ,
\]
मिलता है, जो \emph{\omterm{thm:b2:waves-on-strings:equation}{दालांबेर समीकरण}} है और जिसका सामान्य हल $y = f(x - ct)
+ g(x + ct)$ है: अर्थात् दाईं ओर चलती एक आकृति और बाईं ओर चलती एक आकृति
का, दोनों $c$ चाल से और बिना विरूपण के, अध्यारोपण।
\end{theorem}

\begin{proof}
$x$ और $x + \dd x$ के बीच का अवयव अलग कीजिए, जिसका द्रव्यमान $\mu\,\dd x$
है। उसके पड़ोसी उसे तार की स्पर्शरेखा के अनुदिश तनाव $T$ से खींचते हैं:
क्षैतिज दिशा में दोनों सिरों पर $T\cos\theta \approx T$ (क्षैतिज गति नहीं है, अतः
$T$ पूरे तार में एक-सा है), और ऊर्ध्वाधर दिशा में $T\sin\theta \approx T\partial_xy$, जिससे
कुल बल $T[\partial_xy(x + \dd x) - \partial_xy(x)] = T\,\partial_x^2y\,\dd x$ मिलता है।
न्यूटन: $\mu\,\dd x\,\partial_t^2y = T\partial_x^2y\,\dd x$। हल के लिए
$u = x - ct$, $v = x + ct$ रखिए: समीकरण $\partial_u\partial_vy = 0$ बन जाता है
(शृंखला-नियम), अतः $\partial_vy$ केवल $v$ पर निर्भर है, और $y = f(u) + g(v)$;
और उल्टे, ऐसा हर $y$ उसे संतुष्ट करता है। छोड़ा गया गुरुत्व तब तक
उचित है जब तक $T \gg \mu gL$।
\end{proof}

\section{अप्रगामी तरंगें और विधाएँ}

\begin{proposition}[दोनों सिरों पर बँधे तार की विधाएँ]\label{prop:b2:waves-on-strings:modes}
\index{प्रसामान्य विधाएँ}\index{संनादी}\index{अप्रगामी तरंग}
$x = 0$ और $x = L$ पर बँधा कोई तार ये ज्यावक्रीय अप्रगामी-तरंग हल
स्वीकार करता है (\emph{\omterm{prop:b2:waves-on-strings:modes}{प्रसामान्य विधाएँ}}):
\[
y_n(x, t) = A_n\sin\frac{n\pi x}{L}\cos(\omega_nt - \varphi_n) , \qquad
f_n = \frac{\omega_n}{2\pi} = n\,\frac{c}{2L} = n f_1 , \quad n = 1, 2, 3, \dots
\]
--- अर्थात् मूल विधा $f_1 = c/2L$ और उसकी \emph{संनादी}। विधा $n$ के
सिरों के बीच $n - 1$ निस्पंद होते हैं। तार की कोई भी गति
अध्यारोपण $y = \sum_ny_n$ है, जिसमें आयाम और कलाएँ आरंभिक आकृति और वेग
से तय होती हैं (एक फूरिये श्रेणी, जैसा इस वर्ष का गणित-खंड
स्थापित करता है): और संनादियों का यही मिश्रण स्वर का
\emph{स्वर-गुण} है।
\end{proposition}

\begin{proof}
$y = F(x)\cos(\omega t - \varphi)$ खोजिए: $F'' + (\omega/c)^2F = 0$, और $F(0) = 0$ से
$F = A\sin(\omega x/c)$ मिलता है, तथा $F(L) = 0$ के लिए $\omega L/c = n\pi$ चाहिए। हर विधा
दो प्रगामी तरंगों का योग भी है: $\sin kx\cos\omega t = \tfrac12[\sin(kx -
\omega t) + \sin(kx + \omega t)]$ --- अर्थात् तरंग और उसका परावर्तन।
ज्याओं की पूर्णता फूरिये की प्रमेय है।
\end{proof}

\begin{example}[मेल्डे का प्रयोग]\label{ex:b2:waves-on-strings:melde}
\index{मेल्डे का प्रयोग}
कोई तार एक सिरे पर आवृत्ति $f$ के कंपित्र से चलाया जाता है, और दूसरा सिरा
किसी घिरनी पर से जाकर लटके द्रव्यमान $m$ से बँधा है (अतः $T = mg$): जब भी
$f$ किसी $f_n = (n/2L)
\sqrt{mg/\mu}$ के बराबर होता है, बड़ी \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} उभर आती है --- अर्थात् विधा $n$ का अनुनाद। $L = \qty{1.2}{m}$, $\mu =
\qty{1.0}{g/m}$ और $f = \qty{50}{Hz}$ के लिए, $n = 1, 2, 3$ फंदे देने वाले द्रव्यमान
$m_n = 4L^2f^2\mu/n^2g = 1.47/n^2$ kg हैं: \qty{1.47}{kg}, \qty{367}{g}, \qty{163}{g}।
गिटार के तार को बारहवें परदे पर दबाने से $L$ आधा और
$f_1$ दुगुना हो जाता है: अर्थात् अष्टक; और बीच में बिना दबाए हलके से छूने पर
विषम विधाएँ मर जाती हैं और दूसरा संनादी बचता है --- यही फ़्लाजोले है।
\end{example}

\section{ऊर्जा, शक्ति और प्रतिबाधा}

\begin{proposition}[तार-तरंग की ऊर्जा]\label{prop:b2:waves-on-strings:energy}
\index{ऊर्जा घनत्व (तार)}\index{ऊर्जा अभिवाह (तार)}\index{प्रतिबाधा (तार)}
कोई तार प्रति एकांक लंबाई इतनी ऊर्जा और इतनी शक्ति ($+x$ की ओर
ऊर्जा-अभिवाह) ले जाता है:
\[
e = \tfrac12\mu\Bigl(\frac{\partial y}{\partial t}\Bigr)^2 + \tfrac12T\Bigl(\frac{\partial y}{\partial x}\Bigr)^2 ,
\qquad
\mathcal P = -T\,\frac{\partial y}{\partial x}\,\frac{\partial y}{\partial t} ,
\]
और ये स्थानिक संतुलन $\partial_te + \partial_x\mathcal P = 0$ का पालन करते हैं। किसी
प्रगामी तरंग $y = f(x - ct)$ के लिए गतिज और स्थितिज घनत्व
बराबर होते हैं और $\mathcal P = Z\,(\partial_ty)^2$, जहाँ
\[
Z = \sqrt{T\mu} = \mu c = \frac Tc ,
\]
तार की \emph{प्रतिबाधा} है (\unit{kg/s}): अर्थात् आगे पड़ी वस्तु पर तार
द्वारा लगाए गए अनुप्रस्थ बल $-T\partial_xy$ का अनुप्रस्थ वेग $\partial_ty$ से
अनुपात। $A$ आयाम की कोई ज्यावक्रीय तरंग
माध्य शक्ति $\langle\mathcal P\rangle = \tfrac12Z\omega^2A^2$ ले जाती है।
\end{proposition}

\begin{proof}
स्थितिज पद तनाव द्वारा अवयव को खींचने में किया गया कार्य है:
उसकी लंबाई
\[
\dd x\sqrt{1 + (\partial_xy)^2} \approx \dd x\bigl(1 + \tfrac12(\partial_xy)^2\bigr) ,
\]
है, अतः अतिरिक्त लंबाई गुणा $T$ $\tfrac12T(\partial_xy)^2\dd x$ देता है। $x$ के
पार $+x$ की ओर संचरित शक्ति वह अनुप्रस्थ बल है जो
बायाँ भाग दाएँ भाग पर लगाता है, $-T\partial_xy$, गुणा वेग
$\partial_ty$। संतुलन:
\[
\partial_te = \mu\,\partial_ty\,\partial_t^2y + T\,\partial_xy\,\partial_x\partial_ty ,
\qquad
-\partial_x\mathcal P = T\,\partial_x^2y\,\partial_ty + T\,\partial_xy\,\partial_x\partial_ty ,
\]
जो \omterm{thm:b2:waves-on-strings:equation}{तरंग-समीकरण} से बराबर हैं। $f(x - ct)$ के लिए: $\partial_xy = -\partial_ty/c$, अतः $\mathcal P = (T/c)
(\partial_ty)^2$ और दोनों घनत्व $\tfrac12\mu(\partial_ty)^2$ के बराबर होते हैं।
\end{proof}

\begin{proposition}[किसी संधि पर परावर्तन और पारगमन]\label{prop:b2:waves-on-strings:junction}
\index{परावर्तन गुणांक (तार)}\index{पारगमन गुणांक (तार)}
$A_i$ आयाम की कोई तरंग $Z_1$ प्रतिबाधा के तार से आकर $Z_2$ प्रतिबाधा के
उस तार पर पहुँचती है जो $x = 0$ पर जुड़ा है। संधि पर
विस्थापन और अनुप्रस्थ बल की संततता ये
आयाम-गुणांक देती है:
\[
r = \frac{A_r}{A_i} = \frac{Z_1 - Z_2}{Z_1 + Z_2} , \qquad
t = \frac{A_t}{A_i} = \frac{2Z_1}{Z_1 + Z_2} ,
\]
और ऊर्जा-अंश $R = r^2$, $\mathcal T = 4Z_1Z_2/(Z_1 + Z_2)^2$, जिनके लिए
$R + \mathcal T = 1$। बँधा सिरा ($Z_2 \to \infty$) $r = -1$ से परावर्तित करता है
(स्पंद उलटा लौटता है), और मुक्त सिरा ($Z_2 = 0$) $r = +1$ से;
और बराबर प्रतिबाधाएँ सब कुछ पार कर देती हैं।
\end{proposition}

\begin{proof}
ज्यावक्रीय तरंगें $y_1 = A_i\cos(\omega t - k_1x) + A_r\cos(\omega t + k_1x)$, $x < 0$ के लिए,
और $y_2 = A_t\cos(\omega t - k_2x)$, $x > 0$ के लिए, दोनों ओर वही $\omega$।
$x = 0$ पर: $y_1 = y_2$ से $A_i + A_r = A_t$ मिलता है; और बल $-T\partial_xy$ का
संतत होना (द्रव्यमानहीन संधि) $Z_1(A_i - A_r) = Z_2A_t$ देता है ($T_1k_1 = Z_1\omega$

\end{proof}

\section{परमाणुओं की शृंखला से प्रत्यास्थ छड़ तक}

\begin{proposition}[परमाणुओं की शृंखला; यंग गुणांक]\label{prop:b2:waves-on-strings:chain}
\index{परमाणुओं की शृंखला}\index{यंग गुणांक}\index{हुक का नियम}\index{अनुदैर्घ्य तरंग}
एक-से $m$ द्रव्यमानों की कोई पंक्ति, जो $a$ अंतराल पर रखे और $k$ दृढ़ता की
स्प्रिंगों से जुड़े हैं: यदि $u_n(t)$ पंक्ति के अनुदिश द्रव्यमान $n$ का विस्थापन हो,
तो
\[
m\ddot u_n = k(u_{n+1} - u_n) - k(u_n - u_{n-1}) = k(u_{n+1} - 2u_n + u_{n-1}) .
\]
जब विस्थापन एक द्रव्यमान से अगले तक धीरे-धीरे बदलता है (अर्थात्
$\lambda \gg a$ तरंगदैर्घ्यों पर), तब $u_n(t) \to u(x, t)$ और $u_{n\pm1} - u_n \approx
\pm a\,\partial_xu + \tfrac12a^2\partial_x^2u$, और शृंखला \omterm{thm:b2:waves-on-strings:equation}{दालांबेर
समीकरण} का पालन करती है:
\[
\frac{\partial^2u}{\partial t^2} = c^2\frac{\partial^2u}{\partial x^2} , \qquad
c = a\sqrt{\frac km} = \sqrt{\frac{E}{\rho}} ,
\]
जहाँ $\rho = m/a^3$ ऐसी शृंखलाओं से घनीय व्यूह में बने किसी ठोस का
घनत्व है और $E = k/a$ उसका \emph{\omterm{prop:b2:waves-on-strings:chain}{यंग गुणांक}}: अर्थात् $S$ काट और $\ell$
लंबाई की छड़ की दृढ़ता $kS/a^2\cdot a/\ell = ES/\ell$ है,
अर्थात् \emph{\omterm{prop:b2:waves-on-strings:chain}{हुक का नियम}} $F/S = E\,\Delta\ell/\ell$ (प्रतिबल $\sigma = E\varepsilon$)।
\end{proposition}

\begin{proof}
हर स्प्रिंग अपने सिरों के विस्थापनों के अंतर से खिंचती है।
टेलर-प्रसार रखिए: $m\partial_t^2u = ka^2\partial_x^2u$। प्रति परमाणु
$a$ भुजा का घन $\rho = m/a^3$ देता है; और $S$ काट की छड़ में
$S/a^2$ समांतर शृंखलाएँ होती हैं, हर एक में श्रेणीबद्ध $\ell/a$ स्प्रिंगें, जिनकी कुल दृढ़ता
$(S/a^2)(ka/\ell) = (k/a)S/\ell$ है।
\end{proof}

\begin{proposition}[छड़ में अनुदैर्घ्य तरंगें]\label{prop:b2:waves-on-strings:rod}
\index{ठोसों में ध्वनि}
सीधे सातत्य पर: किसी छड़ का $x$ और $x +
\dd x$ के बीच का अवयव, जो $u(x, t)$ से विस्थापित है, $\varepsilon = \partial_xu$ से
विकृत होता है और हर फलक पर प्रतिबल $\sigma = E\partial_xu$ से खिंचता है; और
न्यूटन $\rho\,\partial_t^2u =
E\,\partial_x^2u$ देता है: अर्थात् अनुदैर्घ्य (संपीडन) तरंगें $c = \sqrt{E/\rho}$ से चलती हैं
--- इस्पात में \qty{5100}{m/s} ($E = \qty{200}{GPa}$), ऐलुमिनियम में \qty{5000}{m/s},
ग्रेनाइट में \qty{3700}{m/s}, और लकड़ी में रेशे के अनुदिश \qty{3500}{m/s}।
प्रति एकांक क्षेत्रफल प्रतिबाधा $Z = \rho c = \sqrt{E\rho}$ है, और वही
संधि-सूत्र लागू होते हैं --- और इसी से पराश्रव्य तरंगें
पटरी की दरार ढूँढ़ लेती हैं।
\end{proposition}

\begin{proof}
अवयव पर कुल बल: $S[\sigma(x + \dd x) - \sigma(x)] = SE\partial_x^2u\,\dd x$;
और द्रव्यमान $\rho S\dd x$।
\end{proof}

\begin{remark}[शृंखला जो जानती है, छड़ नहीं जानती]\label{rem:b2:waves-on-strings:dispersion}
यथार्थ शृंखला-समीकरण में $u_n = A\cos(\omega t - kna)$ रखने पर
$\omega = 2\sqrt{k/m}\,|\sin(ka/2)|$ मिलता है: $ka \ll 1$ के लिए यह $\omega = ck$ है, अर्थात्
अपरिक्षेपी \omterm{thm:b2:waves-on-strings:equation}{तरंग-समीकरण}, पर तरंगदैर्घ्य के $2a$ के पास आते ही
आवृत्ति $2\sqrt{k/m}$ पर संतृप्त हो जाती है और कला तथा समूह वेग
अलग हो जाते हैं --- यही इस खंड का पहला \emph{परिक्षेपण संबंध} है, और
\cref{ch:b2:dispersion-wave-packets} का विषय। $a = \qty{0.25}{nm}$ और
$c = \qty{5}{km/s}$ के लिए कटाव $\omega_{\max} = 2c/a = \qty{4e13}{rad/s}$
(\qty{6}{THz}) पर पड़ता है: ठोस अपने परमाणुओं की स्प्रिंगों के दिए से
तेज़ कंप ही नहीं सकता।
\end{remark}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-on-strings:11}
\emph{शृंखला का परिक्षेपण।} (क) शृंखला-समीकरण में $u_n = A\cos(\omega t - kna)$
रखकर $\omega(k) = 2\omega_0|\sin(ka/2)|$, $\omega_0 =
\sqrt{k/m}$ निकालिए। (ख) कला वेग $\omega/k$ और समूह वेग $\dd\omega/\dd k$;
$ka \to 0$ के लिए उनकी उभयनिष्ठ सीमा; और $ka = \pi$ पर उनके मान। (ग)
$ka = \pi$ पर तरंग कैसी दिखती है? $2a$ से छोटी तरंगदैर्घ्य की कोई
तरंग क्यों नहीं होती? (घ) इस्पात के लिए ($c = \qty{5.1}{km/s}$, $a =
\qty{0.25}{nm}$): $\omega_0$, अधिकतम आवृत्ति, और \qty{1}{MHz} की
पराश्रव्य तरंग की तरंगदैर्घ्य --- क्या वह परिक्षेपी है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-on-strings:12}
\emph{लटकती रस्सी, चटकता चाबुक।} (क) $L$ लंबाई की कोई एकसमान रस्सी
छत से लटकी है: मुक्त सिरे से $x$ ऊँचाई पर तनाव; और स्थानिक
तरंग-चाल $c(x)$। (ख) किसी स्पंद को नीचे से ऊपर तक जाने में लगा समय;
$L = \qty{10}{m}$ के लिए मान; और स्पंद का आकार क्यों बदलता है। (ग) कोई
चाबुक पतला होता जाता है, अर्थात् नोक की ओर $\mu$ घटता है, और प्रहार के दौरान
तनाव (लगभग) अचर रहता है: उसके अनुदिश चाल कैसे बदलती है? (घ)
यह मानकर कि स्पंद की ऊर्जा संरक्षित रहती है और उसकी
अवधि अचर रहती है, दिखाइए कि उसके अनुप्रस्थ वेग का आयाम
$\mu^{-1/4}$ की तरह बढ़ता है, और ऐसे चाबुक की नोक की चाल आँकिए जिसमें
$\mu$ मूठ से नोक तक $10^4$ गुना गिरता हो और हाथ
\qty{10}{m/s} देता हो। (चटकना एक ध्वनि-विस्फोट है।)
\end{exercise}

\section{समस्या: पियानो, हथौड़े से ध्वनि-पट्ट तक}

\begin{problem}[{सप्ताहांत समस्या --- कोई पियानो-तार खोलकर देखा गया: उसका
तनाव और दृढ़ता, हथौड़े का प्रहार और उससे बनते संनादी,
वह सेतु जो उसकी ऊर्जा ध्वनि-पट्ट को रिसा देता है, और कोई पटरी जो
वही समीकरण ढोती है}]
\label{pb:b2:waves-on-strings:1}

\medskip
\noindent\textbf{भाग I --- तार।}
किसी पियानो का A$_4$ तार (\qty{440}{Hz}): इस्पात, $E = \qty{200}{GPa}$, $\rho =
\qty{7800}{kg/m^3}$, व्यास $d = \qty{1.0}{mm}$, कंपन-लंबाई $L =
\qty{38}{cm}$। मंद्र से ऊपर के हर स्वर के लिए ऐसे तीन तार होते हैं; और पियानो में
$230$ तार हैं।
\begin{enumerate}
  \item तार का रैखिक द्रव्यमान घनत्व, तरंग-चाल और तनाव।
  \item तार में तनन-प्रतिबल; तार \qty{2.0}{GPa} पर टूटता है: सुरक्षा
        का अवकाश, और मिलाने वाले का आख़िरी घुमाव ही ख़तरनाक क्यों होता
        है।
  \item सब तार एक ही तनाव पर मानकर ढाँचे पर कुल तनाव
        आँकिए।
  \item सबसे नीचा तार (A$_0$, \qty{27.5}{Hz}) \qty{1.9}{m} लंबा है; यदि वह
        उसी तनाव पर सादा इस्पात-तार होता, तो उसे कौन-सा $\mu$ और
        कौन-सा व्यास चाहिए होता? वह वास्तव में ताँबे से लपेटा गया इस्पात-क्रोड
        है: समझाइए।
  \item A$_4$ तार की विधाएँ: पहले आठ संनादियों की
        आवृत्तियाँ; इनमें से कौन-से मूल विधा से संवादी अंतराल पर पड़ते हैं
        (अष्टक, पंचम, चतुर्थ, वृहत् तृतीय) और कौन-से
        नहीं?
  \item तार अपनी मूल विधा में प्रस्पंद पर \qty{0.50}{mm} आयाम से
        कंपित होता है: संचित ऊर्जा (किसी विधा की ऊर्जा
        $\tfrac14\mu L\omega^2A^2$ है)।
  \item मूल विधा को दो प्रगामी तरंगों के योग के रूप में लिखिए; उनका
        आयाम और हर एक की ले जाई गई माध्य शक्ति दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- हथौड़ा।}
हथौड़ा तार को $x_0 = L/8$ पर मारता है और उसे $t = 0$ पर वेग $v_0$ देता है,
$w$ चौड़ाई के किसी छोटे खंड पर जो $x_0$ के चारों ओर है, और विस्थापन कोई नहीं।
\begin{enumerate}[resume]
  \item सामान्य हल $y = \sum_nB_n\sin(n\pi x/L)\sin(\omega_nt)$ लिखिए
        और ज्याओं की लांबिकता से दिखाइए कि $B_n =
        \dfrac{2}{L\omega_n}\int_0^L\dot y(x, 0)\sin\dfrac{n\pi x}L\,\dd x$।
  \item सँकरे प्रहार ($w \ll L/n$) के लिए दिखाइए कि $B_n \approx \dfrac{2v_0w}
        {L\omega_n}\sin\dfrac{n\pi x_0}{L}$।
  \item $x_0 = L/8$ के लिए कौन-से संनादी अनुपस्थित हैं? पियानो बनाने
        वाले $L/7$ से $L/8$ के पास क्यों मारते हैं (प्रश्न 5 देखिए)?
  \item प्रगामी-तरंग चित्र में गति का वर्णन कीजिए: प्रहार-बिंदु से क्या
        निकलता है, सिरों पर क्या होता है, और
        कितने समय बाद आरंभिक अवस्था लौट आती है।
  \item वास्तविक तार कुछ कड़ा होता है: उसकी विधाएँ $f_n = nf_1
        \sqrt{1 + Bn^2}$ हैं, जहाँ $B = \pi^3Ed^4/(64TL^2)$ (स्वीकृत)।
        $B$ और आठवें संनादी की तीव्रता, प्रतिशत में और
        सेंट में, निकालिए (\num{1} सेंट = $2^{1/1200}$ का आवृत्ति-अनुपात)।
  \item पियानो मिलाने वाले अष्टकों को ``खींचते'' क्यों हैं (ऊँचे स्वर
        कुछ तीव्र मिलाते हैं)?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- सेतु और ध्वनि-पट्ट।}
तार ध्वनि-पट्ट पर चिपके एक सेतु से होकर जाता है; और ध्वनि-पट्ट
सेतु पर $Z_b = \qty{2000}{kg/s}$ प्रतिबाधा के तार जैसा बरतता है।
\begin{enumerate}[resume]
  \item तार की प्रतिबाधा $Z_s$; सेतु पर आयाम का परावर्तन गुणांक।
  \item हर परावर्तन पर ध्वनि-पट्ट तक पहुँचती तरंग-ऊर्जा का
        अंश।
  \item सेतु पर प्रति सेकंड परावर्तनों की संख्या; उससे ऊर्जा-क्षय का
        समय-नियतांक और ध्वनि के \qty{60}{dB} (ऊर्जा में $10^6$ का गुणक)
        गिरने का समय निकालिए।
  \item प्रश्न 6 के प्रहार के ठीक बाद ध्वनि-पट्ट में रिसती शक्ति;
        किसी धीमी बातचीत की शक्ति, लगभग \qty{1e-5}{W} ध्वनि, से
        तुलना कीजिए।
  \item अकेला तार लगभग अश्रव्य क्यों है, और ध्वनि-पट्ट इसका क्या
        करता है? (हवा की प्रतिबाधा
        $\rho c \approx \qty{400}{kg.m^{-2}.s^{-1}}$ सोचिए, जो छोटे
        पृष्ठ से और बड़े पृष्ठ से अलग दिखती है।)
  \item अवमंदक उठा हो तो A$_3$ तार (\qty{220}{Hz}) तब बजने लगता है जब
        A$_4$ बजाया जाए: क्यों?
  \item कोई पियानो बनाने वाला लंबी गूँज चाहता है: $Z_b$ बढ़ाना चाहिए या
        घटाना, और उससे क्या खोता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- पटरी।}
कोई इस्पात-पटरी ($E = \qty{200}{GPa}$, $\rho = \qty{7800}{kg/m^3}$, काट
\qty{70}{cm^2})।
\begin{enumerate}[resume]
  \item संपीडन तरंगों की चाल; किसी गाड़ी की ध्वनि को पटरी के अनुदिश
        और हवा से होकर \qty{5}{km} जाने में लगा समय।
  \item हथौड़े का प्रहार \qty{25}{m} की पटरी-खंड में, जिसका एक सिरा
        मुक्त है, संपीडन-स्पंद भेजता है: प्रतिध्वनि का समय; और क्या वह
        प्रतिध्वनि संपीडन है या विरलन?
  \item कोई \qty{2}{MHz} की पराश्रव्य जाँच दरारें ढूँढ़ती है: इस्पात में
        तरंगदैर्घ्य; इस्पात--वायु दरार पर आयाम का परावर्तन गुणांक
        (हवा की प्रतिबाधा $\qty{400}{kg.m^{-2}.s^{-1}}$); और कोई पतली
        दरार इतनी अच्छी तरह क्यों दिखती है।
  \item भूकंप: चट्टान में संपीडन (P) तरंगें \qty{6.0}{km/s} से और
        अपरूपण (S) तरंगें \qty{3.5}{km/s} से चलती हैं। कोई केंद्र
        S तरंग P तरंग के \qty{20}{s} बाद लिखता है: स्रोत तक
        दूरी।
  \item एक सारणी में इस समस्या की पाँचों तरंग-चालें (तार, पटरी,
        चट्टान P, चट्टान S, हवा) उस दृढ़ता और उस जड़त्व के साथ दीजिए
        जो हर एक को तय करते हैं।
\end{enumerate}
\end{problem}
