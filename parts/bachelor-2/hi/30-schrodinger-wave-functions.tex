\chapter{श्रोडिंगर समीकरण और तरंग फलन}\label{ch:b2:schrodinger-wave-functions}

इलेक्ट्रॉनों को एक-एक कर किसी झिरी-युग्म पर दागिए और लिखते जाइए कि हर एक
कहाँ गिरता है: यहाँ एक बिंदु, वहाँ एक बिंदु, प्रत्यक्षतः यादृच्छिक --- और दस
हज़ार बिंदुओं के बाद, यंग के प्रयोग की फ्रिंजें, मानो हर इलेक्ट्रॉन
दोनों झिरियों से गुज़रा हो और अपने साथ व्यतिकरण किया हो। वर्ष 1 के खंड का अंतिम
अध्याय इस विचित्रता से मिला था: प्रकाश फोटॉनों में आता है,
पदार्थ की कोई तरंगदैर्घ्य $\lambda = h/p$ है, और स्थिति तथा
संवेग दोनों तीखे नहीं हो सकते। जो उसने नहीं दिया वह है
\emph{समीकरण} --- वह नियम जो कहता है कि किसी कण से जुड़ी तरंग
कैसे विकसित होती है, जैसे न्यूटन का नियम कहता है कि कोई प्रक्षेप-पथ कैसे विकसित होता है। वह
समीकरण श्रोडिंगर का है (1926), और उसके साथ क्वांटम वर्णन
कोई परिकलनीय सिद्धांत बन जाता है: कोई कोई \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{तरंग फलन}}
$\psi(x,t)$ लिखता है, जिसके मापांक का वर्ग कण को पाने की प्रायिकता
है, और कोई कोई रैखिक आंशिक अवकल समीकरण हल करता है। यह
अध्याय समीकरण कहता है, समझाता है कि $\psi$ का क्या अर्थ है और क्या
नहीं, उसकी \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाएँ} तथा मुक्त कण के \omterm{prop:b2:dispersion-wave-packets:group}{तरंग-पुंज} ढूँढ़ता है, और दिखाता है कि न्यूटन की यांत्रिकी कहाँ फिर उभरती है --- हर उस
चीज़ के लिए जो किसी अणु से भारी हो।

\section{वर्ष 1 का खंड हमें क्या छोड़ गया}

\begin{proposition}[कण तरंगें हैं]\label{prop:b2:schrodinger-wave-functions:recap}
आवृत्ति $\nu$ के किसी फोटॉन की ऊर्जा $E = h\nu = \hbar\omega$ और
संवेग $p = h/\lambda = \hbar k$ है; संवेग $p$ का कोई भौतिक कण
तरंगदैर्घ्य $\lambda = h/p$ की किसी तरंग जैसा बरतता है (डी ब्रॉय), और उसका $E =
\hbar\omega$ तथा $p = \hbar k$ भी --- क्रिस्टलों से इलेक्ट्रॉन \omterm{def:b2:diffraction:huygens}{विवर्तन},
न्यूट्रॉन और अणु तक की व्यतिकरणमिति उसकी पुष्टि करते हैं; और कोई
अवस्था $\Delta x\,\Delta p
\ge \hbar/2$ से तीखी स्थिति और संवेग नहीं रख सकती (हाइज़ेनबर्ग)। $\hbar = h/2\pi = \qty{1.05e-34}{J.s}$ के साथ: \qty{100}{eV} पर किसी
इलेक्ट्रॉन का $\lambda = \qty{0.12}{nm}$ है, यानी किसी परमाणु का
आकार; \qty{1}{mm/s} पर \qty{1e-15}{kg} के किसी धूलकण का $\lambda =
\qty{7e-16}{m}$ है, किसी नाभिक से छोटा --- वह चिरसम्मत ढंग से बरतता है।
\end{proposition}

\begin{proof}
वर्ष 1 के खंड से याद किया गया; अब का काम तरंग की
गतिकी है।
\end{proof}

\section{तरंग फलन}

\begin{definition}[तरंग फलन और बोर्न का नियम]\label{def:b2:schrodinger-wave-functions:psi}
$x$ के अनुदिश चलते किसी कण की अवस्था किसी सम्मिश्र
\emph{तरंग फलन}\index{तरंग फलन} $\psi(x,t)$ से वर्णित होती है, ऐसा कि
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
कण को $x$ और $x + \dd x$ के बीच समय $t$ पर पाने की प्रायिकता है
(\emph{बोर्न का नियम}\index{बोर्न का नियम}); अतः
\emph{प्रसामान्यन} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, और
प्रत्याशा मान $\langle x\rangle = \int x|\psi|^2\dd x$, $\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$। तरंग फलन
\emph{अध्यारोपण सिद्धांत}\index{अध्यारोपण सिद्धांत} मानते हैं: यदि
$\psi_1$ और $\psi_2$ संभव अवस्थाएँ हों, तो $\alpha\psi_1 +
\beta\psi_2$ भी (प्रसामान्यित), और तब प्रायिकताओं में
व्यतिकरण पद $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$ रहता है। कोई
वैश्विक कला गुणक $\eu^{\iu\theta}$ कुछ नहीं बदलता; दो अध्यारोपित घटकों के बीच
कोई सापेक्ष कला व्यतिकरण बदल देती है।
\end{definition}

\begin{remark}[$\psi$ जो नहीं है]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ न तो $\vect E$ जैसा दिक् में फैला कोई चिरसम्मत क्षेत्र है, न
पदार्थ का कोई बादल: इलेक्ट्रॉन सदा पूरा ही मिलता है, किसी एक बिंदु पर।
$\psi$ कोई \emph{प्रायिकता आयाम} है, वह वस्तु जो
व्यतिकरण करती है; केवल $|\psi|^2$ प्रेक्षित होता है, और केवल सांख्यिकीय रूप से,
सर्वसम रूप से तैयार कणों पर प्रयोग दोहराकर। द्विझिरी के
बिंदु एकल इलेक्ट्रॉन हैं; फ्रिंजें $|\psi_1 +
\psi_2|^2$ हैं।
\end{remark}

\section{श्रोडिंगर समीकरण}

\begin{theorem}[श्रोडिंगर समीकरण]\label{thm:b2:schrodinger-wave-functions:equation}
द्रव्यमान $m$ के किसी कण का, जो स्थितिज ऊर्जा $V(x)$ में हो, कोई \omterm{def:b2:schrodinger-wave-functions:psi}{तरंग
फलन} होता है जो यह मानता है
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
यह रैखिक है (अध्यारोपण), समय में प्रथम कोटि का (किसी एक क्षण पर \omterm{def:b2:schrodinger-wave-functions:psi}{तरंग फलन}
उसे बाद के सभी समयों पर तय कर देता है), और सम्मिश्र ($\psi$
वास्तविक नहीं हो सकता: वह $\iu$ अनिवार्य है)। यह कोई \emph{अभिगृहीत} है,
जो अपने परिणामों से न्यायसंगत ठहरता है।
\end{theorem}

\begin{proof}
कोई प्रमाण नहीं, बल्कि कोई प्रेरणा। तीखे संवेग
$p = \hbar k$ और ऊर्जा $E = \hbar\omega$ का कोई मुक्त कण कोई समतल तरंग $\psi =
A\eu^{\iu(kx - \omega t)}$ होना चाहिए, जिसके लिए $\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ और $-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$।
चिरसम्मत संबंध $E = p^2/2m$ इस प्रकार समीकरण
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$ से पुनः बनता है, और $E = p^2/2m + V$ के लिए
स्वाभाविक व्यापकीकरण $V\psi$ जोड़ देता है। ध्यान दीजिए कि
समतल तरंगों के अध्यारोपणों पर टिकने के लिए समीकरण को रैखिक होना ही पड़ता है, और
समय में प्रथम कोटि का किसी $\iu$ के साथ ताकि समय-निर्भरता $\eu^{-\iu\omega t}$ हो, न कि
$\cos\omega t$ --- कोई वास्तविक तरंग समीकरण
$E^2 \propto p^4$ देता।
\end{proof}

\begin{proposition}[प्रायिकता का संरक्षण]\label{prop:b2:schrodinger-wave-functions:current}
प्रायिकता घनत्व $\rho = |\psi|^2$ और \emph{प्रायिकता
धारा}\index{प्रायिकता धारा}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
$\partial_t\rho + \partial_xj = 0$ मानते हैं: प्रायिकता न सृजित होती है न
नष्ट, और $\int|\psi|^2\dd x$ $1$ के बराबर बना रहता है। किसी समतल तरंग
$A\eu^{\iu kx}$ के लिए, $j = |A|^2\hbar k/m = \rho v$: घनत्व गुणा वेग, जैसा
किसी भी प्रवाह के लिए।
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$;
समीकरण और उसके संयुग्मी से, $\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$, $\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$; $V$ वाले पद कट जाते हैं और बाकी $(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$ है।
\end{proof}

\begin{proposition}[संवेग, प्रत्याशा मान और चिरसम्मत सीमा]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
किसी अवस्था का संवेग संकारक $\hat p =
-\iu\hbar\,\partial_x$ से मिलता है (जो किसी समतल तरंग पर $\hbar k$ देता है): $\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, और ऊर्जा $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$ से। प्रत्याशा मान यह मानते हैं
(एहरनफ़ेस्ट)
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
\omterm{prop:b2:dispersion-wave-packets:group}{तरंग-पुंज} का केंद्र न्यूटन का नियम ठीक-ठीक तब मानता है जब $V$
अधिक से अधिक द्विघाती हो, और लगभग तब जब भी पुंज उस मापक की तुलना में
सँकरा हो जिस पर $V$ बदलता है --- वही चिरसम्मत सीमा,
जो किसी भी स्थूल वस्तु के लिए टिकती है क्योंकि $\hbar$ इतना छोटा है।
\end{proposition}

\begin{proof}
$\langle x\rangle = \int x|\psi|^2$ का अवकलन कीजिए, $\partial_t\rho =
-\partial_xj$ लीजिए और खंडशः समाकलन कीजिए: $\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$। दूसरा संबंध वही परिकलन एक
कोटि ऊपर है (यहाँ मान लिया गया)।
\end{proof}

\section{स्थायी अवस्थाएँ}

\begin{theorem}[स्थायी अवस्थाएँ]\label{thm:b2:schrodinger-wave-functions:stationary}
यदि $V$ समय पर निर्भर न हो, तो समीकरण के
पृथक्कृत रूप के हल होते हैं
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(वह \emph{समय-निरपेक्ष श्रोडिंगर समीकरण}\index{श्रोडिंगर
समीकरण!समय-निरपेक्ष}): वे \emph{स्थायी अवस्थाएँ}\index{स्थायी
अवस्था}, सुपरिभाषित ऊर्जा $E$ की, जिनका प्रायिकता घनत्व
$|\varphi|^2$ हिलता नहीं। वे ऊर्जाएँ $E$ जिनके लिए $\varphi$
स्वीकार्य हो (परिबद्ध, किसी बद्ध अवस्था के लिए प्रसामान्यनीय) वही
\emph{ऊर्जा स्तर} हैं; व्यापक हल कोई अध्यारोपण $\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$ है, और दो स्तरों के किसी अध्यारोपण का
घनत्व \emph{बोर आवृत्ति}\index{बोर
आवृत्ति} $\nu_{21} = (E_2 - E_1)/h$ पर दोलन करता है --- उस प्रकाश की आवृत्ति जिसे
परमाणु उत्सर्जित या अवशोषित करता है।
\end{theorem}

\begin{proof}
$\varphi(x)f(t)$ रखिए: $\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
$t$ का कोई फलन $x$ के किसी फलन के बराबर, अतः कोई अचर $E$; $f
= \eu^{-\iu Et/\hbar}$। दो स्तरों के लिए हमें मिलता है $|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$।
\omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाओं} की पूर्णता मान ली गई है।
\end{proof}

\section{मुक्त कण: समतल तरंगें और तरंग-पुंज}

\begin{proposition}[मुक्त कण]\label{prop:b2:schrodinger-wave-functions:free}
$V = 0$ के लिए \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाएँ} समतल तरंगें $\eu^{\iu(kx -
\omega t)}$ हैं, जिनमें
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{सम}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
पदार्थ तरंगें \emph{प्रकीर्णी} हैं; \omterm{def:b2:dispersion-wave-packets:relation}{कला वेग} का कोई भौतिक
अर्थ नहीं, और \emph{\omterm{prop:b2:dispersion-wave-packets:group}{समूह वेग}} ही कण का वेग है। कोई समतल
तरंग प्रसामान्यनीय नहीं --- पूर्णतः तीखे संवेग का कोई कण
समूचे दिक् में फैला होता है --- अतः कोई असली कण कोई \emph{तरंग-पुंज}\index{तरंग-पुंज} है, किसी पट्टी $\Delta k$ पर समतल तरंगों का कोई
अध्यारोपण, $\Delta x \sim 1/\Delta k$ पर स्थानीकृत, $v_{\text{सम}}$ पर चलता और
\emph{फैलता} हुआ: आरंभिक चौड़ाई $\sigma_0$ के किसी \omterm{prop:b2:laser:gaussian}{गाउसी पुंज} की
समय $t$ पर चौड़ाई होती है
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
और वह $\Delta x\,\Delta p = \hbar/2$ को $t = 0$ पर मानता है (न्यूनतम अनुमत)
तथा बाद में अधिक।
\end{proposition}

\begin{proof}
प्रकीर्णन संबंध समीकरण से निकलता है; $v_{\text{सम}}$ वैसे ही जैसे
\cref{ch:b2:dispersion-wave-packets} में। फैलाव का सूत्र
पुंज के फ़ूरिए संश्लेषण से आता है --- हर घटक $\eu^{\iu kx}$
कला $\eu^{-\iu\hbar k^2t/2m}$ बटोरता है, और किसी सम्मिश्र गुणांक के साथ गाउसी
समाकल बढ़ती चौड़ाई की कोई गाउसी देता है; फ़ूरिए
समाकल वर्ष 3 के गणित खंड का औज़ार है और हम परिणाम
मान लेते हैं। भौतिक रूप से: पुंज में संवेग $\Delta p
= \hbar/2\sigma_0$ पर फैले हैं, अतः वेग $\Delta v = \hbar/2m\sigma_0$ पर फैले,
और $t$ के बाद तेज़ घटक धीमों से $\Delta v\,t$ आगे निकल चुके हैं
--- सूत्र ठीक यही कहता है।
\end{proof}

\begin{example}[फैलता कौन है]\label{ex:b2:schrodinger-wave-functions:spreading}
$\sigma_0 = \qty{1}{nm}$ के भीतर स्थानीकृत कोई इलेक्ट्रॉन: $\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- किसी एक नैनोसेकंड के बाद उसका पुंज \qty{60}{\micro m}
चौड़ा है: अकेला छोड़ा इलेक्ट्रॉन टिका नहीं रहता; किसी परमाणु में बद्ध होकर वह
नहीं फैलता क्योंकि विभव उसे थामे रहता है (कोई \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्था})।
\qty{1}{\micro m} तक स्थानीकृत \qty{1e-15}{kg} का कोई धूलकण: $\tau =
\qty{2e7}{s}$, आठ महीने; कोई गोली: ब्रह्मांड की आयु से भी लंबा।
क्वांटम फैलाव असली है, और हर उस चीज़ के लिए अदृश्य जिसे कोई देख सके।
\end{example}

\begin{remark}[ऊर्जा और समय]\label{rem:b2:schrodinger-wave-functions:energytime}
कोई अवस्था जो केवल $\Delta t$ तक रहे --- जीवनकाल $\tau$ का कोई उत्तेजित परमाणु,
लंबाई $c\tau$ की कोई \omterm{prop:b2:scalar-light-model:coherence}{तरंग-रेल} --- $\Delta E \sim \hbar/\Delta t$ पर फैली
ऊर्जाओं का कोई अध्यारोपण है: किसी रेल की अवधि और उसकी
\omterm{prop:b2:scalar-light-model:coherence}{वर्णक्रमी चौड़ाई} के बीच \cref{ch:b2:scalar-light-model} का फ़ूरिए संबंध,
अब $E = \hbar\omega$ पर लगाया गया। \qty{10}{ns} जीवनकाल के किसी परमाण्विक स्तर की
चौड़ाई $\Delta E = \qty{7e-8}{eV}$ है, यानी \qty{16}{MHz} की कोई नैसर्गिक
रेखा-चौड़ाई; \cref{ch:b2:laser} की डॉप्लर चौड़ाई
(\qty{1.5}{GHz}) उसे किसी गैस में छिपा देती है, पर किसी ठंडे परमाणु या किसी विरामस्थ
आयन में नहीं।
\end{remark}

\begin{method}[तरंग फलनों के साथ काम]\label{meth:b2:schrodinger-wave-functions:method}
(1) प्रसामान्यित कीजिए; प्रायिकताएँ उस क्षेत्र पर $\int|\psi|^2$ हैं। (2)
समय-निरपेक्ष $V$: $\varphi(x)\eu^{-\iu Et/\hbar}$ ढूँढ़िए,
स्थायी समीकरण हल कीजिए, सीमा-प्रतिबंध लगाइए --- वही स्तर।
(3) विकास के लिए \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाओं} का अध्यारोपण कीजिए; दो स्तर
$(E_2 - E_1)/h$ पर विस्पंदित होते हैं। (4) मुक्त गति: $v_{\text{सम}} = p/m$, फैलाव काल
$2m\sigma_0^2/\hbar$। (5) धाराएँ: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$;
$A\eu^{\iu kx} + B\eu^{-\iu kx}$ के लिए, $j = (|A|^2 - |B|^2)\hbar k/m$ (कोई क्रॉस पद नहीं)।
(6) चिरसम्मत सीमा: एहरनफ़ेस्ट, और उपकरण के आकार के सामने
$\lambda$।
\end{method}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{फैलाव, व्युत्पन्न।} पुंज $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
वह अध्यारोपण $\int g(k)\eu^{\iu kx}\dd k$ है जिसमें $g(k) \propto
\eu^{-\sigma_0^2k^2}$ (मान लिया: किसी गाउसी का फ़ूरिए रूपांतर कोई
गाउसी है, चौड़ाई $1/2\sigma_0$, $k$ में)। (क) $\psi(x,t)$ लिखिए, हर घटक को
उसकी कला $\eu^{-\iu\hbar k^2t/2m}$ देकर। (ख) चरघातांक में वर्ग पूरा
कीजिए और $\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$ मान लीजिए धनात्मक वास्तविक भाग वाले सम्मिश्र $\alpha$ के लिए: दिखाइए कि
$|\psi|^2$ चौड़ाई $\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$ की कोई गाउसी है। (ग) $\tau$ की व्याख्या $\Delta v = \Delta p/m$ के ज़रिए कीजिए। (घ) निर्वात में
फोटॉनों का कोई पुंज क्यों नहीं फैलता?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{ऊर्जा--समय।} (क) $\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$ की तरह क्षय होती कोई अवस्था
(प्रायिकता $\eu^{-t/\tau}$) ऊर्जाओं का कोई अध्यारोपण है: यह मानकर
कि उसका ऊर्जा-वितरण पूर्ण चौड़ाई $\Gamma =
\hbar/\tau$ की कोई लोरेंत्ज़ी है, $\Gamma$ दीजिए $\tau = \qty{10}{ns}$ (परमाणु), \qty{1e-8}{s},
\qty{1e-23}{s} (कोई हैड्रॉनी अनुनाद, \unit{MeV} में) के लिए। (ख) \qty{10}{ns}
स्तर की नैसर्गिक रेखा-चौड़ाई आवृत्ति में; डॉप्लर चौड़ाई
\qty{1.5}{GHz} से तुलना कीजिए। (ग) नैसर्गिक चौड़ाई कोई कैसे देखता है (ठंडे परमाणु,
पाशित आयन)? (घ) उत्सर्जित \omterm{prop:b2:scalar-light-model:coherence}{तरंग-रेल} की
\omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} से संबंध जोड़िए।
\end{exercise}

\section{समस्या: किसी नली में इलेक्ट्रॉन और किसी तरंग-पुंज का फैलाव}

\begin{problem}[{सप्ताहांत समस्या --- कब कोई इलेक्ट्रॉन कण है,
कब तरंग}]\label{pb:b2:schrodinger-wave-functions:1}
आँकड़े: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $m_{\text{p}} = \qty{1.67e-27}{kg}$।

\textbf{भाग I --- कैथोड-किरण नली।}
इलेक्ट्रॉन नगण्य ऊर्जा के साथ किसी गरम कैथोड से निकलते हैं, $U = \qty{20}{kV}$ से
त्वरित होते हैं, किसी \qty{0.3}{mm} द्वारक से गुज़रते हैं, और \qty{40}{cm} दूर किसी
परदे से टकराते हैं।
\begin{enumerate}
  \item चाल और संवेग (अनापेक्षिकीय); डी ब्रॉय तरंगदैर्घ्य।
  \item गतिज ऊर्जा $4\%$ है $mc^2$ की: क्या इस समस्या के लिए
        अनापेक्षिकीय व्यवहार स्वीकार्य है?
  \item परदे तक उड़ान-काल।
  \item द्वारक से \omterm{def:b2:diffraction:huygens}{विवर्तन}: कोणीय फैलाव $\sim\lambda/a$ और
        वह धब्बा जो वह परदे पर बनाता है; किसी पिक्सेल
        (\qty{0.3}{mm}) से तुलना कीजिए।
  \item द्वारक पर हाइज़ेनबर्ग: $\Delta p_y \gtrsim \hbar/2a$ वही
        फैलाव देता है --- जाँचिए।
  \item पुंज के अनुदिश \qty{0.3}{mm} चौड़ा कोई पुंज: फैलाव काल
        $2m\sigma_0^2/\hbar$ और परदे पर चौड़ाई।
  \item विक्षेपण प्लेटें कोई अनुप्रस्थ बल लगाती हैं: एहरनफ़ेस्ट से,
        $\langle y\rangle$ कैसे चलता है? निष्कर्ष निकालिए कि नली कोई
        चिरसम्मत युक्ति है।
  \item इलेक्ट्रॉन तरंग का \omterm{def:b2:dispersion-wave-packets:relation}{कला वेग}; यह कोई विरोधाभास क्यों नहीं कि
        वह इलेक्ट्रॉन की चाल से भिन्न है?
\end{enumerate}

\textbf{भाग II --- \omterm{prop:b2:dispersion-wave-packets:group}{तरंग-पुंज}।}
\begin{enumerate}[resume]
  \item चौड़ाई $\sigma_0$ के किसी \omterm{prop:b2:laser:gaussian}{गाउसी पुंज} के लिए: $\Delta p$ और
        वेग-फैलाव $\Delta v$।
  \item $\sigma_0 = \qty{0.1}{nm}$, \qty{1}{nm}, \qty{1}{\micro m} वाले किसी इलेक्ट्रॉन के लिए;
        \qty{1}{nm} वाले किसी प्रोटॉन के लिए; \qty{10}{nm} वाले किसी
        विषाणु (\qty{1e-17}{kg}) के लिए फैलाव काल।
  \item किसी हाइड्रोजन परमाणु में कोई इलेक्ट्रॉन \qty{0.1}{nm} तक स्थानीकृत है
        और फैलता नहीं: क्यों?
  \item $\sigma_0 = \qty{1}{nm}$ वाला \qty{1}{eV} पर कोई मुक्त इलेक्ट्रॉन: उसकी चौड़ाई
        दुगुनी होने से पहले वह कितनी दूर चलता है? तरंगदैर्घ्य से
        तुलना कीजिए।
  \item किसी धातु में, चालन इलेक्ट्रॉन समूचे क्रिस्टल पर $\lambda \approx
        \qty{0.5}{nm}$ की तरंगें हैं: वे किस अर्थ में
        ``अस्थानीकृत'' हैं?
  \item \qty{100}{keV} पर किसी इलेक्ट्रॉन सूक्ष्मदर्शी का $\lambda \approx
        \qty{4}{pm}$ है, फिर भी वह केवल \qty{0.1}{nm} विभेदित करता है: क्यों (उसके चुंबकीय
        लेंस लगभग \qty{10}{mrad} का अर्ध-कोण स्वीकारते हैं)?
  \item वह कसौटी दीजिए जो तय करती है कि किसी वस्तु को
        \omterm{def:b2:schrodinger-wave-functions:psi}{तरंग फलन} से बरता जाए या न्यूटन से।
\end{enumerate}

\textbf{भाग III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाएँ} और संक्रमण।}
कोई इलेक्ट्रॉन $0 < x < L = \qty{1}{nm}$ में परिरुद्ध है; $\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$।
\begin{enumerate}[resume]
  \item इलेक्ट्रॉनवोल्ट में $E_1$, $E_2$, $E_3$; बोर आवृत्तियाँ
        $\nu_{21}$, $\nu_{31}$ और तरंगदैर्घ्य।
  \item जाँचिए कि $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ समय-निरपेक्ष है;
        मूल अवस्था में इलेक्ट्रॉन सबसे संभाव्य कहाँ है?
  \item $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$ के लिए: $|\psi|^2$ $t = 0$ और $t = T_{21}/2$ पर; खाका खींचिए।
  \item $\langle x\rangle(t)$ ($\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$ दिया गया):
        छलकन का आयाम।
  \item दोलायमान द्विध्रुव $e\langle x\rangle$ विकिरित करता है: किस
        आवृत्ति पर, और वह वर्णक्रमी रेखाओं तथा
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाओं} की स्थायित्व के बारे में क्या कहता है?
  \item अवस्था $\psi$ की ऊर्जा: क्या वह $E_1$ है, $E_2$, या कुछ
        और? कोई मापन क्या देता है?
\end{enumerate}

\textbf{भाग IV --- व्याख्या।}
\begin{enumerate}[resume]
  \item एकल-इलेक्ट्रॉन द्विझिरी प्रयोग में, क्या यादृच्छिक है
        और क्या नहीं?
  \item $A\eu^{\iu kx}$ की \omterm{prop:b2:schrodinger-wave-functions:current}{प्रायिकता धारा} लिखिए और उसे प्रति इकाई
        लंबाई $n$ इलेक्ट्रॉनों की किसी किरण-पुंज से चाल $v$ पर जोड़िए।
  \item श्रोडिंगर समीकरण को सम्मिश्र और समय में प्रथम कोटि का क्यों
        होना ही चाहिए (दो कारण)?
  \item पाँच पंक्तियों में सारांश दीजिए: \omterm{def:b2:schrodinger-wave-functions:psi}{तरंग फलन}, समीकरण,
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाएँ}, पुंज, और चिरसम्मत सीमा।
\end{enumerate}
\end{problem}
