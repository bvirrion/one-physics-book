\chapter{दृढ़ पिंड की यांत्रिकी}\label{ch:b2:rigid-body-mechanics}

फुटपाथ से किसी साइकिल के पहिये को देखिए: सड़क के पास वाली तीलियाँ लगभग
स्थिर और स्पष्ट दिखती हैं, ऊपर वाली तीलियाँ साइकिल-चालक से दुगुनी चाल से
चलती हुई धुँधली। पहिये के हर बिंदु का वेग अलग है, फिर भी पहिया एक ही
वस्तु है --- उसके बिंदु आपसी दूरियाँ बनाए रखते हैं, और यही एक प्रतिबंध
उनकी सारी गतियों को दो सदिशों में, एक स्थानांतरण और एक घूर्णन में, बाँध
देता है। वर्ष 1 के खंड में स्थिर अक्ष के परितः घूमते ठोस का विशेष प्रसंग
लिया गया था; यह अध्याय किसी \omterm{def:b2:rigid-body-mechanics:solid}{दृढ़ पिंड} की सामान्य शुद्धगतिकी, उसकी गतिज
ऊर्जा और कोणीय संवेग, उसकी गति के नियम, और वह नियम देता है जो हर पहिये,
टायर, ब्रेक और सीढ़ी पर राज करता है: शुष्क घर्षण का कूलॉम नियम।

\section{दृढ़ पिंड की शुद्धगतिकी}

\begin{definition}[दृढ़ पिंड]\label{def:b2:rigid-body-mechanics:solid}
\index{दृढ़ पिंड}
\emph{दृढ़ पिंड} (या ठोस) बिंदुओं का ऐसा निकाय है जिसके बिंदुओं की आपसी
दूरियाँ अचर रहती हैं: $\|\vect{AB}\|$ उसके हर बिंदु-युग्म $A$, $B$ के लिए
समय पर निर्भर नहीं करता। जिस तंत्र में ठोस का हर बिंदु विरामावस्था में
हो, वह \emph{ठोस से बद्ध तंत्र} कहलाता है।
\end{definition}

\begin{theorem}[दृढ़ पिंड का वेग क्षेत्र]\label{thm:b2:rigid-body-mechanics:field}
\index{घूर्णन सदिश}\index{वेग क्षेत्र (दृढ़ पिंड)}
हर क्षण एक सदिश $\vect\Omega$ ऐसा होता है, जो ठोस का \emph{\omterm{thm:b2:rigid-body-mechanics:field}{घूर्णन
सदिश}} कहलाता है (मात्रक \unit{rad/s}), कि उसके किन्हीं दो बिंदुओं
$A$ और $M$ के वेग किसी तंत्र $\mathcal R$ में
\[
\vect v_M = \vect v_A + \vect\Omega\wedge\vect{AM} .
\]
को संतुष्ट करते हैं। $\vect\Omega$ $A$ के चुनाव पर निर्भर नहीं करता; ठोस
में स्थिर कोई सदिश $\vect u$ $\dd\vect u/\dd t = \vect\Omega\wedge\vect u$ के अनुसार बदलता है।
\end{theorem}

\begin{proof}
मान लीजिए $(\vect e_1, \vect e_2, \vect e_3)$ ठोस से बद्ध कोई लांबिक
मात्रक आधार है। $\vect e_i\cdot\vect e_j = \delta_{ij}$ से गुणांक
$a_{ij} = \dot{\vect e}_i\cdot\vect e_j$ $a_{ij} + a_{ji} = 0$ को संतुष्ट करते हैं: अतः
प्रतिचित्रण $\vect u \mapsto \dot{\vect u}$ का आव्यूह (यहाँ $\vect u$ ठोस
में स्थिर है, $\vect u = \sum u_i\vect e_i$ और $u_i$ अचर) विषम-सममित है,
और कोई भी विषम-सममित $3\times3$ आव्यूह
$\vect u \mapsto \vect\Omega\wedge\vect u$ का आव्यूह होता है, जहाँ $\Omega_1 = a_{23}$,
$\Omega_2 = a_{31}$, $\Omega_3 = a_{12}$। इसे $\vect u =
\vect{AM}$ पर लगाइए: $\vect v_M - \vect v_A = \vect\Omega\wedge\vect{AM}$। यदि
$\vect\Omega'$ भी यही काम करता, तो $(\vect\Omega - \vect\Omega')\wedge
\vect{AM} = \vect 0$ प्रत्येक $M$ के लिए होता, अतः $\vect\Omega' = \vect\Omega$।
\end{proof}

\begin{remark}[तीन गतियाँ]\label{rem:b2:rigid-body-mechanics:motions}
\index{स्थानांतरण}\index{स्थिर अक्ष के परितः घूर्णन}
\emph{स्थानांतरण}: $\vect\Omega = \vect 0$, सब बिंदुओं का वेग एक ही
होता है (गति सीधी रेखा में हो, यह आवश्यक नहीं: विशाल-चक्र की कुर्सी
वृत्त पर स्थानांतरित होती है)। \emph{स्थिर अक्ष $\Delta$ के परितः घूर्णन}:
$\Delta$ के बिंदु विरामावस्था में रहते हैं, $\vect\Omega = \omega\,\vect e_\Delta$,
और अक्ष से $r$ दूरी पर का बिंदु $r|\omega|$ चाल से वृत्त पर चलता है
--- यही वर्ष 1 के खंड का प्रसंग था। सामान्य गति दोनों को जोड़ती है:
$\vect v_M = \vect v_G + \vect\Omega\wedge\vect{GM}$, अर्थात् द्रव्यमान
केंद्र के वेग से स्थानांतरण और उसके ऊपर
$G$ के परितः घूर्णन।
\end{remark}

\begin{definition}[बिना फिसले लुढ़कना]\label{def:b2:rigid-body-mechanics:rolling}
\index{बिना फिसले लुढ़कना}\index{सर्पण वेग}
दो ठोस $S_1$ और $S_2$ किसी बिंदु $I$ पर स्पर्श करते हैं। $S_1$ का $S_2$ पर
\emph{सर्पण वेग} $\vect v_{\text{सर्पण}} = \vect v_{I \in S_1}
- \vect v_{I \in S_2}$ है, अर्थात् उन दो भौतिक बिंदुओं के वेगों का अंतर
जो इस क्षण $I$ पर संपाती हैं। $S_1$ उस दशा में $S_2$ पर
\emph{बिना फिसले लुढ़कता} है जब $\vect v_{\text{सर्पण}} = \vect 0$ हो। स्थिर धरातल पर बिना फिसले
लुढ़कते $R$ त्रिज्या के पहिये के लिए, जिसके केंद्र का वेग $v_G$ और कोणीय
वेग $\omega$ है, यह प्रतिबंध
$v_G = R\omega$ बनता है (चिह्न ऐसे चुने गए हैं कि दाईं ओर लुढ़कता पहिया
दक्षिणावर्त घूमे)।
\end{definition}

\begin{proof}
\cref{thm:b2:rigid-body-mechanics:field} को पहिये पर $G$ और स्पर्श-बिंदु
$I$ के बीच लगाइए: $\vect v_{I\in S_1} = \vect v_G + \vect\Omega
\wedge\vect{GI}$; अब $\vect v_G = v_G\vect e_x$, $\vect\Omega = -\omega
\vect e_z$ और $\vect{GI} = -R\vect e_y$ रखने पर यह $(v_G - R\omega)\vect e_x$ होता है,
जिसका शून्य होना आवश्यक है।
\end{proof}

\begin{example}[लुढ़कते पहिये का वेग क्षेत्र]\label{ex:b2:rigid-body-mechanics:wheel}
लुढ़कते पहिये के लिए $\vect v_M = \vect\Omega\wedge\vect{IM}$: अर्थात् हर
क्षण पहिया \emph{अपने स्पर्श-बिंदु के परितः घूमता} है। स्पर्श-बिंदु
विरामावस्था में है, केंद्र $v = R\omega$ से चलता है, शिखर $2v$ से, और
$h$ ऊँचाई पर नेमि का बिंदु $\omega\sqrt{2Rh}$ चाल से चलता है, और वह भी
$\vect{IM}$ के लंबवत। वाल्व चक्रज खींचता है; चलनी से उछला पत्थर नेमि-बिंदु की चाल से,
अर्थात् $2v$ तक की चाल से, निकलता है
--- इसीलिए ट्रकों के मडगार्ड मायने रखते हैं।
\end{example}

\section{ठोस की गतिज राशियाँ}

\begin{proposition}[संवेग, कोणीय संवेग, गतिज ऊर्जा]\label{prop:b2:rigid-body-mechanics:kinetic}
\index{जड़त्व आघूर्ण}\index{कोणीय संवेग (ठोस)}\index{गतिज ऊर्जा (ठोस)}
$M$ द्रव्यमान और $G$ द्रव्यमान केंद्र वाले किसी ठोस के लिए, तंत्र $\mathcal R$ में:
\begin{itemize}
  \item उसका संवेग $\vect p = M\vect v_G$ है;
  \item यदि वह किसी अक्ष $\Delta$ के परितः ($\mathcal R$ में स्थिर, या
        $G$ से होकर जाती और दिशा में स्थिर) कोणीय वेग
        $\omega$ से घूमता है, तो उस अक्ष के \emph{अनुदिश} उसका कोणीय संवेग
        $L_\Delta = J_\Delta\,\omega$ होता है, जहाँ
        \[
        J_\Delta = \sum_i m_i r_i^2 = \int r^2\,\dd m
        \]
        $\Delta$ के परितः उसका \emph{\omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण}} है ($r$ अक्ष से दूरी;
        मात्रक \unit{kg.m^2});
  \item उसकी गतिज ऊर्जा (क्योनिग) $E_k = \tfrac12Mv_G^2 +
        E_k^*$ है, जहाँ भारकेंद्रीय तंत्र की गतिज ऊर्जा
        $E_k^* = \tfrac12J_{G}\omega^2$ होती है, जब घूर्णन $\omega$ पर
        $G$ से होकर जाती अक्ष के परितः हो --- और $E_k = \tfrac12J_\Delta\omega^2$,
        जब घूर्णन स्थिर अक्ष $\Delta$ के परितः हो।
\end{itemize}
\end{proposition}

\begin{proof}
संवेग और क्योनिग की प्रमेयें वर्ष 1 के खंड में किसी भी निकाय के लिए
सिद्ध की गई थीं। $\Delta$ के परितः घूर्णन के लिए, जहाँ यह अक्ष $O$ से होकर
जाती है, अक्ष से $r_i$ दूरी पर का बिंदु अक्ष के चारों ओर वृत्त पर $r_i\omega$ चाल से चलता
है; $O$ के परितः उसका कोणीय संवेग $\vect e_\Delta$ पर प्रक्षेपित करने पर
$m_ir_i^2\omega$ मिलता है (शेष घटक, जो सममित ठोस के लिए कट जाते हैं, यहाँ
अनावश्यक हैं); अब योग कीजिए। गतिज ऊर्जा: $\sum\tfrac12m_ir_i^2
\omega^2$।
\end{proof}

\begin{remark}[क्या स्वीकृत है]\label{rem:b2:rigid-body-mechanics:tensor}
किसी भी घूर्णन $\vect\Omega$ के लिए पूरा सदिश $\vect L_G$ साधारणतः
$\vect\Omega$ के समांतर नहीं होता: $\vect L_G = \mathbf J\,\vect\Omega$
होता है, जहाँ $\mathbf J$ \emph{जड़त्व प्रदिश} है, अर्थात् एक सममित आव्यूह
जिसके अभिलक्षणिक सदिश मुख्य अक्ष कहलाते हैं। इस अध्याय का हर ठोस किसी
स्थिर अक्ष के परितः या $G$ से होकर जाती किसी सममिति-अक्ष के परितः घूमता
है, जिसके लिए $\vect L = J\vect\Omega$ ठीक-ठीक सही है; सामान्य प्रसंग
वर्ष 3 के खंड का है।
\end{remark}

\begin{proposition}[जड़त्व आघूर्ण]\label{prop:b2:rigid-body-mechanics:inertia}
\index{ह्यूगेंस की प्रमेय}\index{समांतर-अक्ष प्रमेय}
$M$ द्रव्यमान वाले समांगी ठोसों के लिए, $G$ से होकर जाती अक्ष के परितः:
$\ell$ लंबाई की पतली छड़ (अक्ष लंबवत), $J = \tfrac1{12}M\ell^2$; $R$ त्रिज्या
का पतला वलय या बेलनाकार कोश (अपनी अक्ष), $J = MR^2$; चक्रिका या ठोस
बेलन, $J = \tfrac12MR^2$; ठोस गोला, $J = \tfrac25MR^2$; गोलीय
कोश, $J = \tfrac23MR^2$। \emph{\omterm{prop:b2:rigid-body-mechanics:inertia}{ह्यूगेंस की प्रमेय}} (समांतर अक्ष): अक्ष
$\Delta$ के परितः, जो अक्ष $\Delta_G$ (जो $G$ से होकर जाती है) के समांतर
और उससे $d$ दूरी पर है,
\[
J_\Delta = J_{\Delta_G} + Md^2 .
\]
\end{proposition}

\begin{proof}
छड़ के लिए: $\int_{-\ell/2}^{\ell/2}x^2\,(M/\ell)\dd x = M\ell^2/12$।
अब चक्रिका, जिसके छल्ले $(2M/R^2)r\,\dd r$ द्रव्यमान के हैं:
$\int_0^R r^2(2M/R^2)r\,\dd r = MR^2/2$। और गोला: सममिति से
$J = \tfrac23\int r^2\dd m$ (यहाँ $r$ केंद्र से दूरी है,
क्योंकि तीनों अक्षों पर औसत लेने पर $x^2 + y^2 = \tfrac23(x^2 + y^2 + z^2)$ होता है)
और $\int r^2\dd m = \int_0^R r^2(3M/R^3)r^2\dd r = \tfrac35MR^2$।
ह्यूगेंस: $\vect r_i$ को वह सदिश मानिए जो $\Delta$ से $m_i$ तक अक्ष के
लंबवत जाता है, और $\vect r_i = \vect d + \vect r_i^*$ रखिए; तब $\sum m_ir_i^2 = Md^2
+ 2\vect d\cdot\sum m_i\vect r_i^* + \sum m_ir_i^{*2}$ मिलता है, और बीच वाला योग
$G$ की परिभाषा से लुप्त हो जाता है।
\end{proof}

\begin{example}[एक गतिपालक चक्र]\label{ex:b2:rigid-body-mechanics:flywheel}
\qty{50}{kg} की और \qty{30}{cm} त्रिज्या की इस्पात-चक्रिका: $J = \tfrac12 \times
50 \times 0.09 = \qty{2.25}{kg.m^2}$; \qty{3000}{rpm} पर ($\omega =
\qty{314}{rad/s}$) वह $\tfrac12J\omega^2 = \qty{111}{kJ}$ संचित करती है, जो
\qty{50}{km/h} पर चलती छोटी कार की गतिज ऊर्जा है। ऊर्जा-भंडारण के
गतिपालक चक्र निर्वात में चुंबकीय धारकों पर कार्बन-रेशे के घूर्णक
\qty{50000}{rpm} पर घुमाते हैं: ऊर्जा $\omega^2$ की तरह बढ़ती है, और सीमा
नेमि की तनन-सामर्थ्य लगाती है।
\end{example}

\section{ठोस की गतिकी}

\begin{theorem}[ठोस की गति के नियम]\label{thm:b2:rigid-body-mechanics:laws}
\index{कोणीय संवेग प्रमेय (ठोस)}\index{ठोस पर बलों की शक्ति}
किसी गैलीलीय तंत्र में, बाह्य बलों $\vect F_i$ के अधीन, जो बिंदुओं
$A_i$ पर लगे हैं, किसी ठोस के लिए:
\begin{enumerate}
  \item (संवेग) $M\,\dd\vect v_G/\dd t = \sum\vect F_i$;
  \item ($G$ के परितः, या किसी स्थिर बिंदु $O$ के परितः, कोणीय संवेग)
        $\dd\vect L_G/\dd t = \sum\vect{GA_i}\wedge\vect F_i$; किसी स्थिर
        अक्ष $\Delta$ पर, या $G$ से होकर जाती स्थिर दिशा वाली उस अक्ष पर
        जिसके परितः ठोस $\omega$ से घूमता है, प्रक्षेपित करने पर:
        $J_\Delta\,\dd\omega/\dd t = \mathcal M_\Delta$, जहाँ दायाँ पक्ष अक्ष
        के परितः बाह्य बलों के आघूर्णों का योग है;
  \item (शक्ति) किसी ठोस पर बलों के समुच्चय की शक्ति
        $\mathcal P = \vect R\cdot\vect v_A + \vect{\mathcal M}_A\cdot
        \vect\Omega$ होती है, जहाँ $A$ ठोस का कोई भी बिंदु है, $\vect R$
        बलों का योग और $\vect{\mathcal M}_A$ $A$ के परितः उनका कुल आघूर्ण है;
        \emph{किसी ठोस के आंतरिक बलों की शक्ति शून्य होती है}, और
        $\dd E_k/\dd t = \mathcal P_{\text{बाह्य}}$।
\end{enumerate}
\end{theorem}

\begin{proof}
(1) और (2) बिंदु-निकायों के लिए वर्ष 1 के खंड की प्रमेयें ही हैं
($G$ के परितः कोणीय संवेग प्रमेय तब भी लागू होती है जब $G$ त्वरित हो,
क्योंकि भारकेंद्रीय तंत्र के जड़त्वीय बलों का आघूर्ण $G$ के परितः शून्य
होता है)। (3) $\vect v_{A_i} = \vect v_A + \vect\Omega
\wedge\vect{AA_i}$ रखने पर: $\sum\vect F_i\cdot\vect v_{A_i} = \vect R\cdot\vect v_A
+ \sum\vect F_i\cdot(\vect\Omega\wedge\vect{AA_i}) = \vect R\cdot\vect v_A +
\vect\Omega\cdot\sum\vect{AA_i}\wedge\vect F_i$ (मिश्र गुणन)। आंतरिक
बल: उनका योग और उनका कुल आघूर्ण दोनों लुप्त होते हैं (क्रिया--प्रतिक्रिया,
एक ही रेखा पर), अतः किसी ठोस पर उनकी शक्ति शून्य है --- पर किसी
विरूपणीय निकाय पर \emph{नहीं}। तब गतिज ऊर्जा प्रमेय में केवल बाह्य शक्ति बचती है।
\end{proof}

\begin{definition}[आदर्श धुरी]\label{def:b2:rigid-body-mechanics:pivot}
\index{धुरी (आदर्श)}\index{धुरी-धारक}
जब धारक के स्पर्श-बलों का आघूर्ण $\Delta$ के परितः शून्य हो, तब कोई ठोस
स्थिर अक्ष $\Delta$ के परितः \emph{आदर्श धुरी} पर (अर्थात् आदर्श धारक
में) घूमता है; तब वे कोई शक्ति नहीं देते (अक्ष पर $\vect v_A = \vect 0$,
$\mathcal M_\Delta = 0$)। वास्तविक धारक इसमें घर्षण-बलाघूर्ण
$-\Gamma_f\operatorname{sgn}\omega$ जोड़ देता है, या श्यान बलाघूर्ण $-h\omega$।
\end{definition}

\begin{proposition}[भौतिक लोलक]\label{prop:b2:rigid-body-mechanics:pendulum}
\index{भौतिक लोलक}\index{संयुक्त लोलक}
$M$ द्रव्यमान का कोई ठोस आदर्श धुरी पर किसी क्षैतिज स्थिर अक्ष $\Delta$
के परितः झूलता है, और उसका द्रव्यमान केंद्र अक्ष से $d$ दूरी पर है।
$\theta$ को नीचे की ऊर्ध्वाधर से $\vect{OG}$ का कोण मानने पर,
\[
J_\Delta\ddot\theta = -Mgd\sin\theta ,
\]
मिलता है, अतः छोटे दोलनों का आवर्तकाल $T = 2\pi\sqrt{J_\Delta/Mgd}$ है
--- जो $\ell_{\text{तुल्य}} = J_\Delta/Md$ लंबाई के सरल लोलक का आवर्तकाल है।
\end{proposition}

\begin{proof}
$\Delta$ के परितः कोणीय संवेग प्रमेय: भार का आघूर्ण
$-Mgd\sin\theta$ है, धुरी का शून्य। अब रैखिकीकरण कीजिए।
\end{proof}

\begin{example}[मीटर-पट्टी]\label{ex:b2:rigid-body-mechanics:rule}
एक सिरे पर टँगी मीटर-पट्टी: $J = \tfrac13M\ell^2$ (ह्यूगेंस से,
$\tfrac1{12}M\ell^2$ लेकर), $d = \ell/2$, $\ell_{\text{तुल्य}} = \tfrac23\ell =
\qty{0.667}{m}$ और $T = \qty{1.64}{s}$ --- अर्थात् पूरी पट्टी का लोलक अपने
केंद्र पर टँगे बिंदु-द्रव्यमान (\qty{1.42}{s}) से धीमे और सिरे पर टँगे
बिंदु-द्रव्यमान (\qty{2.0}{s}) से तेज़ धड़कता है।
\end{example}

\section{स्पर्श-बल और कूलॉम घर्षण}

\begin{definition}[स्पर्श-बल: अभिलंब और स्पर्शीय]\label{def:b2:rigid-body-mechanics:contact}
\index{अभिलंब प्रतिक्रिया}\index{घर्षण बल}
स्पर्श-बिंदु $I$ पर किसी आधार द्वारा ठोस पर लगाया गया बल दो भागों में
बँटता है: \emph{अभिलंब प्रतिक्रिया} $\vect N$, जो उभयनिष्ठ स्पर्श-समतल
के लंबवत और ठोस की ओर दिष्ट होती है ($N \ge 0$: आधार केवल धकेल सकता है),
और स्पर्श-समतल में स्थित \emph{स्पर्शीय घटक} $\vect T$, जो
\emph{घर्षण बल} है।
\end{definition}

\begin{theorem}[शुष्क घर्षण के कूलॉम नियम]\label{thm:b2:rigid-body-mechanics:coulomb}
\index{कूलॉम के घर्षण नियम}\index{घर्षण गुणांक}\index{स्थैतिक घर्षण}\index{गतिज घर्षण}
स्पर्श में दो शुष्क ठोसों के लिए, जिनका \emph{\omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण गुणांक}} $f$ पदार्थों
और उनकी पृष्ठ-दशा पर निर्भर करता है, पर न स्पर्श-क्षेत्रफल पर और न (अच्छे
सन्निकटन में) चाल पर:
\begin{itemize}
  \item यदि ठोस एक-दूसरे पर फिसलते हैं ($\vect v_{\text{सर्पण}} \ne
        \vect 0$), तो \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} \omterm{def:b2:rigid-body-mechanics:rolling}{सर्पण वेग} के विपरीत दिशा में होता है
        और उसका परिमाण $\|\vect T\| = f\|\vect N\|$ होता है (\omterm{thm:b2:rigid-body-mechanics:coulomb}{गतिज घर्षण});
  \item यदि वे नहीं फिसलते, तो $\|\vect T\| \le f\|\vect N\|$ (\omterm{thm:b2:rigid-body-mechanics:coulomb}{स्थैतिक
        घर्षण}): तब \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} उस सीमा के भीतर वही होता है जो गति के शेष
        नियम माँगते हैं, और उसकी दिशा पहले से ज्ञात नहीं होती।
\end{itemize}
\emph{आदर्श स्पर्श} $f = 0$ का आदर्शीकरण है: $\vect T = \vect 0$।
\end{theorem}
\admitted

\begin{remark}[कूलॉम के नियमों को पढ़ना]\label{rem:b2:rigid-body-mechanics:reading}
\index{घर्षण शंकु}
पहला नियम एक समीकरण है, दूसरा एक असमिका --- और घर्षण की समस्याओं की सारी
कठिनाई इसी में है। \emph{विधि} यह है कि पहले फिसलन न होने की
\emph{कल्पना} कीजिए, $\vect T$ और $\vect N$ निकालिए, और
$\|\vect T\| \le f\|\vect N\|$ की \emph{जाँच} कीजिए; जाँच विफल हो तो ठोस
फिसलता है, सर्पण के विपरीत $\|\vect T\| = f\|\vect N\|$ होता है, और
समस्या फिर से हल की जाती है। ज्यामितीय रूप से, स्पर्श-बल को
$\varphi$ अर्ध-कोण के \emph{\omterm{rem:b2:rigid-body-mechanics:reading}{घर्षण शंकु}} के भीतर
रहना पड़ता है, जहाँ $\tan\varphi
= f$ है। प्रारूपिक मान: सूखे डामर पर रबर $f \approx 0.8$--$1$, गीले
डामर पर $0.4$--$0.6$, बर्फ पर $0.1$; इस्पात पर इस्पात सूखा $0.15$, चिकनाई
के साथ $0.05$; वास्तव में स्थैतिक गुणांक गतिज गुणांक से कुछ बड़ा होता है,
इसीलिए एक बार शुरू हो जाने पर फिसलन रोकना कठिन होता है।
\end{remark}

\begin{proposition}[स्पर्श-बलों की शक्ति; बिना फिसले लुढ़कना]\label{prop:b2:rigid-body-mechanics:power}
\index{लोटनिक घर्षण}
स्थिर आधार पर चलते ठोस पर स्पर्श-बलों की शक्ति
$\mathcal P = \vect T\cdot\vect v_{I\in S}$ होती है (अभिलंब बल
स्पर्श-बिंदु के वेग के लंबवत है, जो स्पर्श-समतल में रहता है)।
\emph{\omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले लुढ़कने}} में यह शक्ति शून्य होती है: तब घर्षण कोई कार्य
नहीं करता, यद्यपि गति के लिए वह अनिवार्य हो सकता है।
फिसलन में वह $-f N v_{\text{सर्पण}} < 0$ होती है: अर्थात् ऊष्मा।
\end{proposition}

\begin{proof}
$\mathcal P = (\vect N + \vect T)\cdot\vect v_{I\in S}$ है और $\vect
v_{I\in S} = \vect v_{\text{सर्पण}}$ स्पर्श-समतल में रहता है, अतः केवल
$\vect T$ का योगदान बचता है; \omterm{def:b2:rigid-body-mechanics:rolling}{सर्पण वेग} शून्य होने पर वह शून्य है, और
ठोसों के फिसलने पर वह $-fNv_{\text{सर्पण}}$ है ($\vect T$ $\vect
v_{\text{सर्पण}}$ के विपरीत)। वास्तविक लुढ़कन
टायर और धरातल के विरूपण से थोड़ी ऊर्जा खोती है (\emph{\omterm{prop:b2:rigid-body-mechanics:power}{लोटनिक घर्षण}}, जिसका
प्रतिरूप छोटा बलाघूर्ण $\Gamma_r = \mu_rNR$ या तुल्य बल $\mu_rN$ है, और
सड़क पर टायर के लिए $\mu_r \approx 0.01$): यह एक अलग और कहीं छोटा
प्रभाव है।
\end{proof}

\begin{method}[आनत तल पर लुढ़कता ठोस]\label{met:b2:rigid-body-mechanics:incline}
\index{आनत तल पर लुढ़कना}
$R$ त्रिज्या, $M$ द्रव्यमान और अपनी अक्ष के परितः $J =
kMR^2$ \omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण} वाला कोई समांगी ठोस ($k = 1$ वलय, $\tfrac12$ चक्रिका, $\tfrac25$ गोला)
$\alpha$ कोण के आनत तल पर, जिसका \omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण गुणांक}
$f$ है, छोड़ा जाता है। \omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले लुढ़कने} की कल्पना कीजिए:
\begin{enumerate}
  \item ढाल के अनुदिश संवेग: $M\dot v = Mg\sin\alpha - T$; अभिलंब:
        $N = Mg\cos\alpha$;
  \item $G$ के परितः कोणीय संवेग: $kMR^2\dot\omega = TR$;
  \item फिसलन नहीं: $v = R\omega$, अतः $T = kM\dot v$ और
        \[
        \dot v = \frac{g\sin\alpha}{1 + k} , \qquad T = \frac{k}{1 + k}\,
        Mg\sin\alpha ;
        \]
  \item जाँच: $T \le fN \iff \tan\alpha \le f\,\dfrac{1 + k}{k}$।
        इससे अधिक ढाल पर ठोस फिसलता है, $T = fMg\cos\alpha$, $\dot v =
        g(\sin\alpha - f\cos\alpha)$ और $\dot\omega = fg\cos\alpha/kR$ हो जाते हैं।
\end{enumerate}
ऊर्जा-जाँच (फिसलन नहीं, घर्षण की शक्ति शून्य): $\tfrac12(1 + k)Mv^2 =
Mgh$ से $v = \sqrt{2gh/(1 + k)}$ मिलता है --- अर्थात् गोला चक्रिका को
हराता है और चक्रिका वलय को, चाहे द्रव्यमान और त्रिज्या कुछ भी हों।
\end{method}

\begin{example}[कार को त्वरित होने के लिए घर्षण क्यों चाहिए]\label{ex:b2:rigid-body-mechanics:car}
अग्र-चालित कार: इंजन अगले पहियों पर बलाघूर्ण लगाता है; कार पर लगा
एकमात्र बाह्य क्षैतिज बल टायरों पर सड़क का घर्षण है, और यही बल कार को
त्वरित करता है। पूरा भार चालक पहियों पर हो तो सूखे डामर पर अधिकतम त्वरण
$fg \approx \qty{8}{m/s^2}$ होता है --- और उसका आधा, यदि चालक धुरी पर
आधा ही भार टिका हो; बर्फ पर $\qty{1}{m/s^2}$। इंजन का कार्य सड़क नहीं करती
(उसका स्पर्श-बिंदु तो विरामावस्था में है): वह आंतरिक कार्य है, जिसे
संचरण-तंत्र गतिज ऊर्जा में बदलता है --- ठीक वैसे ही जैसे चलते हुए
व्यक्ति को ऊर्जा उसकी माँसपेशियों के आंतरिक बल देते हैं, धरती नहीं।
\end{example}

\begin{example}[फिसलन से लुढ़कन तक]\label{ex:b2:rigid-body-mechanics:bowling}
बोलिंग की गेंद बिना चक्रण के $v_0$ से फिसलती हुई छोड़ी जाती है। \omterm{thm:b2:rigid-body-mechanics:coulomb}{गतिज
घर्षण} $fMg$ केंद्र को धीमा करता है ($\dot v = -fg$) और गेंद को चक्रण देता है
($\tfrac25MR^2\dot\omega = fMgR$, $\dot\omega = 5fg/2R$), जब तक $v = R\omega$ न हो जाए:
यह $t_1 = 2v_0/7fg$ पर होता है, और तब $v_1 = \tfrac57v_0$। उसके बाद वह
बिना फिसले अचर चाल से लुढ़कती है। चाल का दो-सातवाँ भाग, और गतिज ऊर्जा का
$1 - (\tfrac57)^2\tfrac75 = \tfrac27$ भाग, फिसलन में ऊष्मा बनकर खो जाता है
--- $f$ चाहे कुछ भी हो; $f$ केवल यह तय करता है कि कितनी जल्दी।
\end{example}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:1}
\omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण}: (क) \qty{1.0}{m} लंबी, \qty{0.50}{kg} की छड़, पहले उसके

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:2}
कोई साइकिल \qty{70}{cm} व्यास के पहियों पर \qty{18}{km/h} से चलती है।
पहियों का कोणीय वेग; धरातल-तंत्र में वाल्व की चाल जब वह पहिये के शिखर
पर, तल पर, और सामने (धुरी की ऊँचाई पर) हो; साइकिल के ढाँचे के सापेक्ष
चलनी की चाल। चलनी में फँसा पत्थर शिखर पर छूट जाता है: वह किस चाल से
निकलता है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:3}
कोई गतिपालक चक्र \qty{25}{cm} त्रिज्या की \qty{40}{kg} इस्पात-चक्रिका है।
(क) \qty{6000}{rpm} पर संचित ऊर्जा। (ख) उसे विराम से \qty{2.0}{min} में
घुमा देने के लिए आवश्यक अचर बलाघूर्ण; घुमाव के अंत में शक्ति। (ग) किसी
बस में उसे \qty{10}{s} तक \qty{20}{kW} देना है: वह किस चाल तक धीमा हो
जाएगा? (घ) तेज़ गतिपालक चक्र निर्वात में क्यों चलाए जाते हैं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:rigid-body-mechanics:4}
$R$ त्रिज्या की एक समांगी चक्रिका अपनी नेमि के किसी बिंदु से जाती क्षैतिज
अक्ष के परितः ऊर्ध्वाधर तल में झूलती है। उस अक्ष के परितः \omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण};
$R = \qty{20}{cm}$ के लिए छोटे दोलनों का आवर्तकाल; तुल्य सरल लोलक की
लंबाई। वही प्रश्न, जब अक्ष केंद्र से $R/2$ दूरी पर हो। सबसे छोटे
आवर्तकाल के लिए अक्ष कहाँ होनी चाहिए?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:5}
समान त्रिज्या के एक वलय, एक चक्रिका और एक गोले को \qty{2.0}{m} ऊँचे,
\qty{20}{\degree} ढाल वाले आनत तल के शिखर से एक साथ छोड़ा जाता है, $f =
0.30$। (क) जाँचिए कि तीनों बिना फिसले लुढ़कते हैं। (ख) प्रत्येक की तल पर
चाल; पहुँचने का क्रम। (ग) प्रत्येक का लिया गया समय। (घ) दिखाइए कि गोले पर
\omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} $\tfrac27Mg\sin\alpha$ है और वह कोई कार्य नहीं करता। (ङ) किस ढाल
पर वलय फिसलने लगेगा? गोला?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:6}
\emph{चक्रिका-ब्रेक।} किसी पहिये और उसकी ब्रेक-चक्रिका का $J = \qty{1.2}{kg.m^2}$
है और वे \qty{900}{rpm} पर घूमते हैं; दो पैड चक्रिका पर \qty{12}{cm} की
औसत त्रिज्या पर दबते हैं, प्रत्येक \qty{800}{N} अभिलंब बल से, $f = 0.40$।
ब्रेक-बलाघूर्ण; कोणीय मंदन; रुकने का समय और चक्करों की संख्या;
निकली ऊष्मा, और \qty{1.5}{kg} इस्पात-चक्रिका का ताप-वर्धन
($c = \qty{470}{J/(kg.K)}$) यदि वह सारी ऊष्मा अपने पास रखे।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:7}
\emph{वास्तविक घिरनी वाली एटवुड मशीन।} $m_1 = \qty{2.0}{kg}$
और $m_2 = \qty{1.5}{kg}$ द्रव्यमान एक रस्सी से, आदर्श धुरी पर टिकी $R =
\qty{10}{cm}$ त्रिज्या और $J = \qty{5.0e-3}{kg.m^2}$ \omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण} वाली
घिरनी पर, लटके हैं; रस्सी घिरनी पर नहीं फिसलती। (क) दोनों
रस्सी-तनाव अलग क्यों हैं? (ख) दोनों द्रव्यमानों और घिरनी के
समीकरण; त्वरण। (ग) दोनों तनाव। (घ) द्रव्यमानहीन घिरनी से तुलना
कीजिए; ऊर्जा से त्वरण फिर से निकालिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:8}
\emph{सीढ़ी।} $\ell$ लंबाई और $m$ द्रव्यमान की एक समांगी सीढ़ी क्षैतिज से
$\theta$ कोण पर किसी चिकनी ऊर्ध्वाधर दीवार से टिकी है;
धरातल का \omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण गुणांक} $f$ है। (क) बल और पाद के परितः उनके आघूर्ण;
साम्य के समीकरण। (ख) दिखाइए कि साम्य के लिए
$\tan\theta \ge 1/2f$ आवश्यक है। (ग) $f = 0.40$ के लिए न्यूनतम कोण। (घ)
\qty{60}{\degree} पर लगी सीढ़ी पर $3m$ द्रव्यमान का व्यक्ति चढ़ता है: वह
($\ell$ के किस अंश तक) कितनी ऊपर जा सकता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:rigid-body-mechanics:9}
\emph{पश्च-चक्रण।} किसी बिलियर्ड-गेंद को ऐसे मारा जाता है कि वह
केंद्र-चाल $v_0$ और पश्च-चक्रण $\omega_0$ (लुढ़कन के विपरीत घूर्णन) लेकर
निकले। \omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण गुणांक} $f$। (क) फिसलते समय $v$ और $\omega$ के
समीकरण। (ख) वह समय जिस पर वह \omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले लुढ़कने} लगती है और तब
उसकी चाल, $v_0$ और $R\omega_0$ के फलन के रूप में। (ग) गेंद के खिलाड़ी की
ओर लौट आने के लिए $R\omega_0$ पर प्रतिबंध। (घ) अंक:
$v_0 = \qty{2.0}{m/s}$, $R\omega_0 = \qty{6.0}{m/s}$, $f = 0.2$,
$R = \qty{2.9}{cm}$।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:10}
\emph{फिरकी।} कोई फिरकी (द्रव्यमान $M$, अपनी अक्ष के परितः \omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण} $J = kMR^2$,
बाहरी त्रिज्या $R$) क्षैतिज मेज़ पर रखी है; $r < R$ त्रिज्या की उसकी
भीतरी नाभि पर लिपटे धागे को क्षैतिज दिशा में बल
$F$ से खींचा जाता है, और धागा नाभि के \emph{नीचे} से निकलता है। (क)
\omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले लुढ़कना} मानकर संवेग और कोणीय-संवेग के
समीकरण लिखिए और केंद्र का त्वरण निकालिए; फिरकी किस ओर
लुढ़कती है? (ख) आवश्यक \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल}, और \omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले लुढ़कने} के लिए $F$ पर
प्रतिबंध। (ग) वही प्रश्न, जब धागा क्षैतिज से
$\beta$ कोण ऊपर की ओर खींचा जाए; दिखाइए कि $\cos\beta = r/R$ होने पर
फिरकी स्थिर रहती है, और इसे ज्यामितीय रूप से समझाइए (स्पर्श-बिंदु से
जाती तात्क्षणिक अक्ष)। (घ) $\cos\beta < r/R$ पर क्या होता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:11}
\emph{बिलियर्ड-गेंद कहाँ मारें।} कोई क्यू विराम में पड़ी गेंद को क्षैतिज
संवेग-आवेग $P$ (बहुत छोटी अवधि का बल) देती है; गेंद की त्रिज्या $R$ है
और प्रहार मेज़ से $h$ ऊँचाई पर होता है। (क) प्रहार के तुरंत बाद संवेग और कोणीय संवेग
(प्रहार के दौरान घर्षण नगण्य)। (ख) दिखाइए कि $h = \tfrac75R$ होने पर गेंद
तुरंत बिना फिसले लुढ़कती है, इससे ऊपर उसमें अग्र-चक्रण होता है और
नीचे पश्च-चक्रण। (ग) $h = R$ (केंद्र पर प्रहार) के लिए, $f = 0.2$ और
$v_0 = \qty{3}{m/s}$ पर गेंद लुढ़कने से पहले कितनी दूर फिसलती है? (घ)
बिलियर्ड-मेज़ की गद्दी $\tfrac75R$ ऊँचाई पर क्यों बनाई जाती है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:rigid-body-mechanics:12}
\emph{कटोरे में बेलन।} $r$ त्रिज्या का कोई ठोस बेलन $R > r$ त्रिज्या के
स्थिर बेलनाकार पृष्ठ के भीतर बिना फिसले लुढ़कता है; दोनों अक्ष
क्षैतिज और समांतर हैं। $\theta$ को ऊर्ध्वाधर से केंद्र-रेखा का कोण
मानिए। (क) $\dot\theta$ के पदों में बेलन का अपनी ही अक्ष के परितः कोणीय
वेग व्यक्त कीजिए (संकेत: स्पर्श-बिंदु
विरामावस्था में है)। (ख) गतिज और स्थितिज ऊर्जा; दिखाइए कि छोटे दोलनों का
आवर्तकाल $T = 2\pi\sqrt{3(R - r)/2g}$ है। (ग) कटोरे में बिना घर्षण फिसलते
बिंदु से तुलना कीजिए। (घ) आयाम $\theta_0$ (छोटे कोण) पर \omterm{def:b2:rigid-body-mechanics:rolling}{बिना फिसले
लुढ़कने} के लिए न्यूनतम \omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण
गुणांक}।
\end{exercise}

\section{समस्या: पहियों पर चलता वाहन}

\begin{problem}[{सप्ताहांत समस्या --- कार अपने टायरों से क्या माँगती है:
लुढ़कना, त्वरण, ब्रेक, और वह एक बल जो यह सब करता है}]
\label{pb:b2:rigid-body-mechanics:1}

$M = \qty{1200}{kg}$ द्रव्यमान की किसी कार का धुरांतर $L = \qty{2.6}{m}$ है;
उसका द्रव्यमान केंद्र $h = \qty{0.55}{m}$ ऊँचाई पर और क्षैतिज दूरियों
$a = \qty{1.1}{m}$ (अगली धुरी से) तथा $b = \qty{1.5}{m}$ (पिछली धुरी से)
पर है। चारों पहियों में से हर एक की त्रिज्या $R = \qty{0.31}{m}$, द्रव्यमान $m =
\qty{15}{kg}$ और अपनी धुरी के परितः \omterm{prop:b2:rigid-body-mechanics:kinetic}{जड़त्व आघूर्ण} $J = \qty{1.0}{kg.m^2}$ है।
टायर--सड़क \omterm{thm:b2:rigid-body-mechanics:coulomb}{घर्षण गुणांक} $f = 0.80$ (सूखा); लोटनिक-घर्षण
गुणांक $\mu_r = 0.012$। $g = \qty{9.81}{m/s^2}$।

\medskip
\noindent\textbf{भाग I --- लुढ़कना।}
\begin{enumerate}
  \item कार $v = \qty{90}{km/h}$ से चलती है। पहियों का कोणीय वेग;
        सड़क-तंत्र में टायर के शिखर, तल और सामने वाले बिंदुओं
        के वेग।
  \item कार की कुल गतिज ऊर्जा, जिसमें पूरे की स्थानांतरी गति और चारों
        पहियों का घूर्णन अलग-अलग दिखाइए; पहियों के घूर्णन का अंश
        कितना है?
  \item समतल धरातल पर विराम में, संवेग और कोणीय-संवेग के समीकरणों से
        अभिलंब बल $N_f$ (दोनों अगले पहिये मिलाकर)
        और $N_r$ (दोनों पिछले) --- कौन-सी धुरी अधिक भार उठाती है?
  \item हर पहिये पर लोटनिक-घर्षण बलाघूर्ण $\mu_rNR$ है। \qty{90}{km/h}
        पर \omterm{prop:b2:rigid-body-mechanics:power}{लोटनिक घर्षण} से क्षयित शक्ति; वायुगतिक कर्षण
        $\tfrac12\rho C_xSv^2$ से तुलना कीजिए, जिसमें $C_xS =
        \qty{0.70}{m^2}$, $\rho = \qty{1.2}{kg/m^3}$।
  \item इंजन बंद कर दिया जाता है और कार समतल धरातल पर लुढ़कती जाती है:
        $v(t)$ का समीकरण लिखिए (पहियों के जड़त्व को प्रभावी
        द्रव्यमान के रूप में जोड़िए) और कर्षण को नगण्य मानकर
        \qty{90}{km/h} से \qty{45}{km/h} तक धीमा होने की दूरी आँकिए; फिर
        केवल कर्षण लेकर।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- त्वरण।}
कार अग्र-चालित है; इंजन दोनों अगले पहियों पर मिलाकर बलाघूर्ण $\Gamma$
लगाता है। यहाँ कर्षण और \omterm{prop:b2:rigid-body-mechanics:power}{लोटनिक घर्षण} की उपेक्षा कीजिए।
\begin{enumerate}[resume]
  \item पूरी कार पर लगे बाह्य बल खींचिए। कौन-सा बल उसे
        त्वरित करता है? कार पर सड़क की शक्ति क्या है, और
        गतिज ऊर्जा कहाँ से आती है?
  \item कार का संवेग समीकरण, अगले पहियों का अपनी धुरी के परितः
        कोणीय-संवेग समीकरण, और बिना फिसलने का
        प्रतिबंध लिखिए; दिखाइए कि त्वरण $\dot v =
        \Gamma/R\,/\,(M + 4J/R^2)$ है और प्रभावी द्रव्यमान निकालिए।
  \item पूरी कार के लिए द्रव्यमान केंद्र के परितः कोणीय संवेग
        (पहियों के कोणीय संवेग नगण्य मानिए): दिखाइए कि
        त्वरण $\dot v$ पर भार-स्थानांतरण $N_r - N_r^0 =
        Mh\dot v/L$ है --- अर्थात् अगली धुरी हलकी होती है, पिछली भारी।
  \item चालक (अगले) पहियों को घिसटाए बिना अधिकतम त्वरण:
        दिखाइए कि $\dot v_{\max} = fgb/(L + fh)$ है और उसका मान
        निकालिए; पश्च-चालित कार के लिए वही, $\dot v_{\max} =
        fga/(L - fh)$। कौन-सी रचना अधिक ज़ोर से त्वरित होती है, और
        ड्रैगस्टर भार पीछे क्यों रखते हैं?
  \item $\dot v_{\max}$ के लिए पहियों पर आवश्यक इंजन-बलाघूर्ण;
        \qty{50}{km/h} पर उससे मिलती शक्ति।
  \item वही कार बर्फ पर ($f = 0.10$): अधिकतम त्वरण और
        \qty{50}{km/h} तक पहुँचने का समय।
  \item अगले टायरों पर \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} टायर के साथ क्या करता है
        --- फिसलन या नहीं --- और बर्फ पर घिसटता पहिया कार को पकड़ बनाए
        रखने वाले पहिये से \emph{कम} त्वरण क्यों देता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- ब्रेक।}
\begin{enumerate}[resume]
  \item ब्रेक हर पहिये पर बलाघूर्ण लगाते हैं। दिखाइए कि लुढ़कते रहने
        वाले पहिये के लिए कार पर ब्रेक-बल टायर पर सड़क का
        घर्षण ही है, और यह कि ब्रेक-बलाघूर्ण
        प्रति पहिया $fN_iR$ से परिबद्ध है।
  \item मंदन $\gamma$ पर भार-स्थानांतरण: अगली धुरी का भार
        $N_f = Mg\,b/L + Mh\gamma/L$; $\gamma = fg$ पर अंक।
  \item चारों पहिये फिसलन की सीमा पर होने पर अधिकतम मंदन;
        \qty{90}{km/h} से सूखी सड़क पर, और गीली सड़क ($f = 0.45$) पर,
        रुकने की दूरी।
  \item यदि ब्रेक ऐसे लगाए जाएँ कि अगला और पिछला बलाघूर्ण बराबर हों,
        तो ज़ोर से ब्रेक लगाने पर कौन-सी धुरी पहले जाम होती है? जाम हुई
        पिछली धुरी खतरनाक क्यों है (फिसलते टायर पर \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} की
        दिशा सोचिए)?
  \item कोई पहिया जाम होकर फिसलता है: सड़क से मिलता मंदन, और
        $f_{\text{गतिज}} =
        0.7$ बनाम $f_{\text{स्थैतिक}} = 0.8$ लेने पर शिखर ब्रेक-बल का कितना
        अंश खो जाता है। समझाइए कि जाम-रोधी तंत्र क्या करता है और वह
        ब्रेक को स्पंदित क्यों करता है।
  \item \qty{90}{km/h} से ब्रेक में निकली ऊष्मा; चार \qty{6}{kg} इस्पात
        चक्रिकाओं का ताप-वर्धन ($c = \qty{470}{J/(kg.K)}$);
        पहाड़ी सड़कों पर आपात-ढलान क्यों बनाई जाती हैं?
  \item कोई विद्युत कार अपनी मोटर से \qty{70}{\%} दक्षता पर ब्रेक-ऊर्जा
        वापस लेती है: \qty{90}{km/h} से हर बार रुकने पर वापस मिली
        ऊर्जा; \qty{50}{kWh} की बैटरी कितने ऐसे रुकावों के
        बराबर है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- मोड़ और ज़मीन से उठा पहिया।}
\begin{enumerate}[resume]
  \item कार समतल सड़क पर
        $\rho = \qty{50}{m}$ त्रिज्या के वृत्त पर $v$ से मुड़ती है:
        \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बलों} को $Mv^2/\rho$ देना पड़ता है। सूखी और गीली सड़क पर
        अधिकतम मोड़-चाल।
  \item गति की दिशा के अनुदिश द्रव्यमान केंद्र के परितः कोणीय संवेग:
        दिखाइए कि बाहरी पहियों पर $\Delta N =
        Mv^2h/\rho w$ का अतिरिक्त भार आता है, जहाँ $w = \qty{1.5}{m}$
        ट्रैक-चौड़ाई है; वह चाल जिस पर भीतरी पहिये उठ जाते हैं
        ($\Delta N = Mg/2$), और उसकी फिसलन-चाल से तुलना: यह कार पहले
        पलटती है या फिसलती है?
  \item मोड़ $\beta$ कोण पर बंकित है। दिखाइए कि चाल
        $v = \sqrt{g\rho\tan\beta}$ पर घर्षण की आवश्यकता ही नहीं रहती;
        $\beta = \qty{10}{\degree}$ के लिए मान।
  \item कोई पहिया ज़मीन से उठाकर हाथ से \qty{60}{rpm} तक
        घुमाया जाता है; उसका धारक \qty{0.05}{N.m} का घर्षण-बलाघूर्ण
        लगाता है। वह कितनी देर घूमता है? खोई गतिज ऊर्जा।
  \item घूमते पहिये को सड़क पर रख दिया जाता है (कार विराम में, पहिया
        \qty{60}{rpm} पर, $N = \qty{3}{kN}$): रुकने से पहले फिसलन का समय,
        और वह दूरी जितनी कार खिसकती यदि वह स्वतंत्र रूप से लुढ़क
        सकती (कार को स्वतंत्र पहियों पर $M$ द्रव्यमान मानिए, शेष
        पहियों का जड़त्व छोड़ दीजिए)।
  \item एक सारणी में सार दीजिए: चारों दशाएँ (लुढ़कना, त्वरण,
        ब्रेक, मोड़), प्रत्येक में काम करने वाला बल और
        कूलॉम की वह सीमा जिसका उसे पालन करना है।
\end{enumerate}
\end{problem}
