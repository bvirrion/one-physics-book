\chapter{तरलों में ध्वनि-तरंगें}\label{ch:b2:sound-waves}

बिजली की चमक और गरज के बीच के सेकंड गिनिए और तीन से भाग दीजिए: वही
किलोमीटरों में दूरी है। ध्वनि हवा को एक तिहाई किलोमीटर प्रति सेकंड से
पार करती है, पानी को डेढ़ किलोमीटर से, और कोई पराश्रव्य स्कैनर शरीर के
भीतर से आती प्रतिध्वनियों का समय दस लाखवें सेकंड तक नापकर अपना चित्र
खींचता है। यह अध्याय \cref{ch:b2:euler-bernoulli,ch:b2:fluid-kinematics}
की तरल-यांत्रिकी को विराम में पड़े उस तरल पर लगाता है जिसे कोई छोटी
दाब-लहर विक्षुब्ध करती है, और वहाँ \omterm{thm:b2:waves-on-strings:equation}{दालांबेर समीकरण} फिर मिलता है: \omterm{thm:b2:sound-waves:equation}{ध्वनि
की चाल}, ऊर्जा कौन ले जाता है, तेज़ कितना तेज़ है, ध्वनि आकाश में कैसे
फैलती है, और गुज़रता सायरन अपना स्वर क्यों बदल देता है।

\section{ध्वानिक सन्निकटन}

\begin{definition}[ध्वानिक विक्षोभ]\label{def:b2:sound-waves:approximation}
\index{ध्वानिक सन्निकटन}\index{अधिदाब}
एकसमान दाब $P_0$ और घनत्व $\rho_0$ का कोई तरल विराम में है और
विक्षुब्ध किया जाता है: $P = P_0 + p(M, t)$, $\rho = \rho_0 + \rho_1(M, t)$, वेग
$\vect v(M, t)$। \emph{ध्वानिक सन्निकटन} छोटी राशियों $p$
(\emph{अधिदाब}, या ध्वानिक दाब), $\rho_1$ और $\vect v$ में केवल पहली
कोटि रखता है: $|p| \ll P_0$, $|\rho_1| \ll \rho_0$, $v \ll c$,
और हर \omterm{def:b2:fluid-kinematics:particle}{तरल कण} जो रूपांतरण भोगता है वह इतना तेज़ होता है
कि वह कोई ऊष्मा नहीं बदलता: वह \emph{समएन्ट्रॉपी} है, जिसकी
संपीड्यता $\chi_S = \frac1\rho\bigl(\frac{\partial\rho}{\partial P}\bigr)_S$ है।
गुरुत्व छोड़ दिया गया है।
\end{definition}

\begin{theorem}[ध्वनि-तरंगें]\label{thm:b2:sound-waves:equation}
\index{ध्वनि की चाल}\index{तरंग-समीकरण (ध्वानिकी)}
पहली कोटि तक \omterm{thm:b2:euler-bernoulli:euler}{ऑयलर समीकरण}, द्रव्यमान-संरक्षण और समएन्ट्रॉपी
नियम ये पढ़े जाते हैं:
\[
\rho_0\frac{\partial\vect v}{\partial t} = -\operatorname{\vect{grad}}p , \qquad
\frac{\partial\rho_1}{\partial t} + \rho_0\operatorname{div}\vect v = 0 , \qquad
\rho_1 = \rho_0\chi_Sp ,
\]
और मिलकर वे \omterm{def:b2:sound-waves:approximation}{अधिदाब} के लिए \omterm{thm:b2:waves-on-strings:equation}{तरंग-समीकरण} देते हैं,
\[
\Delta p = \frac1{c^2}\frac{\partial^2p}{\partial t^2} , \qquad
c = \frac1{\sqrt{\rho_0\chi_S}} ,
\]
($\rho_1$ और $\vect v$ के लिए भी वही)। आदर्श गैस के लिए $\chi_S =
1/\gamma P_0$ और
\[
c = \sqrt{\frac{\gamma P_0}{\rho_0}} = \sqrt{\frac{\gamma RT}{M}} :
\]
\qty{20}{\degreeCelsius} पर हवा में \qty{343}{m/s}, और प्रति केल्विन
\qty{0.6}{m/s} बढ़ता हुआ; हीलियम में \qty{1000}{m/s}; पानी में \qty{1480}{m/s} ($\chi_S =
\qty{4.5e-10}{Pa^{-1}}$)।
\end{theorem}

\begin{proof}
ऑयलर: संवहनी पद $\rho(\vect v\cdot\operatorname{\vect{grad}})\vect v$
दूसरी कोटि का है, $\rho\,\partial_t\vect v \approx \rho_0\,\partial_t\vect v$, और
$\operatorname{\vect{grad}}P = \operatorname{\vect{grad}}p$। द्रव्यमान: $\partial_t\rho +
\operatorname{div}(\rho\vect v) \approx \partial_t\rho_1 + \rho_0\operatorname{div}\vect v$।
समएन्ट्रॉपी नियम: पहली कोटि तक $\dd\rho = \rho\chi_S\,\dd P$। पहले समीकरण का
अपसरण और दूसरे का समय-अवकलज लीजिए:
$\rho_0\partial_t\operatorname{div}\vect v = -\Delta p = -\partial_t^2\rho_1 = -\rho_0\chi_S
\partial_t^2p$। आदर्श गैस के लिए, समएन्ट्रॉपी रूपांतरण में $PV^\gamma$
अचर होने से $\dd\rho/\rho = \dd P/\gamma P$ मिलता है, अतः $\chi_S = 1/\gamma P$,
और $P_0/\rho_0 = RT/M$।
\end{proof}

\begin{remark}[समएन्ट्रॉपी क्यों]\label{rem:b2:sound-waves:laplace}
न्यूटन ने \omterm{thm:b2:sound-waves:equation}{ध्वनि की चाल} समतापी संपीड्यता से निकाली थी,
$\sqrt{P_0/\rho_0} = \qty{290}{m/s}$, जो $15\%$ कम है; लाप्लास ने देखा कि
संपीडन इतने तेज़ हैं कि गरम शिखरों और ठंडी घाटियों के बीच ऊष्मा बह ही
नहीं सकती --- एक आवर्तकाल में ऊष्मा कुछ माइक्रोमीटर विसरित होती है
(\cref{ch:b2:heat-conduction}), जो तरंगदैर्घ्य के सामने कुछ नहीं ---
अतः रूपांतरण रुद्धोष्म और उत्क्रमणीय है, और $\gamma$ आ जाता है।
इस तरह \omterm{thm:b2:sound-waves:equation}{ध्वनि की चाल} $\gamma$ का सीधा मापन है: हवा के लिए $1.40$,
आर्गन के लिए $1.67$।
\end{remark}

\section{समतल तरंगें, प्रतिबाधा, तीव्रता}

\begin{proposition}[समतल प्रगामी ध्वनि-तरंग]\label{prop:b2:sound-waves:plane}
\index{ध्वानिक प्रतिबाधा}\index{ध्वानिक तीव्रता}\index{ऊर्जा घनत्व (ध्वानिक)}
किसी समतल तरंग $p = f(x - ct)$ के लिए, जो $+x$ की ओर चलती है, तरल का
वेग $x$ के अनुदिश और \omterm{def:b2:sound-waves:approximation}{अधिदाब} के साथ कला में होता है, तथा
\[
p = Z\,v , \qquad Z = \rho_0c = \sqrt{\rho_0/\chi_S} ,
\]
जहाँ $Z$ माध्यम की \emph{\omterm{prop:b2:sound-waves:plane}{ध्वानिक प्रतिबाधा}} है
(\unit{kg.m^{-2}.s^{-1}}): हवा के लिए \qty{410}{kg.m^{-2}.s^{-1}}, पानी के लिए \qty{1.5e6}{kg.m^{-2}.s^{-1}},
कोमल ऊतक के लिए $\sim\qty{1.6e6}{kg.m^{-2}.s^{-1}}$,
और इस्पात के लिए \qty{4e7}{kg.m^{-2}.s^{-1}}। तरंग यह ऊर्जा घनत्व और
यह तीव्रता ($+x$ की ओर प्रति एकांक क्षेत्रफल शक्ति) ले जाती है:
\[
e = \tfrac12\rho_0v^2 + \tfrac12\chi_Sp^2 , \qquad I = p\,v ,
\]
जो $\partial_te + \partial_xI = 0$ का पालन करते हैं; और प्रगामी तरंग के लिए $e$ के
दोनों आधे बराबर होते हैं तथा $I = p^2/Z = Zv^2$। $p_m$ दाब-आयाम की किसी
ज्यावक्रीय तरंग की माध्य तीव्रता $\langle I\rangle = p_m^2/2Z$ होती है।
\end{proposition}

\begin{proof}
$p = f(x - ct)$ के लिए, $\rho_0\partial_tv = -\partial_xp = f'/c\cdot c\dots$: ठीक-ठीक,
$\rho_0\partial_tv = -f'(x - ct)$, और $v = f(x - ct)/\rho_0c$ इसे संतुष्ट करता है।
$e$ का स्थितिज पद तरल को संपीडित करने में संचित कार्य है
(एकांक आयतन के किसी समएन्ट्रॉपी संपीडन में $\dd V/V = -\chi_S\,\dd p$
से $\int p\,\chi_S\dd p = \tfrac12\chi_Sp^2$ संचित होता है); और $I$ प्रति एकांक क्षेत्रफल वह दर है जिससे
दाब-बल $pS$ आगे के तरल पर, जो $v$ से चलता है, कार्य करता है।
संतुलन: $\partial_te = \rho_0v\partial_tv + \chi_Sp\partial_tp = -v\partial_xp - p\partial_xv =
-\partial_x(pv)$, दोनों रैखिकीकृत समीकरणों से।
\end{proof}

\begin{definition}[ध्वनि-स्तर]\label{def:b2:sound-waves:decibel}
\index{डेसिबल}\index{ध्वनि-स्तर}\index{श्रवण देहली}
$I$ माध्य तीव्रता की किसी तरंग का \emph{ध्वनि-स्तर}
\[
L = 10\log_{10}\frac I{I_0} \ \text{dB} , \qquad I_0 = \qty{1e-12}{W/m^2}
\]
है --- यहाँ \qty{1}{kHz} पर श्रवण की देहली, जो हवा में
दाब-आयाम $p_0 \approx \qty{2e-5}{Pa}$ (वर्ग-माध्य-मूल) $= 2 \times 10^{-10}$
वायुमंडल से मेल खाती है। तीव्रता दुगुनी करने पर \qty{3}{dB} जुड़ते हैं; दस गुना पर \qty{10}{dB};
और दाब-आयाम सौ गुना करने पर \qty{40}{dB}। शांत कमरा
\qty{30}{dB}, बातचीत \qty{60}{dB}, व्यस्त सड़क \qty{80}{dB}, रॉक-संगीत
\qty{110}{dB}, पीड़ा \qty{120}{dB} ($I = \qty{1}{W/m^2}$, $p_m = \qty{29}{Pa}$)।
\end{definition}

\begin{example}[ध्वनि कितनी छोटी होती है]\label{ex:b2:sound-waves:small}
देहली पर, \qty{1}{kHz}: $p_m = \qty{2.8e-5}{Pa}$, $v_m = p_m/Z =
\qty{7e-8}{m/s}$, और विस्थापन-आयाम $\xi_m = v_m/\omega =
\qty{1e-11}{m}$ --- अर्थात् किसी परमाणु के व्यास का दसवाँ भाग: कर्ण-पटल
परमाणु से छोटी गतियाँ भाँप लेता है। \qty{120}{dB} पर $\xi_m = \qty{11}{\micro m}$,
जो तब भी अदृश्य है, और $p_m/P_0 = 3 \times 10^{-4}$: अर्थात् सबसे तेज़ ध्वनियाँ भी
नन्हे विक्षोभ हैं, और इसीलिए रैखिक सिद्धांत इतना अच्छा चलता है।
\end{example}

\section{गोलीय तरंगें}

\begin{proposition}[गोलीय तरंगें और व्युत्क्रम-वर्ग नियम]\label{prop:b2:sound-waves:spherical}
\index{गोलीय तरंग}\index{व्युत्क्रम-वर्ग नियम (ध्वनि)}
हर दिशा में बराबर विकिरण करता कोई छोटा स्रोत यह \omterm{prop:b2:sound-waves:spherical}{गोलीय तरंग}
बनाता है:
\[
p(r, t) = \frac{A}{r}\,f\Bigl(t - \frac rc\Bigr) ,
\]
जिसका आयाम $1/r$ की तरह और तीव्रता $1/r^2$ की तरह गिरती है:
$\mathcal P$ ध्वानिक शक्ति के स्रोत के लिए $I = \mathcal P/4\pi r^2$, अर्थात्
दूरी दुगुनी होने पर हर बार $-\qty{6}{dB}$। स्रोत से दूर
तरंग स्थानिक रूप से समतल होती है, और $v = p/Z$ त्रिज्यीय।
\end{proposition}

\begin{proof}
गोलीय निर्देशांकों में, केवल $r$ के फलन के लिए, $\Delta p =
\frac1r\frac{\partial^2(rp)}{\partial r^2}$ (स्वीकृत; \cref{ch:b2:maxwell-equations}),
अतः $u = rp$ एक-विमीय \omterm{thm:b2:waves-on-strings:equation}{तरंग-समीकरण} $\partial_r^2u = \partial_t^2u/c^2$ का पालन करता है,
जिससे बाहर जाती तरंग के लिए $u = Af(t - r/c)$ मिलता है। $r$ त्रिज्या के
गोले को पार करती शक्ति $4\pi r^2I$ है, जो कुछ अवशोषित न हो तो अचर रहती है।
\end{proof}

\begin{example}[एक सायरन, एक बातचीत]\label{ex:b2:sound-waves:siren}
\qty{10}{W} का सायरन: $I = 10/4\pi r^2$, अर्थात् \qty{1}{m} पर \qty{0.8}{W/m^2} (\qty{119}{dB},
पीड़ा), \qty{100}{m} पर \qty{79}{dB}, \qty{1}{km} पर \qty{59}{dB} --- जो शहर भर में सुनाई देता है।
कोई आवाज़ लगभग \qty{10}{\micro W} विकिरित करती है: \qty{1}{m} पर \qty{60}{dB},
और \qty{10}{m} पर \qty{40}{dB}। हवा ध्वनि सोखती भी है (ऊँची आवृत्ति पर
अधिक, \qty{10}{kHz} पर प्रति सौ मीटर कुछ \omterm{def:b2:sound-waves:decibel}{डेसिबल}), और इसीलिए
दूर की गरज नीची गड़गड़ाहट में सुनाई देती है।
\end{example}

\section{डॉप्लर प्रभाव}

\begin{theorem}[डॉप्लर प्रभाव]\label{thm:b2:sound-waves:doppler}
\index{डॉप्लर प्रभाव}
कोई स्रोत आवृत्ति $f$ पर उत्सर्जन करता है और उस रेखा के अनुदिश $v_s$ चाल
से चलता है जो उसे किसी प्रेक्षक से जोड़ती है, और प्रेक्षक $v_o$ से चलता है;
दोनों चालें तब धनात्मक गिनी जाती हैं जब वे पास आ रहे हों; और ध्वनि विराम
में पड़ी हवा में $c$ से चलती है। प्रेक्षक यह आवृत्ति पाता है:
\[
f' = f\,\frac{c + v_o}{c - v_s} .
\]
चलता स्रोत अपने आगे के तरंगाग्र सिकोड़ देता है ($\lambda' = (c - v_s)/f$)
और पीछे वालों को फैला देता है; और चलता प्रेक्षक तरंगाग्रों से
$(c + v_o)/\lambda$ दर से मिलता है। $v_s, v_o \ll c$ के लिए दोनों $\Delta f/f \approx
(v_s + v_o)/c$ देते हैं।
\end{theorem}

\begin{proof}
चलता स्रोत: $t$ और $t + 1/f$ पर उत्सर्जित तरंगाग्र गति की दिशा में
$c/f - v_s/f$ से अलग होते हैं, अतः आगे की तरंगदैर्घ्य
$\lambda' = (c - v_s)/f$ है और किसी स्थिर प्रेक्षक को मिलती आवृत्ति
$c/\lambda'$। चलता प्रेक्षक: तरंगदैर्घ्य $\lambda = c/f$ है पर अग्र
सापेक्ष चाल $c + v_o$ से गुज़रते हैं। अब दोनों जोड़िए।
\end{proof}

\begin{example}[सायरन, रडार बंदूक, लाल कोशिकाएँ]\label{ex:b2:sound-waves:dopplerex}
\qty{25}{m/s} से गुज़रती किसी ऐंबुलेंस का \qty{700}{Hz} सायरन: पास आते हुए \qty{755}{Hz},
दूर जाते हुए \qty{651}{Hz} --- अर्थात् अष्टक के छठे भाग की गिरावट,
वही जानी-पहचानी ``नी-नॉ'' ढलान। कोई पुलिस-रडार बंदूक किसी कार से परावर्तित
\qty{24}{GHz} की रेडियो-तरंग पर वही प्रभाव नापती है (तरंग दो बार
विस्थापित होती है, कार के पाने और फिर उत्सर्जित करने से): \qty{30}{m/s} पर $\Delta f = 2fv/c =
\qty{4.8}{kHz}$। किसी धमनी पर ताकती \qty{5}{MHz} की पराश्रव्य जाँच
रक्त-कोशिकाओं को \qty{60}{\degree} पर \qty{0.5}{m/s} के लिए $\Delta f = 2fv\cos\theta/c \approx \qty{1.6}{kHz}$
पर सुनती है: अर्थात् एक श्रव्य सीटी, जिसका स्वर ही
रक्त की चाल है। और दूर जाती किसी आकाशगंगा की वर्णक्रम-रेखाएँ
$\Delta\lambda/\lambda = v/c$ से लाल की ओर खिसकती हैं --- यही सूत्र कम चाल पर प्रकाश के लिए भी चलता है,
और उससे आगे आपेक्षिकता का सुधार लगता है।
\end{example}

\begin{remark}[पराध्वनिक]\label{rem:b2:sound-waves:mach}
\index{मैक शंकु}\index{ध्वनि-विस्फोट}
$v_s > c$ पर सूत्र टूट जाता है (आगे $f' < 0$): अर्थात् कोई ध्वनि स्रोत से
पहले नहीं पहुँचती। पथ के अनुदिश उत्सर्जित तरंगाग्रों का शंक्वाकार
आवरण होता है, जिसका अर्ध-कोण $\theta$ है और $\sin\theta = c/v_s$ (\emph{\omterm{rem:b2:sound-waves:mach}{मैक शंकु}});
और शंकु द्वारा ढोई गई दाब-उछाल ही \omterm{rem:b2:sound-waves:mach}{ध्वनि-विस्फोट} है, जो ज़मीन पर खड़े
श्रोता तक वायुयान के गुज़र जाने के बाद ही पहुँचती है --- मैक $2$ पर और
\qty{10}{km} ऊँचाई पर $\theta = \qty{30}{\degree}$ होता है और विस्फोट सिर के
ऊपर से गुज़रने के $10\,\text{km}/\tan\theta/v_s \approx \qty{25}{s}$ बाद
आता है। गोली की तड़ाक और चाबुक की नोक वही शंकु हैं।
\end{remark}

\begin{method}[ध्वानिकी में परिमाण की कोटियाँ]\label{met:b2:sound-waves:method}
कोई \omterm{def:b2:sound-waves:decibel}{ध्वनि-स्तर} $L$ दिया हो तो: $I = I_0\,10^{L/10}$; फिर $p_m = \sqrt{2ZI}$,
$v_m = p_m/Z$, $\xi_m = v_m/\omega$; और किसी स्रोत की शक्ति $4\pi r^2I$ है, यदि वह
हर दिशा में विकिरण करता हो (किसी कड़े फ़र्श पर आधी, जो
अर्ध-आकाश में परावर्तित कर देता है)। $r_1$ से $r_2$ जाने पर $L$
$20\log_{10}(r_1/r_2)$ से बदलता है। $n$ असंबद्ध बराबर स्रोत जोड़ने पर
$10\log_{10}n$ जुड़ता है; और किसी बिंदु पर कला में दो संबद्ध स्रोत \qty{6}{dB} जोड़ते हैं।
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:4}
चमक के \qty{4.5}{s} बाद गरज सुनाई देती है: दूरी। किसी जहाज़ का भोंपू

\qty{5}{MHz} पराश्रव्य प्रतिध्वनि ऊतक (\qty{1540}{m/s}) में \qty{130}{\micro s} बाद
लौटती है: गहराई।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:5}
\qty{500}{Hz} का कोई रेल-भोंपू; $c = \qty{340}{m/s}$। (क) प्लैटफ़ॉर्म पर खड़े

की ओर चल रही है और भोंपू विराम में है; फिर दोनों \qty{40}{m/s} से एक-दूसरे
की ओर: इनकी तुलना \qty{80}{m/s} की सापेक्ष चाल और एक पिंड के विराम में होने से
कीजिए। (ग) भोंपू वाली गाड़ी के चालक को आगे की किसी दीवार से लौटती
प्रतिध्वनि की कौन-सी आवृत्ति सुनाई देती है।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:6}

वही दाब, पर बहुत भिन्न तीव्रताएँ:
टिप्पणी दीजिए। (पानी के भीतर के स्तर \qty{1}{\micro Pa} के सापेक्ष दिए जाते हैं: उस मापनी पर
\qty{1}{Pa} कितना है?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:7}

स्वातंत्र्य-कोटियों की संख्या से उसका अर्थ बताइए। (ग) \qty{1}{kHz} पर एक आवर्तकाल में
ऊष्मा कितनी दूर विसरित होती है, यह आँकिए (हवा की ऊष्मीय विसरणशीलता

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:8}
\emph{ऑर्गन-नलियाँ।} किसी नली में \omterm{def:b2:sound-waves:approximation}{अधिदाब} खुले सिरे पर शून्य होता है
और वेग बंद सिरे पर। (क) दोनों सिरों पर खुली नली में \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंगें}:
दिखाइए कि विधाएँ $f_n = nc/2L$ हैं; और एक सिरे पर बंद नली में
$f_n = (2n - 1)c/4L$। (ख) हर दशा में \qty{440}{Hz} की मूल विधा के लिए
लंबाइयाँ। (ग) हर नली में कौन-से संनादी होते हैं? (घ) वास्तविक
खुला सिरा नली से लगभग $0.6$ त्रिज्या आगे बैठता है (सिरा-संशोधन): किसी
\qty{4}{cm} की नली के लिए $L$ पर और $f$ पर संशोधन।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:9}
\qty{2.0}{kW} का कोई वैद्युत सायरन अपनी शक्ति का $20\%$ ध्वनि में बदल देता है,

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:10}
\emph{अप्रगामी ध्वनि-तरंग की ऊर्जा।} $L$ लंबाई की किसी बंद नलिका में
\omterm{def:b2:sound-waves:approximation}{अधिदाब} $p = p_m\cos(kx)\cos(\omega t)$ है, $k = n\pi/L$। (क) रैखिकीकृत \omterm{thm:b2:euler-bernoulli:euler}{ऑयलर समीकरण} से $v(x,

ऊर्जा-घनत्व; दिखाइए कि वे समय में समपाद-अंतर पर दोलन करते हैं और
आकाश में एक चौथाई तरंगदैर्घ्य से खिसके होते हैं। (ग) नलिका में कुल ऊर्जा

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:11}
\emph{\omterm{rem:b2:sound-waves:mach}{ध्वनि-विस्फोट}।} कोई वायुयान क्षैतिज दिशा में $v = 1.6\,c$ से उड़ता है,
ऊँचाई $h = \qty{12}{km}$ पर, जहाँ $c = \qty{300}{m/s}$ है। (क) गोलीय तरंगाग्रों के आवरण से
मैक कोण निकालिए। (ख) सिर के ऊपर से गुज़रने और ज़मीन पर विस्फोट के
बीच का समय (ऊँचाई के साथ $c$ का बदलना छोड़ दीजिए)। (ग) वायुयान के
गुज़रते समय ज़मीन पर उस पट्टी की चौड़ाई जो विस्फोट सुनती है, यदि शंकु का
निशान पथ के $\pm\qty{30}{km}$ के भीतर सुनाई देता हो: थल के ऊपर पराध्वनिक
उड़ान पर रोक क्यों है? (घ) \qty{800}{m/s} की कोई राइफ़ल-गोली: मैक कोण; और
प्रक्षेप-पथ से \qty{10}{m} दूर खड़े प्रेक्षक को क्या, और किस क्रम में, सुनाई देता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:12}
\emph{डॉप्लर पराश्रव्य।} कोई जाँच $f = \qty{5.0}{MHz}$ पर ऊतक ($c = \qty{1540}{m/s}$)
में उत्सर्जन करती है; और लाल कोशिकाएँ किसी वाहिका के अनुदिश $v$ से चलती हैं, जो
किरण-पुंज से $\theta$ कोण बनाती है। (क) किसी कोशिका को मिलती आवृत्ति (चलता
प्रेक्षक)। (ख) वह इसी आवृत्ति को चलते स्रोत की तरह फिर उत्सर्जित (प्रकीर्ण) करती है:
जाँच पर लौटती आवृत्ति; दिखाइए कि $\Delta f = 2fv\cos\theta/c$, जब $v \ll c$। (ग)
$\Delta f$, $v = \qty{0.50}{m/s}$, $\theta = \qty{60}{\degree}$ के लिए; और $\theta = \qty{90}{\degree}$ के लिए
--- तब चलाने वाला क्या करता है? (घ) जाँच \qty{50}{Hz} तक विभेद कर सकती है:
वेग का विभेदन। डॉप्लर संकेत केवल दिखाने के बदले सुनाया क्यों जाता है?
\end{exercise}

\section{समस्या: गरज से प्रतिध्वनि-लेखी तक}

\begin{problem}[{सप्ताहांत समस्या --- खुली हवा में, किसी संगीत-भवन में,
शरीर के भीतर और गुज़रते सायरन से ध्वनि}]
\label{pb:b2:sound-waves:1}

हवा: $\gamma = 1.40$, $M = \qty{29}{g/mol}$, और \qty{15}{\degreeCelsius} पर
$Z = \qty{410}{kg.m^{-2}.s^{-1}}$; $R = \qty{8.31}{J/(mol.K)}$।

\medskip
\noindent\textbf{भाग I --- तूफ़ान।}
\begin{enumerate}
  \item \qty{15}{\degreeCelsius} और \qty{30}{\degreeCelsius} पर \omterm{thm:b2:sound-waves:equation}{ध्वनि की चाल};
        ``प्रति किलोमीटर तीन सेकंड'' के नियम को उचित ठहराइए।
  \item चमक के \qty{4.0}{s} बाद गरज आती है: गिरने की जगह की
        दूरी। चमक क्षणिक होने पर भी गरज कई सेकंड चलती है:
        क्यों?
  \item गिरने की जगह के पास स्तर \qty{120}{dB} तक पहुँचता है: हवा का दाब-आयाम
        और वेग-आयाम।
  \item \qty{120}{dB} पर \qty{2}{s} की किसी गरज में दीवार के प्रति वर्ग मीटर पर
        मिली ऊर्जा।
  \item किसी साफ़ रात में ज़मीन ठंडी होती है और नीचे की हवा ऊपर की
        हवा से ठंडी। ध्वनि-किरणें परतों के बीच स्नेल के नियम का पालन करती हैं,
        $\sin i/c = \text{अचर}$: ऊपर की ओर छोड़ी गई किरणें नीचे मुड़ती हैं या ऊपर?
        रात में और पानी के ऊपर दूर की ध्वनियाँ इतनी अच्छी क्यों सुनाई देती हैं?
  \item दिन में उल्टा: \qty{3}{km} दूर का श्रोता कुछ नहीं सुनता।
        यदि $c$ प्रति \qty{500}{m} ऊँचाई पर \qty{1}{\%} गिरे तो \qty{3}{km} के बाद
        किसी क्षैतिज किरण के मुड़ने का कोण आँकिए
        (मुड़ना एकसमान मानिए: किरण एक वृत्त है)।

  \item हवा ऊँची आवृत्तियाँ नीची से कहीं अधिक सोखती है: दूर की गरज
        गड़गड़ाती और पास की चटकती क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- संगीत-समारोह।}
कोई ध्वनिविस्तारक \qty{100}{W} वैद्युत शक्ति पाता है और उसका
$2\%$ ध्वनि में बदल देता है, जो उसके सामने के अर्ध-आकाश में बराबर
विकिरित होती है।
\begin{enumerate}[resume]
  \item ध्वानिक शक्ति; \qty{10}{m} पर तीव्रता और स्तर।
  \item वहाँ दाब-आयाम; और \qty{100}{Hz} पर वेग- तथा विस्थापन-आयाम।
  \item \qty{1}{m} पर स्तर (क्या वह सुरक्षित है?) और वह दूरी जिस पर संगीत
        गिरकर \qty{60}{dB} हो जाता है।
  \item मंच के चारों ओर फैले दस एक-से ध्वनिविस्तारक, जो कला में नहीं
        हैं: \qty{10}{m} पर स्तर। और कला में चलाए गए दो ध्वनिविस्तारक, जो
        अपनी अक्ष पर बैठे श्रोता से बराबर दूरी पर हैं: वहाँ स्तर।
  \item भवन (आयतन \qty{10000}{m^3}) में \qty{2000}{m^2} की दीवारें और श्रोता हैं, जो उन पर पड़ती ध्वनि का $30\%$ सोख
        लेते हैं: साम्य में भवन में संचित ध्वानिक ऊर्जा और
        अनुरणन काल (संगीत रुकने के बाद स्तर के \qty{60}{dB} गिरने का
        समय) आँकिए, और इसके लिए उत्सर्जित शक्ति तथा अवशोषित शक्ति के
        बीच संतुलन लीजिए (विसरित क्षेत्र के लिए $I_{\text{दीवार}} \approx ce/4$, स्वीकृत)।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- प्रतिध्वनि-लेखी।}
कोमल ऊतक: $c = \qty{1540}{m/s}$, $Z = \qty{1.6e6}{kg.m^{-2}.s^{-1}}$; हड्डी
$Z = \qty{7e6}{kg.m^{-2}.s^{-1}}$; हवा $\qty{400}{kg.m^{-2}.s^{-1}}$। अभिलंब
आपतन पर किसी अंतरापृष्ठ पर परावर्तित दाब-आयाम आपतित आयाम का $r = (Z_2 -
Z_1)/(Z_2 + Z_1)$ गुना होता है (\cref{ch:b2:wave-interfaces} में
व्युत्पन्न); और ऊतक ध्वनि को प्रति सेंटीमीटर तथा प्रति मेगाहर्ट्ज़ लगभग
\qty{0.5}{dB} क्षीण करता है।
\begin{enumerate}[resume]
  \item \qty{3}{MHz}, \qty{5}{MHz}, \qty{10}{MHz} पर ऊतक में तरंगदैर्घ्य; और हर एक
        जितना छोटा विवरण खोल सकता है वह लगभग एक तरंगदैर्घ्य है।
  \item \qty{12}{cm} गहरे किसी अंग से प्रतिध्वनि का विलंब; प्रतिध्वनियाँ
        गड्डमड्ड किए बिना स्पंद भेजने की अधिकतम दर; और
        $200$ रेखाओं का चित्र बनाने में लगा समय।
  \item ऊतक--वायु अंतरापृष्ठ पर परावर्तित दाब का, और ऊर्जा का, अंश;
        ऊतक--हड्डी अंतरापृष्ठ पर; और वसा
        ($Z = \qty{1.38e6}{kg.m^{-2}.s^{-1}}$) तथा पेशी
        ($\qty{1.70e6}{kg.m^{-2}.s^{-1}}$) के बीच। जेल क्यों, और हड्डियाँ
        छाया क्यों डालती हैं?
  \item \qty{5}{MHz} पर किरण-पुंज पेशी से मिलने से पहले \qty{2}{cm} वसा पार
        करता है: उस अंतरापृष्ठ से जाँच तक लौटती उत्सर्जित ऊर्जा का अंश
        (परावर्तन और आने-जाने का
        क्षीणन), \omterm{def:b2:sound-waves:decibel}{डेसिबल} में।
  \item प्रश्न 12 वाले अंग के लिए तीनों आवृत्तियों पर आने-जाने का
        क्षीणन \omterm{def:b2:sound-waves:decibel}{डेसिबल} में; यदि स्कैनर \qty{80}{dB} तक की हानि
        सँभाल सके, तो कौन-सी आवृत्तियाँ चलेंगी? विभेदन बनाम गहराई का
        सौदा बताइए।
  \item जाँच \qty{5}{MHz} पर \qty{1}{\micro s} के स्पंद भेजती है: एक स्पंद कितनी
        तरंगदैर्घ्य लंबा है, और उससे गहराई का विभेदन कैसे
        सीमित होता है?
  \item \qty{5}{MHz} पर \qty{0.40}{m/s} की किसी धमनी पर डॉप्लर-विधा, किरण-पुंज
        \qty{60}{\degree} पर: विस्थापन; और चलाने वाले को क्या सुनाई देता है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- सायरन।}
\begin{enumerate}[resume]
  \item कोई ऐंबुलेंस \qty{700}{Hz} के सायरन के साथ \qty{90}{km/h} से गुज़रती है:
        पास आते और दूर जाते हुए सुनाई देती आवृत्तियाँ; और उनके बीच का
        अंतराल अर्ध-स्वरों में (अर्ध-स्वर $2^{1/12}$ का अनुपात है)।
  \item श्रोता किसी कार में है जो पास आती ऐंबुलेंस की ओर \qty{15}{m/s} से
        जा रही है: सुनाई देती आवृत्ति।
  \item फुटपाथ पर खड़े श्रोता को ठीक \qty{700}{Hz} किस क्षण सुनाई देते हैं?
  \item ऐंबुलेंस के आगे उसके तरंगाग्र: तरंगदैर्घ्य; और ऐंबुलेंस के सापेक्ष
        उसके आगे तथा पीछे \omterm{thm:b2:sound-waves:equation}{ध्वनि की चाल}।
  \item \qty{24}{GHz} की कोई रडार बंदूक किसी कार से \qty{3.2}{kHz} का विस्थापन पढ़ती है:
        कार की चाल (तरंग दो बार विस्थापित होती है)।
  \item सार दीजिए: इस समस्या की छह चालें (हवा में ध्वनि,
        ऊतक में, ऐंबुलेंस, कार, लाल कोशिकाएँ, प्रकाश) और वह
        एक सूत्र जो इन सबको सँभाल लेता है।
\end{enumerate}
\end{problem}
