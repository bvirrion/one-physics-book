\section{आवेश और धारा के घनत्व}

\begin{definition}[आवेश घनत्व; धारा घनत्व]\label{def:b2:charges-currents-conduction:densities}
\index{आवेश घनत्व}\index{धारा घनत्व}\index{धारा (j का अभिवाह)}
\omterm{def:b2:fluid-kinematics:particle}{मध्यदर्शी पैमाने} पर पदार्थ कोई \emph{आयतन आवेश घनत्व}
$\rho(M, t)$ (\unit{C/m^3}) ढोता है: अर्थात् आवेश $\rho\,\dd\tau$, जो $\dd\tau$ में है। कोई पृष्ठ
कोई \emph{पृष्ठ घनत्व} $\sigma$ (\unit{C/m^2}) ढो सकता है, और कोई तार कोई रैखिक
घनत्व $\lambda$ (\unit{C/m})। यदि जाति $i$ के वाहक (आवेश
$q_i$, संख्या-घनत्व $n_i$) माध्य वेग $\vect v_i$ से चलें, तो
\emph{धारा घनत्व}
\[
\vect j = \sum_in_iq_i\,\vect v_i \qquad (\unit{A/m^2}) ,
\]
होता है, और किसी दिष्ट पृष्ठ $S$ से होकर \emph{धारा} अभिवाह
$I = \iint_S\vect j\cdot\vect n\,\dd S$ है: अर्थात् प्रति एकांक समय $S$ को पार करता
आवेश। किसी एक ही जाति के लिए $\vect j = \rho_m\vect v$, जहाँ $\rho_m = nq$
गतिशील आवेश का घनत्व है।
\end{definition}

\begin{proof}
$\dd t$ में जाति $i$ के जो वाहक $\dd S$ पार करते हैं वे वही हैं जो
$\dd S$ आधार और $\vect v_i\dd t$ जनक वाले तिरछे बेलन में हैं: संख्या
$n_i\vect v_i\cdot\vect n\,\dd S\,\dd t$, और आवेश उसका $q_i$ गुना।
\end{proof}

\begin{example}[इलेक्ट्रॉन कितने धीरे बहते हैं]\label{ex:b2:charges-currents-conduction:drift}
\index{अपवाह वेग}
ताँबे में प्रति परमाणु एक चालन-इलेक्ट्रॉन होता है: $n = \rho_{\text{Cu}}N_A/M =
8900 \times 6.02 \times 10^{23}/0.0635 = \qty{8.5e28}{m^{-3}}$। \qty{1.5}{mm^2} के तार में
\qty{10}{A} की धारा $j = \qty{6.7e6}{A/m^2}$ है, और
\emph{\omterm{ex:b2:charges-currents-conduction:drift}{अपवाह वेग}} $v = j/ne = 6.7 \times 10^6/(8.5 \times 10^{28} \times 1.6 \times
10^{-19}) = \qty{0.5}{mm/s}$ --- अर्थात् प्रति मीटर दो घंटे। लैंप तुरंत इसलिए जलता
है कि जो \emph{क्षेत्र} इलेक्ट्रॉनों को धकेलता है वह पूरे तार पर
नैनोसेकंडों में खड़ा हो जाता है (वह लगभग प्रकाश की चाल से चलता है, जैसा
\cref{ch:b2:dispersion-wave-packets} की केबल ने दिखाया):
अतः सारे इलेक्ट्रॉन एक साथ चल पड़ते हैं, जैसे भरी नली में पानी।
\end{example}

\section{आवेश का संरक्षण}

\begin{theorem}[आवेश का स्थानिक संरक्षण]\label{thm:b2:charges-currents-conduction:conservation}
\index{आवेश-संरक्षण (स्थानिक)}\index{सांतत्य समीकरण (आवेश)}
हर बिंदु पर,
\[
\frac{\partial\rho}{\partial t} + \operatorname{div}\vect j = 0 ;
\]
होता है; और किसी स्थिर आयतन $V$ पर, जो बंद पृष्ठ $\Sigma$ से घिरा हो, समाकलित करने पर
$\dd Q_V/\dd t = -I_{\text{निर्गम}}$: अर्थात् भीतर का आवेश केवल उतना बदलता है जितना
सीमा को पार करता है। किसी \emph{स्थायी दशा} ($\partial_t\rho = 0$) में
$\operatorname{div}\vect j = 0$: अतः $\vect j$ का अभिवाह किसी धारा-नलिका के अनुदिश
संरक्षित रहता है, तार की हर काट से वही धारा गुज़रती है, और
किसी संधि में घुसती धाराएँ निकलती धाराओं के बराबर होती हैं (किरखोफ़ का संधि नियम)।
\end{theorem}

\begin{proof}
शब्दशः \cref{thm:b2:fluid-kinematics:continuity} वाला द्रव्यमान-संतुलन,
जिसमें $\rho$ \omterm{def:b2:charges-currents-conduction:densities}{आवेश घनत्व} है और $\vect j$ उसकी धारा: किसी स्थिर संदूक के छह
फलकों से बाहर जाता आवेश का कुल निर्गम, प्रति एकांक आयतन,
$\operatorname{div}\vect j$ है, और कोई आवेश बनता नहीं। समाकल रूप
संदूकों का योग है (भीतरी फलक कट जाते हैं)। दो काटों
$S_1$, $S_2$ के बीच की किसी नलिका के लिए, जिसके पार्श्व से कोई अभिवाह न हो, $I_1 = I_2$; और कोई संधि
कई तारों वाला कोई छोटा आयतन है: $\sum I_{\text{आगम}} = \sum I_{\text{निर्गम}}$।
\end{proof}

\begin{remark}[धारा कहाँ रुकती है: संधारित्र]\label{rem:b2:charges-currents-conduction:capacitor}
संधारित्र की किसी प्लेट को खिलाता तार धारा की ऐसी नलिका है जो समाप्त हो
जाती है: प्लेट में $\vect j$ का अभिवाह $I = \dd Q/\dd t$ है, और आवेश जमा होता जाता है
--- अतः वहाँ $\operatorname{div}\vect j \ne 0$, और दशा स्थायी नहीं।
फिर भी वही धारा $I$ दूसरी ओर के तार में बहती है: अतः कुछ तो होगा
जो शेष को अंतराल के पार ले जाता है। \cref{ch:b2:maxwell-equations}
उसका नाम लेता है --- विस्थापन धारा --- और आवेश-संरक्षण को
मैक्सवेल के समीकरणों की आधारशिला बना देता है।
\end{remark}

\section{ड्रूडे प्रतिरूप और ओम का नियम}

\begin{proposition}[ड्रूडे प्रतिरूप; स्थानिक ओम नियम]\label{prop:b2:charges-currents-conduction:drude}
\index{ड्रूडे प्रतिरूप}\index{ओम का नियम (स्थानिक)}\index{चालकता}\index{प्रतिरोधकता}\index{गतिशीलता}
किसी धातु के चालन-इलेक्ट्रॉनों को मुक्त कण मानिए (द्रव्यमान $m$,
आवेश $-e$, घनत्व $n$), जो क्षेत्र $\vect E$ अनुभव करते हैं और, औसतन,
हर $\tau$ सेकंड में जालक से संघट्टों द्वारा अपना अपवाह-संवेग खो देते हैं
--- अर्थात् माध्य वेग पर कोई \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल} $-m\vect v/\tau$:
\[
m\frac{\dd\vect v}{\dd t} = -e\vect E - \frac m\tau\vect v .
\]
किसी स्थायी क्षेत्र में अपवाह कुछ ही $\tau$ में $\vect v = -e\tau
\vect E/m$ पर ठहर जाता है, और \omterm{def:b2:charges-currents-conduction:densities}{धारा घनत्व}
\[
\vect j = \gamma\,\vect E , \qquad \gamma = \frac{ne^2\tau}{m} :
\]
हो जाता है: यही \emph{स्थानिक ओम नियम} है, जिसमें $\gamma$ \emph{चालकता} है
(\unit{S/m}; और उसका व्युत्क्रम $\rho_e = 1/\gamma$ \emph{प्रतिरोधकता},
\unit{\ohm.m})। क्षेत्र से अपवाह-चाल का अनुपात $\mu = e\tau/m$
\emph{गतिशीलता} है (\unit{m^2/(V.s)})। $L$ लंबाई और
एकसमान काट $S$ वाले किसी चालक के लिए, जो एकसमान $\vect j$ ढोता हो, $E = j/\gamma$ को
लंबाई के अनुदिश समाकलित करने पर $U = EL = (L/\gamma S)I$ मिलता है: अर्थात् वर्ष 1 के खंड का
प्रतिरोध $R = L/\gamma S
= \rho_eL/S$।
\end{proposition}

\begin{proof}
$\vect v = -e\tau\vect E/m$ लेकर, $\vect j = -ne\vect v = (ne^2\tau/m)\vect E$।
क्षणिक दौर $\vect v(t) = \vect v_\infty(1 - \eu^{-t/\tau})$ $\tau \sim
\qty{1e-14}{s}$ चलता है, जो हर परिपथीय काल-पैमाने पर तात्क्षणिक है। प्रतिरोध
सिरों के बीच $\int\vect E\cdot\dd\vect l$ से निकलता है।
\end{proof}

\begin{example}[ताँबा, नापी गई चालकता से]\label{ex:b2:charges-currents-conduction:copper}
ताँबा: $\gamma = \qty{6.0e7}{S/m}$, $n = \qty{8.5e28}{m^{-3}}$: अतः $\tau = m\gamma/ne^2 =
9.1 \times 10^{-31} \times 6 \times 10^7/(8.5 \times 10^{28} \times 2.56 \times 10^{-38}) = \qty{2.5e-14}{s}$
--- अर्थात् संघट्टों के बीच पच्चीस फ़ेम्टोसेकंड। धातु में इलेक्ट्रॉनों की यादृच्छिक
चाल लगभग \qty{1.6e6}{m/s} होती है (यह कोई क्वांटम प्रभाव है, न कि
ऊष्मीय $\sqrt{3k_BT/m} \approx \qty{1e5}{m/s}$), अतः माध्य मुक्त पथ $\ell
= v\tau \approx \qty{40}{nm}$ है, अर्थात् सौ परमाणु-अंतराल: इलेक्ट्रॉन स्वयं
परमाणुओं से नहीं टकराते (कोई पूर्ण क्रिस्टल उनके लिए पारदर्शी है)
बल्कि उसके दोषों और ऊष्मीय कंपनों से --- और इसीलिए
धातु को गरम करने या मिश्रित करने पर $\gamma$ गिर जाता है। गतिशीलता $e\tau/m =
\qty{4.4e-3}{m^2/(V.s)}$: अर्थात् \qty{4}{mm/s} प्रति वोल्ट प्रति मीटर।
\end{example}

\begin{proposition}[ऊँची आवृत्ति पर ओम का नियम]\label{prop:b2:charges-currents-conduction:acdrude}
\index{सम्मिश्र चालकता}
किसी ज्यावक्रीय क्षेत्र $\underline{\vect E}\,\eu^{\iu\omega t}$ के लिए ड्रूडे समीकरण
$\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$ देता है, जहाँ
\[
\underline\gamma(\omega) = \frac{\gamma_0}{1 + \iu\omega\tau} , \qquad \gamma_0 = \frac{ne^2\tau}{m} :
\]
अर्थात् चालकता तब तक वास्तविक और अपने दिष्ट मान के बराबर रहती है जब तक $\omega\tau
\ll 1$, अर्थात् ताँबे के लिए लगभग \qty{1e13}{Hz} तक --- यानी पूरे
रेडियो और सूक्ष्मतरंग परास में; और अवरक्त तथा दृश्य में ($\omega\tau \gg 1$)
वह काल्पनिक हो जाती है, $\underline\gamma \approx ne^2/\iu m\omega$: तब इलेक्ट्रॉन
क्षेत्र से कला-बाहर, स्वतंत्र रूप से दोलन करते हैं और धातु
\cref{ch:b2:waves-in-media} के प्लाज़्मा जैसी बरतती है।
\end{proposition}

\begin{proof}
$\iu\omega m\underline{\vect v} = -e\underline{\vect E} - m\underline{\vect v}/\tau$, जिससे
$\underline{\vect v} = -e\tau\underline{\vect E}/m(1 + \iu\omega\tau)$ और $\underline{\vect j}
= -ne\underline{\vect v}$।
\end{proof}

\begin{remark}[चालक, अर्धचालक, विद्युत्-अपघट्य]\label{rem:b2:charges-currents-conduction:materials}
\index{अर्धचालक}\index{विद्युत्-अपघट्य}
वही नियम $\vect j = \sum n_iq_i\mu_i\vect E$ हर चालक के लिए चलता है;
जो बदलता है वह $n$ और $\mu$ है। धातुएँ: $n \sim 10^{29}$, $\gamma \sim 10^7$
S/m। अर्धचालक: $n \sim 10^{16}$--$10^{23}$, जो अपमिश्रण से तय होता है और ताप के साथ
तेज़ी से चढ़ता है, तथा दो प्रकार के वाहक (इलेक्ट्रॉन और
कोटर) --- और यहीं से थर्मिस्टर तथा डायोड आते हैं। विद्युत्-अपघट्य: दोनों चिह्नों के
आयन, $\mu \sim 10^{-8}$--$10^{-7}$ m$^2$/(V s); समुद्र का पानी $\gamma \approx \qty{5}{S/m}$,
नल का पानी $10^{-2}$, शुद्ध पानी $10^{-5}$। विद्युतरोधी (काँच, बहुलक):
$\gamma \lesssim 10^{-12}$ S/m। वर्ष 1 के खंड का हॉल प्रभाव, जिसका
स्थानिक रूप $\vect E_H = -\vect j\wedge\vect B/nq$ है, $n$ और $q$ का
चिह्न नापता है: ऐलुमिनियम या ज़िंक में वाहक धनात्मक निकलते हैं ---
अर्थात् कोटर --- और कोई चिरसम्मत प्रतिरूप इसे समझा नहीं पाता।
\end{remark}

\section{ऊर्जा: स्थानिक जूल नियम}

\begin{proposition}[स्थानिक जूल नियम]\label{prop:b2:charges-currents-conduction:joule}
\index{जूल नियम (स्थानिक)}
विद्युत क्षेत्र वाहकों को प्रति एकांक आयतन और प्रति एकांक समय
शक्ति $\vect j\cdot\vect E$ देता है; और किसी ओमी चालक में यह
\[
p = \vect j\cdot\vect E = \gamma E^2 = \frac{j^2}\gamma \ \ (\unit{W/m^3}) ,
\]
है, जिसका सारा भाग संघट्टों से ऊष्मा में बदल जाता है (अपवाह की गतिज ऊर्जा
नगण्य और अचर है)। किसी तार पर समाकलित करने पर $\int j^2/\gamma\,\dd\tau
= (L/\gamma S)I^2 = RI^2$।
\end{proposition}

\begin{proof}
हर वाहक पर बल $q\vect E$ दर $q\vect E\cdot\vect v$ से कार्य करता है;
और $n\,\dd\tau$ वाहकों पर योग लेने पर: $nq\vect v\cdot\vect E\,\dd\tau = \vect j\cdot
\vect E\,\dd\tau$ (चुंबकीय बल कोई कार्य नहीं करता)। ड्रूडे की स्थायी
दशा में यह शक्ति उसी के बराबर है जो घर्षण-पद से क्षय होती है, अर्थात्
जालक को सौंप दी जाती है।
\end{proof}

\begin{example}[गरम होता तार]\label{ex:b2:charges-currents-conduction:heating}
\qty{10}{A} पर \qty{1.5}{mm^2} का तार: $p = j^2/\gamma = (6.7 \times 10^6)^2/6 \times 10^7 =
\qty{7.4e5}{W/m^3}$, अर्थात् प्रति मीटर \qty{1.1}{W} --- जिसे चारों ओर की हवा आसानी से
निकाल ले जाती है। अपने हाल पर छोड़ दिया जाए (कोई शीतलन नहीं) तो ताँबा ($c = \qty{385}{J/(kg.K)}$,
$\rho = \qty{8900}{kg/m^3}$) $p/\rho c = \qty{0.2}{K/s}$ से गरम होता। उसी तार में \qty{100}{A}
पर दर सौ गुना बड़ी, \qty{20}{K/s}, हो जाती है: और वह
मिनट भर में पिघल जाता है --- यही फ़्यूज़ का सिद्धांत है, जिसका पतला तार
पहले पिघलने के लिए ही बनाया जाता है।
\end{example}

\begin{proposition}[किसी चालक में आवेश का शिथिलन]\label{prop:b2:charges-currents-conduction:relaxation}
\index{शिथिलन काल (चालक)}
किसी ओमी चालक में कोई आयतन \omterm{def:b2:charges-currents-conduction:densities}{आवेश घनत्व} $\rho(t) =
\rho_0\eu^{-t/\tau_r}$ की तरह क्षय होता है, जहाँ
\[
\tau_r = \frac{\varepsilon_0}{\gamma} :
\]
है: अर्थात् ताँबे के लिए \qty{1.5e-19}{s}, समुद्र के पानी के लिए \qty{2e-12}{s}, काँच के लिए \qty{10}{s}।
$1/\tau_r$ कोटि की आवृत्तियों तक कोई चालक \emph{कोई आयतन
आवेश} नहीं ढोता: कोई भी अतिरिक्त आवेश उसकी सतह पर दौड़ जाता है। अतः शब्द
``चालक'' काल-पैमाने पर निर्भर करता है: समुद्र का पानी रेडियो-तरंगों के लिए चालक
है और दृश्य प्रकाश के लिए विद्युतरोधी।
\end{proposition}

\begin{proof}
आवेश-संरक्षण, $\vect j = \gamma\vect E$ और गाउस का नियम स्थानिक
रूप में, $\operatorname{div}\vect E = \rho/\varepsilon_0$ (\cref{ch:b2:maxwell-equations}):
$\partial_t\rho = -\gamma\operatorname{div}\vect E = -(\gamma/\varepsilon_0)\rho$।
\end{proof}

\begin{method}[स्थानिक से समाकल नियम तक]\label{met:b2:charges-currents-conduction:method}
किसी मनमाने आकार के चालक का प्रतिरोध निकालने के लिए: (1) $\vect j$ लिखने के लिए
सममिति काम में लीजिए (स्थायी दशा में उसका अभिवाह संरक्षित है, अतः किसी
नलिका में $j \times$ काट $= I$); (2) $\vect E = \vect j/\gamma$; (3) किसी
एक इलेक्ट्रोड से दूसरे तक किसी रेखा के अनुदिश $\vect E$ समाकलित कीजिए: $U = \int\vect E\cdot
\dd\vect l$, और $R = U/I$; (4) जूल शक्ति $\int j^2/\gamma\,
\dd\tau = RI^2$ से जाँचिए। यही क़दम किसी केबल का रिसाव, किसी तड़ित-चालक के
चारों ओर की ज़मीन का प्रतिरोध, या किसी शंक्वाकार स्पर्श का प्रतिरोध दे देते हैं।
\end{method}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:6}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:10}
\emph{भू-संयोजन।} $a = \qty{0.50}{m}$ त्रिज्या का कोई अर्धगोलीय इलेक्ट्रोड $\gamma =
\qty{0.010}{S/m}$ चालकता की मिट्टी की सतह के बराबर गाड़ा गया है;
उसमें कोई धारा $I$ घुसती है और अर्ध-आकाश में त्रिज्यीय रूप से फैल जाती है।
(क) $\vect j(r)$, $\vect E(r)$, और विभव $V(r)$, जहाँ $V(\infty) = 0$।
(ख) भू-संयोजन का प्रतिरोध, $R = 1/2\pi\gamma a$; मान। (ग) \qty{20}{kA} की
कोई बिजली छड़ पर गिरती है: छड़ का विभव;
$r = \qty{10}{m}$ और \qty{10.7}{m} पर विभव; और वहाँ खड़े किसी व्यक्ति के पैरों के बीच
``पग-वोल्टता'' (पैर \qty{0.7}{m} दूर, त्रिज्या के अनुदिश)।
(घ) पास गिरती बिजली से गाय-भैंस, जिनकी चारों टाँगें दूर-दूर होती हैं,
लोगों की अपेक्षा अधिक क्यों मरती हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:11}
\emph{हॉल संवेदक।} $t$ मोटाई और $w$ चौड़ाई की कोई पट्टी $I$ ढोती है,
$x$ के अनुदिश, और क्षेत्र $B$ में, जो $z$ के अनुदिश है। (क) $\vect E_H = -\vect j\wedge\vect B
/nq$ से दिखाइए कि चौड़ाई के पार हॉल वोल्टता $V_H = IB/nqt$ है। (ख)
ताँबे की पट्टी, $t = \qty{0.10}{mm}$, \qty{1}{A}, \qty{1}{T}: $V_H$। (ग) उसी ज्यामिति का
$n = \qty{1e22}{m^{-3}}$ वाला कोई अपमिश्रित अर्धचालक: $V_H$; और संवेदक
अर्धचालक क्यों काम में लेते हैं। (घ) $n = \qty{1e22}{m^{-3}}$, $t =
\qty{20}{\micro m}$ वाला कोई संवेदक \qty{1}{mA} से खिलाया जाता है और \qty{10}{\micro V} पढ़ता है: नापा गया क्षेत्र।
$V_H$ का चिह्न क्या बताता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:12}
\emph{विद्युत्-अपघट्य।} पानी में Na$^+$ और Cl$^-$ आयनों की गतिशीलताएँ
$\mu_+ = \qty{5.2e-8}{m^2/(V.s)}$ और $\mu_- = \qty{7.9e-8}{m^2/(V.s)}$ हैं। (क)
दिखाइए कि $\gamma = ne(\mu_+ + \mu_-)$, जब प्रति एकांक आयतन $n$ आयन-युग्म हों।
(ख) समुद्र का पानी, \qty{0.5}{mol/L} सोडियम क्लोराइड: $\gamma$; नापे गए
\qty{5}{S/m} से तुलना कीजिए। (ग) \qty{10}{V/m} पर अपवाह चालें; कौन-सा आयन धारा का
अधिक भाग ढोता है? (घ) समुद्र के पानी का कोई स्तंभ, \qty{1}{cm^2} का और \qty{10}{cm} लंबा,
\qty{1}{A} ढोता है: वोल्टता, शक्ति, और उसे \qty{10}{K} गरम करने में लगा समय ($c =
\qty{4.0}{kJ/(kg.K)}$); वैद्युतकणसंचलन के जेल ठंडे क्यों रखे जाते हैं?
\end{exercise}

\section{समस्या: ताँबे का तार, परमाणु से ग्रिड तक}

\begin{problem}[{सप्ताहांत समस्या --- घर के तार का एक मीटर, चार पैमानों पर
जाँचा गया: इलेक्ट्रॉन, स्थानिक नियम, परिपथ, और
विद्युत-लाइन}]
\label{pb:b2:charges-currents-conduction:1}

ताँबा: मोलर द्रव्यमान \qty{63.5}{g/mol}, घनत्व \qty{8900}{kg/m^3}, प्रति परमाणु एक
चालन-इलेक्ट्रॉन, \qty{20}{\degreeCelsius} पर चालकता $\gamma = \qty{6.0e7}{S/m}$,
और चढ़ती प्रतिरोधकता $\rho_e(T) = \rho_e(20)[1 +
\alpha(T - 20)]$, जहाँ $\alpha = \qty{4.0e-3}{K^{-1}}$; $c = \qty{385}{J/(kg.K)}$। तार:
काट $S = \qty{2.5}{mm^2}$, लंबाई \qty{1}{m}, धारा \qty{16}{A}।

\medskip
\noindent\textbf{भाग I --- इलेक्ट्रॉन।}
\begin{enumerate}
  \item चालन-इलेक्ट्रॉनों का संख्या-घनत्व $n$।
  \item \omterm{def:b2:charges-currents-conduction:densities}{धारा घनत्व} और \omterm{ex:b2:charges-currents-conduction:drift}{अपवाह वेग}; किसी इलेक्ट्रॉन को वह मीटर पार करने में
        कितना समय चाहिए।
  \item ड्रूडे का $\tau$ और गतिशीलता; \qty{1.6e6}{m/s} की यादृच्छिक
        चाल के साथ माध्य मुक्त पथ; वह कितने परमाणु-अंतराल (\qty{0.26}{nm})
        है, और इससे यह क्या पता चलता है कि इलेक्ट्रॉन किससे
        टकराते हैं?
  \item ऊष्मीय कंपन $T$ के साथ बढ़ते हैं, अतः मोटे तौर पर $\tau \propto 1/T$:
        कमरे के ताप के पास इसे $\alpha$ से जाँचिए ($\dd\rho_e/\rho_e =
        \dd T/T$ से $\alpha = 1/293$ मिलता)।
  \item तार में विद्युत क्षेत्र; उस मीटर पर कुल वोल्टता; और
        प्रति इलेक्ट्रॉन अपवाह की गतिज ऊर्जा की $k_BT$ से तुलना ---
        क्या इलेक्ट्रॉन-गैस धारा से विचलित होती है?
  \item प्रति सेकंड तार की किसी काट को पार करते इलेक्ट्रॉनों की संख्या।
  \item तार के उस मीटर में चालन-इलेक्ट्रॉनों का आवेश; उन सब पर मिलकर
        क्षेत्र जो बल लगाता है; और उसकी तार के भार से तुलना। वह बल
        कहाँ जा पहुँचता है (संघट्ट सोचिए)?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- स्थानिक नियम।}
\begin{enumerate}[resume]
  \item प्रति एकांक आयतन और प्रति मीटर जूल शक्ति।
  \item हवा में तार प्रति एकांक पृष्ठ $h(T - T_a)$ दर से ऊष्मा खोता है,
        जहाँ $h = \qty{10}{W/(m^2.K)}$ (संवहन): साम्य में
        ताप-वृद्धि; और जाँचिए कि प्रतिरोधकता का सुधार
        मायने रखता है या नहीं।
  \item वही तार ऐसे विद्युतरोधन में गड़ा हो जो केवल $h =
        \qty{2}{W/(m^2.K)}$ जाने दे: प्रतिरोधकता का सुधार जोड़कर नई
        ताप-वृद्धि (संतुलन हल कीजिए)।
  \item बिना किसी शीतलन के छोड़ दिया जाए तो \qty{16}{A} पर और \qty{160}{A}
        (कोई लघु-परिपथ) पर ताप-वृद्धि की दर; दूसरी दशा में गलनांक
        (\qty{1085}{\degreeCelsius}) तक पहुँचने का समय; और संस्थापन की
        रक्षा कौन करता है?
  \item \qty{50}{Hz} पर $\omega\tau = ?$ क्या ओम का नियम प्रभावित होता है? (\cref{ch:b2:waves-in-media}
        का त्वचा-प्रभाव अलग बात है।)
  \item ताँबे का शिथिलन काल $\varepsilon_0/\gamma$ लिखिए; तार के भीतर \omterm{def:b2:charges-currents-conduction:densities}{आवेश घनत्व} के
        लिए इसका क्या अर्थ है, और वह आवेश कहाँ बैठता है जो
        तार के भीतर क्षेत्र $E$ बनाता है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- परिपथ में आवेश का संरक्षण।}
तार किसी स्विच से होकर किसी संधारित्र $C = \qty{100}{\micro F}$ को खिलाता है; और
धारा $I(t) = I_0\eu^{-t/\tau_c}$ है, जहाँ $I_0 = \qty{16}{A}$, $\tau_c = \qty{1}{ms}$।
\begin{enumerate}[resume]
  \item प्लेट तक पहुँचता आवेश; और अंतिम वोल्टता।
  \item प्लेट के चारों ओर किसी बंद पृष्ठ पर $\dd Q/\dd t = -I_{\text{निर्गम}}$
        लगाइए: कौन-सी ओर चालन-धारा ढोती है और कौन-सी नहीं?
        अतः अंतराल में किस राशि का बदलना आवश्यक है?
  \item $t = \qty{0.5}{ms}$ पर प्लेट-आवेश के बदलने की दर; और यदि
        प्लेटें \qty{1}{cm^2} की हों तथा हवा में \qty{0.1}{mm} दूर, तो उनके बीच क्षेत्र के
        बदलने की दर।
  \item दिखाइए कि $\varepsilon_0\,\dd E/\dd t \times$ प्लेट-क्षेत्रफल तार की धारा के
        बराबर है --- अर्थात् वही राशि जिसे \cref{ch:b2:maxwell-equations}
        ऐंपियर के नियम में जोड़ता है।
  \item तार काट दिया जाता है और दोनों सिरे हवा में \qty{1}{mm} दूर हैं:
        कौन-सी धारा बहती है (हवा का $\gamma \approx \qty{1e-14}{S/m}$)? किस
        क्षेत्र पर हवा का भंजन होता है (\qty{3}{MV/m}): अंतराल पर कौन-सी वोल्टता
        चिंगारी देती है?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- लाइन।}
\qty{10}{mm^2} ताँबे की कोई \qty{230}{V} लाइन, \qty{500}{m} लंबी (जाना और लौटना:
तार का \qty{1}{km}), किसी खेत को \qty{40}{A} देती है।
\begin{enumerate}[resume]
  \item लाइन का प्रतिरोध, वोल्टता-गिरावट और खोई शक्ति; तथा
        पहुँचाई गई शक्ति का अंश।
  \item दोनों सिरों पर परिणामित्र लगाकर वही शक्ति \qty{20}{kV} पर:
        धारा, गिरावट और हानि; और इससे निष्कर्ष निकालिए कि संचरण
        ऊँची वोल्टता पर क्यों होता है।
  \item खंभों पर ऐलुमिनियम ($\gamma = \qty{3.6e7}{S/m}$, \qty{2700}{kg/m^3}) काम में
        लिया जाता है: ताँबे के \qty{10}{mm^2} जितने प्रतिरोध के लिए
        काट; दोनों के लिए प्रति किलोमीटर द्रव्यमान; और ऊपर ऐलुमिनियम
        तथा दीवार में ताँबा क्यों जीतता है।
  \item लाइन के चारों ओर कसा कोई हॉल संवेदक उसका क्षेत्र पढ़ता है: किसी
        क्लैंप मापी के लिए, जिसमें $n = \qty{1e22}{m^{-3}}$, $t = \qty{0.1}{mm}$ हो, जो
        \qty{5}{mA} से खिलाया जाए और लोहे के क्रोड से संकेंद्रित \qty{0.1}{T} क्षेत्र में हो,
        हॉल वोल्टता।
  \item किसी गरम दिन के भार के बाद तार \qty{80}{\degreeCelsius} पर बैठता है:
        तब उसका प्रतिरोध, और अतिरिक्त हानि।
  \item लाइन पर बिजली गिरती है: \qty{100}{\micro s} के लिए \qty{20}{kA}। तार के
        \qty{1}{km} में क्षयित ऊर्जा; कुछ भी न निकले तो ताप-वृद्धि;
        और \omterm{def:b2:charges-currents-conduction:densities}{धारा घनत्व} --- भाग I से तुलना कीजिए।
  \item सार दीजिए: वे पाँच राशियाँ ($n$, $\tau$, $\gamma$, $j$, $E$) जो
        हर पैमाने पर तार का वर्णन करती हैं, और वह नियम जो हर
        युग्म को जोड़ता है।
\end{enumerate}
\end{problem}
