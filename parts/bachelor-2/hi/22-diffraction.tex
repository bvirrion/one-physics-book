\chapter{फ्राउनहोफ़र विवर्तन और दैशिक छनन}\label{ch:b2:diffraction}

दो उँगलियों के बीच की झिरी से किसी सड़क-दीपक को सिकोड़कर देखिए और वह
दीपक फ्रिंजों की किसी पट्टी में खिंच जाता है; सबसे महीन दूरदर्शी से किसी
तारे को देखिए और आपको कोई बिंदु नहीं, बल्कि मंद वृत्तों से घिरी कोई छोटी
चकती दिखती है। किसी छिद्र से गुज़रता प्रकाश फैल जाता है --- छिद्र जितना
सँकरा, फैलाव उतना चौड़ा --- और कोई लेंस उसे इस फैलाव से अधिक कसकर फ़ोकस नहीं
कर सकता। यही \emph{\omterm{def:b2:diffraction:huygens}{विवर्तन}} है, और यही कारण है कि कोई सूक्ष्मदर्शी किसी
विषाणु को नहीं देख पाता, किसी दूरदर्शी की विभेदन-शक्ति उसके व्यास के साथ
बढ़ती है, और कोई लेज़र-पुंज समांतर नहीं रह सकती। यह अध्याय दूर-क्षेत्र में
किसी द्वारक की विवर्तन-आकृति निकालता है, वह सीमा जो वह हर प्रकाशिक यंत्र
पर लगाती है, वह ढंग जिससे वह व्यतिकरण से जुड़ती है, और वह विचार --- आबे का
--- जो किसी लेंस के फ़ोकस तल को ऐसी जगह बना देता है जहाँ किसी प्रतिबिंब को
छाना जा सके।

\section{ह्यूगेन्स--फ्रेनेल सिद्धांत और फ्राउनहोफ़र दशा}

\begin{definition}[ह्यूगेन्स--फ्रेनेल सिद्धांत]\label{def:b2:diffraction:huygens}
\index{ह्यूगेन्स--फ्रेनेल सिद्धांत}\index{विवर्तन}\index{फ्राउनहोफ़र विवर्तन}
किसी \omterm{def:b2:scalar-light-model:path}{तरंगाग्र} का (या किसी तरंग से जगमगाए किसी द्वारक का) हर बिंदु किसी
गौण स्रोत की तरह बरतता है, जो वहाँ की तरंग के साथ एक ही कला में कोई गोलीय
तरंगिका उत्सर्जित करता है; और आगे की तरंग तरंगिकाओं का संसक्त योग है। किसी
द्वारक $\Sigma$ के लिए, जो आयाम $a$ की किसी समतल तरंग से जगमगाया हो, किसी दूर बिंदु तक
पहुँचता आयाम $\underline s(M) = K\iint_\Sigma\eu^{-\iu\varphi(P, M)}\dd S$ है, जहाँ
$\varphi(P, M)$ द्वारक-बिंदु $P$ से $M$ तक के पथ की कला है
($K$ कोई अचर)। \emph{फ्राउनहोफ़र} (दूर-क्षेत्र) विवर्तन वह हाल है
जहाँ $M$ अनंत पर हो --- या किसी लेंस के फ़ोकस तल में --- ताकि
द्वारक के सारे बिंदुओं से $M$ की ओर जाती किरणें समांतर हों,
दिशा $\vect u$ में, और
\[
\varphi(P, M) = \varphi_0 - \frac{2\pi}\lambda\,\vect{OP}\cdot\vect u , \qquad
\underline s(\vect u) \propto \iint_\Sigma\eu^{\iu(2\pi/\lambda)\vect{OP}\cdot\vect u}\dd S .
\]
आकृति अनंत पर देखी जाती है, या किसी लेंस के फ़ोकस $F'$ पर,
जहाँ दिशा $\vect u$, छोटे कोणों $(\alpha, \beta)$ की, बिंदु
$(f\alpha, f\beta)$ पर उतरती है।
\end{definition}
\admitted

\begin{remark}[कितना दूर काफ़ी दूर है?]\label{rem:b2:diffraction:fresnelnumber}
\index{फ्रेनेल संख्या}
आकार $D$ के किसी द्वारक के लिए, जो बिना किसी लेंस के दूरी $L$ पर देखा जाए,
किरणें तब पर्याप्त समांतर हैं जब
\cref{exo:b2:scalar-light-model:12} का द्विघाती पद नगण्य हो, $D^2/L \ll \lambda$ ---
अर्थात् \emph{\omterm{rem:b2:diffraction:fresnelnumber}{फ्रेनेल संख्या}} $D^2/\lambda L \ll 1$। \qty{500}{nm} पर \qty{0.1}{mm} की किसी झिर्री
को $L \gg \qty{2}{cm}$ चाहिए; और \qty{1}{cm} के किसी द्वारक को $L \gg \qty{200}{m}$ ---
इसीलिए लेंस, जो अनंत को अपने फ़ोकस तल पर ले आता है। और पास
(फ्रेनेल \omterm{def:b2:diffraction:huygens}{विवर्तन}) आकृतियाँ अधिक जटिल होती हैं; मसलन किसी सीधे किनारे की
छाया में उसके जगमगाए पहलू के अनुदिश फ्रिंजें होती हैं।
\end{remark}

\section{अकेली झिर्री और वृत्ताकार द्वारक}

\begin{proposition}[किसी झिर्री से विवर्तन]\label{prop:b2:diffraction:slit}
\index{अकेली-झिर्री विवर्तन}\index{ज्यावक्रज फलन}
अभिलंब आपतन पर जगमगाई चौड़ाई $b$ की कोई झिर्री ($y$ के अनुदिश लंबी)
दिशा $\theta$ में (झिर्री के लंबवत तल में) यह
तीव्रता भेजती है
\[
I(\theta) = I_0\,\operatorname{sinc}^2\Bigl(\frac{\pi b\sin\theta}\lambda\Bigr) , \qquad \operatorname{sinc}x = \frac{\sin x}x :
\]
अर्थात् कोणीय अर्ध-चौड़ाई $\lambda/b$ का कोई चमकीला केंद्रीय अधिकतम (पहले शून्य
$\sin\theta = \pm\lambda/b$ पर), जो प्रकाश का $90\%$ रखता है, और \omterm{thm:b2:gratings:nwaves}{गौण अधिकतम}
लगभग $\sin\theta = \pm1.43\lambda/b$, $\pm2.46\lambda/b$, \dots पर, जिनकी सापेक्ष
तीव्रताएँ $4.7\%$, $1.6\%$, \dots हैं --- झिर्री जितनी सँकरी, आकृति उतनी चौड़ी।
फ़ोकस दूरी $f$ के किसी लेंस के फ़ोकस तल में
केंद्रीय अधिकतम $2\lambda f/b$ चौड़ा होता है।
\end{proposition}

\begin{proof}
$\underline s(\theta) \propto \int_{-b/2}^{b/2}\eu^{\iu kx\sin\theta}\dd x = b\,\operatorname{sinc}(kb\sin\theta/2)$,
जहाँ $k = 2\pi/\lambda$; वर्ग कीजिए। शून्य वहाँ जहाँ $b\sin\theta = m\lambda$, $m \ne 0$: तब झिर्री के
दोनों आधों की तरंगिकाएँ जोड़ी-दर-जोड़ी कट जाती हैं।
\end{proof}

\begin{proposition}[वृत्ताकार द्वारक; एरी चकती]\label{prop:b2:diffraction:airy}
\index{एरी चकती}\index{रैले कसौटी (विभेदन)}\index{विभेदन-शक्ति (यंत्र)}
व्यास $D$ का कोई वृत्ताकार द्वारक कोई केंद्रीय चमकीली चकती देता है,
\emph{\omterm{prop:b2:diffraction:airy}{एरी चकती}}, जिसके चारों ओर मंद वलय हैं; और पहला अँधेरा वलय यहाँ है
\[
\sin\theta = 1.22\,\frac\lambda D ,
\]
और चकती प्रकाश का $84\%$ रखती है। व्यास $D$ की प्रवेश-पुतली वाला हर
यंत्र --- आँख, कैमरा, दूरदर्शी, सूक्ष्मदर्शी का
अभिदृश्यक --- किसी बिंदु स्रोत को उसी कोणीय त्रिज्या की किसी \omterm{prop:b2:diffraction:airy}{एरी चकती} में
बदल देता है; और दो बिंदु तब \emph{सुलझते} हैं (रैले की कसौटी) जब उनकी
कोणीय दूरी $1.22\lambda/D$ से बड़ी हो: यही यंत्र की
विभेदन-शक्ति है।
\end{proposition}
\admitted

\begin{example}[आँख, दूरदर्शी, सूक्ष्मदर्शी]\label{ex:b2:diffraction:instruments}
आँख ($D = \qty{3}{mm}$, $\lambda = \qty{550}{nm}$): $1.22\lambda/D = \qty{2.2e-4}{rad}$, अर्थात् लगभग

तीक्ष्णता है: विकास ने रेटिना की कोशिकाओं को \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा से मिला
दिया। \qty{2.4}{m} का कोई दूरदर्शी: \qty{0.06}{''}, अर्थात् हज़ार गुना
बेहतर, यदि वायुमंडल दे (ज़मीन से वह नहीं देता:
\qty{1}{''} की "दृश्यता", इसीलिए अंतरिक्ष-दूरदर्शी और अनुकूली प्रकाशिकी)। कोई

\section{विवर्तन और व्यतिकरण साथ-साथ}

\begin{proposition}[परिमित चौड़ाई की यंग की झिर्रियाँ; जालक का आवरण]\label{prop:b2:diffraction:combined}
\index{आवरण (विवर्तन)}\index{ग़ायब कोटियाँ}
चौड़ाई $b$ की दो झिर्रियाँ, जिनके केंद्र $a$ दूर हैं, देती हैं
\[
I(\theta) = 4I_0\,\operatorname{sinc}^2\Bigl(\frac{\pi b\sin\theta}\lambda\Bigr)\cos^2\Bigl(\frac{\pi a\sin\theta}\lambda\Bigr) :
\]
अर्थात् यंग की फ्रिंजें (दूरी $\lambda/a$, $\sin\theta$ में) अकेली-झिर्री के आवरण
(चौड़ाई $2\lambda/b$) के नीचे; $a/b$ फ्रिंजें केंद्रीय लोब भर देती हैं, और
$a\sin\theta = p\lambda$ वाली वे फ्रिंजें जो आवरण के किसी शून्य
($b\sin\theta = m\lambda$) पर पड़ें \emph{ग़ायब} रहती हैं। $N$ झिर्रियों का कोई जालक,
जिनकी चौड़ाई $b$ हो, भी अपने मुख्य अधिकतमों पर वही आवरण ओढ़ लेता है ---
जिसे \cref{ch:b2:gratings} का ब्लेज़ काम में ली जा रही कोटि की ओर झुका देता है।
\end{proposition}

\begin{proof}
द्वारक का समाकल दोनों झिर्रियों पर उसी समाकल के योग में बँट जाता है,
जो $\pm a/2$ से खिसका है: $\underline s \propto b\,\operatorname{sinc}(\pi b\sin\theta/\lambda)
\times2\cos(\pi a\sin\theta/\lambda)$। $N$ झिर्रियों के लिए दूसरा गुणक
$N$-तरंग योग है।
\end{proof}

\section{फ़ूरिये तल और दैशिक छनन}

\begin{proposition}[लेंस कोई फ़ूरिये विश्लेषक]\label{prop:b2:diffraction:fourier}
\index{फ़ूरिये तल}\index{दैशिक आवृत्ति}\index{दैशिक छनन}\index{आबे का सिद्धांत}
किसी लेंस के अगले फ़ोकस तल में रखी पारगम्यता $t(x, y)$ की कोई पारदर्शिका,
किसी समतल तरंग से जगमगाई हुई, पिछले फ़ोकस तल में यह
आयाम बनाती है
\[
\underline s(X, Y) \propto \iint t(x, y)\,\eu^{-\iu2\pi(xX + yY)/\lambda f}\dd x\,\dd y
\]
--- अर्थात् उसका द्विविम फ़ूरिये रूपांतर, जिसमें \emph{\omterm{prop:b2:diffraction:fourier}{फ़ूरिये तल}} का हर बिंदु
$(X, Y)$ कोणों $(X/f, Y/f)$ पर विवर्तित प्रकाश बटोरता है, अर्थात् वस्तु की
\emph{दैशिक आवृत्तियाँ} $(X/\lambda f, Y/\lambda f)$: महीन ब्योरे (ऊँची
आवृत्तियाँ) अक्ष से दूर, मोटे उसके पास। कोई दूसरा लेंस उस तल से प्रतिबिंब
फिर बना देता है; और वहाँ जो कुछ रोका या बदला जाए वह प्रतिबिंब से छन जाता है
(\emph{\omterm{prop:b2:diffraction:fourier}{दैशिक छनन}}, सूक्ष्मदर्शी का आबे-सिद्धांत): अक्ष पर कोई सूचिछिद्र
केवल नीची आवृत्तियाँ रखता है (चिकनाना, किसी लेज़र-पुंज को "साफ़" करना);
अक्ष पर कोई रोक एकसमान पृष्ठभूमि हटा देती है और
किनारे रख छोड़ती है (कृष्ण-क्षेत्र, स्ट्रायोस्कोपी); और अक्ष पर कोई \omterm{prop:b2:plane-waves-polarization:plates}{चतुर्थांश-तरंग पट्टिका}
कला-वस्तुओं को, जो नहीं तो अदृश्य हैं, तीव्रता-विभेद में बदल देती है
(कला-विभेद, ज़ेर्निके)।
\end{proposition}

\begin{proof}[औचित्य]
फ्राउनहोफ़र समाकल में $\vect{OP}\cdot\vect u = x\alpha + y\beta$, जहाँ
दिशा $(X, Y) = (f\alpha, f\beta)$ पर उतरती है; और पारगम्यता
हर गौण स्रोत को तौल देती है। (फ़ूरिये रूपांतर ख़ुद वर्ष 3 के गणित के खंड में
पढ़ा जाता है; यहाँ वह केवल इस समाकल का नाम है।)
आवर्त $d$ की किसी आवर्ती वस्तु के लिए रूपांतर $X = p\lambda f/d$ पर जालक की
कोटियों का समुच्चय है: और प्रतिबिंब आवर्त तभी दिखा सकता है जब
कम-से-कम कोटियाँ $\pm1$ लेंस से पार जाएँ --- यही आबे की शर्त है, जो
फिर से सूक्ष्मदर्शी के विभेदन की सीमा है।
\end{proof}

\begin{example}[किसी पुंज को साफ़ करना, अदृश्य को देखना]\label{ex:b2:diffraction:filtering}
कोई लेज़र-पुंज धूल और लहरें ढोती है: उसे कुछ माइक्रोमीटर के किसी
सूचिछिद्र से होकर फ़ोकस कीजिए --- अर्थात् साफ़ गाउसी पुंज की केंद्रीय \omterm{prop:b2:diffraction:airy}{एरी चकती}
जितना --- और बाक़ी सब, जो बाहर पड़ता है, रुक जाता है;
और फिर से संरेखित पुंज चिकनी है: कोई \emph{दैशिक छन्ना}। कोई जीवित
कोशिका पारदर्शी है; वह प्रकाश की केवल कला बदलती है। अक्षीय धब्बा
रोक देने पर उसके किनारे काले पर चमकीले हो जाते हैं; और ज़ेर्निके की पट्टिका
अविवर्तित प्रकाश को चौथाई तरंग खिसका देती है ताकि वह विवर्तित प्रकाश से
तीव्रता में व्यतिकरण करे: तब कोशिका का भीतरी भाग दिखने लगता है,
और जीवविज्ञान को अपना सूक्ष्मदर्शी मिल गया (1953 का नोबेल पुरस्कार)।
\end{example}

\begin{method}[विवर्तन के आकलन]\label{met:b2:diffraction:method}
(1) कोणीय फैलाव $\sim\lambda/b$, आकार $b$ के किसी द्वारक के लिए ($1.22\lambda/D$ किसी वृत्त के लिए);
और दूरी $L$ पर या फ़ोकस दूरी $f$ पर आकृति का आकार
$\lambda L/b$, $\lambda f/b$। (2) फ्राउनहोफ़र शर्त $b^2/\lambda L \ll 1$ जाँचिए या
कोई लेंस काम में लीजिए। (3) जोड़िए: \omterm{def:b2:diffraction:huygens}{विवर्तन} आवरण $\times$ व्यतिकरण
फ्रिंजें। (4) विभेदन: दो बिंदुओं को $\Delta\theta > 1.22\lambda/D$ चाहिए; और किसी आवर्ती
वस्तु को अपनी पहली कोटियाँ पुतली में घुसानी पड़ती हैं। (5) \omterm{prop:b2:diffraction:fourier}{फ़ूरिये
तल} वाला चित्र: कौन-सा ब्योरा हटाने के लिए क्या रोका जाए।
\end{method}

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffraction:11}
\emph{कृष्ण-क्षेत्र और कला-विभेद।} किसी पारदर्शी वस्तु का
$t(x) = \eu^{\iu\varphi(x)} \approx 1 + \iu\varphi(x)$ है, जहाँ $|\varphi| \ll 1$। (क) उसके साधारण
प्रतिबिंब की तीव्रता: दिखाइए कि वह पहली कोटि तक एकसमान है --- अर्थात् अदृश्य। (ख)
\omterm{prop:b2:diffraction:fourier}{फ़ूरिये तल} में "1" अक्षीय धब्बा है और $\iu\varphi$ उसके चारों ओर
विवर्तित प्रकाश: यदि कोई रोक अक्षीय धब्बा रोक दे तो प्रतिबिंब की
तीव्रता $\propto\varphi^2$ --- कृष्ण-क्षेत्र; और उसकी दुर्बलता। (ग) और यदि इसके बजाय
अक्षीय धब्बा $\pi/2$ से विलंबित कर दिया जाए ($\iu$ से गुणा): प्रतिबिंब $\propto|1 +
\varphi|^2 \approx 1 + 2\varphi$ --- कला-विभेद, $\varphi$ में रैखिक। (घ)
ज़ेर्निके की पट्टिका अक्सर अवशोषी भी क्यों होती है (वह अक्षीय धब्बा क्षीण कर देती है)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffraction:12}
\emph{किसी वर्णक्रमलेखी की झिर्री-प्रतिबिंब और जालक का विभेदन।} चौड़ाई
$W$ का कोई जालक वर्णक्रमलेखी का द्वारक है: दिखाइए कि
उस द्वारक की \omterm{def:b2:diffraction:huygens}{विवर्तन} आकृति, $\operatorname{sinc}^2(\pi W\sin\theta/\lambda)$
($\theta \approx 0$ पर निकलती पुंज के लिए), की कोणीय अर्ध-चौड़ाई
$\lambda/W$ है, और यह कि परिक्षेपण $p/a\cos
\theta$ से तरंगदैर्घ्य में बदलने पर वह $\Delta\lambda = \lambda/pN$ है --- अर्थात्
\cref{ch:b2:gratings} की विभेदन-शक्ति, \omterm{def:b2:diffraction:huygens}{विवर्तन} की तरह देखी गई। (ख) व्यास
$D$ का कोई दूरदर्शी और चौड़ाई $W$ का कोई जालक: जालक को
$D\,f_{\text{संरेखक}}/f_{\text{दूरदर्शी}}$ से चौड़ा होना चाहिए: क्यों? (ग) एक वाक्य में समझाइए कि
प्रकाशिकी की हर "विभेदन-शक्ति" $\sim$(द्वारक का आकार)/$\lambda$ क्यों है।
(घ) झिर्री क्या जोड़ती है, और वह कब हावी हो जाती है?
\end{exercise}

\section{समस्या: क्या सुलझाया जा सकता है}

\begin{problem}[{सप्ताहांत समस्या --- चार पैमानों पर विवर्तन की सीमा:
आँख, कोई दूरदर्शी, कोई सूक्ष्मदर्शी, और वह लेज़र-पुंज जिसे काम में लेने से
पहले साफ़ करना पड़ता है}]
\label{pb:b2:diffraction:1}

\medskip
\noindent\textbf{भाग I --- आँख।}
पुतली दिन में $D = \qty{3}{mm}$, रात में \qty{7}{mm}; $\lambda = \qty{550}{nm}$;
रेटिना की शंकु कोशिकाएँ दृष्टि-गर्त पर \qty{2}{\micro m} की दूरी पर हैं, और आँख की फ़ोकस
दूरी \qty{17}{mm} है।
\begin{enumerate}
  \item दिन में \omterm{prop:b2:diffraction:airy}{एरी चकती} की कोणीय त्रिज्या; रेटिना पर उसकी रैखिक
        त्रिज्या; शंकु-दूरी से तुलना कीजिए --- क्या रेटिना
        प्रकाशिकी से मिला हुआ है?
  \item \qty{25}{cm} पर सुलझे दो बिंदुओं की सबसे छोटी दूरी; और
        \qty{10}{m} पर; क्या आप बाँह भर दूरी पर \qty{1}{mm} का कोई पाठ पढ़ सकते हैं?
  \item रात में पुतली खुलकर \qty{7}{mm} हो जाती है: तब \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा;
        फिर भी दृष्टि और बदतर क्यों हो जाती है (पूरे द्वारक पर लेंस के
        विपथन, और शंकुओं से मोटी शलाकाएँ)?
  \item \qty{1.5}{m} की दूरी पर किसी गाड़ी की दो अग्रबत्तियाँ: वह दूरी जिस पर वे
        एक ही बत्ती में मिल जाती हैं।
  \item आँख के आगे थामा \qty{1}{mm} का कोई छेद: \omterm{def:b2:diffraction:huygens}{विवर्तन} क्या करता है,
        और फिर भी कोई सूचिछिद्र किसी निकट-दृष्टि आँख की मदद क्यों करता है
        (गहराई-क्षेत्र)?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- दूरदर्शी।}
\qty{1}{m} का कोई दूरदर्शी, $f = \qty{10}{m}$, और \qty{10}{\micro m} पिक्सलों वाला
कोई कैमरा; $\lambda = \qty{550}{nm}$।
\begin{enumerate}[resume]
  \item चाप-सेकंडों में \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा; संसूचक पर एरी
        त्रिज्या; और प्रति \omterm{prop:b2:diffraction:airy}{एरी चकती} पिक्सल।
  \item वायुमंडल प्रतिबिंबों को \qty{1}{''} तक धुँधला कर देता है: दूरदर्शी का
        कितना विभेदन खो जाता है? और वह व्यास जिसके लिए
        \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा दृश्यता के बराबर हो।
  \item अनुकूली प्रकाशिकी \qty{2.2}{\micro m} पर वायुमंडल को सुधार देती है:
        वहाँ \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा; अवरक्त में यह आसान क्यों है?
  \item \qty{0.15}{''} की दूरी पर घटकों वाला कोई युग्म तारा: क्या वह
        \qty{2.2}{\micro m} पर अनुकूली प्रकाशिकी से सुलझता है? और अंतरिक्ष से \qty{550}{nm} पर?
  \item चंद्रमा \qty{3.8e5}{km} दूर है: यह दूरदर्शी सबसे छोटा कौन-सा गड्ढा
        सुलझा सकता (केवल \omterm{def:b2:diffraction:huygens}{विवर्तन}); और किसी चंद्र-यान (\qty{4}{m}) का
        निशान?
  \item रात की आँख (\qty{7}{mm}) के सापेक्ष दूरदर्शी की प्रकाश-संचय
        शक्ति; वह क्या ख़रीदती है, और क्या नहीं?
  \item गौण दर्पण थामती चार पंखियों से होकर किसी तारे का
        प्रकाश: प्रतिबिंब का वर्णन कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- सूक्ष्मदर्शी।}
\omterm{prop:b2:guided-waves:fibre}{संख्यात्मक द्वारक} $\mathrm{NA} = n\sin\alpha = 0.9$ (सूखा) या $1.4$
(तेल) का कोई अभिदृश्यक, $\lambda = \qty{550}{nm}$; और सुलझी दूरी $0.61\lambda/\mathrm{NA}$ है।
\begin{enumerate}[resume]
  \item हर अभिदृश्यक के साथ सुलझी दूरी; और प्रति मिलीमीटर रेखा-युग्मों
        की संख्या।
  \item \qty{0.2}{\micro m} पर भीतरी संरचना वाला \qty{1}{\micro m} का कोई जीवाणु:
        हर एक से क्या दिखता है? और \qty{0.1}{\micro m} का कोई विषाणु?
  \item आबे: आवर्त $d$ की कोई आवर्ती संरचना तभी प्रतिबिंबित होती है जब उसकी
        $\sin\theta = \lambda/d$ वाली पहली कोटियाँ अभिदृश्यक में घुसें: न्यूनतम $d$,
        $\mathrm{NA} = 1.4$ के लिए; और $0.61\lambda/\mathrm{NA}$ से तुलना कीजिए।
  \item स्लाइड और अभिदृश्यक के बीच तेल विभेदन क्यों सुधार
        देता है?
  \item \qty{250}{nm} पर पराबैंगनी और \qty{4}{pm} पर कोई इलेक्ट्रॉन सूक्ष्मदर्शी:
        उसी सूत्र में सुलझी दूरियाँ (इलेक्ट्रॉनों के लिए
        व्यावहारिक NA \qty{0.01}{} है): तुलना कीजिए।
  \item कोई कोशिका पारदर्शी है: उज्ज्वल क्षेत्र में उसका प्रतिबिंब ख़ाली है।
        दो वाक्यों में समझाइए कि अभिदृश्यक के \omterm{prop:b2:diffraction:fourier}{फ़ूरिये तल} में रखी
        कला-विभेद पट्टिका क्या करती है।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- पुंज।}
\qty{532}{nm} का कोई लेज़र, पुंज का व्यास \qty{1.5}{mm}, को फैलाकर
\qty{30}{mm} करना और साफ़ करना है।
\begin{enumerate}[resume]
  \item कच्ची पुंज का \omterm{def:b2:diffraction:huygens}{विवर्तन} अर्ध-कोण ($\sim1.22\lambda/D$); और बिना किसी
        प्रकाशिकी के \qty{100}{m} के बाद उसका व्यास।
  \item $f_1 = \qty{10}{mm}$ का कोई लेंस उसे फ़ोकस करता है: फ़ोकस-धब्बे का व्यास;
        और चुना गया सूचिछिद्र-व्यास (धब्बे का $1.5\times$)।
  \item $f_2$ का कोई दूसरा लेंस उसे \qty{30}{mm} पर फिर संरेखित करता है: $f_2$; नया
        \omterm{def:b2:diffraction:huygens}{विवर्तन} कोण; और \qty{1}{km} के बाद व्यास।
  \item पहले लेंस पर \qty{100}{\micro m} की धूल किस कोण पर विवर्तित करती है;
        वह प्रकाश सूचिछिद्र-तल पर कहाँ पड़ता है, और उसका क्या
        होता है?
  \item सूचिछिद्र किसी गाउसी पुंज का $99\%$ पार जाने देता है: खोई शक्ति, और
        वह कहाँ जाती है।
  \item समांतर बने रहने के लिए फैली हुई पुंज को \emph{बड़ा} क्यों होना
        चाहिए ($\lambda/D$ को उस दूरी से जोड़िए जिस पर पुंज दुगुनी हो जाए,
        $\sim D^2/\lambda$)?
  \item इस समस्या की चारों सीमाएँ किसी एक सारणी में समेटिए: द्वारक,
        तरंगदैर्घ्य, $\lambda/D$, और वह क्या सीमित करती है।
\end{enumerate}
\end{problem}
