\chapter{विभव कूप, रोधिकाएँ और सुरंगन}\label{ch:b2:potential-wells-tunneling}

किसी कटोरे में कोई गेंद जितनी ऊर्जा उसे दी गई हो उसी से आगे-पीछे लुढ़कती है;
किसी परमाणु में कोई इलेक्ट्रॉन, किसी नाभिक में कोई न्यूक्लिऑन, किसी
नैनोमीटर-आकार के क्रिस्टल में कोई इलेक्ट्रॉन केवल कुछ ही ऊर्जाएँ रख सकते हैं, और उनमें
सबसे नीची शून्य नहीं होती। कोई गेंद किसी ऐसी पहाड़ी को पार नहीं कर सकती जो उसकी ऊर्जा से
ऊँची हो; कोई $\alpha$ कण किसी नाभिक से किसी ऐसी कूलॉम दीवार से होकर निकल जाता है जिस पर वह
कभी चढ़ ही नहीं सकता था, किसी क्रमवीक्षी सुरंगन सूक्ष्मदर्शी की सुई
निर्वात के किसी अंतराल के पार कोई धारा खींचती है, और अमोनिया का नाइट्रोजन परमाणु
अपने तीन हाइड्रोजनों के तल से होकर चौबीस अरब
बार प्रति सेकंड झूल जाता है। यह अध्याय श्रोडिंगर समीकरण को
सरलतम विभवों में हल करता है --- अनंत और परिमित दीवारों वाले कूप, कोई सोपान, कोई
रोधिका, कोई द्वि-कूप --- और उनमें वे दो महान क्वांटम
तथ्य पाता है जिन्हें चिरसम्मत यांत्रिकी उपजा नहीं सकती: बद्ध ऊर्जाओं का \emph{क्वांटन},
और चिरसम्मत रूप से वर्जित क्षेत्रों से होकर \emph{सुरंगन},
अपनी उस चरघातांकी संवेदिता के साथ जो रोधिका की चौड़ाई तथा ऊँचाई के प्रति है,
जिससे वह पिकोमीटरों का कोई पैमाना बन सकता है और जिससे
रेडियोऐक्टिव जीवनकाल तीस कोटियों तक फैले हैं।

\section{अनंत कूप}

\begin{proposition}[अनंत वर्गाकार कूप]\label{prop:b2:potential-wells-tunneling:infinite}
अभेद्य दीवारों से $0 < x < L$ में परिरुद्ध किसी कण ($V = 0$
भीतर, $V = \infty$ बाहर, अतः दीवारों पर $\varphi = 0$) की
\omterm{thm:b2:schrodinger-wave-functions:stationary}{स्थायी अवस्थाएँ} और ऊर्जाएँ हैं
\[
\varphi_n(x) = \sqrt{\frac{2}{L}}\sin\frac{n\pi x}{L}, \qquad
E_n = \frac{n^2h^2}{8mL^2} = n^2E_1, \qquad n = 1, 2, 3, \dots
\]
ऊर्जाएँ \emph{क्वांटित}\index{ऊर्जा का क्वांटन} हैं,
$n^2$ की तरह बढ़ती हैं, और सबसे नीची शून्य नहीं: वह \emph{परिरोध
ऊर्जा}\index{परिरोध ऊर्जा} $E_1 = h^2/8mL^2$ ही
स्थानीकरण की क़ीमत है ($\Delta x \sim L$ $\Delta p \sim h/2L$ पर बाध्य करता है)।
$\varphi_n$ लांबिक हैं, $\int\varphi_n\varphi_m\dd x = \delta_{nm}$, और कूप की कोई भी
अवस्था उनका कोई अध्यारोपण है।
\end{proposition}

\begin{proof}
भीतर, $\varphi'' = -k^2\varphi$ जिसमें $k^2 = 2mE/\hbar^2$: $\varphi = A\sin kx
+ B\cos kx$; $\varphi(0) = 0$ $B$ को मार देता है, $\varphi(L) = 0$ को $kL = n\pi$ चाहिए
--- किसी डोरी की \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंगें}, जिनमें $E = \hbar^2k^2/2m$।
प्रसामान्यन $A = \sqrt{2/L}$ देता है; और लांबिकता वही है जो
ज्याओं की।
\end{proof}

\begin{example}[परिमाण की कोटियाँ, और रंग]\label{ex:b2:potential-wells-tunneling:orders}
$E_1 = h^2/8mL^2$: \qty{1}{nm} में कोई इलेक्ट्रॉन, \qty{0.38}{eV};
\qty{0.1}{nm} (कोई परमाणु) में, \qty{38}{eV} --- परमाण्विक ऊर्जाओं का मापक; \qty{5}{fm}
में कोई न्यूक्लिऑन, \qty{8}{MeV} --- नाभिकीय ऊर्जाओं का मापक; किसी बक्से में
कोई कंचा, $10^{-63}\,\unit{J}$ --- कोई ध्यान नहीं देगा। कुछ नैनोमीटर चौड़ा कोई
अर्धचालक नैनोक्रिस्टल (कोई \emph{क्वांटम
बिंदु}) अपने इलेक्ट्रॉनों को परिरुद्ध करता है: \omterm{prop:b2:potential-wells-tunneling:infinite}{परिरोध ऊर्जा}
क्रिस्टल के अंतराल में जुड़ जाती है, अतः बिंदु जितना छोटा, उत्सर्जित प्रकाश उतना नीला ---
वही पदार्थ \qty{6}{nm} पर लाल और \qty{3}{nm} पर हरा दमकता है,
केवल आकार से समस्वरित।
\end{example}

\section{परिमित कूप}

\begin{proposition}[परिमित वर्गाकार कूप]\label{prop:b2:potential-wells-tunneling:finite}
$V = 0$ जब $|x| < a$ और बाहर $V = V_0$, तो $0 < E < V_0$ वाली कोई अवस्था
भीतर दोलन करती है ($k = \sqrt{2mE}/\hbar$) और बाहर क्षय होती है ($\kappa =
\sqrt{2m(V_0 - E)}/\hbar$): दीवारों के परे $\varphi \propto \eu^{-\kappa|x|}$।
$\varphi$ और $\varphi'$ का मिलान $x = \pm a$ पर, सम तथा विषम
अवस्थाओं के लिए, देता है
\[
k\tan ka = \kappa \quad\text{(सम)}, \qquad
-k\cot ka = \kappa \quad\text{(विषम)}, \qquad\text{जिसमें}\quad
(ka)^2 + (\kappa a)^2 = \frac{2mV_0a^2}{\hbar^2} \equiv R^2 .
\]
आलेखीय रूप से, हल वक्रों $\eta =
\xi\tan\xi$, $\eta = -\xi\cot\xi$ के वृत्त $\xi^2 + \eta^2 = R^2$ से कटान हैं
($\xi = ka$, $\eta = \kappa a$): \emph{सदा कम से कम एक बद्ध
अवस्था} होती है (सम मूल अवस्था), और कुल मिलाकर $1 + \lfloor 2R/\pi\rfloor$।
कण वर्जित क्षेत्र में गहराई $1/\kappa$ पर \emph{भेदन} करता है,
और स्तर उसी चौड़ाई के अनंत कूप वालों से
नीचे पड़ते हैं।
\end{proposition}

\begin{proof}
सम हल भीतर $A\cos kx$ और बाहर $B\eu^{-\kappa|x|}$ है;
$\varphi'/\varphi$ का सांतत्य $x = a$ पर $-k\tan ka = -\kappa$ देता है। विषम वाले के लिए
$\sin$ के साथ वही। वृत्त $k$ और
$\kappa$ की परिभाषा है। $\xi\tan\xi$ की हर शाखा, जो $\xi = n\pi$ से शुरू हो, और
$-\xi\cot\xi$ की, जो $(n + \tfrac12)\pi$ से शुरू हो, वृत्त से एक बार मिलती है यदि वह
उसके भीतर से शुरू हो: यही गिनती है।
\end{proof}

\begin{example}[किसी नैनोमीटर कूप में कोई इलेक्ट्रॉन]\label{ex:b2:potential-wells-tunneling:finite}
$V_0 = \qty{1}{eV}$, चौड़ाई $2a = \qty{1}{nm}$: $R = a\sqrt{2mV_0}/\hbar = 0.5 \times
10^{-9} \times 5.1 \times 10^9 = 2.6$, अतः $1 + \lfloor 2R/\pi\rfloor = 2$ बद्ध
अवस्थाएँ; मूल अवस्था अनंत कूप की \qty{0.38}{eV} के बजाय \qty{0.26}{eV} के
पास पड़ती है, और दीवारों में \qty{0.2}{nm} रिसती है
($1/\kappa = \hbar/\sqrt{2m(V_0 - E)}$)। ऐसे कूप, कुछ नैनोमीटर मोटी
अर्धचालक परतों के रूप में उगाए जाकर, \cref{ch:b2:laser} के डायोड
लेज़रों का हृदय हैं।
\end{example}

\section{सोपान और रोधिका: सुरंगन}

\begin{proposition}[विभव सोपान]\label{prop:b2:potential-wells-tunneling:step}
ऊर्जा $E$ का कोई कण जो $x < 0$ से आकर सोपान $V = V_0$ पर पड़े,
$x > 0$ के लिए:
\begin{itemize}
\item $E > V_0$: बाईं ओर $\varphi = \eu^{\iu k_1x} + r\eu^{-\iu k_1x}$, दाईं ओर $t\eu^{\iu
      k_2x}$, जिनमें $k_{1,2} = \sqrt{2m(E - V_{0,\text{बाएँ/दाएँ}})}/
      \hbar$; $\varphi$ और $\varphi'$ का सांतत्य $r = (k_1 - k_2)/(k_1
      + k_2)$ तथा परावर्तन प्रायिकता (धाराओं का अनुपात) $R = r^2$,
      $T = 1 - R = 4k_1k_2/(k_1 + k_2)^2$ देता है --- कोई कण \emph{किसी ऐसे सोपान से
      परावर्तित हो सकता है जिस पर चढ़ने की ऊर्जा उसके पास है}, ठीक वैसे ही जैसे किसी डोरी पर कोई तरंग
      प्रतिबाधा के किसी परिवर्तन से (\cref{ch:b2:wave-interfaces});
\item $E < V_0$: दाईं ओर $\varphi = t\eu^{-\kappa x}$, $\kappa = \sqrt{2m(V_0
      - E)}/\hbar$ --- कोई \emph{क्षयी तरंग}\index{क्षयी तरंग
      (क्वांटम)}; $|r| = 1$ (पूर्ण परावर्तन, किसी कला-विस्थापन के साथ), सोपान में कोई
      धारा नहीं बहती, पर कण वहाँ $1/\kappa$ पर क्षय होती किसी
      प्रायिकता के साथ मिलता है।
\end{itemize}
\end{proposition}

\begin{proof}
$x = 0$ पर \omterm{thm:b2:fluid-kinematics:continuity}{सांतत्य समीकरण} लिखिए और हल कीजिए; धाराओं के लिए
$j = (|A|^2 - |B|^2)\hbar k/m$ लीजिए (\cref{ch:b2:schrodinger-wave-functions})।
$E < V_0$ के लिए, $r = (k - \iu\kappa)/(k + \iu\kappa)$, जिसका मापांक एक है।
\end{proof}

\begin{theorem}[किसी रोधिका से होकर सुरंगन]\label{thm:b2:potential-wells-tunneling:tunnel}
ऊर्जा $E < V_0$ का कोई कण जो ऊँचाई $V_0$ और चौड़ाई $a$ की किसी आयताकार
रोधिका से मिले, इस प्रायिकता के साथ पारगत होता है
\[
T = \Big[1 + \frac{V_0^2\sinh^2(\kappa a)}{4E(V_0 - E)}\Big]^{-1}
\ \approx\ 16\,\frac{E}{V_0}\Big(1 - \frac{E}{V_0}\Big)\eu^{-2\kappa a}
\quad (\kappa a \gg 1), \qquad
\kappa = \frac{\sqrt{2m(V_0 - E)}}{\hbar} .
\]
यही वह \emph{सुरंग प्रभाव}\index{सुरंग प्रभाव} है: चिरसम्मत रूप से
असंभव, क्वांटम रूप से चरघातांकी रूप से छोटा --- और चौड़ाई $a$, ऊँचाई $V_0 - E$ तथा
द्रव्यमान $m$ के प्रति चरघातांकी रूप से संवेदी। शीर्ष से \qty{1}{eV} नीचे किसी इलेक्ट्रॉन के लिए, $\kappa = \qty{5.1}{nm^{-1}}$:
$T \sim 10^{-2}$ जब $a = \qty{0.5}{nm}$, $10^{-4}$ \qty{1}{nm} के लिए,
$10^{-9}$ \qty{2}{nm} के लिए; किसी प्रोटॉन के लिए $\kappa$ $43$ गुना बड़ा है और इन चौड़ाइयों पर
कुछ भी नहीं गुज़रता।
\end{theorem}

\begin{proof}
$\varphi = \eu^{\iu kx} + r\eu^{-\iu kx}$ $x < 0$ के लिए, भीतर $A\eu^{\kappa x} + B\eu^{-\kappa x}$,
$t\eu^{\iu kx}$ $x > a$ के लिए; चार सांतत्य प्रतिबंध ($\varphi$
तथा $\varphi'$ के, $0$ और $a$ पर) चार अज्ञातों के लिए; $A$, $B$, $r$ का निराकरण
$1/|t|^2 = 1 + (k^2 + \kappa^2)^2\sinh^2(\kappa a)/4k^2\kappa^2$ देता है, जो
$k^2 = 2mE/\hbar^2$, $\kappa^2 = 2m(V_0 - E)/\hbar^2$ के साथ वही सूत्र है।
$\kappa a \gg 1$ के लिए, $\sinh\kappa a \approx \eu^{\kappa a}/2$। किसी मनमाने आकार की
रोधिका के लिए चरघातांक वर्जित क्षेत्र के पार
$2\int\kappa(x)\dd x$ बन जाता है (गैमो गुणक, मान लिया गया)।
\end{proof}

\section{द्वि-कूप}

\begin{proposition}[सुरंग विपाटन]\label{prop:b2:potential-wells-tunneling:double}
किसी रोधिका से अलग किए दो सर्वसम कूप: यदि रोधिका
अभेद्य होती तो हर कूप का वही मूल स्तर $E_0$ होता, दो बार।
सुरंगन उन्हें युग्मित कर देता है और द्वि-कूप की दो निम्नतम अवस्थाएँ
\emph{सममित} तथा \emph{प्रतिसममित} संयोजन हैं,
$\varphi_\pm \approx (\varphi_{\text{बा}} \pm \varphi_{\text{दा}})/\sqrt2$, जिनकी ऊर्जाएँ
$E_0 \mp \Delta/2$ हैं: स्तर किसी मात्रा $\Delta \propto
\eu^{-\kappa a}$ से \emph{विपाटित} हो जाता है (सुरंगन आयाम, उसका वर्ग नहीं)। बाएँ कूप में
रखा कोई कण, $\psi = (\varphi_+ + \varphi_-)/\sqrt2$, स्थायी
नहीं है: वह कूपों के बीच आवृत्ति
$\nu = \Delta/h$ पर दोलन करता है --- \emph{सुरंगन दोलन}, यानी
\cref{ch:b2:schrodinger-wave-functions} के द्विस्तरीय विस्पंद।
\end{proposition}

\begin{proof}
सममित संयोजन का रोधिका के नीचे कोई निस्पंद नहीं और कोई नीची
वक्रता है, अतः कोई नीची ऊर्जा; प्रतिसममित वाले का कोई निस्पंद है और वह
ऊँचा पड़ता है; विपाटन रोधिका के नीचे स्थानीकृत फलनों के
अतिच्छादन के अनुपात में है, $\eu^{-\kappa a}$। विस्पंदन
वही द्विस्तरीय अध्यारोपण है जो पहले ही निकाला जा चुका।
\end{proof}

\begin{method}[कूप और रोधिकाएँ]\label{meth:b2:potential-wells-tunneling:method}
(1) $\varphi$ को क्षेत्र-दर-क्षेत्र लिखिए: $\eu^{\pm\iu kx}$ जहाँ $E > V$,
$\eu^{\pm\kappa x}$ जहाँ $E < V$, जिनमें $k, \kappa = \sqrt{2m|E - V|}/\hbar$।
(2) हर सीमा पर $\varphi$ और $\varphi'$ का मिलान कीजिए (किसी अनंत दीवार पर,
$\varphi = 0$)। (3) बद्ध अवस्थाएँ: मिलान के हल केवल
विविक्त $E$ के लिए होते हैं --- उन्हें आलेखीय रूप से गिनिए। (4) प्रकीर्णन: धाराएँ
$(|A|^2 - |B|^2)\hbar k/m$ $R$ और $T$ देती हैं। (5) सुरंगन: $T \sim
\eu^{-2\kappa a}$; पहले परिमाण की कोटि, फिर पूर्वगुणक। (6)
द्वि-कूप: विपाटन $\propto\eu^{-\kappa a}$, विस्पंद $\Delta/h$ पर।
\end{method}

$b$, $G$, $T$। (ख) टक्करों की आवृत्ति $v/2R_{\text{n}}$ और जीवनकाल। (ग)
$E = 4$ और \qty{6}{MeV} के लिए दोहराइए: जीवनकालों का अनुपात। (घ) नाभिक
$\alpha$ कण क्यों उत्सर्जित करते हैं, प्रोटॉनों या $^{12}$C के बजाय?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:10}
\emph{अमोनिया।} विपाटन $\Delta = \qty{1e-4}{eV}$। (क) प्रतिलोमन आवृत्ति
और सुरंगन दोलन का आवर्तकाल। (ख) N परमाणु एक ओर
तैयार किया जाता है: $\psi(t)$ और उसे दूसरी ओर पाने की प्रायिकता
लिखिए। (ग) ND$_3$ \qty{1.6}{GHz} पर सुरंगन करता है: विपाटनों के अनुपात से $\kappa a$
निकालिए, यदि $\Delta \propto \eu^{-\kappa a}$ और $\kappa
\propto\sqrt m$ हो ($m_{\text{D}} = 2m_{\text{H}}$, रोधिका अपरिवर्तित)। (घ) कोई काइरल अणु वर्षों तक वाम-हस्त क्यों रह
सकता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:11}
\emph{एसटीएम।} $\Phi = \qty{4.5}{eV}$, $d = \qty{0.5}{nm}$, अभिनति \qty{0.1}{V},
$I = \qty{1}{nA}$। (क) $\kappa$, प्रति \qty{0.1}{nm} $I$ का गुणक, और
ऊँचाई विभेदन यदि $I$ $1\%$ तक मापा जाए। (ख) धारा में प्रति सेकंड
इलेक्ट्रॉन। (ग) नोक का शीर्ष परमाणु अपने पड़ोसियों से \qty{0.25}{nm} अधिक
पास है (पार्श्वतः \qty{0.25}{nm} दूर, अतः $d' = \sqrt{d^2 + 0.25^2}$):
वह धारा का कितना अंश ढोता है। (घ) \qty{1}{cm} का कोई इस्पात ढाँचा
$\alpha L\Delta T$ से फैलता है, जिसमें $\alpha = \qty{1e-5}{K^{-1}}$: किसी पिकोमीटर को कैसा तापीय
स्थायित्व चाहिए, और अभिकल्प उससे कैसे पार पाते हैं?
\end{exercise}

\emph{कोई क्वांटम-कूप \omterm{def:b2:laser:cavity}{लेज़र}।} \qty{0.3}{eV} ऊँची रोधिकाओं के बीच GaAs (अंतराल

\section{समस्या: एसटीएम, अल्फ़ा कण और अमोनिया मेज़र}

\begin{problem}[{सप्ताहांत समस्या --- तीन सुरंगें}]\label{pb:b2:potential-wells-tunneling:1}
आँकड़े: $\hbar = \qty{1.05e-34}{J.s}$, $h = \qty{6.63e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $m_\alpha = \qty{6.64e-27}{kg}$, $e^2/4\pi\varepsilon_0 =
\qty{1.44}{MeV.fm}$, \qty{1}{eV} $= \qty{1.60e-19}{J}$।

\textbf{भाग I --- क्रमवीक्षी सुरंगन सूक्ष्मदर्शी।}
नोक और नमूना टंगस्टन के हैं ($\Phi = \qty{4.5}{eV}$); अंतराल $d$; अभिनति
\qty{50}{mV}; धारा $I = I_0\eu^{-2\kappa d}$ है।
\begin{enumerate}
  \item $\kappa$ उन इलेक्ट्रॉनों के लिए जो फ़र्मी स्तर पर निर्वात
        रोधिका $\Phi$ का सामना करें।
  \item $I$ के परिवर्तन का गुणक $\Delta d = \qty{0.1}{nm}$ के लिए; और
        \qty{1}{pm} के लिए।
  \item प्रतिपुष्टि $I$ को $1\%$ तक नियत रखती है: ऊँचाई का रव।
  \item $I = \qty{1}{nA}$: प्रति सेकंड इलेक्ट्रॉन; दो इलेक्ट्रॉनों के बीच
        औसत समय, सुरंगन ``काल'' $\sim\hbar/\Phi$ से तुलना करके।
  \item शीर्ष परमाणु और उसके पड़ोसी पार्श्वतः \qty{0.25}{nm} दूर,
        $d = \qty{0.5}{nm}$: शीर्ष परमाणु का हिस्सा; परमाणुओं को प्रतिबिंबित करने के लिए
        एक ही परमाणु क्यों पर्याप्त है।
  \item \qty{0.1}{nm} ऊँचा कोई अधिशोषित परमाणु: नियत-ऊँचाई विधा में
        $I$ का परिवर्तन; नियत-धारा विधा क्यों वरीय है।
  \item \qty{2}{cm} का कोई धातु ढाँचा, $\alpha = \qty{1e-5}{K^{-1}}$: वह तापीय
        परिवर्तन जो नोक को \qty{1}{pm} खिसका दे; उपकरण कैसे
        निपटते हैं (सममिति, गति, कम-प्रसार पदार्थ)?
  \item \qty{1}{V} की अभिनति के साथ रोधिका अब आयताकार नहीं रहती:
        उसका खाका खींचिए और बताइए कि $I$ $V$ के साथ रैखिक से
        तेज़ बढ़ता है या धीमे।
  \item किसी ऑक्सीकृत टुकड़े पर कार्य-फलन \qty{4.5}{eV} के बजाय
        \qty{3}{eV} है: किसी पूर्णतः समतल सतह पर नियत-धारा विधा में
        नोक कितना पीछे हटती है? तब एसटीएम किसका प्रतिबिंब बनाता
        है?
\end{enumerate}

\textbf{भाग II --- किसी नाभिक $Z = 92$, $A = 238$ का अल्फ़ा क्षय।}
वह $\alpha$ ($E = \qty{4.2}{MeV}$) त्रिज्या $R_{\text{n}} = \qty{8}{fm}$ के किसी
नाभिक में चलता है; बाहर, कूलॉम ऊर्जा $V(r) =
2(Z - 2)e^2/4\pi\varepsilon_0r$ है।
\begin{enumerate}[resume]
  \item $R_{\text{n}}$ पर रोधिका की ऊँचाई; और चिरसम्मत मोड़-बिंदु
        $b$।
  \item भीतर $\alpha$ की चाल और दीवार पर उसकी टक्करों की
        आवृत्ति।
  \item गैमो चरघातांक $G = (2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b}
        - \sqrt{(R_{\text{n}}/b)(1 - R_{\text{n}}/b)}]$ और $T = \eu^{-G}$।
  \item क्षय दर और अर्ध-आयु; मापे गए $4.5 \times
        10^9$ वर्षों से तुलना कीजिए।
  \item $E = \qty{5.2}{MeV}$ के साथ वही (कोई दूसरा समस्थानिक): अर्ध-आयु;
        और गाइगर--नटाल संवेदिता।
  \item \qty{8}{fm} के बजाय $R_{\text{n}} = \qty{9}{fm}$ लेकर:
        अर्ध-आयु पर प्रभाव; ऐसे आकलनों की परिशुद्धि पर टिप्पणी कीजिए।
  \item उसी नाभिक से किसी प्रोटॉन (आवेश $1$, द्रव्यमान $1/4$) का
        सुरंगन क्यों नहीं देखा जाता, और किसी $^{12}$C
        (आवेश $6$) का भी क्यों नहीं?
  \item वह $\alpha$ पूरे \qty{4.2}{MeV} के साथ उभरता है यद्यपि वह
        किसी \qty{30}{MeV} की दीवार के ``नीचे से गुज़रा'': क्या ऊर्जा संरक्षित है?
\end{enumerate}

\textbf{भाग III --- अमोनिया मेज़र।}
N परमाणु H$_3$ तल से होकर सुरंगन करता है; मूल स्तर
$\Delta = \qty{9.8e-5}{eV}$ से विपाटित है।
\begin{enumerate}[resume]
  \item प्रतिलोमन संक्रमण की आवृत्ति और तरंगदैर्घ्य।
  \item सममित और प्रतिसममित अवस्थाओं को ``N ऊपर'' तथा ``N नीचे''
        के पदों में लिखिए; कौन-सी नीची है, और क्यों?
  \item ``N ऊपर'' तैयार किया गया कोई अणु: $\psi(t)$, समय $t$ पर
        ``नीचे'' की प्रायिकता, और आवर्तकाल।
  \item \qty{300}{K} पर दोनों स्तरों का बोल्ट्ज़मान अनुपात: क्या
        तापीय जनसंख्या कोई मेज़र दे सकती है? प्रतिलोमन कैसे पाया जाता है
        (कोई अवस्था-चयनी वैद्युत क्षेत्र)?
  \item जूल में फोटॉन ऊर्जा; \qty{1e-10}{W} के किसी निर्गम के लिए प्रति सेकंड
        कितने अणुओं को उत्सर्जित करना पड़ेगा।
  \item ND$_3$ \qty{1.6}{GHz} पर सुरंगन करता है: $\kappa a$ निकालिए NH$_3$ के
        लिए, $\Delta \propto \eu^{-\kappa a}$ मानकर $\kappa \propto \sqrt{m}$ के साथ।
  \item मेज़र पहली परमाण्विक घड़ी थी: कोई सुरंगन आवृत्ति
        कोई अच्छी घड़ी क्यों है और कौन-से प्रभाव उसे खिसकाते हैं?
  \item सारांश दीजिए: परिरोध से क्वांटन, क्षयण से सुरंगन,
        और चरघातांक किससे तय होते हैं।
\end{enumerate}
\end{problem}
