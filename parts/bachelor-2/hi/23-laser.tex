\chapter{लेज़र: उद्दीपित उत्सर्जन और गाउसी पुंजें}\label{ch:b2:laser}

किसी व्याख्यान-पर्दे पर कोई लाल बिंदु, दुकान की गल्ले पर छड़ी-संकेत पढ़ता
यंत्र, वह पुंज जो कोई चक्रिका पढ़ती है, किसी गाड़ी का ढाँचा वेल्ड करती है,
महासागरों के नीचे इंटरनेट ले चलती है, किसी स्वच्छपटल को सुधारती है और
चंद्रमा तक की दूरी मिलीमीटर तक नापती है: इनमें से हर एक कोई \emph{\omterm{def:b2:laser:cavity}{लेज़र}}
है, प्रकाश का ऐसा स्रोत जो किसी लौ या दीपक जैसा नहीं। उसका प्रकाश अरब में
एक भाग तक एक ही रंग का है, किसी कमरे भर मिलीमीटरों की पुंज में बना रहता है,
और कुछ तरंगदैर्घ्य चौड़े किसी धब्बे पर फ़ोकस किया जा सकता है, जहाँ उसकी
विकिरण-तीव्रता सूर्य की सतह से दस लाख गुना ऊपर चली जाती है। इसमें से कुछ भी
किसी नई तरह के परमाणु से नहीं आता: वह उस प्रक्रिया से आता है जिसे आइंस्टाइन
ने 1917 में पहचाना --- \emph{\omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन}}, अर्थात् किसी ऐसे फ़ोटॉन का
उत्सर्जन जो पहले से मौजूद किसी फ़ोटॉन जैसा ही हो --- और पदार्थ को इस तरह
सजाने से कि यह प्रक्रिया \omterm{def:b2:laser:processes}{अवशोषण} पर भारी पड़े, और फिर प्रवर्धक को दो दर्पणों
के बीच रख देने से कि वह ख़ुद को खिलाता रहे। यह अध्याय \omterm{def:b2:laser:cavity}{लेज़र} को तीन क़दमों में
बनाता है: प्रकाश और परमाणुओं की कोई जनसंख्या ऊर्जा कैसे बदलते हैं (\omterm{def:b2:laser:processes}{आइंस्टाइन
गुणांक}), यह विनिमय कब प्रवर्धन करता है (\omterm{prop:b2:laser:gain}{जनसंख्या प्रतिलोमन} और लब्धि), और
प्रवर्धक कब दोलक बन जाता है (कोटर, उसकी देहली और उसकी विधाएँ)। और अंत में
वह जो पुंज निकलती है उसका रूप, \omterm{prop:b2:laser:gaussian}{गाउसी पुंज}, जिसकी \omterm{prop:b2:laser:gaussian}{कटि} और \omterm{def:b2:maxwell-equations:operators}{अपसरण} तय करते हैं कि
कोई \omterm{def:b2:laser:cavity}{लेज़र} दूरी पर और किसी फ़ोकस पर क्या कर सकता है।

\section{प्रकाश और पदार्थ: तीन प्रक्रियाएँ}

\begin{definition}[अवशोषण, स्वतःस्फूर्त और उद्दीपित उत्सर्जन]\label{def:b2:laser:processes}
दो ऊर्जा-स्तरों $E_1 < E_2$, जनसंख्याओं (संख्या
घनत्वों) $N_1$, $N_2$ वाले परमाणु लीजिए, और आवृत्ति $\nu$ का प्रकाश जिसमें
$h\nu = E_2 - E_1$ और वर्णक्रमी ऊर्जा घनत्व $u(\nu)$ (प्रति एकांक
आयतन प्रति एकांक आवृत्ति ऊर्जा)। तीन प्रक्रियाएँ ऊर्जा बदलती हैं:
\begin{itemize}
\item \emph{अवशोषण}\index{अवशोषण (प्रकाश का)}: $E_1$ में कोई परमाणु
      कोई फ़ोटॉन सोखकर $E_2$ पर चढ़ जाता है, प्रति एकांक आयतन और समय
      दर $B_{12}\,u(\nu)\,N_1$ से;
\item \emph{स्वतःस्फूर्त उत्सर्जन}\index{स्वतःस्फूर्त उत्सर्जन}: $E_2$ में
      कोई परमाणु अपने आप $E_1$ पर गिर जाता है, और किसी यादृच्छिक
      दिशा में यादृच्छिक कला वाला कोई फ़ोटॉन उत्सर्जित करता है, दर $A_{21}N_2$ से ---
      और $1/A_{21} = \tau$ उस स्तर का जीवनकाल है;
\item \emph{उद्दीपित उत्सर्जन}\index{उद्दीपित उत्सर्जन}: आवृत्ति $\nu$
      का कोई फ़ोटॉन $E_2$ में किसी परमाणु के पास से गुज़रकर उसे गिरने को
      उकसा देता है, और वह पहले जैसा ही \emph{एक-सा} दूसरा फ़ोटॉन उत्सर्जित करता है (वही
      आवृत्ति, दिशा, कला और ध्रुवण), दर
      $B_{21}\,u(\nu)\,N_2$ से।
\end{itemize}
$A_{21}$, $B_{12}$, $B_{21}$ उस संक्रमण के \emph{आइंस्टाइन
गुणांक}\index{आइंस्टाइन गुणांक} हैं।
\end{definition}

\begin{proposition}[आइंस्टाइन संबंध]\label{prop:b2:laser:einstein}
अपभ्रष्ट न हुए स्तरों के लिए,
\[
B_{12} = B_{21} \equiv B, \qquad
\frac{A_{21}}{B_{21}} = \frac{8\pi h\nu^3}{c^3}.
\]
\omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन} और \omterm{def:b2:laser:processes}{अवशोषण} का गुणांक वही है; और
स्वतःस्फूर्त दर उद्दीपित दर के सापेक्ष $\nu^3$ की तरह बढ़ती है।
\end{proposition}

\begin{proof}
परमाणुओं को ताप $T$ पर विकिरण के साथ साम्य में रखिए। दो
तथ्य आगे के अध्यायों से उधार लिए गए हैं: साम्य पर दो स्तरों की जनसंख्याएँ
$N_2/N_1 = \eu^{-h\nu/k_BT}$ के अनुपात में होती हैं (बोल्ट्समान
गुणक, \cref{ch:b2:boltzmann-factor}), और साम्य के
विकिरण का प्लांक वर्णक्रमी घनत्व
$u(\nu) = (8\pi h\nu^3/c^3)/(\eu^{h\nu/k_BT} - 1)$ है
(\cref{ch:b2:thermal-radiation})। साम्य पर स्तरों की जनसंख्याएँ
स्थिर रहती हैं: $B_{12}uN_1 = A_{21}N_2 + B_{21}uN_2$, अतः
\[
u = \frac{A_{21}}{B_{12}\,N_1/N_2 - B_{21}}
  = \frac{A_{21}}{B_{12}\eu^{h\nu/k_BT} - B_{21}}.
\]
यह हर $T$ पर प्लांक के व्यंजक के बराबर होना चाहिए: $T \to \infty$
सीमा (जहाँ $u \to \infty$) $B_{12} = B_{21}$ थोप देती है, और
फिर तुलना $A_{21}/B_{21} = 8\pi h\nu^3/c^3$ दे देती है। ये
गुणांक परमाणु के गुण हैं, अतः ये संबंध साम्य के बाहर भी
टिके रहते हैं।
\end{proof}

\begin{remark}[दीपक प्रवर्धन क्यों नहीं करते]\label{rem:b2:laser:lamps}
\qty{300}{K} पर किसी गैस में, किसी दृश्य संक्रमण के लिए ($h\nu \approx
\qty{2}{eV}$, $k_BT \approx \qty{0.025}{eV}$), $N_2/N_1 = \eu^{-80} \approx
10^{-35}$: अर्थात् कोई परमाणु उत्तेजित है ही नहीं, और प्रकाश केवल सोखा जाता है। कोई
विसर्जन या कोई लौ $E_2$ को भर देती है, पर फिर भी $N_2 < N_1$ के साथ, और
हर उत्सर्जित फ़ोटॉन अवशोषणों से गिनती में हार जाता है; इसके अलावा साम्य पर
स्वतःस्फूर्त और \omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन} का अनुपात,
$A_{21}/(B_{21}u) = \eu^{h\nu/k_BT} - 1$, दृश्य में खगोलीय रूप से बड़ा है ---
अर्थात् किसी दीपक का प्रकाश स्वतःस्फूर्त है: यादृच्छिक कलाएँ, सारी
दिशाएँ, और रेखा की पूरी चौड़ाई। केवल रेडियो और सूक्ष्मतरंग
आवृत्तियों पर ($h\nu \ll k_BT$) \omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन} स्वाभाविक रूप से टक्कर लेता है;
और इस तरह का पहला यंत्र सचमुच मेसर था, \qty{24}{GHz} पर (1954), प्रकाशिक \omterm{def:b2:laser:cavity}{लेज़र} (1960) से पहले।
\end{remark}

\section{प्रवर्धन: जनसंख्या प्रतिलोमन और लब्धि}

\begin{proposition}[किसी माध्यम की लब्धि]\label{prop:b2:laser:gain}
तीव्रता $I$ की कोई पुंज, आवृत्ति $\nu$ पर, माध्यम को पार करती हुई
\omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन} से ऊर्जा पाती है और \omterm{def:b2:laser:processes}{अवशोषण} से खोती है
(\omterm{def:b2:laser:processes}{स्वतःस्फूर्त उत्सर्जन} सारी दिशाओं में जाता है और पुंज में मुश्किल से कुछ
जोड़ता है)। लंबाई $\dd z$ भर,
\[
\dd I = \sigma\,(N_2 - N_1)\,I\,\dd z, \qquad
I(z) = I(0)\,\eu^{g z}, \qquad g = \sigma(N_2 - N_1),
\]
जहाँ $\sigma$ (कोई क्षेत्रफल, अर्थात् संक्रमण का \emph{अनुप्रस्थ काट}\index{अनुप्रस्थ काट
(किसी संक्रमण का)}, $\sigma \propto B\,h\nu/c$) संक्रमण की
प्रबलता नापता है। माध्यम \emph{प्रवर्धन} करता है ($g > 0$)
तभी जब
\[
N_2 > N_1 :
\]
अर्थात् कोई \emph{जनसंख्या प्रतिलोमन}\index{जनसंख्या प्रतिलोमन}। उसके बिना
($N_2 < N_1$, अर्थात् साम्य पर हर माध्यम का हाल) पुंज
सोख ली जाती है।
\end{proposition}

\begin{proof}
प्रति एकांक आयतन और समय, $B u (N_2 - N_1)$ संक्रमण पुंज में कोई फ़ोटॉन $h\nu$
जोड़ (या हटा) देते हैं; और $u = I/c$ के साथ (प्रति एकांक आवृत्ति, रेखा की
चौड़ाई भर), प्रति एकांक आयतन पाई गई शक्ति
$B h\nu (N_2 - N_1) I/c$ है, और यही $\dd I/\dd z$ है; $\sigma =
Bh\nu/c$ परिभाषित कीजिए (रेखा-आकृति के गुणक सहित)।
\end{proof}

\begin{remark}[दो स्तर प्रतिलोमित क्यों नहीं हो सकते]\label{rem:b2:laser:twolevel}
किसी दो-स्तरीय तंत्र को $\nu$ पर प्रकाश से जितना चाहे पंप कीजिए:
पंप $N_1 > N_2$ होने पर सोखता है और $N_2 > N_1$ होने पर उत्सर्जन उकसाता है,
और जनसंख्याएँ अधिक-से-अधिक बराबरी की ओर जाती हैं, $N_2 = N_1$, जहाँ
माध्यम पारदर्शी है पर कुछ भी लब्धि नहीं देता। किसी प्रतिलोमन को कम-से-कम
तीन स्तर चाहिए: भू-स्तर से किसी अल्पजीवी स्तर तक पंप कीजिए, जो
तेज़ी से ऊपरी \omterm{def:b2:laser:cavity}{लेज़र} स्तर में क्षय हो जाए, और वह \emph{दीर्घजीवी}
(अर्ध-स्थायी) है और इसलिए जनसंख्या संचित कर लेता है। किसी
\emph{तीन-स्तरीय}\index{तीन-स्तरीय लेज़र} योजना (माणिक, पहला
\omterm{def:b2:laser:cavity}{लेज़र}) में निचला \omterm{def:b2:laser:cavity}{लेज़र} स्तर भू-अवस्था है, अतः प्रतिलोमन से पहले आधे से अधिक
परमाणु उठाने पड़ते हैं: पंप निर्मम होता है। किसी
\emph{चार-स्तरीय}\index{चार-स्तरीय लेज़र} योजना (हीलियम--नियोन, Nd:YAG, अधिकतर
\omterm{def:b2:laser:cavity}{लेज़र}) में निचला \omterm{def:b2:laser:cavity}{लेज़र} स्तर कोई उत्तेजित स्तर है जो तेज़ी से
भू-अवस्था में ख़ाली हो जाता है; वह हर समय लगभग ख़ाली रहता है, और ऊपरी स्तर में कोई छोटी
जनसंख्या भी पहले से कोई प्रतिलोमन है --- अतः देहली
नीची रहती है।
\end{remark}

\begin{proposition}[लब्धि की संतृप्ति]\label{prop:b2:laser:saturation}
प्रतिलोमन को कोई पंप बनाए रखता है, जो प्रति एकांक आयतन दर $R$ से ऊपरी
स्तर को परमाणु देता है, क्षय $1/\tau$ और
\omterm{def:b2:laser:processes}{उद्दीपित उत्सर्जन} $\sigma I (N_2 - N_1)/h\nu$ के सामने। किसी चार-स्तरीय
माध्यम ($N_1 \approx 0$) में स्थायी अवस्था है
\[
N_2 = \frac{R\tau}{1 + I/I_{\text{संत}}}, \qquad
g = \frac{g_0}{1 + I/I_{\text{संत}}}, \qquad
I_{\text{संत}} = \frac{h\nu}{\sigma\tau} :
\]
अर्थात् \emph{लघु-संकेत लब्धि} $g_0 = \sigma R\tau$ को ख़ुद पुंज ही घटा देती है
जब $I$ \emph{संतृप्ति
तीव्रता}\index{संतृप्ति तीव्रता} $I_{\text{संत}}$ के पास पहुँचे --- अर्थात् प्रवर्धक
अरैखिक है, और यही अरैखिकता \omterm{def:b2:laser:cavity}{लेज़र} की शक्ति तय
करेगी।
\end{proposition}

\begin{proof}
$\dd N_2/\dd t = R - N_2/\tau - \sigma I N_2/h\nu = 0$.
\end{proof}

\section{दोलक: कोटर, देहली और विधाएँ}

\begin{definition}[लेज़र कोटर]\label{def:b2:laser:cavity}
कोई \emph{लेज़र}\index{लेज़र} लंबाई $\ell$ का कोई प्रवर्धक माध्यम है
जो दो दर्पणों (परावर्तकताएँ $R_1$, $R_2$) के बीच $L$ दूरी पर
रखा हो: अर्थात् कोई फ़ाब्री--पेरो कोटर (\cref{ch:b2:gratings})। एक दर्पण,
\emph{निर्गम युग्मक}, कुछ प्रतिशत तक पारगामी है ($T_2 = 1 - R_2$)
और पुंज को बाहर जाने देता है। कोटर का चक्कर लगाता प्रकाश
माध्यम से दो बार प्रवर्धित होता है और दर्पणों तथा
बाक़ी हानियों (प्रकीर्णन, \omterm{def:b2:laser:processes}{अवशोषण}, \omterm{def:b2:diffraction:huygens}{विवर्तन}) से क्षीण।
\end{definition}

\begin{theorem}[दोलन की शर्त]\label{thm:b2:laser:threshold}
सम्मिश्र आयाम $\underline s$ की कोई तरंग एक चक्कर के बाद ख़ुद को फिर बना
लेती है यदि
\[
\underline s\,\cdot\, \eu^{g\ell}\sqrt{R_1}\,\eu^{g\ell}\sqrt{R_2}\,
\eu^{-\alpha\cdot 2L}\,\eu^{2\iu kL} = \underline s
\]
($\alpha$: प्रति एकांक लंबाई बाक़ी हानियाँ)। इससे दो शर्तें।
\begin{itemize}
\item \emph{आयाम}: $R_1R_2\,\eu^{2(g\ell - \alpha L)} \ge 1$, अर्थात्
      लब्धि को \emph{देहली}\index{देहली (लेज़र)} तक पहुँचना चाहिए
      \[
      g_{\text{देहली}} = \frac{1}{\ell}\Big(\alpha L - \tfrac12\ln(R_1R_2)\Big)
      \approx \frac{1}{2\ell}\,(T_2 + \text{प्रति चक्कर हानियाँ}) ;
      \]
\item \emph{कला}: $2kL = 2\pi p$, अर्थात् $\nu_p = p\,c/2L$ --- \omterm{def:b2:laser:cavity}{लेज़र}
      केवल कोटर की \emph{अनुदैर्घ्य विधाओं}\index{अनुदैर्घ्य
      विधाएँ} पर दोलन करता है, जिनकी दूरी $c/2L$ है, और उनमें भी केवल
      उन पर जो लब्धि-वक्र के नीचे पड़ें जहाँ $g(\nu) \ge g_{\text{देहली}}$ हो।
\end{itemize}
\end{theorem}

\begin{proof}
चक्कर के गुणक का आयाम और कला दोनों तुच्छ होने चाहिए; और
सन्निकटन $-\ln R \approx 1 - R = T$ काम में लेता है, $T \ll 1$ के लिए।
\end{proof}

\begin{proposition}[देहली के ऊपर स्थायी अवस्था]\label{prop:b2:laser:steady}
किसी एक परमाणु के \omterm{def:b2:laser:processes}{स्वतःस्फूर्त उत्सर्जन} से शुरू होकर, हर वह विधा जिसके लिए
$g_0 > g_{\text{देहली}}$ हो, चक्कर-दर-चक्कर चरघातांकी रूप से बढ़ती है;
और जैसे-जैसे उसकी तीव्रता बढ़ती है, लब्धि संतृप्त होती जाती है
(\cref{prop:b2:laser:saturation}) जब तक वह ठीक हानियों के बराबर न हो जाए:
\[
g(I) = g_{\text{देहली}} \quad\Longrightarrow\quad
I_{\text{कोटर}} = I_{\text{संत}}\Big(\frac{g_0}{g_{\text{देहली}}} - 1\Big),
\qquad P_{\text{निर्गम}} = T_2\,I_{\text{कोटर}}\,A .
\]
लब्धि देहली के मान पर \emph{जकड़} जाती है; और देहली के ऊपर पंप किया गया हर
अतिरिक्त परमाणु कोई निर्गम फ़ोटॉन बन जाता है, अतः निर्गम शक्ति
देहली-पंप के ऊपर पंप के साथ रैखिक रूप से बढ़ती है।
\end{proposition}

\begin{remark}[लेज़र कोई पुनर्भरण दोलक है]\label{rem:b2:laser:feedback}
\cref{ch:b2:feedback-oscillators} के वीन-सेतु दोलक से
तुलना कीजिए: कोई प्रवर्धक (प्रतिलोमित माध्यम),
कोई आवृत्ति-वरणात्मक \omterm{def:b2:feedback-oscillators:loop}{पुनर्भरण} (कोटर, जो केवल अपनी विधाएँ जाने देता है),
कोई शुरू होने की शर्त (\omterm{def:b2:feedback-oscillators:loop}{पाश-लब्धि} $> 1$, अर्थात् $g_0 > g_{\text{देहली}}$),
रव से शुरुआत (यहाँ \omterm{def:b2:laser:processes}{स्वतःस्फूर्त उत्सर्जन}), और कोई ऐसा आयाम जो
प्रवर्धक की अरैखिकता (लब्धि की संतृप्ति) तय करती है। \omterm{def:b2:laser:cavity}{लेज़र} इसी
कुल का प्रकाशिक सदस्य है।
\end{remark}

\section{गाउसी पुंज}

\begin{proposition}[गाउसी पुंज]\label{prop:b2:laser:gaussian}
मुड़े दर्पणों वाला कोई स्थायी कोटर जो पुंज उत्सर्जित करता है वह (मूल
अनुप्रस्थ विधा में) कोई \emph{गाउसी पुंज}\index{गाउसी
पुंज} है: दूरी $z$ पर, उसके सबसे सँकरे परिच्छेद (\emph{कटि}\index{कटि (किसी पुंज की)},
त्रिज्या $w_0$) से, तीव्रता का
परिच्छेद है
\[
I(r,z) = \frac{2P}{\pi w(z)^2}\,\exp\Big(-\frac{2r^2}{w(z)^2}\Big),
\qquad
w(z) = w_0\sqrt{1 + \Big(\frac{z}{z_{\text{R}}}\Big)^2},
\qquad
z_{\text{R}} = \frac{\pi w_0^2}{\lambda},
\]
जहाँ $P$ कुल शक्ति है और $z_{\text{R}}$ \emph{रैले
लंबाई}\index{रैले लंबाई}: $\pm z_{\text{R}}$ भर पुंज अपनी \omterm{prop:b2:laser:gaussian}{कटि} के
$\sqrt2$ के भीतर रहती है, और उससे बहुत परे इस अर्ध-कोण से
फैलती है
\[
\theta = \frac{\lambda}{\pi w_0} .
\]
पुंज परअक्षीय सन्निकटन में तरंग समीकरण का कोई
हल है; हम उसका रूप स्वीकार करके काम में लेते हैं। उसके \omterm{def:b2:scalar-light-model:path}{तरंगाग्र} \omterm{prop:b2:laser:gaussian}{कटि} पर समतल
और दूर गोलीय होते हैं।
\end{proposition}

\begin{proof}
इस स्तर पर स्वीकृत (गणना गाउसी द्वारक-वितरण का फ़ूरिये-प्रकाशिकी वाला
संचरण है); और उसका मुख्य गुण वही है जो
\cref{ch:b2:diffraction} में \omterm{def:b2:diffraction:huygens}{विवर्तन} पहले ही दे चुका है: आकार $w_0$ का कोई
द्वारक $\lambda/w_0$ भर फैलता है, यहाँ ठीक-ठीक
गुणक $1/\pi$ के साथ।
\end{proof}

\begin{proposition}[किसी पुंज को फ़ोकस और फैलाना]\label{prop:b2:laser:focus}
त्रिज्या $w$ की कोई \omterm{prop:b2:laser:gaussian}{गाउसी पुंज} (जो लेंस पर अपने दूर-क्षेत्र से कहीं बड़ी
हो, अर्थात् लगभग संरेखित) फ़ोकस दूरी
$f$ के किसी लेंस पर पड़कर फ़ोकस पर इस \omterm{prop:b2:laser:gaussian}{कटि} तक फ़ोकस होती है
\[
w_0' = \frac{\lambda f}{\pi w}
\]
जिसकी \omterm{prop:b2:laser:gaussian}{रैले लंबाई} $\pi w_0'^2/\lambda$ है। वहाँ शिखर
विकिरण-तीव्रता $2P/\pi w_0'^2$ है। और उलटे, आवर्धन $M$ का कोई अफ़ोकल दूरदर्शी
त्रिज्या $w$ की किसी पुंज को त्रिज्या $Mw$ की पुंज में बदल देता है,
जिसका \omterm{def:b2:maxwell-equations:operators}{अपसरण} $M$ गुना छोटा है: अर्थात् किसी पुंज को दूर भेजने के लिए पहले उसे
चौड़ा कीजिए।
\end{proposition}

\begin{proof}
लेंस समतल \omterm{def:b2:scalar-light-model:path}{तरंगाग्र} को $f$ पर अभिसरित होते किसी गोले में बदल देता है;
और त्रिज्या $w$ के किसी द्वारक का \omterm{def:b2:diffraction:huygens}{विवर्तन} कोणीय फैलाव
$\lambda/\pi w$ देता है, अतः फ़ोकस-धब्बा $f\lambda/\pi w$ --- जो
$\theta = \lambda/\pi w_0'$ को उलटा पढ़ने से मेल खाता है।
\end{proof}

\begin{example}[काटना, पढ़ना, ताकना]\label{ex:b2:laser:focus}
\qty{10.6}{\micro m} पर कोई \qty{1}{kW} कार्बन-डाइऑक्साइड \omterm{def:b2:laser:cavity}{लेज़र}, पुंज-त्रिज्या
\qty{10}{mm}, $f = \qty{100}{mm}$ से फ़ोकस: $w_0' = \qty{34}{\micro m}$,
शिखर विकिरण-तीव्रता $2P/\pi w_0'^2 = \qty{5e11}{W/m^2}$ --- इस्पात उबल जाता है।
कोई \qty{1}{mW} संकेतक, \qty{633}{nm} पर $w_0 = \qty{0.5}{mm}$: $\theta =
\qty{0.4}{mrad}$, \qty{50}{m} पर \qty{4}{cm} का कोई धब्बा, और निर्गम पर
\qty{2.5}{kW/m^2} की विकिरण-तीव्रता --- धूप से भी ऊपर, और इसीलिए कोई
मिलीवाट भी कभी आँख में नहीं जाना चाहिए: आँख का लेंस उसे रेटिना पर
\qty{10}{\micro m} के किसी धब्बे पर, लाखों वाट प्रति वर्ग
मीटर पर, फ़ोकस कर देता। \qty{650}{nm} पर कोई चक्रिका-पाठक \omterm{prop:b2:guided-waves:fibre}{संख्यात्मक द्वारक}
$0.6$ के किसी लेंस से लगभग $\lambda/2\mathrm{NA} \approx \qty{0.5}{\micro m}$ पर फ़ोकस करता है,
अर्थात् किसी गड्ढे जितना।
\end{example}

\begin{remark}[लेज़र-प्रकाश ख़ास क्यों है]\label{rem:b2:laser:special}
\emph{दिशिकता}: एक ही अनुप्रस्थ विधा, और \omterm{def:b2:diffraction:huygens}{विवर्तन} की सीमा पर \omterm{def:b2:maxwell-equations:operators}{अपसरण}।
\emph{एकवर्णता}: एक या कुछ ही कोटर-विधाएँ, हर एक
किसी मेगाहर्ट्ज़ से भी सँकरी --- अर्थात् मीटरों से किलोमीटरों की संसक्ति लंबाइयाँ
(\cref{ch:b2:scalar-light-model})। \emph{\omterm{prop:b2:two-wave-interference:spatial}{दैशिक संसक्ति}}: पूरी
पुंज एक ही तरंग है, अतः वह कहीं भी अपने साथ व्यतिकरण करती है (होलोग्राफ़ी,
व्यतिकरणमिति)। \emph{चमक}: विवर्तन-सीमित किसी पुंज में कोई मिलीवाट
प्रति एकांक घन कोण और पट्ट-चौड़ाई सूर्य को मात दे देता है। स्पंदों में
\emph{शक्ति}: किसी नैनोसेकंड में कोई जूल कोई गीगावाट है, और किसी फ़ेम्टोसेकंड में कोई
पेटावाट। और चूँकि पुंज किसी दोलक का निर्गम है, उसे गीगाहर्ट्ज़ पर
माडुलित किया जा सकता है --- वही \omterm{prop:b2:guided-waves:fibre}{प्रकाशिक तंतु} का वाहक
(\cref{ch:b2:guided-waves})।
\end{remark}

\begin{method}[लेज़र के आकलन]\label{meth:b2:laser:method}
(1) फ़ोटॉन ऊर्जा $h\nu = hc/\lambda$ और फ़ोटॉन दर $P/h\nu$।
(2) देहली: $g_{\text{देहली}}\ell \approx \tfrac12(T_2 + \text{हानियाँ})$;
माध्यम की लघु-संकेत लब्धि $g_0\ell$ से तुलना कीजिए।
(3) विधाएँ: लब्धि की चौड़ाई के सामने $c/2L$; दोलन करती विधाएँ गिनिए।
(4) शक्ति: लब्धि देहली पर जकड़ जाती है; निर्गम $\propto$ (पंप $-$
देहली-पंप)। (5) पुंज: $z_{\text{R}} = \pi w_0^2/\lambda$, $\theta =
\lambda/\pi w_0$, फ़ोकस-धब्बा $\lambda f/\pi w$, शिखर विकिरण-तीव्रता
$2P/\pi w_0^2$। (6) सुरक्षा: रेटिना की विकिरण-तीव्रता की सूर्य वाली से
तुलना कीजिए।
\end{method}

\begin{problem}[{सप्ताहांत समस्या --- कोई लेज़र बनाया, परखा और

  \item \omterm{def:b2:laser:cavity}{लेज़र} रेखा की फ़ोटॉन ऊर्जा और आवृत्ति।
  \item He और Ne के स्तर \qty{0.05}{eV} से भिन्न हैं: $k_BT$ से
        तुलना कीजिए और समझाइए कि टक्कर से He $\to$ Ne स्थानांतरण
        क्यों दक्ष है।
  \item बिना विसर्जन के \qty{400}{K} पर Ne के ऊपरी स्तर का भू-अवस्था से
        बोल्ट्समान अनुपात: टिप्पणी कीजिए।
  \item दोनों जीवनकालों से समझाइए कि ऊपरी स्तर के भरते ही Ne के
        स्तरों के बीच प्रतिलोमन बनाए रखना आसान क्यों है।
  \item रेखा की डॉप्लर चौड़ाई: \qty{400}{K} पर नियोन की सबसे संभावित
        चाल और कोटि में $\Delta\nu \approx \nu\,v/c$
        (ठीक-ठीक $\Delta\nu = 2\nu\sqrt{2k_BT\ln2/Mc^2}$)।
  \item $\Delta N = N_2 - N_1 = \qty{3e15}{m^{-3}}$ के साथ: लघु-संकेत
        लब्धि गुणांक और \qty{30}{cm} भर प्रति चक्कर लब्धि।
\end{enumerate}

\textbf{भाग II --- कोटर।}
\begin{enumerate}[resume]
  \item देहली-लब्धि गुणांक; और प्रश्न 6 के सापेक्ष गुंजाइश।
  \item विधाओं की दूरी; और कितनी विधाएँ दोलन कर सकती हैं।
  \item कोटर के भीतर शक्ति और नली में कोटर के भीतर विकिरण-तीव्रता।
  \item स्थायी अवस्था में लब्धि को क्या तय करता है, और अतिरिक्त
        पंप ऊर्जा कहाँ जाती है?
  \item भीतर विद्युत् शक्ति और बाहर प्रकाशिक शक्ति: दक्षता; और वह इतनी
        कम क्यों है, इसके दो कारण।
  \item शुरुआत: प्रति चक्कर शुद्ध लब्धि $2(g_0 - g_{\text{देहली}})L$,
        स्थायी अवस्था में कोटर में फ़ोटॉनों की संख्या, किसी एक फ़ोटॉन से
        चक्करों की संख्या, और शुरू होने का समय।
  \item नली गरम होकर \qty{1}{\micro m} लंबी हो जाती है: किसी विधा का
        आवृत्ति-खिसकाव; और किसी एक-विधा \omterm{def:b2:laser:cavity}{लेज़र} के लिए परिणाम।
  \item नली \omterm{prop:b2:wave-interfaces:brewster}{ब्रूस्टर कोण} पर रखी खिड़कियों से बंद है: इससे निर्गम के
        ध्रुवण का क्या होता है, और इससे कोई हानि क्यों नहीं
        जुड़ती?
\end{enumerate}

\textbf{भाग IV --- चंद्रमा तक।}
कोई परास-केंद्र \qty{532}{nm} पर \qty{100}{ps} की \qty{100}{mJ} स्पंदें
\qty{1}{m} के किसी दूरदर्शी से दाग़ता है; और वायुमंडल बाहर जाती पुंज को
\qty{1}{''} ($\qty{4.8e-6}{rad}$) तक फैला देता है।
\begin{enumerate}[resume]
  \item प्रति स्पंद फ़ोटॉन और शिखर शक्ति।
  \item चंद्रमा (\qty{3.8e8}{m}) पर धब्बे की त्रिज्या;
        विवर्तन-सीमित धब्बे से तुलना कीजिए।
  \item \qty{0.1}{m^2} की कोई परावर्तक पंक्ति स्पंद का कौन-सा अंश
        रोक लेती है, और कितने फ़ोटॉन?
  \item पंक्ति के \qty{4}{cm} के कोने-घन प्रकाश को फैलाव

        अंश, और प्रति स्पंद भाँपे गए फ़ोटॉन (कुल प्रकाशिक

        प्रति स्पंद दूरी की परिशुद्धता; $10^4$ लौटावों के
        बाद; और चंद्रमा जिस \qty{3.8}{cm} प्रति वर्ष से दूर जा रहा है उससे तुलना कीजिए।
