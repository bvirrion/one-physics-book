\chapter{ऊष्मागतिकीय विभव}\label{ch:b2:thermodynamic-potentials}

रबर का कोई फ़ीता तेज़ी से खींचिए और उसे अपने होंठ से छुआइए: वह गरम है;
उसे सिकुड़ने दीजिए और वह ठंडा है। उस पर कोई भार लटकाइए और उसे किसी
बाल-सुखाने वाले से गरम कीजिए: वह \emph{छोटा हो जाता है}। इस्पात का कोई
स्प्रिंग ऐसा नहीं बरतता, और कारण यह है कि रबर के फ़ीते का तनाव ऊर्जा की
नहीं बल्कि एंट्रॉपी की बात है --- खिंची हुई शृंखलाओं के पास ख़ुद को सजाने
के कम तरीक़े बचते हैं। ऐसी चीज़ों पर तर्क करने के लिए जानना पड़ता है कि नियत
ताप पर, या नियत ताप और दाब पर, कोई तंत्र क्या करना चाहता है, और
उससे कितना कार्य खींचा जा सकता है: और इन प्रश्नों का उत्तर आंतरिक ऊर्जा तथा
एंट्रॉपी बड़ी भद्दी तरह देते हैं, क्योंकि वे किसी विलगित तंत्र की भाषा
बोलते हैं। यह अध्याय प्रयोगशाला की शर्तों के लिए बने वे दो फलन लाता है,
\emph{हेल्महोल्ट्स \omterm{def:b2:thermodynamic-potentials:FG}{मुक्त ऊर्जा}} $F = U -
TS$ और \emph{गिब्स \omterm{def:b2:thermodynamic-potentials:FG}{मुक्त एंथैल्पी}} $G = H - TS$, दिखाता है कि वे
वापस पाए जा सकने वाले कार्य को नापते हैं और साम्य की ओर घटते हैं, उनके
अवकलों से वे मैक्सवेल संबंध निकालता है जो ऐसी राशियाँ जोड़ देते हैं जिन्हें
कोई जुड़ी हुई सोचता भी नहीं, और यह सब दो प्रावस्थाओं के
साम्य पर लगाता है, जहाँ $G$ तय करता है कि कौन-सी प्रावस्था जीतती है और
क्लापेरों का सूत्र लौटा देता है।

\section{मूलभूत सर्वसमिका और उसके चर}

\begin{proposition}[मूलभूत सर्वसमिका]\label{prop:b2:thermodynamic-potentials:identity}
नियत संघटन के किसी बंद तंत्र के लिए, आंतरिक ऊर्जा,
जिसे एंट्रॉपी और आयतन का कोई फलन माना जाए, यह मानती है
\[
\dd U = T\,\dd S - P\,\dd V, \qquad
T = \Big(\frac{\partial U}{\partial S}\Big)_V, \quad
P = -\Big(\frac{\partial U}{\partial V}\Big)_S ,
\]
और \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी} $H = U + PV$ मानती है $\dd H = T\,\dd S + V\,\dd P$।
हर फलन, अपने \emph{स्वाभाविक चरों}\index{स्वाभाविक
चर} में --- $(S,V)$ $U$ के लिए, $(S,P)$ $H$ के लिए --- तंत्र की पूरी
ऊष्मागतिकी समेटे है: उसके आंशिक अवकलज बाक़ी
अवस्था-चर दे देते हैं। दूसरा नियम किसी विलगित तंत्र
(नियत $U$, $V$) के साम्य को अधिकतम $S$ की अवस्था की तरह चुन लेता है।
\end{proposition}

\begin{proof}
किसी उत्क्रमणीय रूपांतरण के लिए पहला नियम, $\dd U = \delta Q_{\text{उत्क्र}}
+ \delta W_{\text{उत्क्र}} = T\,\dd S - P\,\dd V$, केवल अवस्था-फलन
लेता है और इसलिए साम्य अवस्थाओं के बीच किसी भी अत्यल्प बदलाव पर
टिका रहता है। $\dd(PV)$ जोड़िए $H$ के लिए।
\end{proof}

\begin{remark}[नए फलन क्यों]\label{rem:b2:thermodynamic-potentials:why}
प्रयोग नियत एंट्रॉपी पर नहीं किए जाते। किसी रसायनज्ञ की कुप्पी
कमरे के ताप और वायुमंडल के दाब पर बैठी है; और किसी तापस्थापी में कोई
पिस्टन नियत $T$ पर काम करता है। हमें ऐसे फलन चाहिए जिनके
\omterm{prop:b2:thermodynamic-potentials:identity}{स्वाभाविक चर} $(T,V)$ और $(T,P)$ हों, और जो उन
शर्तों के अधीन वही भूमिका निभाएँ जो $S$ किसी विलगित तंत्र के लिए निभाता है: यानी बताएँ
कि तंत्र किस ओर बदलता है और वह कितना कार्य दे सकता है। और कोई
\emph{लजांद्र रूपांतर} --- $TS$ घटाकर $S$ की जगह $T$ को
स्वतंत्र चर बना लेना --- उन्हें बना देता है।
\end{remark}

\section{मुक्त ऊर्जा और मुक्त एंथैल्पी}

\begin{definition}[हेल्महोल्ट्स मुक्त ऊर्जा, गिब्स मुक्त एंथैल्पी]\label{def:b2:thermodynamic-potentials:FG}
\[
F = U - TS \quad\text{(\emph{मुक्त ऊर्जा}\index{मुक्त ऊर्जा})}, \qquad
G = H - TS = U + PV - TS \quad\text{(\emph{मुक्त एंथैल्पी}\index{मुक्त एंथैल्पी})}.
\]
दोनों विस्तीर्ण अवस्था-फलन हैं, जो जूल में नापे जाते हैं।
\end{definition}

\begin{theorem}[अवकल और व्युत्पन्न राशियाँ]\label{thm:b2:thermodynamic-potentials:dFdG}
\[
\dd F = -S\,\dd T - P\,\dd V, \qquad
\dd G = -S\,\dd T + V\,\dd P ;
\]
अतः, \omterm{prop:b2:thermodynamic-potentials:identity}{स्वाभाविक चरों} $(T,V)$ में, जो $F$ के हैं, और $(T,P)$ में, जो $G$ के हैं,
\[
S = -\Big(\frac{\partial F}{\partial T}\Big)_V, \quad
P = -\Big(\frac{\partial F}{\partial V}\Big)_T, \qquad
S = -\Big(\frac{\partial G}{\partial T}\Big)_P, \quad
V = \Big(\frac{\partial G}{\partial P}\Big)_T ,
\]
और (गिब्स--हेल्महोल्ट्स) $U = F - T(\partial F/\partial T)_V =
-T^2\,\partial(F/T)/\partial T$, $H = -T^2\,\partial(G/T)/\partial T$।
$F(T,V)$ या $G(T,P)$ जानना सब कुछ जानना है: अवस्था-समीकरण,
एंट्रॉपी, ऊर्जा, ऊष्मा-धारिताएँ।
\end{theorem}

\begin{proof}
$\dd F = \dd U - T\dd S - S\dd T = -S\dd T - P\dd V$; और वैसे ही $G$ के लिए
$\dd H$ से। आंशिक अवकलज पढ़ लीजिए; और गिब्स--हेल्महोल्ट्स के लिए,
$F - T\,\partial_TF = F + TS = U$ तथा $\partial_T(F/T) = (T\partial_TF -
F)/T^2 = -U/T^2$।
\end{proof}

\begin{example}[पूर्ण गैस]\label{ex:b2:thermodynamic-potentials:perfect}
$n$ मोलों के लिए, $U = nc_{V}T + U_0$ और $S = nc_V\ln T + nR\ln V + S_0$
देते हैं $F(T,V) = -nRT\ln V + f(T)$: तब $P = -\partial_VF = nRT/V$
--- अर्थात् अवस्था-समीकरण $F$ से निकल आता है --- और $S = -\partial_TF$
एंट्रॉपी लौटा देता है। चरों $(T,P)$ में, $G(T,P) = nRT\ln P +
g(T)$, अतः $V = \partial_PG = nRT/P$, और \emph{मोलर} \omterm{def:b2:thermodynamic-potentials:FG}{मुक्त एंथैल्पी}
$\mu(T,P) = \mu^\circ(T) + RT\ln(P/P^\circ)$ --- अर्थात् गैस का \emph{रासायनिक
विभव}\index{रासायनिक विभव}, जो उसके दाब के साथ बढ़ता है: और गैस ऊँचे $\mu$ से
नीचे की ओर बहती है, यहाँ ऊँचे से नीचे दाब की ओर।
\end{example}

\section{अधिकतम कार्य और बदलाव की दिशा}

\begin{theorem}[एकतापी रूपांतरण]\label{thm:b2:thermodynamic-potentials:monothermal}
$T_0$ पर किसी एक ही तापस्थापी से ऊष्मा बदलता कोई तंत्र, जिसका
आरंभिक और अंतिम ताप $T_0$ के बराबर हो, किसी भी रूपांतरण में
यह कार्य पाता है
\[
W \ge \Delta F , \qquad\text{अर्थात्}\qquad W_{\text{वापस}} = -W \le -\Delta F :
\]
अर्थात् $F$ की कमी नियत ताप पर तंत्र से \emph{वापस पाया जा सकने वाला अधिकतम
कार्य} है (जो किसी उत्क्रमणीय रूपांतरण में मिलता है) ---
इसीलिए ``मुक्त'' ऊर्जा: $U$ का वह भाग जो कार्य में बदला जा सके, और
बाक़ी, $TS$, बँधा हुआ। और यदि रूपांतरण एकदाबी भी हो
(बाहर दाब $P_0$, आरंभिक तथा अंतिम दाब $P_0$), तो वायुमंडल के कार्य के
\emph{अलावा} कार्य (\omterm{def:b2:open-systems:cv}{उपयोगी कार्य}, मसलन विद्युत्)
$W_{\text{उप}} \ge \Delta G$ मानता है: अर्थात् $-\Delta G$ नियत $T$ और $P$ पर वापस पाया जा
सकने वाला अधिकतम \omterm{def:b2:open-systems:cv}{उपयोगी कार्य} है (किसी ईंधन-सेल का, किसी
बैटरी का)।
\end{theorem}

\begin{proof}
पहला नियम: $\Delta U = W + Q$। एक तापस्थापी के साथ दूसरा नियम: $\Delta S
= Q/T_0 + S_{\text{सृजित}}$, $S_{\text{सृजित}} \ge 0$, अतः $Q \le T_0\Delta S$। अतः
$W = \Delta U - Q \ge \Delta U - T_0\Delta S = \Delta(U - T_0S) = \Delta F$, दोनों सिरों पर
$T = T_0$ लेकर। और वायुमंडल का कार्य $-P_0\Delta V$
अलग करने पर, $W = W_{\text{उप}} - P_0\Delta V$ तथा $W_{\text{उप}} \ge \Delta U + P_0
\Delta V - T_0\Delta S = \Delta G$।
\end{proof}

\begin{theorem}[बदलाव और साम्य की कसौटियाँ]\label{thm:b2:thermodynamic-potentials:criteria}
नियत $T$ और $V$ पर थामा और कोई कार्य न पाता कोई तंत्र केवल इस तरह
बदल सकता है कि \emph{$F$ घटे}; और उसका साम्य न्यूनतम $F$ की अवस्था है।
नियत $T$ और $P$ पर थामा और कोई \omterm{def:b2:open-systems:cv}{उपयोगी कार्य} न पाता कोई तंत्र
केवल इस तरह बदल सकता है कि \emph{$G$ घटे}; और उसका साम्य
न्यूनतम $G$ की अवस्था है। (और आम तौर पर, $T_0$, $P_0$ के किसी परिवेश के संपर्क में
किसी तंत्र के लिए राशि $U - T_0S + P_0V$ घटती है।)
\end{theorem}

\begin{proof}
पिछले प्रमेय में $W = 0$ (या $W_{\text{उप}} = 0$) लीजिए: $\Delta F
\le 0$ (या $\Delta G \le 0$), और बराबरी किसी उत्क्रमणीय, यानी
साम्य, बदलाव के लिए।
\end{proof}

\begin{example}[समतापी प्रसार, असली और आदर्श]\label{ex:b2:thermodynamic-potentials:expansion}
\qty{300}{K} पर गैस का कोई मोल तापस्थापी के संपर्क में \qty{1}{L} से \qty{10}{L}
तक फैलता है: $\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$। कोई
उत्क्रमणीय प्रसार \qty{5.7}{kJ} वापस देता है; \qty{1}{bar} के वायुमंडल के सामने कोई
प्रसार केवल $P_0\Delta V = \qty{0.9}{kJ}$; और निर्वात में कोई
मुक्त प्रसार, कुछ नहीं। और \qty{5.7}{kJ} वही अधिकतम है जो
गैस तथा उसका तापस्थापी अवस्था के इस बदलाव के लिए दे सकते हैं, मशीन चाहे
कितनी ही चतुर हो।
\end{example}

\section{मैक्सवेल संबंध}

\begin{theorem}[मैक्सवेल संबंध]\label{thm:b2:thermodynamic-potentials:maxwell}
चूँकि $\dd F$ और $\dd G$ यथार्थ अवकल हैं (मिश्रित दूसरे अवकलजों पर
श्वार्ट्स का प्रमेय),
\[
\Big(\frac{\partial S}{\partial V}\Big)_T = \Big(\frac{\partial P}{\partial T}\Big)_V,
\qquad
\Big(\frac{\partial S}{\partial P}\Big)_T = -\Big(\frac{\partial V}{\partial T}\Big)_P .
\]
वे उसे, जो नापा नहीं जा सकता (कि नियत ताप पर एंट्रॉपी आयतन या
दाब के साथ कैसे बदलती है), अवस्था-समीकरण के रास्ते बता देते हैं। दो
परिणाम:
\[
\Big(\frac{\partial U}{\partial V}\Big)_T = T\Big(\frac{\partial P}{\partial T}\Big)_V - P,
\qquad
C_P - C_V = T\Big(\frac{\partial P}{\partial T}\Big)_V\Big(\frac{\partial V}{\partial T}\Big)_P
= \frac{TV\alpha^2}{\kappa_T} ,
\]
जहाँ $\alpha = V^{-1}(\partial V/\partial T)_P$ प्रसार गुणांक है और
$\kappa_T = -V^{-1}(\partial V/\partial P)_T$ समतापी संपीड्यता।
\end{theorem}

\begin{proof}
$\partial^2F/\partial T\partial V$ दोनों क्रमों में निकालिए: $-\partial_VS =
-\partial_TP$। $G$ के लिए वही। फिर $\dd U = T\dd S - P\dd V$ नियत $T$ पर
देता है $(\partial_VU)_T = T(\partial_VS)_T - P$; और $C_P - C_V$
$S(T,V(T,P))$ लिखने से निकलता है: $(\partial_TS)_P = (\partial_TS)_V +
(\partial_VS)_T(\partial_TV)_P$, जिसे $T$ से गुणा कीजिए।
\end{proof}

\begin{example}[संबंध क्या खोल देते हैं]\label{ex:b2:thermodynamic-potentials:reveal}
किसी पूर्ण गैस के लिए, $T(\partial_TP)_V - P = 0$: अर्थात् ऊर्जा
आयतन पर निर्भर नहीं करती (जूल का नियम फिर मिला) और $C_P - C_V = nR$। किसी फ़ान
डेर वाल्स गैस के लिए, $(\partial_VU)_T = a/V_{\text{मोलर}}^2$: अर्थात् आकर्षणों का
भीतरी दाब। द्रव जल के लिए, $(\partial_TP)_V = \alpha/\kappa_T = 2.1
\times 10^{-4}/4.6 \times 10^{-10} = \qty{4.6e5}{Pa/K}$: किसी दृढ़ भरे पात्र में जल को
एक केल्विन गरम करना दाब \qty{4.6}{bar} चढ़ा देता है
(कोई भरी बोतल कभी बंद मत कीजिए), और $(\partial_VU)_T \approx \qty{1400}{bar}$ ---
अर्थात् द्रव की संसक्ति; फिर भी $c_P - c_V = Tv\alpha^2/\kappa_T \approx
\qty{30}{J/kg/K}$, यानी $1\%$ से भी कम, $c_P$ का: अर्थात् संघनित पदार्थ के लिए दोनों
ऊष्मा-धारिताएँ लगभग मिल जाती हैं।
\end{example}

\begin{proposition}[रबर की एंट्रॉपिक प्रत्यास्थता]\label{prop:b2:thermodynamic-potentials:rubber}
लंबाई $L$ के किसी फ़ीते के लिए, जो तनाव $f$ के अधीन हो, $\dd U = T\dd S + f\dd L$
और $\dd F = -S\dd T + f\dd L$, जिससे $(\partial S/\partial L)_T =
-(\partial f/\partial T)_L$ और $(\partial U/\partial L)_T = f - T(\partial f/
\partial T)_L$। रबर, अच्छे सन्निकटन में, $f = T\,\varphi(L)$ मानता है,
जिसमें $\varphi > 0$ हो $L > L_0$ के लिए: तब $(\partial U/\partial L)_T = 0$ --- अर्थात्
खींचने पर ऊर्जा बदलती नहीं --- और $(\partial S/\partial L)_T =
-\varphi(L) < 0$: अतः तनाव \emph{एंट्रॉपिक} है, $f = -T(\partial S/\partial
L)_T$, और फ़ीता इसलिए पीछे खींचता है कि उसकी खिंची शृंखलाओं के पास कम
विन्यास बचते हैं। परिणाम: समतापी रूप से खींचने पर वह ऊष्मा छोड़ता है
($Q = T\Delta S < 0$); रुद्धोष्म रूप से खींचने पर वह गरम होता है; और अचर भार के
नीचे गरम करने पर वह सिकुड़ता है (नियत $f/T$ का अर्थ है नियत $\varphi(L)$, अतः
नियत $f$ पर $\varphi$ को गिरना पड़ता है जैसे-जैसे $T$ चढ़े)।
\end{proposition}

\begin{proof}
$\dd F$ से मैक्सवेल संबंध; और बाक़ी प्रतिस्थापन है। और
सूक्ष्म मूल \cref{ch:b2:particle-diffusion} की यादृच्छिक चाल है:
$N$ कड़ियों की किसी शृंखला के, जिनकी लंबाई $a$ हो, सिरे-से-सिरे की दूरी
प्रसरण $Na^2$ के किसी गाउसी की तरह बँटी है, अतः विस्तार $L$ पर
विन्यासों की संख्या $\Omega \propto \eu^{-L^2/2Na^2}$ है, $S = S_0 -
k_BL^2/2Na^2$ और $f = k_BTL/Na^2$: अर्थात् कोई ऐसा स्प्रिंग जिसकी दृढ़ता
\emph{ताप के समानुपाती} है।
\end{proof}

\section{दो प्रावस्थाओं के बीच साम्य}

\begin{theorem}[सह-अस्तित्व की शर्त]\label{thm:b2:thermodynamic-potentials:coexistence}
नियत $T$ और $P$ पर कोई शुद्ध पदार्थ, जो द्रव्यमानों $m_1$, $m_2$ की दो
प्रावस्थाओं में बँटा हो और जिनकी प्रति एकांक द्रव्यमान \omterm{def:b2:thermodynamic-potentials:FG}{मुक्त एंथैल्पियाँ} $g_1(T,P)$,
$g_2(T,P)$ हों, $G = m_1g_1 + m_2g_2$ रखता है; और साम्य पर $G$ न्यूनतम है, अतः
\begin{itemize}
\item यदि $g_1 < g_2$ हो तो सारा पदार्थ प्रावस्था 1 में चला जाता है (स्थायी
      प्रावस्था वही है जिसका $g$ \emph{नीचा} हो);
\item दोनों प्रावस्थाएँ केवल वहीं साथ रहती हैं जहाँ
      \[
      g_1(T,P) = g_2(T,P) ,
      \]
      अर्थात् $T$ और $P$ के बीच कोई संबंध: यानी प्रावस्था-आरेख का
      सह-अस्तित्व वक्र।
\end{itemize}
उस वक्र के अनुदिश (क्लापेरों),
\[
\frac{\dd P}{\dd T} = \frac{s_2 - s_1}{v_2 - v_1} = \frac{L_{1\to2}}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
$\dd G = (g_1 - g_2)\dd m_1$ नियत $T$, $P$ पर ($m_1 + m_2$ नियत रखकर):
अतः $G$ घटता है जब द्रव्यमान नीचे $g$ वाली प्रावस्था में जाए, जब तक वह
प्रावस्था अकेली न रह जाए या $g_1 = g_2$ न हो जाए। $g_1 = g_2$ का वक्र के अनुदिश
अवकलन कीजिए, हर प्रावस्था के लिए $\dd g = -s\,\dd T + v\,\dd P$ के साथ: $-s_1\dd T +
v_1\dd P = -s_2\dd T + v_2\dd P$; और $s_2 - s_1 = L/T$, क्योंकि नियत $T$, $P$ पर
संक्रमण उत्क्रमणीय है। वर्ष 1 के खंड ने यह सूत्र किसी कार्नो चक्र से
पाया था; यहाँ वह $G$ से टपक पड़ता है।
\end{proof}

\begin{remark}[फ़ान डेर वाल्स और मैक्सवेल की रचना]\label{rem:b2:thermodynamic-potentials:maxwellconstruction}
फ़ान डेर वाल्स समीकरण $(P + a/v^2)(v - b) = RT$ अपने
क्रांतिक ताप $T_{\text{क्रां}} = 8a/27Rb$ से नीचे ऐसे समताप देता है जिनमें कोई पाश हो। और
असली तरल पाश के पीछे नहीं चलता: $P_{\text{संत}}(T)$ पर वह
द्रव से वाष्प वाली शाखा पर कूद जाता है, और $P_{\text{संत}}$ $g_{\text{द्रव}}
= g_{\text{वाष्प}}$ से तय होता है, अर्थात् पाश के अनुदिश $\int_{\text{द्रव}}^{\text{वाष्प}}v\,\dd P = 0$
से: यानी क्षैतिज रेखा \emph{बराबर क्षेत्रफल} काट देती है। पाश के वे भाग
जहाँ $(\partial P/\partial v)_T > 0$ हो यांत्रिक रूप से अस्थायी हैं; और
उनके तथा रेखा के बीच वाले अर्ध-स्थायी --- यानी वह अतितापित
द्रव जो किसी सूक्ष्मतरंग में ``उछल पड़ता'' है, और कुहरा-कक्ष की अतिसंतृप्त वाष्प।
\end{remark}

\begin{method}[विभवों को काम में लेना]\label{meth:b2:thermodynamic-potentials:method}
(1) बंधन पहचानिए: विलगित $\to$ $S$ अधिकतम; नियत $T,V$ $\to$
$F$ न्यूनतम; नियत $T,P$ $\to$ $G$ न्यूनतम। (2) अधिकतम कार्य: $-\Delta F$ नियत $T$ पर,
और $-\Delta G$ नियत $T,P$ पर \omterm{def:b2:open-systems:cv}{उपयोगी कार्य} का। (3) $S$ का कोई अवकलज
नियत $T$ पर चाहिए? कोई मैक्सवेल संबंध और अवस्था-समीकरण लीजिए।
(4) दो प्रावस्थाएँ: $g$ की तुलना कीजिए; सह-अस्तित्व $g_1 = g_2$; ढाल के लिए
क्लापेरों; और ढाल के चिह्न के लिए $\Delta v$ का चिह्न। (5) \omterm{ex:b2:thermodynamic-potentials:perfect}{रासायनिक
विभव} $\mu = g$ प्रति मोल: पदार्थ नीचे $\mu$ की ओर बहता है।
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:1}
पूर्ण गैस के $n$ मोलों के लिए, $F(T,V)$ और $G(T,P)$ अकेले $T$ के फलनों

कि $S = -\partial_TF$ जानी हुई एंट्रॉपी देता है।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:2}
गैस का कोई मोल, \qty{300}{K}, किसी तापस्थापी में \qty{1}{L} से \qty{10}{L}

एंट्रॉपी।
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:3}

(L/T_{\text{गलन}})(T_{\text{गलन}} - T)$ है और उसे
\qty{-5}{} तथा $+\qty{5}{\celsius}$ पर निकालिए ($L = \qty{334}{kJ/kg}$)। हर हाल में कौन-सी प्रावस्था
स्थायी है; क्या दूसरी असंभव है?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:4}
जल की क्लापेरों ढालें: गलन ($\Delta v = -\qty{9.1e-5}{m^3/kg}$) और
\qty{100}{\celsius} पर वाष्पन ($L = \qty{2.26}{MJ/kg}$, $\Delta v =
\qty{1.67}{m^3/kg}$)। \qty{70}{bar} के नीचे गलनांक; और \qty{0.7}{bar} तथा
\qty{2}{bar} पर क्वथनांक।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:5}
\emph{मैक्सवेल।} (क) अध्याय के दोनों मैक्सवेल संबंध निकालिए
और $\dd U$ तथा $\dd H$ वाले दोनों भी। (ख) दिखाइए $(\partial U/\partial V)_T =
T(\partial P/\partial T)_V - P$; और उसे किसी पूर्ण गैस तथा किसी फ़ान
डेर वाल्स गैस के लिए आँकिए। (ग) द्रव जल: $\alpha = \qty{2.1e-4}{K^{-1}}$, $\kappa_T =
\qty{4.6e-10}{Pa^{-1}}$: किसी दृढ़ भरे पात्र में प्रति केल्विन दाब की चढ़ाई,
और $(\partial U/\partial V)_T$। (घ) काँच की कोई भरी, बंद बोतल
\qty{10}{K} गरम की जाती है: टिप्पणी कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:6}
\emph{$C_P - C_V$.} (क) $C_P - C_V = TV\alpha^2/\kappa_T$ निकालिए। (ख)
पूर्ण गैस। (ग) \qty{300}{K} पर जल; ताँबा ($\alpha = \qty{5e-5}{K^{-1}}$,
$\kappa_T = \qty{7e-12}{Pa^{-1}}$, $\rho = \qty{8900}{kg/m^3}$); और
$c_P$ (\qty{4180}{} तथा \qty{385}{J/kg/K}) से तुलना कीजिए। (घ) किसी ठोस की
``एक ही'' ऊष्मा-धारिता की बात क्यों की जा सकती है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:7}
\emph{रबर।} कोई फ़ीता $f = CT(L - L_0)$ मानता है, जिसमें $C = \qty{1}{N/m/K}$।
(क) $(\partial S/\partial L)_T$ और $(\partial U/\partial L)_T$। (ख) \qty{300}{K} पर
$L_0$ से $L_0 + \qty{0.1}{m}$ तक समतापी रूप से खींचने पर एंट्रॉपी का
बदलाव, ऊष्मा और कार्य। (ग) इसके बजाय रुद्धोष्म रूप से खींचने पर (ऊष्मा-धारिता
\qty{4}{J/K}): ताप का बदलाव। (घ) किसी नियत भार के साथ लटकाकर
\qty{30}{K} गरम करने पर: $L - L_0$ का सापेक्ष बदलाव।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:8}
\emph{किसी पोखर का वाष्पन।} आंशिक दाब $p$ पर जल-वाष्प का \omterm{ex:b2:thermodynamic-potentials:perfect}{रासायनिक विभव}
$\mu_{\text{वा}} = \mu_{\text{वा}}^\circ(T) +
RT\ln(p/p^\circ)$ है, और वह द्रव वाले के बराबर तब होता है जब $p = p_{\text{संत}}(T)$।
(क) दिखाइए कि $p < p_{\text{संत}}$ होने पर द्रव वाष्पित होता है और
$p > p_{\text{संत}}$ होने पर वाष्प संघनित; और सापेक्ष आर्द्रता परिभाषित कीजिए। (ख)
$\mu_{\text{द्रव}} - \mu_{\text{वा}}$, $50\%$ आर्द्रता पर, \qty{293}{K}। (ग) सूखी हवा में
\qty{20}{\celsius} पर भी धुले कपड़े क्यों सूख जाते हैं, और $100\%$ आर्द्रता पर कभी क्यों नहीं?
(घ) ओस: वह पहले किन सतहों पर बनती है, और क्यों?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:9}
\emph{फ़ान डेर वाल्स।} $(P + a/v^2)(v - b) = RT$ (मोलर)। (क) क्रांतिक
बिंदु निकालिए ($\partial_vP = \partial_v^2P = 0$): $v_{\text{क्रां}} = 3b$,
$T_{\text{क्रां}} = 8a/27Rb$, $P_{\text{क्रां}} = a/27b^2$। (ख) कार्बन डाइऑक्साइड,
$T_{\text{क्रां}} = \qty{304}{K}$, $P_{\text{क्रां}} = \qty{73.8}{bar}$: $a$ और $b$। (ग)
$g$ से मैक्सवेल का बराबर-क्षेत्रफल नियम न्यायसंगत ठहराइए। (घ) पाश के कौन-से भाग
अस्थायी हैं, कौन-से अर्ध-स्थायी, और अर्ध-स्थायी वालों को कौन-सी परिघटनाएँ
दिखाती हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:10}
\emph{क्लाउज़ियस--क्लापेरों समाकलित।} वाष्पन के लिए $v_{\text{द्रव}}$ छोड़ दीजिए
और वाष्प को पूर्ण मानिए। (क) दिखाइए $\dd\ln P/\dd T = L_{\text{मोलर}}/RT^2$
($L_{\text{मोलर}}$ मोलर) और अचर $L_{\text{मोलर}}$ के लिए समाकलित कीजिए। (ख) जल,
$L_{\text{मोलर}} = \qty{40.7}{kJ/mol}$: \qty{100}{\celsius} पर \qty{1}{bar} से
\qty{20}{\celsius} पर $P_{\text{संत}}$; और नापे गए \qty{23}{mbar} से तुलना कीजिए।
(ग) त्रिक-बिंदु का दाब (\qty{0.01}{\celsius}) पूर्वबताइए और उसकी
\qty{611}{Pa} से तुलना कीजिए। (घ) दिखाइए कि वही नियम $\Delta g$ पर लगाए गए
गिब्स--हेल्महोल्ट्स से निकल आता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:11}
\emph{परासरण।} किसी विलायक का \omterm{ex:b2:thermodynamic-potentials:perfect}{रासायनिक विभव} किसी तनु
विलेय से गिर जाता है: $\mu = \mu^\circ - RTx_{\text{घु}}$ ($x_{\text{घु}}$ विलेय का मोल
अंश)। कोई झिल्ली केवल विलायक को जाने देती है; एक ओर शुद्ध विलायक,
दूसरी ओर विलयन, वही $T$। (क) दिखाइए कि साम्य के लिए
विलयन पर कोई अतिरिक्त दाब $\Pi$ चाहिए, जिसमें $\Pi v_{\text{मोलर}} =
RTx_{\text{घु}}$, अर्थात् $\Pi = cRT$ (फ़ान 'ट हॉफ़)। (ख) समुद्री जल, घुले कणों का $c \approx
\qty{1.1}{mol/L}$, \qty{293}{K}: $\Pi$। (ग) कोई घन मीटर मीठा जल निकालने का
न्यूनतम कार्य; असली संयंत्रों (\qty{3}{kWh/m^3}) से तुलना कीजिए। (घ) पौधों की
कोशिकाएँ शुद्ध जल में क्यों फट जाती हैं और नमकीन जल में सिकुड़ जाती हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:12}
\emph{शृंखला।} $N$ कड़ियों की कोई बहुलक शृंखला, जिनकी लंबाई $a$ हो: फ़ीते के अनुदिश
लंबाई $L$ के सिरे-से-सिरे सदिश वाले विन्यासों की
संख्या $\Omega \propto \eu^{-L^2/2Na^2}$ है (यादृच्छिक चाल)। (क) $S(L)$
और, नियत $U$ पर, तनाव $f = -T\,\partial S/\partial L$। (ख) शृंखला की
दृढ़ता और उसकी ताप-निर्भरता। (ग) $N = 1000$, $a =
\qty{0.5}{nm}$, \qty{300}{K}, $L = \qty{10}{nm}$: $f$; और \qty{30}{N} देने के लिए
समांतर में कितनी शृंखलाएँ? (घ) प्रति घन मीटर $10^{26}$ शृंखलाओं वाले, हर एक
$N = 100$ की, किसी रबर का \omterm{prop:b2:waves-on-strings:chain}{यंग गुणांक} आँकिए।
\end{exercise}

\section{समस्या: रबर का फ़ीता और जल का प्रावस्था-आरेख}

\begin{problem}[{सप्ताहांत समस्या --- वह एंट्रॉपी जो खींचती है, और वे
प्रावस्थाएँ जो होड़ करती हैं}]\label{pb:b2:thermodynamic-potentials:1}
\textbf{भाग I --- रबर का फ़ीता।}
\qty{2}{g} द्रव्यमान (ऊष्मा-धारिता \qty{4}{J/K}) के किसी फ़ीते का तनाव
$f = CT(L - L_0)$ है, जिसमें $C = \qty{1}{N/m/K}$; और \qty{300}{K} तथा $L - L_0 =
\qty{0.1}{m}$ पर $f = \qty{30}{N}$।
\begin{enumerate}
  \item फ़ीते के लिए $\dd U$ और $\dd F$ लिखिए ($\dd U = T\dd S + f\dd L$)।
  \item मैक्सवेल संबंध $(\partial S/\partial L)_T = -(\partial
        f/\partial T)_L$ निकालिए।
  \item दिखाइए कि इस फ़ीते के लिए $(\partial U/\partial L)_T = 0$: उसका क्या
        अर्थ है?
  \item \qty{300}{K} पर $L_0$ से $L_0 + \qty{0.1}{m}$ तक समतापी खिंचाव:
        $\Delta S$, ऊष्मा $Q$, कार्य $W$; और $\Delta U = 0$ जाँचिए।
  \item वही खिंचाव रुद्धोष्म रूप से किया जाए: ताप का बदलाव।
  \item फ़ीता कोई नियत भार थामे है; उसे \qty{30}{K} गरम किया जाता है: नया
        $L - L_0$। ``रबर इंजन'' का वर्णन कीजिए (रबर के आरों वाला कोई पहिया
        जो एक ओर से गरम किया जाए)।
  \item $N$ कड़ियों की किसी शृंखला का, जिनकी लंबाई $a$ हो, $\Omega \propto \eu^{-L^2/2Na^2}$ है:
        $S(L)$ और $f(L,T)$; दृढ़ता $T$ के साथ क्यों बढ़ती है?
  \item $N = 1000$, $a = \qty{0.5}{nm}$, $L = \qty{10}{nm}$, \qty{300}{K}: किसी एक
        शृंखला के लिए $f$ और \qty{30}{N} के लिए शृंखलाओं की संख्या।
  \item इस्पात का कोई स्प्रिंग उलटा क्यों बरतता है (नियत लंबाई पर तनाव
        $T$ के साथ गिरता है)?
\end{enumerate}

\textbf{भाग II --- जल का प्रावस्था-आरेख।}
आँकड़े: $L_{\text{गलन}} = \qty{334}{kJ/kg}$, $v_{\text{हिम}} = \qty{1.091e-3}{m^3/kg}$,
$v_{\text{द्रव}} = \qty{1.000e-3}{m^3/kg}$; \qty{100}{\celsius} पर $L_{\text{वाष्प}} = \qty{2.26}{MJ/kg}$,
$v_{\text{वाष्प}} = \qty{1.67}{m^3/kg}$; $L_{\text{मोलर}} = \qty{40.7}{kJ/mol}$;
त्रिक बिंदु \qty{0.01}{\celsius}, \qty{611}{Pa}; क्रांतिक बिंदु \qty{647}{K},
\qty{221}{bar}।
\begin{enumerate}[resume]
  \item \qty{1}{bar} पर बर्फ़, द्रव और वाष्प के लिए $g(T)$ खींचिए; ढालों
        का क्रम न्यायसंगत ठहराइए और संक्रमण ढूँढ़िए।
  \item $g_1 = g_2$ से क्लापेरों का सूत्र निकालिए।
  \item गलन-वक्र की ढाल; उसका चिह्न, और वह असामान्य क्यों है।
  \item कोई स्केटर: \qty{1}{cm^2} पर \qty{700}{N}: गलनांक का
        बदलाव; और स्केटिंग की लोकप्रिय व्याख्या पर निष्कर्ष निकालिए।
  \item बर्फ़ के \qty{3}{km} के नीचे ($\rho = \qty{917}{kg/m^3}$): दाब और
        गलनांक; और हिम-चादर के लिए परिणाम।
  \item \qty{100}{\celsius} पर वाष्पन-वक्र की ढाल; \qty{0.7}{bar} और
        \qty{2}{bar} पर क्वथनांक।
  \item पूर्ण वाष्प और अचर $L_{\text{मोलर}}$ के साथ क्लाउज़ियस--क्लापेरों
        समाकलित कीजिए; \qty{20}{\celsius} पर और त्रिक बिंदु पर
        $P_{\text{संत}}$; और परिशुद्धता पर टिप्पणी कीजिए।
  \item त्रिक बिंदु पर, ऊर्ध्वपातन और वाष्पन वक्रों की ढालों की
        तुलना कीजिए ($L_{\text{ऊर्ध्व}} = L_{\text{गलन}} + L_{\text{वाष्प}}$)।
  \item \qty{-10}{\celsius} पर अतिशीतित जल: $g_{\text{द्रव}} - g_{\text{हिम}}$;
        बादल की बूँदें \qty{-40}{\celsius} तक द्रव क्यों रह सकती हैं?
  \item क्रांतिक बिंदु पर $\Delta v$, $L$ और ढाल का क्या होता है; और
        कोई अतिक्रांतिक तरल क्या है?
\end{enumerate}

\textbf{भाग III --- अनुप्रयोग।}
\begin{enumerate}[resume]
  \item किसी फ़ान डेर वाल्स समताप पर मैक्सवेल की बराबर-क्षेत्रफल रचना को
        $g$ के सांतत्य से समझाइए।
  \item किसी सूक्ष्मतरंग में अतितापित जल ``उछल पड़ता'' है: समताप की कौन-सी
        शाखा, और ख़तरा क्यों?
  \item हिम-शुष्कन: $(P,T)$ आरेख में पथ का वर्णन कीजिए और
        जल बिना पिघले क्यों निकल जाता है।
  \item सूखी हवा में \qty{20}{\celsius} पर जल की कोई तश्तरी क्यों वाष्पित होती है
        और संतृप्त हवा में क्यों रुक जाती है --- \omterm{ex:b2:thermodynamic-potentials:perfect}{रासायनिक विभवों} के पदों में?
  \item \qty{2}{bar} का कोई दाब-कुकर: भोजन इतना अधिक तेज़ी से क्यों पकता है
        (रसायन की दर मोटे तौर पर हर \qty{10}{K} पर दुगुनी हो
        जाती है)?
  \item समेटिए: किस बंधन के लिए कौन-सा विभव, और मैक्सवेल संबंध
        क्या ख़रीद देते हैं।
\end{enumerate}
\end{problem}
