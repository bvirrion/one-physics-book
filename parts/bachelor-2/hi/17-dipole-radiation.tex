\chapter{द्विध्रुव विकिरण और प्रकीर्णन}\label{ch:b2:dipole-radiation}

आकाश नीला क्यों है, और सूर्य से समकोण पर सबसे नीला क्यों --- और वहीं
ध्रुवित भी, जैसा मधुमक्खियाँ और छायाकार जानते हैं? किसी रेडियो मीनार को
दसियों मीटर ऊँचा क्यों होना पड़ता है, और वह सीधे ऊपर की ओर कुछ भी विकिरित
क्यों नहीं करती? दोनों प्रश्नों का उत्तर एक ही है: कोई दोलन करता विद्युत्
आवेश विद्युत्चुंबकीय तरंगें विकिरित करता है, ऐसी शक्ति के साथ जो आवृत्ति की
चौथी घात की तरह बढ़ती है और ऐसी आकृति के साथ जो दोलन की अक्ष के अनुदिश लुप्त
हो जाती है। यह अध्याय \emph{दोलन करते द्विध्रुव} का, अर्थात् सबसे सरल विकिरक
व्यवस्था का, क्षेत्र बताता है, उससे ऐंटेना और किसी त्वरित आवेश का विकिरण
निकालता है, और उसे हवा के अणुओं पर लगाता है, जिनमें हर एक कोई नन्हा ऐंटेना है
जो सूर्य के प्रकाश को फिर से उत्सर्जित कर देता है: \omterm{prop:b2:dipole-radiation:rayleigh}{रैले प्रकीर्णन}।

\section{किसी दोलन करते द्विध्रुव का क्षेत्र}

\begin{theorem}[किसी विद्युत् द्विध्रुव का विकिरण क्षेत्र]\label{thm:b2:dipole-radiation:field}
\index{दोलन करता द्विध्रुव}\index{विकिरण क्षेत्र}\index{विलंबन}
मूल बिंदु पर रखा आघूर्ण $\vect p(t) = p_0\cos\omega t\,\vect e_z$ और आकार $a$ का
कोई द्विध्रुव विकिरित करता है। \emph{\omterm{thm:b2:dipole-radiation:field}{विकिरण क्षेत्र}} में --- अर्थात् उन दूरियों पर जहाँ $r \gg
\lambda = 2\pi c/\omega \gg a$ --- द्विध्रुव की अक्ष के इर्द-गिर्द गोलीय निर्देशांकों $(r,
\theta, \varphi)$ में उसका क्षेत्र है
\[
\vect E(M, t) = -\frac{\mu_0\,\omega^2p_0}{4\pi}\,\frac{\sin\theta}r\,\cos\bigl(\omega(t - r/c)\bigr)\,\vect e_\theta ,
\qquad
\vect B = \frac{\vect e_r\wedge\vect E}c :
\]
अर्थात् कोई \omterm{prop:b2:sound-waves:spherical}{गोलीय तरंग}, स्थानीय रूप से समतल --- $(\vect E, \vect B, \vect e_r)$ कोई
दक्षिणावर्ती त्रयी जिसमें $B = E/c$ --- जो यात्रा-काल
$r/c$ से \emph{विलंबित} है, जिसका आयाम $1/r$ की तरह गिरता है, जो द्विध्रुव के त्वरण
$\ddot p = -\omega^2p$ के और $\sin\theta$ के समानुपाती है: अर्थात्
विषुवतीय तल में अधिकतम, अक्ष के अनुदिश शून्य; और अक्ष वाले तल में
ध्रुवित।
\end{theorem}
\admitted

\begin{remark}[सूत्र को पढ़ना]\label{rem:b2:dipole-radiation:reading}
\omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल के समीकरणों} के यथार्थ हल में (विलंबित विभवों से, जो इस खंड से
परे है) $1/r^2$ और
$1/r^3$ के पद भी होते हैं; अंतिम वर्ष 1 के खंड वाला अर्ध-स्थैतिक द्विध्रुव क्षेत्र है,
$\vect E \propto p/4\pi\varepsilon_0r^3$, जो \emph{निकट
क्षेत्र} $r \ll \lambda$ में हावी रहता है; $r \sim \lambda/2\pi$ पर दोनों तुलनीय हैं, और उससे परे
केवल $1/r$ वाला पद बचता है। हर गुणक का कोई कारण है। $1/r$: किसी गोले से
पार शक्ति, $\propto E^2r^2$, $r$ पर निर्भर नहीं होनी चाहिए। $\omega^2p_0$:
विरामावस्था में या एकसमान गति में किसी आवेश का क्षेत्र विकिरक नहीं है; विकिरण
\emph{त्वरण} करता है। $\sin\theta$: अक्ष पर बैठे किसी प्रेक्षक को
आवेश दृष्टि-रेखा के अनुदिश आता-जाता दिखता है, बिना किसी
अनुप्रस्थ त्वरण के; जबकि विषुवतीय तल में पूरा
त्वरण अनुप्रस्थ है। विलंबन: $M$ का क्षेत्र समय $t$ पर
वही दिखाता है जो द्विध्रुव ने $t - r/c$ पर किया था।
\end{remark}

\section{विकिरित शक्ति}

\begin{proposition}[किसी द्विध्रुव की शक्ति; लार्मर का सूत्र]\label{prop:b2:dipole-radiation:power}
\index{विकिरित शक्ति (द्विध्रुव)}\index{लार्मर सूत्र}\index{विकिरण आकृति}
द्विध्रुव के क्षेत्र का माध्य पॉयंटिंग सदिश त्रिज्य है,
\[
\langle\vect\Pi\rangle = \frac{\mu_0\,\omega^4p_0^2}{32\pi^2c}\,\frac{\sin^2\theta}{r^2}\,\vect e_r ,
\]
और विकिरित कुल माध्य शक्ति है
\[
\mathcal P = \frac{\mu_0\,\omega^4p_0^2}{12\pi c} = \frac{p_0^2\omega^4}{12\pi\varepsilon_0c^3} .
\]
किसी अकेले आवेश $q$ के लिए, जिसका त्वरण $a$ हो (अनापेक्षिकीय),
तात्क्षणिक शक्ति $\mathcal P = q^2a^2/6\pi\varepsilon_0c^3$ है (लार्मर)।
\end{proposition}

\begin{proof}
$\langle\Pi\rangle = \langle E^2\rangle/\mu_0c$, जिसमें $\langle\cos^2\rangle = \tfrac12$;
गोले पर समाकलित कीजिए:
\[
\int\sin^2\theta\,r^2\dd\Omega = 2\pi r^2\int_0^\pi\sin^3\theta\,\dd\theta = \tfrac83\pi r^2 ,
\qquad
\mathcal P = \frac{\mu_0\omega^4p_0^2}{32\pi^2c}\cdot\frac{8\pi}3 = \frac{\mu_0\omega^4p_0^2}{12\pi c} .
\]
लार्मर: $p = qz$, अतः $\ddot p = qa$;
माध्य $\tfrac12\omega^4p_0^2 = \langle\ddot p^2\rangle$ की जगह
तात्क्षणिक $q^2a^2$ रखने पर $\mathcal P = \mu_0q^2a^2/6\pi c$ मिलता है।
\end{proof}

\begin{example}[कोई ऐंटेना और कोई तार]\label{ex:b2:dipole-radiation:antenna}
\index{ऐंटेना}\index{विकिरण प्रतिरोध}\index{अर्ध-तरंग द्विध्रुव}
लंबाई $\ell \ll \lambda$ का कोई छोटा तार, जो धारा $I_0\cos\omega t$ ले चलता है, कोई
द्विध्रुव है जिसका $\dot p = I_0\ell\cos\omega t$, अर्थात् $p_0 = I_0\ell/\omega$: वह
$\mathcal P = \mu_0\omega^2I_0^2\ell^2/12\pi c = \tfrac12R_{\text{विकिरण}}I_0^2$ विकिरित करता है, जिसमें
\emph{\omterm{ex:b2:dipole-radiation:antenna}{विकिरण प्रतिरोध}} $R_{\text{विकिरण}} = \dfrac{2\pi}3\sqrt{\dfrac{\mu_0}{\varepsilon_0}}
\Bigl(\dfrac\ell\lambda\Bigr)^2 \approx 790\,(\ell/\lambda)^2\ \Omega$ --- शक्ति परिपथ से ऐसे
निकल जाती है मानो किसी प्रतिरोधक से होकर, पर कुछ भी गरम किए बिना। \qty{1}{MHz} पर
\qty{1}{m} का कोई तार ($\lambda = \qty{300}{m}$): $R_{\text{विकिरण}} = \qty{9}{m\ohm}$, जो
उसके ताँबे के प्रतिरोध से भी कम है, अर्थात् कोई ख़राब ऐंटेना; \qty{100}{MHz} पर: \qty{88}{\ohm} (यहाँ
छोटे-द्विध्रुव का सूत्र खींचा जा रहा है), अर्थात् कोई अच्छा। \emph{\omterm{ex:b2:dipole-radiation:antenna}{अर्ध-तरंग
द्विध्रुव}}, $\ell = \lambda/2$ जिसमें धारा का वितरण ज्यावक्रीय हो, का
$R_{\text{विकिरण}} = \qty{73}{\ohm}$ है (स्वीकृत) और उसकी आकृति प्रारंभिक द्विध्रुव
वाली के पास है --- वही मानक ऐंटेना, रेडियो मीनारों से लेकर पुराने
दूरदर्शनों के खरगोश-कानों तक। \qty{50}{Hz} पर \qty{1}{m} का कोई मुख्य लाइन का तार:
$R_{\text{विकिरण}} \approx \qty{2e-11}{\ohm}$ --- अर्थात् \cref{ch:b2:maxwell-equations}
के अर्ध-स्थायी परिपथ कुछ भी विकिरित नहीं करते।
\end{example}

\begin{example}[विकिरित करता कोई इलेक्ट्रॉन]\label{ex:b2:dipole-radiation:electron}
किसी X-किरण नलिका में \qty{50}{keV} का कोई इलेक्ट्रॉन ($v = \qty{1.2e8}{m/s}$)
टंग्स्टन के \qty{1}{\micro m} में रुक जाता है: $a \sim v^2/2d = \qty{7e21}{m/s^2}$, लार्मर शक्ति
$\qty{1.2e-4}{W}$, $\qty{1.6e-14}{s}$ तक: \qty{2e-18}{J}, अर्थात् उसकी ऊर्जा का लगभग $0.02\%$
नलिका के "मंदन विकिरण" (ब्रेमस्ट्राहलुंग) के रूप में विकिरित
--- बाक़ी ऊष्मा है, और ऐनोड ठंडा किया जाता है। किसी हाइड्रोजन परमाणु का
चिरसम्मत इलेक्ट्रॉन, जो $v^2/r = \qty{9e22}{m/s^2}$ से त्वरित है,
\qty{5e-8}{W} विकिरित करता और \qty{1e-11}{s} में सर्पिल खाता हुआ नाभिक में गिर जाता:
चिरसम्मत भौतिकी परमाणुओं को मना कर देती है, और \cref{ch:b2:schrodinger-wave-functions}
उन्हें बचा लेता है।
\end{example}

\section{रैले प्रकीर्णन: नीला आकाश}

\begin{proposition}[किसी बद्ध इलेक्ट्रॉन से प्रकीर्णन]\label{prop:b2:dipole-radiation:rayleigh}
\index{रैले प्रकीर्णन}\index{प्रकीर्णन अनुप्रस्थ काट}\index{टामसन प्रकीर्णन}
आयाम $E_0$ और आवृत्ति $\omega$ की कोई तरंग किसी ऐसे परमाणु पर पड़ती है जिसे
अनुनाद $\omega_0 \gg \omega$ वाले किसी प्रत्यास्थ रूप से बद्ध इलेक्ट्रॉन का प्रतिरूप दिया गया है
(\cref{ch:b2:waves-in-media}): इलेक्ट्रॉन आयाम
$eE_0/m\omega_0^2$ से दोलन करता है, परमाणु क्षेत्र के साथ कला में कोई द्विध्रुव $p_0 = e^2E_0/m\omega_0^2$
बन जाता है, और शक्ति $\mathcal P = p_0^2\omega^4/12\pi
\varepsilon_0c^3$ फिर से विकिरित करता है। उस शक्ति और आपाती तीव्रता का अनुपात है
\emph{\omterm{prop:b2:dipole-radiation:rayleigh}{प्रकीर्णन अनुप्रस्थ काट}}
\[
\sigma = \frac{\mathcal P}{\varepsilon_0cE_0^2/2} = \frac{8\pi}3\,r_e^2\Bigl(\frac\omega{\omega_0}\Bigr)^4 ,
\qquad r_e = \frac{e^2}{4\pi\varepsilon_0mc^2} = \qty{2.8e-15}{m} :
\]
अर्थात् रैले का नियम, $\sigma \propto \omega^4 \propto 1/\lambda^4$। किसी \emph{मुक्त}
इलेक्ट्रॉन ($\omega_0 = 0$) के लिए अनुप्रस्थ काट $\sigma_T = 8\pi r_e^2/3 =
\qty{6.65e-29}{m^2}$ है, जो आवृत्ति से स्वतंत्र है (\omterm{prop:b2:dipole-radiation:rayleigh}{टामसन प्रकीर्णन})।
\end{proposition}

\begin{proof}
$\omega \ll \omega_0$ और नगण्य अवमंदन वाले \omterm{prop:b2:waves-in-media:lorentz}{लोरेंत्स प्रतिरूप} से,
$\underline{\vect r} = -e\underline{\vect E}/m\omega_0^2$; शक्ति में $p_0$ डालिए और
$I = \varepsilon_0cE_0^2/2$ से भाग दीजिए; और $r_e$ अचरों को समेट लेता है। मुक्त
इलेक्ट्रॉन के लिए, $\vect r = e\vect E/m\omega^2$ (विपरीत कला में), $p_0 = e^2E_0/m\omega^2$, और
$\omega^4$ कट जाते हैं।
\end{proof}

\begin{example}[नीला आकाश, लाल सूर्यास्त, सफ़ेद बादल]\label{ex:b2:dipole-radiation:sky}
बैंगनी प्रकाश ($\qty{400}{nm}$) लाल से $(700/400)^4 = 9.4$ गुना अधिक
प्रकीर्ण होता है: अतः प्रकीर्ण धूप से जगमगाता आकाश नीला है (बैंगनी नहीं:
सूर्य बैंगनी कम उत्सर्जित करता है और आँख उसे ठीक से देख नहीं पाती)। सूर्यास्त पर सीधा
प्रकाश दोपहर से चालीस गुना अधिक हवा पार करता है और अपना नीला
और जगहों के आकाश को दे बैठता है: वह लाल हो जाता है। हवा का हर अणु
\qty{550}{nm} पर $\sigma \approx \qty{5e-31}{m^2}$ से प्रकीर्ण करता है; और $n = \qty{2.5e25}{m^{-3}}$ के साथ
माध्य मुक्त पथ $1/n\sigma$ \qty{80}{km} है: अर्थात् वायुमंडल से नीचे उतरते हुए हरे
प्रकाश का दसवाँ भाग प्रकीर्ण हो जाता है, और बैंगनी का
तिहाई। बादल की बूँदें, जो $\lambda$ से कहीं बड़ी हैं, रैले
प्रकीर्णक नहीं हैं: वे सारे रंग एक-से प्रकीर्ण करती हैं --- सफ़ेद।
\end{example}

\begin{proposition}[प्रकीर्णन से ध्रुवण]\label{prop:b2:dipole-radiation:polarization}
\index{प्रकीर्णन से ध्रुवण}
$x$ के अनुदिश चलता \omterm{def:b2:plane-waves-polarization:states}{प्राकृतिक प्रकाश} $yz$ तल में द्विध्रुव प्रेरित करता है।
$y$ के अनुदिश देखते किसी प्रेक्षक को (पुंज से \qty{90}{\degree} पर)
$y$ के अनुदिश (उसकी दृष्टि-रेखा) द्विध्रुव-घटकों से कोई विकिरण नहीं मिलता
और $z$ के अनुदिश वालों से पूरा: अर्थात् प्रकीर्ण प्रकाश $z$ के अनुदिश
\emph{रैखिक रूप से ध्रुवित} है, उस तल के लंबवत जिसमें
पुंज और दृष्टि-रेखा हैं। और कोणों पर वह
आंशिक रूप से ध्रुवित है, जिसकी मात्रा $(1 - \cos^2\chi)/(1 + \cos^2\chi)$ है,
प्रकीर्णन कोण $\chi$ के लिए।
\end{proposition}

\begin{proof}
आपाती \omterm{def:b2:plane-waves-polarization:states}{प्राकृतिक प्रकाश} का $\vect E$ $y$ और $z$ के अनुदिश बराबर माध्य घटक
रखता है; $y$ के अनुदिश द्विध्रुव $y$ की ओर विकिरित नहीं करता ($\sin
\theta = 0$), जबकि $z$ के अनुदिश वाला $\propto\sin\theta = 1$ विकिरित करता है। कोण $\chi$ पर
$y$ घटक का योगदान तीव्रता में $\cos^2\chi$ से घट जाता है।
\end{proof}

\begin{remark}[नीला प्रकाश हम तक पहुँचता ही क्यों है]\label{rem:b2:dipole-radiation:why}
किसी पूर्ण एकसमान माध्य में पड़ोसी अणुओं से प्रकीर्ण तरंगें आगे को छोड़
हर दिशा में विनाशी व्यतिकरण कर लेतीं
(अपवर्तनांक वही आगे-प्रकीर्ण तरंग है)। हवा के घनत्व के
\emph{उतार-चढ़ाव} --- अणु किसी जालक पर नहीं बैठे --- ही कोई शुद्ध प्रकीर्ण तीव्रता
छोड़ जाते हैं, जो अणुओं की संख्या के समानुपाती है, जैसा ऊपर एक के लिए
निकाला गया (आइंस्टाइन, 1910)। वही यादृच्छिकता आकाश के प्रकाश को बिंदु-बिंदु
पर असंसक्त बना देती है।
\end{remark}

\begin{method}[विकिरण के आकलन]\label{met:b2:dipole-radiation:method}
(1) द्विध्रुव पहचानिए: $p_0 = q\,\times$ आयाम, या किसी तार के लिए $I_0\ell/\omega$।
(2) दूरी $r$ पर क्षेत्र: $E \approx \mu_0\omega^2p_0\sin\theta/4\pi r$; और शक्ति:
$\mu_0\omega^4p_0^2/12\pi c$। (3) किसी त्वरित आवेश के लिए लार्मर लीजिए; किसी
ऐंटेना के लिए \omterm{ex:b2:dipole-radiation:antenna}{विकिरण प्रतिरोध}। (4) प्रकीर्णन के लिए माध्यम के उत्तर
(\omterm{prop:b2:waves-in-media:lorentz}{लोरेंत्स प्रतिरूप}) से प्रेरित द्विध्रुव निकालिए और विकिरित शक्ति को
आपाती तीव्रता से भाग दीजिए। (5) शून्य याद रखिए:
दोलन की अक्ष के अनुदिश कुछ नहीं, और इसीलिए
\qty{90}{\degree} पर ध्रुवण तथा \cref{ch:b2:wave-interfaces} का \omterm{prop:b2:wave-interfaces:brewster}{ब्रूस्टर कोण}।
\end{method}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:9}
\emph{दो ऐंटेना।} $d$ दूरी पर, $x$ के अनुदिश, दो ऊर्ध्वाधर
द्विध्रुव एक ही आयाम से खिलाए जाते हैं। (क) एक ही कला में, $d = \lambda/2$: किन
क्षैतिज दिशाओं में उनके दूर-क्षेत्र जुड़ते हैं, किनमें कट जाते हैं?
(चौड़ी-ओर पंक्ति)। (ख) विपरीत कला में, $d = \lambda/2$: वही (सिरा-दिशा)।
(ग) समपाद में, $d = \lambda/4$: दिखाइए कि पंक्ति केवल एक ही
ओर विकिरित करती है। (घ) AM प्रसारण केंद्र कई मीनारें क्यों लेते हैं, और
कोई कला-पंक्ति रडार बिना हिले अपनी पुंज कैसे मोड़ लेता है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:10}
\emph{विकिरण अवमंदन।} लोरेंत्स इलेक्ट्रॉन (\cref{ch:b2:waves-in-media}),
जो $\omega_0$ पर आयाम $x_0$ से दोलन कर रहा है, लार्मर के सूत्र से विकिरित करता है।
(क) माध्य विकिरित शक्ति; दोलक की ऊर्जा $\tfrac12m\omega_0^2x_0^2$;
दिखाइए कि ऊर्जा दर $\Gamma = e^2\omega_0^2/6\pi\varepsilon_0mc^3$ से क्षय होती है।
(ख) सोडियम रेखा (\qty{589}{nm}) के लिए $\Gamma$ का मान: \cref{exo:b2:waves-in-media:9}
में काम में लिए गए $\Gamma$ से तुलना कीजिए। (ग) जीवनकाल $1/\Gamma$ और रेखा की
प्राकृतिक चौड़ाई, $\Delta\lambda = \lambda^2\Gamma/2\pi c$। (घ) परमाणविक दोलक का गुणता गुणक
$\omega_0/\Gamma$।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:11}
\emph{मीनार।} कोई AM केंद्र \qty{1}{MHz} पर \qty{50}{kW} किसी चालक ज़मीन पर खड़ी
$\lambda/4$ ऊँचाई की ऊर्ध्वाधर मीनार से विकिरित करता है (ज़मीन किसी
दर्पण की तरह काम करती है, जिससे मीनार किसी \omterm{ex:b2:dipole-radiation:antenna}{अर्ध-तरंग द्विध्रुव} की आधी बन जाती है: $R_{\text{विकिरण}} \approx
\qty{36}{\ohm}$, और शक्ति केवल ऊपरी अर्ध-आकाश में जाती है)। (क)
ऊँचाई; तली पर धारा। (ख) क्षैतिज तल में \qty{10}{km} पर क्षेत्र-आयाम
(द्विध्रुव आकृति लीजिए, ऊपरी अर्ध-गोले पर $\mathcal P$
$= \int\langle\Pi\rangle\dd S$; और ठीक-ठाक आकलन के लिए क्षितिज पर $\langle\Pi\rangle = 1.5\,
\mathcal P/2\pi r^2$ लीजिए)। (ग) वहाँ \qty{1}{m} के किसी अभिग्राही छड़ में
विद्युत् वाहक बल। (घ) ज़मीन की चालकता क्यों मायने रखती है, और मीनार के नीचे
त्रिज्य ताँबे के तार क्यों गाड़े जाते हैं?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:12}
\emph{प्रकीर्णन और अपवर्तनांक।} किसी विरल गैस के लिए सारे अणुओं की
आगे-प्रकीर्ण तरंगें संसक्त रूप से जुड़ती हैं और अपवर्तनांक बना देती हैं,
$n - 1 = N\alpha/2\varepsilon_0$, जिसमें $\alpha = e^2/m\omega_0^2$ ध्रुवणता है; और
रैले अनुप्रस्थ काट $\sigma = (8\pi/3)r_e^2(\omega/\omega_0)^4$ है। (क)
$\sigma$ को $n$, $N$ और $\lambda$ के पदों में लिखिए: $\sigma = 8\pi^3(n^2 - 1)^2/3N^2\lambda^4$। (ख)
\qty{550}{nm} पर हवा के लिए ($n - 1 = 2.9 \times 10^{-4}$, $N = \qty{2.5e25}{m^{-3}}$):
$\sigma$ और माध्य मुक्त पथ $1/N\sigma$। (ग) वायुमंडल की ऊर्ध्वाधर प्रकाशिक गहराई
(तुल्य ऊँचाई \qty{8}{km}); \cref{exo:b2:dipole-radiation:8} के $0.1$ से तुलना
कीजिए। (घ) इस संबंध (रैले,
1899) से आवोगाद्रो संख्या का कोई मान क्यों निकल आया?
\end{exercise}

\section{समस्या: मीनार, आकाश और इलेक्ट्रॉन}

\begin{problem}[{सप्ताहांत समस्या --- तीन ऐंटेना: कोई प्रसारण
मीनार, धूप में हवा का कोई अणु, और कोई मुक्त इलेक्ट्रॉन}]
\label{pb:b2:dipole-radiation:1}

\medskip
\noindent\textbf{भाग I --- मीनार।}
$f = \qty{1.0}{MHz}$ का कोई मध्यम-तरंग केंद्र किसी चालक ज़मीन पर खड़ी चतुर्थांश-तरंग
मीनार से $\mathcal P = \qty{100}{kW}$ विकिरित करता है ($R_{\text{विकिरण}} = \qty{36}{\ohm}$,
और ऊपरी अर्ध-आकाश पर किसी ऊर्ध्वाधर द्विध्रुव की आकृति, जैसा
\cref{exo:b2:dipole-radiation:11} में)।
\begin{enumerate}
  \item तरंगदैर्घ्य और मीनार की ऊँचाई; उसकी तली पर धारा का आयाम;
        और $R_{\text{विकिरण}}$ पर वोल्टता।
  \item तुल्य द्विध्रुव आघूर्ण का आयाम, मीनार को $p_0 = I_0\ell_{\text{प्रभावी}}/\omega$ वाला
        कोई छोटा द्विध्रुव मानते हुए, जहाँ चतुर्थांश-तरंग मीनार के लिए $\ell_{\text{प्रभावी}} \approx
        \lambda/2\pi$ (स्वीकृत)।
  \item क्षैतिज तल में \qty{10}{km} और \qty{100}{km} पर माध्य तीव्रता
        (ऊपरी अर्ध-गोले वाली आकृति लीजिए, जो क्षितिज पर अधिकतम है,
        $\langle\Pi\rangle = 1.5\mathcal P/2\pi r^2$); और क्षेत्र-आयाम।
  \item \qty{100}{km} पर \qty{1}{m} की किसी ऊर्ध्वाधर छड़ में विद्युत् वाहक बल; और
        किसी फ़ेराइट-छड़ ऐंटेना में ($N = 100$ फेरों की कोई कुंडली,
        \qty{1}{cm^2}, प्रभावी पारगम्यता $100$), जो तरंग के $\vect B$ के सामने हो: तुलना कीजिए।
  \item \qty{100}{km} पर प्रभावी क्षेत्रफल $A = 1.5\lambda^2/4\pi$ (द्विध्रुव वाला)
        के किसी ऐंटेना के साथ किसी मिलान वाले अभिग्राही को मिलती शक्ति।
  \item मीनार के चालकों का कुल हानि-प्रतिरोध
        \qty{2}{\ohm} है: दक्षता; और ऊष्मा के रूप में खोई शक्ति।
  \item क्या \qty{10}{km} पर कोई अभिग्राही \omterm{thm:b2:dipole-radiation:field}{विकिरण क्षेत्र} में है? और मीनार से
        \qty{100}{m} पर वाला?
  \item मध्यम-तरंग संकेत रात में और दूर तक क्यों पहुँचते हैं?
        (\cref{ch:b2:waves-in-media} की D परत याद कीजिए।)
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- आकाश।}
समुद्र-तल पर हवा: $N = \qty{2.5e25}{m^{-3}}$, रैले अनुप्रस्थ काट $\sigma =
\qty{4.6e-31}{m^2}\,(550\,\text{nm}/\lambda)^4$; ज़मीन पर धूप \qty{1.0}{kW/m^2};
और समुद्र-तल के घनत्व पर वायुमंडल की तुल्य मोटाई
\qty{8}{km}।
\begin{enumerate}[resume]
  \item \qty{450}{nm} और \qty{650}{nm} पर अनुप्रस्थ काट; और अनुपात।
  \item तीनों रंगों का माध्य मुक्त पथ; ऊर्ध्वाधर प्रकाशिक गहराई
        $\tau = N\sigma H$; और शिरोबिंदु पर सीधे पुंज से बाहर प्रकीर्ण हर रंग
        का अंश।
  \item ज़मीन पर हवा के प्रति घन मीटर से हरे प्रकाश की प्रकीर्ण
        शक्ति (\qty{550}{nm} के आसपास \qty{300}{W/m^2} धूप लीजिए);
        किस घन कोण में, और किस आकृति के साथ?
  \item सूर्य से \qty{90}{\degree} पर देखा हवा का \qty{1}{m^2} अनुप्रस्थ काट का कोई स्तंभ:
        उस स्तंभ से प्रेक्षक तक आते आकाश-प्रकाश की तीव्रता आँकिए
        (प्रति एकांक घन कोण उसकी ओर प्रकीर्ण शक्ति, \qty{8}{km} भर पर),
        और सीधे सूर्य से तुलना कीजिए; क्या आकाश की चमक ठीक कोटि की
        है (सूर्य की लगभग \qty{1e-5}{})?
  \item उस स्तंभ के प्रकाश के ध्रुवण की मात्रा; असल में वह कम क्यों
        है, और कुछ कीट उसे कैसे काम में लेते हैं?
  \item सूर्यास्त पर पथ $38$ वायुमंडल है: नीले और लाल के पारगत
        अंश; सूर्य का रंग; और सिर के ऊपर आकाश नीला क्यों बना
        रहता है।
  \item चंद्रमा पर पूरी धूप में भी आकाश काला क्यों है?
  \item मंगल पर हवा सौ गुना पतली है पर महीन धूल से भरी है
        (माइक्रोमीटर के कण): मंगल का आकाश मटमैला पीला और उसका
        सूर्यास्त नीला क्यों है?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- इलेक्ट्रॉन।}
\begin{enumerate}[resume]
  \item $r_e$ से टामसन अनुप्रस्थ काट।
  \item किसी X-किरण नलिका में \qty{100}{keV} का कोई इलेक्ट्रॉन ($v = \qty{1.6e8}{m/s}$,
        उसे अनापेक्षिकीय मानिए) टंग्स्टन में \qty{2}{\micro m} पर रोक दिया
        जाता है: त्वरण, लार्मर शक्ति, अवधि, विकिरित ऊर्जा
        और गतिज ऊर्जा में उसका अंश।
  \item नलिका जो सबसे छोटी तरंगदैर्घ्य उत्सर्जित कर सकती है (कोई फ़ोटॉन
        जो इलेक्ट्रॉन की पूरी ऊर्जा ले ले)।
  \item नलिका \qty{20}{mA} पर चलती है: विकिरित शक्ति; ऐनोड में ऊष्मा;
        और ऐनोड घूमता क्यों है।
  \item बोर त्रिज्या (\qty{5.3e-11}{m}, चाल \qty{2.2e6}{m/s}) पर किसी प्रोटॉन का
        चक्कर लगाता कोई चिरसम्मत इलेक्ट्रॉन: त्वरण, लार्मर
        शक्ति, और अपने \qty{13.6}{eV} को विकिरित कर देने का समय --- अर्थात्
        चिरसम्मत परमाणु का जीवनकाल।
  \item असली परमाणु सदा टिके रहते हैं: बताइए कि यहाँ चिरसम्मत भौतिकी
        क्या चूक जाती है, और \cref{ch:b2:schrodinger-wave-functions}
        किसी स्थायी अवस्था के बारे में क्या कहेगा।
  \item \qty{100}{m} त्रिज्या के किसी सिंक्रोट्रॉन में $v \approx c$ पर कोई इलेक्ट्रॉन:
        अनापेक्षिकीय \omterm{prop:b2:dipole-radiation:power}{लार्मर सूत्र} कितनी शक्ति देता है? (असली
        शक्ति $\gamma^4$ गुना बड़ी है, जिसमें $\gamma \sim 10^4$: वही
        सिंक्रोट्रॉन प्रकाश स्रोत।)
  \item वही इलेक्ट्रॉन \qty{1}{kW/m^2} की धूप में विराम पर: वह जो
        शक्ति फिर से विकिरित करता है (टामसन); और कोई वाट प्रकीर्ण
        करने के लिए ऐसे कितने इलेक्ट्रॉन चाहिए।
  \item तीनों ऐंटेना को किसी सारणी में समेटिए: द्विध्रुव आघूर्ण,
        आवृत्ति, शक्ति, आकृति।
\end{enumerate}
\end{problem}
