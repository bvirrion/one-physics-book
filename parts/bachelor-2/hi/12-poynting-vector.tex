\chapter{विद्युत्चुंबकीय ऊर्जा और पॉयनटिंग सदिश}\label{ch:b2:poynting-vector}

कोई बैटरी ताँबे के दो तारों से होकर किसी बल्ब को जलाती है। ऊर्जा कहाँ से
चलती है? ताँबे के भीतर से नहीं, यह पता चलता है: वहाँ इलेक्ट्रॉन
मिलीमीटर प्रति सेकंड से बहते हैं और लगभग कुछ भी नहीं ढोते। ऊर्जा
\emph{तारों के चारों ओर के आकाश} से होकर, विद्युत्चुंबकीय क्षेत्र में
बहती है, और बल्ब के तंतु में उसकी बग़लों से घुसती है। सूर्य किसी धूप वाली
छत के हर वर्ग मीटर को उसी तरह किलोवाट भर पहुँचाता है, और वह भी पंद्रह
करोड़ किलोमीटर निर्वात के पार। यह अध्याय \omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल के समीकरणों} से
विद्युत्चुंबकीय ऊर्जा का बहीखाता निकालता है --- अर्थात् किसी क्षेत्र में
कितनी ऊर्जा संचित है, और वह सदिश, पॉयनटिंग का, जो बताता है कि वह कहाँ
जाती है --- और उसे किसी तार, संधारित्र, कुंडली और प्रकाश-पुंज पर लगाता है।

\section{क्षेत्र का ऊर्जा-संतुलन}

\begin{theorem}[पॉयनटिंग की प्रमेय]\label{thm:b2:poynting-vector:poynting}
\index{पॉयनटिंग सदिश}\index{पॉयनटिंग की प्रमेय}\index{विद्युत्चुंबकीय ऊर्जा घनत्व}
\emph{\omterm{thm:b2:poynting-vector:poynting}{विद्युत्चुंबकीय ऊर्जा घनत्व}} और \emph{\omterm{thm:b2:poynting-vector:poynting}{पॉयनटिंग
सदिश}} इस तरह परिभाषित कीजिए:
\[
u = \frac{\varepsilon_0E^2}2 + \frac{B^2}{2\mu_0} \quad (\unit{J/m^3}) , \qquad
\vect\Pi = \frac{\vect E\wedge\vect B}{\mu_0} \quad (\unit{W/m^2}) .
\]
\omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल के समीकरण} हर बिंदु पर यह देते हैं:
\[
\frac{\partial u}{\partial t} + \operatorname{div}\vect\Pi = -\vect j\cdot\vect E ,
\]
और किसी भी स्थिर आयतन $V$ के लिए, जो $\Sigma$ से घिरा हो:
\[
\frac{\dd}{\dd t}\iiint_Vu\,\dd\tau = -\iint_\Sigma\vect\Pi\cdot\vect n\,\dd S - \iiint_V\vect j\cdot\vect E\,\dd\tau :
\]
अर्थात् $V$ के भीतर की क्षेत्र-ऊर्जा उतनी घटती है जितनी सीमा से बाहर
बहती है --- $\vect\Pi$ का अभिवाह --- और जितनी क्षेत्र आवेशों को देता है
($\vect j\cdot\vect E$, किसी चालक में जूल शक्ति)। $\vect\Pi$
ऊर्जा-अभिवाह घनत्व है: अर्थात् उसके अभिलंब एकांक पृष्ठ को पार करती
शक्ति।
\end{theorem}

\begin{proof}
मैक्सवेल--ऐंपियर से $\vect j = \operatorname{\vect{curl}}\vect B/\mu_0 - \varepsilon_0\partial_t
\vect E$, अतः $\vect j\cdot\vect E = \vect E\cdot\operatorname{\vect{curl}}\vect B/\mu_0 -
\partial_t(\varepsilon_0E^2/2)$। सर्वसमिका $\operatorname{div}(\vect E\wedge\vect B) = \vect B
\cdot\operatorname{\vect{curl}}\vect E - \vect E\cdot\operatorname{\vect{curl}}\vect B$ (उसे
घटकों पर जाँचिए) और मैक्सवेल--फ़ैराडे से $\vect E\cdot\operatorname{\vect{curl}}
\vect B = -\operatorname{div}(\vect E\wedge\vect B) - \vect B\cdot\partial_t\vect B = -\mu_0
\operatorname{div}\vect\Pi - \partial_t(B^2/2)$। अब जोड़िए: $\vect j\cdot\vect E = -\operatorname{div}
\vect\Pi - \partial_tu$। फिर $V$ पर समाकलित कीजिए और \omterm{thm:b2:maxwell-equations:theorems}{अपसरण प्रमेय} लगाइए।
\end{proof}

\begin{remark}[क्या तय है और क्या नहीं]\label{rem:b2:poynting-vector:interpretation}
प्रमेय केवल बंद पृष्ठों से $\vect\Pi$ का अभिवाह और कुल $u$ तय करती है;
और $\vect\Pi$ में कोई भी अपसरण-रहित क्षेत्र जोड़ देने से कोई प्रेक्षणीय
बात नहीं बदलती। ऊपर वाला चुनाव सबसे सरल है, वही प्रकाश के ढोए संवेग से
मेल खाता है (नीचे), और वही सर्वत्र
काम में लिया जाता है। घनत्व $\varepsilon_0E^2/2$ और $B^2/2\mu_0$ वर्ष 1 के खंड के
संधारित्र और प्रेरक की ऊर्जाएँ हैं,
जो अब आकाश के हर बिंदु को सौंप दी गई हैं: \qty{1}{MV/m} का कोई क्षेत्र (हवा
की भंजन-सीमा) \qty{4.4}{J/m^3} संचित करता है, और \qty{1}{T} का कोई क्षेत्र
\qty{0.4}{MJ/m^3} --- अर्थात् एक लाख गुना अधिक, और इसीलिए मेगाजूल तक
ऊर्जा-भंडारण संधारित्रों में नहीं, चुंबकों में होता है।
\end{remark}

\section{ऊर्जा कहाँ बहती है: तीन परिपथ}

\begin{example}[प्रतिरोधी तार]\label{ex:b2:poynting-vector:wire}
$a$ त्रिज्या का कोई बेलनाकार तार स्थायी धारा $I$ ढोता है, अपनी अक्ष के
अनुदिश क्षेत्र $E = j/\gamma$ के अधीन; और उसके ठीक बाहर $B = \mu_0I/2\pi a$
(दिगंशी)। पृष्ठ पर $\vect\Pi = \vect E\wedge\vect B/\mu_0$
\emph{त्रिज्या के अनुदिश भीतर की ओर} संकेत करता है, जिसका परिमाण $EI/2\pi a$ है; और किसी
$L$ लंबाई के पार्श्व पृष्ठ से उसका अभिवाह $EI/2\pi a\times2\pi aL = (EL)I = UI =
RI^2$ है: अर्थात् जूल ऊष्मा तार में उसकी बग़लों से घुसती है, जिसे
चारों ओर का क्षेत्र पहुँचाता है, न कि इलेक्ट्रॉन तार के नीचे ढोकर।
भीतर, $\Pi(r) = (E/\mu_0)(\mu_0Ir/2\pi a^2) = EIr/2\pi a^2$: अर्थात् भीतर की ओर बहाव
अक्ष की ओर घटता जाता है, क्योंकि ऊर्जा परत-दर-परत जमा होती जाती है।
\end{example}

\begin{example}[संधारित्र और परिनालिका]\label{ex:b2:poynting-vector:capsol}
वृत्ताकार प्लेटों वाला कोई संधारित्र, आवेशित होता हुआ: $\vect E$ अक्षीय, $\vect B =
\mu_0\varepsilon_0(r/2)\dot E\,\vect e_\theta$ (\cref{ch:b2:maxwell-equations}), अतः
$\vect\Pi = -\varepsilon_0(r/2)E\dot E\,\vect e_r$, भीतर की ओर; और
$R$ त्रिज्या तथा $d$ ऊँचाई के बेलन से उसका अभिवाह $\varepsilon_0\pi R^2d\,E\dot E = \dd(\tfrac12
\varepsilon_0E^2\,\pi R^2d)/\dd t$ है: अर्थात् विद्युत ऊर्जा किनारे से घुसती है। कोई
परिनालिका, जिसकी धारा बढ़ती हो: $\vect B$ अक्षीय, प्रेरित $\vect E = -(r/2)
\dot B\,\vect e_\theta$, $\vect\Pi = -(rB\dot B/2\mu_0)\vect e_r$, फिर भीतर की ओर, और
अभिवाह $\dd(B^2/2\mu_0\times\text{आयतन})/\dd t$। हर दशा में ऊर्जा
उपकरण तक उसके बाहर के क्षेत्र से होकर पहुँचती है; और तार केवल
क्षेत्र को राह दिखाते हैं।
\end{example}

\begin{remark}[किसी स्थैतिक क्रॉस-क्षेत्र में ऊर्जा]\label{rem:b2:poynting-vector:static}
किसी स्थायी चुंबक के क्षेत्र में रखे किसी आवेशित संधारित्र की प्लेटों के
बीच हर जगह $\vect\Pi = \vect E\wedge\vect B/\mu_0 \ne \vect 0$ होता है,
यद्यपि कुछ बदलता नहीं और कुछ गरम नहीं होता। इसमें कोई विरोध नहीं:
वहाँ $\vect\Pi$ का \omterm{def:b2:maxwell-equations:operators}{अपसरण} शून्य है, उसकी रेखाएँ अपने ऊपर ही बंद होती हैं, और
किसी भी बंद पृष्ठ से उसका अभिवाह लुप्त हो जाता है --- और प्रमेय किसी ऐसे
परिचालित बहाव के बारे में कुछ नहीं कहती जो कुछ पहुँचाता ही नहीं। केवल बंद
पृष्ठों से अभिवाह ही भौतिक हैं।
\end{remark}

\section{प्रकाश: तीव्रता और विकिरण दाब}

\begin{proposition}[किसी समतल तरंग की ऊर्जा]\label{prop:b2:poynting-vector:wave}
\index{तीव्रता (विद्युत्चुंबकीय)}\index{विकिरण दाब}\index{विकिरण-तीव्रता}
निर्वात में किसी समतल तरंग के लिए --- अर्थात् \cref{ch:b2:plane-waves-polarization} का हल,
जिसमें $\vect B = \vect k\wedge\vect E/\omega$, $B = E/c$ और $\vect E \perp \vect B$ ---
विद्युत और चुंबकीय ऊर्जा-घनत्व बराबर होते हैं, तथा
\[
u = \varepsilon_0E^2 , \qquad \vect\Pi = \varepsilon_0cE^2\,\vect u = cu\,\vect u ,
\]
होता है, जहाँ $\vect u$ संचरण की दिशा है: अर्थात् ऊर्जा $c$ से चलती है।
$E_0$ आयाम की किसी ज्यावक्रीय तरंग के लिए \emph{तीव्रता}
(विकिरण-तीव्रता) माध्य अभिवाह है,
\[
I = \langle\Pi\rangle = \frac{\varepsilon_0cE_0^2}2 = \frac{E_0^2}{2\mu_0c} ,
\]
और तरंग संवेग-घनत्व $\vect\Pi/c^2$ ढोती है (स्वीकृत), अतः
वह किसी पृष्ठ पर, जिस पर वह अभिलंब आपतन से टकराए,
\emph{\omterm{prop:b2:poynting-vector:wave}{विकिरण दाब}} $P = I/c$ लगाती है यदि वह सोख लिया जाए और $2I/c$ यदि परावर्तित हो।
\end{proposition}

\begin{proof}
$B^2/2\mu_0 = E^2/2\mu_0c^2 = \varepsilon_0E^2/2$; $\vect E\wedge\vect B/\mu_0 = E^2/\mu_0c =
\varepsilon_0cE^2$, $\vect k$ के अनुदिश; और $\langle\cos^2\rangle = \tfrac12$। संवेग-घनत्व
$\vect\Pi/c^2$ सोखते पृष्ठ के आवेशों पर लोरेंत्ज़ बल से निकलता है
(या आपेक्षिकता से: प्रकाश के लिए $p = E/c$): प्रति एकांक समय आता संवेग
$\Pi S\,\dd t/c^2\times c$, किसी पृष्ठ $S$ पर, कोई
बल $\Pi S/c$ है।
\end{proof}

\begin{example}[सूर्य का प्रकाश, लेज़र, सूक्ष्मतरंग]\label{ex:b2:poynting-vector:orders}
वायुमंडल के शिखर पर सूर्य का प्रकाश, $I = \qty{1.36}{kW/m^2}$: $E_0 =
\sqrt{2I/\varepsilon_0c} = \qty{1.0}{kV/m}$, $B_0 = E_0/c = \qty{3.4}{\micro T}$, ऊर्जा
घनत्व \qty{4.5}{\micro J/m^3}, दाब \qty{4.5}{\micro Pa} --- अर्थात् किसी वर्ग किलोमीटर पर
\qty{0.7}{kg} भर का बल, फिर भी किसी सौर-पाल को मोड़ने और किसी धूमकेतु की धूल को
उसकी पूँछ में उड़ा देने के लिए काफ़ी। \qty{1}{mm^2} के धब्बे पर \qty{1}{mW} का कोई
लेज़र सूचक: \qty{1}{kW/m^2}, अर्थात् सूर्य जैसा; और \qty{1}{\micro m^2} पर केंद्रित:
\qty{1}{GW/m^2}, $E_0 = \qty{0.9}{MV/m}$, हवा के भंजन के पास। और
$\qty{30}{L}$ भरती किसी \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंग} वाला \qty{1}{kW} का सूक्ष्मतरंग-चूल्हा: ऊर्जा
घनत्व $\sim 10^{-7}$ J/m$^3$ और $E_0$ कुछ किलोवोल्ट प्रति मीटर।
\end{example}

\begin{method}[ऊर्जा का बहीखाता]\label{met:b2:poynting-vector:method}
(1) $\vect E$ और $\vect B$ निकालिए (स्थैतिकी, अर्ध-स्थैतिकी या तरंगें)। (2)
$\vect\Pi = \vect E\wedge\vect B/\mu_0$ और $u$ बनाइए। (3) उपकरण के चारों ओर कोई
बंद पृष्ठ चुनिए और $\vect\Pi$ का अभिवाह निकालिए: वह जूल शक्ति और भीतर संचित
ऊर्जा के बदलने की दर के योग के बराबर होना चाहिए
--- अर्थात् क्षेत्रों की एक जाँच। (4) किसी तरंग के लिए $\Pi$ का माध्य
तीव्रता है; दाब के लिए उसे $c$ से बाँटिए, और फ़ोटॉन-अभिवाह के लिए
फ़ोटॉन-ऊर्जा $hf$ से।
\end{method}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:3}

\begin{exercise}[$\star$]\label{exo:b2:poynting-vector:4}
\qty{100}{m} $\times$ \qty{100}{m} का कोई सौर-पाल, जो पूरा परावर्तक हो,

गुरुत्व (\qty{5.9e-3}{N/kg}) से तुलना।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:5}
$a = \qty{1}{mm}$ त्रिज्या का कोई ताँबे का तार \qty{10}{A} ढोता है ($\gamma =
\qty{6e7}{S/m}$)। (क) भीतर $E$, पृष्ठ पर $B$, और पृष्ठ पर $\vect\Pi$
(दिशा और परिमाण)। (ख) तार के \qty{1}{m} में $\vect\Pi$ का अभिवाह;
$RI^2$ से तुलना कीजिए। (ग) तार के भीतर $\Pi(r)$, और $r$ तथा $r + \dd r$ के बीच
जमा शक्ति; जाँचिए कि वह $j^2/\gamma\cdot2\pi r\,\dd r$ है।
(घ) ऊर्जा आती कहाँ से है --- बैटरी और तार के बीच के क्षेत्र का
वर्णन कीजिए।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:6}
वृत्ताकार प्लेटों ($R$, $d$) का कोई संधारित्र, जो $I$ से आवेशित होता है। (क) प्लेटों के बीच $\vect E$, $\vect B$,
$\vect\Pi$। (ख) पार्श्व बेलन से $\vect\Pi$ का अभिवाह;
दिखाइए कि वह $\dd W_e/\dd t$ के बराबर है। (ग) $R = \qty{5}{cm}$, $d =
\qty{1}{mm}$, $I = \qty{1}{A}$ के लिए, उस क्षण जब $E = \qty{1}{MV/m}$: किनारे पर $\Pi$
और भीतर आती शक्ति। (घ) क्या चुंबकीय ऊर्जा नगण्य है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:7}
कोई लंबी परिनालिका ($n$, त्रिज्या $R$), जिसकी धारा $I(t)$ बढ़ती है। (क)
भीतर प्रेरित $\vect E$; $\vect\Pi$; और दिशा। (ख) प्रति एकांक लंबाई पार्श्व पृष्ठ से
अभिवाह, और $\dd(B^2/2\mu_0\cdot\pi R^2)/\dd t$। (ग)
परिनालिका के ठीक बाहर (जहाँ $\vect B = \vect 0$) $\vect\Pi$ क्या है?
तब ऊर्जा भीतर कैसे आती है? (सोचिए कि धारा कहाँ बहती है:
वेष्टन कोई पतली चादर है; उसके ठीक भीतर का पृष्ठ लीजिए।)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:8}
\emph{किसी केबल की शक्ति कहाँ चलती है।} कोई \omterm{prop:b2:dispersion-wave-packets:coax}{समाक्ष केबल} (त्रिज्याएँ $a$,
$b$) क्रोड में स्थायी धारा $I$ ढोती है और चोटी में वापस, उनके बीच
वोल्टता $U$ के अधीन (पूर्ण चालक)। (क) $\vect E$ और $\vect B$ परावैद्युत
($a < r < b$) में, गाउस और ऐंपियर से। (ख) $\vect\Pi$:
दिशा और परिमाण। (ग) $\vect\Pi$ को छल्ले $a < r
< b$ पर समाकलित कीजिए: दिखाइए कि परिणाम $UI$ है --- अर्थात् पूरी शक्ति
विद्युतरोधी में चलती है। (घ) कुछ प्रतिरोधी क्रोड के साथ $\vect\Pi$ किस
दिशा में झुकता है, और क्रोड में उसका अभिवाह किसके बराबर होता है?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:poynting-vector:9}

उड़ा दिया जाता है। (ग) धूमकेतुओं की पूँछें सूर्य से दूर की ओर क्यों होती हैं, और वे

\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:poynting-vector:10}

परिपथ की चुंबकीय ऊर्जा आरंभ और अंत में संतुलन में क्यों नहीं

\end{exercise}

जाती हैं, और उन्हें कौन बंद करता है? (घ) कोई सुझाव: अंतराल में रखे किसी
छोटे अवशोषक से ऊर्जा-बहाव नापिए --- क्या वह गरम होगा? क्यों नहीं?

\end{exercise}

\section{समस्या: किसी लाइन, किसी पट्ट और किसी चूल्हे की ऊर्जा}

\begin{problem}[{सप्ताहांत समस्या --- विद्युत्चुंबकीय ऊर्जा का पीछा:
किसी विद्युत-लाइन से किसी घर तक, सूर्य से किसी सौर पट्ट तक,
और किसी मैग्नेट्रॉन से पानी के किसी प्याले तक}]
\label{pb:b2:poynting-vector:1}

\medskip
\noindent\textbf{भाग I --- लाइन।}
कोई समाक्ष शक्ति-केबल (क्रोड-त्रिज्या $a = \qty{1.0}{cm}$, चोटी-त्रिज्या $b =
\qty{3.0}{cm}$, परावैद्युत $\varepsilon_r = 2.3$) दिष्ट धारा $I =
\qty{500}{A}$ ढोती है, $U = \qty{20}{kV}$ पर; और क्रोड की चालकता $\gamma =
\qty{6e7}{S/m}$ है।
\begin{enumerate}
  \item परावैद्युत में क्षेत्र $\vect E(r)$ और $\vect B(r)$ (पहले पूर्ण
        चालक मानकर); और मान, $r = a$ पर।
  \item परावैद्युत में \omterm{thm:b2:poynting-vector:poynting}{पॉयनटिंग सदिश}; और छल्लेदार काट से उसका अभिवाह:
        जाँचिए कि वह $UI$ के बराबर है।
  \item ऊर्जा-घनत्व $\varepsilon_0\varepsilon_rE^2/2$ और $B^2/2\mu_0$, $r = a$ पर
        (किसी परावैद्युत में विद्युत ऊर्जा-घनत्व $\varepsilon_r$ ढोता है);
        कौन भारी पड़ता है, और कितना?
  \item ऊर्जा प्रकाश की चाल के किस अंश से ``चलती'' है
        ($\Pi/u$ लीजिए, $r = a$ पर)? $c/\sqrt{\varepsilon_r}$ से तुलना कीजिए।
  \item अब क्रोड प्रतिरोधी है: उसके भीतर क्षेत्र $E_z$; और उसके पृष्ठ के
        ठीक बाहर \omterm{thm:b2:poynting-vector:poynting}{पॉयनटिंग सदिश} को कोई त्रिज्यीय
        घटक मिल जाता है: उसका परिमाण, और वह प्रति मीटर क्रोड को जितनी शक्ति
        पहुँचाता है; प्रति मीटर $RI^2$ के सामने जाँचिए।
  \item $\vect\Pi$ के त्रिज्यीय और अक्षीय घटकों का अनुपात, $r =
        a$ पर:
        इससे यह क्या पता चलता है कि ऊर्जा कहाँ जाती है?
  \item केबल की प्रति मीटर चुंबकीय ऊर्जा, $\int B^2/2\mu_0\,\dd\tau$, और
        प्रति मीटर विद्युत ऊर्जा, $\int\varepsilon_0\varepsilon_rE^2/2\,\dd\tau$;
        और \cref{ch:b2:dispersion-wave-packets} के प्रति मीटर प्रेरकत्व तथा
        धारिता के साथ उन्हें $\tfrac12\Lambda I^2$ और $\tfrac12\Gamma U^2$ के रूप में फिर से पाइए।
  \item \qty{10}{km} की कोई लाइन: पहुँचाई गई शक्ति, खोई शक्ति, और अंश।
        $U = \qty{200}{kV}$ तथा $I = \qty{50}{A}$ (वही शक्ति) के साथ दोहराइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पट्ट।}
किसी साफ़ दिन दोपहर सूर्य का प्रकाश: $I = \qty{1.0}{kW/m^2}$; \qty{1.6}{m^2} का
कोई प्रकाश-वोल्टीय पट्ट, जिसकी दक्षता $20\%$ है; और औसतन \qty{2.0}{eV}
की फ़ोटॉन-ऊर्जा।
\begin{enumerate}[resume]
  \item पुंज में $E_0$, $B_0$, ऊर्जा घनत्व और फ़ोटॉन-अभिवाह।
  \item दी गई वैद्युत शक्ति; निकालने लायक़ ऊष्मा; और यदि पट्ट दोनों फलकों से
        $h = \qty{10}{W/(m^2.K)}$ पर ऊष्मा खोए तो ताप-वृद्धि।
  \item पट्ट पर \omterm{prop:b2:poynting-vector:wave}{विकिरण दाब} (अवशोषी); बल; और उसकी
        उसके भार (\qty{18}{kg}) से तुलना।
  \item प्रति दिन (\qty{6}{h} तुल्य पूर्ण धूप) और प्रति वर्ष मिली ऊर्जा;
        और किसी घर की \qty{3500}{kWh} प्रति वर्ष की आपूर्ति हेतु आवश्यक
        क्षेत्रफल।
  \item पट्ट \qty{10.7}{A} पर \qty{30}{V} देता है: शक्ति जाँचिए;
        और इन्वर्टर तक जाते दो-तार केबल के \qty{5}{m} पर पॉयनटिंग
        अभिवाह --- क्षेत्र वाले चित्र में वह शक्ति कहाँ से
        चलती है?
  \item पृथ्वी की कक्षा पर सूर्य \qty{1.36}{kW/m^2} विकिरित करता है: सूर्य की
        कुल शक्ति; और वह द्रव्यमान जो वह प्रति सेकंड बदल देता है ($E = mc^2$,
        $c = \qty{3e8}{m/s}$)।
  \item प्रति सेकंड पट्ट से टकराते फ़ोटॉनों की संख्या।
  \item पृथ्वी पर फ़ोटॉन-संवेग का अभिवाह: पृथ्वी पर सूर्य का कुल
        विकिरण बल (चक्रिका-त्रिज्या \qty{6400}{km}, अवशोषी);
        गुरुत्व (\qty{3.5e22}{N}) से तुलना कीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- चूल्हा।}
कोई सूक्ष्मतरंग-चूल्हा \qty{2.45}{GHz} पर \qty{800}{W} किसी \qty{30}{L} की गुहा में
पहुँचाता है; और \qty{250}{g} पानी का कोई प्याला उसका $80\%$ सोख लेता है।
\begin{enumerate}[resume]
  \item तरंगदैर्घ्य; क्या गुहा उसके सामने बड़ी है? दीवारें धातु की
        क्यों होती हैं?
  \item पानी को \qty{20}{\degreeCelsius} से \qty{80}{\degreeCelsius} तक गरम करने में लगा समय
        ($c = \qty{4.18}{kJ/(kg.K)}$); और प्रति एकांक आयतन सोखी गई शक्ति।
  \item यदि शक्ति प्याले की \qty{50}{cm^2} काट से होकर किसी समतल तरंग की
        तरह बहती, तो तीव्रता और $E_0$; असली क्षेत्र
        तुलनीय आयाम की \omterm{prop:b2:waves-on-strings:modes}{अप्रगामी तरंगें} हैं --- गरम धब्बे कितनी दूर-दूर
        होते हैं, और तश्तरी क्यों घूमती है?
  \item यदि गुहा का गुणता गुणक खाली होने पर
        $Q = 2\pi\times\text{(संचित ऊर्जा)}/\text{(प्रति आवर्तकाल खोई ऊर्जा)}
        \approx 10^3$ हो: संचित ऊर्जा, माध्य ऊर्जा-घनत्व, $E_0$
        --- और हवा के भंजन (\qty{3}{MV/m}) से तुलना।
  \item पानी इसलिए सोखता है कि उसके अणु द्विध्रुव हैं और
        \qty{2.45}{GHz} पर चलाए जाते हैं: तब स्थानिक जूल-जैसी शक्ति $\tfrac12\omega\varepsilon_0
        \varepsilon''E_0^2$ है, जहाँ पानी के लिए $\varepsilon'' \approx 10$: प्रश्न 16 के
        शक्ति-घनत्व के लिए पानी में आवश्यक $E_0$; क्या वह संगत है?
  \item \qty{2.45}{GHz} पर फ़ोटॉन-ऊर्जा eV में, और प्याले द्वारा प्रति सेकंड
        सोखे गए फ़ोटॉनों की संख्या; सूक्ष्मतरंगें गरम तो कर सकती हैं पर
        आयनित क्यों नहीं (आयनन को लगभग \qty{10}{eV} चाहिए)।
  \item दीवारों पर गुहा-क्षेत्र का \omterm{prop:b2:poynting-vector:wave}{विकिरण दाब} (ऊर्जा-घनत्व $u$ की
        कोटि का): क्या उसकी चिंता करने लायक़ है?
  \item कोई खाली चूल्हा (कोई भार नहीं) पॉयनटिंग की भाषा में
        क्षति का ख़तरा क्यों उठाता है?
  \item लाइन, पट्ट और चूल्हे के लिए एक सारणी में सार दीजिए: ऊर्जा का
        वाहक, $\vect\Pi$ की दिशा, शक्ति, और उसे कौन
        बदलता है।
\end{enumerate}
\end{problem}
