\chapter{ऊष्मागतिकीय संतुलन और खुले तंत्र}\label{ch:b2:open-systems}

वर्ष 1 के खंड की ऊष्मागतिकी बंद तंत्रों से निपटती थी --- किसी बेलन में गैस
का कोई नियत द्रव्यमान, जिसे दबाया, गरम किया, फैलाया जाता है। पर जो मशीनें
असल में हमारे घर गरम करती हैं, हमारा भोजन ठंडा करती हैं और हमारी बिजली
बनाती हैं, उन्हें कोई \emph{प्रवाह} पार करता है: किसी बिजलीघर की टरबाइन से
भाप प्रति सेकंड सैकड़ों किलोग्राम बहती है, किसी ऊष्मा-पंप के संपीडक,
संघनित्र, वाल्व और वाष्पित्र से कोई प्रशीतक घूमता है, और किसी जेट इंजन के
संपीडक तथा दहन-कक्ष से हवा बहती है। हर घटक कोई \emph{\omterm{def:b2:open-systems:cv}{खुला
तंत्र}} है, अर्थात् दिक् का कोई ऐसा इलाक़ा जिसमें पदार्थ घुसता और निकलता
रहता है, और उसके लिए पहला तथा दूसरा नियम फिर से लिखने पड़ते हैं। यह
फिर-से-लिखना छोटा है --- कुंजी यह है कि तंत्र में धकेला गया तरल, अपनी
आंतरिक ऊर्जा के अलावा, उसे धकेलने में किया गया कार्य भी ले चलता है, और
इसीलिए किसी प्रवाह की स्वाभाविक ऊर्जा \emph{\omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी}} है, आंतरिक ऊर्जा नहीं।
यह अध्याय \omterm{def:b2:open-systems:cv}{स्थायी प्रवाह} में खुले तंत्रों के लिए द्रव्यमान, ऊर्जा और
एंट्रॉपी के संतुलन खड़े करता है, उन्हें प्रारंभिक यंत्रों पर लगाता है ---
\omterm{prop:b2:open-systems:devices}{तुंड}, \omterm{prop:b2:open-systems:devices}{थ्रॉटल}, संपीडक, टरबाइन, \omterm{prop:b2:open-systems:devices}{ऊष्मा-विनिमयक} --- और उन यंत्रों को उन दो
चक्रों में जोड़ता है जिन्हें सप्ताहांत समस्या नापती है: कोई
ऊष्मा-पंप और कोई भाप-बिजलीघर।

\section{खुले तंत्र और द्रव्यमान संतुलन}

\begin{definition}[खुला तंत्र, नियंत्रण आयतन, स्थायी प्रवाह]\label{def:b2:open-systems:cv}
कोई \emph{खुला तंत्र}\index{खुला तंत्र} दिक् का कोई नियत इलाक़ा है,
अर्थात् \emph{नियंत्रण आयतन}\index{नियंत्रण आयतन} $\Sigma$, जो किसी ऐसी
सतह से घिरा हो जिससे होकर तरल घुसता और निकलता है (प्रवेश तथा निर्गम
काट) और जिससे होकर बाहर के साथ ऊष्मा तथा कार्य बदले जा सकें;
और ख़ुद प्रवाह के अलावा चलते पुरज़ों (किसी धुरी, किसी पिस्टन) से पाया गया
कार्य \emph{उपयोगी कार्य}\index{उपयोगी कार्य}
$W_{\text{उप}}$ (धुरी कार्य) है। प्रवाह तब \emph{स्थायी}\index{स्थायी
प्रवाह} है जब $\Sigma$ के भीतर हर क्षेत्र काल-निरपेक्ष हो: तब $\Sigma$ में
समाए द्रव्यमान, ऊर्जा और एंट्रॉपी अचर हैं।
\end{definition}

\begin{proposition}[द्रव्यमान संतुलन]\label{prop:b2:open-systems:mass}
क्षेत्रफल $S$ के किसी काट से होकर, जहाँ घनत्व $\rho$ का तरल उसके अभिलंब
माध्य चाल $c$ से चले, \emph{द्रव्यमान प्रवाह
दर}\index{द्रव्यमान प्रवाह दर} $\dot m = \rho c S$ (\unit{kg/s} में) बहती है। किसी
\omterm{def:b2:open-systems:cv}{स्थायी प्रवाह} में जो घुसता है वही निकलता है: $\sum_{\text{भीतर}}\dot m = \sum_{\text{बाहर}}
\dot m$; और किसी एक धारा के लिए $\dot m$ प्रवेश तथा निर्गम पर वही है,
$\rho_1c_1S_1 = \rho_2c_2S_2$ --- अर्थात् \cref{ch:b2:fluid-kinematics} का
\omterm{thm:b2:fluid-kinematics:continuity}{सांतत्य समीकरण}, काट पर समाकलित।
\end{proposition}

\begin{proof}
$\Sigma$ में द्रव्यमान अचर है, अतः फ्लक्स संतुलित होते हैं।
\end{proof}

\section{किसी खुले तंत्र के लिए पहला नियम}

\begin{theorem}[किसी स्थायी एक-धारा प्रवाह का ऊर्जा संतुलन]\label{thm:b2:open-systems:firstlaw}
\omterm{def:b2:fluid-kinematics:flowrate}{द्रव्यमान प्रवाह दर} $\dot m$ के किसी \omterm{def:b2:open-systems:cv}{स्थायी प्रवाह} के लिए, जो अवस्था 1 में
घुसे और अवस्था 2 में निकले, तथा उपयोगी (धुरी) शक्ति
$P_{\text{उप}}$ और तापीय शक्ति $P_{\text{ताप}}$ पाए,
\[
\dot m\,\Big[(h_2 - h_1) + \tfrac12(c_2^2 - c_1^2) + g(z_2 - z_1)\Big]
= P_{\text{उप}} + P_{\text{ताप}} ,
\]
या, तंत्र को पार करते तरल के प्रति एकांक द्रव्यमान,
\[
\Delta h + \Delta\big(\tfrac12c^2\big) + g\,\Delta z = w_{\text{उप}} + q .
\]
\emph{एंथैल्पी}\index{एंथैल्पी (किसी प्रवाह में)} $h = u + Pv$ आंतरिक ऊर्जा
की जगह ले लेती है: $Pv$ \emph{प्रवाह कार्य} है, अर्थात् किसी एकांक द्रव्यमान
को भीतर धकेलने के लिए ऊपरी तरल का किया कार्य, घटा उसे बाहर धकेलने का।
\end{theorem}

\begin{proof}
$\Sigma$ के भीतर के तरल, $t$ पर, और घुसने को तैयार टुकड़े $\dd m = \dot m\,\dd t$
से बने बंद तंत्र के पीछे चलिए; $t + \dd t$ पर वह
$\Sigma$ के भीतर के तरल (वही अवस्था, \omterm{def:b2:open-systems:cv}{स्थायी}) और निकल चुके टुकड़े $\dd m$
से बना है। उसकी ऊर्जा
$\dd m\,[(u_2 + \tfrac12c_2^2 + gz_2) - (u_1 + \tfrac12c_1^2 + gz_1)]$ से बदली। उसने
$P_{\text{उप}}\dd t + P_{\text{ताप}}\dd t$ पाया, और साथ में प्रवेश पर दाब-बलों का
कार्य, $P_1S_1 \times c_1\dd t = P_1v_1\,\dd m$
(टुकड़े को भीतर धकेलते हुए), घटा निर्गम पर वाला, $P_2v_2\,\dd m$। इस बंद
तंत्र के लिए पहला नियम, $u + Pv = h$ के साथ, परिणाम दे देता है।
\end{proof}

\begin{remark}[पहला नियम काम में लेना]\label{rem:b2:open-systems:using}
किसी आदर्श गैस के लिए $\Delta h = c_p\Delta T$; किसी द्रव के लिए $\Delta h \approx
c\,\Delta T + v\,\Delta P$ (दूसरा पद पंप का कार्य है); और किसी वाष्प के लिए
$h$ सारणियों या चार्टों से पढ़ लिया जाता है। गतिज पद प्रायः
नगण्य है: \qty{10}{m/s} पर $c^2/2 = \qty{50}{J/kg}$, हवा के एक केल्विन के
\qty{1}{kJ/kg} के सामने --- वह केवल \omterm{prop:b2:open-systems:devices}{तुंडों} और
टरबाइन के फलकों में मायने रखता है, जहाँ चालें सैकड़ों मीटर प्रति सेकंड तक पहुँचती हैं। और
स्थितिज पद, प्रति मीटर $g\Delta z \approx \qty{10}{J/kg}$, किसी बाँध के पानी के लिए मायने रखता है,
किसी गैस-चक्र में कभी नहीं। कई प्रवेशों और निर्गमों के साथ
संतुलन ऐसा पढ़ा जाता है: $\sum_{\text{बाहर}}\dot m\,h - \sum_{\text{भीतर}}\dot m\,h =
P_{\text{उप}} + P_{\text{ताप}}$ (गतिज और स्थितिज पद छोड़े गए)।
\end{remark}

\section{किसी खुले तंत्र के लिए दूसरा नियम}

\begin{theorem}[किसी स्थायी प्रवाह का एंट्रॉपी संतुलन]\label{thm:b2:open-systems:secondlaw}
किसी \omterm{def:b2:open-systems:cv}{स्थायी} एक-धारा प्रवाह के लिए, जो तापीय शक्तियाँ $P_{\text{ताप},i}$ उन
स्रोतों के साथ बदलता हो जो तापों $T_i$ पर हैं,
\[
\dot m\,(s_2 - s_1) = \sum_i\frac{P_{\text{ताप},i}}{T_i} + \dot S_{\text{सृजित}},
\qquad \dot S_{\text{सृजित}} \ge 0 ;
\]
और प्रति एकांक द्रव्यमान, $\Delta s = s_{\text{विनिमय}} + s_{\text{सृजित}}$। कोई रुद्धोष्म
उत्क्रमणीय प्रवाह \emph{सम-एंट्रॉपी} है; और किसी रुद्धोष्म असली प्रवाह का $s_2 >
s_1$ होता है। किसी टरबाइन की \emph{सम-एंट्रॉपी दक्षता}\index{सम-एंट्रॉपी दक्षता}
$\eta_{\text{ट}} = w_{\text{असली}}/w_{\text{सम}} = (h_1 - h_2)/(h_1
- h_{2s})$ है और किसी संपीडक की $\eta_{\text{सं}} = (h_{2s} - h_1)/(h_2 -
h_1)$, जहाँ $2s$ वह अवस्था है जो उसी निर्गम दाब पर सम-एंट्रॉपी रूप से
पहुँची जाए; और दोनों प्रायः $0.8$--$0.9$ होते हैं। और अनुत्क्रमणीयता को खोया
कार्य, परिवेश के ताप $T_0$ के सापेक्ष,
प्रति एकांक द्रव्यमान $T_0\,s_{\text{सृजित}}$ है।
\end{theorem}

\begin{proof}
पहले जैसा ही बंद तंत्र: उसकी एंट्रॉपी $\dd m\,(s_2 -
s_1)$ से बदलती है, वह स्रोतों से $\sum P_{\text{ताप},i}\dd t/T_i$ पाता है, और
बाक़ी बनाई जाती है।
\end{proof}

\section{प्रारंभिक यंत्र}

\begin{proposition}[तुंड, थ्रॉटल, संपीडक, टरबाइन, विनिमयक]\label{prop:b2:open-systems:devices}
जब तक न कहा जाए, सब रुद्धोष्म, \omterm{def:b2:open-systems:cv}{स्थायी}, और गतिज तथा स्थितिज पद छोड़े गए,
जब तक वही मुद्दा न हों।
\begin{itemize}
\item \emph{तुंड}\index{तुंड} (कोई धुरी नहीं, कोई ऊष्मा नहीं): $\tfrac12c_2^2 -
      \tfrac12c_1^2 = h_1 - h_2$ --- अर्थात् \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी} चाल बन जाती है; और कोई
      \emph{विसारक} उलटा करता है।
\item \emph{थ्रॉटल}\index{थ्रॉटल} (कोई वाल्व, कोई सरंध्र डाट; कोई धुरी नहीं,
      कोई ऊष्मा नहीं, चालें छोटी): $h_2 = h_1$, अर्थात् \emph{\omterm{thm:b2:open-systems:firstlaw}{सम-एंथैल्पी}}। किसी
      आदर्श गैस के लिए $T_2 = T_1$; किसी असली गैस के लिए ताप आम तौर पर
      गिरता है (जूल--टॉमसन प्रभाव); और संतृप्ति के पास किसी द्रव के लिए कोई
      अंश भभककर वाष्प बन जाता है --- वही हर प्रशीतक का प्रसार-वाल्व।
\item \emph{संपीडक, पंप, टरबाइन}\index{टरबाइन} (रुद्धोष्म
      मशीनें): $w_{\text{उप}} = h_2 - h_1$ --- संपीडक के लिए धनात्मक,
      टरबाइन के लिए ऋणात्मक; और किसी द्रव-पंप के लिए $w_{\text{उप}} \approx
      v(P_2 - P_1)$, जो नन्हा है क्योंकि $v$ छोटा है।
\item \emph{ऊष्मा-विनिमयक}\index{ऊष्मा-विनिमयक} (दो धाराएँ, कोई
      धुरी नहीं, पूरा-का-पूरा रुद्धोष्म): $\dot m_{\text{क}}(h_{\text{क},2} -
      h_{\text{क},1}) = -\dot m_{\text{ख}}(h_{\text{ख},2} - h_{\text{ख},1})$
      --- जो एक खोता है वही दूसरा पाता है; और विनिमय एंट्रॉपी बनाता है,
      जब तक धाराएँ हर जगह एक ही ताप पर न हों
      (प्रति-प्रवाह उसे घटा देता है)।
\item \emph{मिश्रण कक्ष}: $\sum\dot m_{\text{भीतर}}h_{\text{भीतर}} = \dot
      m_{\text{बाहर}}h_{\text{बाहर}}$।
\end{itemize}
\end{proposition}

\begin{proof}
हर एक उपयुक्त पदों को शून्य रखकर लिखा पहला नियम ही है।
\end{proof}

\begin{proposition}[वाष्प-संपीडन प्रशीतन और ऊष्मा-पंप]\label{prop:b2:open-systems:fridge}
कोई प्रशीतक इनसे होकर घूमता है: (1$\to$2) कोई रुद्धोष्म संपीडक
जो वाष्प को वाष्पित्र के दाब से संघनित्र के
दाब तक चढ़ा दे, $w = h_2 - h_1$; (2$\to$3) कोई संघनित्र जहाँ वह गरम पक्ष को
$q_{\text{संघ}} = h_2 - h_3$ छोड़ दे (ऊष्मा-पंप के लिए कमरा, किसी फ़्रिज के लिए
रसोई) और द्रव बनकर निकले; (3$\to$4) कोई \omterm{prop:b2:open-systems:devices}{थ्रॉटल}, $h_4
= h_3$; (4$\to$1) कोई वाष्पित्र जहाँ वह ठंडे पक्ष से $q_{\text{वाष्प}} = h_1 - h_4$
सोखे। निष्पादन गुणांक हैं
\[
\mathrm{COP}_{\text{शीतलन}} = \frac{h_1 - h_4}{h_2 - h_1}, \qquad
\mathrm{COP}_{\text{तापन}} = \frac{h_2 - h_3}{h_2 - h_1} = \mathrm{COP}_{\text{शीतलन}} + 1 ,
\]
जो कार्नो के $T_{\text{ठंडा}}/(T_{\text{गरम}} - T_{\text{ठंडा}})$ और
$T_{\text{गरम}}/(T_{\text{गरम}} - T_{\text{ठंडा}})$ से बँधे हैं; और चक्र सबसे आसानी से
$(\log P, h)$ आरेख पर पढ़ा जाता है, जहाँ दोनों विनिमय तथा कार्य
क्षैतिज खंड हैं।
\end{proposition}

\begin{proof}
चारों पहले-नियम संतुलन; और पूरे चक्र का ऊर्जा-संतुलन
$q_{\text{संघ}} = q_{\text{वाष्प}} + w$ दे देता है, जिससे
दोनों COP के बीच का संबंध।
\end{proof}

\begin{proposition}[भाप (रैंकिन) चक्र]\label{prop:b2:open-systems:rankine}
जल इनसे होकर घूमता है: (1$\to$2) कोई पंप, $w_{\text{पंप}} \approx v(P_2 -
P_1)$, यानी कुछ \unit{kJ/kg}; (2$\to$3) ऊँचे दाब पर कोई बॉयलर जहाँ उसे
गरम, वाष्पित और अतितापित किया जाए, $q_{\text{भीतर}} = h_3 - h_2$;
(3$\to$4) कोई टरबाइन जो भाप को संघनित्र के दाब तक फैलाए,
$w_{\text{ट}} = h_3 - h_4$ (कोई हज़ार \unit{kJ/kg}); (4$\to$1) नीचे दाब पर कोई संघनित्र
जहाँ गीली भाप फिर द्रव बन जाए और शीतलन-जल को
$q_{\text{बाहर}} = h_4 - h_1$ फेंक दे। दक्षता है
\[
\eta = \frac{w_{\text{ट}} - w_{\text{पंप}}}{q_{\text{भीतर}}} = 1 - \frac{q_{\text{बाहर}}}{q_{\text{भीतर}}} ,
\]
व्यवहार में लगभग $0.35$--$0.45$; और ऊष्मा किसी माध्य
ताप $\bar T = q_{\text{भीतर}}/(s_3 - s_2)$ पर जोड़ी जाती है, जो बॉयलर की
चोटी से कहीं नीचे है, और इसीलिए अतितापन, पुनस्तापन तथा ऊँचे दाब
$\eta$ चढ़ा देते हैं, और इसीलिए निकास इतना सूखा रहना चाहिए ($x_4 \gtrsim 0.88$) कि
आख़िरी फलक बूँदों के कटाव से बचे रहें।
\end{proposition}

\begin{proof}
हर घटक पर पहला नियम, फिर चक्र पर: $w_{\text{ट}} - w_{\text{पंप}}
= q_{\text{भीतर}} - q_{\text{बाहर}}$।
\end{proof}

\begin{remark}[दूसरे चक्र, और कोई क्षणिक]\label{rem:b2:open-systems:other}
गैस टरबाइन (संपीडक, दहन-कक्ष, टरबाइन: ब्रेटन
चक्र) की आदर्श दक्षता $1 - r^{-(\gamma-1)/\gamma}$ है, दाब-अनुपात $r$ के लिए,
और वह अपना निकास कई सौ डिग्री पर फेंकती है;
उस निकास को किसी भाप-चक्र को खिलाना (संयुक्त चक्र) $60\%$ तक पहुँचता है।
खुले तंत्रों के क्षणिक भी होते हैं: $T_0$ की किसी लाइन से किसी निर्वातित दृढ़
टंकी को भरना $T = \gamma T_0$ पर ख़त्म होता है (प्रवाह कार्य $Pv$
आंतरिक ऊर्जा बन जाता है), हवा के लिए \qty{120}{K} अधिक गरम --- और इसीलिए
कोई गोताखोरी सिलिंडर भरते समय गरम हो जाता है।
\end{remark}

\begin{method}[खुले-तंत्र के संतुलन]\label{meth:b2:open-systems:method}
(1) \omterm{def:b2:open-systems:cv}{नियंत्रण आयतन} खींचिए; प्रवेश, निर्गम, धुरी और ऊष्मा सूचीबद्ध कीजिए।
(2) द्रव्यमान: $\sum\dot m_{\text{भीतर}} = \sum\dot m_{\text{बाहर}}$। (3) प्रति एकांक
द्रव्यमान ऊर्जा: $\Delta h + \Delta c^2/2 + g\Delta z = w_{\text{उप}} + q$, जिसमें
गैसों के लिए $\Delta h = c_p\Delta T$, वाष्पों के लिए सारणियाँ, और पंपों के लिए
$v\Delta P$। (4) एंट्रॉपी: $\Delta s = \sum q_i/T_i + s_{\text{सृजित}}$, सम-एंट्रॉपी
संदर्भ अवस्थाएँ, दक्षताएँ, और खोया $T_0s_{\text{सृजित}}$। (5) चक्र:
चार्ट पढ़िए, पाश भर जोड़िए, और कार्नो से तुलना कीजिए।
\end{method}

(क) कोई रेडिएटर \qty{1}{kg/s} जल को \qty{90}{} से \qty{80}{\celsius} तक ठंडा करता है;

हवा का द्रव्यमान प्रवाह और आयतन प्रवाह। (ख) कोई भाप-टरबाइन, \qty{10}{kg/s},

जल को \qty{1}{} से \qty{60}{bar} तक चढ़ाता है: प्रति किलोग्राम कार्य, और

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:5}

(घ) किसी जल-नाल से \qty{100}{m} गिरता जल: $g\Delta z$ के सामने

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:6}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:7}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:8}

\begin{exercise}[$\star\star$]\label{exo:b2:open-systems:9}

\begin{exercise}[$\star\star\star$]\label{exo:b2:open-systems:10}
\emph{कोई टंकी भरना।} कोई दृढ़, निर्वातित, विद्युतरोधित टंकी किसी ऐसी लाइन से
जोड़ी जाती है जो $T_0$, $P_0$ पर कोई आदर्श गैस ले चलती है। (क) भरने के दौरान टंकी का

पानी में धीरे-धीरे क्यों भरा जाता है?
\end{exercise}

\qty{144}{kJ/kg}$। (क) दूसरे चरण की निर्गम शुष्कता और \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी}। (ख)

\section{समस्या: कोई घरेलू ऊष्मा-पंप और कोई भाप-बिजलीघर}

\begin{problem}[{सप्ताहांत समस्या --- खुले-तंत्र संतुलनों से नापी गई
दो मशीनें}]\label{pb:b2:open-systems:1}
\textbf{भाग I --- ऊष्मा-पंप।}
प्रशीतक के आँकड़े (\unit{kJ/kg}, \unit{kJ/kg/K}): \qty{0}{\celsius} पर
संतृप्त वाष्प ($P = \qty{2.9}{bar}$): $h_1 = 399$, $s_1 = 1.727$;
\qty{0}{\celsius} पर संतृप्त द्रव: $h_{\text{f}} = 200$, $s_{\text{f}} = 1.00$;
संघनित्र के दाब \qty{10.2}{bar} ($T_{\text{संत}} = \qty{40}{\celsius}$) पर:
सम-एंट्रॉपी संपीडन का अंत $h_{2s} = 426$, संतृप्त द्रव $h_3 = 256$,
$s_3 = 1.19$; और वहाँ अतितापित वाष्प का $c_p \approx \qty{1.0}{kJ/kg/K}$।
संपीडक $\eta_{\text{सं}} = 0.8$; और घर की माँग \qty{8}{kW}।
\begin{enumerate}
  \item चक्र को किसी $(\log P, h)$ चार्ट पर खींचिए और चारों
        घटकों को उस प्रक्रिया के साथ बताइए जो हर एक करता है।
  \item \omterm{prop:b2:open-systems:devices}{थ्रॉटल} के बाद अवस्था 4: \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी} और वाष्प का अंश।
  \item वाष्पित्र में प्रति किलोग्राम सोखी गई ऊष्मा।
  \item संपीडक का कार्य, सम-एंट्रॉपी और असली; $h_2$ और
        $T_2$ का कोई आकलन।
  \item संघनित्र में छोड़ी गई ऊष्मा; दोनों COP; और उनका
        संबंध जाँचिए।
  \item \qty{0}{} और \qty{40}{\celsius} के बीच कार्नो COP; मशीन कौन-सा
        अंश पा लेती है?
  \item प्रशीतक का द्रव्यमान प्रवाह और विद्युत् शक्ति (मोटर की
        दक्षता $0.9$)।
  \item संपीडक में और \omterm{prop:b2:open-systems:devices}{थ्रॉटल} में प्रति किलोग्राम बनी एंट्रॉपी
        ($s_4 = s_{\text{f}} + x\,(s_1 - s_{\text{f}})$); कौन-सा
        बुरा है?
\end{enumerate}

\textbf{भाग II --- ठंडा मौसम और अभिकल्प के चुनाव।}
\begin{enumerate}[resume]
  \item बाहर \qty{-10}{\celsius} पर वाष्पित्र
        \qty{-15}{\celsius} पर चलता है: तापन का कार्नो COP; और असली
        COP उससे तेज़ी से क्यों गिरता है (दाब-अनुपात और
        वाष्प के घनत्व के बारे में सोचिए)?
  \item बाहरी कुंडली पर पाला क्यों जमता है, और तुषार-हरण चक्र की
        क़ीमत क्या है?
  \item \qty{35}{\celsius} पर फ़र्श-तापन बनाम \qty{55}{\celsius} पर
        ऊष्मक: वाष्पित्र \qty{0}{\celsius} पर लेकर कार्नो COP;
        निष्कर्ष निकालिए।
  \item दक्षता $0.40$ के किसी बिजलीघर में जले प्रति किलोवाट-घंटा ईंधन के लिए,
        इस ऊष्मा-पंप ($\mathrm{COP}
        = 4$) से मिलती ऊष्मा की तुलना
        उस गैस-बॉयलर से कीजिए जो दक्षता $0.9$ पर वही ईंधन सीधे
        जलाए।
\end{enumerate}

\textbf{भाग III --- भाप-संयंत्र।}
बॉयलर \qty{60}{bar}, \qty{500}{\celsius}: $h_3 = 3422$, $s_3 = 6.88$;
\qty{60}{bar} पर, $T_{\text{संत}} = \qty{276}{\celsius}$, $h_{\text{f}} = 1213$,
$h_{\text{g}} = 2784$; संघनित्र \qty{0.05}{bar}, \qty{33}{\celsius}:
$h_{\text{f}} = 138$, $h_{\text{fg}} = 2423$, $s_{\text{f}} = 0.476$, $s_{\text{g}} =
8.394$; द्रव $v = \qty{1e-3}{m^3/kg}$; टरबाइन $\eta_{\text{ट}} = 0.85$;
और शुद्ध विद्युत् निर्गम \qty{600}{MW}।
\begin{enumerate}[resume]
  \item चक्र को किसी $(T,s)$ आरेख पर खींचिए और प्रक्रियाएँ बताइए।
  \item पंप का कार्य और $h_2$।
  \item बॉयलर में जोड़ी गई ऊष्मा, जिसे पूर्वतापन, वाष्पन
        और अतितापन में बाँटिए।
  \item सम-एंट्रॉपी प्रसार: निर्गम शुष्कता, \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी}, कार्य।
  \item असली प्रसार: कार्य, निर्गम \omterm{thm:b2:open-systems:firstlaw}{एंथैल्पी} और शुष्कता; और शुष्कता
        क्यों मायने रखती है।
  \item चक्र की दक्षता; \qty{773}{} और \qty{306}{K} के बीच कार्नो;
        ऊष्मा जोड़ने का माध्य ताप $\bar T = q_{\text{भीतर}}/(s_3
        - s_2)$ और वह कार्नो दक्षता जो वह देता।
  \item भाप का द्रव्यमान प्रवाह; संघनित्र में फेंकी गई ऊष्मा; \qty{10}{K}
        की चढ़ाई के लिए शीतलन-जल का प्रवाह, या वह जल-द्रव्यमान जो कोई शीतलन
        मीनार प्रति सेकंड वाष्पित करती है (\qty{2.4}{MJ/kg})।
  \item टरबाइन में प्रति किलोग्राम और प्रति सेकंड बनी एंट्रॉपी;
        और \qty{306}{K} पर खोया कार्य, टरबाइन के कार्य के
        अंश की तरह।
  \item \qty{10}{bar} पर \qty{500}{\celsius} तक पुनस्तापन दक्षता को
        लगभग $0.43$ और निर्गम शुष्कता को $0.92$ तक चढ़ा देता है:
        दोनों असर गुणात्मक रूप से समझाइए।
\end{enumerate}

\textbf{भाग IV --- पूरी शृंखला।}
\begin{enumerate}[resume]
  \item भट्ठी अपनी ऊष्मा भाप को लगभग \qty{1500}{K} की धुआँ-गैसों से
        सौंपती है: उस अंतरण में प्रति सेकंड बनी एंट्रॉपी
        (\qty{1500}{K} पर ऊष्मा, $\bar T$ पर भाप में); टरबाइन वाली से
        तुलना कीजिए।
  \item संघनित्र अपनी ऊष्मा \qty{306}{K} के जल को फेंकता है ---
        जो अंततः \qty{293}{K} के परिवेश तक पहुँचती है: वहाँ प्रति सेकंड
        बनी एंट्रॉपी।
  \item अनुत्क्रमणीयता के तीनों स्रोतों (भट्ठी, टरबाइन,
        संघनित्र) को क्रम दीजिए और बताइए कि अभियंताओं का ज़ोर कहाँ जाता है।
  \item खुले-तंत्र की विधि पाँच पंक्तियों में समेटिए।
\end{enumerate}
\end{problem}
