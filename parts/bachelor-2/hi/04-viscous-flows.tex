\chapter{श्यान प्रवाह}\label{ch:b2:viscous-flows}

शहद का मर्तबान उड़ेलिए और वह धीमी, गाढ़ी धार में गिरता है; पानी का
मर्तबान उड़ेलिए और वह पलक झपकते ख़त्म। दोनों में चम्मच चलाइए: शहद रोकता
है, और हर चाल पर रोकता ही रहता है; पानी तब तक कुछ नहीं कहता जब तक आप तेज़
न चलें, और फिर वह भँवर बनाने लगता है। यह अंतर \emph{\omterm{def:b2:viscous-flows:viscosity}{श्यानता}} है ---
तरल का आंतरिक घर्षण --- और \omterm{def:b2:viscous-flows:viscosity}{श्यानता} तथा जड़त्व की यह लड़ाई, जो एक ही
संख्या में सिमटती है, रेनल्ड्स की संख्या में, तय करती है कि कोई प्रवाह
सुव्यवस्थित परतों में रेंगेगा या विक्षोभ में उलट-पुलट जाएगा। यह अध्याय
\omterm{thm:b2:euler-bernoulli:euler}{ऑयलर समीकरण} में श्यान बल जोड़ता है, हर अभियंता के काम आने वाले दोनों
\omterm{def:b2:viscous-flows:reynolds}{स्तरीय प्रवाह} हल करता है, और समझाता है कि ऊँची \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} पर
\omterm{def:b2:viscous-flows:viscosity}{श्यानता} दीवार की पतली परत में छिप क्यों जाती है और फिर भी हर
कार, जहाज़ और वायुयान का अधिकांश ईंधन क्यों खा जाती है।

\section{श्यानता और नेवियर--स्टोक्स समीकरण}

\begin{definition}[न्यूटनी श्यानता]\label{def:b2:viscous-flows:viscosity}
\index{श्यानता}\index{गतिक श्यानता}\index{शुद्धगतिक श्यानता}\index{अपरूपण प्रतिबल}\index{न्यूटनी तरल}
समतल अपरूपण प्रवाह $\vect v = v_x(y)\,\vect e_x$ में, किसी पृष्ठ
$y = \text{अचर}$ के ऊपर का तरल उसके नीचे के तरल पर (किसी पृष्ठ-अवयव
$\dd S$ के माध्यम से) यह स्पर्शीय बल लगाता है:
\[
\dd\vect F = \eta\,\frac{\dd v_x}{\dd y}\,\dd S\;\vect e_x ,
\]
और इस तरह धीमी परत को आगे खींचता है और बदले में स्वयं पीछे खिंचता है।
\emph{अपरूपण प्रतिबल} $\tau = \eta\,\dd v_x/\dd y$ है; और गुणांक
$\eta$ (मात्रक \unit{Pa.s}) तरल की \emph{गतिक श्यानता} है,
जबकि $\nu = \eta/\rho$ (\unit{m^2/s}) उसकी \emph{शुद्धगतिक श्यानता}। जिस तरल
के लिए $\eta$ अपरूपण-दर पर निर्भर न करे वह \emph{न्यूटनी} है:
पानी, हवा, तेल, ऐल्कोहॉल; रक्त, रंग, चटनी और बहुलक-द्रव
नहीं। \qty{20}{\degreeCelsius} पर परिमाण की कोटि: हवा $\eta =
\qty{1.8e-5}{Pa.s}$ ($\nu = \qty{1.5e-5}{m^2/s}$), पानी $\qty{1.0e-3}{Pa.s}$
($\nu = \qty{1.0e-6}{m^2/s}$), जैतून का तेल $\qty{0.08}{Pa.s}$, ग्लिसरॉल
$\qty{1.5}{Pa.s}$, शहद $\sim\qty{10}{Pa.s}$। द्रव गरम होने पर पतले पड़ते
हैं, गैसें गाढ़ी।
\end{definition}

\begin{proposition}[श्यान बल घनत्व; अ-सर्पण प्रतिबंध]\label{prop:b2:viscous-flows:density}
\index{अ-सर्पण प्रतिबंध}\index{श्यान बल घनत्व}
समतल अपरूपण प्रवाह में, $y$ और $y + \dd y$ के बीच तरल की किसी परत पर
कुल श्यान बल प्रति एकांक आयतन $\eta\,\dd^2v_x/\dd y^2$ होता है: अर्थात्
\omterm{def:b2:viscous-flows:viscosity}{श्यानता} $\eta\,\Delta\vect v$ घनत्व के आयतन-बल की तरह काम करती है (यहाँ
लाप्लासीय घटक-दर-घटक लिया गया है), और यह परिणाम किसी भी
\omterm{cor:b2:fluid-kinematics:incompressible}{असंपीड्य प्रवाह} के लिए सही है। किसी ठोस दीवार पर तरल \emph{चिपक} जाता
है: उसका वेग दीवार के वेग के बराबर हो जाता है (\emph{\omterm{prop:b2:viscous-flows:density}{अ-सर्पण प्रतिबंध}}),
और इसीलिए पंखे की पंखुड़ी पर धूल टिकी रहती है और दीवार श्यान कर्षण अनुभव करती है।
\end{proposition}

\begin{proof}
परत को ऊपर से $\eta\,v_x'(y + \dd y)\dd S$ खींचता है और नीचे से
$-\eta\,v_x'(y)\dd S$: कुल $\eta v_x''\,\dd y\,\dd S$। सामान्य
रूप स्वीकृत है (वह प्रतिबल को वेग-प्रवणताओं का सममित रैखिक फलन लिखने से
निकलता है)। \omterm{prop:b2:viscous-flows:density}{अ-सर्पण प्रतिबंध}
प्रायोगिक है।
\end{proof}

\begin{theorem}[नेवियर--स्टोक्स समीकरण]\label{thm:b2:viscous-flows:ns}
\index{नेवियर--स्टोक्स समीकरण}
कोई असंपीड्य \omterm{def:b2:viscous-flows:viscosity}{न्यूटनी तरल} इसका पालन करता है:
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g + \eta\,\Delta\vect v ,
\qquad \operatorname{div}\vect v = 0 ,
\]
और हर दीवार पर \omterm{prop:b2:viscous-flows:density}{अ-सर्पण प्रतिबंध} के साथ। \omterm{thm:b2:euler-bernoulli:euler}{ऑयलर समीकरण}
$\eta = 0$ की दशा है।
\end{theorem}
\admitted

\begin{example}[मेज़ पर अपरूपण प्रतिबल]\label{ex:b2:viscous-flows:table}
किसी पट्ट और मेज़ के बीच तेल की \qty{0.1}{mm} परत ($\eta = \qty{0.1}{Pa.s}$),
और पट्ट \qty{1}{m/s} से सरकता हुआ: $\tau = \eta U/h = \qty{1}{kPa}$, अर्थात्
वर्ग मीटर के दसवें भाग पर \qty{100}{N} का बल --- यही स्नेहित धारक का सिद्धांत
है, जहाँ श्यान कर्षण सौ गुना बड़े शुष्क घर्षण की जगह ले लेता है।
चम्मच पर \qty{1}{mm} मोटा शहद, \qty{10}{Pa.s}, जो
\qty{1}{mm/s} से बहता है: $\tau = \qty{10}{Pa}$, ठीक उतना ही जितना प्रति एकांक
क्षेत्रफल एक मिलीमीटर शहद का भार --- और इसीलिए वह उसी चाल से बहता है।
\end{example}

\section{रेनल्ड्स संख्या}

\begin{definition}[रेनल्ड्स संख्या]\label{def:b2:viscous-flows:reynolds}
\index{रेनल्ड्स संख्या}\index{स्तरीय प्रवाह}\index{विक्षुब्ध प्रवाह}
अभिलक्षणिक चाल $U$ और लंबाई $L$ वाले किसी प्रवाह के लिए \emph{रेनल्ड्स
संख्या}
\[
\mathrm{Re} = \frac{\rho UL}{\eta} = \frac{UL}{\nu}
\]
\omterm{thm:b2:viscous-flows:ns}{नेवियर--स्टोक्स समीकरण} में जड़त्व-पद
$\rho(\vect v\cdot\operatorname{\vect{grad}})\vect v \sim \rho U^2/L$ और
श्यान पद $\eta\Delta\vect v \sim \eta U/L^2$ की परिमाण-कोटियों का
अनुपात है। $\mathrm{Re} \ll 1$ पर जड़त्व नगण्य है और प्रवाह
\emph{मंथर} होता है (श्यान, उत्क्रमणीय, चिकना); $\mathrm{Re} \gg 1$ पर
\omterm{def:b2:viscous-flows:viscosity}{श्यानता} दीवारों के पास को छोड़ नगण्य है और प्रवाह \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल}
जैसा होता है --- जब तक कि, $10^3$ कोटि की देहली से आगे (नली में व्यास के
आधार पर लगभग $2000$ से $3000$), वह \emph{विक्षुब्ध} न हो जाए:
अस्थायी, अराजक, और हर पैमाने पर भँवरों से भरा। इन दोनों के बीच
प्रवाह \emph{स्तरीय} होता है: स्थायी और परतदार।
\end{definition}

\begin{example}[रोज़मर्रा के प्रवाहों की रेनल्ड्स संख्याएँ]\label{ex:b2:viscous-flows:revalues}
कोई जीवाणु (\qty{1}{\micro m}, \qty{30}{\micro m/s}, पानी में): $\mathrm{Re} =
3 \times 10^{-5}$ --- वह जड़त्वहीन संसार में रहता है, जहाँ उसका
कशाभिका रोकते ही वह एक परमाणु-व्यास के भीतर रुक जाता है। केशिका में रक्त:
$10^{-3}$; गिरता धूल-कण: $10^{-2}$; सुनहरी मछली: $10^3$; तैराक:
$10^6$; \qty{30}{m/s} पर कार (\qty{4}{m}): $8 \times 10^6$; यात्री-विमान: $10^8$;
गल्फ़ स्ट्रीम: $10^{11}$। समान ज्यामिति और समान
\omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} वाले दो प्रवाह \emph{समरूप} होते हैं: और इसी से जहाज़ का
\qty{1}{m} का प्रतिरूप कर्षण-टंकी में, या वायुयान पवन-सुरंग में, परखा जा सकता है।
\end{example}

\section{स्तरीय प्रवाह: कुएत, प्वाज़ॉय, स्टोक्स}

\begin{proposition}[नली में प्वाज़ॉय प्रवाह]\label{prop:b2:viscous-flows:poiseuille}
\index{प्वाज़ॉय प्रवाह}\index{द्रवीय प्रतिरोध}
$R$ त्रिज्या और $L$ लंबाई की किसी क्षैतिज बेलनाकार नली में, जिसे उसके
सिरों के बीच दाब-अंतर $\Delta P = P_1 - P_2$ चलाता है, स्थायी
\omterm{def:b2:viscous-flows:reynolds}{स्तरीय प्रवाह} यह है:
\[
v(r) = \frac{\Delta P}{4\eta L}\,(R^2 - r^2) , \qquad
D_V = \frac{\pi R^4}{8\eta L}\,\Delta P :
\]
अर्थात् परवलयिक प्रोफ़ाइल, जो अक्ष पर अधिकतम है, और माध्य चाल
$\langle v\rangle = v_{\max}/2$। प्रवाह दर
दाब-गिरावट के समानुपाती है (प्वाज़ॉय का नियम), जिसमें \emph{\omterm{prop:b2:viscous-flows:poiseuille}{द्रवीय प्रतिरोध}}
$R_h = \Delta P/D_V = 8\eta L/\pi R^4$ है --- और वह त्रिज्या की \emph{चौथी घात}
के व्युत्क्रमानुपाती। दीवार पर \omterm{def:b2:viscous-flows:viscosity}{अपरूपण प्रतिबल} $\tau_w = \Delta P\,R/2L = 4\eta\langle v
\rangle/R$ है। यह तब तक मान्य है जब तक $\mathrm{Re} = 2R\langle v\rangle/\nu \lesssim 2000$।
\end{proposition}

\begin{proof}
स्थायी, $\vect v = v(r)\vect e_z$ (अतः संवहनी पद लुप्त हो जाता है: $\vect v
\cdot\operatorname{\vect{grad}}$ $z$ के अनुदिश काम करता है, जिसके अनुदिश कुछ
नहीं बदलता)। $z$ के अनुदिश नेवियर--स्टोक्स: $0 = -\partial_zP + \eta\Delta v$, और
त्रिज्यीय दिशा में $\partial_rP = 0$: अतः $P$ केवल $z$ पर निर्भर है, $\Delta v$ केवल $r$ पर, इसलिए
दोनों पक्ष किसी अचर के बराबर हैं, $\partial_zP = -\Delta P/L$। बेलनाकार
निर्देशांकों में $\Delta v = \frac1r\frac{\dd}{\dd r}\bigl(r\frac{\dd v}{\dd r}\bigr)$
(कथित; यह किसी बेलनाकार कोश पर श्यान बलों का संतुलन है,
$2\pi rL\,\eta v'$, जो $r$ पर और $r + \dd r$ पर लगता है)। दो बार समाकलन करने पर, $v$
को अक्ष पर परिमित और $v(R) = 0$ लेकर, परवलय मिलता है; और $D_V = \int_0^Rv\,2\pi
r\,\dd r$। दीवार का प्रतिबल $-\eta v'(R)$ है; या समतुल्य रूप से, दाब-बल
$\pi R^2\Delta P$ दीवार के घर्षण $2\pi RL\tau_w$ को संतुलित करता है।
\end{proof}

\begin{example}[चौथी घात]\label{ex:b2:viscous-flows:fourth}
\qty{0.4}{mm} भीतरी व्यास और \qty{3}{cm} लंबाई की कोई सिरिंज-सुई,
पानी, और अंगूठे का दाब \qty{20}{kPa}: $D_V = \pi(2 \times 10^{-4})^4 \times 2 \times

सोलह गुना अधिक देती है। जिस धमनी को कोई पट्टिका उसकी त्रिज्या के $70\%$ तक
सँकरा कर दे, उसका प्रतिरोध $(1/0.7)^4 = 4.2$ गुना हो जाता है: शरीर उसकी
भरपाई ऊपर का दाब बढ़ाकर करता है, और यही एक कारण है कि धमनी-काठिन्य
और उच्च रक्तचाप साथ-साथ चलते हैं।
\end{example}

\begin{proposition}[स्टोक्स कर्षण]\label{prop:b2:viscous-flows:stokes}
\index{स्टोक्स का नियम}\index{सीमांत वेग}\index{अवसादन}
$R$ त्रिज्या का कोई गोला जब $v$ चाल से किसी ऐसे तरल में चलता है जहाँ
$\mathrm{Re} = 2\rho vR/\eta \ll 1$ हो, तो उस पर कर्षण
\[
F = 6\pi\eta Rv ,
\]
लगता है, जो उसके वेग के विपरीत होता है। $\rho_s$ घनत्व का गोला
$\rho$ घनत्व के तरल में गिरते हुए \emph{\omterm{prop:b2:viscous-flows:stokes}{सीमांत वेग}} $v_\infty =
\dfrac{2R^2(\rho_s - \rho)g}{9\eta}$ तक पहुँचता है --- यही अवसादन,
अपकेंद्रित्र और मिलिकन की तेल-बूँदों का नियम है।
\end{proposition}

\begin{proof}
कर्षण स्वीकृत है (वह गोले के चारों ओर मंथर-प्रवाह समीकरणों का यथार्थ हल
है, जिसका दो-तिहाई श्यान घर्षण और एक-तिहाई
दाब है)। सीमांत चाल: भार में से उत्प्लावन घटाकर, $\tfrac43\pi R^3(\rho_s -
\rho)g$, कर्षण के बराबर होता है; और बाद में $\mathrm{Re} \ll 1$ सदा जाँच लीजिए।
\end{proof}

\begin{example}[कुहरा, वर्षा और धूल]\label{ex:b2:viscous-flows:fog}
हवा में \qty{10}{\micro m} व्यास की कुहरे की बूँद $2 \times (5 \times
10^{-6})^2 \times 10^4/(9 \times 1.8 \times 10^{-5}) = \qty{3}{mm/s}$ से गिरती है ($\mathrm{Re} = 2 \times
10^{-3}$): ज़रा-सा ऊर्ध्व-प्रवाह उसे टिकाए रखता है, और इसीलिए बादल
तैरते हैं। \qty{1}{mm} की वर्षा-बूँद \qty{30}{m/s} और $\mathrm{Re} \approx 2000$ देती:
वहाँ स्टोक्स नहीं चलता, और अगले अनुच्छेद का द्विघाती कर्षण
प्रेक्षित \qty{4}{m/s} देता है। \qty{10}{\micro m} का धूल-कण ($\rho_s =
\qty{2500}{kg/m^3}$) \qty{8}{mm/s} से बैठता है, अर्थात् दो मिनट में एक मीटर ---
और उससे महीन धूल घंटों लटकी रहती है।
\end{example}

\section{ऊँची रेनल्ड्स संख्या: सीमांत परतें और कर्षण}

\begin{proposition}[सीमांत परत]\label{prop:b2:viscous-flows:bl}
\index{सीमांत परत}
ऊँची \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} पर किसी पिंड के चारों ओर का प्रवाह, दीवार के
अनुदिश पतली \emph{\omterm{prop:b2:viscous-flows:bl}{सीमांत परत}} को छोड़, हर जगह \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल} जैसा होता है;
उस परत में वेग अपने बाहरी मान $U$ से घटकर शून्य हो जाता है (अ-सर्पण)।
किसी सपाट पट्ट के अनुदिश उसकी मोटाई इस तरह बढ़ती है:
\[
\delta(x) \sim \sqrt{\frac{\nu x}{U}} = \frac{x}{\sqrt{\mathrm{Re}_x}} ,
\qquad \mathrm{Re}_x = \frac{Ux}{\nu} ,
\]
जहाँ $x$ अग्र किनारे से दूरी है; और परत स्वयं
$\mathrm{Re}_x \sim 5 \times 10^5$ के आगे विक्षुब्ध हो जाती है। दीवार पर श्यान घर्षण,
और किसी कुंठित पिंड के पीछे परत का \emph{पृथक्करण} (जो कम दाब की
विक्षुब्ध पश्चधारा छोड़ जाता है), कर्षण के दो स्रोत हैं।
\end{proposition}

\begin{proof}
परिमाण की कोटि: कोई \omterm{def:b2:fluid-kinematics:particle}{तरल कण} पट्ट के अनुदिश चलने में $x/U$ समय बिताता
है, और उतने में दीवार का प्रभाव तरल में विसरित होता है --- श्यान पद
$\nu\,\partial^2v/\partial y^2$ संवेग का विसरण है, जिसकी विसरणशीलता $\nu$ है
और जो $\sqrt{\nu t}$ दूरी तक समय $t$ में पहुँचता है
(\cref{ch:b2:particle-diffusion})। अतः $\delta \sim \sqrt{\nu x
/U}$। पूरा (प्रांटल) सिद्धांत स्वीकृत है।
\end{proof}

\begin{proposition}[ऊँची रेनल्ड्स संख्या पर कर्षण]\label{prop:b2:viscous-flows:drag}
\index{कर्षण गुणांक}\index{कर्षण संकट}
$S$ अग्र-क्षेत्रफल के किसी पिंड पर, जो किसी तरल में $v$ से चलता हो,
कर्षण इस तरह लिखा जाता है:
\[
F = \tfrac12\rho v^2\,S\,C_x ,
\]
जहाँ \emph{\omterm{prop:b2:viscous-flows:drag}{कर्षण गुणांक}} $C_x$ आकार पर और
$\mathrm{Re}$ पर निर्भर करता है। गोले के लिए $C_x = 24/\mathrm{Re}$, जब $\mathrm{Re} \ll 1$ (स्टोक्स,
जहाँ $\mathrm{Re}$ व्यास पर आधारित है), फिर लगभग अचर $0.4$--$0.5$,
$\mathrm{Re} \approx 10^3$ से $2 \times 10^5$ तक, और वहाँ वह एकाएक
गिरकर लगभग $0.1$ हो जाता है (\emph{\omterm{prop:b2:viscous-flows:drag}{कर्षण संकट}}: \omterm{prop:b2:viscous-flows:bl}{सीमांत परत} विक्षुब्ध हो
जाती है, सतह से अधिक देर चिपकी रहती है और पश्चधारा सिकुड़ जाती है ---
गोल्फ़ की गेंद के गड्ढे इसे जल्दी छेड़ देते हैं)। धारारेखित पिंडों का
$C_x \approx 0.05$ होता है; कार का $0.3$; साइकिल-सवार का $0.9$; प्रवाह के सामने
रखे सपाट पट्ट का $1.2$। $v$ से चलने के लिए आवश्यक शक्ति $Fv \propto v^3$ है।
\end{proposition}
\admitted

\begin{remark}[पृष्ठ तनाव]\label{rem:b2:viscous-flows:capillarity}
\index{पृष्ठ तनाव}\index{लाप्लास दाब}\index{केशिका लंबाई}
किसी द्रव की सीमाओं पर एक दूसरा बल बसता है: उसकी मुक्त सतह
$\gamma$ तनाव की खिंची झिल्ली की तरह बरतती है (\unit{N/m}; पानी के लिए
\qty{0.072}{N/m}), क्योंकि सतह के अणुओं के आधे पड़ोसी नहीं होते।
$R$ त्रिज्या की बूँद के भीतर दाब बाहर के दाब से \emph{\omterm{rem:b2:viscous-flows:capillarity}{लाप्लास दाब}}
$2\gamma/R$ जितना अधिक होता है (\qty{0.1}{mm} की बूँद में \qty{1.4}{kPa},
और \qty{10}{nm} की में \qty{14}{bar}); और \emph{\omterm{rem:b2:viscous-flows:capillarity}{केशिका लंबाई}}
$\sqrt{\gamma/\rho g} = \qty{2.7}{mm}$ (पानी के लिए) से नीचे \omterm{rem:b2:viscous-flows:capillarity}{पृष्ठ तनाव} गुरुत्व को हरा देता है ---
यही ओस की बूँदों का, गिलास के नवचंद्रक का, और केशिका नली में चढ़ाव
($h = 2\gamma/\rho gr$, जुरिन) का आकार है। इसी से छोटे कीट पानी पर चलते
हैं, गीले ब्रश के बाल आपस में चिपक जाते हैं, और द्रव की धार
टूटकर बूँदें बन जाती है।
\end{remark}

\begin{method}[कौन-सी दशा?]\label{met:b2:viscous-flows:regime}
श्यान प्रवाह की किसी भी गणना से पहले $\mathrm{Re}$ निकालिए। $\mathrm{Re} \ll 1$:
स्टोक्स और मंथर प्रवाह, कर्षण $\propto v$, दाब-गिरावट $\propto$ प्रवाह
दर। नली में $\mathrm{Re}$ $\sim\!2000$ तक: स्तरीय, प्वाज़ॉय। उससे ऊपर:
विक्षुब्ध --- वहाँ प्वाज़ॉय हानि को बुरी तरह कम आँकता है; इसके बदले
चिकनी नलियों के लिए प्रयोगसिद्ध नियम $\Delta P = \lambda(L/D)\,\tfrac12\rho\langle v\rangle^2$ लीजिए, जहाँ $\lambda
\approx 0.3/\mathrm{Re}^{1/4}$, और पिंडों के लिए $F = \tfrac12\rho v^2SC_x$।
हर दशा में बाद में जाँचिए कि उत्तर से निकली $\mathrm{Re}$
मानी गई दशा से मेल खाती है।
\end{method}

\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:viscous-flows:6}

0.37x/\mathrm{Re}_x^{1/5}$ मानकर पिछले सिरे पर सीमांत-परत की मोटाई का आकलन।
\end{exercise}

सतह की चाल।
\end{exercise}

\section{समस्या: रक्त, कच्चा तेल और वर्षा}

\begin{problem}[{सप्ताहांत समस्या --- वही श्यान नियम किसी धमनी में,
किसी हज़ार किलोमीटर की पाइपलाइन में, और किसी गिरती बूँद में}]
\label{pb:b2:viscous-flows:1}

\medskip
\noindent\textbf{भाग I --- रक्त।}
रक्त: $\rho = \qty{1060}{kg/m^3}$, $\eta = \qty{3.5e-3}{Pa.s}$ (न्यूटनी
माना गया); हृद्-निर्गम $D_V = \qty{5.0}{L/min}$।
\begin{enumerate}
  \item महाधमनी, त्रिज्या \qty{1.25}{cm}: माध्य चाल, \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या}, दशा।
  \item यदि महाधमनी का प्रवाह प्वाज़ॉय का होता, तो अधिकतम चाल और
        दीवार का \omterm{def:b2:viscous-flows:viscosity}{अपरूपण प्रतिबल} क्या होते? (अंतःस्तर
        \qty{1}{Pa} कोटि के प्रतिबल भाँप लेता है।)
  \item \qty{4.0}{\micro m} त्रिज्या की कोई केशिका रक्त को
        \qty{0.30}{mm/s} से ले जाती है: \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या}; उसकी
        \qty{1.0}{mm} लंबाई पर दाब-गिरावट, पास्कल में और पारे के मिलीमीटरों में
        (\qty{1}{mmHg} = \qty{133}{Pa})।
  \item किसी लाल कोशिका को केशिका पार करने में लगा समय; गैस-विनिमय
        के लिए यही सही कोटि क्यों है?
  \item धमनिकाएँ (त्रिज्या \qty{10}{\micro m}, लंबाई \qty{2.0}{mm}, $10^8$
        समांतर में): पूरे समुच्चय का प्रतिरोध और पूरे हृद्-निर्गम पर
        उस पर दाब-गिरावट।
  \item धमनी से शिरा तक कुल गिरावट \qty{13}{kPa} है: वाम निलय द्वारा दी गई
        यांत्रिक शक्ति। \qty{100}{W} के शरीर से
        तुलना कीजिए।
  \item कोई संकीर्णन किसी हृद्-धमनी की त्रिज्या थोड़ी लंबाई में $40\%$
        घटा देता है: उसका प्रतिरोध कितने गुना बढ़ता है; और आगे की
        हृद्-पेशी को विराम में तो पूर्ति मिलती रह सकती है, पर
        श्रम के दौरान क्यों नहीं।
  \item कड़े श्रम के दौरान हृद्-निर्गम \qty{25}{L/min} तक पहुँच जाता है:
        महाधमनी में \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} और दशा।
  \item रक्त न्यूटनी नहीं है: उसकी आभासी \omterm{def:b2:viscous-flows:viscosity}{श्यानता} ऊँची
        अपरूपण-दर पर घटती है और नीची पर बढ़ती है। ऊपर की किन
        वाहिकाओं पर इसका सबसे अधिक असर पड़ता है, और किस दिशा में?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- पाइपलाइन।}
कच्चे तेल की कोई पाइपलाइन: व्यास $D = \qty{1.2}{m}$, लंबाई $L = \qty{1300}{km}$,
प्रवाह $D_V = \qty{3.0}{m^3/s}$; तेल $\rho = \qty{850}{kg/m^3}$, और गरम रखे जाने पर $\eta =
\qty{2.0e-2}{Pa.s}$। विक्षुब्ध घर्षण: $\Delta P = \lambda
(L/D)\tfrac12\rho\langle v\rangle^2$, $\lambda = 0.316/\mathrm{Re}^{1/4}$।
\begin{enumerate}[resume]
  \item माध्य चाल और \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या}; दशा।
  \item प्रति किलोमीटर और पूरी लाइन पर दाब-गिरावट।
  \item नली \qty{80}{bar} सह लेती है: लाइन पर पंप-केंद्रों की
        न्यूनतम संख्या।
  \item कुल पंपन-शक्ति; पहुँचाए गए प्रति घन मीटर तेल पर ऊर्जा,
        और तेल की अपनी ऊर्जा (\qty{36}{GJ/m^3}) से तुलना।
  \item प्वाज़ॉय का नियम कौन-सी गिरावट बताता? विक्षोभ की क़ीमत पर
        टिप्पणी दीजिए (और इस पर भी कि कुछ पाइपलाइनों में विक्षोभ को
        टालने वाले बहुलक क्यों मिलाए जाते हैं)।
  \item जाड़े में बिना गरम किया तेल $\eta = \qty{2.0}{Pa.s}$ तक पहुँच जाता:
        नई \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} और दशा; उपयुक्त नियम से लाइन पर
        दाब-गिरावट। अभिकल्प के लिए निष्कर्ष (असली
        लाइन ऊष्मारोधी है और तेल गरम रखा जाता है)।
  \item तेल का एक सिरे से दूसरे तक पारगमन-काल।
  \item गरम, विक्षुब्ध दशा में दाब-गिरावट से दीवार का \omterm{def:b2:viscous-flows:viscosity}{अपरूपण
        प्रतिबल}; नली की दीवार पर कुल \omterm{def:b2:rigid-body-mechanics:contact}{घर्षण बल}।
  \item वही प्रवाह \qty{1.0}{m} की नली में: विक्षुब्ध दाब-गिरावट कितने
        गुना बदलती है?
  \item गरम तेल में बालू का \qty{0.10}{mm} का कण ($\rho_s = \qty{2650}{kg/m^3}$)
        बहता है: स्टोक्स की अवसादन चाल, कण की \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स
        संख्या}, नली पार करके बैठने में लगा समय। इनकी तुलना
        विक्षुब्ध वेग-उतार-चढ़ावों से कीजिए, जो $\langle v
        \rangle$ के कुछ प्रतिशत हैं: क्या चलती पाइपलाइन में बालू बैठती है? रुकी
        हुई में?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- बूँदें।}
हवा: $\rho = \qty{1.2}{kg/m^3}$, $\eta = \qty{1.8e-5}{Pa.s}$; पानी $\rho_w =
\qty{1000}{kg/m^3}$, $\gamma = \qty{0.072}{N/m}$।
\begin{enumerate}[resume]
  \item बादल की बूँद, व्यास \qty{10}{\micro m}: स्टोक्स की सीमांत चाल,
        \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या}, \qty{1}{km} गिरने में लगा समय; बादल क्यों टिका रहता है।
  \item फुहार, \qty{0.20}{mm}: स्टोक्स की चाल और \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या}; क्या
        स्टोक्स अब भी भरोसे लायक़ है?
  \item वर्षा-बूँद, \qty{2.0}{mm}: दिखाइए कि स्टोक्स विफल है, और $C_x = 0.5$ के
        साथ सीमांत चाल निकालिए। \qty{1}{km} से गिरने में लगा समय।
  \item \qty{1}{cm/s} के ऊर्ध्व-प्रवाह में बादल की बूँद चढ़ती है या
        गिरती है? फुहार-बूँद को कौन-सा ऊर्ध्व-प्रवाह टिकाए रखेगा?
  \item तीनों बूँदों के भीतर \omterm{rem:b2:viscous-flows:capillarity}{लाप्लास दाब}; हर एक के लिए
        \omterm{thm:b2:euler-bernoulli:bernoulli}{गतिक दाब} $\tfrac12\rho v_\infty^2$ से तुलना कीजिए, और समझाइए कि
        बड़ी वर्षा-बूँदें लगभग \qty{5}{mm} से ऊपर चपटी होकर क्यों टूट जाती हैं।
  \item एक सारणी में इस समस्या के आठों प्रवाहों के लिए
        \omterm{def:b2:viscous-flows:reynolds}{रेनल्ड्स संख्या} और उन्हें चलाने वाला नियम दीजिए।
\end{enumerate}
\end{problem}
