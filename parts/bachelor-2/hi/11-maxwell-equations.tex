\chapter{मैक्सवेल के समीकरण}\label{ch:b2:maxwell-equations}

1865 में जेम्स क्लर्क मैक्सवेल ने वे नियम लिख डाले जो कूलॉम, ऐंपियर और
फ़ैराडे ने एक-एक करके पाए थे, ऐंपियर वाले में ऐसा पद जोड़ दिया जो अब तक
किसी प्रयोग ने माँगा न था, और पाया कि उसके समीकरण ऐसी तरंगें स्वीकार करते
हैं जो उस चाल से चलती हैं जिसे वह विद्युत् और चुंबकत्व के नियतांकों से
निकाल सकता था --- अर्थात् प्रकाश की चाल। विद्युत्, चुंबकत्व और प्रकाशिकी
एक ही विषय निकले। वर्ष 1 के खंड ने वे नियम पृष्ठों और पाशों पर समाकलों के
रूप में कहे थे; यह अध्याय उन्हें चार \emph{स्थानिक} समीकरणों के रूप में
लिखता है, जो हर बिंदु और हर क्षण पर सही हैं, उन सदिश संकारकों के साथ जो
ऐसा लिखना संभव बनाते हैं, और उनके पहले परिणाम निकालता है: \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा},
विभव, किसी अंतरापृष्ठ पर क्षेत्रों का व्यवहार, और वे सन्निकटन जिनके
अंतर्गत वर्ष 1 के खंड के परिपथ यथार्थ हैं।

\section{सदिश संकारक}

\begin{definition}[प्रवणता, अपसरण, कर्ल, लाप्लासीय]\label{def:b2:maxwell-equations:operators}
\index{प्रवणता}\index{अपसरण}\index{कर्ल}\index{लाप्लासीय}\index{नाबला}
किसी अदिश क्षेत्र $f$ और किसी सदिश क्षेत्र $\vect A$ के लिए, कार्तीय
निर्देशांकों में, सांकेतिक सदिश $\vect\nabla = (\partial_x, \partial_y,
\partial_z)$ के साथ:
\begin{align*}
\operatorname{\vect{grad}}f &= \vect\nabla f = (\partial_xf,\ \partial_yf,\ \partial_zf) ,\\
\operatorname{div}\vect A &= \vect\nabla\cdot\vect A = \partial_xA_x + \partial_yA_y + \partial_zA_z ,\\
\operatorname{\vect{curl}}\vect A &= \vect\nabla\wedge\vect A = (\partial_yA_z - \partial_zA_y,\ \partial_zA_x - \partial_xA_z,\ \partial_xA_y - \partial_yA_x) ,\\
\Delta f &= \operatorname{div}\operatorname{\vect{grad}}f = \partial_x^2f + \partial_y^2f + \partial_z^2f ,
\end{align*}
और $\Delta\vect A$ हर कार्तीय घटक पर लिया गया लाप्लासीय है।
इनका अर्थ निर्देशांकों से स्वतंत्र है: $\operatorname{\vect{grad}}f
\cdot\dd\vect l = \dd f$ $f$ का परिवर्तन है, $\dd\vect l$ के अनुदिश; $\operatorname{div}
\vect A$ किसी छोटे आयतन के पृष्ठ से $\vect A$ का बाहर जाता अभिवाह है,
प्रति एकांक आयतन; और $\operatorname{\vect{curl}}\vect A\cdot\vect n$ $\vect A$ का
परिचालन है, किसी ऐसे छोटे पाश के चारों ओर जिसका अभिलंब $\vect n$ हो, प्रति एकांक
क्षेत्रफल।
\end{definition}

\begin{theorem}[अपसरण और स्टोक्स की प्रमेयें]\label{thm:b2:maxwell-equations:theorems}
\index{अपसरण प्रमेय}\index{ऑस्ट्रोग्रैडस्की की प्रमेय}\index{स्टोक्स की प्रमेय}
किसी बंद पृष्ठ $\Sigma$ के लिए, जो आयतन $V$ को घेरता है, और किसी दिष्ट
बंद वक्र $C$ के लिए, जो पृष्ठ $S$ को घेरता है (अभिलंब $\vect n$ दक्षिण-हस्त नियम से),
\[
\iint_\Sigma\vect A\cdot\vect n\,\dd S = \iiint_V\operatorname{div}\vect A\,\dd\tau ,
\qquad
\oint_C\vect A\cdot\dd\vect l = \iint_S\operatorname{\vect{curl}}\vect A\cdot\vect n\,\dd S .
\]
\end{theorem}
\admitted

\begin{remark}[क्या स्वीकृत है और वह विश्वसनीय क्यों है]\label{rem:b2:maxwell-equations:whyadmitted}
दोनों प्रमेयें वर्ष 3 के गणित-खंड में सिद्ध हैं; इस वर्ष का
गणित-खंड उनके समतल रूप (ग्रीन--रीमान) और बहु-समाकलों पर रुक जाता है।
पर उनकी बात वही है जो हमने \cref{ch:b2:fluid-kinematics} में काम में ली:
आयतन को छोटे संदूकों से भर दीजिए (और पाश के पृष्ठ को छोटे पाशों से) ---
तब भीतरी फलकों के अभिवाह (और भीतरी किनारों के परिचालन)
जोड़े-जोड़े में कट जाते हैं, और केवल सीमा बचती है। परिभाषाओं से दो
सर्वसमिकाएँ निकलती हैं जो लगातार काम आती हैं:
$\operatorname{\vect{curl}}\operatorname{\vect{grad}}f = \vect 0$,
$\operatorname{div}\operatorname{\vect{curl}}\vect A = 0$, और
$\operatorname{\vect{curl}}\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}
\operatorname{div}\vect A - \Delta\vect A$। सममिति वाले क्षेत्रों के लिए
उपयोगी सूत्र ये हैं: किसी त्रिज्यीय क्षेत्र $A_r(r)\vect e_r$ के लिए $\operatorname{div}
= \frac1{r^2}\partial_r(r^2A_r)$ (गोलीय) या $\frac1r\partial_r(rA_r)$
(बेलनाकार); किसी अक्ष के चारों ओर किसी दिगंशी क्षेत्र $A_\theta(r)\vect e_\theta$ के
लिए, $(\operatorname{\vect{curl}})_z = \frac1r\partial_r(rA_\theta)$; और $f(r)$ के लिए,
$\Delta f = \frac1{r^2}\partial_r(r^2\partial_rf) = \frac1r\partial_r^2(rf)$ (गोलीय),
$\frac1r\partial_r(r\partial_rf)$ (बेलनाकार)।
\end{remark}

\section{चारों समीकरण}

\begin{theorem}[मैक्सवेल के समीकरण]\label{thm:b2:maxwell-equations:maxwell}
\index{मैक्सवेल के समीकरण}\index{मैक्सवेल--गाउस समीकरण}\index{मैक्सवेल--फ़ैराडे समीकरण}\index{मैक्सवेल--ऐंपियर समीकरण}\index{विस्थापन धारा}
निर्वात में (और ऐसे पदार्थ में जिसका वर्णन उसके कुल आवेश तथा धारा के
घनत्वों $\rho$ और $\vect j$ से हो), विद्युत और चुंबकीय क्षेत्र हर बिंदु और
हर समय इनका पालन करते हैं:
\begin{align*}
\operatorname{div}\vect E &= \frac\rho{\varepsilon_0} &&\text{(मैक्सवेल--गाउस)} , &
\operatorname{\vect{curl}}\vect E &= -\frac{\partial\vect B}{\partial t} &&\text{(मैक्सवेल--फ़ैराडे)} ,\\
\operatorname{div}\vect B &= 0 &&\text{(मैक्सवेल--अभिवाह)} , &
\operatorname{\vect{curl}}\vect B &= \mu_0\vect j + \mu_0\varepsilon_0\frac{\partial\vect E}{\partial t} &&\text{(मैक्सवेल--ऐंपियर)} ,
\end{align*}
जहाँ $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $\mu_0 = 4\pi \times \qty{1e-7}{H/m}$ (और
$\varepsilon_0\mu_0 = 1/c^2$)। क्षेत्र आवेशों पर लोरेंत्ज़
बल $\vect F = q(\vect E + \vect v\wedge\vect B)$ से काम करते हैं। पद $\varepsilon_0\partial_t\vect E$
मैक्सवेल का \emph{\omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा}} घनत्व है।
\end{theorem}

\begin{proof}[वर्ष 1 के खंड के समाकल नियमों से तुल्यता]
पहले दोनों पर \omterm{thm:b2:maxwell-equations:theorems}{अपसरण प्रमेय} लगाइए: $\iint_\Sigma\vect E\cdot
\vect n\,\dd S = Q_{\text{भीतरी}}/\varepsilon_0$ (गाउस), और किसी भी बंद पृष्ठ से
$\vect B$ का अभिवाह शून्य। अंतिम दोनों पर स्टोक्स लगाइए:
$\oint_C\vect E\cdot\dd\vect l = -\dd\Phi_B/\dd t$ (फ़ैराडे, किसी स्थिर पाश के लिए)
और $\oint_C\vect B\cdot\dd\vect l = \mu_0I_{\text{घिरी}} + \mu_0\varepsilon_0\,\dd\Phi_E/\dd t$
(ऐंपियर, पूरा किया हुआ)। और उल्टे, समाकल नियम, जो हर
आयतन और पाश के लिए सही हैं, स्थानिक नियमों को दे देते हैं। इस तरह स्थानिक समीकरण
वर्ष 1 के नियम और एक नया पद हैं, जिसे नीचे उचित ठहराया गया है; और उन्हें
विद्युत्चुंबकत्व की मूल-कल्पना मान लिया जाता है।
\end{proof}

\begin{proposition}[विस्थापन धारा क्यों आवश्यक है]\label{prop:b2:maxwell-equations:displacement}
\omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल--ऐंपियर समीकरण} का \omterm{def:b2:maxwell-equations:operators}{अपसरण} लेने और
$\operatorname{div}\operatorname{\vect{curl}} = 0$ तथा मैक्सवेल--गाउस काम में लेने पर
\[
0 = \mu_0\operatorname{div}\vect j + \mu_0\varepsilon_0\,\partial_t\operatorname{div}\vect E
= \mu_0\Bigl(\operatorname{div}\vect j + \frac{\partial\rho}{\partial t}\Bigr) :
\]
मिलता है, अर्थात् समीकरणों में आवेश-संरक्षण समाया है। विस्थापन-पद के
बिना वे सदा $\operatorname{div}\vect j = 0$ माँगते
--- जो किसी आवेशित होते संधारित्र में झूठ है, जहाँ चालन-धारा
प्लेट पर रुक जाती है। वहाँ, प्लेटों के बीच, $\varepsilon_0\partial_t\vect E$ ठीक
उतनी धारा ढोता है जितनी तार में आती है, और चुंबकीय क्षेत्र
अंतराल के चारों ओर वैसे ही परिचालित होता है जैसे किसी तार के चारों ओर।
\end{proposition}

\begin{proof}
दोनों पक्षों का \omterm{def:b2:maxwell-equations:operators}{अपसरण}, जैसा कहा गया; और संधारित्र के लिए, प्लेटों के बीच
किसी पृष्ठ से $\varepsilon_0\partial_t\vect E$ का अभिवाह
$\varepsilon_0S\,\dd E/\dd t = \dd Q/\dd t = I$ है।
\end{proof}

\begin{example}[आवेशित होते संधारित्र के भीतर क्षेत्र]\label{ex:b2:maxwell-equations:capacitor}
$R$ त्रिज्या की वृत्ताकार प्लेटें, अंतराल $d \ll R$, जो धारा
$I$ से आवेशित होती हैं: उनके बीच $E = Q/\varepsilon_0\pi R^2$ एकसमान है और $\dd E/
\dd t = I/\varepsilon_0\pi R^2$ से बढ़ता है। सममिति से $\vect B = B(r)\vect e_\theta$; और $r < R$
त्रिज्या के किसी वृत्त पर ऐंपियर--मैक्सवेल (उसे कोई चालन-धारा पार नहीं करती): $2\pi rB
= \mu_0\varepsilon_0\pi r^2\,\dd E/\dd t$, अतः $B = \mu_0Ir/2\pi R^2$ --- अर्थात् चक्रिका पर
फैली एकसमान धारा $I$ का क्षेत्र, जो $r = R$ पर बाहर के तार के
$\mu_0I/2\pi r$ से संतत मिलता है। $I = \qty{1}{A}$, $R = \qty{5}{cm}$ के लिए: $B(R) =
\qty{4}{\micro T}$, जो नापा जा सकता है; अर्थात् मैक्सवेल का पद असली है।
\end{example}

\begin{remark}[रैखिकता; समीकरण क्या कहते हैं]\label{rem:b2:maxwell-equations:linearity}
समीकरण क्षेत्रों और उनके स्रोतों में रैखिक हैं: अतः क्षेत्र
अध्यारोपित होते हैं। इनमें से दो (गाउस, फ़ैराडे--अभिवाह) कोई स्रोत नहीं ढोते और
संरचना कहते हैं --- अर्थात् विद्युत क्षेत्र-रेखाएँ आवेशों पर शुरू और समाप्त होती हैं,
और चुंबकीय क्षेत्र-रेखाएँ कभी समाप्त नहीं होतीं; बाक़ी दो कहते हैं कि आवेश और
धाराएँ, तथा हर क्षेत्र का परिवर्तन, क्षेत्रों को कैसे बनाते हैं। और साथ
पढ़ने पर: बदलता $\vect B$ परिचालित $\vect E$ बनाता है और बदलता
$\vect E$ परिचालित $\vect B$: अतः दोनों बिना कहीं किसी आवेश के एक-दूसरे को
थामे रख सकते हैं --- यही \cref{ch:b2:plane-waves-polarization} की तरंग है, और
\omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा} ही उसे संभव बनाती है।
\end{remark}

\section{विभव}

\begin{proposition}[अदिश और सदिश विभव]\label{prop:b2:maxwell-equations:potentials}
\index{सदिश विभव}\index{अदिश विभव}\index{गेज}\index{कूलॉम गेज}\index{प्वासों समीकरण}
चूँकि $\operatorname{div}\vect B = 0$ और $\operatorname{\vect{curl}}\vect E = -\partial_t
\vect B$, अतः क्षेत्र किसी \emph{\omterm{prop:b2:maxwell-equations:potentials}{सदिश विभव}} $\vect A$ और किसी
\emph{\omterm{prop:b2:maxwell-equations:potentials}{अदिश विभव}} $V$ से व्युत्पन्न होते हैं:
\[
\vect B = \operatorname{\vect{curl}}\vect A , \qquad
\vect E = -\operatorname{\vect{grad}}V - \frac{\partial\vect A}{\partial t} ,
\]
और वे किसी \emph{गेज रूपांतरण} $\vect A \to \vect A +
\operatorname{\vect{grad}}\chi$, $V \to V - \partial_t\chi$ तक ही तय होते हैं, जो क्षेत्रों को
अपरिवर्तित छोड़ देता है। \emph{\omterm{prop:b2:maxwell-equations:potentials}{कूलॉम गेज}} $\operatorname{div}\vect A = 0$ में गाउस का
नियम \emph{\omterm{prop:b2:maxwell-equations:potentials}{प्वासों समीकरण}}
\[
\Delta V = -\frac\rho{\varepsilon_0} ,
\]
बन जाता है, और स्थायी दशा में ऐंपियर का नियम $\Delta\vect A = -\mu_0\vect j$।
स्थैतिकी में $V$ वर्ष 1 के खंड का स्थिरवैद्युत विभव है।
\end{proposition}

\begin{proof}
कोई अपसरण-रहित क्षेत्र कोई \omterm{def:b2:maxwell-equations:operators}{कर्ल} होता है (स्वीकृत, और उसी तरह उल्टा
कथन भी कि कोई कर्ल-रहित क्षेत्र कोई \omterm{def:b2:maxwell-equations:operators}{प्रवणता} होता है, दोनों सरल संबद्ध
प्रदेशों के लिए वर्ष 2 के गणित-खंड से); तब $\operatorname{\vect{curl}}
(\vect E + \partial_t\vect A) = \vect 0$, अतः $\vect E + \partial_t\vect A$ कोई \omterm{def:b2:maxwell-equations:operators}{प्रवणता} है।
गेज: $\operatorname{\vect{curl}}\operatorname{\vect{grad}}\chi = \vect 0$, और
$\partial_t\operatorname{\vect{grad}}\chi$ वाले पद $\vect E$ में कट जाते हैं। प्वासों: $\operatorname{div}
\vect E = -\Delta V - \partial_t\operatorname{div}\vect A$। स्थैतिकी: $\operatorname{\vect{curl}}
\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}\operatorname{div}\vect A - \Delta\vect A
= \mu_0\vect j$।
\end{proof}

\begin{example}[दो सदिश विभव]\label{ex:b2:maxwell-equations:vectorpot}
कोई एकसमान क्षेत्र $\vect B_0$ $\vect A = \tfrac12\vect B_0\wedge\vect r$ से व्युत्पन्न होता है
(जाँचिए: $\operatorname{\vect{curl}}(\tfrac12\vect B_0\wedge\vect r) = \vect B_0$, और
$\operatorname{div}\vect A = 0$); और किसी अनंत परिनालिका के भीतर यह $A_\theta =
Br/2$ है, बाहर $A_\theta = BR^2/2r$ --- अर्थात् ऐसा विभव जो वहाँ लुप्त नहीं होता
जहाँ क्षेत्र लुप्त होता है, और जिसका किसी पाश पर परिचालन उससे होकर
$\vect B$ का अभिवाह है: $\oint\vect A\cdot\dd\vect l = \Phi_B$ (स्टोक्स)। तब फ़ैराडे का
नियम प्रेरित क्षेत्र के लिए $\vect E = -\partial_t\vect A$ पढ़ा जाता है।
\end{example}

\section{अंतरापृष्ठ पर क्षेत्र}

\begin{proposition}[सीमा-संबंध]\label{prop:b2:maxwell-equations:boundary}
\index{सीमा-संबंध (क्षेत्र)}\index{पृष्ठ धारा}
पृष्ठ \omterm{def:b2:charges-currents-conduction:densities}{आवेश घनत्व} $\sigma$ और पृष्ठ \omterm{def:b2:charges-currents-conduction:densities}{धारा घनत्व} $\vect j_s$ (\unit{A/m}) ढोते
किसी पृष्ठ के दोनों ओर, जहाँ $\vect n_{12}$ माध्यम 1 से माध्यम 2 की ओर मात्रक
अभिलंब है,
\[
\vect E_2 - \vect E_1 = \frac\sigma{\varepsilon_0}\,\vect n_{12} , \qquad
\vect B_2 - \vect B_1 = \mu_0\,\vect j_s\wedge\vect n_{12} :
\]
होता है: अर्थात् $\vect E$ का स्पर्शीय घटक और $\vect B$ का अभिलंब घटक
संतत रहते हैं; $\vect E$ का अभिलंब घटक $\sigma/\varepsilon_0$ से कूदता है,
और $\vect B$ का स्पर्शीय घटक $\mu_0j_s$ से। किसी पूर्ण चालक के पृष्ठ पर
(भीतर $\vect E = \vect B = \vect 0$, जैसा \cref{ch:b2:wave-interfaces} की
दशाओं में), $\vect E$ अभिलंब और $\sigma\vect n/
\varepsilon_0$ के बराबर होता है,
तथा $\vect B$ स्पर्शीय और $\mu_0\vect j_s\wedge\vect n$ के बराबर।
\end{proposition}

\begin{proof}
अभिलंब घटक: पृष्ठ पर टिकी किसी चपटी डिबिया पर, जिसका क्षेत्रफल $\dd S$ हो
और ऊँचाई लुप्त होती हो, गाउस का नियम और अभिवाह-नियम: $(E_{2n}
- E_{1n})\dd S = \sigma\dd S/\varepsilon_0$, $(B_{2n} - B_{1n})\dd S = 0$। स्पर्शीय
घटक: पृष्ठ पर टिके किसी छोटे आयत पर, जिसकी लंबाई $\dd l$ हो और ऊँचाई
लुप्त होती हो, फ़ैराडे और ऐंपियर के नियम: आयत से होकर
$\partial_t\vect B$ और $\partial_t\vect E$ के अभिवाह उसके क्षेत्रफल के साथ लुप्त हो
जाते हैं, पर कोई \omterm{prop:b2:maxwell-equations:boundary}{पृष्ठ धारा} $j_s\,\dd l$ उसे पार करती है:
अतः $(E_{2t} - E_{1t})\dd l = 0$, $(B_{2t} - B_{1t})\dd l = \mu_0j_s\dd l$, उस
स्पर्शीय दिशा के लिए जो $\vect j_s$ के लंबवत है।
\end{proof}

\begin{example}[आवेशित समतल, धारा-चादर, परिनालिका]\label{ex:b2:maxwell-equations:sheets}

अर्थात् वर्ष 1 का परिणाम एक ही पंक्ति में। $j_s$ धारा की कोई समतल चादर: $B =

परिनालिका, जो $I$ ढोती हो, $j_s = nI$ की कोई बेलनाकार धारा-चादर है: और उसके दोनों ओर उछाल
$\mu_0nI$, तथा बाहर $B = 0$, से भीतर $B = \mu_0nI$ मिलता है।
\end{example}

\section{अर्ध-स्थायी दशाएँ}

\begin{definition}[अर्ध-स्थायी सन्निकटन]\label{def:b2:maxwell-equations:arqs}
\index{अर्ध-स्थायी सन्निकटन}\index{ARQS}
$L$ आकार का कोई निकाय, जो आवृत्ति $f$ पर चलाया जाए, तब \emph{अर्ध-स्थायी}
होता है जब $L \ll \lambda = c/f$: तब क्षेत्रों को उसे पार करने में जो समय $L/c$
लगता है वह आवर्तकाल के सामने नगण्य है, और उसके सारे बिंदु स्रोतों को
एक ही क्षण पर ``देखते'' हैं। दो सीमाएँ काम आती हैं:
\begin{itemize}
  \item \emph{चुंबकीय} अर्ध-स्थायी दशा (धाराओं, कुंडलियों, परिणामित्रों,
        प्रेरण वाले परिपथ): यहाँ ऐंपियर के नियम से \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन
        धारा} छोड़ दी जाती है, $\operatorname{\vect{curl}}\vect B
        = \mu_0\vect j$ (अतः $\operatorname{div}\vect j = 0$ और धाराएँ बंद
        परिपथों में बहती हैं), जबकि फ़ैराडे का नियम पूरा रखा जाता है;
  \item \emph{वैद्युत} अर्ध-स्थायी दशा (कोई आवेशित होता
        संधारित्र): यहाँ फ़ैराडे का पद छोड़ दिया जाता है, $\vect E$ तात्क्षणिक
        आवेशों का स्थिरवैद्युत क्षेत्र है, जबकि $\vect B$ निकालने के लिए
        \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा} रखी जाती है।
\end{itemize}
\end{definition}

\begin{proof}[औचित्य]
चुंबकीय दशा में किसी चालक में ऐंपियर के नियम के दोनों पदों की तुलना
कीजिए: $\mu_0\varepsilon_0\partial_t E \sim \mu_0\varepsilon_0\omega E$ बनाम $\mu_0\gamma E$,
अर्थात् अनुपात $\varepsilon_0\omega/\gamma = \omega\tau_r \sim 10^{-17}$, ताँबे में \qty{50}{Hz} पर;
और चालक के बाहर \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा}
$\varepsilon_0\omega E$ कोटि की है, जहाँ $E \sim \omega AB$, अर्थात् चालन-धारा के
प्रभाव से $(\omega L/c)^2 = (L/\lambda)^2 \times 4\pi^2$ गुना छोटी। वैद्युत
दशा में प्रेरित क्षेत्र $\omega BL \sim \omega^2\varepsilon_0\mu_0L^2E$ $E$ से उसी
$(L/\lambda)^2$ गुना छोटा है।
\end{proof}

\begin{example}[परिपथ और ऐंटेना]\label{ex:b2:maxwell-equations:sizes}
\qty{50}{Hz} का कोई मुख्य-लाइन परिपथ ($\lambda = \qty{6000}{km}$) किसी देश के
आकार तक अर्ध-स्थायी रहता है --- और इसी से किरखोफ़ के नियम तथा
परिणामित्र चलते हैं। \qty{10}{cm} का कोई परिपथ लगभग
\qty{100}{MHz} तक अर्ध-स्थायी है; और \qty{1}{GHz} ($\lambda = \qty{30}{cm}$) पर उसके तार ऐंटेना बन जाते हैं,
धाराएँ तार के अनुदिश भिन्न हो जाती हैं, और परिपथ विकिरण करने लगता है: यही
\cref{ch:b2:dipole-radiation} का और सूक्ष्मतरंग-अभिकल्प का प्रदेश है, जहाँ किसी पट्ट पर
कोई पथ एक \omterm{prop:b2:dispersion-wave-packets:coax}{संचरण-रेखा} होता है।
\end{example}

\begin{method}[स्थानिक समीकरणों का प्रयोग]\label{met:b2:maxwell-equations:method}
(1) क्षेत्रों की दिशा और चर तय करने के लिए स्रोतों की सममितियाँ और
अपरिवर्त्यताएँ काम में लीजिए (जैसा वर्ष 1 के खंड में)।
(2) रूप चुनिए: जहाँ पर्याप्त सममिति हो वहाँ समाकल (किसी सुचुने पृष्ठ या
पाश पर गाउस, ऐंपियर); और जहाँ कोई अवकल समीकरण चाहिए वहाँ स्थानिक
($V$ के लिए प्वासों, क्षेत्रों के लिए विसरण या तरंग समीकरण)।
(3) किसी समय-निर्भर समस्या में दशा तय कीजिए:
स्थैतिक, अर्ध-स्थायी चुंबकीय या वैद्युत, या पूरा मैक्सवेल। (4)
अंतरापृष्ठों पर सीमा-संबंध लीजिए --- केवल वहीं
पृष्ठ आवेश और धाराएँ आती हैं। (5)
$\operatorname{div}\vect B = 0$ और विमाएँ जाँचिए।
\end{method}

\omterm{def:b2:maxwell-equations:operators}{प्रवणताएँ} हैं, कौन \omterm{def:b2:maxwell-equations:operators}{कर्ल}, और कौन दोनों में से कुछ नहीं?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:2}

\section{समस्या: संधारित्र, बीटाट्रॉन और विभव}

\begin{problem}[{सप्ताहांत समस्या --- मैक्सवेल के समीकरण काम पर: किसी एक
आवेशित होते संधारित्र में, उस त्वरक में जो अकेले फ़ैराडे के नियम से चलता
है, और उन विभवों में जो उन्हें हल करते हैं}]
\label{pb:b2:maxwell-equations:1}

\medskip
\noindent\textbf{भाग I --- आवेशित होता संधारित्र।}
वृत्ताकार प्लेटें, त्रिज्या $R = \qty{10}{cm}$, अंतराल $d = \qty{1.0}{mm}$,
हवा; और अक्ष के अनुदिश खिलाते तार में धारा $I(t) = I_0\cos\omega t$, जहाँ $I_0 = \qty{1.0}{A}$, $f = \qty{1.0}{MHz}$।
किनारे के प्रभाव छोड़ दिए गए हैं।
\begin{enumerate}
  \item प्लेटों के बीच आवेश $Q(t)$ और क्षेत्र $E(t)$; तथा
        $E$ का आयाम।
  \item प्लेटों के बीच विस्थापन-धारा घनत्व; और जाँचिए कि
        किसी प्लेट से उसका अभिवाह $I(t)$ के बराबर है।
  \item प्लेटों के बीच चुंबकीय क्षेत्र $B(r, t)$, $r < R$ के लिए, समाकल
        ऐंपियर--मैक्सवेल नियम से; और $r = R$ पर आयाम।
  \item $r > R$ के लिए क्षेत्र (अब भी प्लेटों के तलों के बीच):
        उसी $r$ पर तार के क्षेत्र से तुलना कीजिए; क्या $r = R$ पर कोई
        असंततता है?
  \item बदलता $\vect B$ फ़ैराडे के नियम से विद्युत क्षेत्र में कोई सुधार
        $\vect E_1$ प्रेरित करता है: $\operatorname{\vect{curl}}\vect E_1
        = -\partial_t\vect B$ से, $\vect E_1 = E_1(r)\vect e_z$ (अक्षीय) लेकर, दिखाइए कि
        $E_1(r) - E_1(0) = \dfrac{\mu_0\varepsilon_0r^2}4\,\dfrac{\dd^2E}{\dd t^2}$, और
        $|E_1(R) - E_1(0)|/|E|$ आँकिए।
  \item \qty{1}{MHz} पर वैद्युत अर्ध-स्थायी विवेचन की मान्यता पर
        निष्कर्ष निकालिए; किस आवृत्ति पर यह सुधार
        $10\%$ तक पहुँच जाएगा? (यथार्थ हल कोई बेसेल फलन है: तब
        संधारित्र कोई गुहा बन चुका होता है।)
  \item ऊर्जा: प्लेटों के बीच संचित विद्युत ऊर्जा $\tfrac12\varepsilon_0E^2$ का
        अधिकतम मान; चुंबकीय ऊर्जा $\int B^2/2\mu_0\,\dd\tau$, $E = 0$ वाले क्षण पर;
        और उनका अनुपात तथा प्रश्न 5 के आकलन से उसका
        संबंध।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- बीटाट्रॉन।}
कोई विद्युत्चुंबक सममित क्षेत्र $\vect B = B(r, t)\vect e_z$ बनाता है, जो
$z$ अक्ष के परितः सममित है; और इलेक्ट्रॉन $r_0$ त्रिज्या के किसी निर्वात वलय में घूमते हैं।
\begin{enumerate}[resume]
  \item समाकल रूप में मैक्सवेल--फ़ैराडे से दिखाइए कि
        कक्षा पर विद्युत क्षेत्र $E_\theta(r_0) = -\dfrac1{2\pi r_0}
        \dfrac{\dd\Phi}{\dd t}$ है, जहाँ $\Phi$ कक्षा से होकर अभिवाह है।
  \item किसी इलेक्ट्रॉन (आवेश $-e$) का स्पर्शीय गति-समीकरण:
        परिमाण में $\dd p/\dd t = eE_\theta$; और प्रति चक्कर पाई गई ऊर्जा
        $e\,|\dd\Phi/\dd t|$।
  \item वृत्त पर त्रिज्यीय साम्य: $p = eB(r_0)r_0$। $r_0$ के
        अचर रहने के लिए दिखाइए कि $B(r_0, t) = \Phi(t)/2\pi r_0^2$, अर्थात्
        कक्षा पर क्षेत्र उसके भीतर के माध्य क्षेत्र का आधा होना चाहिए।
  \item किसी एकसमान क्षेत्र से इलेक्ट्रॉन त्वरित क्यों नहीं किए जा सकते?
        यह प्रतिबंध व्यवहार में कैसे पूरा किया जाता है (ध्रुव-खंडों का
        आकार)?
  \item $r_0 = \qty{0.50}{m}$, और $B(r_0)$ \qty{5}{ms} में $0$ से \qty{0.40}{T} तक
        चढ़ाया गया: अंतिम संवेग; अंतिम ऊर्जा (इन
        अति-आपेक्षिकीय इलेक्ट्रॉनों के लिए $E = pc$ लीजिए, $\qty{1}{MeV} = \qty{1.6e-13}{J}$);
        प्रति चक्कर ऊर्जा और चक्करों की संख्या; और तय की गई दूरी।
  \item वृत्त पर त्वरित होने पर इलेक्ट्रॉन विकिरण करते हैं
        (\cref{ch:b2:dipole-radiation}): और यही बीटाट्रॉनों को कुछ
        सौ \unit{MeV} तक सीमित रखता है। इलेक्ट्रॉनों की ऊर्जा अंततः कहाँ से
        आती है, और \omterm{thm:b2:maxwell-equations:maxwell}{मैक्सवेल के समीकरणों} के किस पद से होकर?
  \item त्वरण के बाद इलेक्ट्रॉन किसी लक्ष्य पर फेंके जाते हैं और
        X-किरणें बनाते हैं: बीटाट्रॉन 1950 के दशक की विकिरण-चिकित्सा मशीन
        क्यों था, और उसकी जगह क्या ले आया?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- विभव और प्वासों।}
\begin{enumerate}[resume]
  \item $\vect E$ और $\vect B$ को $V$ तथा $\vect A$ के पदों में लिखिए, और जाँचिए
        कि तब मैक्सवेल--फ़ैराडे तथा मैक्सवेल--अभिवाह अपने आप
        संतुष्ट हो जाते हैं।
  \item $a$ त्रिज्या का कोई गोला एकसमान घनत्व $\rho_0$ ढोता है: गोलीय
        सममिति में \omterm{prop:b2:maxwell-equations:potentials}{प्वासों समीकरण} लिखिए, उसे भीतर और बाहर हल कीजिए
        (अनंत पर $V \to 0$, तथा $V$ और $V'$ $a$ पर संतत), और
        वर्ष 1 के खंड के क्षेत्र फिर से पाइए।
  \item गोले की स्थितिज ऊर्जा, $\tfrac12\int\rho V\,\dd\tau$: मान
        $3Q^2/20\pi\varepsilon_0a$; और किसी यूरेनियम नाभिक ($Q = 92e$, $a =
        \qty{7.4}{fm}$) के लिए \unit{MeV} में --- अर्थात् वह ऊर्जा जो उसके टूटने पर निकलती है।
  \item वैद्युत अर्ध-स्थायी दशा में भाग I के संधारित्र के लिए
        प्लेटों के बीच $V$ दीजिए। दिखाइए कि कोई \emph{अक्षीय}
        \omterm{prop:b2:maxwell-equations:potentials}{सदिश विभव} $A_z(r, t)\,\vect e_z$, जिसके लिए $(\operatorname{\vect{curl}}
        \vect A)_\theta = -\partial_rA_z$, प्रश्न 3 का क्षेत्र $B_\theta$ फिर से देता है,
        और $A_z$ निकालिए, $r < R$ के लिए ($A_z(0) = 0$ लीजिए)।
  \item दिखाइए कि प्रश्न 5 का सुधार $\vect E_1$ ठीक-ठीक $-\partial_t\vect A$ है,
        उस $A_z$ के लिए, किसी अचर तक।
  \item गेज: कूलॉम प्रतिबंध $\operatorname{div}\vect A = 0$ आपके $A_z(r)$ के लिए
        सही उतरता है --- क्यों? और कौन-सा दूसरा $\vect A$ वही $\vect B$ देता?
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- अंतरापृष्ठ और दशाएँ।}
\begin{enumerate}[resume]
  \item अधिकतम $E$ वाले क्षण पर भाग I की प्लेटों पर पृष्ठ आवेश;
        और प्लेट के भीतरी फलक पर सीमा-संबंध जाँचिए (धातु के भीतर
        क्षेत्र शून्य)।
  \item धारा प्लेट में अक्ष से किनारे तक त्रिज्यीय रूप से फैलती है:
        प्लेट में पृष्ठ \omterm{def:b2:charges-currents-conduction:densities}{धारा घनत्व} $j_s(r)$; और सीमा-संबंध से प्लेट के
        बाहरी फलक के ठीक बाहर चुंबकीय क्षेत्र, जिसकी तुलना प्रश्न 3 में
        भीतरी ओर मिले क्षेत्र से कीजिए।
  \item वर्गीकृत कीजिए: \qty{1}{MHz} पर प्लेटें (वैद्युत अर्ध-स्थायी?), \qty{20}{cm} का
        खिलाता तार-पाश (चुंबकीय अर्ध-स्थायी?), और वही संधारित्र
        \qty{1}{GHz} पर; $L/\lambda$ से उचित ठहराइए।
  \item बीटाट्रॉन के वलय में चुंबकीय \omterm{def:b2:maxwell-equations:arqs}{अर्ध-स्थायी सन्निकटन}
        इतना अच्छा क्यों है (आकार, \qty{100}{Hz} की चढ़ाई), जबकि वहाँ
        विद्युत क्षेत्र ही अनिवार्य है?
  \item एक सारणी में सार दीजिए कि मैक्सवेल के किस समीकरण (स्थानिक या
        समाकल) ने हर भाग का उत्तर दिया, और कौन-सा पद --- \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन
        धारा}, फ़ैराडे, गाउस, सीमा-उछाल --- निर्णायक रहा।
\end{enumerate}
\end{problem}
