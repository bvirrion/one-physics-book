\chapter{दो-तरंग व्यतिकरण}\label{ch:b2:two-wave-interference}

कोई साबुन का बुलबुला किसी रंगहीन द्रव का बना है, फिर भी वह रंगों से
झिलमिलाता है; किसी गीली सड़क पर तेल की झिल्ली वही इंद्रधनुषी पट्टियाँ
दिखाती है; और किसी व्याध-पतंग के पंख हरे तथा बैंगनी चमकते हैं। इनमें से
कोई भी रंग किसी वर्णक से नहीं आता: वे एक ही प्रकाश-तरंग की दो प्रतियों से
आते हैं, जो किसी पतली झिल्ली के दोनों फलकों से परावर्तित होकर, किसी विलंब
के साथ पहुँचती हैं और या तो एक-दूसरे को पुष्ट करती हैं या रद्द कर देती हैं
--- अर्थात् दो तरंगों का \emph{व्यतिकरण}। यंग ने 1801 में, दो झिर्रियों
और एक मोमबत्ती से, उसे इस बात का प्रमाण बना दिया कि प्रकाश कोई तरंग है;
यह अध्याय सूत्र, यंग की फ्रिंजों की ज्यामिति, वे दो शर्तें --- कालिक और
\omterm{prop:b2:two-wave-interference:spatial}{दैशिक संसक्ति} --- जिनके अधीन फ्रिंजें बची रहती हैं, और पतली झिल्ली के
रंग सामने रखता है।

\section{व्यतिकरण का सूत्र}

\begin{theorem}[दो-तरंग व्यतिकरण]\label{thm:b2:two-wave-interference:formula}
\index{व्यतिकरण}\index{पथांतर}\index{व्यतिकरण की कोटि}\index{विभेद (फ्रिंज)}
तीव्रताओं $I_1$, $I_2$ और कलांतर $\Delta\varphi$ वाली दो संसक्त \omterm{def:b2:scalar-light-model:scalar}{एकवर्णी
तरंगें} किसी बिंदु $M$ पर वहाँ तीव्रता देती हैं
\[
I(M) = I_1 + I_2 + 2\sqrt{I_1I_2}\,\cos\Delta\varphi(M) , \qquad
\Delta\varphi = \frac{2\pi}{\lambda_0}\,\delta(M) ,
\]
जहाँ $\delta$ स्रोत से $M$ तक के दोनों रास्तों के बीच \emph{प्रकाशिक \omterm{thm:b2:two-wave-interference:formula}{पथांतर}}
है। तीव्रता वहाँ अधिकतम है ($I_1 + I_2 +
2\sqrt{I_1I_2}$) जहाँ $\delta = p\lambda_0$ हो, $p$ कोई पूर्णांक --- अर्थात्
\emph{\omterm{thm:b2:two-wave-interference:formula}{व्यतिकरण की कोटि}} --- और वहाँ न्यूनतम ($I_1 + I_2 - 2\sqrt{I_1I_2}$) जहाँ $\delta = (p +
\tfrac12)\lambda_0$। फ्रिंजों का \emph{विभेद} (दृश्यता) है
\[
V = \frac{I_{\max} - I_{\min}}{I_{\max} + I_{\min}} = \frac{2\sqrt{I_1I_2}}{I_1 + I_2} ,
\]
जो बराबर तीव्रताओं के लिए $1$ है, और तब न्यूनतम बिलकुल अँधेरे होते हैं:
$I = 2I_0(1 + \cos\Delta\varphi) = 4I_0\cos^2(\Delta\varphi/2)$।
\end{theorem}

\begin{proof}
$s = a_1\cos(\omega t - \varphi_1) + a_2\cos(\omega t - \varphi_2)$; और $\langle s^2\rangle$ वैसे ही जैसे
\cref{ch:b2:scalar-light-model} में, जहाँ संसूचन के दौरान $\Delta\varphi$ स्थिर है;
$I_i = \tfrac12Ka_i^2$। स्रोत से साथ निकली और \omterm{def:b2:scalar-light-model:path}{प्रकाशिक पथ} $L_1$, $L_2$ तय
करती दो तरंगों का कलांतर
$2\pi(L_2 - L_1)/\lambda_0$ है (और हर उस परावर्तन के लिए $\pi$ जोड़कर जो चिह्न
उलट दे, \cref{ch:b2:wave-interfaces})।
\end{proof}

\begin{remark}[तीन शर्तें]\label{rem:b2:two-wave-interference:conditions}
\index{संसक्ति (दैशिक)}\index{संसक्ति (कालिक)}
फ्रिंजें तभी होती हैं जब (1) दोनों तरंगें \emph{एक ही स्रोत} से आएँ
(किसी एक कंपन की दो प्रतियाँ, नहीं तो $\Delta\varphi$ यादृच्छिक है); (2)
\omterm{thm:b2:two-wave-interference:formula}{पथांतर} \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} से छोटा हो, $|\delta| < \ell_c$
(\emph{कालिक संसक्ति}: उससे परे प्रतियाँ अलग-अलग
तरंग-रेलों की हैं); (3) स्रोत इतना छोटा हो कि उसके अलग-अलग बिंदु
खिसकी हुई ऐसी फ्रिंज-व्यवस्थाएँ न बनाएँ जो एक-दूसरे को धो डालें
(\emph{\omterm{prop:b2:two-wave-interference:spatial}{दैशिक संसक्ति}}, नीचे)। और कोई चौथी, जो प्रायः पूरी रहती है:
दोनों तरंगों के ध्रुवण लंबवत न हों (वे तीव्रता में जुड़ जातीं,
मशालों की तरह)।
\end{remark}

\section{यंग की झिर्रियाँ}

\begin{proposition}[तरंगाग्र-विभाजन: यंग का प्रयोग]\label{prop:b2:two-wave-interference:young}
\index{यंग की झिर्रियाँ}\index{फ्रिंज-चौड़ाई}\index{तरंगाग्र-विभाजन}
कोई बिंदु स्रोत $S$ (या कोई झिर्री) दो सँकरी समांतर झिर्रियों
$S_1$, $S_2$ को जगमगाता है, जो $a$ दूरी पर और $S$ से बराबर दूरी पर हैं; और कोई पर्दा
दूरी $D \gg a$ पर खड़ा है। पर्दे के उस बिंदु पर जो ऊँचाई $x$ पर है ($a$ के
समांतर) \omterm{thm:b2:two-wave-interference:formula}{पथांतर} है
\[
\delta = S_2M - S_1M \approx \frac{ax}D ,
\]
और पर्दा सीधी, बराबर दूरी वाली फ्रिंजें दिखाता है, जो $x_p =
p\lambda D/a$ पर चमकीली हैं, और जिनकी \emph{फ्रिंज-चौड़ाई}
\[
i = \frac{\lambda D}a .
\]
फ्रिंजें झिर्रियों के पीछे हर जगह होती हैं (वे \emph{स्थानीकृत
नहीं} हैं); और झिर्रियों के बाद फ़ोकस दूरी $f$ का कोई लेंस रख देने पर वे
उसके फ़ोकस तल में $i = \lambda f/a$ के साथ दिखती हैं (फ्रिंजें "अनंत पर")।
तीव्रता $4I_0\cos^2(\pi ax/\lambda D)$ है।
\end{proposition}

\begin{proof}
$S_2M^2 - S_1M^2 = [(x + a/2)^2 - (x - a/2)^2] = 2ax$ और $S_2M + S_1M \approx 2D$,
अतः $\delta \approx 2ax/2D$। दो कोटियाँ $\lambda$ से अलग हैं, अर्थात् $\Delta x = \lambda D/a$ से।
लेंस के साथ, झिर्रियों से अक्ष से $\theta$ कोण पर निकलती दोनों तरंगों का
\omterm{thm:b2:two-wave-interference:formula}{पथांतर} $a\sin\theta \approx a\theta$ है और वे $x = f\theta$ पर मिलती हैं।
\end{proof}

\begin{example}[आँकड़े, और श्वेत प्रकाश]\label{ex:b2:two-wave-interference:numbers}
$a = \qty{0.5}{mm}$, $D = \qty{2}{m}$, $\lambda = \qty{600}{nm}$: $i = \qty{2.4}{mm}$ --- देखने भर
को काफ़ी बड़ी, और इसीलिए \omterm{prop:b2:two-wave-interference:young}{यंग की झिर्रियाँ} पास-पास तथा पर्दा
दूर होना चाहिए। श्वेत प्रकाश के साथ हर रंग अपनी अलग फ्रिंजें बनाता है, जो सब
केंद्र पर ($\delta = 0$) संपाती हैं: अर्थात् कोई श्वेत केंद्रीय फ्रिंज, फिर
रंगीन फ्रिंजें जैसे-जैसे कोटियाँ खिसकती जाती हैं (लाल $i$ बैंगनी का $1.8$ गुना
है), और कुछ कोटियों के बाद कोई एकसमान सफ़ेदी लिया क्षेत्र --- क्योंकि
श्वेत प्रकाश की \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} कोई माइक्रोमीटर, अर्थात् दो-तीन
तरंगदैर्घ्य है। उस "श्वेत" क्षेत्र पर ताना गया कोई वर्णक्रममापी फिर भी
उन तरंगदैर्घ्यों पर अँधेरी पट्टियाँ देखता है जिनके लिए $\delta = (p + \tfrac12)\lambda$: अर्थात् कोई
\emph{खाँचेदार वर्णक्रम}।
\end{example}

\begin{proposition}[दैशिक संसक्ति: स्रोत छोटा होना चाहिए]\label{prop:b2:two-wave-interference:spatial}
\index{दैशिक संसक्ति}\index{विस्तृत स्रोत}
किसी \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} का हर बिंदु अपनी अलग फ्रिंज-व्यवस्था बनाता है; और
अनुप्रस्थ रूप से $b$ खिसका कोई स्रोत-बिंदु, जो झिर्रियों से पहले दूरी $L$ पर हो,
फ्रिंजों को $x = bD/L$ खिसका देता है (\omterm{thm:b2:two-wave-interference:formula}{पथांतर}
$ab/L$ पा लेता है)। चौड़ाई $b$ का कोई स्रोत फ्रिंजों को एक फ्रिंज-चौड़ाई भर फैला देता है,
और विभेद तब लुप्त हो जाता है जब
\[
\frac{ab}L \approx \lambda , \qquad\text{अर्थात् जब झिर्रियाँ स्रोत को इस कोण के नीचे देखें } \frac bL \gtrsim \frac\lambda a .
\]
दोनों झिर्रियों को प्रकाश के \emph{संसक्ति क्षेत्रफल} के भीतर होना चाहिए ---
अर्थात् उस इलाक़े के भीतर जिस पर स्रोत से आई तरंग की कोई सुपरिभाषित
कला हो, और जिसकी चौड़ाई $\lambda L/b$ है। धूप ($b/L = \qty{0.5}{\degree} = \qty{9}{mrad}$)
संसक्ति चौड़ाई $\lambda/0.009 \approx \qty{60}{\micro m}$ देती है: यंग को उसे चौड़ा करने के लिए कोई
सूचिछिद्र चाहिए था।
\end{proposition}

\begin{proof}
ऊँचाई $b$ पर किसी स्रोत-बिंदु के लिए अतिरिक्त पथ $SS_2 - SS_1 \approx -ab/L$
$ax/D$ में जुड़ जाता है: अतः आकृति $x = bD/L$ खिसक जाती है। खिसके हुए
$\cos^2$ रूपों को चौड़ाई $b$ के स्रोत पर समाकलित करने से विभेद
$|\operatorname{sinc}(\pi ab/\lambda L)|$ मिलता है, जो $ab/L = \lambda$ पर शून्य है।
\end{proof}

\section{पतली झिल्लियाँ: आयाम-विभाजन}

\begin{proposition}[किसी पतली झिल्ली में व्यतिकरण]\label{prop:b2:two-wave-interference:film}
\index{पतली-झिल्ली व्यतिकरण}\index{आयाम-विभाजन}\index{समान नति की फ्रिंजें}\index{समान मोटाई की फ्रिंजें}
हवा में अपवर्तनांक $n$ और मोटाई $e$ की कोई झिल्ली, जो आपतन $i$ पर जगमगाई हो
(अपवर्तन कोण $r$), अपने दोनों फलकों से परावर्तित दो तरंगें लौटाती है,
जिनका \omterm{thm:b2:two-wave-interference:formula}{पथांतर}
\[
\delta = 2ne\cos r \ (+\tfrac{\lambda_0}2) ,
\]
जहाँ अतिरिक्त आधा तरंगदैर्घ्य इस बात का हिसाब है कि पहले फलक पर सघन माध्यम
पर परावर्तन चिह्न उलट देता है और दूसरे पर नहीं
(\cref{ch:b2:wave-interfaces})। फ्रिंजें $2ne\cos r =
(p + \tfrac12)\lambda_0$ पर चमकीली हैं, $2ne\cos r = p\lambda_0$ पर अँधेरी --- और
लुप्त होती मोटाई की कोई झिल्ली \emph{कुछ भी} परावर्तित नहीं करती (किसी बहती साबुन
की झिल्ली का काला ऊपरी सिरा)। एकसमान मोटाई की झिल्ली के लिए फ्रिंजें केवल $i$ पर
निर्भर हैं और अनंत पर दिखती हैं (\emph{समान नति} के वलय); और धीरे-धीरे
बदलती मोटाई की किसी पच्चर के लिए, लगभग-अभिलंब आपतन में देखी जाने पर, वे
बराबर $e$ की समोच्च रेखाओं के पीछे चलती हैं और झिल्ली पर स्थानीकृत हैं
(\emph{समान मोटाई} की फ्रिंजें): कोण $\alpha$ की कोई वायु-पच्चर
$\lambda_0/2\alpha$ की दूरी पर सीधी फ्रिंजें दिखाती है; और किसी समतल पर रखा त्रिज्या $R$ का कोई लेंस
\emph{न्यूटन के वलय} देता है, जो $r_p = \sqrt{p\lambda_0R}$ पर अँधेरे हैं।
\end{proposition}

\begin{proof}
ज्यामिति: पहले फलक पर अपवर्तित तरंग झिल्ली को दो बार पार करती है
(काँच में $2e/\cos r$, \omterm{def:b2:scalar-light-model:path}{प्रकाशिक पथ} $2ne/\cos r$) जबकि पहले फलक पर परावर्तित
तरंग हवा में $2e\tan r\sin i$ आगे बढ़ती है; और अंतर
$2ne/\cos r - 2e\tan r\sin i = 2ne\cos r$ है, जहाँ $\sin i = n\sin r$।
पच्चर के लिए, $e = \alpha x$ और क्रमिक अँधेरी फ्रिंजें $2\alpha x = p\lambda$ पर; और लेंस
के लिए, $e \approx r^2/2R$।
\end{proof}

\begin{example}[साबुन, तेल, व्याध-पतंग]\label{ex:b2:two-wave-interference:colours}
अभिलंब आपतन में देखी \qty{300}{nm} मोटी कोई साबुन की झिल्ली ($n = 1.33$):
$2ne = (p + \tfrac12)\lambda$ पर चमकीली, अर्थात् $\lambda = 798/(p + \tfrac12)$ nm: दृश्य के भीतर \qty{532}{nm}
(हरा, $p = 1$), और \qty{400}{nm} पर कोई अँधेरी पट्टी; अतः
झिल्ली हरी दिखती है। जैसे-जैसे वह बहती और पतली होती है, रंग उस
क्रम से गुज़रता है और काले पर ख़त्म होता है। तेल की कोई झिल्ली
(\qty{400}{nm}, $n = 1.45$) जल ($1.33$) पर: दूसरा परावर्तन, जो कम अपवर्तनांक की ओर है,
चिह्न नहीं उलटता, अतः फिर $\delta = 2ne + \lambda/2$ और झिल्ली $2ne =
(p + \tfrac12)\lambda$ पर चमकीली है: \qty{464}{nm} ($p = 2$), नीला। किसी
तितली के पंख, किसी भृंग के कवच या किसी संहत चक्रिका की इंद्रधनुषी झलक इसी तरह के
व्यतिकरण हैं (चक्रिका की, बहुत सारी तरंगों की: \cref{ch:b2:gratings}),
और इसीलिए वे कोण के साथ बदलती हैं --- $\delta$ $\cos r$ पर निर्भर है ---
जबकि किसी वर्णक का रंग नहीं बदलता।
\end{example}

\begin{method}[कोई व्यतिकरण समस्या हल करना]\label{met:b2:two-wave-interference:method}
(1) स्रोत से $M$ तक के दोनों रास्ते पहचानिए और प्रकाशिक
\omterm{thm:b2:two-wave-interference:formula}{पथांतर} $\delta(M)$ निकालिए, और चिह्न उलटने वाले हर परावर्तन के लिए $\lambda/2$
जोड़िए। (2) $I = I_1 + I_2 + 2\sqrt{I_1I_2}\cos(2\pi\delta/\lambda)$;
$\delta = p\lambda$ पर चमकीली। (3) फ्रिंजों की दूरी $\delta(M + \Delta M) - \delta(M) =
\lambda$ से। (4) कालिक संसक्ति ($\delta_{\max} < \ell_c$, दिखती फ्रिंजों की संख्या
$\sim\lambda/\Delta\lambda$) और \omterm{prop:b2:two-wave-interference:spatial}{दैशिक संसक्ति} (स्रोत का कोण
$< \lambda/a$) जाँचिए। (5) बताइए कि फ्रिंजें कहाँ स्थानीकृत हैं (दो बिंदु
स्रोतों के लिए हर जगह; समान-मोटाई फ्रिंजों के लिए झिल्ली पर; और समान-नति
वालों के लिए अनंत पर)।
\end{method}

मिट जाती हैं (माइकेल्सन, 1920, माउंट विल्सन)। (ख) \qty{2.5}{m} का कोई अकेला
दूरदर्शी चकती को क्यों नहीं सुलझा सकता (उसकी सीमा $1.22\lambda/d$ है)?
(ग) आधुनिक व्यतिकरणमापी \qty{100}{m} दूर के दूरदर्शियों को जोड़ते हैं: नापा जा सकने
वाला सबसे छोटा व्यास। (घ) दोनों पथ \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} के भीतर बराबर क्यों होने
चाहिए, और \qty{10}{nm} के किसी छन्ने के लिए वह कितनी लंबी है?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:two-wave-interference:12}
\emph{समान-नति के वलय।} मोटाई $e =
\qty{2.0}{mm}$ और अपवर्तनांक $1.5$ की कोई समांतर पट्टिका किसी चौड़े सोडियम स्रोत से जगमगाई है; और
परावर्तित प्रकाश $f = \qty{50}{cm}$ के किसी लेंस से फ़ोकस किया जाता है। (क) दिखाइए कि
वलय उन कोणों $i_p$ पर हैं जिनके लिए $2ne\cos r_p = (p + \tfrac12)\lambda$ (चमकीले),
और यह कि केंद्र के पास $\cos r \approx 1 - i^2/2n^2$। (ख) केंद्र पर कोटि;
क्या केंद्र चमकीला है? (ग) फ़ोकस तल में पहले तीन चमकीले
वलयों की त्रिज्याएँ। (घ) ये वलय किसी \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} के साथ क्यों दिखते हैं जबकि
यंग की फ्रिंजों को कोई सँकरा चाहिए? (अनंत पर स्थानीकरण।)
\end{exercise}

\section{समस्या: फ्रेनेल के दर्पण और तेल की कोई झिल्ली}

\begin{problem}[{सप्ताहांत समस्या --- 1816 का कोई व्यतिकरणमापी और किसी
पोखर पर कोई इंद्रधनुष: एक ही तरंग को चीरने के दो तरीक़े और वे दोनों
संसक्तियाँ जो दोनों को सीमित करती हैं}]
\label{pb:b2:two-wave-interference:1}

\medskip
\noindent\textbf{भाग I --- फ्रेनेल के दर्पण।}
किसी रेखा के अनुदिश छूते दो समतल दर्पण बहुत छोटा कोण $\alpha =
\qty{5.0e-3}{rad}$ बनाते हैं; उस रेखा के समांतर कोई झिर्री-स्रोत $S$ उससे $d =
\qty{20}{cm}$ पर है, और पर्दा उससे $D = \qty{1.5}{m}$ आगे। $\lambda =
\qty{589}{nm}$ (सोडियम)।
\begin{enumerate}
  \item दिखाइए कि दोनों दर्पण दो आभासी प्रतिबिंब $S_1$, $S_2$ देते हैं,
        $S$ के, जो संपर्क-रेखा पर केंद्रित त्रिज्या $d$ के किसी वृत्त पर हैं
        और $a = 2\alpha d$ से अलग; और मान।
  \item $S_1$ तथा $S_2$ से आती तरंगें व्यतिकरण क्यों करती हैं, जबकि
        दो अलग दीपकों की न करतीं?
  \item पर्दे पर फ्रिंज-चौड़ाई (स्रोतों से पर्दे तक की दूरी
        $\approx d + D$); और उस \qty{5}{mm} चौड़े इलाक़े भर में फ्रिंजों की संख्या
        जहाँ दोनों परावर्तित पुंजें एक-दूसरे पर पड़ती हैं।
  \item उस इलाक़े के किनारे पर \omterm{thm:b2:two-wave-interference:formula}{व्यतिकरण की कोटि}; क्या वह दीपक की
        \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} (\qty{2}{cm}) से बँधी सीमा के नीचे है?
  \item स्रोत-झिर्री \qty{20}{\micro m} चौड़ी है: उसके दोनों किनारों से बनी
        फ्रिंजों का खिसकाव (हर स्रोत-बिंदु का प्रतिबिंब दो
        आभासी बिंदुओं में बनता है; फ्रिंज-व्यवस्था $bD'/d$ खिसकती है, जहाँ $D' =
        d + D$); क्या विभेद बचा रहता है ($i$ से तुलना कीजिए)?
  \item झिर्री चौड़ी करके \qty{0.1}{mm} कर दी जाती है: विभेद,
        $|\operatorname{sinc}(\pi ab/\lambda d)|$ काम में लेकर।
  \item दोनों दर्पण प्रकाश का $90\%$ और $80\%$ परावर्तित करते हैं: फ्रिंजों
        का विभेद।
  \item एक दर्पण के आगे \qty{10}{\micro m} मोटी ($n = 1.5$) कोई काँच की पट्टिका
        रख दी जाती है: आकृति का खिसकाव, फ्रिंजों में; क्या उसका
        सोडियम प्रकाश में पीछा किया जा सकता है? और श्वेत प्रकाश में?
  \item सोडियम दीपक की जगह श्वेत प्रकाश रख दीजिए: पर्दे का वर्णन कीजिए;
        केंद्र के हर ओर कितनी रंगीन फ्रिंजें?
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- तेल की झिल्ली।}
तेल ($n = 1.45$) की कोई झिल्ली किसी गीली सड़क ($n_{\text{जल}} = 1.33$) पर फैलती है,
जो बादलों भरे आकाश से जगमगाई है और लगभग-अभिलंब आपतन में देखी जाती है।
\begin{enumerate}[resume]
  \item दोनों फलकों पर परावर्तन: कौन चिह्न उलटता है? \omterm{thm:b2:two-wave-interference:formula}{पथांतर};
        और किसी चमकीले रंग की शर्त।
  \item उस सबसे पतले इलाक़े की मोटाई जो लाल दिखे ($\lambda =
        \qty{650}{nm}$); हरा (\qty{550}{nm}); बैंगनी (\qty{420}{nm})।
  \item \qty{600}{nm} मोटा कोई इलाक़ा: कौन-से दृश्य तरंगदैर्घ्य
        पुष्ट होते हैं, कौन-से रद्द; और उसका रंग।
  \item झिल्ली रंगों की पट्टियाँ क्यों दिखाती है (सड़क के अनुदिश क्या
        बदलता है), और पट्टियाँ मोटाई की समोच्च रेखाओं के पीछे क्यों चलती हैं?
  \item झिल्ली वाष्पन से पतली होती जाती है: किसी दिए बिंदु पर रंग-क्रम
        के बदलाव का वर्णन कीजिए; और यहाँ $e \to 0$ पर क्या दिखता है
        --- चमकीला या अँधेरा? हवा में किसी साबुन की झिल्ली से तुलना कीजिए।
  \item किसी बहती साबुन की झिल्ली का ऊपरी सिरा फटने से ठीक पहले काला पड़
        जाता है: वहाँ मोटाई (कुछ नैनोमीटर) और काला क्यों।
  \item \qty{60}{\degree} आपतन में देखी गई: तेल में अपवर्तन कोण,
        और प्रश्न 11 के चमकीले तरंगदैर्घ्यों का खिसकाव।
  \item लगभग \qty{1}{\micro m} \omterm{prop:b2:scalar-light-model:coherence}{संसक्ति लंबाई} वाले दिन के प्रकाश के लिए, किस
        मोटाई के परे रंग किसी एकसमान धूसर में मिट जाते हैं?
  \item आकाश का प्रकाश अध्रुवित है और बहुत सारी दिशाओं से आता है:
        आकृति वैसे धुल क्यों नहीं जाती जैसे कोई \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} यंग की
        फ्रिंजें धो डालता है? (झिल्ली पर स्थानीकरण।)
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- दो संसक्तियाँ।}
\begin{enumerate}[resume]
  \item कालिक संसक्ति: सोडियम प्रकाश के साथ फ्रेनेल के दर्पणों के लिए
        दिखती अधिकतम कोटि; और किसी लेज़र ($\ell_c = \qty{10}{m}$) के साथ।
  \item \omterm{prop:b2:two-wave-interference:spatial}{दैशिक संसक्ति}: दर्पणों के लिए अधिकतम स्रोत-चौड़ाई
        ($ab/d < \lambda/4$ से) और यंग की झिर्रियों के लिए, $a = \qty{1}{mm}$ के
        साथ, $L = \qty{40}{cm}$ पर।
  \item समझाइए कि पतली झिल्ली कोई \omterm{prop:b2:two-wave-interference:spatial}{विस्तृत स्रोत} क्यों झेल लेती है जबकि
        \omterm{prop:b2:two-wave-interference:young}{यंग की झिर्रियाँ} नहीं: दो अलग स्रोत-बिंदुओं के लिए झिल्ली की दोनों
        किरणें खींचिए और दिखाइए कि वे फिर भी झिल्ली पर ही मिलती हैं।
  \item सोडियम रेखा ख़ुद \qty{0.02}{nm} चौड़ी है: दर्पण अधिक-से-अधिक कितनी
        फ्रिंजें दिखा सकते हैं ($N \approx \lambda/\Delta\lambda$ लीजिए)?
  \item कोई विद्यार्थी सोडियम दीपक की जगह कोई हरी LED ($\Delta\lambda =
        \qty{30}{nm}$) रख देता है: दर्पणों के साथ दिखती फ्रिंजों की संख्या।
  \item किसी लेज़र के साथ फ्रिंजें पूरे अध्यारोपण-इलाक़े भर पूरे विभेद
        के साथ फैलती हैं पर पर्दा "चित्तीदार" है: क्यों?
  \item किसी एक सारणी में, दर्पणों और झिल्ली के लिए: तरंग कैसे चीरी
        जाती है, फ्रिंजें कहाँ स्थानीकृत हैं, उनकी संख्या को क्या सीमित
        करता है, और कोई चौड़ा स्रोत क्या कर देता है।
\end{enumerate}
\end{problem}
