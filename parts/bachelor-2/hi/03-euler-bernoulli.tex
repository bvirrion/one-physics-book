\chapter{आदर्श तरल: ऑयलर और बर्नूली}\label{ch:b2:euler-bernoulli}

काग़ज़ की एक शीट किनारे से पकड़िए और उसके ऊपर से फूँक मारिए: वह उठ जाती
है। स्नानघर में नल खोलिए और परदा भीतर की ओर झुक आता है। कोई पंख तीन
सौ टन सँभाल लेता है, क्योंकि हवा उसकी ऊपरी सतह पर नीचे की सतह से तेज़
बहती है। तीनों एक ही कथन हैं: \emph{तरल जहाँ तेज़
चलता है, वहाँ उसका दाब कम होता है}। यह अध्याय सरलतम प्रतिरूप में ---
\emph{आदर्श} तरल में, जो दाब तो अनुभव करता है पर घर्षण नहीं --- किसी
\omterm{def:b2:fluid-kinematics:particle}{तरल कण} के लिए न्यूटन का दूसरा नियम लिखता है, और उससे \omterm{thm:b2:euler-bernoulli:bernoulli}{बर्नूली की प्रमेय}
निकालता है, जो किसी \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} का ऊर्जा-संतुलन है और जिसके प्रयोग नली के
प्रवाहमापी से लेकर वायुयान के चाल-सूचक तक फैले हैं।

\section{ऑयलर समीकरण}

\begin{proposition}[किसी तरल कण पर दाब-बल]\label{prop:b2:euler-bernoulli:pressure}
\index{दाब-बल (आयतन घनत्व)}
$\dd\tau$ आयतन के किसी कण पर चारों ओर के तरल द्वारा लगाए गए
दाब-बलों का परिणामी
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
होता है: अर्थात् दाब $-\operatorname{\vect{grad}}P$ घनत्व के आयतन-बल की
तरह काम करता है, जो ऊँचे दाब से नीचे दाब की ओर धकेलता है।
\end{proposition}

\begin{proof}
\cref{thm:b2:fluid-kinematics:continuity} वाले संदूक पर,
$x$ और $x + \dd x$ वाले फलक $+P(x)\dd y\dd z$ और $-P(x + \dd x)\dd y\dd z$
पाते हैं, $x$ के अनुदिश: कुल $-\partial_xP\,\dd x\dd y\dd z$; और इसी तरह $y$ तथा $z$ के लिए।
\end{proof}

\begin{definition}[आदर्श तरल]\label{def:b2:euler-bernoulli:perfect}
\index{आदर्श तरल}
\emph{आदर्श तरल} वह है जिसमें पड़ोसी कणों के बीच एकमात्र स्पर्श-बल
दाब है --- हर पृष्ठ के अभिलंब, और कोई स्पर्शीय (श्यान) प्रतिबल नहीं।
यह एक आदर्शीकरण है, जो --- जैसा अगला अध्याय बताता है --- दीवारों से दूर
और ऊँची रेनल्ड्स संख्या पर मान्य है: दीवार से दूर नली में पानी, पतली
सीमांत परत के बाहर पंख के चारों ओर हवा, या समुद्र पर कोई तरंग।
\end{definition}

\begin{theorem}[ऑयलर समीकरण]\label{thm:b2:euler-bernoulli:euler}
\index{ऑयलर समीकरण}
किसी गैलीलीय तंत्र में गुरुत्व के अधीन $\rho$ घनत्व का कोई \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल}
हर बिंदु पर इसका पालन करता है:
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
और दाईं ओर कोई भी अन्य आयतन-बल घनत्व $\vect f_v$ (वैद्युत, या
अगैलीलीय तंत्र में जड़त्वीय) जोड़ दिया जाता है। विराम में यह
वर्ष 1 के खंड के द्रवस्थैतिक नियम $\operatorname{\vect{grad}}P = \rho\vect g$
पर सिमट आता है।
\end{theorem}

\begin{proof}
$\rho\,\dd\tau$ द्रव्यमान के कण के लिए न्यूटन का दूसरा नियम: उसका
त्वरण \omterm{thm:b2:fluid-kinematics:material}{द्रव्य अवकलज} है
(\cref{thm:b2:fluid-kinematics:material}), और बल हैं दाब का
परिणामी तथा भार $\rho\vect g\,\dd\tau$; अब $\dd\tau$ से भाग दीजिए।
\end{proof}

\begin{example}[त्वरित होती टंकी]\label{ex:b2:euler-bernoulli:tank}
$\vect a$ से त्वरित होते ट्रक पर रखी टंकी का पानी ट्रक के तंत्र में
विराम में है; वहाँ जड़त्वीय बल $-\rho\vect a$ के साथ ऑयलर
$\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$ देता है: अर्थात् दाब
``प्रभावी गुरुत्व'' $\vect g - \vect a$ के अनुदिश बढ़ता है, और मुक्त
सतह (जो एक समदाब पृष्ठ है) उसके लंबवत रहकर
$\arctan(a/g)$ कोण से पीछे की ओर झुक जाती है --- $a = \qty{2}{m/s^2}$ के लिए \qty{11}{\degree}।
$\Omega$ से घूमती बाल्टी में अपकेंद्र बल $\rho\Omega^2r\,\vect e_r$
$P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$ देता है और मुक्त सतह
परवलयज $z = \Omega^2r^2/2g$ होती है।
\end{example}

\section{बर्नूली की प्रमेय}

\begin{theorem}[बर्नूली]\label{thm:b2:euler-bernoulli:bernoulli}
\index{बर्नूली की प्रमेय}\index{स्थगन दाब}\index{गतिक दाब}
एकसमान घनत्व के किसी \emph{आदर्श}, \emph{असंपीड्य} तरल के \emph{स्थायी}
प्रवाह में, एकसमान गुरुत्व वाले किसी गैलीलीय तंत्र में, राशि
\[
P + \tfrac12\rho v^2 + \rho gz
\]
\emph{हर \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश} अचर रहती है। और यदि प्रवाह
\emph{अघूर्णी} भी हो, तो वह पूरे तरल में एक ही अचर होती है।
पद $\tfrac12\rho v^2$ \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{गतिक दाब}} है; और $P + \tfrac12\rho v^2$
\emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{स्थगन दाब}} है, अर्थात् वह दाब जिस तक तरल पहुँचता यदि उसे अपनी
\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} पर विराम में ला दिया जाता।
\end{theorem}

\begin{proof}
स्थायी ऑयलर, जिसमें $(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$ और $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$, तथा $\rho$ एकसमान:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
दायाँ पक्ष $\vect v$ के लंबवत है, अतः बर्नूली राशि की प्रवणता का
\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश कोई घटक नहीं है: इसलिए वह राशि
उसके अनुदिश अचर रहती है। और यदि $\vect\omega = \vect 0$ हो तो प्रवणता
हर जगह लुप्त हो जाती है।
\end{proof}

\begin{remark}[बर्नूली क्या कहती है और क्या नहीं]\label{rem:b2:euler-bernoulli:meaning}
$\rho$ से भाग देने पर प्रमेय यह पढ़ी जाती है: प्रति एकांक द्रव्यमान
यांत्रिक ऊर्जा $v^2/2 + gz$, और उसमें $P/\rho$ जोड़कर, किसी कण के पथ पर संरक्षित रहती है।
दाब वाला पद वह कार्य है जो पीछे के तरल का दाब कण \emph{पर} करता है,
घटा वह कार्य जो कण आगे के तरल पर करता है: अर्थात् किसी दाब-क्षेत्र में
बिना घर्षण धकेले गए कण के लिए ऊर्जा-प्रमेय। शर्तें चार हैं: \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल} (कोई
श्यान हानि नहीं), स्थायी, असंपीड्य, और \emph{\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश}। भिन्न
धारा-रेखाओं के दो बिंदुओं की तुलना केवल \omterm{def:b2:fluid-kinematics:vorticity}{अघूर्णी प्रवाह} में
मान्य है। यह ``तेज़ हवा खींच लेती है'' नहीं कहती: यह कहती है कि जो कण
(दाब-गिरावट से) \emph{त्वरित} हो चुका है वह अब कम दाब पर है
--- कारण दाब-क्षेत्र है, चाल परिणाम।
\end{remark}

\begin{remark}[संपीड्य तरल]\label{rem:b2:euler-bernoulli:compressible}
गैस के लिए घनत्व बदलता है, पर यदि प्रवाह स्थायी, आदर्श और
समएन्ट्रॉपी हो, तो वही उपपत्ति \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश $h + v^2/2 + gz$ का
संरक्षण देती है, जहाँ $h$ प्रति एकांक द्रव्यमान एन्थैल्पी है (आदर्श गैस
के लिए $h = c_PT$): यही रूप \cref{ch:b2:open-systems} में टोंटियों और
टरबाइनों के लिए काम आता है। ध्वनि की चाल से काफ़ी नीचे की चालों पर
(हवा में $v \lesssim \qty{100}{m/s}$) घनत्व $5\%$ से भी कम बदलता है
और असंपीड्य प्रमेय सटीक रहती है।
\end{remark}

\section{अनुप्रयोग}

\begin{proposition}[वेंचुरी मापी]\label{prop:b2:euler-bernoulli:venturi}
\index{वेंचुरी प्रभाव}\index{वेंचुरी मापी}
किसी क्षैतिज नली में काट $S_1$ से घटकर $S_2 < S_1$ होने पर
चाल बढ़ती है ($v_2 = v_1S_1/S_2$, द्रव्यमान-संरक्षण से) और दाब
गिरता है:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
दाब-अंतर को दाबमापी से नापने पर प्रवाह दर मिल जाती है: यही
\emph{\omterm{prop:b2:euler-bernoulli:venturi}{वेंचुरी मापी}} है। यही प्रभाव कार्बोरेटर, रंग-फुहार, बुन्सन जलक
की वायु-प्रवेशिका और गुज़रती रेलगाड़ी के ``खिंचाव'' को चलाता है।
\end{proposition}

\begin{proof}
अचर ऊँचाई पर किसी मध्य \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश बर्नूली, और $v_2 =
v_1S_1/S_2$; अब $v_1$ के लिए हल कीजिए।
\end{proof}

\begin{proposition}[पीटो नली]\label{prop:b2:euler-bernoulli:pitot}
\index{पीटो नली}
प्रवाह की ओर मुँह किए, दूर के सिरे पर बंद कोई नली तरल को अपने मुँह पर
विराम में ले आती है (एक \emph{स्थगन बिंदु}); और बग़ल में, प्रवाह के
समांतर, दूसरा छिद्र स्थैतिक दाब $P$ पढ़ता है। अंतर
\omterm{thm:b2:euler-bernoulli:bernoulli}{गतिक दाब} है:
\[
P_{\text{स्थगन}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{स्थगन}} - P)/\rho} .
\]
हर वायुयान अपनी वायुचाल इसी तरह नापता है।
\end{proposition}

\begin{proof}
उस \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश बर्नूली जो स्थगन बिंदु पर समाप्त होती है, जहाँ
$v = 0$; बग़ल के छिद्र प्रवाह को बिगाड़ते नहीं, जो अविक्षुब्ध
$v$ और $P$ पर गुज़रता है।
\end{proof}

\begin{proposition}[टोरिचेली का सूत्र]\label{prop:b2:euler-bernoulli:torricelli}
\index{टोरिचेली का सूत्र}\index{निःस्राव}
कोई बड़ी खुली टंकी मुक्त सतह से $h$ गहराई पर बने छोटे छेद से खाली होती
है; धार इस चाल से निकलती है:
\[
v = \sqrt{2gh} ,
\]
अर्थात् सतह से मुक्त पतन की चाल। प्रवाह दर $D_V \approx
\alpha s\sqrt{2gh}$ है, जहाँ $s$ छेद का क्षेत्रफल है और $\alpha \approx 0.6$
तीखे किनारे वाले छिद्र का \emph{संकुचन गुणांक} (\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखाएँ} छेद के
आगे भी अभिसरित होती रहती हैं)।
\end{proposition}

\begin{proof}
मुक्त सतह से (जो विराम में है, क्योंकि टंकी बड़ी है:
$v_{\text{सतह}} = v\,s/S \ll v$; दाब $P_0$) धार तक (छेद के ठीक बाहर दाब
$P_0$) बर्नूली: $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$। प्रवाह
अर्ध-स्थायी है: $h$ धीरे-धीरे बदलता है।
\end{proof}

\begin{proposition}[पंख पर उत्थापन]\label{prop:b2:euler-bernoulli:lift}
\index{उत्थापन}\index{कुट्टा--जुकोव्स्की प्रमेय}\index{मैग्नस प्रभाव}
पंख के चारों ओर के प्रवाह में हवा ऊपरी सतह पर निचली सतह से तेज़
गुज़रती है; इसलिए दाब ऊपर नीचे की अपेक्षा कम रहता है, और परिणामी
\emph{उत्थापन} $F_L =
\tfrac12\rho v^2SC_L$ है ($S$ पंख का क्षेत्रफल, और $C_L \approx 0.3$--$1.5$
उत्थापन गुणांक, जो परिच्छेदिका और आक्रमण कोण पर निर्भर करता है)।
चाल का यह अंतर परिच्छेदिका के चारों ओर वेग के किसी \emph{परिचालन}
$\Gamma$ के तुल्य है, और प्रति एकांक विस्तार उत्थापन $F_L' =
\rho v\Gamma$ है (कुट्टा--जुकोव्स्की, स्वीकृत)। घूमती गेंद हवा को अपने
साथ खींच लेती है और उसी तरह बग़ल में उठ जाती है: यही कटी हुई टेनिस-गेंद
या मुड़ते फ़्री-किक का \emph{\omterm{prop:b2:euler-bernoulli:lift}{मैग्नस प्रभाव}} है।
\end{proposition}

\begin{proof}
ठीक ऊपर और ठीक नीचे की उन धारा-रेखाओं पर बर्नूली, जो एक ही
प्रवाह-पूर्व दशा से चलती हैं: $P_{\text{नीचे}} - P_{\text{ऊपर}} = \tfrac12\rho
(v_{\text{ऊपर}}^2 - v_{\text{नीचे}}^2)$; अब जीवा पर समाकलन कीजिए। ऊपर का प्रवाह
तेज़ क्यों है --- तीखा पश्च किनारा पिछले स्थगन बिंदु को वहीं बाँध देता
है और परिचालन तय कर देता है (कुट्टा प्रतिबंध)
--- यह अकेले आदर्श-तरल प्रतिरूप से परे है; उसके लिए अगले अध्याय की
सीमांत परत चाहिए।
\end{proof}

\begin{proposition}[एक अस्थायी प्रवाह: यू-नली]\label{prop:b2:euler-bernoulli:utube}
\index{यू-नली का दोलन}\index{अस्थायी बर्नूली}
कुल $L$ लंबाई का कोई द्रव-स्तंभ एकसमान काट की यू-नली भरता है;
साम्य से $z$ विस्थापित होने पर वह इस तरह दोलन करता है:
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
अर्थात् $L/2$ लंबाई के लोलक की तरह। और साधारणतः, विभव $\varphi$ वाले
किसी अस्थायी पर \omterm{def:b2:fluid-kinematics:vorticity}{अघूर्णी प्रवाह} के लिए ऑयलर समाकलित होकर
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$ देता है, जिसमें
वही अचर पूरे तरल में एक-सा रहता है (\omterm{prop:b2:euler-bernoulli:utube}{अस्थायी बर्नूली})।
\end{proposition}

\begin{proof}
चाल $v = \dot z$ नली के अनुदिश एकसमान है (असंपीड्य,
एकसमान काट)। ऑयलर को नली की अक्ष पर प्रक्षेपित कीजिए और स्तंभ के
अनुदिश एक मुक्त सतह से दूसरी तक समाकलित कीजिए: दाब वाले पद
कट जाते हैं (दोनों सिरों पर $P_0$), $\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$,
संवहनी पद $\rho\,\Delta(v^2/2)$ लुप्त हो जाता है (दोनों सिरों पर वही चाल),
और गुरुत्व वाला पद $\rho g\cdot2z$ देता है। अतः $\rho L\ddot z = -2\rho gz$।
सामान्य कथन $\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ से निकलता है, जो $\vect v = \operatorname{\vect{grad}}\varphi$ के लिए सही है।
\end{proof}

\begin{example}[नली चालू करना]\label{ex:b2:euler-bernoulli:start}
$\ell$ लंबाई की कोई क्षैतिज नली अचर शीर्ष $h$ वाले जलाशय से किसी खुले
सिरे तक जाती है; कपाट $t = 0$ पर खोला जाता है। नली के अनुदिश $v$
एकसमान लेकर, जलाशय की सतह से निर्गम तक समाकलित ऑयलर
$\ell\,\dd v/\dd t = gh - \tfrac12v^2$ देता है, जिसका हल
$v = V\tanh(Vt/2\ell)$ है, जहाँ $V = \sqrt{2gh}$: अर्थात् प्रवाह $2\ell/V$
समय-नियतांक के साथ टोरिचेली की चाल तक पहुँचता है --- \qty{200}{m} शीर्ष
के नीचे \qty{400}{m} के जलनाल के लिए \qty{13}{s}।
\end{example}

\begin{method}[बर्नूली का प्रयोग]\label{met:b2:euler-bernoulli:method}
(1) चारों शर्तें जाँचिए; तय कीजिए कि प्रवाह अघूर्णी है या नहीं
(किसी अवरोधक के चारों ओर एकसमान प्रवाह-पूर्व धारा: हाँ; वेग-प्रोफ़ाइल
वाली नली में प्रवाह: नहीं --- एक ही \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} पर टिकिए)। (2) किसी
\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} पर ऐसे दो बिंदु चुनिए जहाँ अधिक से अधिक राशियाँ ज्ञात हों: कोई मुक्त
सतह ($P = P_0$, और काट बड़ी हो तो $v \approx 0$), खुली हवा में कोई
धार ($P = P_0$), कोई स्थगन बिंदु ($v = 0$)। (3) चालों को जोड़ने के लिए
द्रव्यमान-संरक्षण $vS = \text{अचर}$ जोड़िए। (4) अस्थायी
प्रवाहों के लिए इसके बदले रेखा के अनुदिश ऑयलर को समाकलित कीजिए।
\end{method}

ट्रक पर रखी है। (क) \qty{3.0}{m/s^2} से त्वरित होने पर: मुक्त सतह का कोण,
और पिछली तथा अगली दीवार के बीच पानी की ऊँचाई का
अंतर। (ख) तल के दोनों कोनों के बीच दाब-अंतर। (ग)
\qty{6}{m/s^2} से ब्रेक लगाने पर --- क्या पानी तल से \qty{0.8}{m} ऊपर के ढक्कन
तक पहुँच जाता है? (घ) वही ट्रक \qty{40}{m} त्रिज्या के मोड़ पर
\qty{15}{m/s} से: सतह का झुकाव।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
\qty{15}{cm} त्रिज्या की कोई बाल्टी, जिसमें \qty{20}{cm} गहरा पानी है, अपनी
अक्ष के परितः \num{2.0} चक्कर प्रति सेकंड घुमाई जाती है। (क) मुक्त सतह का आकार और
समीकरण। (ख) नेमि और केंद्र के बीच ऊँचाई का अंतर। (ग) आयतन
संरक्षित रहने पर केंद्र और नेमि पर पानी की ऊँचाई। (घ) किस
घूर्णन दर पर तल का केंद्र सूख जाता है? (ङ) वही परवलयज उत्तम
दूरदर्शी-दर्पण क्यों बनाता है (पारे का घूमता तालाब)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}
\emph{साइफन।} कोई नली टंकी से पानी को मुक्त सतह से \qty{2.0}{m} ऊपर के
शिखर के ऊपर से ले जाकर उससे \qty{3.0}{m} नीचे के निर्गम तक पहुँचाती है। (क) निर्गम चाल
और, \qty{2}{cm} की नली के लिए, प्रवाह दर। (ख) शिखर पर दाब। (ग)
साइफन के चलने के लिए शिखर की अधिकतम ऊँचाई (पानी का वाष्प
दाब, \qty{2.3}{kPa}, छूना नहीं चाहिए)। (घ) नली में हवा भरी हो तो साइफन
क्यों नहीं चलता?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}
\qty{1000}{kg} का कोई हलका वायुयान समुद्र-तल पर \qty{15}{m^2} के पंख पर
\qty{60}{m/s} से उड़ता है। (क) पंख के दोनों फलकों के बीच
माध्य दाब-अंतर; उत्थापन गुणांक। (ख) यदि पंख के नीचे चाल
\qty{58}{m/s} हो, तो पंख के ऊपर वह कितनी है (वही प्रवाह-पूर्व
दशा लीजिए)? (ग) \qty{10}{m} के विस्तार के लिए कुट्टा--जुकोव्स्की सूत्र से
परिच्छेदिका के चारों ओर परिचालन। (घ) यदि
$C_L$ $1.4$ से अधिक न हो सके तो स्टॉल चाल।
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{धमनी-संकीर्णन।} रक्त ($\rho = \qty{1060}{kg/m^3}$) \qty{0.50}{cm^2} काट की
धमनी में वायुमंडलीय दाब से \qty{13}{kPa} ऊपर, \qty{0.50}{m/s} से
बहता है। कोई पट्टिका उसे \qty{0.10}{cm^2} तक सँकरा कर देती है। (क) संकीर्णन में
चाल और दाब। (ख) धमनी के बाहर के ऊतक लगभग
\qty{1}{kPa} पर हैं: क्या धमनी बैठ सकती है? (ग) यदि वह और सँकरी हो तो चाल
और बढ़ती है: इस भगदड़ को समझाइए और बताइए कि ऐसा संकीर्णन फड़फड़ा क्यों सकता है।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{स्थगन ताप।} स्थायी समएन्ट्रॉपी प्रवाह में किसी आदर्श गैस के लिए
\omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के अनुदिश $c_PT + v^2/2$ संरक्षित रहता है। (क) दिखाइए कि
वायुयान की नोक पर विराम में लाई गई हवा $T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$ तक पहुँचती है, जहाँ $M = v/c$ मैक संख्या है और $c =
\sqrt{\gamma RT/M_{\text{mol}}}$ ध्वनि की चाल। (ख) \qty{220}{K} की हवा में
$M = 0.85$ पर उड़ते यात्री-विमान की नोक का ताप; $M = 2$ पर उड़ते किसी
पराध्वनिक वायुयान का; और $M = 25$ पर किसी पुनःप्रवेश कैप्सूल का (क्या सूत्र
तब भी अर्थपूर्ण है?)। (ग) दिखाइए कि असंपीड्य पीटो सूत्र
$M = 0.85$ पर चाल को अधिक आँकता है, और लगभग कितना (यहाँ
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ को $M^2$ में दूसरी कोटि तक फैलाइए)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{चालू करने का क्षणिक दौर।} $\ell = \qty{50}{m}$ लंबाई और एकसमान
काट की कोई नली किसी बड़े जलाशय (निर्गम से $h = \qty{5.0}{m}$ ऊपर शीर्ष)
से खुले सिरे पर लगे कपाट तक जाती है। कपाट $t = 0$ पर खोला जाता है। (क)
$\ell\,\dd v/\dd t = gh - v^2/2$ पाने के लिए नली के अनुदिश ऑयलर समाकलित कीजिए।
(ख) उसे हल कीजिए: $v = V\tanh(Vt/2\ell)$, जहाँ $V = \sqrt{2gh}$। (ग) अंतिम
चाल का $90\%$ पाने में लगा समय। (घ) अब कपाट को पूरे प्रवाह से
\qty{0.5}{s} में एकाएक बंद किया जाता है: $\rho\ell\,\dd v/\dd t$ से माध्य मंदन
और कपाट पर अधिदाब आँकिए --- और बताइए कि वास्तविक
बंदी इससे कहीं अधिक क्यों देती है (अगले अध्याय)।
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{किसी भँवर की मुक्त सतह।} पानी \cref{ch:b2:fluid-kinematics} के
\omterm{ex:b2:fluid-kinematics:planeflows}{बिंदु भँवर} की तरह घूमता है, $v_\theta = \Gamma/2\pi r$, अघूर्णी,
और दूर मुक्त सतह $z = 0$ पर है। (क) बर्नूली किन्हीं दो बिंदुओं के बीच
क्यों लगाई जा सकती है? (ख) मुक्त सतह का आकार $z(r)$। (ग)
किसी रैंकिन क्रोड के भीतर ($r < a$, $v_\theta = \Omega r$, घूर्णी) इसके बदले
घूमते तंत्र में ऑयलर लीजिए या सीधे: दिखाइए कि $z = z(a) +
\Omega^2(r^2 - a^2)/2g$। (घ) $\Gamma = \qty{0.03}{m^2/s}$ और $a = \qty{5}{mm}$ वाले
टब-भँवर के लिए कीप की कुल गहराई; और किसी बवंडर के लिए
($\Gamma = \qty{3e4}{m^2/s}$, $a = \qty{60}{m}$, हवा में: उत्तर को इसके बदले
केंद्र पर दाब-गिरावट के रूप में लिखिए)।
\end{exercise}

\section{समस्या: बाँध, जलनाल और टरबाइन}

\begin{problem}[{सप्ताहांत समस्या --- जलाशय से धार तक एक जलविद्युत
संयंत्र: दाब, चालें, शक्तियाँ, हानियाँ और
जल-आघात}]
\label{pb:b2:euler-bernoulli:1}

किसी बाँध के पीछे जलाशय में \qty{150}{m} गहरा पानी है; उसकी मुक्त सतह
किसी टरबाइन की टोंटियों से $H = \qty{200}{m}$ ऊपर है। $D = \qty{3.0}{m}$
व्यास और $\ell = \qty{400}{m}$ लंबाई का एक जलनाल (इस्पात की नली)
अभिकल्पित प्रवाह $D_V = \qty{60}{m^3/s}$ को टरबाइन-गृह तक उतारता है,
जहाँ कोई टोंटी उसे वायुमंडलीय दाब $P_0 =
\qty{1.0}{bar}$ पर मुक्त धार में बदल देती है, और वह धार किसी पेल्टन चक्र
की बाल्टियों पर टकराती है। $\rho =
\qty{1000}{kg/m^3}$, $g = \qty{9.81}{m/s^2}$; और जब तक अन्यथा न कहा जाए,
तरल आदर्श है।

\medskip
\noindent\textbf{भाग I --- विराम में और अभिकल्पित प्रवाह पर।}
\begin{enumerate}
  \item बाँध के तल पर दाब, और बाँध की ऊपर से नीचे तक
        \qty{1}{m} चौड़ी किसी ऊर्ध्वाधर पट्टी पर बल।
  \item टरबाइन के कपाट बंद होने पर, टरबाइन-गृह में जलनाल के
        भीतर दाब।
  \item अभिकल्पित प्रवाह पर जलनाल में चाल; वहाँ \omterm{thm:b2:euler-bernoulli:bernoulli}{गतिक दाब};
        और जलाशय की सतह से बर्नूली लेकर जलनाल के टरबाइन-गृह
        वाले सिरे पर (टोंटी से पहले) दाब।
  \item मुक्त धार की चाल; टोंटी की काट और व्यास।
  \item धार द्वारा ले जाई गई गतिज शक्ति; दिखाइए कि वह $\rho gD_VH$ के
        बराबर है और उसका मान निकालिए।
  \item अभिकल्पित प्रवाह पर जलनाल में समाए पानी का द्रव्यमान और गतिज
        ऊर्जा; नली में किसी कण का पारगमन-काल।
  \item दिखाइए कि अभिकल्पित प्रवाह पर शीर्ष $H$ जलनाल के तल पर दाब
        और पानी की गतिज ऊर्जा के बीच बँट जाता है, और
        टोंटी पहले को दूसरे में बदल देती है: टोंटी के ठीक बाद दाब
        कितना है?
  \item \omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल} में शीर्ष-हानि नहीं होती, वास्तविक में होती है: किसी
        नली में हानि $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ है, जहाँ चिकने
        इस्पात के लिए $\lambda
        \approx 0.015$। पानी के मीटरों में शीर्ष-हानि, और
        शक्ति का खोया हुआ अंश।
  \item कोई अभियंता इस्पात बचाने के लिए \qty{2}{m} का जलनाल सुझाता है।
        प्रश्न 7 फिर से कीजिए; टिप्पणी दीजिए।
\end{enumerate}

\medskip
\noindent\textbf{भाग II --- प्रवाह नापना।}
जलनाल में \qty{2.0}{m} के गले वाला एक वेंचुरी बनाया गया है।
\begin{enumerate}[resume]
  \item अभिकल्पित प्रवाह पर गले में चाल और नली तथा गले के बीच
        दाब-गिरावट।
  \item यह गिरावट पारे के दाबमापी पर पढ़ी जाती है ($\rho_{\text{Hg}} =
        \qty{13600}{kg/m^3}$): ऊँचाई का अंतर।
  \item दिखाइए कि प्रवाह दर दाबमापी के पाठ के वर्गमूल के
        समानुपाती है, और अभिकल्पित प्रवाह के आधे पर पाठ
        दीजिए।
  \item दाब-द्वार दीवार के साथ सटे क्यों होने चाहिए और प्रवाह में
        बाहर निकले हुए क्यों नहीं?
  \item नली का एक भाग, भूमि-रूप के कारण, जलाशय की सतह से \qty{6}{m}
        \emph{ऊपर} के शिखर पर से जाता है। अभिकल्पित प्रवाह पर वहाँ दाब;
        क्या होता है, और अभिकल्पक को क्या करना पड़ेगा?
\end{enumerate}

\medskip
\noindent\textbf{भाग III --- अस्थायी दशाएँ।}
जलाशय का पृष्ठ-क्षेत्रफल $A = \qty{2.0}{km^2}$ है।
\begin{enumerate}[resume]
  \item अभिकल्पित प्रवाह पर जलाशय का तल जिस दर से गिरता है, उसे
        सेंटीमीटर प्रति घंटे में।
  \item टरबाइन बंद कर दी जाती है और टोंटी (काट प्रश्न 4 वाली)
        हवा में खुली छोड़ दी जाती है: जलाशय गुरुत्व से खाली होता है।
        द्रव्यमान-संतुलन लिखिए और तल $h(t)$ निकालिए (अर्ध-स्थायी
        टोरिचेली प्रवाह)।
  \item तल को \qty{10}{m} गिरने में लगा समय; अचर अभिकल्पित प्रवाह पर
        लगते समय से तुलना कीजिए।
  \item संयंत्र चालू करना: जलनाल भरा है, टोंटी $t = 0$ पर खोली जाती
        है। जलनाल के अनुदिश चाल एकसमान लेकर और टोंटी की ज्यामिति
        छोड़कर, नली के अनुदिश ऑयलर समाकलित करके
        $\ell\,\dd v/\dd t = gH - v^2/2$ पाइए, उसे हल कीजिए, और
        समय-नियतांक दीजिए।
  \item इसी चालू होने के दौरान, जिस क्षण $v$ अपने अंतिम मान का आधा
        है, जलनाल के तल पर दाब कितना है?
        स्थायी मान से अंतर के चिह्न को समझाइए।
\end{enumerate}

\medskip
\noindent\textbf{भाग IV --- कपाट बंद करना।}
\begin{enumerate}[resume]
  \item टरबाइन का कपाट $\tau = \qty{10}{s}$ में बंद किया जाता है; मानिए कि
        जलनाल का पानी एकसमान रूप से मंदित होता है। नली के अनुदिश
        ऑयलर लेकर कपाट पर अधिदाब बार में आँकिए।
  \item तेज़ बंदी के लिए पानी असंपीड्य नहीं रहता: कोई दाब-तरंग
        नली में $c = \qty{1400}{m/s}$ से ऊपर की ओर चलती है (इस्पात की नली
        में पानी में ध्वनि की चाल --- \cref{ch:b2:sound-waves}),
        और अधिदाब $\Delta P = \rho c\,\Delta v$ होता है (जुकोव्स्की)।
        अभिकल्पित प्रवाह से एकाएक रुकने पर मान; स्थैतिक दाब से
        तुलना कीजिए। तरंग को जलाशय तक जाकर लौटने में कितना समय
        लगता है, और इससे ``तेज़'' के अर्थ के बारे में क्या
        पता चलता है?
  \item जलनाल की रक्षा के लिए एक \emph{उछाल-टंकी} --- $A_s = \qty{100}{m^2}$
        काट का ऊर्ध्वाधर खुला कूपक --- वहाँ जोड़ी जाती है जहाँ जलाशय से
        आती $L = \qty{2000}{m}$ लंबाई और $A =
        \qty{7.0}{m^2}$ काट की क्षैतिज सुरंग जलनाल से मिलती है। एकाएक
        बंदी के बाद सुरंग का पानी चलता रहता है और कूपक में
        तल $z$ चढ़ जाता है। सुरंग और कूपक के बीच द्रव्यमान-संरक्षण
        लिखिए।
  \item सुरंग के अनुदिश ऑयलर (\omterm{def:b2:euler-bernoulli:perfect}{आदर्श तरल}) समाकलित करके दिखाइए कि
        $z$ $\ddot z + \dfrac{gA}{LA_s}z = 0$ का पालन करता है; दोलन का
        आवर्तकाल।
  \item अभिकल्पित प्रवाह से बंदी के बाद कूपक में तल की अधिकतम चढ़ाई
        (ऊर्जा-तर्क: सुरंग के पानी की गतिज ऊर्जा कूपक में
        स्थितिज ऊर्जा बन जाती है)।
  \item सार दीजिए: इस संयंत्र में मिले पाँच दाब (बाँध पर स्थैतिक,
        टरबाइन पर स्थैतिक, नली में गतिक, जल-आघात,
        उछाल) और वह नियम जो हर एक को देता है।
\end{enumerate}
\end{problem}
