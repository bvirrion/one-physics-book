\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:2}

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:4}

\begin{solution}{exo:b2:boltzmann-factor:5}
(a) \cref{prop:b2:boltzmann-factor:twolevel}. (b) $\eu^{-x} < 1$ voor $T > 0$;
$N_2 > N_1$ zou $T < 0$ vergen --- een \omterm{prop:b2:laser:gain}{bevolkingsinversie}. (c) $0$ en
$Nk_B\ln 2$. (d) $x = 1.93$: $0.13$; bij \qty{77}{K}, $x = 7.5$: $5 \times 10^{-4}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:6}
(a) Afgeleide van $N\varepsilon/(\eu^x + 1)$. (b) $x = 2.40$, $C = 0.44\,Nk_B$.
(c) $\propto x^2\eu^{-x} \to 0$: bevroren; $\propto 1/T^2 \to 0$: beide niveaus vol,
niets meer om op te nemen. (d) $k_BT = 0.42\varepsilon$: \qty{0.5}{mK}; onder een
millikelvin dragen de kernspins het meeste van de warmtecapaciteit en
vertragen zij elke koeling.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:7}

\begin{solution}{exo:b2:boltzmann-factor:8}

\begin{solution}{exo:b2:boltzmann-factor:9}
(a) Integreer $f$ partieel; de hoofdterm is $(2/\sqrt\pi)x\eu^{-x^2}$. (b)
Helium $x = 5.5$: $5 \times 10^{-13}$; stikstof $x = 14.5$: $\eu^{-211}$.
(c) $x =
4.9$: $10^{-10}$ per botsingstijd --- weg; $10^{-17}$ vergt

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:10}
(a) $n = n_0\eu^{m\omega^2r^2/2k_BT}$. (b)
$\eu^{\Delta m\omega^2r^2/2k_BT} = \eu^{0.24} =
1.27$. (c)
$H = k_BT/m'g = \qty{16}{\micro m}$; het tellen van bolletjes op
verscheidene hoogten geeft $k_B$, dus $N_A = R/k_B$. (d) Ionen volgen
$\eu^{\mp e\varphi/k_BT}$ rond een lading; gelineariseerd in de \omterm{prop:b2:maxwell-equations:potentials}{vergelijking
van Poisson}
geeft dat $\varphi'' = \varphi/\lambda_{\text{D}}^2$: de potentiaal wordt voorbij
$\lambda_{\text{D}}$ afgeschermd.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:11}
(a) $Z = 1/(1 - \eu^{-x})$, $\langle E\rangle = h\nu/(\eu^x - 1)$. (b) $k_BT$:
gelijke verdeling; $h\nu\eu^{-h\nu/k_BT}$: bevroren. (c)
$u_\nu = (8\pi\nu^2/c^3)\langle E
\rangle$; Rayleigh--Jeans geeft elke mode
$k_BT$. (d) Een mode kan niet met
minder dan één kwantum worden aangeslagen.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:12}
(a) Alleen $x = \mu B/k_BT$; $\ln 2$ en $0$. (b) $x = 0.67$: $S/Nk_B = 0.51$, dus
$0.18\,k_B$ per spin weggenomen. (c) $B/T$ constant: \qty{10}{mK}. (d) Het eigen
veld van de spins (millitesla) vervangt $B$ op het eind en begrenst de koeling;
het meten van $M = n\mu\tanh(\mu B/k_BT)$ geeft $T$.
\end{solution}

\begin{solution}{pb:b2:boltzmann-factor:1}
\textbf{1.} $\dd P/\dd z = -nmg = -(mg/k_BT)P$; $H = \qty{8.4}{km}$.

\textbf{2.} $\eu^{-mgz/k_BT}$: de potentiële energie van één molecuul, de
temperatuur van de lucht.

\textbf{3.} \qty{0.70}{}, \qty{0.52}{}, \qty{0.35}{bar}; $H\ln 2 = \qty{5.8}{km}$.

\textbf{4.} $P_0/g = \qty{1.03e4}{kg/m^2}$; $\times 4\pi R^2$:
\qty{5.3e18}{kg}; $\rho_0H =
1.22 \times 8400 = \qty{1.03e4}{kg/m^2}$.

\textbf{5.} $5.3 \times 10^{18}/(29 \times 10^{-3}/N_A) = 1.1 \times 10^{44}$.

\textbf{6.} $H_{\text{O}_2} = 7.6$, $H_{\text{N}_2} = \qty{8.7}{km}$: de
verhouding op \qty{50}{km}
zou $\eu^{-50/7.6 + 50/8.7} = 0.44$ van haar waarde aan de grond zijn;
turbulentie mengt
sneller dan diffusie scheidt.

\textbf{7.} Lager: koudere lucht is dichter en de druk daalt sneller.

\textbf{8.} \qty{2.5e25}{m^{-3}}; $\eu^{-11.9}$: \qty{1.7e20}{m^{-3}} ---
geen rand, alleen een
afspraak.

\textbf{9.} $11.2\sqrt{6370/6870} = \qty{10.8}{km/s}$.

\textbf{10.} Helium \qty{2040}{}, \qty{2300}{m/s}; stikstof \qty{770}{},
\qty{870}{m/s}.

\textbf{11.} $x = 5.3$: $4 \times 10^{-12}$; stikstof $\eu^{-196}$: geen.

\textbf{12.}
$\tfrac14 \times 10^{12} \times 2300 \times 4 \times 10^{-12} \times \tfrac12 \approx
10^3$
atomen per vierkante meter per seconde; over de aarde en een jaar,
$2 \times 10^{25}$ atomen --- \qty{0.1}{kg}.

\textbf{13.}
$5 \times 10^{-6} \times 1.1 \times 10^{44} = 5.5 \times 10^{38}$ atomen,
\qty{3.7e12}{kg};
tegen \qty{0.1}{kg/jaar} zou de verblijftijd absurd zijn.

\textbf{14.} Een stationaire toestand vergt een verlies van
\qty{3e6}{kg/jaar}: een verblijftijd
van ongeveer een miljoen jaar, wat de aanvaarde waarde is --- de
thermische schatting bij \qty{1000}{K} is dus veel te klein: de ontsnapping wordt
beheerst door hetere episoden en door niet-thermische (ionische) processen.

\textbf{15.} $x = 3.75$: $\eu^{-14}$ in plaats van $\eu^{-28}$ --- een
factor $10^6$:
het verlies is buitengewoon gevoelig voor de temperatuur van de exosfeer.

\textbf{16.} De maan is heet en licht: alles ontsnapt in
geologische tijd; Titan is koud, $x = 11$ voor stikstof.

\textbf{17.} $p_\pm = \eu^{\pm x}/2\cosh x$; $M = n\mu\tanh x$.

\textbf{18.} $x = \mu B/k_BT$: \qty{4.2}{A/m}, \qty{310}{A/m}, \qty{1.6e4}{A/m}
(verzadiging \qty{1.9e4}{A/m}); Curie voor $x \ll 1$, dat wil zeggen
$T \gg \qty{70}{mK}$.

\textbf{19.} $M$ hangt alleen van $T$ af zodra $B$ bekend is; het steilst nabij
$x \approx 1$, $T \approx \mu B/k_B = \qty{70}{mK}$; men meet de
susceptibiliteit met een spoel.

\textbf{20.} $\langle E\rangle = -N\mu B\tanh x$; $C = Nk_Bx^2/\cosh^2x$,
piekend bij
$x = 1.2$: \qty{56}{mK}.

\textbf{21.} $Nk_B\ln 2$ en $0$: alle spins richten zich naar het veld.

\textbf{22.} \qty{10}{mK}; het eigen veld van de spins van een millitesla
vervangt $B$
en legt de bodem vast.

\textbf{23.} $x = 1$ bij $\mu_{\text{N}}B/k_B = \qty{0.37}{mK}$: Curie geldt tot een
millikelvin, en de thermometer is bruikbaar tot microkelvin.

\textbf{24.} $\Delta\nu/\nu = \sqrt{8k_BT\ln2/mc^2} \propto \sqrt T$: de breedte
van een lijn meet de gastemperatuur.

\textbf{25.} Dampkring, ontsnappend gas, spins: $\eu^{-E/k_BT}$ met $E = mgz$,
$\tfrac12mv^2$, $\mp\mu B$.
\end{solution}
