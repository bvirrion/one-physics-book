\begin{solution}{exo:b2:gratings:5}
(a) Nullen bij $\varphi = 2\pi m/N$, $m = 1, \dots, N - 1$: $N - 1$ nullen,
$N - 2$ maxima
ertussen. (b) Bij $N\varphi/2 = 3\pi/2$ is
$I \approx I_0/\sin^2(3\pi/2N) \approx I_0(2N/3\pi)^2$:
$4/9\pi^2 = 0.045$ van $N^2I_0$. (c) $N = 3$: één enkel nevenmaximum bij
$\varphi = \pi$ met hoogte $I_0$, een negende van het \omterm{thm:b2:gratings:nwaves}{hoofdmaximum} $9I_0$;
$N = 6$: vier
nevenmaxima van enkele procenten. (d) De middenlob van $\sin^2x/x^2$
bevat $90\%$ van de energie.
\end{solution}

\begin{solution}{exo:b2:gratings:6}

\begin{solution}{exo:b2:gratings:9}

$\lambda^2/2L = \qty{0.036}{nm}$. (b)
$\mathcal F

\end{solution}

\begin{solution}{exo:b2:gratings:10}
(a) $|\sin\theta - \sin\theta_0| \le 2$. (b) De uiterste stralen verschillen in weg
$W(\sin\theta - \sin\theta_0)$, hoogstens $2W$. (c)
$2 \times 0.3/5 \times 10^{-7} = 1.2 \times 10^6$; $c/R
= \qty{250}{m/s}$.
(d) Het midden van een lijn wordt tot op een honderdste van haar
breedte bepaald, en duizend lijnen middelen uit met $\sqrt{1000}$: meters per
seconde, met heldhaftige stabiliteit.
\end{solution}

\begin{solution}{exo:b2:gratings:12}
(a) Buren verschillen in fase $2\pi a\sin\theta/\lambda$ door de weg en $-\psi$
door de voeding; het \omterm{thm:b2:gratings:nwaves}{hoofdmaximum} bij $\varphi = 0$. (b) Eerste nul bij
$\Delta\varphi = 2\pi/N$: $\Delta\theta = \lambda/Na\cos\theta$. (c) Een
tweede \omterm{thm:b2:gratings:nwaves}{hoofdmaximum} bij
$\varphi = \pm2\pi$ vergt $|\sin\theta \pm \lambda/a| \le 1$: alleen als
$a \le \lambda/2$ is dat voor elke
stuurhoek onmogelijk. (d) $Na = \qty{0.96}{m}$: \qty{1.8}{\degree}; $50$ stappen,
\qty{50}{\micro s}.
\end{solution}

\begin{solution}{pb:b2:gratings:1}
\textbf{1.} $a = \qty{0.833}{\micro m}$; $N = 144\,000$.

\textbf{2.} $\sin\theta = \sin20^\circ - \lambda/a$ (de orde ligt aan de
andere kant van
de normaal): \qty{550}{nm} bij \qty{18.5}{\degree}, \qty{400}{nm} bij
\qty{7.9}{\degree}, \qty{700}{nm} bij
\qty{29.9}{\degree}: \qty{22}{\degree} spectrum.

\textbf{3.} $1/a\cos\theta = \qty{1.27e-3}{rad/nm}$; $f_2\times$:
\qty{1.27}{mm/nm}, dat wil zeggen
\qty{0.79}{nm/mm}, \qty{0.012}{nm} per pixel.

\textbf{4.} $R = N = 144\,000$: \qty{0.0038}{nm}, een derde pixel.

\textbf{5.} \qty{400}{nm} van de tweede orde landt waar \qty{800}{nm} van de
eerste orde
zou landen, voorbij het rood: geen overlap in het zichtbare; de overlap zou
bij \qty{350}{nm} beginnen, wat een glasfilter wegneemt.

\textbf{6.} \qty{22}{\degree} $\times$ \qty{1}{m} $= \qty{38}{cm}$: een
detector bestrijkt een strook;
het tralie wordt gedraaid om die te kiezen.

\textbf{7.} $\lambda = 2a\sin17^\circ = \qty{490}{nm}$.

\textbf{8.} $0.8 \times 0.4 = 32\%$.

\textbf{9.} Hoeken \qty{26.5}{\degree}, \qty{14.0}{\degree},
\qty{10.3}{\degree}: $+14$, $-8$ en
\qty{-14}{cm} vanaf de plaats van \qty{550}{nm}.

\textbf{10.} \qty{0.18}{nm} is $15$ pixels: gescheiden.

\textbf{11.} $50$ pixels uit elkaar, elk $4$ pixels breed.

\textbf{12.} \qty{100}{\micro m}, $6.7$ pixels, \qty{0.08}{nm} --- twintig keer de
grens van het tralie: spleetbegrensd.

\textbf{13.} Het beeld is \qty{97}{\micro m} breed: een spleet van
\qty{50}{\micro m} laat
de helft van het licht door.

\textbf{14.} Minder licht bij dezelfde scherpte: de fotonruis groeit.

\textbf{15.} $\Delta\lambda = \lambda v/c = \qty{0.066}{nm}$, $5.5$ pixels.

\textbf{16.} De snelheid van de aarde verschuift elke lijn door het jaar
heen over ten hoogste $\pm5.5$
pixels; zij is uit de efemeriden bekend tot veel
beter dan de $0.3\%$ die voor \qty{100}{m/s} nodig is.

\textbf{17.} \qty{2.2e-5}{nm}, $0.002$ pixel --- ver onder de lijnbreedte. Een
zwaartepunt tot op een honderdste van een lijn van \qty{7}{px} is
\qty{0.07}{px}; duizend
lijnen geven $\sqrt{1000}$: $0.002$ pixel --- het signaal, maar net. Werkelijke
instrumenten voegen oplossend vermogen en stabiliteit toe.

\textbf{18.}
$\Delta\sin\theta = (\lambda/a) \times 10^{-5} = 6.6 \times 10^{-6}$,
$\Delta\theta = \qty{7e-6}{rad}$,
\qty{7}{\micro m}: een halve pixel per kelvin --- millikelvins voor \qty{10}{m/s}.

\textbf{19.} Haar lijnen, op hetzelfde ogenblik op dezelfde pixels
opgenomen, geven de golflengteschaal en volgen elke drift.

\textbf{20.} $a = \qty{3.33}{\micro m}$, $N = 36\,000$, dezelfde hoek
(\qty{18.5}{\degree}), $R = 4N
= 144\,000$, dezelfde dispersie, VSB
$\lambda/4 = \qty{137}{nm}$: oplossend vermogen en

\textbf{21.} VSB $\lambda^2/2L = \qty{0.15}{nm}$ ($12$ pixels);
$\mathcal F = 30$; pieken
\qty{0.005}{nm} breed: een kam van scherpe lijnen over de detector, een
golflengteliniaal.

\textbf{22.} $2 \times 10^{-4}$ tegen \qty{1.3e-3}{rad/nm}: zes keer minder, en het
oplossend vermogen van een prisma ($b\,\dd n/\dd\lambda \sim 10^4$) en zijn
niet-lineaire schaal
verliezen ook; tralies werken bovendien in het ultraviolet.

\textbf{23.} $R \le 2W/\lambda$: $4 \times 10^6$ voor een meter bij
\qty{500}{nm} --- breedte en
hoek zijn de enige wegen omhoog.

\textbf{24.} $10^4 \times 0.012 \times 0.32 = 38$ fotonen per pixel per
seconde; $3800$
in \qty{100}{s}: SRV $\approx 60$.

\textbf{25.} Dispersie, bepaald door $p/a\cos\theta$ en de brandpuntsafstand
van de camera;
oplossend vermogen, door de belichte breedte en de hoeken ($pN$);
\omterm{prop:b2:gratings:fabryperot}{vrij spectraal bereik}, door de orde ($\lambda/p$).
\end{solution}
