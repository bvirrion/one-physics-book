\section*{Hoofdstuk \ref{ch:b2:thermodynamic-potentials} ---
Thermodynamische potentialen}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$, $G = nRT\ln P + g(T)$; $P = -\partial_VF = nRT/V$,
$V = \partial_PG = nRT/P$; $-\partial_TF = nR\ln V - f'(T)$, de entropie
$nC_V\ln T + nR\ln V + \text{const}$ als $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$, $\Delta U = 0$,
$\Delta S = \qty{19.1}{J/K}$.
Omkeerbaar: \qty{5.7}{kJ}, $S_{\text{c}} = 0$; tegen \qty{1}{bar}: \qty{0.9}{kJ},
$S_{\text{c}} = 19.1 - 900/300 = \qty{16}{J/K}$; vacuüm: niets,
$S_{\text{c}} =
\qty{19.1}{J/K}$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$ met $\Delta h \approx L$,
$\Delta s \approx L/T_{\text{smelt}}$.
Bij \qty{-5}{\celsius}: $g_{\text{vl}} - g_{\text{ijs}} = +\qty{6.1}{kJ/kg}$, ijs
stabiel, vloeistof metastabiel (onderkoeling bestaat); bij $+\qty{5}{\celsius}$:
\qty{-6.1}{kJ/kg}, vloeistof stabiel --- oververhit ijs wordt niet waargenomen, zijn
oppervlak smelt eerst.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(a) Materie gaat naar de lagere $\mu$: $\mu_{\text{v}} < \mu_{\text{vl}}$
wanneer $p <
p_{\text{verz}}$; vochtigheid $= p/p_{\text{verz}}$. (b)
$-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
in het voordeel van de damp. (c) Verdamping vergt alleen $p < p_{\text{verz}}$,
geen koken; bij $100\%$ zijn de potentialen gelijk. (d) Op de koudste
oppervlakken, waarvan $p_{\text{verz}}(T_{\text{s}})$ onder de $p$ van de
lucht is gezakt
--- die welke naar de nachthemel stralen.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(a) Standaard. (b)
$b = RT_{\text{c}}/8P_{\text{c}} = \qty{4.3e-5}{m^3/mol}$,
$a =
27R^2T_{\text{c}}^2/64P_{\text{c}} = \qty{0.36}{Pa.m^6/mol^2}$. (c)
$\dd g = v\,\dd P$
bij vaste $T$ en $g$ gelijk aan de twee uiteinden: $\int v\,\dd P = 0$ rond de
lus. (d) Onstabiel waar $\partial_vP > 0$; metastabiel ertussen: de
oververhitte vloeistof (opspatten) en de oververzadigde damp (mist die
op kiemen wacht, nevelkamer).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}
(a) $\dd P/\dd T = L_{\text{m}}P/RT^2$:
$\ln(P/P_0) = -(L_{\text{m}}/R)(1/T - 1/T_0)$.
(b) \qty{28}{mbar} tegen \qty{23}{mbar} --- $L$ is groter bij \qty{20}{\celsius}.
(c) \qty{820}{Pa} tegen \qty{611}{Pa}. (d)
$\partial(\Delta g/T)/\partial T = -L/T^2$
en $\Delta g = 0$ langs de kromme geven dezelfde helling.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(a) Gelijke $\mu$ over het membraan:
$\mu^\circ(P + \Pi) - RTx_{\text{s}} =
\mu^\circ(P)$, dat wil zeggen
$v_{\text{m}}\Pi = RTx_{\text{s}}$, $\Pi = cRT$. (b) \qty{27}{bar}.
(c) $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$ per kubieke meter; werkelijke
installaties
vier keer meer. (d) Water stroomt naar de lagere potentiaal: de
cel met veel opgeloste stof in, en er in pekel weer uit.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(a) $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$. (b) $k_BT/Na^2 \propto T$. (c)
\qty{1.7e-13}{N}; $1.8 \times 10^{14}$. (d) $E \sim nk_BT \approx \qty{0.4}{MPa}$.
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$; $\dd F = -S\dd T + f\dd L$.

\textbf{2.} Schwarz op $\dd F$.

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$: de energie
verandert niet; de trekkracht is entropisch.

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$; $Q = \qty{-1.5}{J}$;
$W = +\qty{1.5}{J}$; som
nul.

\textbf{5.} $+\qty{0.4}{K}$.

\textbf{6.} $L - L_0 \propto 1/T$: \qty{9.1}{cm}, een samentrekking van
\qty{9}{mm}; de
verwarmde spaken worden korter, het zwaartepunt van de velg verschuift, het wiel
draait.

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$; de entropieboete wordt
in $k_BT$ betaald.

\textbf{8.} \qty{1.7e-13}{N}; $1.8 \times 10^{14}$ ketens.

\textbf{9.} Haar kracht is energetisch ($\partial_LU \ne 0$); opwarmen verweekt
de bindingen, zodat $f$ met $T$ daalt.

\textbf{10.} Hellingen $-s$, met
$s_{\text{gas}} > s_{\text{vl}} > s_{\text{ijs}}$; de
laagste lijn is de stabiele fase; kruisingen bij $T_{\text{smelt}}$ en
$T_{\text{damp}}$.

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}.

\textbf{12.} $\qty{-1.3e7}{Pa/K}$; negatief omdat $v_{\text{vl}} < v_{\text{ijs}}$.

\textbf{13.} \qty{70}{bar}: \qty{-0.5}{K} --- zo schaatst men niet bij
\qty{-10}{\celsius}; wrijving en de vloeistofachtige oppervlaktelaag doen het.

\textbf{14.} \qty{270}{bar}, \qty{-2}{K}: met de geothermische flux smelt de
bodem en glijdt de kap.

\textbf{15.} \qty{3.6}{kPa/K}; \qty{92}{\celsius}; ongeveer \qty{127}{\celsius}
(in werkelijkheid \qty{120}{}).

\textbf{16.} \qty{28}{mbar} (\qty{23}{}), \qty{820}{Pa} (\qty{611}{}): $L$
groeit naarmate $T$
daalt; fouten van $20$--$30\%$.

\textbf{17.} Dezelfde $\Delta v$, zodat de verhouding van de hellingen
$L_{\text{sub}}/L_{\text{damp}}
= 1.15$ is: de sublimatiekromme is steiler
--- een knik in het tripelpunt.

\textbf{18.} \qty{12}{kJ/kg}; ijs moet een kiem vormen, en een kleine kiem kost
meer oppervlakte-energie dan zij oplevert: zuivere druppeltjes wachten, tot
\qty{-40}{\celsius}.

\textbf{19.} $\Delta v \to 0$, $L \to 0$, hun verhouding eindig: de kromme houdt op;
daarvoorbij één fluïdum dat continu van vloeistofachtig naar gasachtig overgaat.

\textbf{20.} $\dd g = v\,\dd P$ bij vaste $T$; $g$ gelijk aan de twee uiteinden van
de horizontale: $\int v\,\dd P = 0$, gelijke oppervlakken.

\textbf{21.} De tak van de oververhitte vloeistof: geen belkiem in een
gladde kop; een verstoring brengt haar ineens aan de kook.

\textbf{22.} Invriezen, tot onder \qty{611}{Pa} pompen, zacht verwarmen: de weg
kruist de sublimatiekromme, nooit de smeltkromme; de structuur
blijft behouden.

\textbf{23.} $\mu_{\text{v}}(p) < \mu_{\text{vl}}$ wanneer
$p < p_{\text{verz}}$; gelijk bij
verzadiging.

\textbf{24.} \qty{120}{\celsius} in plaats van \qty{100}{}: vier keer sneller.

\textbf{25.} $S$ voor geïsoleerd, $F$ voor vaste $T,V$, $G$ voor vaste $T,P$;
$-\Delta F$, $-\Delta G$ de terugwinbare arbeid; de betrekkingen van Maxwell
maken van
afgeleiden van $S$ de toestandsvergelijking.
\end{solution}
