\begin{solution}{pb:b2:open-systems:1}
\textbf{1.} Compressor (adiabatisch, $w = \Delta h$), condensor (isobaar, warmte
eruit), \omterm{prop:b2:open-systems:devices}{smoorventiel} (isenthalp), verdamper (isobaar, warmte erin).

\textbf{2.} $h_4 = \qty{256}{kJ/kg}$, $x_4 = 0.28$.

\textbf{3.} \qty{143}{kJ/kg}.

\textbf{4.} \qty{27}{} en \qty{34}{kJ/kg}; $h_2 = 433$;
$T_2 \approx 45 + 7 \approx
\qty{52}{\celsius}$.

\textbf{5.} \qty{177}{kJ/kg}; $\mathrm{COP}_{\text{h}} = 5.2$,
$\mathrm{COP}_{\text{c}} = 4.2$;
$5.2 = 4.2 + 1$.

\textbf{6.} $7.8$ en $6.8$; twee derde.

\textbf{7.} \qty{0.045}{kg/s}; \qty{1.5}{kW}, \qty{1.7}{kW} aan het stopcontact.

\textbf{8.} Compressor: $c_p\ln(325/318) \approx \qty{0.022}{kJ/kg/K}$;
\omterm{prop:b2:open-systems:devices}{smoorventiel}: $s_4 - s_3 = 1.204 - 1.19 = \qty{0.014}{kJ/kg/K}$ --- de compressor.

\textbf{9.} $313/55 = 5.7$; de drukverhouding verdubbelt vrijwel (meer arbeid
per kilogram) en de dampdichtheid halveert (minder massa per slag):

\textbf{10.} De batterij staat onder \qty{0}{\celsius} in vochtige lucht; rijp
isoleert haar; het ontdooien keert de kringloop elk uur enkele minuten om en
kost enkele procenten.

\textbf{11.} $308/35 = 8.8$ tegen $328/55 = 6.0$: hoe lager de
condensortemperatuur, hoe beter --- vloerverwarming.

\textbf{12.} \qty{1}{kWh} brandstof $\to$ \qty{0.4}{kWh} elektriciteit $\to$
\qty{1.6}{kWh} warmte, tegen \qty{0.9}{kWh} van de ketel.

\textbf{13.} Pomp, ketel (isobare verwarming, verdamping, oververhitting),
