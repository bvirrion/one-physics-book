\section*{Hoofdstuk \ref{ch:b2:maxwell-equations} --- De vergelijkingen van Maxwell}

\begin{solution}{exo:b2:maxwell-equations:1}
(a) $\operatorname{div} = 3$, $\operatorname{\vect{curl}} = \vect 0$: de
\omterm{def:b2:maxwell-equations:operators}{gradiënt} van $r^2/2$. (b) $0$ en $(0, 0, 2)$: een rotatie (van
$-\tfrac12(x^2 + y^2)\vect e_z$), geen \omterm{def:b2:maxwell-equations:operators}{gradiënt}. (c) $0$ en $\vect 0$: de
\omterm{def:b2:maxwell-equations:operators}{gradiënt} van $xyz$ (en ook een rotatie). (d)
$\frac1{r^2}\partial_r(r^2/r^2) = 0$, rotatie $\vect 0$: de \omterm{def:b2:maxwell-equations:operators}{gradiënt} van
$-1/r$. (e) $0$ en $(0, -2x, 0)$: een rotatie, geen \omterm{def:b2:maxwell-equations:operators}{gradiënt}.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:2}
(a) Gauss en Faraday: beide leden in \unit{V/m^2} (een \omterm{def:b2:charges-currents-conduction:densities}{ladingsdichtheid}
gedeeld door $\varepsilon_0$, een veldgradiënt, een veld gedeeld door een
tijd); Ampère: beide leden in \unit{T/m}. (b)
$1/\sqrt{8.85 \times 10^{-12} \times 1.257 \times 10^{-6}} = \qty{3.00e8}{m/s}$.
(c) Verhouding $\varepsilon_0\omega/\gamma$: koper $5 \times 10^{-20}$;
zeewater bij \qty{1}{GHz} $1 \times 10^{-2}$ (nog net een geleider); glas
bij \qty{50}{Hz} $3 \times 10^3$: een diëlektricum, waarin de
\omterm{thm:b2:maxwell-equations:maxwell}{verplaatsingsstroom} overheerst.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:3}
$\dd E/\dd t = I/\varepsilon_0\pi R^2 = \qty{1.8e12}{V.m^{-1}.s^{-1}}$;
$j_d = I/\pi R^2 =
\qty{16}{A/m^2}$;
$B(\qty{5}{cm}) = \mu_0Ir/2\pi R^2 = \qty{0.5}{\micro T}$;
$B(\qty{20}{cm}) =
\mu_0I/2\pi r = \qty{0.5}{\micro T}$ --- precies het veld
van de draad.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:4}
(a) $\rho = \varepsilon_0k$. (b)
$\frac1{r^2}\partial_r(\rho_0r^3/3\varepsilon_0) = \rho_0/\varepsilon_0$;
erbuiten geeft Gauss $E = \rho_0R^3/3\varepsilon_0r^2$. (c)
$\operatorname{div}\vect B = 0$: ja;
$\operatorname{\vect{curl}}\vect B = -(B_0/a)\vect e_z$:
$\vect j = -(B_0/\mu_0a)\vect e_z$, een uniforme stroomlaag met dikte langs
$y$.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:5}
(a) $\vect A = \tfrac12(B_yz - B_zy, B_zx - B_xz, B_xy - B_yx)$:
$(\operatorname{\vect{curl}}
\vect A)_x = \tfrac12(B_x + B_x) = B_x$,
enzovoort; $\operatorname{div}\vect A = 0$. (b)
$\vect A' - \vect A = \tfrac12B_0(y, x, 0) = \operatorname{\vect{grad}}(\tfrac12B_0xy)$.
(c) $\frac1r\partial_r(Br^2/2) = B$; $\frac1r\partial_r(BR^2/2) = 0$; beide
zijn gelijk aan $BR/2$ in $R$. (d) $2\pi r\cdot BR^2/2r = \pi R^2B = \Phi$:
een lus buiten de solenoïde, waar $\vect B = \vect 0$, voelt toch
$e = -\dd\Phi/\dd t$ --- via $\vect E =
-\partial_t\vect A$, die daar niet
verdwijnt.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:6}
(a) $E = \sigma/\varepsilon_0 = \qty{1.1e5}{V/m}$, normaal gericht. (b)
$\sigma/\varepsilon_0$ ertussen, nul erbuiten (elke sprong voegt
$\pm\sigma/\varepsilon_0$ toe). (c) $j_s = nI = \qty{1e4}{A/m}$, sprong
$\mu_0j_s = \qty{12.6}{mT} = B$ binnenin. (d) Een lus buiten de torus
omsluit geen netto stroom, dus is $B = 0$ daar; binnenin geeft Ampère
$\mu_0NI/2\pi r$, en de sprong over de wikkeling,
$\mu_0j_s = \mu_0NI/2\pi r$, stemt daarmee overeen.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:7}
(a) $V'' = -\rho_0/\varepsilon_0$:
$V = -\rho_0x^2/2\varepsilon_0 + (U/d + \rho_0d/2\varepsilon_0)x$. (b)
$E = \rho_0x/\varepsilon_0 - U/d - \rho_0d/2\varepsilon_0$, nul in
$x_0 = d/2 + \varepsilon_0U/\rho_0d$ --- nabij het midden bij een grote
ruimtelading. (c)
$\sigma(0) = \varepsilon_0E(0^+) =
-\varepsilon_0U/d - \rho_0d/2$,
$\sigma(d) = -\varepsilon_0E(d^-) = \varepsilon_0U/d - \rho_0d/2$: samen
$-\rho_0d$, wat de ruimtelading neutraliseert. (d) $\rho_0 = 0$: lineaire
$V$, uniforme $E$; $U = 0$: een parabool met top in $d/2$.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:8}
(a) \qty{6000}{km}, \qty{300}{m}, \qty{3}{m}, \qty{12.5}{cm}. (b)
$f < c/10L$: \qty{300}{MHz}; \qty{3}{GHz}; \qty{30}{Hz}. (c)
$(2\pi \times 10^6 \times 0.05/3 \times 10^8)^2/4 =
3 \times 10^{-7}$. (d)
Strikt genomen is het dat niet: bij \qty{50}{Hz} is een net van
\qty{1000}{km} een zesde golflengte, verschillen de fasen erover, en worden
lange lijnen als \omterm{prop:b2:dispersion-wave-packets:coax}{transmissielijnen} behandeld --- elke plaatselijke
schakeling is echter wel quasistationair.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:9}
$\operatorname{div}\operatorname{\vect{curl}}\vect B = 0$ zou
$\operatorname{div}\vect j =
0$ afdwingen, d.w.z.\ $\partial_t\rho = 0$
overal. In de plaat (volume $\pi R^2e$) komt de stroom $I$ binnen en gaat er
niets uit:
$\langle\operatorname{div}\vect j\rangle =
-I/\pi R^2e = \qty{-1.6e5}{A/m^3}$,
$\partial_t\rho = \qty{+1.6e5}{C.m^{-3}.s^{-1}}$.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:10}
(a) $2\pi rE_\theta = -\pi r^2\dot B$: $E_\theta = -r\dot B/2$ binnenin,
$-R^2\dot B/2r$ erbuiten. (b) Haar lijnen zijn gesloten cirkels: zij is geen
\omterm{def:b2:maxwell-equations:operators}{gradiënt}, maar drijft ladingen langs een lus rond --- een elektromotorisch
veld. (c) $e\cdot2\pi r_0
|E_\theta| = e\pi r_0^2\dot B$:
$\pi \times 0.25 \times 10 = \qty{7.9}{eV}$ per omloop; $2.5 \times 10^6$
omlopen. (d) $p = eB(r_0)r_0$ geeft $\dot p = er_0\dot B(r_0)$, terwijl het
geïnduceerde veld $\dot p = e|E_\theta| = e\dot\Phi/2\pi r_0$ geeft:
$B(r_0) = \Phi/2\pi r_0^2 =
\tfrac12\langle B\rangle$.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:11}
(a)
$\operatorname{\vect{curl}}\operatorname{\vect{curl}}\vect E = -\Delta\vect E$
(omdat $\operatorname{div}
\vect E = 0$)
$= -\partial_t\operatorname{\vect{curl}}\vect B = -\mu_0\varepsilon_0\partial_t^2\vect E$.
(b) Hetzelfde met de rollen verwisseld. (c)
$c = 1/\sqrt{\mu_0\varepsilon_0} = \qty{3.0e8}{m/s}$. (d) Weber en
Kohlrausch hadden de verhouding van de elektrostatische en de
elektromagnetische ladingseenheid gemeten, $3.1 \times 10^8$ m/s; Fizeau had
het licht op $3.15 \times 10^8$ m/s gemeten: ``wij kunnen de gevolgtrekking
nauwelijks vermijden dat licht bestaat in de transversale golvingen van
datzelfde medium''.
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:12}
(a)
$\operatorname{\vect{curl}}\operatorname{\vect{curl}}\vect B = -\Delta\vect B = \mu_0\gamma\operatorname{\vect{curl}}
\vect E = -\mu_0\gamma\partial_t\vect B$:
$\Delta\vect B = \mu_0\gamma\partial_t\vect B$, de vorm van de
warmtevergelijking met diffusiecoëfficiënt $1/\mu_0\gamma$. (b)
$\tau \sim \mu_0\gamma d^2$: \qty{7.5}{ms} voor de plaat;
$4\pi \times 10^{-7} \times 5 \times 10^5 \times 9 \times 10^{12} = \qty{6e12}{s}$,
tweehonderdduizend jaar, voor de kern. (c)
$\mu_0\gamma\delta^2 \sim 2/\omega$: $\delta = \sqrt{2/\mu_0
\gamma\omega}$:
\qty{9}{mm} bij \qty{50}{Hz}, \qty{65}{\micro m} bij \qty{1}{MHz}. (d) Het
veld van de kern kan door diffusie niet in minder dan $10^5$ jaar
veranderen: de waargenomen seculaire variatie en de omkeringen vragen om
beweging van het geleidende fluïdum --- een dynamo.
\end{solution}

\begin{solution}{pb:b2:maxwell-equations:1}
\textbf{1.} $Q = (I_0/\omega)\sin\omega t$, $E = Q/\varepsilon_0\pi R^2$:
amplitude $I_0/\omega\varepsilon_0
\pi R^2 = \qty{5.7e5}{V/m}$.

\textbf{2.} $j_d = \varepsilon_0\partial_tE = I(t)/\pi R^2 = 32\cos\omega t$
A/m$^2$; haar flux door de plaat is $I(t)$.

\textbf{3.} $2\pi rB = \mu_0\varepsilon_0\pi r^2\partial_tE$:
$B = \mu_0I(t)r/2\pi R^2$; in $R$: $\mu_0I_0/2\pi R = \qty{2}{\micro T}$.

\textbf{4.} Voor $r > R$ is de hele \omterm{thm:b2:maxwell-equations:maxwell}{verplaatsingsstroom} omsloten:
$B =
\mu_0I/2\pi r$, het veld van de draad; continu in $R$.

\textbf{5.}
$(\operatorname{\vect{curl}}\vect E_1)_\theta = -\partial_rE_1 = -\partial_tB_\theta$,
dus $\partial_rE_1
= \mu_0\varepsilon_0(r/2)\ddot E$ en
$E_1(r) - E_1(0) = \mu_0\varepsilon_0r^2\ddot E/4$; met
$\ddot E = -\omega^2E$:
$|E_1(R) - E_1(0)|/|E| = (\omega R/c)^2/4 = 1 \times 10^{-6}$.

\textbf{6.} Uitstekend. Voor $10\%$ is $(\omega R/c)^2 = 0.4$ nodig,
$f \approx \qty{300}{MHz}$ --- een trilholte, geen condensator meer.

\textbf{7.} $W_e = \tfrac12\varepsilon_0E_0^2\pi R^2d = \qty{4.5e-5}{J}$;
$W_m = \int_0^R(\mu_0I_0r/
2\pi R^2)^2/2\mu_0\cdot2\pi rd\,\dd r = \mu_0I_0^2d/16\pi = \qty{2.5e-11}{J}$;
verhouding $6 \times
10^{-7} \approx (\omega R/c)^2/8$: dezelfde kleine
parameter.

\textbf{8.} $\oint\vect E\cdot\dd\vect l = 2\pi r_0E_\theta = -\dd\Phi/\dd t$.

\textbf{9.} $\dd p/\dd t = e|E_\theta| = (e/2\pi r_0)|\dd\Phi/\dd t|$; per
omloop $2\pi r_0\cdot e|E_\theta| = e|\dd\Phi/\dd t|$.

\textbf{10.} $p = eB(r_0)r_0$ geeft $\dot p = er_0\dot B(r_0)$;
gelijkstellen met 9 geeft
$B(r_0) = \Phi/2\pi r_0^2 = \tfrac12\langle B\rangle$.

\textbf{11.} Een uniform veld heeft $B(r_0) = \langle B\rangle$, twee keer
te veel: de magnetische kracht groeit sneller dan de impuls en de baan
krimpt. De poolschoenen zijn zo gevormd dat het veld sterk is nabij de as en
half zo sterk op de baan.

\textbf{12.} $p = eBr_0 = \qty{3.2e-20}{kg.m/s}$;
$E = pc = \qty{9.6e-12}{J} = \qty{60}{MeV}$;
$\Phi = 2\pi r_0^2B = \qty{0.63}{Wb}$ in \qty{5}{ms}: \qty{126}{eV} per
omloop, $4.8 \times
10^5$ omlopen, \qty{1500}{km} --- bij $c$ in
\qty{5}{ms}: consistent.

\textbf{13.} Uit de voeding van de magneet, aan de elektronen doorgegeven
door de term van Faraday: de veranderende flux maakt het versnellende veld.

\textbf{14.} Tientallen MeV aan elektronen die in een zware trefplaat worden
afgeremd, geven doordringende röntgenstraling; de compacte lineaire
versneller, met hogere dosistempo's, heeft het betatron vervangen.

\textbf{15.} $\operatorname{div}\operatorname{\vect{curl}}\vect A = 0$;
$\operatorname{\vect{curl}}(-\operatorname{\vect{grad}}
V - \partial_t\vect A) = -\partial_t\operatorname{\vect{curl}}\vect A = -\partial_t\vect B$.

\textbf{16.} $\frac1{r^2}(r^2V')' = -\rho_0/\varepsilon_0$:
$V = \rho_0(3a^2 - r^2)/6\varepsilon_0$ binnenin,
$\rho_0a^3/3\varepsilon_0r = Q/4\pi\varepsilon_0r$ erbuiten;
$E = \rho_0r/3\varepsilon_0$ en $Q/4\pi\varepsilon_0r^2$.

\textbf{17.}
$\tfrac12\int\rho V\,\dd\tau = 2\pi\rho_0^2(a^5/6 - a^5/30)/\varepsilon_0 = 4\pi\rho_0^2a^5/
15\varepsilon_0 = 3Q^2/20\pi\varepsilon_0a$;
uranium: \qty{1.6e-10}{J} $\approx \qty{1000}{MeV}$ (bij splijting komen de
\qty{200}{MeV} vrij waarmee twee kleinere bollen goedkoper uitkomen).

\textbf{18.} $V = -E(t)z$ (plus een constante). Met
$\vect A = A_z(r, t)\vect e_z$ geeft
$B_\theta = -\partial_rA_z = \mu_0Ir/2\pi R^2$ dat
$A_z = -\mu_0I(t)r^2/4\pi R^2$.

\textbf{19.}
$-\partial_tA_z = \mu_0\dot Ir^2/4\pi R^2 = \mu_0\varepsilon_0r^2\ddot E/4$
(omdat $\dot I =
\varepsilon_0\pi R^2\ddot E$): de correctie van vraag 5, op
een constante na.

\textbf{20.} $\operatorname{div}\vect A = \partial_zA_z = 0$ omdat $A_z$
alleen van $r$ afhangt; elke $\vect A + \operatorname{\vect{grad}}\chi$.

\textbf{21.} $\sigma = \varepsilon_0E_0 = \qty{5}{\micro C/m^2}$; in het
metaal is $\vect E =
\vect 0$, zodat de sprong $\sigma/\varepsilon_0$
precies het veld in de spleet is.

\textbf{22.} De stroom $I(t)$ voedt de plaat vanaf de as; het deel dat
voorbij $r$ nog moet worden geleverd, is $I(1 - r^2/R^2)$, zodat
$j_s = I(1 -
r^2/R^2)/2\pi r$. Op het buitenvlak geeft Ampère het veld van
de draad, $\mu_0I/
2\pi r$; op het binnenvlak $\mu_0Ir/2\pi R^2$; het
verschil, $\mu_0I(1 -
r^2/R^2)/2\pi r = \mu_0j_s$, is de sprong aan het
grensvlak.

\textbf{23.} Platen: $R/\lambda = 3 \times 10^{-4}$, elektrisch
quasistationair. Lus: $7 \times 10^{-4}$, magnetisch quasistationair. Bij
\qty{1}{GHz}: $R/\lambda = 0.3$ --- een trilholte en een antenne.

\textbf{24.} $\lambda = \qty{3000}{km}$ voor een ring van een meter: de
\omterm{thm:b2:maxwell-equations:maxwell}{verplaatsingsstroom} is volstrekt verwaarloosbaar; het essentiële elektrische
veld is dat van Faraday, dat het magnetische quasistationaire regime
volledig behoudt.

\textbf{25.} I: Ampère--Maxwell (\omterm{thm:b2:maxwell-equations:maxwell}{verplaatsingsstroom}), Faraday voor de
correctie, energiedichtheden. II: Faraday in integraalvorm, lorentzkracht.
III: de potentialen, Poisson (Gauss). IV: de sprongen van $\vect E$ en
$\vect B$ aan het grensvlak; de verhouding van afmeting tot golflengte voor
het regime.
\end{solution}
