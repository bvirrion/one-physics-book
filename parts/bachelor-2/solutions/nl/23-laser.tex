\begin{solution}{exo:b2:laser:5}
(a) Zie \cref{prop:b2:laser:einstein}. (b) \qty{600}{nm}: $h\nu/k_BT
= 80$,
verhouding $5 \times 10^{34}$; \qty{1}{cm}: $4.8 \times 10^{-3}$ ---
\omterm{def:b2:laser:processes}{gestimuleerde emissie} overheerst. (c) Bij microgolffrequenties

\begin{solution}{exo:b2:laser:6}
(a) $I_{\text{s}} = h\nu/\sigma\tau = \qty{1.0e5}{W/m^2}$. (b)
$R =
g_0/\sigma\tau = \qty{3.3e22}{m^{-3}.s^{-1}}$. (c)
$I = I_{\text{s}}(5 - 1) =
\qty{4.2e5}{W/m^2}$;
$P_{\text{uit}} = 0.01 \times 4.2 \times 10^5 \times
10^{-6} = \qty{4.2}{mW}$.
(d)
$I = I_{\text{s}}(\sigma R\tau/g_{\text{th}}
- 1) = (h\nu/g_{\text{th}})(R - g_{\text{th}}/\sigma\tau)$:
lineair in $R$
boven $R_{\text{th}} = g_{\text{th}}/\sigma\tau$.
\end{solution}

\begin{solution}{exo:b2:laser:7}
(a) $c/2L > \qty{1.5}{GHz}$: $L < \qty{10}{cm}$. (b)
$\delta\nu = \nu\,\delta L/L
= \qty{1.6}{GHz}$: drie modeafstanden, de hele
versterkingsbreedte --- de mode
veegt over de versterkingskromme en draagt over aan haar buur. (c) Een

\qty{100}{km}$.
\end{solution}

\begin{solution}{exo:b2:laser:8}

\begin{solution}{exo:b2:laser:11}
(a) Intensiteit $\times 1.04$ om de \qty{2}{ns}. (b)
$N = P\,(2L/c)/h\nu =
6.4 \times 10^8$ fotonen;
$\ln(6.4 \times 10^8)/\ln 1.04 \approx 520$
heen-en-terugreizen, ongeveer \qty{1}{\micro s}. (c) Verzadiging van de
versterking; een mode onder
de drempel verliest per heen-en-terugreis meer dan zij wint en blijft op het
spontane peil. (d) De modes delen één inversie; de sterkste
verzadigt haar en hongert de andere uit; drift van lengte en versterking veranderen
de winnaar --- de uitgang springt van mode naar mode.
\end{solution}

\begin{solution}{pb:b2:laser:1}
\textbf{1.} \qty{1.96}{eV}, \qty{4.74e14}{Hz}.

\textbf{2.} $\qty{0.05}{eV} \approx 1.5\,k_BT$: de warmtebeweging levert
het kleine tekort; de overdracht is vrijwel resonant, dus doeltreffend.

\textbf{3.} $\eu^{-20.66/0.0345} = \eu^{-600} \approx 10^{-260}$: geen
enkel; alleen de
ontlading voedt het niveau.

\textbf{4.} Het onderste niveau loopt tien keer sneller leeg dan het bovenste
vervalt: het is altijd vrijwel leeg --- een \omterm{rem:b2:laser:twolevel}{vierniveauschema}.

\textbf{5.} $v^* = \sqrt{2k_BT/m} = \qty{580}{m/s}$;
$\nu v^*/c \approx \qty{0.9}{GHz}$;
de exacte formule geeft \qty{1.5}{GHz}.

\textbf{6.} $g_0 = \qty{0.09}{m^{-1}}$; $\eu^{0.027}$: $2.7\%$ per doorgang.

\textbf{7.} $g_{\text{th}} = -\ln(0.999 \times 0.99)/0.6 = \qty{0.018}{m^{-1}}$;
marge $5$.

\textbf{8.} \qty{500}{MHz}; drie.

\textbf{9.} $P_{\text{in}} = \qty{1}{mW}/0.01 = \qty{100}{mW}$; \qty{1.3e5}{W/m^2}.

\textbf{10.} De versterking verzadigt tot $g_{\text{th}}$; het overschot wordt
uitgangsfotonen (en \omterm{def:b2:laser:processes}{spontane emissie}, warmte).

\textbf{11.} $\qty{1}{mW}/\qty{7.5}{W} = 1.3 \times 10^{-4}$: het meeste van de
elektronenergie gaat elders heen dan naar het metastabiele helium, en het
foton draagt \qty{1.96}{eV} van de geïnvesteerde \qty{20.66}{eV}.

\textbf{12.} $2 \times 0.072 \times 0.3 = 0.043$;
$N = 0.1 \times 2 \times 10^{-9}/3.14
\times 10^{-19} = 6.4 \times 10^8$;
$20.3/0.043 \approx 470$ heen-en-terugreizen;
\qty{0.9}{\micro s}.

\textbf{13.} $\delta\nu = \nu\,\delta L/L = \qty{1.6}{GHz}$: de mode doorkruist de
hele versterkingskromme --- de \omterm{def:b2:laser:cavity}{laser} springt naar een andere mode tenzij de lengte
wordt gestabiliseerd.

\textbf{14.} De $p$-polarisatie doorloopt een brewstervlak zonder
weerkaatsing; de $s$-polarisatie verliest zo'n $15\%$ per vlak en
haalt de drempel nooit: lineair gepolariseerde uitgang zonder kosten.

\textbf{15.} $z_{\text{R}} = \qty{0.45}{m}$; $\theta = \qty{6.7e-4}{rad}$;
$w(\qty{10}{m})
= \qty{6.7}{mm}$: \qty{13}{mm} in doorsnede.

\textbf{16.} $2P/\pi w_0^2 = \qty{7}{kW/m^2}$: zeven zonnen.

\textbf{17.} $w_0' = \qty{34}{\micro m}$; \qty{5.6e5}{W/m^2}.

\textbf{18.} $w_0' = \qty{11}{\micro m}$; \qty{5e6}{W/m^2}; het netvlies overleeft
\qty{1}{mW} gedurende de \qty{0.25}{s} van de afweerreflex --- klasse 2: zichtbaar,
$\le \qty{1}{mW}$, veilig omdat men knippert.

\textbf{19.} \qty{6.7e-5}{rad}; $w_0 = \qty{3}{mm}$, $z_{\text{R}} = \qty{45}{m}$,
$w(\qty{1}{km}) = \qty{67}{mm}$: \qty{13}{cm}.

\textbf{20.} $c/\Delta\nu$: \qty{300}{m}; \qty{30}{cm}.

\textbf{21.} $2.7 \times 10^{17}$; \qty{1}{GW}.

\textbf{22.} \qty{1.8}{km}; alleen \omterm{def:b2:diffraction:huygens}{buiging}: \qty{130}{m}.

\textbf{23.} $0.1/\pi(1800)^2 = 10^{-8}$; $2.6 \times 10^9$ fotonen.

\textbf{24.} $\lambda/d = \qty{1.3e-5}{rad}$, \qty{5}{km}; $(0.5/5000)^2 = 10^{-8}$:
$25$ fotonen, twee of drie opgespoord.

\textbf{25.} \qty{1.5}{cm}; \qty{0.15}{mm}; de verwijdering wordt binnen
weken gemeten.
\end{solution}
