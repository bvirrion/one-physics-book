\section*{Chapitre \ref{ch:b2:thermodynamic-potentials} --- Potentiels thermodynamiques}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$, $G = nRT\ln P + g(T)$~; $P = -\partial_VF = nRT/V$,
$V = \partial_PG = nRT/P$~; $-\partial_TF = nR\ln V - f'(T)$, soit l'entropie
$nC_V\ln T + nR\ln V + \text{const}$ si $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$, $\Delta U = 0$, $\Delta S = \qty{19.1}{J/K}$.
Réversible~: \qty{5.7}{kJ}, $S_{\text{c}} = 0$~; contre \qty{1}{bar}~: \qty{0.9}{kJ},
$S_{\text{c}} = 19.1 - 900/300 = \qty{16}{J/K}$~; dans le vide~: rien, $S_{\text{c}} =
\qty{19.1}{J/K}$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$ avec $\Delta h \approx L$, $\Delta s \approx L/T_{\text{fus}}$.
À \qty{-5}{\celsius}~: $g_{\text{liq}} - g_{\text{glace}} = +\qty{6.1}{kJ/kg}$, la glace est
stable, le liquide métastable (la surfusion existe)~; à $+\qty{5}{\celsius}$~:
\qty{-6.1}{kJ/kg}, le liquide est stable --- on n'observe pas de glace surchauffée, sa
surface fond d'abord.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:4}
Fusion $\qty{-1.3e7}{Pa/K}$~; vaporisation \qty{3.6}{kPa/K}. \qty{70}{bar}~:
\qty{-0.5}{K}~; \qty{0.7}{bar}~: \qty{92}{\celsius}~; \qty{2}{bar}~: $\approx
\qty{127}{\celsius}$ par la pente locale (\qty{120}{\celsius} en fait~: la
pente croît).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:7}
(a) $-C(L - L_0)$~; $0$. (b) $\Delta S = -C(L - L_0)^2/2 = \qty{-5e-3}{J/K}$~; $Q =
\qty{-1.5}{J}$~; $W = +\qty{1.5}{J}$. (c) $+\qty{0.4}{K}$. (d) $(L - L_0) \propto

\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(a) La matière va vers le $\mu$ le plus bas~: $\mu_{\text{v}} < \mu_{\text{liq}}$ quand $p <
p_{\text{sat}}$~; humidité $= p/p_{\text{sat}}$. (b) $-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
en faveur de la vapeur. (c) L'évaporation ne demande que $p < p_{\text{sat}}$,
non l'ébullition~; à $100\%$ les potentiels sont égaux. (d) Sur les surfaces les plus
froides, dont le $p_{\text{sat}}(T_{\text{s}})$ est tombé sous le $p$ de l'air
--- celles qui rayonnent vers le ciel nocturne.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(a) Classique. (b) $b = RT_{\text{c}}/8P_{\text{c}} = \qty{4.3e-5}{m^3/mol}$, $a =
27R^2T_{\text{c}}^2/64P_{\text{c}} = \qty{0.36}{Pa.m^6/mol^2}$. (c) $\dd g = v\,\dd P$
à $T$ fixée et $g$ égale aux deux bouts~: $\int v\,\dd P = 0$ autour de la
boucle. (d) Instable là où $\partial_vP > 0$~; métastable entre~: le
liquide surchauffé (qui sursaute) et la vapeur sursaturée (brouillard qui
attend des germes, chambre à brouillard).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}
(a) $\dd P/\dd T = L_{\text{m}}P/RT^2$~: $\ln(P/P_0) = -(L_{\text{m}}/R)(1/T - 1/T_0)$.
(b) \qty{28}{mbar} contre \qty{23}{mbar} --- $L$ est plus grande à \qty{20}{\celsius}.
(c) \qty{820}{Pa} contre \qty{611}{Pa}. (d) $\partial(\Delta g/T)/\partial T = -L/T^2$
et $\Delta g = 0$ le long de la courbe donnent la même pente.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(a) Égalité des $\mu$ de part et d'autre de la membrane~: $\mu^\circ(P + \Pi) - RTx_{\text{s}} =
\mu^\circ(P)$, c'est-à-dire $v_{\text{m}}\Pi = RTx_{\text{s}}$, $\Pi = cRT$. (b) \qty{27}{bar}.
(c) $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$ par mètre cube~; les usines réelles
quatre fois plus. (d) L'eau va vers le potentiel le plus bas~: dans la
cellule riche en soluté, hors d'elle dans la saumure.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(a) $S = S_0 - k_BL^2/2Na^2$~; $f = k_BTL/Na^2$. (b) $k_BT/Na^2 \propto T$. (c)
\qty{1.7e-13}{N}~; $1.8 \times 10^{14}$. (d) $E \sim nk_BT \approx \qty{0.4}{MPa}$.
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$~; $\dd F = -S\dd T + f\dd L$.

\textbf{2.} Schwarz appliqué à $\dd F$.

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$~: l'énergie
ne change pas~; la traction est entropique.

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$~; $Q = \qty{-1.5}{J}$~; $W = +\qty{1.5}{J}$~; somme
nulle.

\textbf{5.} $+\qty{0.4}{K}$.

\textbf{6.} $L - L_0 \propto 1/T$~: \qty{9.1}{cm}, une contraction de \qty{9}{mm}~; les
rayons chauffés raccourcissent, le centre de masse de la jante se déplace, la roue
tourne.

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$~; $f = k_BTL/Na^2$~; la pénalité d'entropie se
paie en $k_BT$.

\textbf{8.} \qty{1.7e-13}{N}~; $1.8 \times 10^{14}$ chaînes.

\textbf{9.} Sa force est énergétique ($\partial_LU \ne 0$)~; l'échauffement adoucit
les liaisons, donc $f$ baisse avec $T$.

\textbf{10.} Pentes $-s$, avec $s_{\text{gaz}} > s_{\text{liq}} > s_{\text{glace}}$~; la
ligne la plus basse est la phase stable~; croisements en $T_{\text{fus}}$ et $T_{\text{vap}}$.

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}.

\textbf{12.} $\qty{-1.3e7}{Pa/K}$~; négative parce que $v_{\text{liq}} < v_{\text{glace}}$.

\textbf{13.} \qty{70}{bar}~: \qty{-0.5}{K} --- ce n'est pas ainsi qu'on patine à
\qty{-10}{\celsius}~; le frottement et la couche superficielle quasi liquide le font.

\textbf{14.} \qty{270}{bar}, \qty{-2}{K}~: avec le flux géothermique, la
base fond et la calotte glisse.

\textbf{15.} \qty{3.6}{kPa/K}~; \qty{92}{\celsius}~; environ \qty{127}{\celsius}
(\qty{120}{} en fait).

\textbf{16.} \qty{28}{mbar} (\qty{23}{}), \qty{820}{Pa} (\qty{611}{})~: $L$ croît quand $T$
baisse~; erreurs de $20$--$30\%$.

\textbf{17.} Même $\Delta v$, donc le rapport des pentes vaut $L_{\text{sub}}/L_{\text{vap}}
= 1.15$~: la courbe de sublimation est plus raide --- un coude au point triple.

\textbf{18.} \qty{12}{kJ/kg}~; la glace doit germer, et un petit germe coûte
plus d'énergie de surface qu'il n'en gagne~: les gouttelettes pures attendent, jusqu'à
\qty{-40}{\celsius}.

\textbf{19.} $\Delta v \to 0$, $L \to 0$, leur rapport reste fini~: la courbe s'arrête~;
au-delà, un seul fluide passant continûment de l'allure liquide à l'allure gazeuse.

\textbf{20.} $\dd g = v\,\dd P$ à $T$ fixée~; $g$ égale aux deux bouts de
l'horizontale~: $\int v\,\dd P = 0$, aires égales.

\textbf{21.} La branche du liquide surchauffé~: aucun germe de bulle dans une
tasse lisse~; la moindre perturbation le fait bouillir d'un coup.

\textbf{22.} Congeler, pomper sous \qty{611}{Pa}, réchauffer doucement~: le chemin
traverse la courbe de sublimation, jamais celle de fusion~; la structure
est préservée.

\textbf{23.} $\mu_{\text{v}}(p) < \mu_{\text{liq}}$ quand $p < p_{\text{sat}}$~; égalité à
saturation.

\textbf{24.} \qty{120}{\celsius} au lieu de \qty{100}{}~: quatre fois plus vite.

\textbf{25.} $S$ pour un isolé, $F$ à $T,V$ fixés, $G$ à $T,P$ fixés~;
$-\Delta F$, $-\Delta G$ le travail récupérable~; les relations de Maxwell changent les
dérivées de $S$ en équation d'état.
\end{solution}
