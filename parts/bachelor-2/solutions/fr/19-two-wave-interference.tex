\section*{Chapitre \ref{ch:b2:two-wave-interference} --- Interférences à deux ondes}

\begin{solution}{exo:b2:two-wave-interference:1}
$i = \lambda D/a = \qty{2.4}{mm}$~; $x/i = 4.2$~; dixième frange sombre en $9.5i = \qty{22.5}{mm}$~;
$a = \qty{2}{mm}$~: \qty{0.47}{mm}~; dans l'eau $\lambda/n$~: \qty{1.8}{mm}.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:2}
$1.25I_0 \pm I_0$~: $V = 0.8$. $1.01I_0 \pm 0.2I_0$~: $V = 0.2$ --- le terme d'interférence
va comme le rapport des \emph{amplitudes}, $1/10$, non celui des intensités. Filtre~:
$V = 2\sqrt{0.5}/1.5 = 0.94$.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:3}
$2ne = \qty{665}{nm}$~: brillant en $665/(p + \tfrac12)$~: \qty{443}{nm} (bleu-violet)~;
sombre en $665/p$~: \qty{665}{nm} (rouge)~: un film bleu. $e = \qty{1}{\micro m}$~: $2ne = \qty{2660}{nm}$,
brillant en $760$, $591$, $484$, \qty{409}{nm} --- quatre couleurs à la fois, un
blanc rosé délavé.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:4}
$r_p = \sqrt{p\lambda R}$~: $1.09$, $1.53$, \qty{1.88}{mm}~; $p = r^2/\lambda R = 85$ anneaux dans
\qty{1}{cm}~; le centre est sombre~: épaisseur nulle et une inversion de signe. Eau~:
rayons divisés par $\sqrt n$, $98$ anneaux.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:5}
(a) $i = 0.8$ et \qty{1.4}{mm}~; deuxième rouge \qty{2.8}{mm}, troisième violette \qty{2.4}{mm}.
(b) L'ordre $p$ du rouge, en $1.4p$ mm, rencontre l'ordre $p + 1$ du violet en $0.8(p + 1)$~: dès
$p = 2$ les ordres se recouvrent. (c) $\delta = ax/D = \qty{2.5}{\micro m}$~: manquent les $\lambda = \delta/(p +
\tfrac12)$~: $714$, $556$, \qty{455}{nm}. (d) Centre blanc~; puis des franges à
bord intérieur bleu et bord extérieur rouge~; dès le troisième ordre les couleurs
se mêlent en blanc.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:6}
(a) $b < \lambda L/4a = \qty{59}{\micro m}$. (b) $\pi ab/\lambda L = 6.7$~: contraste $|\sin6.7|/6.7
= 0.06$. (c) Une source large d'un centimètre couvre des centaines d'interfranges de
décalage~: pas de franges. (d) $a < \lambda/\theta = \qty{61}{\micro m}$.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:7}
(a) $a = 2h = \qty{0.4}{mm}$~: $i = \lambda D/a = \qty{1.25}{mm}$. (b) En $x = 0$ les chemins
sont égaux et pourtant la frange est sombre~: la réflexion ajoute $\pi$ (inversion
de signe en incidence rasante sur le milieu le plus dense). (c) $\delta = ax/D <
\qty{50}{\micro m}$~: $100$ franges. (d) La mer est le miroir, l'antenne du haut de
la falaise l'écran~: à mesure que la source monte, les franges défilent et
donnent son élévation.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:8}
(a) $2d = 42\lambda$~: $d = \qty{12.4}{\micro m}$. (b) $100/42 = \qty{2.4}{mm}$. (c) $2nd = p\lambda$~:
$56$ franges. (d) Les franges migrent vers l'extrémité écartée et s'espacent quand $d$
diminue~; un quart de frange vaut $\lambda/8 = \qty{74}{nm}$ sur $d$.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:9}
(a) Seule la première (air $\to$ huile, plus dense)~; huile $\to$ eau va vers un indice
plus bas~: $\delta = 2ne + \lambda/2$, brillant pour $2ne = (p + \tfrac12)\lambda$. (b) $2ne =
\qty{1160}{nm}$~: renforcées $773$ (limite du visible), \qty{464}{nm} (bleu)~;
annulées $1160/p$~: \qty{580}{nm} (jaune)~: du bleu. (c) $2ne = \lambda/2$~: $e = \qty{72}{nm}$.
(d) $\delta = 2ne\cos r$ diminue avec l'angle~: les longueurs d'onde renforcées
se déplacent vers le bleu. (e) Les deux réflexions inversent alors le signe, sans
$\lambda/2$ supplémentaire~: brillant pour $2ne = p\lambda$, sombre pour $(p + \tfrac12)\lambda$ --- c'est la
couche antireflet quart d'onde.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:10}
(a) $(n - 1)e = \qty{10}{\micro m}$, $20$ franges, vers la fente recouverte (où
le chemin géométrique doit raccourcir de \qty{10}{\micro m} pour rétablir $\delta = 0$)~:
$x = (n - 1)eD/a = \qty{2}{cm}$. (b) La frange blanche s'y trouve~; la dispersion du verre
($\Delta n \approx 0.01$ sur le visible, soit \qty{0.2}{\micro m} de chemin)
reste sous $\ell_c$~: encore visible, légèrement colorée. (c) Mesurer le
décalage de la frange blanche~: $e = (\text{décalage})\,a/(n - 1)D$. (d) $e(n - 1)\theta^2
(1 + 1/n)/2 = \qty{63}{nm}$ à \qty{5}{\degree}~: un huitième de frange.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:11}
(a) $\theta = \qty{2.3e-7}{rad}$~: $a = 1.22\lambda/\theta = \qty{3.1}{m}$. (b) Sa limite $1.22\lambda/d =
\qty{0.057}{''}$ dépasse le diamètre de l'étoile, et l'atmosphère brouille
jusqu'à $1''$. (c) $1.22\lambda/100 = \qty{7e-9}{rad} = \qty{1.4}{mas}$. (d) Les franges exigent
$\delta < \ell_c = \lambda^2/\Delta\lambda = \qty{32}{\micro m}$.
\end{solution}

\begin{solution}{exo:b2:two-wave-interference:12}
(a) $\delta = 2ne\cos r + \lambda/2$~; $\cos r = \sqrt{1 - \sin^2i/n^2} \approx 1 - i^2/2n^2$.
(b) $2ne/\lambda = 10186.8$~: le centre, avec $p + \tfrac12$ compris entre $10186.5$ et
$10187.5$, n'est ni brillant ni sombre. (c) $i_p^2 = 2n^2[1 - (p + \tfrac12)\lambda/
2ne]$ pour $p = 10186$, $10185$, $10184$~: $i = 0.011$, $0.024$, \qty{0.032}{rad}, $r = fi
= 5.4$, $11.8$, \qty{15.8}{mm}. (d) Le seul angle $i$ fixe l'ordre~: chaque
point source alimente le même anneau sous le même angle, de sorte que les anneaux, vus
à l'infini, survivent à une source large.
\end{solution}

\begin{solution}{pb:b2:two-wave-interference:1}
\textbf{1.} Chaque miroir image $S$ par symétrie à la distance $d$ de
l'arête de contact~; les deux images sont sur le cercle de rayon $d$, à
l'angle $2\alpha$ l'une de l'autre~: $a = 2\alpha d = \qty{2}{mm}$.

\textbf{2.} $S_1$ et $S_2$ sont deux images de la même vibration~: leurs
sauts de phase sont les mêmes.

\textbf{3.} $i = \lambda(d + D)/a = 589 \times 10^{-9} \times 1.7/2 \times 10^{-3} = \qty{0.50}{mm}$~: dix
franges sur \qty{5}{mm}.

\textbf{4.} $\delta = ax/(d + D) = \qty{2.9}{\micro m}$ au bord, $p = 5$~: très en deçà de
$\ell_c/\lambda = 34\,000$.

\textbf{5.} Décalage $bD'/d = 20 \times 10^{-6} \times 1.7/0.2 = \qty{0.17}{mm}$, soit un tiers de
$i$~: contraste $|\operatorname{sinc}(\pi ab/\lambda d)| = \operatorname{sinc}(1.07) = 0.82$.

\textbf{6.} $\pi ab/\lambda d = 5.3$~: $|\sin5.3|/5.3 = 0.16$.

\textbf{7.} Amplitudes $\sqrt{0.9}$, $\sqrt{0.8}$~: $V = 2\sqrt{0.72}/1.7 = 0.998$.

\textbf{8.} $(n - 1)e = \qty{5}{\micro m}$~: $8.5$ franges~; suivies sans peine en
lumière de sodium~; en lumière blanche la frange blanche se déplace de $8.5i = \qty{4.2}{mm}$
et reste visible (la dispersion du verre ne coûte qu'une fraction de
micromètre).

\textbf{9.} Une frange centrale blanche, deux ou trois franges colorées de
chaque côté, puis un blanc uniforme.

\textbf{10.} Air $\to$ huile inverse, huile $\to$ eau (indice plus bas) non~:
$\delta = 2ne + \lambda/2$~; brillant pour $2ne = (p + \tfrac12)\lambda$.

\textbf{11.} $e = \lambda/4n$~: rouge \qty{112}{nm}, vert \qty{95}{nm}, violet \qty{72}{nm}.

\textbf{12.} $2ne = \qty{1740}{nm}$~: renforcées $1740/(p + \tfrac12)$~: $696$ (rouge), $497$
(bleu-vert), \qty{387}{nm}~; annulées $1740/p$~: $580$ (jaune), \qty{435}{nm}~:
un mélange pourpre de rouge et de bleu-vert.

\textbf{13.} L'épaisseur varie le long de la route~; chaque couleur est brillante
là où $2ne$ prend la bonne valeur, si bien que les bandes sont des lignes de niveau
d'épaisseur.

\textbf{14.} Les ordres diminuent~: les couleurs parcourent la séquence
vers le bord mince~; quand $e \to 0$ l'unique inversion de signe laisse
$\delta = \lambda/2$ pour toutes les couleurs~: sombre --- comme pour un film de savon dans l'air.

\textbf{15.} Quelques nanomètres~: $2ne \ll \lambda$ pour toutes les couleurs, les deux
réflexions sont en opposition de phase et s'annulent~: noir.

\textbf{16.} $\sin r = \sin60^\circ/1.45$, $r = \qty{37}{\degree}$, $\cos r = 0.80$~: $2ne\cos r
= \qty{1390}{nm}$, brillant en $928$, $557$, \qty{398}{nm}~: le rouge de la question 12 a
viré au vert-jaune --- vers le bleu.

\textbf{17.} Lorsque $2ne$ dépasse $\ell_c \approx \qty{1}{\micro m}$, c'est-à-dire au-delà de
\qty{0.4}{\micro m} d'huile environ, les ordres se recouvrent et les couleurs virent au gris.

\textbf{18.} Les deux rayons réfléchis par la lame se rejoignent sur la lame
quelle que soit la direction de la source~: les franges y sont localisées et
une source large et multidirectionnelle ne fait qu'ajouter des motifs identiques.

\textbf{19.} $\ell_c/\lambda$~: $34\,000$ avec le sodium, $1.7 \times 10^7$ avec le laser.

\textbf{20.} $b < \lambda d/4a$~: \qty{15}{\micro m} pour les miroirs, \qty{59}{\micro m} pour
les fentes.

\textbf{21.} Tracer deux points sources~: pour chacun, les deux rayons réfléchis
se recombinent au même point de la lame, avec le même $\delta = 2ne\cos r$
sous incidence quasi normale~; le motif de la lame ne bouge pas, celui d'Young si.

\textbf{22.} $\lambda/\Delta\lambda = 589/0.02 \approx 30\,000$.

\textbf{23.} $550/30 \approx 18$ franges.

\textbf{24.} La lumière cohérente diffusée par les grains de l'écran interfère
avec des phases aléatoires à l'œil~: c'est le tavelage.

\textbf{25.} Miroirs~: \omterm{prop:b2:two-wave-interference:young}{division du front d'onde}, franges partout
(non localisées), nombre limité par $\ell_c/\lambda$, une source large les
brouille. Lame~: \omterm{prop:b2:two-wave-interference:film}{division d'amplitude}, franges sur la lame, couleurs
limitées par $\ell_c \sim \qty{1}{\micro m}$, une source large est sans effet.
\end{solution}
