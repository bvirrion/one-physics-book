\section*{Chapitre \ref{ch:b2:euler-bernoulli} --- Fluides parfaits~: Euler et Bernoulli}

\begin{solution}{exo:b2:euler-bernoulli:1}
$v_2 = 1.5 \times 4 = \qty{6.0}{m/s}$, $P_2 = 3.0 \times 10^5 - 500(36 - 2.25) =
\qty{2.83}{bar}$. À \qty{2}{cm}~: $v = \qty{24}{m/s}$, $P = 3.0 \times 10^5 - 500
(576 - 2.25) = \qty{0.13}{bar}$ --- proche de la pression de vapeur. Un résultat
négatif signifie que le débit supposé est impossible~: l'eau cavite
et l'écoulement est étranglé.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:2}
$v = \sqrt{2 \times 15000/0.74} = \qty{201}{m/s}$~; indiquée $\sqrt{2 \times 15000/
1.22} = \qty{157}{m/s}$. La vitesse indiquée mesure la \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique},
qui est ce que subit l'aile~: la vitesse de décrochage est une vitesse \emph{indiquée}
fixe, à toute altitude.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:3}
Profondeur \qty{2.0}{m}~: $v = \sqrt{2g \times 2} = \qty{6.3}{m/s}$~; $D_V = 0.6 \times 10^{-4}
\times 6.3 = \qty{0.38}{L/s}$~; durée de chute $\sqrt{2/g} = \qty{0.45}{s}$, portée
\qty{2.8}{m}. La portée $= 2\sqrt{h(H - h)}$ est symétrique en $h \leftrightarrow H -
h$~: un trou à la profondeur \qty{1.0}{m} (\qty{2}{m} au-dessus du sol) atteint le même
point.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:4}
$\Delta P = \tfrac12 \times 1.2 \times 900 = \qty{540}{Pa}$, pression extérieure plus basse~:
force résultante \qty{54}{kN} \emph{vers le haut}~; le toit pèse \qty{49}{kN}~: il s'envole.
(Les toits réels cèdent d'abord sur les bords, là où l'écoulement décolle.)
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:5}
(a) $\tan\theta = 3/9.81$, $\theta = \qty{17}{\degree}$~; $2 \times 0.306 = \qty{0.61}{m}$
(plus haut à l'arrière). (b) $\Delta P = \rho aL = \qty{6.0}{kPa}$. (c) $\tan\theta =
0.61$, écart \qty{1.22}{m}~: l'avant monte de \qty{0.61}{m} au-dessus de la moyenne
\qty{0.40}{m}~: \qty{1.01}{m} $>$ \qty{0.8}{m}, l'eau touche le couvercle. (d) $a = v^2/R =
\qty{5.6}{m/s^2}$~: $\theta = \qty{30}{\degree}$, le côté extérieur est plus haut.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:6}
(a) Paraboloïde $z = z_0 + \Omega^2r^2/2g$, $\Omega = 4\pi = \qty{12.6}{rad/s}$. (b)
$\Omega^2R^2/2g = 158 \times 0.0225/19.6 = \qty{0.18}{m}$. (c) La hauteur moyenne d'un
paraboloïde sur le disque est à mi-hauteur~: centre $0.20 - 0.09 = \qty{0.11}{m}$, bord
\qty{0.29}{m}. (d) Le centre se découvre quand $\Omega^2R^2/4g = h_0$~: $\Omega = \sqrt{4gh_0}/R =
\qty{18.7}{rad/s} = 3.0$ tours par seconde. (e) Un paraboloïde concentre
les rayons parallèles en un point, à $f = g/2\Omega^2$~; les télescopes à miroir liquide
font tourner du mercure.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:7}
(a) $v = \sqrt{2g \times 3} = \qty{7.7}{m/s}$~; $D_V = \pi \times 10^{-4} \times 7.7 =
\qty{2.4}{L/s}$. (b) $P = P_0 - \rho g(2) - \tfrac12\rho v^2 = 100 - 19.6 - 29.4 =
\qty{51}{kPa}$. (c) $P_0 - \rho gh_c - \tfrac12\rho v^2 \ge \qty{2.3}{kPa}$~: $h_c \le
\qty{7.0}{m}$. (d) L'air est compressible et léger~: aucune colonne liquide
continue ne transmet la pression, l'eau reste de part et d'autre.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:8}

\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:9}
(a) $v_2 = \qty{2.5}{m/s}$~; $\Delta P = \tfrac12 \times 1060 \times (6.25 - 0.25) =
\qty{3.2}{kPa}$, $P_2 = \qty{9.8}{kPa}$. (b) Encore au-dessus de \qty{1}{kPa}~: non. (c) À
\qty{0.05}{cm^2}, $v = \qty{5}{m/s}$ et $\Delta P = \qty{13}{kPa}$~: $P_2 \approx 0$, sous
la pression des tissus --- la paroi s'affaisse, l'écoulement cesse, la
pression remonte, elle se rouvre~: un battement (le souffle qu'entend le médecin).
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:10}
(a) $c_PT_0 = c_PT + v^2/2$ avec $c_P = \gamma R/(\gamma - 1)M_{\text{mol}}$ et
$c^2 = \gamma RT/M_{\text{mol}}$~: $v^2/2c_PT = \tfrac12(\gamma - 1)M^2$. (b) $M =
0.85$~: $T_0 = 220 \times 1.145 = \qty{252}{K}$~; $M = 2$~: $220 \times 1.8 = \qty{396}{K}$~;
$M = 25$~: $220 \times 126 = \qty{28000}{K}$ --- dénué de sens~: l'air se dissocie
et s'ionise, $c_P$ n'est plus constant (les températures d'arrêt réelles valent
quelques milliers de kelvins). (c) $P_0/P = (1 + \tfrac{\gamma - 1}2M^2)^{\gamma/(\gamma
- 1)} \approx 1 + \tfrac{\gamma}2M^2 + \tfrac{\gamma}8M^4$, et $\gamma PM^2 = \rho v^2$~:
$P_0 - P = \tfrac12\rho v^2(1 + M^2/4)$. Lire $v$ sur $\sqrt{2(P_0 - P)/\rho}$
la surestime de $\sqrt{1 + M^2/4} - 1 \approx 9\%$ à $M = 0.85$.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:11}
(a) De la surface ($v \approx 0$, $P_0$, hauteur $h$) à la sortie ($v$, $P_0$,
hauteur $0$), avec $\int\partial_tv\,\dd s = \ell\,\dd v/\dd t$. (b) $\dd v/(V^2 -
v^2) = \dd t/2\ell$, $V^2 = 2gh$~: $\operatorname{artanh}(v/V) = Vt/2\ell$. (c) $V =
\qty{9.9}{m/s}$~; $\tanh x = 0.9$ pour $x = 1.47$~: $t = 2\ell x/V = \qty{15}{s}$. (d)
$\dd v/\dd t \approx \qty{-20}{m/s^2}$~: $\Delta P = \rho\ell \times 20 = \qty{9.9}{bar}$.
L'eau est légèrement compressible~: une fermeture rapide lance une onde
de pression, $\Delta P = \rho c\Delta v \approx \qty{140}{bar}$.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:12}
(a) Stationnaire, parfait, incompressible et irrotationnel~: une seule constante
pour tout l'écoulement. (b) Sur la surface $P = P_0$~: $\tfrac12\rho v^2 + \rho gz =
0$, $z = -\Gamma^2/8\pi^2gr^2$. (c) Euler radial~: $\partial_rP = \rho\Omega^2r$,
hydrostatique verticale~: la surface vérifie $z(r) - z(a) = \Omega^2(r^2 -
a^2)/2g$. (d) Baignoire~: $\Omega = \Gamma/2\pi a^2 = \qty{191}{rad/s}$~; $z(a) = -\Omega^2a^2/
2g = \qty{-4.7}{cm}$, et le cœur ajoute \qty{4.7}{cm}~: \qty{9.3}{cm} de profondeur.
Tornade~: $\Delta P = \tfrac12\rho v_{\max}^2$ à l'extérieur, autant à l'intérieur~:
$\rho v_{\max}^2 = 1.2 \times 6400 = \qty{7.7}{kPa}$ sous la pression ambiante au centre.
\end{solution}

\begin{solution}{pb:b2:euler-bernoulli:1}
\textbf{1.} $\rho g \times 150 = \qty{14.7}{bar}$ au-dessus de l'atmosphère~; $\tfrac12\rho gH_d^2
\times \qty{1}{m} = \qty{1.1e8}{N}$.

\textbf{2.} $\rho gH = \qty{19.6}{bar}$ en relatif, \qty{20.6}{bar} en absolu.

\textbf{3.} $v = 60/7.07 = \qty{8.5}{m/s}$~; $\tfrac12\rho v^2 = \qty{0.36}{bar}$~; $P =
P_0 + \rho gH - \tfrac12\rho v^2 = \qty{20.3}{bar}$ en absolu.

\textbf{4.} $v_j = \sqrt{2gH} = \qty{62.6}{m/s}$~; $s = 60/62.6 = \qty{0.96}{m^2}$, $d =
\qty{1.1}{m}$.

\textbf{5.} $\tfrac12\rho D_Vv_j^2 = \tfrac12\rho D_V(2gH) = \rho gD_VH = \qty{118}{MW}$.

\textbf{6.} $m = \rho S\ell = \qty{2.8e6}{kg}$, $E_k = \tfrac12mv^2 = \qty{1.0e8}{J}$~;
transit $\ell/v = \qty{47}{s}$.

\textbf{7.} $\rho gH = (P - P_0) + \tfrac12\rho v^2$~: \qty{19.3}{bar} de pression
et \qty{0.36}{bar} d'énergie cinétique~; dans l'injecteur la pression tombe
à $P_0$ tandis que $\tfrac12\rho v^2$ monte à $\rho gH$~: juste après, $P = P_0$.

\textbf{8.} $\Delta h = \lambda(\ell/D)v^2/2g = 0.015 \times 133 \times 3.67 = \qty{7.3}{m}$,
soit $3.7\%$ de la puissance.

\textbf{9.} $v = \qty{19.1}{m/s}$, $v^2/2g = \qty{18.6}{m}$, $\Delta h = 0.015 \times 200 \times
18.6 = \qty{56}{m}$~: $28\%$ de perte --- l'acier économisé se paie chaque
seconde en énergie.

\textbf{10.} $v_t = 60/\pi = \qty{19.1}{m/s}$~; $\Delta P = 500(19.1^2 - 8.49^2) =
\qty{146}{kPa}$.

\textbf{11.} $\Delta P = (\rho_{\text{Hg}} - \rho)g\,\Delta h$~: $\Delta h = 146000/(12600
\times 9.81) = \qty{1.18}{m}$.

\textbf{12.} $\Delta P \propto v^2 \propto D_V^2$~: $D_V \propto \sqrt{\Delta h}$~; la moitié du
débit en donne le quart, \qty{0.30}{m}.

\textbf{13.} Une prise en saillie fait face à l'écoulement ou le perturbe et lit une part
de la \omterm{thm:b2:euler-bernoulli:bernoulli}{pression dynamique} (un Pitot), non la pression statique.

\textbf{14.} $P = P_0 - \rho g \times 6 - \tfrac12\rho v^2 = 100 - 59 - 36 = \qty{5}{kPa}$~:
proche de la pression de vapeur~; l'air dissous se dégage et l'eau peut
bouillir (cavitation), rompant la colonne. Abaisser la crête, ou installer un
reniflard et une pompe à vide, ou réduire le débit.

\textbf{15.} $D_V/A = 60/2 \times 10^6 = \qty{3e-5}{m/s} = \qty{11}{cm/h}$.

\textbf{16.} $A\,\dd h/\dd t = -s\sqrt{2gh}$~: $\sqrt h = \sqrt{h_0} - (s/A)\sqrt{g/2}\,t$.

\textbf{17.} $\Delta t = (A/s)\sqrt{2/g}(\sqrt{200} - \sqrt{190}) = 2.09 \times 10^6 \times
0.45 \times 0.36 = \qty{3.4e5}{s} = \qty{3.9}{jours}$~; à débit nominal constant
$10A/D_V = \qty{3.3e5}{s}$~: presque autant, l'injecteur ayant été dimensionné
pour l'écoulement de Torricelli à $h \approx H$.

\textbf{18.} $\ell\,\dd v/\dd t = gH - v^2/2$~: $v = V\tanh(Vt/2\ell)$, $V = \qty{62.6}{m/s}$,
constante de temps $2\ell/V = \qty{13}{s}$.

\textbf{19.} À $v = V/2$~: $\dd v/\dd t = (gH - V^2/8)/\ell = \qty{3.7}{m/s^2}$~;
$P = P_0 + \rho gH - \tfrac12\rho v^2 - \rho\ell\,\dd v/\dd t = 1.0 + 19.6 - 4.9 - 14.7
= \qty{1.0}{bar}$~: l'essentiel de la charge sert à accélérer la colonne, de sorte que
la pression au bas de la conduite est très inférieure à sa valeur en régime établi.

\textbf{20.} $|\dd v/\dd t| = 8.49/10 = \qty{0.85}{m/s^2}$~: $\Delta P = \rho\ell \times 0.85 =
\qty{3.4}{bar}$.

\textbf{21.} $\Delta P = 1000 \times 1400 \times 8.49 = \qty{119}{bar}$, six fois la
pression statique. L'aller-retour de l'onde dure $2\ell/c = \qty{0.57}{s}$~: une
fermeture plus rapide est «~brutale~» et subit toute la surpression de
Joukowsky~; les fermetures plus lentes en subissent moins.

\textbf{22.} $Av = A_s\,\dd z/\dd t$.

\textbf{23.} Euler le long de la galerie, du lac (pression
$P_0 + \rho g\times$profondeur) au pied du puits (pression $P_0 + \rho g
(\text{profondeur} + z)$)~: $\rho L\,\dd v/\dd t = -\rho gz$~; avec $v = (A_s/A)\dot z$~:
$\ddot z + (gA/LA_s)z = 0$~; $T = 2\pi\sqrt{LA_s/gA} = 2\pi\sqrt{2000 \times 100/
68.7} = \qty{340}{s} \approx \qty{5.7}{min}$.

\textbf{24.} $\tfrac12\rho ALv^2 = \tfrac12\rho gA_sz_{\max}^2$ avec $v = 60/7 =
\qty{8.6}{m/s}$~: $z_{\max} = v\sqrt{AL/gA_s} = \qty{32}{m}$.

\textbf{25.} Barrage~: hydrostatique, $\rho gh$. Turbine, vannes fermées~: de
même, $\rho gH$. Conduite en service~: Bernoulli, $\tfrac12\rho v^2$ prélevé
sur la charge statique. Coup de bélier~: compressibilité, $\rho c\Delta v$.
Cheminée d'équilibre~: Euler non stationnaire le long de la galerie, une oscillation de période
$2\pi\sqrt{LA_s/gA}$.
\end{solution}
