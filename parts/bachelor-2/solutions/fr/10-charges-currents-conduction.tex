\begin{solution}{exo:b2:charges-currents-conduction:5}
(a) Du courant entre, rien ne sort~: $\dd Q/\dd t = I$. (b) La charge s'accumule~: $\partial_t
\rho \ne 0$. (c) $\vect j = (I/4\pi r^2)\vect e_r$~: $\operatorname{div}\vect j = \frac1{r^2}
\partial_r(I/4\pi) = 0$. (d) Seulement à l'électrode, là où le courant est
injecté depuis le fil --- une source de flux.
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:6}
(a) $j = I/2\pi rL$, $E = j/\gamma$, radial. (b) $U = \int_a^bE\,\dd r = I\ln(b/a)/
2\pi\gamma L$~: $R = \ln3.5/(2\pi \times 10^{-13} \times 10^3) = \qty{2e9}{\ohm}$ par kilomètre.
(c) \qty{0.5}{\micro A}, \qty{0.5}{mW}. (d) $RC = \varepsilon_0\varepsilon_r/\gamma = \qty{200}{s}$ pour
$\varepsilon_r = 2.3$~: le câble, chargé puis isolé, se décharge à travers son propre
isolant avec le temps de relaxation de cet isolant.
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:7}
(a) $\omega\tau$~: $8 \times 10^{-12}$, $1.6 \times 10^{-3}$, $4.7$, $79$. (b) $|\underline\gamma|/\gamma_0 =
1/\sqrt{1 + \omega^2\tau^2}$~: $1$, $1$, $0.21$, $0.013$~; phase $-\arctan\omega\tau$~: $0$,
$-\qty{0.09}{\degree}$, $-\qty{78}{\degree}$, $-\qty{89}{\degree}$. (c) $\omega\tau \le 0.14$~: $f \le
\qty{0.9}{THz}$. (d) $\underline\gamma \approx ne^2/\iu m\omega$~: $\underline{\vect j}$ est en retard sur
$\underline{\vect E}$ de \qty{90}{\degree}, $\langle\vect j\cdot\vect E\rangle = 0$~: aucun échauffement
Joule --- les électrons oscillent librement, comme dans un plasma.
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:9}

\begin{solution}{exo:b2:charges-currents-conduction:11}
(a) $E_H = jB/nq$ avec $j = I/wt$~: $V_H = E_Hw = IB/nqt$. (b) $1/(8.5 \times 10^{28}
\times 1.6 \times 10^{-19} \times 10^{-4}) = \qty{0.7}{\micro V}$. (c) \qty{6.3}{mV}~: $V_H \propto 1/n$,
et les semi-conducteurs ont $10^6$ fois moins de porteurs. (d) $B = V_Hnqt/I =
10^{-5} \times 10^{22} \times 1.6 \times 10^{-19} \times 2 \times 10^{-5}/10^{-3} = \qty{0.32}{T}$~; le signe
de $V_H$ donne le signe des porteurs (pour un $\vect B$ connu), ou la
direction de $\vect B$ (pour un capteur connu).
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:12}
(a) $\vect j = ne\mu_+\vect E + (-ne)(-\mu_-\vect E)$~: les deux espèces dérivent en
sens opposés et transportent le courant dans le même sens. (b) $n =
0.5 \times 10^3 \times 6.02 \times 10^{23} = \qty{3e26}{m^{-3}}$~: $\gamma = 3 \times 10^{26} \times 1.6 \times 10^{-19}
\times 1.31 \times 10^{-7} = \qty{6.3}{S/m}$ (les ions se gênent à cette
concentration~: \qty{5}{S/m} mesurés). (c) \qty{0.52}{\micro m/s} et \qty{0.79}{\micro m/s}~:

\qty{200}{W}$ dans \qty{10}{g} d'eau~: \qty{10}{K} en \qty{2}{s} --- d'où les plaques de

\end{solution}

\begin{solution}{pb:b2:charges-currents-conduction:1}
\textbf{1.} $n = \rho N_A/M = \qty{8.4e28}{m^{-3}}$.

\textbf{2.} $j = \qty{6.4e6}{A/m^2}$, $v = j/ne = \qty{0.47}{mm/s}$~; \qty{35}{min} par mètre.

\textbf{3.} $\tau = m\gamma/ne^2 = \qty{2.5e-14}{s}$, $\mu = \qty{4.4e-3}{m^2/(V.s)}$, $\ell =
\qty{40}{nm}$, soit $150$ distances~: les électrons se heurtent aux défauts et aux
vibrations thermiques, non aux atomes.

\textbf{4.} $\rho_e \propto T$ donne $\alpha = 1/293 = \qty{3.4e-3}{K^{-1}}$, proche du
$4 \times 10^{-3}$ mesuré.

\textbf{5.} $E = \qty{0.11}{V/m}$, $U = \qty{0.11}{V}$~; $\tfrac12mv^2 = \qty{1e-37}{J}$ contre
$k_BT = \qty{4e-21}{J}$~: le gaz d'électrons ne remarque pas le courant.

\textbf{6.} $I/e = \qty{1e20}{}$ électrons par seconde.

\textbf{7.} $Q = neSL = \qty{3.4e4}{C}$~; $QE = \qty{3.6}{kN}$, contre un poids de
\qty{0.2}{N}. Le champ tire le réseau positif avec la force opposée~;
les électrons cèdent leur quantité de mouvement au réseau par les chocs~:
le fil dans son ensemble ne ressent rien.

\textbf{8.} $p = j^2/\gamma = \qty{6.8e5}{W/m^3}$, soit \qty{1.7}{W/m}.

\textbf{9.} Surface $\pi d = \qty{5.6e-3}{m^2}$ par mètre~: $\Delta T = 1.7/(10 \times 5.6 \times
10^{-3}) = \qty{30}{K}$~; la résistivité croît de $12\%$, la puissance passe à \qty{1.9}{W}~:
$\Delta T \approx \qty{35}{K}$.

\textbf{10.} $\Delta T = 1.7(1 + 0.004\Delta T)/(2 \times 5.6 \times 10^{-3})$~: $\Delta T(1 - 0.6) = 150$,
$\Delta T \approx \qty{400}{K}$ --- proche de l'emballement thermique~: les câbles enterrés doivent être
déclassés.

\textbf{11.} $p/\rho c = \qty{0.2}{K/s}$~; à \qty{160}{A}, \qty{20}{K/s}~: fusion en
une minute environ (moins, car $\rho_e$ croît)~; le disjoncteur s'ouvre en quelques millisecondes.

\textbf{12.} $\omega\tau = 8 \times 10^{-12}$~: la loi d'Ohm est intacte.

\textbf{13.} $\varepsilon_0/\gamma = \qty{1.5e-19}{s}$~: aucune charge à l'intérieur~; le champ dans le
fil est créé par des charges portées par sa \emph{surface}, dont la densité varie
le long du fil.

\textbf{14.} $Q = I_0\tau_c = \qty{16}{mC}$, $V = Q/C = \qty{160}{V}$.

\textbf{15.} Le côté du fil porte $I$, le côté de l'intervalle aucun courant de
conduction~: la charge croît, et avec elle le champ électrique dans l'intervalle.

\textbf{16.} $I = 16\eu^{-0.5} = \qty{9.7}{A}$~; $E = Q/\varepsilon_0A$, $\dd E/\dd t = I/\varepsilon_0A
= 9.7/(8.85 \times 10^{-12} \times 10^{-4}) = \qty{1.1e16}{V.m^{-1}.s^{-1}}$.

\textbf{17.} $\varepsilon_0A\,\dd E/\dd t = I$~: le courant de déplacement.

\textbf{18.} $j = \gamma E = 10^{-14} \times 2.3 \times 10^5 = \qty{2e-9}{A/m^2}$, soit $10^{-14}$ A
à travers la section~: rien~; claquage à \qty{3}{kV} sur le millimètre.

\textbf{19.} $R = 1000/(6 \times 10^7 \times 10^{-5}) = \qty{1.7}{\ohm}$~; $\Delta U = \qty{67}{V}$~; $P =
\qty{2.7}{kW}$, soit $29\%$ des \qty{9.2}{kW} délivrés.

\textbf{20.} $I = \qty{0.46}{A}$, $\Delta U = \qty{0.8}{V}$, \qty{0.35}{W}~: le transport à
haute tension divise la perte par $(U_2/U_1)^2$.

\textbf{21.} $10 \times 6/3.6 = \qty{17}{mm^2}$~; \qty{89}{kg/km} de cuivre contre
\qty{45}{kg/km} d'aluminium~: plus léger et moins cher sur les pylônes~; dans le mur
la section plus faible du cuivre passe dans les gaines et les bornes.

\textbf{22.} $V_H = IB/nqt = 5 \times 10^{-3} \times 0.1/(10^{22} \times 1.6 \times 10^{-19} \times 10^{-4}) =
\qty{3.1}{mV}$.

\textbf{23.} $+24\%$~: \qty{2.1}{\ohm}, \qty{3.3}{kW} perdus.

\textbf{24.} $RI^2t = 1.7 \times 4 \times 10^8 \times 10^{-4} = \qty{67}{kJ}$ dans \qty{89}{kg}~: \qty{2}{K}.
$j = \qty{2e9}{A/m^2}$, trois cents fois la valeur domestique.

\textbf{25.} $n$ (la matière) et $\tau$ (les chocs) donnent $\gamma = ne^2\tau/m$~;
$\gamma$ relie $j$ et $E$ (Ohm)~; $j$ s'intègre en $I$, $E$ en $U$ (le
circuit)~; $j^2/\gamma$ est la chaleur.
\end{solution}
