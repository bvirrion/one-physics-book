\section*{Chapitre \ref{ch:b2:waves-in-media} --- Ondes électromagnétiques dans les plasmas, les conducteurs et les diélectriques}

\begin{solution}{exo:b2:waves-in-media:1}
$f_p \approx 9\sqrt n$ Hz~: \qty{2.8}{MHz}, \qty{9}{MHz}, \qty{9}{GHz}, \qty{2.6e15}{Hz}
($\lambda_p = \qty{115}{nm}$), \qty{2.8}{kHz}. L'AM à \qty{1}{MHz}~: seulement à travers

\end{solution}

\begin{solution}{exo:b2:waves-in-media:2}

\end{solution}

\begin{solution}{exo:b2:waves-in-media:4}
$f_p = \qty{4.0}{MHz}$~: \qty{5}{MHz} passe~; \qty{2}{MHz} est réfléchie, pénétrant de
$c/\sqrt{\omega_p^2 - \omega^2} = \qty{14}{m}$. Pour $n = \qty{1.5e12}{m^{-3}}$~: \qty{11}{MHz}. La

\end{solution}

\begin{solution}{exo:b2:waves-in-media:5}
(a) $1/\gamma\pi a^2 = \qty{5.3}{m\ohm/m}$. (b) $\delta = \qty{65}{\micro m}$~; $R \approx 1/\gamma2\pi a\delta
= \qty{41}{m\ohm/m}$, soit $7.7$ fois le continu. (c) $\delta = \qty{6.5}{\micro m}$, \qty{0.41}{\ohm/m},
$77$ fois. (d) $R \propto \sqrt f$ croît aussi vite que $L\omega$ dès que la peau
domine, et chaque spire ajoute de la résistance~; le fil de Litz assemble de nombreux
brins isolés plus fins que $\delta$, de sorte que tout le cuivre conduit.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:6}
$\delta_{\text{Al}} = 11.9$, $0.84$, \qty{0.084}{mm}~: $d/\delta = 0.08$, $1.2$, $12$~: $0.7$,
$10$, \qty{103}{dB}. À \qty{50}{Hz} le conducteur est transparent~; le fer de grand
$\mu_r$ détourne le flux (blindage magnétique) et réduit aussi $\delta$ d'un facteur
$\sqrt{\mu_r}$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:7}
(a) $v_0 = eE_0/m\omega$~; $\tfrac12nm\langle v^2\rangle = ne^2E_0^2/4m\omega^2 = \tfrac14\varepsilon_0
E_0^2\,\omega_p^2/\omega^2$. (b) $\tfrac14\varepsilon_0E_0^2$ et $\tfrac14\varepsilon_0E_0^2(1 - \omega_p^2/
\omega^2)$. (c) La somme vaut $\tfrac12\varepsilon_0E_0^2$ (la parenthèse vaut $1$)~; $\langle\Pi
\rangle = E_0^2k/2\mu_0\omega$, et le rapport vaut $c^2k/\omega = v_g$. (d) $\tfrac14\cdot\tfrac14
/\tfrac12 = 1/8$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:8}
(a) $1/v_g = (1/c)(1 - \omega_p^2/\omega^2)^{-1/2} \approx (1 + \omega_p^2/2\omega^2)/c$. (b) $\Delta t
= (L\omega_p^2/2c)(1/\omega_1^2 - 1/\omega_2^2)$ avec $\omega_p^2 = ne^2/\varepsilon_0m$ et $\omega =
2\pi f$~: constante $e^2/8\pi^2\varepsilon_0mc = \qty{1.34e-7}{}$ (SI). (c) $nL = \qty{9.3e23}{m^{-2}}$,
$(1/f_1^2 - 1/f_2^2) = \qty{5.7e-18}{s^2}$~: $\Delta t = \qty{0.72}{s}$. (d) La colonne $nL$
(la mesure de dispersion), donc la distance pour un modèle de $n$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:9}
(a) $\operatorname{Im}\underline\chi = \omega_p^2\Gamma\omega/[(\omega_0^2 - \omega^2)^2 + \Gamma^2\omega^2]$ avec
$\omega_0^2 - \omega^2 \approx 2\omega_0(\omega_0 - \omega)$~: $n'' = \tfrac12\operatorname{Im}\underline\chi$ est
la lorentzienne annoncée. (b) $n''(\omega_0) = \omega_p^2/2\omega_0\Gamma$, $\alpha = 2n''\omega_0/c =
\omega_p^2/\Gamma c$. (c) $\omega_p^2 = \qty{3.2e20}{s^{-2}}$~: $\alpha = \qty{1.8e4}{m^{-1}}$~;
$\ln10/\alpha = \qty{0.13}{mm}$. (d) $n' - 1 \propto (\omega_0 - \omega)/[(\omega_0 - \omega)^2 +
\Gamma^2/4]$~: positif en dessous, négatif au-dessus, décroissant à moins de $\pm\Gamma/2$
de la raie --- anomale en ce domaine.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:10}
(a) $\underline E = E_0\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$, $\underline B = \underline k\underline E/\omega
= (1 - \iu)\underline E/\omega\delta$. (b) $\langle\Pi_x\rangle = \tfrac12\operatorname{Re}(\underline E
\underline B^*)/\mu_0 = E_0^2/2\mu_0\omega\delta$ en $x = 0$. (c) $\int_0^\infty\tfrac12\gamma E_0^2
\eu^{-2x/\delta}\dd x = \gamma E_0^2\delta/4 = E_0^2/2\mu_0\omega\delta$ en utilisant $\gamma = 2/\mu_0\omega\delta^2$.
(d) $H_0 = \sqrt2E_0/\mu_0\omega\delta$, donc $\tfrac12R_sH_0^2 = E_0^2/\gamma\mu_0^2\omega^2\delta^3 =
E_0^2/2\mu_0\omega\delta$. Cuivre à \qty{1}{GHz}~: $R_s = 1/\gamma\delta = \qty{8}{m\ohm}$~;
$\tfrac12 \times 8 \times 10^{-3} \times 10^4 = \qty{40}{W/m^2}$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:11}
(a) $\omega_p = \qty{1.4e16}{rad/s}$, $\lambda_p = 2\pi c/\omega_p = \qty{140}{nm}$. (b) $\omega = \qty{3.8e15}{rad/s}
< \omega_p$~: $1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{23}{nm}$, réflexion totale. (c)
Encore en dessous de $\omega_p$~: réfléchissant~; la transparence n'apparaît qu'au-delà de \qty{140}{nm} (dans le
sodium, \qty{210}{nm}). (d) Les chocs donnent une partie réelle à $\underline\gamma$ et
quelques pour cent d'absorption~; les résonances d'électrons liés dans le visible
(or, cuivre) absorbent le bleu et colorent le métal.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:12}
(a) $\omega\tau = 0.12$~: $\underline\varepsilon_r = 84 - 9.7\iu$~; à \qty{20}{GHz}, $\omega\tau = 1$~:
$45 - 40\iu$. (b) $\underline n \approx 9.2 - 0.53\iu$~; $c/2n''\omega = \qty{1.8}{cm}$~: le four
cuit les deux centimètres extérieurs, la conduction fait le reste. (c) $\tfrac12 \times
1.54 \times 10^{10} \times 8.85 \times 10^{-12} \times 9.7 \times 4 \times 10^6 = \qty{2.6}{MW/m^3}$. (d) À
\qty{20}{GHz} la longueur d'absorption serait d'un millimètre~: seule la peau
cuirait~; \qty{2.45}{GHz} pénètre, et c'est une bande libre.
\end{solution}

\begin{solution}{pb:b2:waves-in-media:1}
\textbf{1.} $f_p = \qty{9}{MHz}$~: le GPS passe, l'onde courte est réfléchie.

\textbf{2.} $k = (\omega/c)\sqrt{1 - \omega_p^2/\omega^2}$~: $v_\varphi = \omega/k \approx c(1 + \omega_p^2/
2\omega^2)$, $v_g = \dd\omega/\dd k = c^2k/\omega \approx c(1 - \omega_p^2/2\omega^2)$.

\textbf{3.} $\Delta t = \int(1/v_g - 1/c)\dd z = \int\omega_p^2\dd z/2c\omega^2 = e^2N_T/2\varepsilon_0
mc\omega^2 = (e^2/8\pi^2\varepsilon_0mc)N_T/f^2 = 1.34 \times 10^{-7} \times 2 \times 10^{17}/2.48 \times
10^{18} = \qty{11}{ns}$~: soit \qty{3.2}{m}.

\textbf{4.} $N_T = (\Delta t_1 - \Delta t_2)/K(1/f_1^2 - 1/f_2^2)$~; l'écart vaut
\qty{7}{ns}~: $1\%$ demande \qty{0.07}{ns}.

\textbf{5.} $v_\varphi$ dépasse $c$ d'autant que $v_g$ lui est inférieure~: la
phase de la porteuse arrive en avance tandis que la modulation arrive en retard. Un
récepteur à phase de porteuse applique la correction avec le signe opposé.

\textbf{6.} \qty{3.2}{m} de jour, \qty{0.3}{m} de nuit.

\textbf{7.} $f_p = \qty{0.9}{MHz}$ avec des chocs~: les ondes moyennes et courtes sont
absorbées (une coupure radio)~; le GPS à \qty{1.5}{GHz} subit une absorption
$\propto 1/\omega^2$, négligeable.

\textbf{8.} $f_1\Delta t = 1.575 \times 10^9 \times 1.1 \times 10^{-8} = 17$ périodes.

\textbf{9.} $\delta = \sqrt{2/\mu_0\gamma\omega} = \qty{10}{\micro m}$~: une paroi d'un millimètre fait
cent \omterm{def:b2:dispersion-wave-packets:complex}{épaisseurs de peau}.

\textbf{10.} $R_s = 1/\gamma\delta = \qty{0.1}{\ohm}$~; $\tfrac12R_sH_0^2 = \qty{45}{W/m^2}$~: \qty{45}{W},
soit $6\%$.

\textbf{11.} $\eu^{-\pi \times 0.5} = 0.21$~: \qty{-14}{dB} en amplitude, \qty{-27}{dB} en
puissance (et le petit trou se couple mal au départ).

\textbf{12.} La lumière, $\lambda = \qty{0.5}{\micro m}$, est deux mille fois plus petite
que les trous et passe~; la micro-onde, $\lambda = \qty{12}{cm}$, est
cent fois plus grande.

\textbf{13.} $\underline n = 2 - 0.005\iu$~; $c/2n''\omega = \qty{1.9}{m}$~: le verre n'est
guère chauffé.

\textbf{14.} $j \approx H_0/\delta = \qty{3e7}{A/m^2}$, $j^2/\gamma = \qty{9e8}{W/m^3}$ dans une
couche de dix micromètres~: les pointes chauffent en quelques millisecondes, émettent des électrons,
ionisent l'air --- des étincelles.

\textbf{15.} $\delta = \qty{18}{\micro m}$~; section $2\pi a\delta = \qty{5.6e-8}{m^2}$ contre
$\pi a^2 = \qty{7.9e-7}{m^2}$~: $R_{\text{ca}} = \qty{0.30}{\ohm/m}$, quatorze fois le continu.

\textbf{16.} $RI^2 = \qty{120}{W/m}$~; surface $\pi d = \qty{3.1e-3}{m^2}$ par mètre~:
$\Delta T \approx \qty{1900}{K}$ --- il fondrait~; de telles bobines sont des tubes refroidis à l'eau.

\textbf{17.} $\delta = \qty{0.14}{mm}$~: le courant induit, et la chaleur, restent
dans un dixième de millimètre~; une impulsion brève suivie d'une trempe durcit
la surface d'un engrenage tandis que le cœur reste tenace.

\textbf{18.} $R = 1/\gamma2\pi a\delta = \qty{0.10}{\ohm/m}$, trois fois moins~; l'intérieur
ne transporte rien --- d'où le tube, avec le fluide de refroidissement là où le
cuivre serait gaspillé.

\textbf{19.} $\delta = a$ pour $f = 1/\pi\mu_0\gamma a^2 = \qty{17}{Hz}$~; en dessous, le courant est
uniforme et la formule du continu s'applique.

\textbf{20.} $n^2 \approx 1 + \omega_p^2/\omega_0^2 + (\omega_p^2/\omega_0^2)(\lambda_0/\lambda)^2$~: la forme de Cauchy
avec $n(\infty)^2 - 1 = \omega_p^2/\omega_0^2 = 1.25$ pour $n(\infty) = 1.5$.

\textbf{21.} $n = 1.531$ et $1.510$, $v = 1.96$ et \qty{1.99e8}{m/s}~; $n_g(550) =
1.550$~; $n_g(400) - n_g(700) = 0.063$~: \qty{210}{ns} sur un kilomètre.

\textbf{22.} $\sin((A + D)/2) = 1.5165 \times 0.5$~: $D = \qty{38.6}{\degree}$~; $\dd D = 2\sin
(A/2)\dd n/\cos((A + D)/2) = 1.53\,\dd n$~: $\Delta D = \qty{1.85}{\degree}$, soit \qty{6.5}{cm} à \qty{2}{m}.

\textbf{23.} À \qty{200}{nm} la fréquence approche la résonance et
$n''$ croît~: opaque. Une faible résonance à \qty{700}{nm} absorbe le rouge~:
un verre vert.

\textbf{24.} Le bleu est plus proche de la résonance ultraviolette~: les électrons
liés répondent davantage, $n$ est plus grand, et la déviation aussi.

\textbf{25.} Couche F à \qty{1.5}{GHz}~: $\underline\gamma$ imaginaire, propagation
avec un faible retard de groupe. Acier à \qty{2.45}{GHz}~: $\underline\gamma$ réel, \omterm{thm:b2:waves-in-media:skin}{effet
de peau}, réflexion. Cuivre à \qty{13.56}{MHz}~: de même, une peau de
\qty{18}{\micro m}. Verre dans le visible~: $\iu\omega\varepsilon_0\chi$, un indice réel, un
milieu transparent et dispersif.
\end{solution}
