\begin{solution}{exo:b2:gratings:1}

\qty{9}{\degree}. À \qty{30}{\degree}~: $\sin\theta = 0.5 + 0.275p$~: $p = +1$ (\qty{51}{\degree}) d'un

l'autre.
\end{solution}

\begin{solution}{exo:b2:gratings:4}

\begin{solution}{exo:b2:gratings:5}
(a) Zéros en $\varphi = 2\pi m/N$, $m = 1, \dots, N - 1$~: $N - 1$ zéros, $N - 2$ maxima
entre eux. (b) En $N\varphi/2 = 3\pi/2$, $I \approx I_0/\sin^2(3\pi/2N) \approx I_0(2N/3\pi)^2$~:
$4/9\pi^2 = 0.045$ de $N^2I_0$. (c) $N = 3$~: un seul \omterm{thm:b2:gratings:nwaves}{maximum secondaire} en
$\varphi = \pi$, de hauteur $I_0$, le neuvième du principal $9I_0$~; $N = 6$~: quatre
maxima secondaires de quelques pour cent. (d) Le lobe central de $\sin^2x/x^2$
retient $90\%$ de l'énergie.
\end{solution}

\begin{solution}{exo:b2:gratings:6}
(a) $p > \lambda_{\min}/(\lambda_{\max} - \lambda_{\min}) = 1.33$~: à partir de $p = 2$. (b) $\lambda/p$. (c)
\qty{25}{nm}~; un prisme croisé avec le réseau empile les ordres les uns au-dessus
des autres. (d) $\Delta\lambda = \lambda^2/2L$, une fraction de nanomètre --- des centaines
de fois moins que pour un réseau.
\end{solution}

\begin{solution}{exo:b2:gratings:7}
(a) $\sin\beta = \lambda/2a = 0.3$, $\beta = \qty{17.5}{\degree}$. (b) La facette seule diffracte
en un large lobe centré sur sa direction spéculaire~; les ordres qui
tombent dedans reçoivent la lumière. (c) Ordre 2 ($2 \times \qty{500}{nm} = \qty{1}{\micro m}$ sous
le même angle). (d) $R = 2W\sin\beta/\lambda = 3.6 \times 10^5$.
\end{solution}

\begin{solution}{exo:b2:gratings:8}

$R = 30\,000$~: \qty{0.018}{nm}. (c) La fente, par un facteur neuf~; égalité en $s = \qty{6}{\micro m}$.

\end{solution}

\begin{solution}{exo:b2:gratings:9}
(a) $p = 2L/\lambda = 16\,700$~; ISL $c/2L = \qty{30}{GHz}$, $\lambda^2/2L = \qty{0.036}{nm}$. (b) $\mathcal F
= \pi\sqrt{0.95}/0.05 = 61$~; \qty{0.5}{GHz}, \qty{0.6}{pm}. (c) $p\mathcal F = 10^6$, quatre
fois celui du réseau de \qty{10}{cm}. (d) Son \omterm{prop:b2:gratings:fabryperot}{intervalle spectral libre} est une fraction
de nanomètre~: les ordres se recouvrent à moins de préfiltrer la lumière
sur l'un d'eux.
\end{solution}

\begin{solution}{exo:b2:gratings:10}
(a) $|\sin\theta - \sin\theta_0| \le 2$. (b) Les rayons extrêmes diffèrent en marche de
$W(\sin\theta - \sin\theta_0)$, au plus $2W$. (c) $2 \times 0.3/5 \times 10^{-7} = 1.2 \times 10^6$~; $c/R
= \qty{250}{m/s}$. (d) Le centre d'une raie se localise au centième de sa
largeur, et mille raies moyennent en $\sqrt{1000}$~: des mètres par
seconde, au prix d'une stabilité héroïque.
\end{solution}

\begin{solution}{exo:b2:gratings:11}
(a) L'onde réfléchie par le plan inférieur parcourt $2d\sin\theta$ de plus.
(b) $\sin\theta = 0.273p$~: \qty{15.8}{\degree}, \qty{33.1}{\degree}, \qty{55.0}{\degree}. (c) $\lambda \gg 2d$~:
aucun angle ne satisfait l'équation~; le verre n'a pas de plans réguliers, d'où
seulement des anneaux diffus --- il est amorphe. (d) $\sin\theta \le 1$.
\end{solution}

\begin{solution}{exo:b2:gratings:12}
(a) Les voisines diffèrent en phase de $2\pi a\sin\theta/\lambda$ par le trajet et de $-\psi$
par l'alimentation~; le \omterm{thm:b2:gratings:nwaves}{maximum principal} en $\varphi = 0$. (b) Premier zéro en
$\Delta\varphi = 2\pi/N$~: $\Delta\theta = \lambda/Na\cos\theta$. (c) Un second \omterm{thm:b2:gratings:nwaves}{maximum principal} en
$\varphi = \pm2\pi$ exige $|\sin\theta \pm \lambda/a| \le 1$~: impossible pour tout angle de
pointage seulement si $a \le \lambda/2$. (d) $Na = \qty{0.96}{m}$~: \qty{1.8}{\degree}~; $50$ pas,
\qty{50}{\micro s}.
\end{solution}

\begin{solution}{pb:b2:gratings:1}
\textbf{1.} $a = \qty{0.833}{\micro m}$~; $N = 144\,000$.

\textbf{2.} $\sin\theta = \sin20^\circ - \lambda/a$ (l'ordre se trouve de l'autre côté de
la normale)~: \qty{550}{nm} à \qty{18.5}{\degree}, \qty{400}{nm} à \qty{7.9}{\degree}, \qty{700}{nm} à
\qty{29.9}{\degree}~: \qty{22}{\degree} de spectre.

\textbf{3.} $1/a\cos\theta = \qty{1.27e-3}{rad/nm}$~; $f_2\times$~: \qty{1.27}{mm/nm}, soit
\qty{0.79}{nm/mm}, \qty{0.012}{nm} par pixel.

\textbf{4.} $R = N = 144\,000$~: \qty{0.0038}{nm}, le tiers d'un pixel.

\textbf{5.} Le \qty{400}{nm} du second ordre tombe là où le \qty{800}{nm} du premier
tomberait, au-delà du rouge~: pas de recouvrement dans le visible~; il commencerait
à \qty{350}{nm}, qu'un filtre de verre élimine.

\textbf{6.} \qty{22}{\degree} $\times$ \qty{1}{m} $= \qty{38}{cm}$~: un détecteur en couvre une tranche~;
on tourne le réseau pour la choisir.

\textbf{7.} $\lambda = 2a\sin17^\circ = \qty{490}{nm}$.

\textbf{8.} $0.8 \times 0.4 = 32\%$.

\textbf{9.} Angles \qty{26.5}{\degree}, \qty{14.0}{\degree}, \qty{10.3}{\degree}~: $+14$, $-8$ et
\qty{-14}{cm} de la position du \qty{550}{nm}.

\textbf{10.} \qty{0.18}{nm} font $15$ pixels~: séparées.

\textbf{11.} $50$ pixels d'écart, chacune large de $4$ pixels.

\textbf{12.} \qty{100}{\micro m}, $6.7$ pixels, \qty{0.08}{nm} --- vingt fois la
limite du réseau~: limité par la fente.

\textbf{13.} L'image fait \qty{97}{\micro m} de large~: une fente de \qty{50}{\micro m} laisse passer
la moitié de la lumière.

\textbf{14.} Moins de lumière pour la même finesse~: le bruit de photons croît.

\textbf{15.} $\Delta\lambda = \lambda v/c = \qty{0.066}{nm}$, $5.5$ pixels.

\textbf{16.} La vitesse de la Terre décale chaque raie de $\pm5.5$
pixels au plus sur l'année~; les éphémérides la donnent bien
mieux que les $0.3\%$ nécessaires pour \qty{100}{m/s}.

\textbf{17.} \qty{2.2e-5}{nm}, $0.002$ pixel --- très en deçà de la largeur de raie. Un
centroïde au centième d'une raie de \qty{7}{px} vaut \qty{0.07}{px}~; mille
raies donnent $\sqrt{1000}$~: $0.002$ pixel --- le signal, tout juste. Les instruments
réels ajoutent du pouvoir de résolution et de la stabilité.

\textbf{18.} $\Delta\sin\theta = (\lambda/a) \times 10^{-5} = 6.6 \times 10^{-6}$, $\Delta\theta = \qty{7e-6}{rad}$,
\qty{7}{\micro m}~: un demi-pixel par kelvin --- des millikelvins pour \qty{10}{m/s}.

\textbf{19.} Ses raies, enregistrées au même instant sur les mêmes
pixels, donnent l'échelle des longueurs d'onde et suivent toute dérive.

\textbf{20.} $a = \qty{3.33}{\micro m}$, $N = 36\,000$, même angle (\qty{18.5}{\degree}), $R = 4N
= 144\,000$, même dispersion, ISL $\lambda/4 = \qty{137}{nm}$~: pouvoir de résolution et
dispersion ne dépendent que de $W$ et de l'angle~; l'\omterm{prop:b2:gratings:fabryperot}{intervalle spectral libre}
a rétréci.

\textbf{21.} ISL $\lambda^2/2L = \qty{0.15}{nm}$ ($12$ pixels)~; $\mathcal F = 30$~; pics
larges de \qty{0.005}{nm}~: un peigne de raies fines en travers du détecteur, une
règle à longueurs d'onde.

\textbf{22.} $2 \times 10^{-4}$ contre \qty{1.3e-3}{rad/nm}~: six fois moins, et le
pouvoir de résolution d'un prisme ($b\,\dd n/\dd\lambda \sim 10^4$) comme son échelle non linéaire
sont perdants aussi~; les réseaux fonctionnent en outre dans l'ultraviolet.

\textbf{23.} $R \le 2W/\lambda$~: $4 \times 10^6$ pour un mètre à \qty{500}{nm} --- largeur et
angle sont les seules voies vers le haut.

\textbf{24.} $10^4 \times 0.012 \times 0.32 = 38$ photons par pixel et par seconde~; $3800$
en \qty{100}{s}~: RSB $\approx 60$.

\textbf{25.} La dispersion, fixée par $p/a\cos\theta$ et la distance focale de la caméra~;
le pouvoir de résolution, par la largeur éclairée et les angles ($pN$)~;
l'\omterm{prop:b2:gratings:fabryperot}{intervalle spectral libre}, par l'ordre ($\lambda/p$).
\end{solution}
