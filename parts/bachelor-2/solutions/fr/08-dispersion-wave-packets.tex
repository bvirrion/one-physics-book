\section*{Chapitre \ref{ch:b2:dispersion-wave-packets} --- Dispersion et paquets d'ondes}

\begin{solution}{exo:b2:dispersion-wave-packets:1}
(a) $k = \qty{0.063}{rad/m}$, $\omega = \sqrt{gk} = \qty{0.79}{rad/s}$, $T = \qty{8.0}{s}$~;
$v_\varphi = \qty{12.5}{m/s}$, $v_g = \qty{6.3}{m/s}$. (b) $\omega = \qty{0.45}{rad/s}$, $k = \omega^2
/g = \qty{0.021}{rad/m}$, $\lambda = \qty{310}{m}$~; $v_\varphi = gT/2\pi = \qty{22}{m/s}$, $v_g =
\qty{11}{m/s}$~; $5 \times 10^6/11 = \qty{4.6e5}{s} = 5.3$ jours. (c) $\lambda = \qty{6}{m}$, $v_g
= \qty{1.6}{m/s}$~: il lui faudrait cinq semaines, et les vagues courtes sont amorties
par la \omterm{def:b2:viscous-flows:viscosity}{viscosité} et brisées par le vent bien avant.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:2}
$\omega = \qty{1.26e8}{rad/s}$, $\omega_c = \qty{6.3e7}{rad/s}$~: $k = \sqrt{\omega^2 - \omega_c^2}/c =
\qty{0.36}{rad/m}$, $v_\varphi = \qty{3.5e8}{m/s}$, $v_g = c^2k/\omega = \qty{2.6e8}{m/s}$,
produit $c^2$. À \qty{5}{MHz}, $\omega < \omega_c$~: $\kappa = \sqrt{\omega_c^2 - \omega^2}/c =
\qty{0.18}{m^{-1}}$, profondeur \qty{5.5}{m}.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:3}
Battement à \qty{4}{Hz}, période des variations d'intensité \qty{0.25}{s}. Eau~: $f = 0.125$ et
\qty{0.119}{Hz}~; enveloppe $\cos(\delta\omega t - \delta kx)$ avec $\delta k = \qty{2.9e-3}{rad/m}$
(battements distants de \qty{1.1}{km}) et vitesse $\delta\omega/\delta k = \qty{6.4}{m/s}$ --- la
\omterm{prop:b2:dispersion-wave-packets:group}{vitesse de groupe} à \qty{105}{m}, $\tfrac12\sqrt{g\lambda/2\pi} = \qty{6.4}{m/s}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:4}
$\ln(b/a) = 1.20$~: $\Lambda = \qty{0.24}{\micro H/m}$, $\Gamma = \qty{106}{pF/m}$, $c = 1/
\sqrt{\Lambda\Gamma} = \qty{2.0e8}{m/s}$, $Z_c = \qty{48}{\ohm}$, \qty{5.0}{ns/m}~; \qty{10}{ns}
$\leftrightarrow$ \qty{2.0}{m}. Air~: $\Gamma = \qty{46}{pF/m}$, $c = c_0$, $Z_c = \qty{72}{\ohm}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:5}
(a) $\omega^2 = c^2k^2 + (EI/\mu)k^4$~; $v_\varphi = \sqrt{c^2 + (EI/\mu)k^2}$~; $v_g = (c^2k +
2(EI/\mu)k^3)/\omega$. (b) $c^2 = \qty{1.12e5}{m^2/s^2}$, $k = \qty{66}{rad/m}$, $(EI/\mu)k^2
= \qty{7.0e3}{m^2/s^2}$~: $v_\varphi/c = \sqrt{1.063} = 1.031$, $+3.1\%$ --- les $53$ cents du
\cref{ch:b2:waves-on-strings}. (c) $v_g > v_\varphi$~: anomale~; les harmoniques
élevées prennent de l'avance et l'attaque d'un pincement devient un bref
glissando descendant.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:6}
(a) $-\mu\omega^2 + \iu\alpha\omega = -Tk^2$~: $\underline k^2 = (\omega^2/c^2)(1 - \iu\alpha/\mu\omega)$.
(b) $\underline k \approx (\omega/c)(1 - \iu\alpha/2\mu\omega)$~: $k' = \omega/c$, $k'' = \alpha/2\mu c =
\alpha/2Z$~; $8.7k''$ dB/m. (c) $c = \qty{100}{m/s}$, $Z = \qty{0.5}{kg/s}$, $\mu\omega = 3.1 \gg
0.02$~: $k'' = \qty{0.02}{m^{-1}}$, $\delta = \qty{50}{m}$, \qty{0.17}{dB/m}. (d) Pas dans cette
limite ($k' = \omega/c$)~; lorsque $\alpha \gtrsim \mu\omega$ la racine carrée ne se
linéarise plus et $k'$ dépend non linéairement de $\omega$~: dispersive (c'est la
limite diffusive du câble).
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:7}
(a) $v_\varphi = \sqrt{\gamma k/\rho}$, $v_g = \tfrac32v_\varphi$~: le groupe devance les
crêtes. (b) $v_\varphi^2 = g/k + \gamma k/\rho$ est minimal en $k_m = \sqrt{\rho g/\gamma}$~:
$\lambda_m = 2\pi\sqrt{\gamma/\rho g} = \qty{1.7}{cm}$, $v_{\min} = (4g\gamma/\rho)^{1/4} =
\qty{23}{cm/s}$. (c) Un sillage est un motif stationnaire par rapport au
bateau, fait d'ondes dont la \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase} égale sa vitesse~; aucune n'est
plus lente que \qty{23}{cm/s}. (d) Régime capillaire, $v_g > v_\varphi$~: les crêtes
naissent à l'avant de l'anneau et meurent à l'arrière.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:8}
(a) $k = \qty{0.126}{rad/m}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-18}{m^2/s}$~: $t_{\text{ét}}
= 2.5 \times 10^5/18 = \qty{1.4e4}{s}$, quatre heures. (b) \qty{10}{ps}~: $t_{\text{ét}} =
4 \times 10^{-6}/0.16 = \qty{25}{\micro s}$, soit \qty{5}{km} à \qty{2e8}{m/s}~; \qty{1}{ns}~: $0.04/0.16 =
\qty{0.25}{s}$, \qty{50000}{km}. Les impulsions courtes (hauts débits) s'étalent en quelques
kilomètres~: les liaisons longues utilisent une fibre compensatrice ou des débits modérés.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:9}
(a) $\partial_x^2v = R\Gamma\,\partial_tv$, $D = 1/R\Gamma = \qty{1.1e12}{m^2/s}$. (b) $\underline k^2
= -\iu R\Gamma\omega$~: $k' = k'' = \sqrt{R\Gamma\omega/2}$~; à \qty{1}{Hz}~: $k = \qty{1.7e-6}{m^{-1}}$,
$v_\varphi = \qty{3.7e6}{m/s}$, $\delta = \qty{600}{km}$~; à \qty{10}{Hz}~: $v_\varphi = \qty{1.2e7}{m/s}$,
$\delta = \qty{190}{km}$. (c) $R\Gamma L^2 = 9 \times 10^{-13} \times 9 \times 10^{12} = \qty{8}{s}$
par point~: environ un mot à la minute. (d) $\Lambda = R\Gamma/G = \qty{9}{H/km}$ --- une
inductance absurde pour un câble sous-marin~; les bobines de charge fonctionnaient sur les lignes
téléphoniques terrestres, dont les $G$ et $R\Gamma$ étaient différents.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:10}
(a) $\partial_te = \dot y(\mu\ddot y + Ky) + Ty'\dot y' = \dot yTy'' + Ty'\dot y' = \partial_x
(Ty'\dot y) = -\partial_x\mathcal P$. (b) $\langle e\rangle = \tfrac14A^2(\mu\omega^2 + Tk^2 + K) =
\tfrac12\mu\omega^2A^2$ (en utilisant $\mu\omega^2 = Tk^2 + K$)~; $\langle\mathcal P\rangle = \tfrac12Tk\omega
A^2$. (c) Rapport $Tk/\mu\omega = c^2k/\omega = v_g$. (d) $y = A\eu^{-\kappa x}\cos\omega t$~:
$\mathcal P \propto \sin2\omega t$, de moyenne nulle --- aucune énergie ne traverse le milieu.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:11}
(a) $\tanh kh \to 1$~: $\omega^2 = gk$~; $kh \ll 1$~: $\omega = \sqrt{gh}\,k$, célérité $\sqrt{gh}$,
non dispersive. (b) $kh = 0.13$~: eau peu profonde~; $c = \sqrt{9.81 \times 4000} = \qty{198}{m/s}$,
$T = \qty{17}{min}$~; \qty{8000}{km} en \qty{11}{h}. (c) $\tfrac12\rho gA^2 = \qty{1.2}{kJ/m^2}$
sur $200 \times 1000$ km$^2$~: \qty{2.5e14}{J}. Green~: $A \propto h^{-1/4}$, $(4000/10)^{1/4}
= 4.5$~: \qty{2.2}{m}, avant redressement et déferlement. (d) Vers $h \approx \lambda/2 =
\qty{50}{m}$~; la portion de crête située en eau moins profonde ralentit, de sorte que la
crête pivote parallèlement aux lignes de niveau du fond --- c'est la réfraction.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:12}
(a) En $x = \ell$~: $v = v_i + v_r$, $i = (v_i - v_r)/Z_c$, $v = R_Li$~: $r = (R_L -
Z_c)/(R_L + Z_c)$. (b) $0$, $-1$, $+1$, $0.5$. (c) L'entrée de la ligne se comporte comme
$Z_c$, si bien que l'oscilloscope voit \qty{0.5}{V} en $t = 0$, puis l'écho $0.5r$ V à
$2\ell/c = \qty{1}{\micro s}$~: rien, \qty{-0.5}{V}, \qty{+0.5}{V}, \qty{+0.25}{V}~; le
générateur adapté absorbe l'écho, de sorte qu'il n'y en a pas de second. (d)
$\tfrac12 \times 2 \times 10^8 \times 1.8 \times 10^{-6} = \qty{180}{m}$.
\end{solution}

\begin{solution}{pb:b2:dispersion-wave-packets:1}
\textbf{1.} $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi = -\Gamma\partial_tv$~; avec
$\eu^{\iu(\omega t - kx)}$~: $\iu kv = (\iu\omega\Lambda + R)i$, $\iu ki = \iu\omega\Gamma v$, d'où
$k^2 = \omega\Gamma(\omega\Lambda - \iu R)$.

\textbf{2.} $\omega = R/\Lambda = \qty{4000}{rad/s}$, soit \qty{640}{Hz}~: à quelques hertz $\Lambda\omega
\ll R$, régime diffusif.

\textbf{3.} $k' = k'' = \sqrt{R\Gamma\omega/2}$ avec $R\Gamma = \qty{5e-13}{s/m^2}$~: à \qty{1}{Hz}~:
$\qty{1.25e-6}{m^{-1}}$, $v_\varphi = \qty{5e6}{m/s}$, $\delta = \qty{800}{km}$~; à \qty{4}{Hz}~:
$\qty{2.5e-6}{m^{-1}}$, $\qty{1e7}{m/s}$, \qty{400}{km}.

\textbf{4.} $8.7L/\delta$~: \qty{38}{dB} à \qty{1}{Hz}, \qty{76}{dB} à \qty{4}{Hz}~: seules les
composantes les plus lentes survivent~; un point net arrive en une longue bosse aplatie.

\textbf{5.} $R\Gamma L^2 = 5 \times 10^{-13} \times 1.2 \times 10^{13} = \qty{6}{s}$~: un point toutes les
six secondes, un mot à la minute (le câble réel, avec de sensibles galvanomètres à
miroir, en faisait quelques-uns).

\textbf{6.} $\Lambda = R/\omega = \qty{80}{mH/km}$. Alors $k^2 = \Gamma R\omega(1 - \iu)$, $k = \sqrt{\Gamma R\omega}
\,2^{1/4}\eu^{-\iu\pi/8}$~: $k' = \qty{3.9e-6}{m^{-1}}$, $k'' = \qty{1.6e-6}{m^{-1}}$~: $v_\varphi =
\qty{6.4e6}{m/s}$, $\delta = \qty{620}{km}$, \qty{49}{dB} sur le câble au lieu de
\qty{76}{dB}.

\textbf{7.} Un point toutes les \qty{6}{s} fait environ \qty{0.2}{bit/s}~: la fibre est
$10^{10}$ à $10^{11}$ fois plus rapide.

\textbf{8.} $v_\varphi = gT/2\pi$, $v_g = gT/4\pi$.

\textbf{9.} \qty{18}{s}~: $v_g = \qty{14.1}{m/s}$, $4 \times 10^6/14.1 = \qty{3.3}{jours}$~;
\qty{6}{s}~: \qty{4.7}{m/s}, \qty{9.9}{jours}~: l'épisode de houle dure plus de six jours.

\textbf{10.} $t = D/v_g = 4\pi D/gT$, donc $1/T = gt/4\pi D$. $\Delta(1/T) = 1/12 - 1/16
= \qty{0.0208}{Hz}$ en \qty{86400}{s}~: $g/4\pi D = \qty{2.4e-7}{s^{-2}}$, $D = \qty{3200}{km}$.

\textbf{11.} $\tfrac12\rho gA^2 = \qty{11}{kJ/m^2}$~; l'énergie se déplace à $v_g = gT/4\pi =
\qty{9.4}{m/s}$~: \qty{103}{kW} par mètre de crête.

\textbf{12.} $10^3 \times 103 \times 0.3 = \qty{31}{MW}$~: dix éoliennes.

\textbf{13.} $\lambda = 2\pi U^2/g = \qty{16}{m}$. L'énergie de chaque composante
se déplace à la moitié de sa \omterm{def:b2:dispersion-wave-packets:relation}{vitesse de phase}, de sorte que le motif est confiné à un dièdre
derrière le bateau (de demi-angle \qty{19.5}{\degree}, quelle que soit sa vitesse).

\textbf{14.} Une crête naît à l'arrière du groupe, le dépasse
(les crêtes vont à $2v_g$ par rapport à l'eau, à $v_g$ par rapport au
groupe) et meurt à l'avant. Le groupe mesure $120 \times 9.4 = \qty{1.1}{km}$
de long, soit cinq longueurs d'onde ($\lambda = gT^2/2\pi = \qty{225}{m}$)~; dix crêtes passent devant la
bouée, chacune ayant vécu environ deux minutes.

\textbf{15.} $kh = 0.13$~: eau peu profonde. $c = \qty{198}{m/s}$, $T = \lambda/c = \qty{17}{min}$,
\qty{8.4}{h}.

\textbf{16.} Non dispersive~: toutes les composantes à \qty{198}{m/s}, arrivant
ensemble en une seule impulsion (la houle, à $\sim\qty{10}{m/s}$, triée par période,
mettrait une semaine).

\textbf{17.} $\tfrac12\rho gA^2 = \qty{1.8}{kJ/m^2}$ sur $500 \times 200$ km$^2$~: \qty{1.8e14}{J},
soit $4\%$ d'une mégatonne.

\textbf{18.} $A = 0.6(4000/20)^{1/4} = \qty{2.3}{m}$~; $c = \sqrt{gh} = \qty{14}{m/s}$, $\lambda =
cT = \qty{14}{km}$. La crête, en eau plus profonde que le creux qui la précède,
va plus vite et le rattrape~: le front se redresse en mascaret.

\textbf{19.} Une élévation de \qty{0.6}{m} en un quart d'heure, sur une pente
de $10^{-5}$~: rien de perceptible.

\textbf{20.} La célérité moyenne, $\sqrt{gh}$ pour la profondeur moyenne~; comme
$\sqrt h$ est concave, les passages peu profonds freinent l'onde plus que les passages profonds
ne l'accélèrent, et la moyenne le long du trajet est inférieure à $\sqrt{g\bar h}$.

\textbf{21.} $\omega' = \tfrac12\sqrt{g/k}$, $\omega'' = -\tfrac14\sqrt{g/k^3}$~; $k = \qty{0.063}{rad/m}$~:
$\omega'' = \qty{-50}{m^2/s}$.

\textbf{22.} $t_{\text{ét}} = 10^6/50 = \qty{2e4}{s}$, soit \qty{5.6}{h}~; le paquet a
parcouru $v_gt = 6.25 \times 2 \times 10^4 = \qty{125}{km}$.

\textbf{23.} Un train long contient une bande étroite de $k$, dont les \omterm{prop:b2:dispersion-wave-packets:group}{vitesses de
groupe} coïncident presque~: les composantes restent groupées plus longtemps. Le
milieu du tsunami est non dispersif~: vitesses égales, aucun étalement.

\textbf{24.} $\omega'' = \hbar/m = \qty{1.2e-4}{m^2/s}$~: $t_{\text{ét}} = 10^{-18}/1.2 \times
10^{-4} = \qty{9e-15}{s}$~; $v_g = \hbar k/m = \qty{7.3e6}{m/s}$~: \qty{60}{nm}.

\textbf{25.} Câble~: $k^2 = -\iu R\Gamma\omega$, $v_\varphi = \sqrt{2\omega/R\Gamma}$, pas de
$v_g$ bien défini, atténue et brouille. Houle~: $\omega^2 = gk$, $v_\varphi = 2v_g$,
s'étale. Tsunami~: $\omega = \sqrt{gh}\,k$, $v_\varphi = v_g$, ni l'un ni l'autre. Électron~:
$\omega = \hbar k^2/2m$, $v_g = 2v_\varphi$, s'étale.
\end{solution}
