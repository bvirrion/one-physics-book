\section*{Chapitre \ref{ch:b2:viscous-flows} --- Écoulements visqueux}

\begin{solution}{exo:b2:viscous-flows:1}
$\tau = \eta U/h = 0.25 \times 2/2 \times 10^{-4} = \qty{2.5}{kPa}$~; $F = \qty{1.25}{kN}$~;
$P = Fv = \qty{2.5}{kW}$. Frottement sec~: $0.3 \times 98 = \qty{29}{N}$ --- ici le
film mince, fortement cisaillé, résiste bien davantage~; la traînée visqueuse croît comme
$U/h$, de sorte que la lubrification est avantageuse aux vitesses plus faibles et avec des films plus épais (et
elle ne grippe jamais).
\end{solution}

\begin{solution}{exo:b2:viscous-flows:2}
(a) $2 \times 10^{-6} \times 3 \times 10^{-5}/10^{-6} = 6 \times 10^{-5}$, rampant. (b)
$1.8 \times 1.5/10^{-6} = 2.7 \times 10^6$, turbulent. (c) $4 \times 30/1.5 \times 10^{-5}
= 8 \times 10^6$, turbulent. (d) $0.025 \times 0.33/3.3 \times 10^{-6} = 2500$,
laminaire à la limite. (e) $900 \times 0.5 \times 0.01/0.1 = 45$, laminaire.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:3}
$D_V = \pi R^4\Delta P/8\eta L = \pi(2 \times 10^{-4})^4 \times 2 \times 10^4/(8 \times 10^{-3}
\times 0.03) = \qty{0.42}{mL/s}$~; \qty{12}{s} pour \qty{5}{mL}. $v_{\max} = \Delta PR^2/
4\eta L = \qty{6.7}{m/s}$, $\langle v\rangle = \qty{3.3}{m/s}$, $\mathrm{Re} = 2R\langle v
\rangle/\nu = 1300$~: laminaire. $\tau_w = \Delta PR/2L = \qty{67}{Pa}$. À \qty{0.8}{mm}~:
seize fois le débit, \qty{6.7}{mL/s} --- mais $\mathrm{Re} \approx 10^4$~: l'écoulement
devient turbulent et le gain est moindre.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:4}
$v_\infty = 2R^2\Delta\rho\,g/9\eta$. (a) \qty{1.3}{cm/s}, $\mathrm{Re} = 0.02$, \qty{75}{s}
par mètre. (b) \qty{0.24}{mm/s}, $\mathrm{Re} = 3 \times 10^{-5}$, \qty{70}{min} par
mètre. (c) \qty{1.2}{m/s} mais $\mathrm{Re} = 16$~: Stokes surestime (vitesse
réelle voisine de \qty{0.7}{m/s}), environ une seconde par mètre.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:5}
(a) $\langle v\rangle = 0.1/0.0707 = \qty{1.4}{m/s}$, $\mathrm{Re} = 4.2 \times 10^5$. (b)
$8\eta LD_V/\pi R^4 = \qty{500}{Pa}$. (c) $\lambda = 0.3/25.5 = 0.0118$, $\Delta P = 0.0118
\times 3333 \times 1000 = \qty{39}{kPa}$~: $80$ fois plus~; $P = \Delta PD_V = \qty{3.9}{kW}$.
(d) Même vitesse, $\mathrm{Re} = 3 \times 10^5$, $\lambda = 0.0128$, $\Delta P = 0.0128 \times 4717
\times 1000 = \qty{60}{kPa}$, \qty{6}{kW}~: plus de paroi par unité de débit, donc plus de perte.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:6}
(a) $v = 2 \times 25 \times 10^{-12} \times 50 \times 9.81/(9 \times 10^{-3}) = \qty{2.7}{\micro m/s}$~;
\qty{5}{cm} en $1.8 \times 10^4$ s, cinq heures. (b) \qty{2.7}{mm/s}, \qty{18}{s}. (c)
$\Omega = \sqrt{1000g/r} = \qty{313}{rad/s} = \qty{3000}{rpm}$. (d) $v \propto R^2$~: les grosses
particules culottent les premières~; des passages successifs à vitesse croissante séparent
les tailles.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:7}
(a) $F = 0.6 \times 144 \times 0.35 = \qty{30}{N}$, $+\qty{4}{N}$~: $P = 34 \times 12 =
\qty{410}{W}$. (b) $47 + 4 = \qty{51}{N}$, \qty{770}{W}. (c) Vitesse relative
\qty{17}{m/s}~: $61 + 4 = \qty{65}{N}$, multiplié par la vitesse sol~: \qty{780}{W}. (d)
Traînée $\qty{21}{N}$~: \qty{300}{W}, soit \qty{110}{W} économisés. (e) $mg\sin\alpha = \qty{39}{N} =
4 + 0.21v^2$~: $v = \qty{13}{m/s}$, \qty{47}{km/h}.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:8}
(a) Golf~: $\tfrac12\rho C_xS = \qty{2.2e-4}{kg/m}$, $v_\infty = \sqrt{0.45/2.2 \times 10^{-4}}
= \qty{45}{m/s}$~; parachutiste $\sqrt{785/0.42} = \qty{43}{m/s}$~; parachute $\sqrt{785/15} =
\qty{7.2}{m/s}$. (b) $m\dot v = mg - kv^2$, soit $\dot v = g(1 - v^2/v_\infty^2)$,
dont la solution partant du repos est $v_\infty\tanh(gt/v_\infty)$. (c) $\tanh x = 0.9$ pour
$x = 1.47$~: $t = 1.47 \times 43/9.81 = \qty{6.5}{s}$~; $z = (v_\infty^2/g)\ln\cosh x =
190 \times 0.83 = \qty{160}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:9}
(a) Taux de cisaillement $\Omega R/e = 0.5/10^{-3} = \qty{500}{s^{-1}}$~; $\Gamma = \tau\cdot2\pi Rh
\cdot R$~: $\tau = 0.02/(2\pi \times 2.5 \times 10^{-3} \times 0.1) = \qty{12.7}{Pa}$. (b) $\eta
= 12.7/500 = \qty{2.5e-2}{Pa.s}$. (c) $\mathrm{Re} = \rho Ue/\eta \approx 1000 \times 0.5 \times
10^{-3}/0.025 = 20$~: laminaire. (d) Un entrefer mince rend le taux de cisaillement uniforme
(approximation plane) et maintient l'\omterm{def:b2:viscous-flows:reynolds}{écoulement laminaire}.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:10}
$D_V = \qty{8.3e-5}{m^3/s}$. (a) $R_h = 8\eta L/\pi R^4 = 8 \times 3.5 \times 10^{-3} \times
0.4/(\pi \times 2.4 \times 10^{-8}) = \qty{1.5e5}{Pa.s/m^3}$, $\Delta P = \qty{12}{Pa}$. (b)
Un capillaire $\qty{3.5e16}{Pa.s/m^3}$~; $10^{10}$ en parallèle~: \qty{3.5e6}{Pa.s/m^3}~;
$\Delta P = \qty{290}{Pa}$ (\qty{2}{mmHg}). (c) Une artériole $8 \times 3.5 \times 10^{-3}
\times 2 \times 10^{-3}/(\pi \times 10^{-20}) = \qty{1.8e15}{Pa.s/m^3}$~; l'ensemble~:
\qty{1.8e7}{Pa.s/m^3}~; $\Delta P = \qty{1.5}{kPa}$~: les artérioles dominent les
deux autres (c'est dans les petits vaisseaux, non dans les gros, que la pression
chute). (d) $P = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, environ $1\%$. (e)
$(1/1.1)^4 = 0.68$~: un tiers de la résistance artériolaire supprimé ---
c'est ainsi que les vasodilatateurs abaissent la pression artérielle.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:11}
(a) $F = b\int_0^L\tau_w\,\dd x = 0.332\rho U^2b\sqrt{\nu/U}\int_0^L\dd x/\sqrt x =
0.664\rho U^2b\sqrt{\nu L/U} = 0.664\rho U^2bL/\sqrt{\mathrm{Re}_L}$. (b) $\mathrm{Re}_L =
3 \times 10^5$ (laminaire)~; $\delta \approx 5\sqrt{\nu L/U} = \qty{9}{mm}$~; $F = 0.664 \times 1000
\times 0.09 \times 1/548 = \qty{0.11}{N}$ par face, \qty{0.22}{N}. (c) $\mathrm{Re}_L = 10^7$~:
la couche est turbulente au-delà des premiers centimètres, la formule laminaire
échoue (la traînée réelle est plusieurs fois plus grande). (d) $\mathrm{Re}_x =
10^9$~: $\delta \approx 0.37 \times 100/63 = \qty{0.6}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:12}
(a) $0 = \rho g + \eta v''(y)$, $v(0) = 0$ (adhérence), $v'(h) = 0$ (aucune contrainte à
la surface libre). (b) On intègre deux fois~: $v = (\rho g/2\eta)(2hy - y^2)$~; $q =
\int_0^hv\,\dd y = \rho gh^3/3\eta$. (c) $v_s = \rho gh^2/2\eta = 1200 \times 9.81 \times
4 \times 10^{-8}/1 = \qty{0.47}{mm/s}$~; $q = \qty{6.3e-8}{m^2/s}$~; \qty{28}{cm} en dix
minutes --- une peinture newtonienne de cette épaisseur coulerait entièrement~; les
peintures réelles sont rhéofluidifiantes et thixotropes, et coulent malgré tout si elles sont trop épaisses
puisque $v_s \propto h^2$. (d) $1400 \times 9.81 \times 4 \times 10^{-6}/20 = \qty{2.7}{mm/s}$.
\end{solution}

\begin{solution}{pb:b2:viscous-flows:1}
\textbf{1.} $S = \qty{4.9}{cm^2}$, $\langle v\rangle = 83/4.9 = \qty{0.17}{m/s}$~; $\mathrm{Re}
= 1060 \times 0.17 \times 0.025/3.5 \times 10^{-3} = 1300$~: laminaire (les pointes pulsatiles
atteignent $\sim 5000$).

\textbf{2.} $v_{\max} = \qty{0.34}{m/s}$~; $\tau_w = 4\eta\langle v\rangle/R = \qty{0.19}{Pa}$.

\textbf{3.} $\mathrm{Re} = 1060 \times 3 \times 10^{-4} \times 8 \times 10^{-6}/3.5 \times 10^{-3} =
7 \times 10^{-4}$~; $\Delta P = 8\eta L\langle v\rangle/R^2 = 8 \times 3.5 \times 10^{-3} \times 10^{-3}
\times 3 \times 10^{-4}/1.6 \times 10^{-11} = \qty{525}{Pa} = \qty{3.9}{mmHg}$.

\textbf{4.} $1/0.3 = \qty{3.3}{s}$~: assez long pour que l'oxygène diffuse sur les quelques
micromètres qui le séparent des tissus (\cref{ch:b2:particle-diffusion}).

\textbf{5.} Une artériole $8\eta L/\pi R^4 = \qty{1.8e15}{Pa.s/m^3}$~; $10^8$ en
parallèle~: \qty{1.8e7}{Pa.s/m^3}~; $\Delta P = 1.8 \times 10^7 \times 8.3 \times 10^{-5} =
\qty{1.5}{kPa} = \qty{11}{mmHg}$.

\textbf{6.} $P = \Delta P\,D_V = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, soit environ
$1\%$ de la puissance de l'organisme.

\textbf{7.} $(1/0.6)^4 = 7.7$. Au repos les vaisseaux en aval se dilatent et
compensent~; à l'effort la demande quadruple ou quintuple et
l'artère rétrécie ne peut la satisfaire~: c'est l'angine de poitrine.

\textbf{8.} Cinq fois la vitesse~: $\mathrm{Re} \approx 6500$, turbulent --- ce sont les
souffles qu'entend un stéthoscope à l'effort.

\textbf{9.} Dans l'écoulement lent et peu cisaillé des petites veines et des capillaires
les globules rouges s'agrègent et la \omterm{def:b2:viscous-flows:viscosity}{viscosité} apparente augmente (l'inverse
dans l'aorte rapide)~; toutes les estimations de chute de pression pour les petits vaisseaux
sont des bornes inférieures.

\textbf{10.} $\langle v\rangle = 3/1.13 = \qty{2.65}{m/s}$~; $\mathrm{Re} = 850 \times 2.65 \times
1.2/0.02 = 1.35 \times 10^5$~: turbulent.

\textbf{11.} $\lambda = 0.316/19.2 = 0.0165$~; $\Delta P/L = (0.0165/1.2) \times \tfrac12 \times
850 \times 7.0 = \qty{41}{Pa/m} = \qty{41}{kPa/km}$~; \qty{534}{bar} sur la ligne.

\textbf{12.} $534/80 = 6.7$~: au moins sept stations.

\textbf{13.} $P = \Delta P\,D_V = 5.34 \times 10^7 \times 3 = \qty{160}{MW}$~;
\qty{53}{MJ/m^3}, soit $0.15\%$ de l'énergie du pétrole.

\textbf{14.} $\Delta P = 32\eta L\langle v\rangle/D^2 = 32 \times 0.02 \times 1.3 \times 10^6
\times 2.65/1.44 = \qty{15}{bar}$~: trente-cinq fois moins. La turbulence coûte
cher~; quelques parties par million de longs polymères la retardent et réduisent
la chute de plusieurs dizaines de pour cent.

\textbf{15.} $\mathrm{Re} = 1350$~: laminaire~; Poiseuille~: $\Delta P = 32 \times 2 \times 1.3 \times
10^6 \times 2.65/1.44 = \qty{1500}{bar}$ --- impossible~; le pétrole doit être maintenu
chaud, d'où le calorifugeage et le chauffage.

\textbf{16.} $1.3 \times 10^6/2.65 = \qty{4.9e5}{s} = \qty{5.7}{jours}$.

\textbf{17.} $\tau_w = (\Delta P/L)D/4 = 41 \times 0.3 = \qty{12}{Pa}$~; $F = \tau_w\pi DL =
\qty{6e7}{N}$.

\textbf{18.} $\Delta P \propto \lambda\langle v\rangle^2/D \propto D^{-4.75}$~: $1.2^{4.75} = 2.4$.

\textbf{19.} $v = 2R^2\Delta\rho\,g/9\eta = 2 \times 2.5 \times 10^{-9} \times 1800 \times 9.81/
0.18 = \qty{0.49}{mm/s}$~; $\mathrm{Re} = 2 \times 10^{-3}$~; \qty{41}{min} pour traverser
\qty{1.2}{m}. Les fluctuations turbulentes de quelques centimètres par seconde
le maintiennent en suspension tant que le pétrole circule~; à l'arrêt il sédimente en
moins d'une heure.

\textbf{20.} $v = 2 \times 2.5 \times 10^{-11} \times 1000 \times 9.81/1.62 \times 10^{-4} =
\qty{3.0}{mm/s}$~; $\mathrm{Re} = 2 \times 10^{-3}$~; \qty{1}{km} en $3.3 \times 10^5$ s, quatre
jours~: le moindre courant ascendant supérieur à quelques millimètres par seconde la maintient.

\textbf{21.} \qty{1.2}{m/s}, $\mathrm{Re} = 16$~: Stokes surestime d'environ un
facteur deux (réel~: \qty{0.7}{m/s}).

\textbf{22.} Stokes donnerait \qty{120}{m/s} et $\mathrm{Re} \sim 10^4$~: absurde.
Traînée quadratique, $\tfrac43\pi R^3\rho_wg = \tfrac12\rho v^2\pi R^2C_x$~: $v = \sqrt{8R
\rho_wg/3\rho C_x} = \qty{6.6}{m/s}$ ($\mathrm{Re} = 900$, cohérent)~; \qty{150}{s} depuis
\qty{1}{km}.

\textbf{23.} \qty{1}{cm/s} $>$ \qty{3}{mm/s}~: la gouttelette monte~; la goutte de bruine
demande environ \qty{1}{m/s}, ce que fournissent les nuages convectifs.

\textbf{24.} $2\gamma/R$~: \qty{29}{kPa}, \qty{1.4}{kPa}, \qty{144}{Pa}~; \omterm{thm:b2:euler-bernoulli:bernoulli}{pressions
dynamiques} $\tfrac12\rho v^2$~: $5 \times 10^{-6}$, $0.3$ et \qty{26}{Pa}. Le rapport
atteint $0.2$ à \qty{2}{mm}~; à \qty{5}{mm} ($v \approx \qty{9}{m/s}$, $\tfrac12\rho v^2 =
\qty{49}{Pa}$ contre \qty{58}{Pa}) la pression aérodynamique aplatit la
goutte et la brise~: aucune goutte de pluie ne dépasse \qty{5}{mm} environ.

\textbf{25.} Aorte $10^3$ (laminaire, près de la transition)~; capillaire $10^{-3}$
(Poiseuille)~; artériole $10^{-2}$ (Poiseuille)~; oléoduc chaud $10^5$
(turbulent, $\lambda(\mathrm{Re})$)~; oléoduc froid $10^3$ (Poiseuille)~; sable dans le
pétrole $10^{-3}$ (Stokes)~; gouttelette de nuage $10^{-3}$ (Stokes)~; goutte de pluie $10^3$
(traînée quadratique, $C_x \approx 0.5$).
\end{solution}
