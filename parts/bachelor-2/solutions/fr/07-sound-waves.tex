\section*{Chapitre \ref{ch:b2:sound-waves} --- Ondes sonores dans les fluides}

\begin{solution}{exo:b2:sound-waves:1}
$c = \sqrt{\gamma RT/M}$~: \qty{331}{m/s}, \qty{343}{m/s}, \qty{299}{m/s}~; hélium
\qty{1010}{m/s}~; CO$_2$ \qty{268}{m/s}. Les cordes vocales vibrent à la même
fréquence, mais les résonances du conduit vocal (qui façonnent le
timbre) varient comme $c$ et montent d'un facteur trois~: la voix paraît
aiguë.
\end{solution}

\begin{solution}{exo:b2:sound-waves:2}

\qty{7e-5}{m/s}$, $\xi_m = v_m/\omega = \qty{22}{nm}$. \qty{120}{dB}~: \qty{1}{W/m^2},
\qty{29}{Pa}, \qty{7}{cm/s}, \qty{22}{\micro m}. Deux~: \qty{63}{dB}~; cent~: \qty{80}{dB}.
\end{solution}

\begin{solution}{exo:b2:sound-waves:3}

\begin{solution}{exo:b2:sound-waves:5}
(a) $500 \times 340/300 = \qty{567}{Hz}$~; $500 \times 340/380 = \qty{447}{Hz}$. (b) $500 \times
380/340 = \qty{559}{Hz}$~; les deux~: $500 \times 380/300 = \qty{633}{Hz}$, contre \qty{654}{Hz}
(source seule à \qty{80}{m/s}) ou \qty{618}{Hz} (observateur seul)~: c'est l'air, et non
la vitesse relative, qui fixe le résultat. (c) Le mur reçoit \qty{567}{Hz}
et les réémet au repos~; le conducteur s'en approche à \qty{40}{m/s}~: $567 \times
380/340 = \qty{633}{Hz}$.
\end{solution}

\begin{solution}{exo:b2:sound-waves:6}
$v_m = p_m/Z$~: \qty{2.4}{mm/s}, \qty{0.67}{\micro m/s}, \qty{25}{nm/s}~; $\xi_m = v_m/\omega$~:
\qty{390}{nm}, \qty{0.11}{nm}, \qty{4}{pm}~; $I = p_m^2/2Z$~: \qty{1.2e-3}{W/m^2} (\qty{91}{dB}),
\qty{3.3e-7}{W/m^2} (\qty{55}{dB}), \qty{1.3e-8}{W/m^2} (\qty{41}{dB}). Un milieu raide
bouge à peine sous une pression donnée et transporte peu de puissance.
\qty{1}{Pa} vaut \qty{120}{dB} par rapport à \qty{1}{\micro Pa}.
\end{solution}

\begin{solution}{exo:b2:sound-waves:7}
(a) $\sqrt{1.013 \times 10^5/1.2} = \qty{290}{m/s}$~; $\times\sqrt{1.4}$~: \qty{343}{m/s}. (b)
$\gamma = c^2M/RT$~: $1.40$ pour l'air, $1.71$ pour l'argon ($5/3$ aux arrondis près)~:
trois degrés de liberté de translation ($\gamma = 1 + 2/3$) contre cinq pour
un gaz diatomique ($1 + 2/5$). (c) $\sqrt{D/f} = \sqrt{2 \times 10^{-8}} = \qty{0.14}{mm}
\ll \lambda = \qty{34}{cm}$~: aucune chaleur ne passe des crêtes aux creux.
\end{solution}

\begin{solution}{exo:b2:sound-waves:8}
(a) Ouvert--ouvert~: $p = p_m\sin kx\cos\omega t$ avec $kL = n\pi$~; fermé--ouvert~:
$p = p_m\cos kx\cos\omega t$ (nœud de vitesse, ventre de pression à l'extrémité
fermée) avec $kL = (2n - 1)\pi/2$. (b) $L = c/2f = \qty{39}{cm}$~; $c/4f = \qty{19.5}{cm}$.
(c) Toutes les harmoniques~; les harmoniques impaires seulement (le son creux de la clarinette).
(d) $0.6 \times 2 = \qty{1.2}{cm}$ par extrémité ouverte~: $L_{\text{eff}} = \qty{41.4}{cm}$, $f =
\qty{414}{Hz}$, $6\%$ trop bas --- le tuyau doit être coupé à \qty{36.6}{cm}.
\end{solution}

\begin{solution}{exo:b2:sound-waves:9}
$\mathcal P = \qty{400}{W}$, $I = \mathcal P/2\pi r^2$. (a) \qty{6.4e-3}{W/m^2}, \qty{98}{dB},
$p_m = \qty{2.3}{Pa}$. (b) $r = \sqrt{400/2\pi} = \qty{8}{m}$~; \qty{70}{dB} à \qty{2.5}{km}.
(c) $98 - 20\log(r/100) - (r - 100)/100 = 70$~: $r \approx \qty{950}{m}$. (d)
$\pi(10)^2/\pi(0.95)^2 \approx 110$ sirènes.
\end{solution}

\begin{solution}{exo:b2:sound-waves:10}
(a) $\rho_0\partial_tv = p_mk\sin kx\cos\omega t$~: $v = (p_m/Z)\sin kx\sin\omega t$. (b)
$e_k = \tfrac12\chi_Sp_m^2\sin^2kx\sin^2\omega t$, $e_p = \tfrac12\chi_Sp_m^2\cos^2kx\cos^2\omega t$~:
l'une est maximale quand l'autre s'annule, dans le temps comme dans l'espace. (c) $E =
S\int_0^L(e_k + e_p)\dd x = \tfrac14\chi_Sp_m^2SL$, constante. (d) $I = pv =
(p_m^2/4Z)\sin2kx\sin2\omega t$~: de moyenne nulle, mais deux fois par période l'énergie
va des ventres de pression aux ventres de vitesse et revient.
\end{solution}

\begin{solution}{exo:b2:sound-waves:11}
(a) Le front émis en $t = 0$ a le rayon $ct$ lorsque la source est à $vt$~:
le cône tangent a $\sin\theta = c/v$. (b) $\theta = \qty{39}{\degree}$~; le cône
atteint le sol sous l'avion lorsque celui-ci s'est éloigné de $h/\tan\theta = \qty{15}{km}$~:
$15000/480 = \qty{31}{s}$ après le passage à la verticale. (c) Un
tapis large de \qty{60}{km} le long de toute la route entend le bang. (d) $\sin\theta
= 340/800$, $\theta = \qty{25}{\degree}$~; le claquement du cône arrive d'abord, la
détonation de la bouche du canon, qui voyage à $c$ depuis l'arme, ensuite.
\end{solution}

\begin{solution}{exo:b2:sound-waves:12}
(a) $f_1 = f(c + v\cos\theta)/c$. (b) $f_2 = f_1c/(c - v\cos\theta) = f(c + v\cos\theta)
/(c - v\cos\theta) \approx f(1 + 2v\cos\theta/c)$. (c) $2 \times 5 \times 10^6 \times 0.5 \times 0.5/1540
= \qty{1.6}{kHz}$~; nul à \qty{90}{\degree}~: incliner la sonde. (d) $\Delta v = 50 \times 1540/
(2 \times 5 \times 10^6 \times 0.5) = \qty{1.5}{cm/s}$. L'oreille suit la hauteur et le timbre en
temps réel~: un sifflement régulier signale un \omterm{def:b2:viscous-flows:reynolds}{écoulement laminaire}, un souffle de la turbulence
derrière un rétrécissement.
\end{solution}

\begin{solution}{pb:b2:sound-waves:1}
\textbf{1.} $c = \sqrt{1.4 \times 8.31 \times 288/0.029} = \qty{340}{m/s}$~; \qty{349}{m/s}
à \qty{30}{\degreeCelsius}. \qty{1}{km} en \qty{2.9}{s}.

\textbf{2.} \qty{1.4}{km}. Le canal fait plusieurs kilomètres~; le son de
ses différentes parties arrive sur plusieurs secondes, avec des échos sur les nuages et
le sol.

\textbf{3.} $I = \qty{1}{W/m^2}$~: $p_m = \sqrt{2ZI} = \qty{29}{Pa}$, $v_m = \qty{7}{cm/s}$.

\textbf{4.} $It = \qty{2}{J/m^2}$.

\textbf{5.} $c$ croît vers le haut~; $\sin i/c$ constant fait croître $i$, de sorte que le
rayon s'incurve vers l'horizontale puis redescend~: le son est canalisé
le long du sol et porte loin. Au-dessus de l'eau l'air le plus bas est refroidi
par l'eau~: même effet.

\textbf{6.} Les rayons s'incurvent vers le haut, laissant une zone d'ombre. Rayon de
courbure $R = c/(\dd c/\dd z) = \qty{50}{km}$~: après \qty{3}{km} le rayon a
tourné de $3/50 = \qty{0.06}{rad} = \qty{3.4}{\degree}$ et s'est élevé de \qty{90}{m}~: l'auditeur
est en dessous.

\textbf{7.} L'absorption croît comme $f^2$~: les hautes fréquences du claquement ont
disparu après quelques kilomètres, le grondement grave subsiste.

\textbf{8.} $\mathcal P = \qty{2}{W}$~; $I = 2/2\pi r^2 = \qty{3.2e-3}{W/m^2}$ à \qty{10}{m}~:
\qty{95}{dB}.

\textbf{9.} $p_m = \sqrt{2 \times 410 \times 3.2 \times 10^{-3}} = \qty{1.6}{Pa}$~; $v_m = \qty{3.9}{mm/s}$~;
$\xi_m = v_m/2\pi f = \qty{6}{\micro m}$.

\textbf{10.} $+\qty{20}{dB}$~: \qty{115}{dB} à \qty{1}{m} --- nocif en quelques
minutes. \qty{60}{dB}~: $r = \sqrt{2/2\pi10^{-6}} = \qty{560}{m}$.

\textbf{11.} Dix incohérents~: $+\qty{10}{dB}$, \qty{105}{dB}. Deux en phase~: la
pression double, $+\qty{6}{dB}$, \qty{101}{dB}.

\textbf{12.} Bilan $\mathcal P = \alpha A\,ce/4$~: $e = 8/(0.3 \times 2000 \times 340) =
\qty{3.9e-5}{J/m^3}$, $E = eV = \qty{0.4}{J}$. Après l'arrêt, $\dd E/\dd t = -(\alpha Ac
/4V)E$~: $\tau = 4V/\alpha Ac = \qty{0.20}{s}$, et \qty{60}{dB} demandent $\tau\ln10^6 =
\qty{2.7}{s}$ (temps de réverbération de Sabine).

\textbf{13.} $\lambda = c/f$~: \qty{0.51}{mm}, \qty{0.31}{mm}, \qty{0.15}{mm}.

\textbf{14.} $2 \times 0.12/1540 = \qty{156}{\micro s}$~; \qty{6.4}{kHz}~; $200$ lignes en
\qty{31}{ms} (trente images par seconde).

\textbf{15.} Tissu--air~: $r = -0.9995$, $99.9\%$ de l'énergie réfléchie
--- d'où le gel, qui supprime le film d'air. Tissu--os~: $r = 0.63$,
$40\%$ de l'énergie~; le reste est absorbé par l'os, qui projette une
ombre. Graisse--muscle~: $r = 0.10$, $1\%$ de l'énergie~: les échos ténus dont
l'image est faite.

\textbf{16.} Fraction réfléchie $1.1\%$ ($-\qty{20}{dB}$)~; aller-retour de
\qty{4}{cm} à \qty{5}{MHz}~: $-\qty{10}{dB}$~; au total $-\qty{30}{dB}$, un millième.

\textbf{17.} Aller-retour de \qty{24}{cm}~: \qty{36}{dB}, \qty{60}{dB}, \qty{120}{dB}~: \qty{3}{MHz}
et \qty{5}{MHz} conviennent, \qty{10}{MHz} non --- les organes profonds sont imagés à
basse fréquence avec une résolution grossière, les organes superficiels à haute
fréquence avec finesse.

\textbf{18.} Cinq longueurs d'onde, soit \qty{1.5}{mm}~: deux interfaces plus proches
que \qty{0.8}{mm} environ (la moitié de l'impulsion) donnent des échos qui se recouvrent.

\textbf{19.} $2 \times 5 \times 10^6 \times 0.4 \times 0.5/1540 = \qty{1.3}{kHz}$~: un sifflement dont
la hauteur monte et descend à chaque battement de cœur.

\textbf{20.} $v = \qty{25}{m/s}$~: $700 \times 340/315 = \qty{756}{Hz}$, $700 \times 340/365 =
\qty{652}{Hz}$~; rapport $1.16$, soit $12\log_21.16 = 2.6$ demi-tons.

\textbf{21.} $700 \times 355/315 = \qty{789}{Hz}$.

\textbf{22.} Au plus court passage, lorsque la vitesse de l'ambulance est
perpendiculaire à la ligne de visée (rigoureusement, lorsque le son qui arrive
maintenant a été émis en ce point).

\textbf{23.} $\lambda' = (340 - 25)/700 = \qty{0.45}{m}$ (contre \qty{0.49}{m})~; le
son s'éloigne de l'ambulance à \qty{315}{m/s} devant et à \qty{365}{m/s}
derrière.

\textbf{24.} $v = c\,\Delta f/2f = 3 \times 10^8 \times 3200/4.8 \times 10^{10} = \qty{20}{m/s} =
\qty{72}{km/h}$.

\textbf{25.} \qty{340}{m/s}, \qty{1540}{m/s}, \qty{25}{m/s}, \qty{20}{m/s}, \qty{0.4}{m/s},
\qty{3e8}{m/s}~: $f' = f(c + v_o)/(c - v_s)$, appliquée une ou deux fois, et pour la
lumière à faible vitesse.
\end{solution}
