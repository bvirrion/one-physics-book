\begin{solution}{exo:b2:boltzmann-factor:1}
$H = k_BT/mg = \qty{8.4}{km}$~; \qty{0.70}{}, \qty{0.35}{}, \qty{0.30}{bar} (les
\qty{0.26}{} réels~: l'air se refroidit avec l'altitude)~; hélium \qty{61}{km}~; Mars

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:2}
Hydrogène~: $4\eu^{-395} \approx 10^{-171}$~; $4\eu^{-19.7} = 10^{-8}$~; $4\eu^{-11.8} = 3 \times
10^{-5}$. Sodium~: $\eu^{-9.8} = 6 \times 10^{-5}$. Rotation~: $\eu^{-0.04} = 0.96$, presque
égal. Spin nucléaire~: $1 - 4 \times 10^{-6}$ --- l'infime polarisation avec laquelle
travaille la résonance magnétique.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:3}

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:4}

$M_{\text{sat}} = n\mu = \qty{9.3e4}{A/m}$~; $\chi = \mu_0n\mu^2/k_BT = 2.6 \times 10^{-4}$.

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:5}
(a) \cref{prop:b2:boltzmann-factor:twolevel}. (b) $\eu^{-x} < 1$ pour $T > 0$~;
$N_2 > N_1$ exigerait $T < 0$ --- une \omterm{prop:b2:laser:gain}{inversion de population}. (c) $0$ et
$Nk_B\ln 2$. (d) $x = 1.93$~: $0.13$~; à \qty{77}{K}, $x = 7.5$~: $5 \times 10^{-4}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:6}
(a) Dérivée de $N\varepsilon/(\eu^x + 1)$. (b) $x = 2.40$, $C = 0.44\,Nk_B$.
(c) $\propto x^2\eu^{-x} \to 0$~: gelé~; $\propto 1/T^2 \to 0$~: les deux niveaux pleins,
plus rien à absorber. (d) $k_BT = 0.42\varepsilon$~: \qty{0.5}{mK}~; sous le
millikelvin les spins nucléaires détiennent l'essentiel de la capacité thermique et
ralentissent tout refroidissement.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:7}
(a) \cref{thm:b2:boltzmann-factor:equipartition}. (b) $\tfrac32R$,
$\tfrac52R$. (c) $C_{\text{vib}}/R = x^2\eu^x/(\eu^x - 1)^2$~: $0.002$, $0.41$, $0.90$~;
$C_V = 2.50R$, $2.91R$, $3.40R$. (d) $0.92$, $0.30$, $0.995$~; diamant $x = 4.3$~:

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:8}
(a), (b) \cref{thm:b2:boltzmann-factor:maxwell}. (c) $\tfrac32k_BT$. (d)
$\tfrac14n\langle v\rangle = \qty{2.9e27}{m^{-2}.s^{-1}}$~: $3 \times 10^{23}$ par centimètre

\begin{solution}{exo:b2:boltzmann-factor:9}
(a) Intégrer $f$ par parties~; le terme dominant est $(2/\sqrt\pi)x\eu^{-x^2}$. (b)
Hélium $x = 5.5$~: $5 \times 10^{-13}$~; azote $x = 14.5$~: $\eu^{-211}$. (c) $x =
4.9$~: $10^{-10}$ par temps de collision --- parti~; $10^{-17}$ demande $x \approx 6.6$,
$T \lesssim \qty{230}{K}$. (d) $x = 11$~: $\eu^{-121}$~; assez froid.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:10}
(a) $n = n_0\eu^{m\omega^2r^2/2k_BT}$. (b) $\eu^{\Delta m\omega^2r^2/2k_BT} = \eu^{0.24} =
1.27$. (c) $H = k_BT/m'g = \qty{16}{\micro m}$~; compter les sphères à
plusieurs hauteurs donne $k_B$, d'où $N_A = R/k_B$. (d) Les ions suivent
$\eu^{\mp e\varphi/k_BT}$ autour d'une charge~; linéarisé dans l'\omterm{prop:b2:maxwell-equations:potentials}{équation de Poisson}
cela donne $\varphi'' = \varphi/\lambda_{\text{D}}^2$~: le potentiel est écranté
au-delà de $\lambda_{\text{D}}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:11}
(a) $Z = 1/(1 - \eu^{-x})$, $\langle E\rangle = h\nu/(\eu^x - 1)$. (b) $k_BT$~:
équipartition~; $h\nu\eu^{-h\nu/k_BT}$~: gelé. (c) $u_\nu = (8\pi\nu^2/c^3)\langle E
\rangle$~; Rayleigh--Jeans donne $k_BT$ à chaque mode. (d) Un mode ne peut être
excité par moins d'un quantum.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:12}
(a) $x = \mu B/k_BT$ seulement~; $\ln 2$ et $0$. (b) $x = 0.67$~: $S/Nk_B = 0.51$, donc
$0.18\,k_B$ retiré par spin. (c) $B/T$ constant~: \qty{10}{mK}. (d) Le champ propre
des spins (le millitesla) remplace $B$ à la fin et plafonne le refroidissement~;
mesurer $M = n\mu\tanh(\mu B/k_BT)$ donne $T$.
\end{solution}

\begin{solution}{pb:b2:boltzmann-factor:1}
\textbf{1.} $\dd P/\dd z = -nmg = -(mg/k_BT)P$~; $H = \qty{8.4}{km}$.

\textbf{2.} $\eu^{-mgz/k_BT}$~: l'énergie potentielle d'une molécule, la
température de l'air.

\textbf{3.} \qty{0.70}{}, \qty{0.52}{}, \qty{0.35}{bar}~; $H\ln 2 = \qty{5.8}{km}$.

\textbf{4.} $P_0/g = \qty{1.03e4}{kg/m^2}$~; $\times 4\pi R^2$~: \qty{5.3e18}{kg}~; $\rho_0H =
1.22 \times 8400 = \qty{1.03e4}{kg/m^2}$.

\textbf{5.} $5.3 \times 10^{18}/(29 \times 10^{-3}/N_A) = 1.1 \times 10^{44}$.

\textbf{6.} $H_{\text{O}_2} = 7.6$, $H_{\text{N}_2} = \qty{8.7}{km}$~: le rapport à \qty{50}{km}
vaudrait $\eu^{-50/7.6 + 50/8.7} = 0.44$ de sa valeur au sol~; la turbulence mélange
plus vite que la diffusion ne sépare.

\textbf{7.} Plus basse~: un air plus froid est plus dense et la pression décroît plus vite.

\textbf{8.} \qty{2.5e25}{m^{-3}}~; $\eu^{-11.9}$~: \qty{1.7e20}{m^{-3}} --- pas de frontière, seulement une
convention.

\textbf{9.} $11.2\sqrt{6370/6870} = \qty{10.8}{km/s}$.

\textbf{10.} Hélium \qty{2040}{}, \qty{2300}{m/s}~; azote \qty{770}{}, \qty{870}{m/s}.

\textbf{11.} $x = 5.3$~: $4 \times 10^{-12}$~; azote $\eu^{-196}$~: aucun.

\textbf{12.} $\tfrac14 \times 10^{12} \times 2300 \times 4 \times 10^{-12} \times \tfrac12 \approx
10^3$ atomes par mètre carré et par seconde~; pour toute la Terre et sur un an,
$2 \times 10^{25}$ atomes --- \qty{0.1}{kg}.

\textbf{13.} $5 \times 10^{-6} \times 1.1 \times 10^{44} = 5.5 \times 10^{38}$ atomes, \qty{3.7e12}{kg}~;
face à \qty{0.1}{kg/an} le temps de séjour serait absurde.

\textbf{14.} Le régime permanent exige une perte de \qty{3e6}{kg/an}~: un temps
de séjour d'environ un million d'années, qui est la valeur admise --- donc
l'estimation thermique à \qty{1000}{K} est bien trop faible~: l'échappement est
dominé par des épisodes plus chauds et par des processus non thermiques (ioniques).

\textbf{15.} $x = 3.75$~: $\eu^{-14}$ au lieu de $\eu^{-28}$ --- un facteur $10^6$~:
la perte est extraordinairement sensible à la température exosphérique.

\textbf{16.} La Lune est chaude et légère~: tout s'échappe en
temps géologique~; Titan est froid, $x = 11$ pour l'azote.

\textbf{17.} $p_\pm = \eu^{\pm x}/2\cosh x$~; $M = n\mu\tanh x$.

\textbf{18.} $x = \mu B/k_BT$~: \qty{4.2}{A/m}, \qty{310}{A/m}, \qty{1.6e4}{A/m}
(saturation \qty{1.9e4}{A/m})~; Curie pour $x \ll 1$, c'est-à-dire $T \gg \qty{70}{mK}$.

\textbf{19.} $M$ ne dépend que de $T$ une fois $B$ connu~; la pente est la plus raide vers
$x \approx 1$, $T \approx \mu B/k_B = \qty{70}{mK}$~; on mesure la
susceptibilité avec une bobine.

\textbf{20.} $\langle E\rangle = -N\mu B\tanh x$~; $C = Nk_Bx^2/\cosh^2x$, culminant en
$x = 1.2$~: \qty{56}{mK}.

\textbf{21.} $Nk_B\ln 2$ et $0$~: tous les spins s'alignent sur le champ.

\textbf{22.} \qty{10}{mK}~; le champ propre des spins, de l'ordre du millitesla, remplace $B$
et fixe le plancher.

\textbf{23.} $x = 1$ en $\mu_{\text{N}}B/k_B = \qty{0.37}{mK}$~: Curie tient jusqu'au
millikelvin, et le thermomètre reste utile au microkelvin.

\textbf{24.} $\Delta\nu/\nu = \sqrt{8k_BT\ln2/mc^2} \propto \sqrt T$~: la largeur
d'une raie mesure la température du gaz.

\textbf{25.} Atmosphère, gaz qui s'échappe, spins~: $\eu^{-E/k_BT}$ avec $E = mgz$,
$\tfrac12mv^2$, $\mp\mu B$.
\end{solution}
