\section*{Chapitre \ref{ch:b2:fluid-kinematics} --- Cinématique des fluides}

\begin{solution}{exo:b2:fluid-kinematics:1}
(a) $\ell/L \sim 7 \times 10^{-8}$~: milieu continu. (b) $0.07$~: cas limite --- les
lois du milieu continu commencent à défaillir (glissement aux parois). (c) $\ell = 70 \times 10^{-9}
/3 \times 10^{-7} = \qty{0.23}{m}$, $\ell/L = 0.23$~: pas de milieu continu~; le gaz est
une pluie de molécules indépendantes (régime moléculaire libre).
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:2}
$a = v\,\dd v/\dd x = v_0^2(1 + x/L)/L$ (de $v_0^2/L$ à $2v_0^2/L$).
$\dd x/\dd t = v_0(1 + x/L)$~: $t = (L/v_0)\ln2 = 0.69\,L/v_0$, entre
$L/2v_0$ et $L/v_0$.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:3}
$D_V = \int_0^Rv_{\max}(1 - r^2/R^2)\,2\pi r\,\dd r = \tfrac12\pi R^2v_{\max}$~;
$\langle v\rangle = v_{\max}/2 = \qty{0.20}{m/s}$~; $D_V = \qty{1.6e-5}{m^3/s} =
\qty{16}{mL/s}$.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:4}
(a) $\operatorname{div} = 0$, $\vect\omega = \vect 0$~: incompressible,
irrotationnel. (b) $\operatorname{div} = 0$, $\omega_z = 2\Omega$~: rotation
en bloc. (c) $\operatorname{div} = 2a \ne 0$, irrotationnel (une source
plane). (d) $\operatorname{div} = 0$, $\omega_z = -k$~: cisaillement. (e)
$\operatorname{div} = b$~: incompressible si et seulement si $b = 0$~; irrotationnel pour
tout $b$.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:5}
$D_V = \qty{0.20}{L/s}$. Tuyau $S = \qty{1.13}{cm^2}$~: $v = \qty{1.8}{m/s}$~; lance
$S = \qty{7.1}{mm^2}$~: $v = \qty{28}{m/s}$. Accélération moyenne $\Delta(v^2/2)/L
= (28^2 - 1.8^2)/0.08 \approx \qty{1e4}{m/s^2} \approx 1000\,g$. Durée $\approx
\int\dd x/v$~: quelques millisecondes (\qty{4}{ms} pour un profil de vitesse linéaire).
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:6}
(a) $\dd y/\dd x = (V/U)\cos\omega t$~: des droites de pente
$(V/U)\cos\omega t$, toutes parallèles, qui basculent avec le temps. (b) $x = Ut$, $y =
(V/\omega)\sin\omega t$~: la sinusoïde $y = (V/\omega)\sin(\omega x/U)$. (c) À
$t = 0$ les \omterm{def:b2:fluid-kinematics:euler}{lignes de courant} sont à \qty{45}{\degree} et la \omterm{def:b2:fluid-kinematics:euler}{trajectoire} quitte
l'origine à \qty{45}{\degree}~; à $t = \pi/2\omega$ les \omterm{def:b2:fluid-kinematics:euler}{lignes de courant} sont
horizontales alors que la \omterm{def:b2:fluid-kinematics:euler}{trajectoire} reste la même sinusoïde. Elles diffèrent parce que
l'écoulement n'est pas stationnaire~: une \omterm{def:b2:fluid-kinematics:euler}{ligne de courant} est un instantané, une \omterm{def:b2:fluid-kinematics:euler}{trajectoire} une histoire.
(d) $\vect a = \partial_t\vect v = (0, -V\omega\sin\omega t)$ --- le champ est
uniforme, donc le terme convectif est nul.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:7}
(a) $D_V = 50 \times 2 \times 1.2 = \qty{120}{m^3/s}$. (b) $120/60 = \qty{2.0}{m/s}$.
(c) $150/150 = \qty{1.0}{m/s}$. (d) $\rho \propto 1/T$ à $P$ constant est divisée par deux~;
$\rho vS$ se conserve~: la vitesse double.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:8}
(a) $\omega_z = \frac1r\frac{\dd(rv_\theta)}{\dd r}$~: $2\Omega$ à l'intérieur, $0$
à l'extérieur. (b) À l'intérieur $\Gamma = 2\pi r\cdot\Omega r = 2\Omega\cdot\pi r^2$, le flux
de la \omterm{def:b2:fluid-kinematics:vorticity}{vorticité}~; à l'extérieur $\Gamma = 2\pi\Omega a^2$, constant, le flux
du cœur entier. (c) $\Omega = v_{\max}/a = \qty{1.4}{rad/s}$~; $\Gamma = 2\pi av_{\max}
= \qty{2.2e4}{m^2/s}$~; à \qty{500}{m}~: $\Omega a^2/r = \qty{7}{m/s}$. (d) Le
cœur tourne comme un solide~; à l'extérieur l'écoulement est irrotationnel~: les particules
font le tour de l'axe sans tourner sur elles-mêmes. La \omterm{def:b2:fluid-kinematics:vorticity}{vorticité} ---
la rotation --- réside dans le cœur.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:9}
(a) $\varphi = \tfrac12k(x^2 - y^2)$, $\Delta\varphi = k - k = 0$. (b) $\dd x/
kx = \dd y/(-ky)$~: $xy = $ const~; $\partial_y\psi = kx = v_x$, $-\partial_x\psi =
-ky = v_y$. (c) $\vect a = (v_x\partial_x + v_y\partial_y)\vect v = k^2(x, y) =
\operatorname{\vect{grad}}\bigl(\tfrac12k^2(x^2 + y^2)\bigr) = \operatorname{\vect{grad}}
(v^2/2)$~: dans un \omterm{def:b2:fluid-kinematics:euler}{écoulement stationnaire} irrotationnel l'accélération est le
gradient de l'énergie cinétique massique~; elle est dirigée à l'opposé du
point d'arrêt --- une particule ralentit en s'approchant de la paroi
selon $y$ et accélère en s'en éloignant selon $x$. (d) $\dot x = kx$, $\dot y = -ky$~:
$x = x_0\eu^{kt}$, $y = y_0\eu^{-kt}$ ($xy$ constant).
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:10}
(a) $\dd x/\dd t = x/(t + \tau)$ donne $x = x_0(1 + t/\tau)$, $\ddot x = 0$~;
\omterm{thm:b2:fluid-kinematics:material}{dérivée particulaire}~: $\partial_tv + v\partial_xv = -x/(t + \tau)^2 + x/(t + \tau)^2
= 0$. (b) $\partial_t\rho + \partial_x(\rho v) = \dot\rho + \rho/(t + \tau) = 0$~: $\rho =
\rho_0\tau/(t + \tau)$. (c) L'écart entre deux particules croît comme $(1 + t/\tau)$
et la masse comprise entre elles est fixe~: $\rho \propto 1/(1 + t/\tau)$. (d) $v =
Hx$ avec $H = 1/(t + \tau)$~: une loi de Hubble, expansion libre (sans
accélération), masse volumique décroissant comme l'inverse du facteur d'échelle.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:11}
(a) $\vect v = \operatorname{\vect{grad}}\varphi = (U + qx/2\pi r^2, qy/2\pi r^2)$~;
la source a $v_r = q/2\pi r$, $\operatorname{div} = \frac1r\dd(rv_r)/\dd r
= 0$ et aucune dépendance en $\theta$, donc $\omega_z = 0$~; des sommes de tels champs sont
encore incompressibles et irrotationnelles. (b) Sur le demi-axe $x$ négatif,
$U - q/2\pi r = 0$~: $x_s = -q/2\pi U$. (c) Au point d'arrêt $\theta =
\pi$, $y = 0$~: $\psi = q/2$~; la \omterm{def:b2:fluid-kinematics:euler}{ligne de courant} $Ur\sin\theta + q\theta/2\pi = q/2$,
soit $r\sin\theta = (q/2\pi U)(\pi - \theta)$ --- une courbe qui part de $x_s$ et
s'ouvre vers l'aval. (d) $\theta \to 0$~: $y \to q/2U$~; $\theta \to 2\pi$ (branche
inférieure)~: $y \to -q/2U$. Le fluide intérieur à cette ligne provient de la
source~; à l'extérieur, le courant uniforme contourne un demi-corps de
largeur asymptotique $q/U$.
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:12}
(a) $\operatorname{div}\vect v = \frac1r\partial_r(-q/2\pi) = 0$~; $\omega_z =
\frac1r\partial_r(\Gamma/2\pi) = 0$. (b) $\dd r/v_r = r\,\dd\theta/v_\theta$ donne
$\dd r/r = -(q/\Gamma)\dd\theta$. (c) $a_r = v_r\partial_rv_r - v_\theta^2/r =
-(q^2 + \Gamma^2)/4\pi^2r^3 = -v^2/r$~; $a_\theta = v_r\partial_rv_\theta + v_rv_\theta/r
= 0$. (d) $\dd r/\dd t = -q/2\pi r$~: $t = \pi(r_0^2 - a^2)/q$~; nombre de tours
$= (\Gamma/2\pi q)\ln(r_0/a)$. (e) $t = \qty{520}{s} \approx \qty{9}{min}$~; $10$
tours.
\end{solution}

\begin{solution}{pb:b2:fluid-kinematics:1}
\textbf{1.} À travers un cylindre de rayon $r$~: $2\pi rh|v_r| = D_V$, donc
$v_r = -q/2\pi r$, $q = D_V/h = \qty{1.5e-3}{m^2/s}$.

\textbf{2.} $rv_\theta = 0.5 \times 0.01 = \qty{5e-3}{m^2/s}$~: $v_\theta = \Gamma/2\pi r$
avec $\Gamma = \qty{3.1e-2}{m^2/s}$.

\textbf{3.} $\operatorname{div}\vect v = \frac1r\partial_r(rv_r) = 0$, $\omega_z =
\frac1r\partial_r(rv_\theta) = 0$.

\textbf{4.} $\dd r/r = -(q/\Gamma)\dd\theta$~: $r = r_0\eu^{-q\theta/\Gamma}$~; nombre de tours
$= (\Gamma/2\pi q)\ln(r_0/a) = 3.3 \times 3.2 \approx 11$.

\textbf{5.} $t = \pi(r_0^2 - a^2)/q = \qty{520}{s} \approx \qty{9}{min}$~; $v_\theta(a) =
\qty{0.25}{m/s}$.

\textbf{6.} $\vect a = -(v^2/r)\vect e_r$~; en $r = a$~: $v^2 = 0.25^2 + 0.012^2$,
$a = \qty{3.1}{m/s^2} \approx 0.3g$.

\textbf{7.} $\Omega_\oplus\sin\lambda\,r = 5 \times 10^{-5} \times 0.5 = \qty{2.5e-5}{m/s}$,
quatre cents fois moins que le brassage~: l'hémisphère ne décide
rien dans une baignoire ordinaire (il décide dans une cuve laissée immobile un jour entier).

\textbf{8.} $v_\theta = \Omega r$ ($r < a$), $\Omega a^2/r$ ($r > a$)~; $\Omega = 80/60
= \qty{1.33}{rad/s}$, $\Gamma = 2\pi av_{\max} = \qty{3.0e4}{m^2/s}$.

\textbf{9.} $\omega_z = 2\Omega = \qty{2.7}{s^{-1}}$ dans le cœur, $0$ à l'extérieur~;
$v_\theta$ croît linéairement jusqu'à \qty{80}{m/s} puis décroît en $1/r$~; $\omega_z$ est un
échelon.

\textbf{10.} $a = v_{\max}^2/a = 6400/60 = \qty{107}{m/s^2} \approx 11g$, vers l'axe.

\textbf{11.} À l'extérieur la \omterm{def:b2:fluid-kinematics:vorticity}{vorticité} est nulle~: un petit élément se translate
le long de son cercle sans tourner sur lui-même --- la roue à aubes
garde son orientation. Dans le cœur (rotation en bloc) elle tourne avec le
fluide, un tour par révolution.

\textbf{12.} $E_{\text{cœur}} = \int_0^a\tfrac12\rho\Omega^2r^2\,2\pi r\,\dd r =
\tfrac14\pi\rho\Omega^2a^4 = \qty{2.2e7}{J/m}$.

\textbf{13.} $E_{\text{ext}} = \int_a^R\tfrac12\rho(\Omega a^2/r)^2\,2\pi r\,\dd r =
\pi\rho\Omega^2a^4\ln(R/a) = 8.7 \times 10^7 \times 4.4 = \qty{3.8e8}{J/m}$~; l'intégrande
$\rho v^2r \propto 1/r$, d'où le logarithme.

\textbf{14.} $\qty{4.0e8}{J/m} \times \qty{1000}{m} = \qty{4e11}{J}$~: environ
\qty{100}{t} de TNT.

\textbf{15.} Afflux $2\pi a \times 300 \times 5 = \qty{5.7e5}{m^3/s}$ à travers
$\pi a^2 = \qty{1.1e4}{m^2}$~: $v_z = \qty{50}{m/s}$.

\textbf{16.} $\omega = 2v_{\max}/a = \qty{2.5e-3}{s^{-1}} = 50f$~; $\Gamma = 2\pi av_{\max}
= \qty{1.3e7}{m^2/s}$.

\textbf{17.} $r_0\cdot fr_0/2 = r(v_\theta + fr/2)$~: $v_\theta = f(r_0^2 - r^2)/2r$~;
\qty{60}{m/s} à \qty{100}{km}, \qty{155}{m/s} à \qty{40}{km}.

\textbf{18.} $f > 0$ dans l'hémisphère nord donne $v_\theta > 0$~:
sens trigonométrique vu de dessus (cyclonique). Sur \qty{500}{km} la
rotation planétaire est la rotation initiale dominante~; dans une baignoire c'est le
brassage.

\textbf{19.} Le frottement sur la mer enlève du moment cinétique, et le
gradient de pression ne peut plus équilibrer l'accélération centripète d'un
anneau toujours plus rapide~: l'air monte avant d'atteindre l'axe, dans le mur
de l'œil, laissant un œil calme.

\textbf{20.} $\rho\,2\pi rHv = 1.2 \times 2\pi \times 5 \times 10^5 \times 10^3 \times 5 =
\qty{1.9e10}{kg/s}$.

\textbf{21.} $\vect v(M, t) = -U\vect e_x + \vect v_0(M + Ut\,\vect e_x)$~: la
structure est advectée, le champ n'est pas stationnaire dans le référentiel du sol.
La translation s'ajoute au tourbillon du côté où $v_\theta$ pointe vers
l'ouest --- le côté nord (à droite) de la \omterm{def:b2:fluid-kinematics:euler}{trajectoire} vers l'ouest.

\textbf{22.} $0.05 \times 10^7/3600 = \qty{139}{m^3/s}$~; $v = 139/40 = \qty{3.5}{m/s}$.

\textbf{23.} $D_V = \qty{83}{cm^3/s}$~: aorte \qty{0.33}{m/s}~; capillaires
\qty{0.33}{mm/s}~; \qty{3}{s} par millimètre.

\textbf{24.} $S\,\dd h/\dd t = -s\sqrt{2gh}$~: $\sqrt h = \sqrt{h_0} -
\frac sS\sqrt{g/2}\,t$, vide en $T = (S/s)\sqrt{2h_0/g} = 10^4 \times 0.45 =
\qty{4500}{s} \approx \qty{75}{min}$.

\textbf{25.} Baignoire~: $r \sim \qty{0.5}{m}$, $v \sim \qty{1}{cm/s}$, $\Gamma =
\qty{3e-2}{m^2/s}$~; tornade~: \qty{60}{m}, \qty{80}{m/s}, $\omega = \qty{2.7}{s^{-1}}$,
$\Gamma = \qty{3e4}{m^2/s}$~; cyclone~: \qty{40}{km}, \qty{50}{m/s}, $\omega =
\qty{2.5e-3}{s^{-1}}$, $\Gamma = \qty{1.3e7}{m^2/s}$. Dans chacun, le moment
cinétique par unité de masse $rv_\theta$ du fluide qui converge se conserve~:
l'afflux les met en rotation.
\end{solution}
