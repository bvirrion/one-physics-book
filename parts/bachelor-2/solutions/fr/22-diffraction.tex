\begin{solution}{exo:b2:diffraction:4}

\begin{solution}{exo:b2:diffraction:5}
(a) $\int_{-b/2}^{b/2}\eu^{\iu kx\sin\theta}\dd x = b\,\operatorname{sinc}(kb\sin\theta/2)$. (b)
$\dd(\sin x/x)/\dd x = 0$ donne $\tan x = x$~; $x = 4.49 = 1.43\pi$, $\operatorname{sinc}^2 =
0.047$. (c) $0.903$. (d) La phase devient $kx(\sin\theta - \sin\theta_0)$~: la
même figure autour de $\theta_0$.
\end{solution}

\begin{solution}{exo:b2:diffraction:7}

\end{solution}

\begin{solution}{exo:b2:diffraction:8}
(a) $\sin\theta = \pm0.275$ (\qty{16}{\degree}), $\pm0.55$ (\qty{33}{\degree}). (b) $\sin\alpha \ge 0.275$.
(c) Ordres $0$, $\pm1$ seulement~: une image sinusoïdale de la bonne période, non
les barres nettes. (d) $\sin\alpha \ge \lambda/2d = 0.14$~: deux fois la résolution.
\end{solution}

\begin{solution}{exo:b2:diffraction:9}
(a) $2.44\lambda f/D = \qty{15}{\micro m}$~; trou \qty{23}{\micro m}. (b) $\lambda/\qty{50}{\micro m} =
\qty{1.3e-2}{rad}$~: \qty{0.25}{mm} hors de l'axe, bloquée. (c) $1\%$~; le trou ne garde que

\end{solution}

\begin{solution}{exo:b2:diffraction:10}
(a) Les amplitudes s'ajoutent en l'onde libre, nulle hors de l'axe~:
$|\underline s_{\text{fente}}| = |\underline s_{\text{bande}}|$ là. (b) Franges distantes de $\lambda L/b =
\qty{16}{mm}$~; mesurer l'interfrange donne $b$. (c) L'ombre de la bande se tient
dans le faisceau brillant non diffracté. (d) Des gouttelettes de diamètre $d$ diffractent
comme des trous~: anneaux en $1.22\lambda/d$, le rouge à l'extérieur~; le rayon donne $d$.
\end{solution}

\begin{solution}{exo:b2:diffraction:11}
(a) $|1 + \iu\varphi|^2 = 1 + \varphi^2$~: uniforme au premier ordre. (b) Sans la
lumière axiale, $|\iu\varphi|^2 = \varphi^2$~: second ordre, faible. (c) $|\iu + \iu\varphi|^2 =
1 + 2\varphi + \dots$~: linéaire. (d) La tache axiale éclipse la lumière
diffractée~; l'atténuer égalise les deux amplitudes et relève le contraste.
\end{solution}

\begin{solution}{exo:b2:diffraction:12}
(a) Demi-largeur $\lambda/W$~; divisée par $p/a\cos\theta$~: $\Delta\lambda = \lambda a\cos\theta/pW \approx
\lambda/pN$. (b) Le faisceau collimaté a le diamètre $Df_{\text{coll}}/f_{\text{tél}}$~;
un réseau plus petit gaspillerait lumière et pouvoir de résolution. (c) Une
résolution est toujours l'angle sur lequel la phase à travers
l'ouverture change d'une longueur d'onde, $\lambda/(\text{taille})$. (d) Sa largeur
d'image, qui domine dès qu'elle dépasse la largeur de \omterm{def:b2:diffraction:huygens}{diffraction} du
réseau --- le cas usuel.
\end{solution}

\begin{solution}{pb:b2:diffraction:1}
\textbf{1.} $1.22\lambda/D = \qty{2.2e-4}{rad}$~; $\times\qty{17}{mm} = \qty{3.8}{\micro m}$~: deux espacements

\textbf{3.} \qty{9.6e-5}{rad}~; les aberrations du cristallin à pleine ouverture et les
bâtonnets plus grossiers, regroupés, annulent le gain.

\textbf{4.} $1.5/2.2 \times 10^{-4} = \qty{7}{km}$.

\textbf{5.} La \omterm{def:b2:diffraction:huygens}{diffraction} s'aggrave jusqu'à \qty{6.7e-4}{rad}, mais le cercle de flou
d'un œil défocalisé rétrécit avec l'ouverture~: la profondeur de champ l'emporte.

\textbf{6.} \qty{6.7e-7}{rad} (\qty{0.14}{''})~; $\times\qty{10}{m} = \qty{6.7}{\micro m}$ de rayon~: un

\textbf{8.} \qty{2.7e-6}{rad} (\qty{0.55}{''})~; les erreurs de phase de l'atmosphère sont
une plus petite fraction d'une longueur d'onde plus grande~: moins de corrections, plus lentes.

\textbf{9.} $0.15 < 0.55$~: non~; à \qty{550}{nm} depuis l'espace, \qty{0.14}{''}~: tout juste.

\textbf{11.} Une croix de quatre aigrettes de \omterm{def:b2:diffraction:huygens}{diffraction}.
