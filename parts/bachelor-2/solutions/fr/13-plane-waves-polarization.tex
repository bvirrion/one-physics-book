\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:4}
$\operatorname{div}\vect E = \partial_xE_x = 0$, $\operatorname{div}\vect B = 0$~;
$\operatorname{\vect{curl}}\vect E = E_0k\sin(\omega t - kz)\,\vect e_y = -\partial_t\vect B$ si et seulement si $\omega =
kc$ --- Faraday fixe à la fois la direction ($\vect e_y$) et la norme
($E_0/c$) de $\vect B$~; $\operatorname{\vect{curl}}\vect B = -(E_0k/c)\sin(\omega t - kz)\,\vect e_x
= \mu_0\varepsilon_0\partial_t\vect E$. Un $\vect E$ selon $\vect e_z$ aurait $\operatorname{div}
\vect E \ne 0$ sans charge~: impossible --- l'onde est transversale.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:5}
(a) $\vect E = E_0(\cos(\omega t - kz), \pm\sin(\omega t - kz), 0)$~: module $E_0$,
angle $\pm(\omega t - kz)$. (b) $(\cos, \sin) + (\cos, -\sin) = (2\cos, 0)$~: rectiligne
selon $x$~; une phase relative $\varphi$ tourne la direction de $\varphi/2$. (c)
$I_0/2$ sous tout angle. (d) Une \omterm{prop:b2:plane-waves-polarization:plates}{lame quart d'onde} transforme la lumière circulaire
en lumière rectiligne, qu'un polariseur éteint ensuite~; la \omterm{def:b2:plane-waves-polarization:states}{lumière naturelle}
reste non polarisée.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:6}
(a) $e = \lambda/4(n_s - n_f) = \qty{16}{\micro m}$~; demi-onde \qty{32}{\micro m}. (b) $\delta =
(\pi/2)(589/450) = \qty{2.1}{rad}$~: lumière elliptique. (c) $\delta \propto e/\lambda$~: une
lame de $m + \tfrac14$ ondes a des $\dd\delta/\dd\lambda$ et $\dd\delta/\dd T$ plus grands d'un facteur
$4m + 1$. (d) $\delta \propto e$~: un micromètre sur \qty{16}{\micro m} fait $6\%$ de $\pi/2$.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:7}
(a) La composante lente change de signe~: $(\cos\alpha, \sin\alpha) \to (\cos\alpha,
-\sin\alpha)$. (b) Une polarisation à l'angle $\alpha$ d'une lame placée à $\beta$ ressort
à $2\beta - \alpha$~: rotation $2\beta$~; \qty{45}{\degree}. (c) Elle inverse le
sens. (d) $I = I_0\sin^22\beta$~: quatre maxima et quatre extinctions
par tour.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:8}
$\cos^{2N}(\pi/2N)$~: $0$, $0.25$, $0.61$, $0.78$, $0.976$, $\to 1$. Un
milieu qui tourne continûment (pouvoir rotatoire, cellule à cristaux liquides)
fait tourner la polarisation sans perte.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:9}
(a) $I_{\max} = I_n/2 + I_p$, $I_{\min} = I_n/2$~; $p = I_p/(I_n + I_p) = (I_{\max} -
I_{\min})/(I_{\max} + I_{\min})$. (b) $p = 4/6 = 0.67$. (c) Le polariseur croisé
laisse passer $I_n/2 = 0.05I$~: $95\%$ supprimés. (d) Non --- les deux donnent une intensité
constante~; il faut une \omterm{prop:b2:plane-waves-polarization:plates}{lame quart d'onde}.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:10}
(a) Phases $\omega t - k(z\cos\theta \pm x\sin\theta)$~: somme $2E_0\cos(kx\sin\theta)\cos(\omega t
- kz\cos\theta)\,\vect e_y$. (b) $\Delta x = \lambda/2\sin\theta$~: \qty{3.6}{\micro m} à \qty{5}{\degree},
$\lambda/2 = \qty{316}{nm}$ à \qty{90}{\degree}. (c) $v_\varphi = c/\cos\theta > c$~: c'est la
phase d'un motif~; l'énergie va selon $z$ à $c\cos\theta$ (et chaque onde à $c$).
(d) $B_z = (\sin\theta/c)(E_1 - E_2)$, $E_y = E_1 + E_2$~: $\langle\Pi_x\rangle \propto
\langle E_1^2 - E_2^2\rangle = 0$~; $\Pi_z \propto \cos\theta(E_1 + E_2)^2$, maximal dans les
plans brillants.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:11}
(a) Sans tension~: tournée de \qty{90}{\degree}, elle traverse le polariseur croisé~:
brillant~; sous tension~: sombre. (b) Fuite $\sin^25^\circ = 7.6 \times 10^{-3}$~: contraste
$130$. (c) L'hélice agit comme un empilement de retardateurs dont $\delta \propto 1/\lambda$~:
exacte à \qty{550}{nm}, la rotation est fausse aux extrémités du spectre,
qui fuient --- l'état sombre paraît violacé. (d) Lorsque le polariseur
extérieur est croisé avec le polariseur de sortie de l'écran~; l'angle est
l'axe de polarisation de l'écran.
\end{solution}

\begin{solution}{exo:b2:plane-waves-polarization:12}
(a) $\vect e_x = \tfrac12[(\vect e_x + \iu\vect e_y) + (\vect e_x - \iu\vect e_y)]$~; après $\ell$
les deux composantes ont les phases $k_L\ell$ et $k_R\ell$, et se recombinent en une
lumière rectiligne à l'angle $(k_L - k_R)\ell/2 = \pi(n_L - n_R)\ell/\lambda$. (b) $n_L -
n_R = \theta\lambda/\pi\ell = 0.116 \times 589 \times 10^{-9}/0.314 = \qty{2.2e-7}{}$. (c)
$0.01/6.65 \times 0.1 = \qty{1.5e-4}{g/mL}$. (d) La rotation de Faraday est fixée par
$\vect B$, non par le sens de propagation~: au retour elle s'ajoute
(\qty{45}{\degree} $+$ \qty{45}{\degree} $=$ \qty{90}{\degree}), et le polariseur d'entrée bloque
la lumière de retour --- c'est un isolateur.
\end{solution}

\begin{solution}{pb:b2:plane-waves-polarization:1}
\textbf{1.} $f = \qty{4.74e14}{Hz}$, $k = \qty{9.9e6}{rad/m}$, $hf = \qty{3.1e-19}{J} =
\qty{1.96}{eV}$~; $3.2 \times 10^{16}$ photons par seconde.

\textbf{2.} $I = 0.01/\pi(0.5 \times 10^{-3})^2 = \qty{1.3e4}{W/m^2}$~; $E_0 = \qty{3.1}{kV/m}$,
$B_0 = \qty{10}{\micro T}$~; $\vect E = E_0\cos(\omega t - kz)\vect e_x$, $\vect B = (E_0/c)\cos(\omega t
- kz)\vect e_y$.

\textbf{3.} $u = I/c = \qty{4.2e-5}{J/m^3}$~; $P/c = \qty{3.3e-11}{J}$ par mètre de
faisceau~; \qty{30}{cm}, soit $4.7 \times 10^5$ longueurs d'onde.

\textbf{4.} $\langle\Pi\rangle = I$ selon $z$, oscillant en $\cos^2$ à $2f$~;
$P/c = \qty{3.3e-11}{N}$ sur une cible noire, \qty{6.7e-11}{N} sur un miroir.

\textbf{5.} $E_0/5 \times 10^{11} = 6 \times 10^{-9}$~; $B_0$ vaut le cinquième du champ
terrestre.

\textbf{6.} $I/c = \qty{4.2e-5}{Pa}$.

\textbf{7.} $v_0 = eE_0/m\omega = \qty{0.18}{m/s}$~; $F_m/F_e = v_0B_0/E_0 = v_0/c = 6 \times
10^{-10}$~: la force magnétique sur des charges lentes est négligeable.

\textbf{8.} $\vect E = E_0[\cos(\omega t - kz)\vect e_x \pm \sin(\omega t - kz)\vect e_y]$ avec
$E_0 = \qty{2.2}{kV/m}$ (même puissance), $\vect B = \vect e_z\wedge\vect E/c$~: l'intensité
est inchangée mais $\Pi = \varepsilon_0cE_0^2$ est désormais \emph{constant} dans le
temps --- le flux d'énergie ne bat plus.

\textbf{9.} $10\cos^230^\circ = \qty{7.5}{mW}$~; $0$~; $7.5 \times \cos^445^\circ = \qty{1.9}{mW}$.

\textbf{10.} Brillant~: $\tfrac12$~; sombre~: $0$.

après le retardateur, une phase $\delta$ entre elles~; la projection sur la direction
croisée vaut $\propto(1 - \eu^{\iu\delta})/2$~: $I \propto \sin^2(\delta/2)$.

\textbf{12.} Noir~; moitié de l'intensité.

\textbf{13.} Une sortie circulaire paraît également brillante à travers des lunettes
de soleil polarisantes, quel que soit l'angle de la tête.

\textbf{14.} $\tfrac12 \times 0.85 \times 0.85 \times \tfrac13 = 12\%$~; le premier polariseur

\textbf{15.} Brillant $0.36$, sombre $\tfrac12 \times 10^{-3}$~: contraste $\approx 700$.

\textbf{16.} Une lumière circulaire, quelle que soit la rotation du filtre~: la
lame séparatrice sensible à la polarisation reçoit la même puissance quelle que soit l'orientation du
filtre.

\textbf{17.} La \omterm{def:b2:plane-waves-polarization:states}{polarisation circulaire} garde son sens quand la tête
