\section*{Chapter \ref{ch:b2:dispersion-wave-packets} --- Dispersion and Wave Packets}

\begin{solution}{exo:b2:dispersion-wave-packets:1}
(a) $k = \qty{0.063}{rad/m}$, $\omega = \sqrt{gk} = \qty{0.79}{rad/s}$, $T = \qty{8.0}{s}$;
$v_\varphi = \qty{12.5}{m/s}$, $v_g = \qty{6.3}{m/s}$. (b) $\omega = \qty{0.45}{rad/s}$, $k = \omega^2
/g = \qty{0.021}{rad/m}$, $\lambda = \qty{310}{m}$; $v_\varphi = gT/2\pi = \qty{22}{m/s}$, $v_g =
\qty{11}{m/s}$; $5 \times 10^6/11 = \qty{4.6e5}{s} = 5.3$ days. (c) $\lambda = \qty{6}{m}$, $v_g
= \qty{1.6}{m/s}$: it would need five weeks, and short waves are damped
by \omterm{def:b2:viscous-flows:viscosity}{viscosity} and broken by the wind long before.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:2}
$\omega = \qty{1.26e8}{rad/s}$, $\omega_c = \qty{6.3e7}{rad/s}$: $k = \sqrt{\omega^2 - \omega_c^2}/c =
\qty{0.36}{rad/m}$, $v_\varphi = \qty{3.5e8}{m/s}$, $v_g = c^2k/\omega = \qty{2.6e8}{m/s}$,
product $c^2$. At \qty{5}{MHz}, $\omega < \omega_c$: $\kappa = \sqrt{\omega_c^2 - \omega^2}/c =
\qty{0.18}{m^{-1}}$, depth \qty{5.5}{m}.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:3}
Beat at \qty{4}{Hz}, loudness period \qty{0.25}{s}. Water: $f = 0.125$ and
\qty{0.119}{Hz}; envelope $\cos(\delta\omega t - \delta kx)$ with $\delta k = \qty{2.9e-3}{rad/m}$
(beats \qty{1.1}{km} apart) and speed $\delta\omega/\delta k = \qty{6.4}{m/s}$ --- the
\omterm{prop:b2:dispersion-wave-packets:group}{group velocity} at \qty{105}{m}, $\tfrac12\sqrt{g\lambda/2\pi} = \qty{6.4}{m/s}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:4}
$\ln(b/a) = 1.20$: $\Lambda = \qty{0.24}{\micro H/m}$, $\Gamma = \qty{106}{pF/m}$, $c = 1/
\sqrt{\Lambda\Gamma} = \qty{2.0e8}{m/s}$, $Z_c = \qty{48}{\ohm}$, \qty{5.0}{ns/m}; \qty{10}{ns}
$\leftrightarrow$ \qty{2.0}{m}. Air: $\Gamma = \qty{46}{pF/m}$, $c = c_0$, $Z_c = \qty{72}{\ohm}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:5}
(a) $\omega^2 = c^2k^2 + (EI/\mu)k^4$; $v_\varphi = \sqrt{c^2 + (EI/\mu)k^2}$; $v_g = (c^2k +
2(EI/\mu)k^3)/\omega$. (b) $c^2 = \qty{1.12e5}{m^2/s^2}$, $k = \qty{66}{rad/m}$, $(EI/\mu)k^2
= \qty{7.0e3}{m^2/s^2}$: $v_\varphi/c = \sqrt{1.063} = 1.031$, $+3.1\%$ --- the $53$ cents of
\cref{ch:b2:waves-on-strings}. (c) $v_g > v_\varphi$: anomalous; the high
harmonics run ahead and the attack of a pluck turns into a brief
descending chirp.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:6}
(a) $-\mu\omega^2 + \iu\alpha\omega = -Tk^2$: $\underline k^2 = (\omega^2/c^2)(1 - \iu\alpha/\mu\omega)$.
(b) $\underline k \approx (\omega/c)(1 - \iu\alpha/2\mu\omega)$: $k' = \omega/c$, $k'' = \alpha/2\mu c =
\alpha/2Z$; $8.7k''$ dB/m. (c) $c = \qty{100}{m/s}$, $Z = \qty{0.5}{kg/s}$, $\mu\omega = 3.1 \gg
0.02$: $k'' = \qty{0.02}{m^{-1}}$, $\delta = \qty{50}{m}$, \qty{0.17}{dB/m}. (d) Not in this
limit ($k' = \omega/c$); when $\alpha \gtrsim \mu\omega$ the square root no longer
linearizes and $k'$ depends non-linearly on $\omega$: dispersive (the
diffusive limit of the cable).
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:7}
(a) $v_\varphi = \sqrt{\gamma k/\rho}$, $v_g = \tfrac32v_\varphi$: the group outruns the
crests. (b) $v_\varphi^2 = g/k + \gamma k/\rho$ is minimal at $k_m = \sqrt{\rho g/\gamma}$:
$\lambda_m = 2\pi\sqrt{\gamma/\rho g} = \qty{1.7}{cm}$, $v_{\min} = (4g\gamma/\rho)^{1/4} =
\qty{23}{cm/s}$. (c) A wake is a pattern stationary with respect to the
boat, made of waves whose \omterm{def:b2:dispersion-wave-packets:relation}{phase velocity} matches its speed; none is
slower than \qty{23}{cm/s}. (d) Capillary regime, $v_g > v_\varphi$: crests are
born at the front of the ring and die at its back.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:8}
(a) $k = \qty{0.126}{rad/m}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-18}{m^2/s}$: $t_{\text{sp}}
= 2.5 \times 10^5/18 = \qty{1.4e4}{s}$, four hours. (b) \qty{10}{ps}: $t_{\text{sp}} =
4 \times 10^{-6}/0.16 = \qty{25}{\micro s}$, \qty{5}{km} at \qty{2e8}{m/s}; \qty{1}{ns}: $0.04/0.16 =
\qty{0.25}{s}$, \qty{50000}{km}. Short pulses (high bit rates) spread within
kilometres: long links use compensating fibre or moderate rates.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:9}
(a) $\partial_x^2v = R\Gamma\,\partial_tv$, $D = 1/R\Gamma = \qty{1.1e12}{m^2/s}$. (b) $\underline k^2
= -\iu R\Gamma\omega$: $k' = k'' = \sqrt{R\Gamma\omega/2}$; \qty{1}{Hz}: $k = \qty{1.7e-6}{m^{-1}}$,
$v_\varphi = \qty{3.7e6}{m/s}$, $\delta = \qty{600}{km}$; \qty{10}{Hz}: $v_\varphi = \qty{1.2e7}{m/s}$,
$\delta = \qty{190}{km}$. (c) $R\Gamma L^2 = 9 \times 10^{-13} \times 9 \times 10^{12} = \qty{8}{s}$
per dot: about a word a minute. (d) $\Lambda = R\Gamma/G = \qty{9}{H/km}$ --- an
absurd inductance for a submarine cable; loading coils worked on land
telephone lines, whose $G$ and $R\Gamma$ were different.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:10}
(a) $\partial_te = \dot y(\mu\ddot y + Ky) + Ty'\dot y' = \dot yTy'' + Ty'\dot y' = \partial_x
(Ty'\dot y) = -\partial_x\mathcal P$. (b) $\langle e\rangle = \tfrac14A^2(\mu\omega^2 + Tk^2 + K) =
\tfrac12\mu\omega^2A^2$ (using $\mu\omega^2 = Tk^2 + K$); $\langle\mathcal P\rangle = \tfrac12Tk\omega
A^2$. (c) Ratio $Tk/\mu\omega = c^2k/\omega = v_g$. (d) $y = A\eu^{-\kappa x}\cos\omega t$:
$\mathcal P \propto \sin2\omega t$, zero mean --- no energy crosses the medium.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:11}
(a) $\tanh kh \to 1$: $\omega^2 = gk$; $kh \ll 1$: $\omega = \sqrt{gh}\,k$, speed $\sqrt{gh}$,
non-dispersive. (b) $kh = 0.13$: shallow; $c = \sqrt{9.81 \times 4000} = \qty{198}{m/s}$,
$T = \qty{17}{min}$; \qty{8000}{km} in \qty{11}{h}. (c) $\tfrac12\rho gA^2 = \qty{1.2}{kJ/m^2}$
over $200 \times 1000$ km$^2$: \qty{2.5e14}{J}. Green: $A \propto h^{-1/4}$, $(4000/10)^{1/4}
= 4.5$: \qty{2.2}{m}, before steepening and run-up. (d) At $h \approx \lambda/2 =
\qty{50}{m}$; the part of a crest in shallower water slows down, so the
crest swings round parallel to the depth contours --- refraction.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:12}
(a) At $x = \ell$: $v = v_i + v_r$, $i = (v_i - v_r)/Z_c$, $v = R_Li$: $r = (R_L -
Z_c)/(R_L + Z_c)$. (b) $0$, $-1$, $+1$, $0.5$. (c) The line's input looks like
$Z_c$, so the scope sees \qty{0.5}{V} at $t = 0$, then the echo $0.5r$ V at
$2\ell/c = \qty{1}{\micro s}$: nothing, \qty{-0.5}{V}, \qty{+0.5}{V}, \qty{+0.25}{V}; the
matched generator absorbs the echo, so there is no second one. (d)
$\tfrac12 \times 2 \times 10^8 \times 1.8 \times 10^{-6} = \qty{180}{m}$.
\end{solution}

\begin{solution}{pb:b2:dispersion-wave-packets:1}
\textbf{1.} $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi = -\Gamma\partial_tv$; with
$\eu^{\iu(\omega t - kx)}$: $\iu kv = (\iu\omega\Lambda + R)i$, $\iu ki = \iu\omega\Gamma v$, hence
$k^2 = \omega\Gamma(\omega\Lambda - \iu R)$.

\textbf{2.} $\omega = R/\Lambda = \qty{4000}{rad/s}$, \qty{640}{Hz}: at a few hertz $\Lambda\omega
\ll R$, diffusive.

\textbf{3.} $k' = k'' = \sqrt{R\Gamma\omega/2}$ with $R\Gamma = \qty{5e-13}{s/m^2}$: \qty{1}{Hz}:
$\qty{1.25e-6}{m^{-1}}$, $v_\varphi = \qty{5e6}{m/s}$, $\delta = \qty{800}{km}$; \qty{4}{Hz}:
$\qty{2.5e-6}{m^{-1}}$, $\qty{1e7}{m/s}$, \qty{400}{km}.

\textbf{4.} $8.7L/\delta$: \qty{38}{dB} at \qty{1}{Hz}, \qty{76}{dB} at \qty{4}{Hz}: only the
slowest components survive; a sharp dot arrives as a long, low hump.

\textbf{5.} $R\Gamma L^2 = 5 \times 10^{-13} \times 1.2 \times 10^{13} = \qty{6}{s}$: a dot every
six seconds, a word a minute (the real cable, with sensitive mirror
galvanometers, managed a few).

\textbf{6.} $\Lambda = R/\omega = \qty{80}{mH/km}$. Then $k^2 = \Gamma R\omega(1 - \iu)$, $k = \sqrt{\Gamma R\omega}
\,2^{1/4}\eu^{-\iu\pi/8}$: $k' = \qty{3.9e-6}{m^{-1}}$, $k'' = \qty{1.6e-6}{m^{-1}}$: $v_\varphi =
\qty{6.4e6}{m/s}$, $\delta = \qty{620}{km}$, \qty{49}{dB} over the cable instead of
\qty{76}{dB}.

\textbf{7.} A dot every \qty{6}{s} is about \qty{0.2}{bit/s}: the fibre is
$10^{10}$ to $10^{11}$ times faster.

\textbf{8.} $v_\varphi = gT/2\pi$, $v_g = gT/4\pi$.

\textbf{9.} \qty{18}{s}: $v_g = \qty{14.1}{m/s}$, $4 \times 10^6/14.1 = \qty{3.3}{days}$;
\qty{6}{s}: \qty{4.7}{m/s}, \qty{9.9}{days}: the swell lasts more than six days.

\textbf{10.} $t = D/v_g = 4\pi D/gT$, so $1/T = gt/4\pi D$. $\Delta(1/T) = 1/12 - 1/16
= \qty{0.0208}{Hz}$ in \qty{86400}{s}: $g/4\pi D = \qty{2.4e-7}{s^{-2}}$, $D = \qty{3200}{km}$.

\textbf{11.} $\tfrac12\rho gA^2 = \qty{11}{kJ/m^2}$; the energy moves at $v_g = gT/4\pi =
\qty{9.4}{m/s}$: \qty{103}{kW} per metre of crest.

\textbf{12.} $10^3 \times 103 \times 0.3 = \qty{31}{MW}$: ten wind turbines.

\textbf{13.} $\lambda = 2\pi U^2/g = \qty{16}{m}$. The energy of each wave component
travels at half its phase speed, so the pattern is confined to a wedge
behind the boat (half-angle \qty{19.5}{\degree}, whatever its speed).

\textbf{14.} A crest is born at the rear of the group, overtakes it
(crests move at $2v_g$ relative to the water, $v_g$ relative to the
group) and dies at the front. The group is $120 \times 9.4 = \qty{1.1}{km}$
long, five wavelengths ($\lambda = gT^2/2\pi = \qty{225}{m}$); ten crests pass the
buoy, each having lived about two minutes.

\textbf{15.} $kh = 0.13$: shallow. $c = \qty{198}{m/s}$, $T = \lambda/c = \qty{17}{min}$,
\qty{8.4}{h}.

\textbf{16.} Non-dispersive: all components at \qty{198}{m/s}, arriving
together as one pulse (the swell, at $\sim\qty{10}{m/s}$, sorted by period,
would take a week).

\textbf{17.} $\tfrac12\rho gA^2 = \qty{1.8}{kJ/m^2}$ over $500 \times 200$ km$^2$: \qty{1.8e14}{J},
$4\%$ of a megatonne.

\textbf{18.} $A = 0.6(4000/20)^{1/4} = \qty{2.3}{m}$; $c = \sqrt{gh} = \qty{14}{m/s}$, $\lambda =
cT = \qty{14}{km}$. The crest, in deeper water than the trough ahead of it,
travels faster and catches up: the front steepens into a bore.

\textbf{19.} A rise of \qty{0.6}{m} over a quarter of an hour, on a slope
of $10^{-5}$: nothing to feel.

\textbf{20.} The mean speed, $\sqrt{gh}$ for the mean depth; since
$\sqrt h$ is concave, shallow stretches slow the wave more than deep
ones speed it up, and the path average is below $\sqrt{g\bar h}$.

\textbf{21.} $\omega' = \tfrac12\sqrt{g/k}$, $\omega'' = -\tfrac14\sqrt{g/k^3}$; $k = \qty{0.063}{rad/m}$:
$\omega'' = \qty{-50}{m^2/s}$.

\textbf{22.} $t_{\text{sp}} = 10^6/50 = \qty{2e4}{s}$, \qty{5.6}{h}; the packet has
moved $v_gt = 6.25 \times 2 \times 10^4 = \qty{125}{km}$.

\textbf{23.} A long train contains a narrow band of $k$, whose group
velocities nearly coincide: the components stay together longer. The
tsunami's medium is non-dispersive: equal speeds, no spreading.

\textbf{24.} $\omega'' = \hbar/m = \qty{1.2e-4}{m^2/s}$: $t_{\text{sp}} = 10^{-18}/1.2 \times
10^{-4} = \qty{9e-15}{s}$; $v_g = \hbar k/m = \qty{7.3e6}{m/s}$: \qty{60}{nm}.

\textbf{25.} Cable: $k^2 = -\iu R\Gamma\omega$, $v_\varphi = \sqrt{2\omega/R\Gamma}$, no
clean $v_g$, attenuates and smears. Swell: $\omega^2 = gk$, $v_\varphi = 2v_g$,
spreads. Tsunami: $\omega = \sqrt{gh}\,k$, $v_\varphi = v_g$, neither. Electron:
$\omega = \hbar k^2/2m$, $v_g = 2v_\varphi$, spreads.
\end{solution}
