\begin{solution}{exo:b2:charges-currents-conduction:5}
(a) Arus masuk, tak ada yang keluar: $\dd Q/\dd t = I$. (b) Muatannya menumpuk: $\partial_t
\rho \ne 0$. (c) $\vect j = (I/4\pi r^2)\vect e_r$: $\operatorname{div}\vect j = \frac1{r^2}
\partial_r(I/4\pi) = 0$. (d) Hanya di elektrodenya, tempat arusnya
disuntikkan dari kawatnya --- sebuah sumber fluks.
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:6}
(a) $j = I/2\pi rL$, $E = j/\gamma$, radial. (b) $U = \int_a^bE\,\dd r = I\ln(b/a)/

(c) \qty{0.5}{\micro A}, \qty{0.5}{mW}. (d) $RC = \varepsilon_0\varepsilon_r/\gamma = \qty{200}{s}$ bagi
$\varepsilon_r = 2.3$: kabelnya, yang terisi dan terkucil, melucut lewat
isolatornya sendiri dengan waktu relaksasi isolator itu.
\end{solution}

\begin{solution}{exo:b2:charges-currents-conduction:7}

1/\sqrt{1 + \omega^2\tau^2}$: $1$, $1$, $0.21$, $0.013$; fasenya $-\arctan\omega\tau$: $0$,

\end{solution}

\begin{solution}{pb:b2:charges-currents-conduction:1}
\textbf{1.} $n = \rho N_A/M = \qty{8.4e28}{m^{-3}}$.

\textbf{2.} $j = \qty{6.4e6}{A/m^2}$, $v = j/ne = \qty{0.47}{mm/s}$; \qty{35}{min} tiap meter.

\textbf{3.} $\tau = m\gamma/ne^2 = \qty{2.5e-14}{s}$, $\mu = \qty{4.4e-3}{m^2/(V.s)}$, $\ell =
\qty{40}{nm}$, $150$ jarak atom: elektronnya bertumbukan dengan cacat dan
getaran termalnya, bukan dengan atomnya.

\textbf{4.} $\rho_e \propto T$ memberi $\alpha = 1/293 = \qty{3.4e-3}{K^{-1}}$, dekat dengan
$4 \times 10^{-3}$ yang terukur.

\textbf{5.} $E = \qty{0.11}{V/m}$, $U = \qty{0.11}{V}$; $\tfrac12mv^2 = \qty{1e-37}{J}$ melawan
$k_BT = \qty{4e-21}{J}$: gas elektronnya tak menyadari arusnya.

\textbf{6.} $I/e = \qty{1e20}{}$ elektron tiap sekon.

\textbf{7.} $Q = neSL = \qty{3.4e4}{C}$; $QE = \qty{3.6}{kN}$, melawan berat
\qty{0.2}{N}. Medannya menarik kisi positifnya dengan gaya yang
berlawanan; elektronnya menyerahkan momentumnya kepada kisinya lewat tumbukan:
kawatnya sebagai keseluruhan tak merasakan apa pun.

\textbf{8.} $p = j^2/\gamma = \qty{6.8e5}{W/m^3}$, \qty{1.7}{W/m}.

\textbf{9.} Permukaannya $\pi d = \qty{5.6e-3}{m^2}$ tiap meter: $\Delta T = 1.7/(10 \times 5.6 \times
10^{-3}) = \qty{30}{K}$; hambatan jenisnya naik $12\%$, dayanya menjadi \qty{1.9}{W}:
$\Delta T \approx \qty{35}{K}$.

\textbf{10.} $\Delta T = 1.7(1 + 0.004\Delta T)/(2 \times 5.6 \times 10^{-3})$: $\Delta T(1 - 0.6) = 150$,
$\Delta T \approx \qty{400}{K}$ --- dekat pelarian termal: kabel yang terkubur harus
diturunkan alanya.

\textbf{11.} $p/\rho c = \qty{0.2}{K/s}$; pada \qty{160}{A}, \qty{20}{K/s}: mencair dalam
sekitar semenit (lebih cepat ketika $\rho_e$ naik); pemutus arusnya membuka dalam milisekon.

\textbf{12.} $\omega\tau = 8 \times 10^{-12}$: hukum Ohmnya tak tersentuh.

\textbf{13.} $\varepsilon_0/\gamma = \qty{1.5e-19}{s}$: tak ada muatan di dalamnya; medan dalam
kawatnya dibuat muatan pada \emph{permukaan}nya, yang rapatnya berubah
sepanjang kawatnya.

\textbf{14.} $Q = I_0\tau_c = \qty{16}{mC}$, $V = Q/C = \qty{160}{V}$.

\textbf{15.} Sisi kawatnya mengemban $I$, sisi celahnya tak mengemban arus
hantaran: muatannya tumbuh, dan bersamanya medan listrik dalam celahnya.

\textbf{16.} $I = 16\eu^{-0.5} = \qty{9.7}{A}$; $E = Q/\varepsilon_0A$, $\dd E/\dd t = I/\varepsilon_0A
= 9.7/(8.85 \times 10^{-12} \times 10^{-4}) = \qty{1.1e16}{V.m^{-1}.s^{-1}}$.

\textbf{17.} $\varepsilon_0A\,\dd E/\dd t = I$: yaitu arus pergeserannya.

\textbf{18.} $j = \gamma E = 10^{-14} \times 2.3 \times 10^5 = \qty{2e-9}{A/m^2}$, $10^{-14}$ A
menembus penampangnya: tak berarti; tembus listriknya pada \qty{3}{kV} menyeberangi milimeternya.

\textbf{19.} $R = 1000/(6 \times 10^7 \times 10^{-5}) = \qty{1.7}{\ohm}$; $\Delta U = \qty{67}{V}$; $P =
\qty{2.7}{kW}$, $29\%$ dari \qty{9.2}{kW} yang terserahkan.

\textbf{20.} $I = \qty{0.46}{A}$, $\Delta U = \qty{0.8}{V}$, \qty{0.35}{W}: penyaluran pada
tegangan tinggi membagi ruginya dengan $(U_2/U_1)^2$.

\textbf{21.} $10 \times 6/3.6 = \qty{17}{mm^2}$; \qty{89}{kg/km} tembaga melawan
\qty{45}{kg/km} aluminium: lebih ringan dan lebih murah di menara; di dalam dinding
penampang tembaga yang lebih kecil muat dalam pipa dan terminalnya.

\textbf{22.} $V_H = IB/nqt = 5 \times 10^{-3} \times 0.1/(10^{22} \times 1.6 \times 10^{-19} \times 10^{-4}) =
\qty{3.1}{mV}$.

\textbf{23.} $+24\%$: \qty{2.1}{\ohm}, \qty{3.3}{kW} hilang.

\textbf{24.} $RI^2t = 1.7 \times 4 \times 10^8 \times 10^{-4} = \qty{67}{kJ}$ ke dalam \qty{89}{kg}: \qty{2}{K}.
$j = \qty{2e9}{A/m^2}$, tiga ratus kali nilai rumah tangganya.

\textbf{25.} $n$ (materinya) dan $\tau$ (tumbukannya) memberi $\gamma = ne^2\tau/m$;
$\gamma$ mengaitkan $j$ dan $E$ (Ohm); $j$ terintegralkan menjadi $I$, $E$ menjadi $U$ (rangkaiannya);
$j^2/\gamma$ adalah kalornya.
\end{solution}
