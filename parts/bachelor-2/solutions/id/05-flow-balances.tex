\begin{solution}{exo:b2:flow-balances:12}
(a) $v_1h_1 = v_2h_2 = q$. (b) Gaya hidrostatik tiap satuan lebar pada sebuah
penampang $\int_0^h\rho g(h - z)\dd z = \tfrac12\rho gh^2$; momentumnya: $\tfrac12\rho g(h_1^2
- h_2^2) = \rho q(v_2 - v_1)$; bagilah dengan $\rho$ lalu pakailah $q = v_1h_1 = v_2h_2$.
(c) Dengan $v_2 = v_1h_1/h_2$ dan $h_1 \ne h_2$: $\tfrac12g(h_1 + h_2)h_2 = v_1^2h_1$,
yakni $(h_2/h_1)^2 + h_2/h_1 - 2\mathrm{Fr}_1^2 = 0$; $h_2 > h_1$ menuntut $\mathrm{Fr}_1 > 1$.
(d) $\Delta H = (h_1 + v_1^2/2g) - (h_2 + v_2^2/2g) = (h_2 - h_1)^3/4h_1h_2$.
$\mathrm{Fr}_1 = 2.86$, $h_2/h_1 = 3.57$, $h_2 = \qty{0.71}{m}$, $v_2 = \qty{1.1}{m/s}$,
$\Delta H = 0.514^3/0.57 = \qty{0.24}{m}$; $\rho gq\,\Delta H = \qty{1.9}{kW}$ tiap meter
lebarnya.
\end{solution}

\begin{solution}{pb:b2:flow-balances:1}
\textbf{1.} $F = (D_a + D_f)v_e = 51 \times 600 = \qty{30.6}{kN}$.

\textbf{2.} $F = 51 \times 600 - 50 \times 250 = \qty{18.1}{kN}$; $Fv = \qty{4.5}{MW}$.

\textbf{3.} $\mathcal P_{\text{kin}} = \tfrac12 \times 51 \times 600^2 - \tfrac12 \times 50 \times 250^2 =
\qty{7.6}{MW}$; $\eta_p = 0.59$. Dengan $D_f = 0$: $\eta_p = 2v(v_e - v)/(v_e^2 - v^2)
= 2v/(v_e + v) = 500/850$.

\textbf{4.} Kalornya \qty{43}{MW}: termalnya $7.6/43 = 18\%$; keseluruhannya $4.5/43 = 10\%$.

\textbf{5.} $\tfrac12 \times 500(v_e'^2 - 250^2) = 7.6 \times 10^6$: $v_e' = \qty{305}{m/s}$;
$F = 500 \times 55 = \qty{27.5}{kN}$; $\eta_p = 500/555 = 0.90$. Menggerakkan lebih banyak udara lebih
perlahan memberi \omterm{prop:b2:flow-balances:pelton}{gaya dorong} lebih besar pada daya yang sama: dari situlah kipasnya.

\textbf{6.} Ia menuntut udara luar sekaligus sebagai fluida kerja dan sebagai
pengoksidasi; sebuah roket membawa keduanya.

\textbf{7.} Udara masuk pada \qty{70}{m/s} (ke belakang dalam kerangka mesinnya)
lalu keluar dengan komponen maju $150\cos45^\circ = \qty{106}{m/s}$:
momentum yang diberikan kepada udaranya mengarah maju, $500(106 + 70) = \qty{88}{kN}$
pengereman; aliran yang sama yang dibuang ke belakang akan memberi $500(150 - 70) =
\qty{40}{kN}$ \omterm{prop:b2:flow-balances:pelton}{gaya dorong}.

\textbf{8.} Selama $\dd t$ sistemnya (roket $m$ + gas $D_m\dd t$ yang disemburkan pada
$v - v_e$): $(m - D_m\dd t)(v + \dd v) + D_m\dd t(v - v_e) - mv = -mg\,\dd t$,
sehingga $m\,\dd v/\dd t = D_mv_e - mg$.

\textbf{9.} $D_m = \qty{1333}{kg/s}$; $F = \qty{4.0}{MN}$; $a = 13.3 - 9.8 =
\qty{3.5}{m/s^2}$ saat lepas landas, $40 - 9.8 = \qty{30}{m/s^2}$ saat habis bakar.

\textbf{10.} $v = v_e\ln(m_0/m) - gt$: $3000\ln3 - 9.81 \times 150 = 3296 - 1472 =
\qty{1.8}{km/s}$; rugi gravitasinya \qty{1.5}{km/s}.

\textbf{11.} Tingkat 1: $3000\ln(300/200) = \qty{1.2}{km/s}$; lepaskan \qty{20}{t}; tingkat

\textbf{12.} $v_e = 7800/\ln3 = \qty{7.1}{km/s}$: tak ada propelan kimia

\textbf{13.} \omterm{thm:b2:flow-balances:momentum}{Fluks momentum} tegak pancarannya $D_mv = \qty{4.0}{MN}$
dihilangkan dan sebesar itu pula diciptakan secara mendatar: gaya \qty{4}{MN} ke bawah

\textbf{22.} $\rho gD_VH = 9810 \times 40 \times 205 = \qty{80}{MW}$; \qty{91}{MW} listrik.

\textbf{23.} $P = P_0 + \rho g(200 + 3.3) + \tfrac12\rho v^2 \approx \qty{21}{bar}$ mutlak,

\textbf{24.} $V = 40 \times 28800 = \qty{1.15e6}{m^3}$, $E = \rho gV \times 200 = \qty{2.3e12}{J}
= \qty{630}{MWh}$; yang terpakai $91 \times 8 = \qty{730}{MWh}$; yang terpungut $630 \times 0.9 \times

\textbf{25.} Momentum: \omterm{prop:b2:flow-balances:pelton}{gaya dorong} mesin jet dan roketnya, gaya
Peltonnya. Momentum sudut: momen gaya turbinnya (\omterm{def:b2:fluid-kinematics:euler}{Euler}). Energi:
\omterm{thm:b2:flow-balances:energy}{tinggi tekan} dan daya pompanya, efisiensi penyimpanannya. Massa: setiap debitnya,
kehilangan massa roketnya.
\end{solution}
