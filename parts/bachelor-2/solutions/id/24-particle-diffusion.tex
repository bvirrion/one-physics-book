\begin{solution}{exo:b2:particle-diffusion:5}

\begin{solution}{exo:b2:particle-diffusion:6}
(a) $n = n_{\text{s}}a/r$, $\Phi = 4\pi D a n_{\text{s}} = \qty{3.6e-8}{mol/s}$.
(b) $1.95 \times 10^{-5}\,\unit{mol}$; \qty{540}{s}. (c) Selubung $1/r$-nya memakan
$\sim a^2/D \approx \qty{2000}{s}$ untuk terbangun dan butirnya menyusut; pengadukan
mengganti selubung setebal $\sim a$ itu dengan sebuah \omterm{prop:b2:viscous-flows:bl}{lapisan batas} $\delta \ll a$
lalu mengalikan fluksnya dengan $a/\delta$. (d) $(rn')' = 0$: $n = A + B\ln r$,
yang tak dapat lenyap di takhingga: tak ada keadaan tegar --- awan sebuah
silinder terus tumbuh (secara logaritmik).
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:7}
(a) $\dd/\dd t\int n = D[n']_{-\infty}^{\infty} = 0$; $\dd/\dd t\int x^2 n = D\int
x^2 n'' = 2D\int n = 2DN$. (b) $\partial_t[n(x,-t)] = -D\,\partial_x^2 n$: tandanya
berbalik. (c) Bagi $t < 0$ ``lebar''-nya $2Dt$ negatif: tak ada keadaan
semacam itu --- pulsanya tak dapat dibuat tak menyebar. (d) $\dd/\dd t\int n^2 = 2D\int nn''
= -2D\int n'^2 \le 0$.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:11}
(a) $n = n_\infty(1 - a/r)$, $\Phi = 4\pi D a n_\infty$. (b) $3.8 \times
10^6$ tiap sekon. (c) Reseptornya berlaku seperti konduktansi yang terparalel lalu
terseri dengan selubung bolanya: $\Phi \approx 4\pi Dan_\infty
\cdot N_{\text{r}}s/(N_{\text{r}}s + \pi a)$; separuh maksimumnya bagi
$N_{\text{r}}s = \pi a$: dengan $s = \qty{1}{nm}$, $N_{\text{r}} \approx
3000$, yang meliputi $N_{\text{r}}s^2/4a^2 \approx 10^{-3}$ permukaannya.
(d) Tiap jenis reseptornya memerlukan luas yang dapat diabaikan bagi penangkapan
yang nyaris maksimal: sebuah sel dapat mengawasi beribu zat sekaligus.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:12}
(a) Beda maju dalam $t$, beda kedua terpusat dalam $x$. (b)
$\alpha = 1/2$: $n_i^{k+1} = (n_{i-1}^k + n_{i+1}^k)/2$ --- tiap partikelnya melompat
ke kiri atau ke kanan dengan peluang $1/2$. (c) Bagi $n_i = (-1)^i$, $n^{k+1} =
(1 - 4\alpha)n^k$: $|1 - 4\alpha| > 1$ ketika $\alpha > 1/2$. (d) $\Delta t \le
\Delta x^2/2D = \qty{3.3}{s}$; sekitar $1100$ langkah.
\end{solution}

\begin{solution}{pb:b2:particle-diffusion:1}
\textbf{1.} $k_BT = \qty{0.118}{eV}$, $E_{\text{a}}/k_BT = 29.2$: $D =
\qty{1.6e-17}{m^2/s}$; pada \qty{1000}{\celsius}: \qty{1.6e-18}{m^2/s} --- sebuah
faktor $10$.

\textbf{2.} Permukaan yang memantulnya ditangani lewat bayangan cermin
pulsanya, yang melipatduakan amplitudonya pada $x > 0$; $\int_0^\infty = (Q/\sqrt{\pi Dt})
\cdot \tfrac12\sqrt{4\pi Dt} = Q$.

\textbf{3.} $Dt = \qty{5.8e-14}{m^2}$; $\sqrt{\pi Dt} = \qty{0.43}{\micro m}$; $n_{\text{s}}
= \qty{2.3e24}{m^{-3}}$.

\textbf{4.} $\eu^{-x^2/4Dt} = 4.3 \times 10^{-4}$: $x_j^2 = 4Dt \times 7.75$, $x_j =
\qty{1.3}{\micro m}$.

\textbf{5.} $n_{\text{s}}$ menjadi separuh, logaritmanya turun menjadi $7.05$, $4Dt$
menjadi empat kali: $x_j = \qty{2.6}{\micro m}$ --- kurang dari dua kalinya.

\textbf{6.} $\delta D/D = (E_{\text{a}}/k_BT)\,\delta T/T = 21\%$; $x_j \propto
\sqrt{Dt}$ kira-kira, jadi $1\%$ pada $x_j$ menuntut $2\%$ pada $D$: \qty{1}{K}.

\textbf{7.} $E_{\text{a}}/k_BT = 134$: $D \approx 10^{-62}\,\unit{m^2/s}$; satu
jarak (\qty{0.25}{nm}) dalam $a^2/D \sim 10^{43}\,\unit{s}$: beku.

\textbf{8.} $E_{\text{a}}/k_BT = 32.8$: $D = \qty{4.3e-19}{m^2/s}$; $\sqrt{Dt} =
\qty{28}{nm}$.

\textbf{9.} $n_0\operatorname{erfc}(x/2\sqrt{Dt})$; $n_B/n_0 = 5 \times 10^{-6}$:
$x \approx 6.4\sqrt{Dt} = \qty{0.18}{\micro m}$.

\textbf{10.} $2n_0\sqrt{Dt/\pi} = \qty{6.3e18}{m^{-2}}$: enam kali $Q$ ---
dosisnya tumbuh sebagai $\sqrt t$ dan ditetapkan waktunya.

\textbf{11.} Suhu rendah: $D$ kecil, dosisnya dikendalikan waktunya,
dangkal; suhu tinggi: dosis tetapnya didorong dalam-dalam dengan cepat.

\textbf{12.} \qty{0.18}{\micro m} melawan \qty{1.3}{\micro m}: tipis.

\textbf{13.} $4 \times 10^{18}$ molekul; $n_{\text{a}} = \qty{2.4e25}{m^{-3}}$.

\textbf{14.} $\ell = \qty{37}{nm}$; $v^* = \qty{206}{m/s}$.

\textbf{15.} $D \approx \qty{2.5e-6}{m^2/s}$.

\textbf{16.} $L^2/D = 10^7\,\unit{s}$, berbulan; jika hanyut: \qty{50}{s}.

\textbf{17.} $n = 8N(4\pi Dt)^{-3/2}\eu^{-r^2/4Dt}$; maksimal pada $t^* = r^2/6D =
\qty{6.7e4}{s}$, delapan belas jam.

\textbf{18.} $n_{\max} = 8N(4\pi Dt^*)^{-3/2}\eu^{-3/2} \approx \qty{2e18}{m^{-3}}$,
jauh di atas ambangnya: tercium, akhirnya.

\textbf{19.} $r^2 = 4Dt\ln 10^6 = \qty{12}{m^2}$: \qty{3.5}{m}.

\textbf{20.} $v^*t/\ell \approx 5 \times 10^{14}$.

\textbf{21.} $\langle x\rangle = 0$, $\langle x^2\rangle = N\ell^2 = \ell^2 t/\tau$;
$D = \ell^2/2\tau$.

\textbf{22.} $\binom{N}{N/2}/2^N$: $1/2$, $3/8$, $5/16$ --- menurun
(sebagai $1/\sqrt{\pi N/2}$): pejalannya mengembara pergi sebagai $\sqrt N$.

\textbf{23.} Awannya berkumpul menjadi sebuah titik --- tak pernah terlihat; ia
melanggar \omterm{thm:b2:particle-diffusion:equation}{persamaan difusinya}, bukan hukum mekanikanya.

\textbf{24.} Tiap tumbukannya terbalikkan, tetapi keadaan awalnya (semua
partikelnya bersama) itu luar biasa: hampir setiap riwayat mikroskopis
darinya menyebar, dan kebalikannya menuntut sebuah persekongkolan semua
kecepatannya. Rerata jalan acaknya hanya menyimpan yang lazim lalu
membuang keterangan itu; ketakterbalikannya bersifat statistik.

\textbf{25.} $L \sim \sqrt{Dt}$, $t \sim L^2/D$; $D \sim \ell v^*/3$, yakni
langkah antara tumbukannya: $10^{-5}$, $10^{-9}$, $10^{-17}\,\unit{m^2/s}$.
\end{solution}
