\begin{solution}{exo:b2:potential-wells-tunneling:7}

\end{solution}

\begin{solution}{exo:b2:potential-wells-tunneling:8}

\begin{solution}{pb:b2:potential-wells-tunneling:1}
\textbf{1.} $\kappa = \sqrt{2m\Phi}/\hbar = \qty{1.09e10}{m^{-1}}$.

\textbf{2.} $\eu^{2.2} = 8.8$; $\eu^{0.022}$: $2.2\%$ tiap pikometer.

\textbf{3.} \qty{0.5}{pm}.

\textbf{4.} $6 \times 10^9$ tiap sekon; terpisah \qty{0.16}{ns}, melawan $\hbar/\Phi
\sim 10^{-16}\,\unit{s}$: satu demi satu.

\textbf{5.} Sekitar separuhnya; sisanya meluruh begitu cepat dengan jaraknya sehingga
citranya adalah citra satu atom.

\textbf{6.} $\times 8.8$; arus tetapnya menjaga ujungnya selamat lalu mengubah
eksponensialnya menjadi sebuah tinggi yang linear.

\textbf{7.} $\Delta T = 10^{-12}/(10^{-5} \times 0.02) = \qty{5e-6}{K}$; dudukan yang
setangkup, kecepatan, bahan bermuai rendah, dan sebuah \omterm{def:b2:feedback-oscillators:loop}{umpan balik} yang mengikuti
hanyutannya.

\textbf{8.} Sebuah trapesium, yang direndahkan pada sisi pengumpulnya: penghalang
efektifnya menipis dan $I$ tumbuh lebih cepat daripada linear.

\textbf{9.} $\kappa' = \qty{0.89e10}{m^{-1}}$; pada \qty{0.5}{nm} eksponennya turun dari
$10.9$ menjadi $8.9$, $I \times 7$: ujungnya menarik diri sebesar $\ln 7/2\kappa' \approx \qty{0.1}{nm}$
--- yaitu sebuah bonggol yang elektronik, bukan geometrik.

\textbf{10.} $V(R_{\text{n}}) = 2 \times 90 \times 1.44/8 = \qty{32}{MeV}$; $b = \qty{62}{fm}$.

\textbf{11.} $v = \qty{1.4e7}{m/s}$; $9 \times 10^{20}$ tumbukan tiap sekon.

\textbf{12.} $G = 96$; $T = \eu^{-96} \approx 10^{-42}$.

\textbf{13.} Lajunya $\sim 10^{-21}\,\unit{s^{-1}}$, waktu paruhnya $\sim 10^{13}$ tahun --- seribu
kali nilai yang terukur: bagi sebuah eksponen seratus, sebuah model
kasar sudah baik jika mendarat dalam beberapa orde besaran.

\textbf{14.} $G = 79$: $\eu^{17} \approx 10^7$ kali lebih pendek, sekitar $10^6$ tahun ---
satu MeV tambahan, tujuh orde besaran.

\textbf{15.} $G = 92$: $\eu^{4.5} \approx 90$ kali lebih pendek --- sefermi jari-jari
adalah dua orde besaran.

\textbf{16.} Sebuah proton tak terbentuk lebih dahulu dengan energi positif (ia
terikat sekitar \qty{7}{MeV}); sebuah $^{12}$C bermuatan tiga kali lipat dan bereksponen
tiga kali lebih besar.

\textbf{17.} Ya: partikel $\alpha$-nya selalu mempunyai \qty{4.2}{MeV}; \omterm{def:b2:schrodinger-wave-functions:psi}{fungsi gelombangnya}
sekadar mempunyai sebuah bagian evanesen di bawah penghalangnya, yang di sana tak ada pengukuran
yang menemukannya dengan energi kinetik negatif.

\textbf{18.} \qty{23.7}{GHz}; \qty{1.27}{cm}.

\textbf{19.} $(|{\uparrow}\rangle \pm |{\downarrow}\rangle)/\sqrt2$; yang
setangkup lebih rendah --- tanpa simpul, kelengkungannya lebih kecil.

\textbf{20.} $\psi = (\varphi_+\eu^{-\iu E_+t/\hbar} + \varphi_-\eu^{-\iu E_-t/\hbar})/\sqrt2$;
$P_{\downarrow} = \sin^2(\pi\nu t)$; periodenya \qty{42}{ps}.

\textbf{21.} $\eu^{-\Delta/k_BT} = 0.996$: nyaris sama, tanpa pembalikan; sebuah
medan listrik yang taksergam memilah berkasnya lalu mengirimkan molekul berkeadaan
atas ke dalam rongganya.

\textbf{22.} \qty{1.6e-23}{J}; $6 \times 10^{12}$ molekul tiap sekon.

\textbf{23.} $\kappa a = 6.5$.

\textbf{24.} Ia ditetapkan molekulnya sendiri, serupa bagi tiap
molekulnya, dan tak peka terhadap luarnya; geserannya datang dari
\omterm{thm:b2:sound-waves:doppler}{efek Doppler} dalam berkasnya, medan listrik liar, tumbukan dan
tarikan rongganya.

\textbf{25.} Pengurungan mengkuantumkan ($E_n \propto n^2/L^2$); keevanesenan meloloskan
gelombangnya ($T \sim \eu^{-2\kappa a}$); eksponennya ditetapkan
$\sqrt{2m(V_0 - E)}/\hbar$ dan lebarnya --- massa, tinggi, jarak.
\end{solution}
