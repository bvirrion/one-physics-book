\begin{solution}{pb:b2:open-systems:1}

keluar), \omterm{prop:b2:open-systems:devices}{katup cekik} (isentalpik), evaporator (isobarik, kalor masuk).

\textbf{2.} $h_4 = \qty{256}{kJ/kg}$, $x_4 = 0.28$.

\textbf{3.} \qty{143}{kJ/kg}.

\textbf{4.} \qty{27}{} dan \qty{34}{kJ/kg}; $h_2 = 433$; $T_2 \approx 45 + 7 \approx
\qty{52}{\celsius}$.

\textbf{5.} \qty{177}{kJ/kg}; $\mathrm{COP}_{\text{h}} = 5.2$, $\mathrm{COP}_{\text{c}} = 4.2$;
$5.2 = 4.2 + 1$.

\textbf{6.} $7.8$ dan $6.8$; dua pertiga.

\textbf{7.} \qty{0.045}{kg/s}; \qty{1.5}{kW}, \qty{1.7}{kW} pada stopkontaknya.

\textbf{8.} Kompresornya: $c_p\ln(325/318) \approx \qty{0.022}{kJ/kg/K}$;

\textbf{9.} $313/55 = 5.7$; nisbah tekanannya nyaris berlipat dua (lebih banyak kerja

\textbf{10.} Kumparannya berada di bawah \qty{0}{\celsius} dalam udara lembap; bunga esnya
menyekatnya; pencairannya membalikkan daurnya beberapa menit tiap jam lalu

kondensornya, makin baik --- yaitu pemanas lantainya.

efisiensinya; rakitlah komponennya lalu bandingkanlah dengan Carnot.
