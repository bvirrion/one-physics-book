\section*{Bab \ref{ch:b2:boltzmann-factor} --- Faktor Boltzmann}

\begin{solution}{exo:b2:boltzmann-factor:1}
$H = k_BT/mg = \qty{8.4}{km}$; \qty{0.70}{}, \qty{0.35}{}, \qty{0.30}{bar} (yang sungguhan
\qty{0.26}{}: karena udaranya mendingin dengan ketinggiannya); helium \qty{61}{km}; Mars
\qty{10.7}{km}.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:2}
Hidrogen: $4\eu^{-395} \approx 10^{-171}$; $4\eu^{-19.7} = 10^{-8}$; $4\eu^{-11.8} = 3 \times
10^{-5}$. Natrium: $\eu^{-9.8} = 6 \times 10^{-5}$. Rotasinya: $\eu^{-0.04} = 0.96$, nyaris
sama. Spin intinya: $1 - 4 \times 10^{-6}$ --- yaitu polarisasi mungil yang
dipakai resonansi magnetik.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:3}

\qty{262}{m/s}; H$_2$: \qty{1580}{}, \qty{1780}{}, \qty{1930}{m/s}; butir asapnya:

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:4}
$x = 2.2 \times 10^{-3}$ dan $0.67$; $M/M_{\text{jenuh}} = 2.2 \times 10^{-3}$ dan $0.59$;
$M_{\text{jenuh}} = n\mu = \qty{9.3e4}{A/m}$; $\chi = \mu_0n\mu^2/k_BT = 2.6 \times 10^{-4}$.
\qty{1.7e6}{A/m} milik besi bersifat kooperatif, bukan sebuah efek Boltzmann.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:5}
(a) \cref{prop:b2:boltzmann-factor:twolevel}. (b) $\eu^{-x} < 1$ bagi $T > 0$;
$N_2 > N_1$ akan menuntut $T < 0$ --- yaitu sebuah \omterm{prop:b2:laser:gain}{pembalikan populasi}. (c) $0$ dan
$Nk_B\ln 2$. (d) $x = 1.93$: $0.13$; pada \qty{77}{K}, $x = 7.5$: $5 \times 10^{-4}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:6}
(a) Turunan $N\varepsilon/(\eu^x + 1)$. (b) $x = 2.40$, $C = 0.44\,Nk_B$.
(c) $\propto x^2\eu^{-x} \to 0$: beku; $\propto 1/T^2 \to 0$: kedua arasnya penuh,
tak ada yang tersisa untuk diserap. (d) $k_BT = 0.42\varepsilon$: \qty{0.5}{mK}; di bawah
semilikelvin spin intinya menahan sebagian besar kapasitas kalornya lalu
memperlambat setiap pendinginan.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:7}
(a) \cref{thm:b2:boltzmann-factor:equipartition}. (b) $\tfrac32R$,
$\tfrac52R$. (c) $C_{\text{vib}}/R = x^2\eu^x/(\eu^x - 1)^2$: $0.002$, $0.41$, $0.90$;
$C_V = 2.50R$, $2.91R$, $3.40R$. (d) $0.92$, $0.30$, $0.995$; intannya $x = 4.3$:
$0.26 \times 3R = \qty{6.5}{J/mol/K}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:8}
(a), (b) \cref{thm:b2:boltzmann-factor:maxwell}. (c) $\tfrac32k_BT$. (d)
$\tfrac14n\langle v\rangle = \qty{2.9e27}{m^{-2}.s^{-1}}$: $3 \times 10^{23}$ tiap sentimeter
persegi tiap sekon; menembus \qty{7.9e-13}{m^2}: $2.3 \times 10^{15}$
molekul tiap sekon, sebuah kebocoran $\qty{1e-5}{Pa.m^3/s}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:9}
(a) Integralkanlah $f$ bagian demi bagian; suku pemimpinnya adalah $(2/\sqrt\pi)x\eu^{-x^2}$. (b)
Helium $x = 5.5$: $5 \times 10^{-13}$; nitrogen $x = 14.5$: $\eu^{-211}$. (c) $x =
4.9$: $10^{-10}$ tiap waktu tumbukan --- lenyap; $10^{-17}$ menuntut $x \approx 6.6$,
$T \lesssim \qty{230}{K}$. (d) $x = 11$: $\eu^{-121}$; cukup dingin.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:10}
(a) $n = n_0\eu^{m\omega^2r^2/2k_BT}$. (b) $\eu^{\Delta m\omega^2r^2/2k_BT} = \eu^{0.24} =
1.27$. (c) $H = k_BT/m'g = \qty{16}{\micro m}$; mencacah bolanya pada
beberapa ketinggian memberi $k_B$, maka $N_A = R/k_B$. (d) Ionnya menuruti
$\eu^{\mp e\varphi/k_BT}$ di sekitar sebuah muatan; yang dilinearkan dalam \omterm{prop:b2:maxwell-equations:potentials}{persamaan Poisson}
ini memberi $\varphi'' = \varphi/\lambda_{\text{D}}^2$: potensialnya terperisai
di luar $\lambda_{\text{D}}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:11}
(a) $Z = 1/(1 - \eu^{-x})$, $\langle E\rangle = h\nu/(\eu^x - 1)$. (b) $k_BT$:
ekuipartisi; $h\nu\eu^{-h\nu/k_BT}$: beku. (c) $u_\nu = (8\pi\nu^2/c^3)\langle E
\rangle$; Rayleigh--Jeans memberi tiap ragamnya $k_BT$. (d) Sebuah ragam tak dapat
tereksitasi oleh kurang daripada satu kuantum.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:12}
(a) $x = \mu B/k_BT$ saja; $\ln 2$ dan $0$. (b) $x = 0.67$: $S/Nk_B = 0.51$, jadi
$0.18\,k_B$ tiap spin yang tersingkirkan. (c) $B/T$ tetap: \qty{10}{mK}. (d) Medan
spinnya sendiri (bermilitesla) menggantikan $B$ di ujungnya lalu membatasi pendinginannya;
mengukur $M = n\mu\tanh(\mu B/k_BT)$ memberi $T$.
\end{solution}

\begin{solution}{pb:b2:boltzmann-factor:1}
\textbf{1.} $\dd P/\dd z = -nmg = -(mg/k_BT)P$; $H = \qty{8.4}{km}$.

\textbf{2.} $\eu^{-mgz/k_BT}$: yaitu energi potensial satu molekul, dan
suhu udaranya.

\textbf{3.} \qty{0.70}{}, \qty{0.52}{}, \qty{0.35}{bar}; $H\ln 2 = \qty{5.8}{km}$.

\textbf{4.} $P_0/g = \qty{1.03e4}{kg/m^2}$; $\times 4\pi R^2$: \qty{5.3e18}{kg}; $\rho_0H =
1.22 \times 8400 = \qty{1.03e4}{kg/m^2}$.

\textbf{5.} $5.3 \times 10^{18}/(29 \times 10^{-3}/N_A) = 1.1 \times 10^{44}$.

\textbf{6.} $H_{\text{O}_2} = 7.6$, $H_{\text{N}_2} = \qty{8.7}{km}$: nisbahnya pada \qty{50}{km}
akan menjadi $\eu^{-50/7.6 + 50/8.7} = 0.44$ nilai tanahnya; pergolakannya mencampur
lebih cepat daripada difusinya memisahkan.

\textbf{7.} Lebih rendah: udara yang lebih dingin lebih rapat dan tekanannya turun lebih cepat.

\textbf{8.} \qty{2.5e25}{m^{-3}}; $\eu^{-11.9}$: \qty{1.7e20}{m^{-3}} --- tak ada tepi, hanya sebuah
kelaziman.

\textbf{9.} $11.2\sqrt{6370/6870} = \qty{10.8}{km/s}$.

\textbf{10.} Helium \qty{2040}{}, \qty{2300}{m/s}; nitrogen \qty{770}{}, \qty{870}{m/s}.

\textbf{11.} $x = 5.3$: $4 \times 10^{-12}$; nitrogennya $\eu^{-196}$: tak ada.

\textbf{12.} $\tfrac14 \times 10^{12} \times 2300 \times 4 \times 10^{-12} \times \tfrac12 \approx
10^3$ atom tiap meter persegi tiap sekon; bagi seluruh Bumi dan setahun,
$2 \times 10^{25}$ atom --- \qty{0.1}{kg}.

\textbf{13.} $5 \times 10^{-6} \times 1.1 \times 10^{44} = 5.5 \times 10^{38}$ atom, \qty{3.7e12}{kg};
terhadap \qty{0.1}{kg/tahun} waktu tinggalnya akan menjadi mustahil.

\textbf{14.} Keadaan tunaknya menuntut rugi \qty{3e6}{kg/tahun}: yaitu waktu
tinggal sekitar sejuta tahun, yang merupakan nilai yang diterima --- jadi
taksiran termalnya pada \qty{1000}{K} jauh terlalu kecil: pelepasannya
dikuasai episode yang lebih panas dan proses tak termal (ionik).

\textbf{15.} $x = 3.75$: $\eu^{-14}$ alih-alih $\eu^{-28}$ --- sebuah faktor $10^6$:
ruginya luar biasa peka terhadap suhu eksosfernya.

\textbf{16.} Bulan itu panas dan ringan: segalanya lepas dalam
waktu geologi; Titan itu dingin, $x = 11$ bagi nitrogen.

\textbf{17.} $p_\pm = \eu^{\pm x}/2\cosh x$; $M = n\mu\tanh x$.

\textbf{18.} $x = \mu B/k_BT$: \qty{4.2}{A/m}, \qty{310}{A/m}, \qty{1.6e4}{A/m}
(kejenuhannya \qty{1.9e4}{A/m}); Curie bagi $x \ll 1$, yakni $T \gg \qty{70}{mK}$.

\textbf{19.} $M$ hanya bergantung pada $T$ begitu $B$ diketahui; tercuram dekat
$x \approx 1$, $T \approx \mu B/k_B = \qty{70}{mK}$; orang mengukur
suseptibilitasnya dengan sebuah kumparan.

\textbf{20.} $\langle E\rangle = -N\mu B\tanh x$; $C = Nk_Bx^2/\cosh^2x$, yang memuncak pada
$x = 1.2$: \qty{56}{mK}.

\textbf{21.} $Nk_B\ln 2$ dan $0$: semua spinnya berjajar dengan medannya.

\textbf{22.} \qty{10}{mK}; medan spinnya sendiri sebesar semilitesla menggantikan $B$
lalu menetapkan lantainya.

\textbf{23.} $x = 1$ pada $\mu_{\text{N}}B/k_B = \qty{0.37}{mK}$: Curie berlaku sampai
semilikelvin, dan termometernya berguna sampai mikrokelvin.

\textbf{24.} $\Delta\nu/\nu = \sqrt{8k_BT\ln2/mc^2} \propto \sqrt T$: lebar
sebuah garis mengukur suhu gasnya.

\textbf{25.} Atmosfer, gas yang lepas, spin: $\eu^{-E/k_BT}$ dengan $E = mgz$,
$\tfrac12mv^2$, $\mp\mu B$.
\end{solution}
