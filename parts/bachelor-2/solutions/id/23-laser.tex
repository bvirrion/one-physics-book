\begin{solution}{exo:b2:laser:5}
(a) Lihatlah \cref{prop:b2:laser:einstein}. (b) \qty{600}{nm}: $h\nu/k_BT
= 80$, nisbahnya $5 \times 10^{34}$; \qty{1}{cm}: $4.8 \times 10^{-3}$ ---
\omterm{def:b2:laser:processes}{pancaran terangsang} menguasai. (c) Pada frekuensi gelombang mikro
\omterm{def:b2:laser:processes}{pancaran spontan} dapat diabaikan dan pembalikan yang kecil sudah memberi
penguatan; teknologinya (rongga, \omterm{prop:b2:guided-waves:plates}{pandu gelombang}) sudah ada. (d) $A \propto
\nu^3$, jadi $\tau \propto \lambda^3$: \qty{10}{\micro s}.
\end{solution}

\begin{solution}{exo:b2:laser:6}
(a) $I_{\text{s}} = h\nu/\sigma\tau = \qty{1.0e5}{W/m^2}$. (b) $R =
g_0/\sigma\tau = \qty{3.3e22}{m^{-3}.s^{-1}}$. (c) $I = I_{\text{s}}(5 - 1) =
\qty{4.2e5}{W/m^2}$; $P_{\text{kel}} = 0.01 \times 4.2 \times 10^5 \times
10^{-6} = \qty{4.2}{mW}$. (d) $I = I_{\text{s}}(\sigma R\tau/g_{\text{th}}
- 1) = (h\nu/g_{\text{th}})(R - g_{\text{th}}/\sigma\tau)$: linear dalam $R$
di atas $R_{\text{th}} = g_{\text{th}}/\sigma\tau$.
\end{solution}

\begin{solution}{exo:b2:laser:7}
(a) $c/2L > \qty{1.5}{GHz}$: $L < \qty{10}{cm}$. (b) $\delta\nu = \nu\,\delta L/L
= \qty{1.6}{GHz}$: tiga jarak ragam, seluruh lebar penguatannya --- ragamnya
menyapu menyeberangi kurva penguatannya lalu menyerahkannya kepada tetangganya. (c) Sebuah
pemanas menahan panjangnya sampai sepersekian kecil $\lambda$, dengan memakai
imbangan dua ragam sebagai sinyal galatnya. (d) $c/\delta\nu \approx
\qty{100}{km}$.
\end{solution}

\begin{solution}{exo:b2:laser:8}
(a) $w_0' = \qty{10}{\micro m}$, $z_{\text{R}} = \qty{0.5}{mm}$, $I = 2P/\pi
w_0'^2 = \qty{6.4e6}{W/m^2}$. (b) $w_0' = \qty{3.4}{\micro m}$, $I =
\qty{5.5e7}{W/m^2}$; citra Matahari: $10^3 \times (3/0.15)^2 =
\qty{4e5}{W/m^2}$ --- penunjuknya seratus kali lebih buruk. (c) Daya bola lampunya
menyebar pada $4\pi$ dan citranya pada retinanya meluas; seluruh daya
penunjuknya mendarat dalam beberapa mikrometer, pada panjang gelombang
kepekaan maksimumnya. (d) $2z_{\text{R}} = \qty{1}{mm}$.
\end{solution}

\begin{solution}{exo:b2:laser:11}
(a) Intensitasnya $\times 1.04$ tiap \qty{2}{ns}. (b) $N = P\,(2L/c)/h\nu =
6.4 \times 10^8$ foton; $\ln(6.4 \times 10^8)/\ln 1.04 \approx 520$ perjalanan
pulang pergi, sekitar \qty{1}{\micro s}. (c) Penjenuhan penguatannya; sebuah ragam di bawah
ambangnya kehilangan lebih banyak tiap perjalanan pulang pergi daripada yang diperolehnya lalu tinggal pada aras
spontannya. (d) Ragamnya berbagi satu pembalikan; yang terkuat
menjenuhkannya lalu melaparkan yang lain; hanyutan panjang dan penguatannya mengubah
pemenangnya --- keluarannya melompat dari ragam ke ragam.
\end{solution}

\begin{solution}{pb:b2:laser:1}
\textbf{1.} \qty{1.96}{eV}, \qty{4.74e14}{Hz}.

\textbf{2.} $\qty{0.05}{eV} \approx 1.5\,k_BT$: gerak termalnya memasok
defisit yang kecil itu; alihnya nyaris resonan, maka mangkus.

\textbf{3.} $\eu^{-20.66/0.0345} = \eu^{-600} \approx 10^{-260}$: tak ada; hanya
lucutannya yang menyuapi arasnya.

\textbf{4.} Aras bawahnya kosong sepuluh kali lebih cepat daripada aras atasnya
meluruh: ia selalu nyaris kosong --- sebuah skema empat aras.

\textbf{5.} $v^* = \sqrt{2k_BT/m} = \qty{580}{m/s}$; $\nu v^*/c \approx \qty{0.9}{GHz}$;
rumus eksaknya memberi \qty{1.5}{GHz}.

\textbf{6.} $g_0 = \qty{0.09}{m^{-1}}$; $\eu^{0.027}$: $2.7\%$ tiap lintasan.

\textbf{7.} $g_{\text{th}} = -\ln(0.999 \times 0.99)/0.6 = \qty{0.018}{m^{-1}}$;
marginnya $5$.

\textbf{8.} \qty{500}{MHz}; tiga.

\textbf{9.} $P_{\text{msk}} = \qty{1}{mW}/0.01 = \qty{100}{mW}$; \qty{1.3e5}{W/m^2}.

\textbf{10.} Penguatannya menjenuh menjadi $g_{\text{th}}$; kelebihannya menjadi
foton keluaran (dan \omterm{def:b2:laser:processes}{pancaran spontan}, kalor).

\textbf{11.} $\qty{1}{mW}/\qty{7.5}{W} = 1.3 \times 10^{-4}$: sebagian besar
energi elektronnya pergi ke tempat lain daripada metastabil heliumnya, dan
fotonnya mengemban \qty{1.96}{eV} dari \qty{20.66}{eV} yang ditanamkan.

\textbf{12.} $2 \times 0.072 \times 0.3 = 0.043$; $N = 0.1 \times 2 \times 10^{-9}/3.14
\times 10^{-19} = 6.4 \times 10^8$; $20.3/0.043 \approx 470$ perjalanan pulang pergi;
\qty{0.9}{\micro s}.

\textbf{13.} $\delta\nu = \nu\,\delta L/L = \qty{1.6}{GHz}$: ragamnya menyeberangi
seluruh kurva penguatannya --- \omterm{def:b2:laser:cavity}{lasernya} melompat ke ragam lain kecuali panjangnya
distabilkan.

\textbf{14.} Polarisasi $p$-nya menyeberangi sebuah permukaan Brewster tanpa
pemantulan; polarisasi $s$-nya kehilangan sekitar $15\%$ tiap permukaan lalu
tak pernah mencapai ambangnya: keluaran yang terpolarisasi linear tanpa biaya.

\textbf{15.} $z_{\text{R}} = \qty{0.45}{m}$; $\theta = \qty{6.7e-4}{rad}$; $w(\qty{10}{m})
= \qty{6.7}{mm}$: \qty{13}{mm} lebarnya.

\textbf{16.} $2P/\pi w_0^2 = \qty{7}{kW/m^2}$: tujuh matahari.

\textbf{17.} $w_0' = \qty{34}{\micro m}$; \qty{5.6e5}{W/m^2}.

\textbf{18.} $w_0' = \qty{11}{\micro m}$; \qty{5e6}{W/m^2}; retinanya bertahan
\qty{1}{mW} selama \qty{0.25}{s} refleks penghindarannya --- kelas 2: tampak,
$\le \qty{1}{mW}$, aman karena orang berkedip.

\textbf{19.} \qty{6.7e-5}{rad}; $w_0 = \qty{3}{mm}$, $z_{\text{R}} = \qty{45}{m}$,
$w(\qty{1}{km}) = \qty{67}{mm}$: \qty{13}{cm}.

\textbf{20.} $c/\Delta\nu$: \qty{300}{m}; \qty{30}{cm}.

\textbf{21.} $2.7 \times 10^{17}$; \qty{1}{GW}.

\textbf{22.} \qty{1.8}{km}; \omterm{def:b2:diffraction:huygens}{difraksi} saja: \qty{130}{m}.

\textbf{23.} $0.1/\pi(1800)^2 = 10^{-8}$; $2.6 \times 10^9$ foton.

\textbf{24.} $\lambda/d = \qty{1.3e-5}{rad}$, \qty{5}{km}; $(0.5/5000)^2 = 10^{-8}$:

\textbf{25.} \qty{1.5}{cm}; \qty{0.15}{mm}; kemunduran jaraknya terukur dalam
beberapa pekan.
\end{solution}
