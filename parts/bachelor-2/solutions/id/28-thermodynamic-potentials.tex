\section*{Bab \ref{ch:b2:thermodynamic-potentials} --- Potensial Termodinamika}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$, $G = nRT\ln P + g(T)$; $P = -\partial_VF = nRT/V$,
$V = \partial_PG = nRT/P$; $-\partial_TF = nR\ln V - f'(T)$, yaitu entropinya
$nC_V\ln T + nR\ln V + \text{tetap}$ jika $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$, $\Delta U = 0$, $\Delta S = \qty{19.1}{J/K}$.
Terbalikkan: \qty{5.7}{kJ}, $S_{\text{c}} = 0$; melawan \qty{1}{bar}: \qty{0.9}{kJ},
$S_{\text{c}} = 19.1 - 900/300 = \qty{16}{J/K}$; hampa: tak ada apa pun, $S_{\text{c}} =
\qty{19.1}{J/K}$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$ dengan $\Delta h \approx L$, $\Delta s \approx L/T_{\text{lebur}}$.
Pada \qty{-5}{\celsius}: $g_{\text{cair}} - g_{\text{es}} = +\qty{6.1}{kJ/kg}$, esnya
mantap, cairannya metastabil (kelewatdinginannya ada); pada $+\qty{5}{\celsius}$:
\qty{-6.1}{kJ/kg}, cairannya mantap --- es lewat panas tak teramati, karena
permukaannya melebur lebih dahulu.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:4}
Peleburannya $\qty{-1.3e7}{Pa/K}$; penguapannya \qty{3.6}{kPa/K}. \qty{70}{bar}:
\qty{-0.5}{K}; \qty{0.7}{bar}: \qty{92}{\celsius}; \qty{2}{bar}: $\approx
\qty{127}{\celsius}$ menurut kemiringan setempatnya (\qty{120}{\celsius} sebenarnya: karena
kemiringannya tumbuh).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:5}
(a) Juga $(\partial T/\partial V)_S = -(\partial P/\partial S)_V$ dan
$(\partial T/\partial P)_S = (\partial V/\partial S)_P$. (b) $0$; $a/v^2$. (c)
$\alpha/\kappa_T = \qty{4.6}{bar/K}$; $T\alpha/\kappa_T - P \approx \qty{1400}{bar}$.
(d) \qty{46}{bar}: botolnya pecah.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:6}

\begin{solution}{exo:b2:thermodynamic-potentials:7}
(a) $-C(L - L_0)$; $0$. (b) $\Delta S = -C(L - L_0)^2/2 = \qty{-5e-3}{J/K}$; $Q =
\qty{-1.5}{J}$; $W = +\qty{1.5}{J}$. (c) $+\qty{0.4}{K}$. (d) $(L - L_0) \propto
1/T$: $-9\%$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(a) Materi pergi ke $\mu$ yang lebih rendah: $\mu_{\text{v}} < \mu_{\text{cair}}$ ketika $p <
p_{\text{jenuh}}$; kelembapannya $= p/p_{\text{jenuh}}$. (b) $-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
yang menguntungkan uapnya. (c) Penguapannya hanya menuntut $p < p_{\text{jenuh}}$,
bukan pendidihan; pada $100\%$ potensialnya sama. (d) Pada permukaan yang paling
dingin, yang $p_{\text{jenuh}}(T_{\text{s}})$-nya telah turun di bawah $p$ udaranya
--- yaitu yang memancar ke langit malamnya.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(a) Baku. (b) $b = RT_{\text{c}}/8P_{\text{c}} = \qty{4.3e-5}{m^3/mol}$, $a =
27R^2T_{\text{c}}^2/64P_{\text{c}} = \qty{0.36}{Pa.m^6/mol^2}$. (c) $\dd g = v\,\dd P$
pada $T$ tetap dan $g$ yang sama di kedua ujungnya: $\int v\,\dd P = 0$ mengelilingi
simpalnya. (d) Takstabil di tempat $\partial_vP > 0$; metastabil di antaranya: yaitu
cairan lewat panasnya (yang meletup) dan uap lewat jenuhnya (kabut yang
menunggu inti, kamar awan).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}
(a) $\dd P/\dd T = L_{\text{m}}P/RT^2$: $\ln(P/P_0) = -(L_{\text{m}}/R)(1/T - 1/T_0)$.
(b) \qty{28}{mbar} melawan \qty{23}{mbar} --- $L$ lebih besar pada \qty{20}{\celsius}.
(c) \qty{820}{Pa} melawan \qty{611}{Pa}. (d) $\partial(\Delta g/T)/\partial T = -L/T^2$
dan $\Delta g = 0$ sepanjang kurvanya memberi kemiringan yang sama.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(a) $\mu$ yang sama menyeberangi membrannya: $\mu^\circ(P + \Pi) - RTx_{\text{s}} =
\mu^\circ(P)$, yakni $v_{\text{m}}\Pi = RTx_{\text{s}}$, $\Pi = cRT$. (b) \qty{27}{bar}.
(c) $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$ tiap meter kubik; pabrik yang sungguhan
empat kali lebih banyak. (d) Air mengalir ke potensial yang lebih rendah: masuk ke
sel yang kaya zat terlarut, keluar darinya dalam air garam.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(a) $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$. (b) $k_BT/Na^2 \propto T$. (c)
\qty{1.7e-13}{N}; $1.8 \times 10^{14}$. (d) $E \sim nk_BT \approx \qty{0.4}{MPa}$.
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$; $\dd F = -S\dd T + f\dd L$.

\textbf{2.} Schwarz pada $\dd F$.

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$: energinya
tak berubah; tarikannya entropik.

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$; $Q = \qty{-1.5}{J}$; $W = +\qty{1.5}{J}$; jumlahnya
nol.

\textbf{5.} $+\qty{0.4}{K}$.

\textbf{6.} $L - L_0 \propto 1/T$: \qty{9.1}{cm}, sebuah pengerutan \qty{9}{mm}; jeruji
yang dipanaskan memendek, pusat massa velgnya bergeser, rodanya
berputar.

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$; denda entropinya
dibayar dalam $k_BT$.

\textbf{8.} \qty{1.7e-13}{N}; $1.8 \times 10^{14}$ rantai.

\textbf{9.} Gayanya bersifat energetik ($\partial_LU \ne 0$); penghangatannya melunakkan
ikatannya, sehingga $f$ turun dengan $T$.

\textbf{10.} Kemiringannya $-s$, dengan $s_{\text{gas}} > s_{\text{cair}} > s_{\text{es}}$; garis
terendahnya adalah fase yang mantap; perpotongannya di $T_{\text{lebur}}$ dan $T_{\text{uap}}$.

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}.

\textbf{12.} $\qty{-1.3e7}{Pa/K}$; negatif karena $v_{\text{cair}} < v_{\text{es}}$.

\textbf{13.} \qty{70}{bar}: \qty{-0.5}{K} --- bukan begitu cara orang meluncur pada
\qty{-10}{\celsius}; gesekan dan lapisan permukaan yang mirip cairan yang melakukannya.

\textbf{14.} \qty{270}{bar}, \qty{-2}{K}: dengan fluks geotermalnya,
alasnya melebur dan lapisannya meluncur.

\textbf{15.} \qty{3.6}{kPa/K}; \qty{92}{\celsius}; sekitar \qty{127}{\celsius}
(\qty{120}{} sebenarnya).

\textbf{16.} \qty{28}{mbar} (\qty{23}{}), \qty{820}{Pa} (\qty{611}{}): $L$ tumbuh ketika $T$
turun; galatnya $20$--$30\%$.

\textbf{17.} $\Delta v$ yang sama, jadi nisbah kemiringannya adalah $L_{\text{sub}}/L_{\text{uap}}
= 1.15$: kurva sublimasinya lebih curam --- sebuah tekukan di titik tripelnya.

\textbf{18.} \qty{12}{kJ/kg}; esnya harus berinti, dan inti yang kecil memakan
lebih banyak energi permukaan daripada yang diperolehnya: tetes murninya menunggu, sampai
\qty{-40}{\celsius}.

\textbf{19.} $\Delta v \to 0$, $L \to 0$, nisbahnya berhingga: kurvanya berakhir;
di luarnya, satu fluida yang beranjak sinambung dari mirip cairan ke mirip gas.

\textbf{20.} $\dd g = v\,\dd P$ pada $T$ tetap; $g$ yang sama di kedua ujung
garis mendatarnya: $\int v\,\dd P = 0$, yaitu luas yang sama.

\textbf{21.} Cabang cairan lewat panasnya: tak ada inti gelembung dalam sebuah
cangkir yang licin; sebuah gangguan mendidihkannya seketika.

\textbf{22.} Bekukanlah, pompalah di bawah \qty{611}{Pa}, hangatkanlah perlahan: lintasannya
menyeberangi kurva sublimasinya, tak pernah kurva peleburannya; strukturnya
terjaga.

\textbf{23.} $\mu_{\text{v}}(p) < \mu_{\text{cair}}$ ketika $p < p_{\text{jenuh}}$; sama pada
kejenuhannya.

\textbf{24.} \qty{120}{\celsius} alih-alih \qty{100}{}: empat kali lebih cepat.

\textbf{25.} $S$ bagi yang terkucil, $F$ bagi $T,V$ tetap, $G$ bagi $T,P$ tetap;
$-\Delta F$, $-\Delta G$ yaitu kerja yang dapat dipulihkan; hubungan Maxwellnya mengubah
turunan $S$ menjadi persamaan keadaannya.
\end{solution}
