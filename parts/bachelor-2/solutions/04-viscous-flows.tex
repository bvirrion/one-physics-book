\section*{Chapter \ref{ch:b2:viscous-flows} --- Viscous Flows}

\begin{solution}{exo:b2:viscous-flows:1}
$\tau = \eta U/h = 0.25 \times 2/2 \times 10^{-4} = \qty{2.5}{kPa}$; $F = \qty{1.25}{kN}$;
$P = Fv = \qty{2.5}{kW}$. Dry friction: $0.3 \times 98 = \qty{29}{N}$ --- here the
thin, fast-sheared film resists far more; the viscous drag grows as
$U/h$, so lubrication pays at lower speeds and with thicker films (and
it never seizes).
\end{solution}

\begin{solution}{exo:b2:viscous-flows:2}
(a) $2 \times 10^{-6} \times 3 \times 10^{-5}/10^{-6} = 6 \times 10^{-5}$, creeping. (b)
$1.8 \times 1.5/10^{-6} = 2.7 \times 10^6$, turbulent. (c) $4 \times 30/1.5 \times 10^{-5}
= 8 \times 10^6$, turbulent. (d) $0.025 \times 0.33/3.3 \times 10^{-6} = 2500$,
borderline laminar. (e) $900 \times 0.5 \times 0.01/0.1 = 45$, laminar.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:3}
$D_V = \pi R^4\Delta P/8\eta L = \pi(2 \times 10^{-4})^4 \times 2 \times 10^4/(8 \times 10^{-3}
\times 0.03) = \qty{0.42}{mL/s}$; \qty{12}{s} for \qty{5}{mL}. $v_{\max} = \Delta PR^2/
4\eta L = \qty{6.7}{m/s}$, $\langle v\rangle = \qty{3.3}{m/s}$, $\mathrm{Re} = 2R\langle v
\rangle/\nu = 1300$: laminar. $\tau_w = \Delta PR/2L = \qty{67}{Pa}$. At \qty{0.8}{mm}:
sixteen times the flow, \qty{6.7}{mL/s} --- but $\mathrm{Re} \approx 10^4$: the flow
turns turbulent and the gain is less.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:4}
$v_\infty = 2R^2\Delta\rho\,g/9\eta$. (a) \qty{1.3}{cm/s}, $\mathrm{Re} = 0.02$, \qty{75}{s}
per metre. (b) \qty{0.24}{mm/s}, $\mathrm{Re} = 3 \times 10^{-5}$, \qty{70}{min} per
metre. (c) \qty{1.2}{m/s} but $\mathrm{Re} = 16$: Stokes overestimates (real
speed about \qty{0.7}{m/s}), about a second per metre.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:5}
(a) $\langle v\rangle = 0.1/0.0707 = \qty{1.4}{m/s}$, $\mathrm{Re} = 4.2 \times 10^5$. (b)
$8\eta LD_V/\pi R^4 = \qty{500}{Pa}$. (c) $\lambda = 0.3/25.5 = 0.0118$, $\Delta P = 0.0118
\times 3333 \times 1000 = \qty{39}{kPa}$: $80$ times more; $P = \Delta PD_V = \qty{3.9}{kW}$.
(d) Same speed, $\mathrm{Re} = 3 \times 10^5$, $\lambda = 0.0128$, $\Delta P = 0.0128 \times 4717
\times 1000 = \qty{60}{kPa}$, \qty{6}{kW}: more wall per unit flow, more loss.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:6}
(a) $v = 2 \times 25 \times 10^{-12} \times 50 \times 9.81/(9 \times 10^{-3}) = \qty{2.7}{\micro m/s}$;
\qty{5}{cm} in $1.8 \times 10^4$ s, five hours. (b) \qty{2.7}{mm/s}, \qty{18}{s}. (c)
$\Omega = \sqrt{1000g/r} = \qty{313}{rad/s} = \qty{3000}{rpm}$. (d) $v \propto R^2$: large
particles pellet first; successive spins at increasing speed separate
sizes.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:7}
(a) $F = 0.6 \times 144 \times 0.35 = \qty{30}{N}$, $+\qty{4}{N}$: $P = 34 \times 12 =
\qty{410}{W}$. (b) $47 + 4 = \qty{51}{N}$, \qty{770}{W}. (c) Relative speed
\qty{17}{m/s}: $61 + 4 = \qty{65}{N}$, times the ground speed: \qty{780}{W}. (d)
Drag $\qty{21}{N}$: \qty{300}{W}, saving \qty{110}{W}. (e) $mg\sin\alpha = \qty{39}{N} =
4 + 0.21v^2$: $v = \qty{13}{m/s}$, \qty{47}{km/h}.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:8}
(a) Golf: $\tfrac12\rho C_xS = \qty{2.2e-4}{kg/m}$, $v_\infty = \sqrt{0.45/2.2 \times 10^{-4}}
= \qty{45}{m/s}$; skydiver $\sqrt{785/0.42} = \qty{43}{m/s}$; parachute $\sqrt{785/15} =
\qty{7.2}{m/s}$. (b) $m\dot v = mg - kv^2$, i.e.\ $\dot v = g(1 - v^2/v_\infty^2)$,
whose solution from rest is $v_\infty\tanh(gt/v_\infty)$. (c) $\tanh x = 0.9$ at
$x = 1.47$: $t = 1.47 \times 43/9.81 = \qty{6.5}{s}$; $z = (v_\infty^2/g)\ln\cosh x =
190 \times 0.83 = \qty{160}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:9}
(a) Shear rate $\Omega R/e = 0.5/10^{-3} = \qty{500}{s^{-1}}$; $\Gamma = \tau\cdot2\pi Rh
\cdot R$: $\tau = 0.02/(2\pi \times 2.5 \times 10^{-3} \times 0.1) = \qty{12.7}{Pa}$. (b) $\eta
= 12.7/500 = \qty{2.5e-2}{Pa.s}$. (c) $\mathrm{Re} = \rho Ue/\eta \approx 1000 \times 0.5 \times
10^{-3}/0.025 = 20$: laminar. (d) A thin gap makes the shear rate uniform
(the plane approximation) and keeps the flow laminar.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:10}
$D_V = \qty{8.3e-5}{m^3/s}$. (a) $R_h = 8\eta L/\pi R^4 = 8 \times 3.5 \times 10^{-3} \times
0.4/(\pi \times 2.4 \times 10^{-8}) = \qty{1.5e5}{Pa.s/m^3}$, $\Delta P = \qty{12}{Pa}$. (b)
One capillary $\qty{3.5e16}{Pa.s/m^3}$; $10^{10}$ in parallel: \qty{3.5e6}{Pa.s/m^3};
$\Delta P = \qty{290}{Pa}$ (\qty{2}{mmHg}). (c) One arteriole $8 \times 3.5 \times 10^{-3}
\times 2 \times 10^{-3}/(\pi \times 10^{-20}) = \qty{1.8e15}{Pa.s/m^3}$; the set:
\qty{1.8e7}{Pa.s/m^3}; $\Delta P = \qty{1.5}{kPa}$: the arterioles dominate the
two (the small vessels, not the large ones, are where the pressure
falls). (d) $P = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, about $1\%$. (e)
$(1/1.1)^4 = 0.68$: a third of the arteriolar resistance removed ---
vasodilators lower the blood pressure this way.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:11}
(a) $F = b\int_0^L\tau_w\,\dd x = 0.332\rho U^2b\sqrt{\nu/U}\int_0^L\dd x/\sqrt x =
0.664\rho U^2b\sqrt{\nu L/U} = 0.664\rho U^2bL/\sqrt{\mathrm{Re}_L}$. (b) $\mathrm{Re}_L =
3 \times 10^5$ (laminar); $\delta \approx 5\sqrt{\nu L/U} = \qty{9}{mm}$; $F = 0.664 \times 1000
\times 0.09 \times 1/548 = \qty{0.11}{N}$ per face, \qty{0.22}{N}. (c) $\mathrm{Re}_L = 10^7$:
the layer is turbulent beyond the first few centimetres, the laminar
formula fails (the real drag is several times larger). (d) $\mathrm{Re}_x =
10^9$: $\delta \approx 0.37 \times 100/63 = \qty{0.6}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:12}
(a) $0 = \rho g + \eta v''(y)$, $v(0) = 0$ (no slip), $v'(h) = 0$ (no stress at
the free surface). (b) Integrate twice: $v = (\rho g/2\eta)(2hy - y^2)$; $q =
\int_0^hv\,\dd y = \rho gh^3/3\eta$. (c) $v_s = \rho gh^2/2\eta = 1200 \times 9.81 \times
4 \times 10^{-8}/1 = \qty{0.47}{mm/s}$; $q = \qty{6.3e-8}{m^2/s}$; \qty{28}{cm} in ten
minutes --- a Newtonian paint of that thickness would run off; real
paints are shear-thinning and thixotropic, and sag anyway when too thick
because $v_s \propto h^2$. (d) $1400 \times 9.81 \times 4 \times 10^{-6}/20 = \qty{2.7}{mm/s}$.
\end{solution}

\begin{solution}{pb:b2:viscous-flows:1}
\textbf{1.} $S = \qty{4.9}{cm^2}$, $\langle v\rangle = 83/4.9 = \qty{0.17}{m/s}$; $\mathrm{Re}
= 1060 \times 0.17 \times 0.025/3.5 \times 10^{-3} = 1300$: laminar (pulsatile peaks
reach $\sim 5000$).

\textbf{2.} $v_{\max} = \qty{0.34}{m/s}$; $\tau_w = 4\eta\langle v\rangle/R = \qty{0.19}{Pa}$.

\textbf{3.} $\mathrm{Re} = 1060 \times 3 \times 10^{-4} \times 8 \times 10^{-6}/3.5 \times 10^{-3} =
7 \times 10^{-4}$; $\Delta P = 8\eta L\langle v\rangle/R^2 = 8 \times 3.5 \times 10^{-3} \times 10^{-3}
\times 3 \times 10^{-4}/1.6 \times 10^{-11} = \qty{525}{Pa} = \qty{3.9}{mmHg}$.

\textbf{4.} $1/0.3 = \qty{3.3}{s}$: long enough for oxygen to diffuse the few
micrometres to the tissue (\cref{ch:b2:particle-diffusion}).

\textbf{5.} One arteriole $8\eta L/\pi R^4 = \qty{1.8e15}{Pa.s/m^3}$; $10^8$ in
parallel: \qty{1.8e7}{Pa.s/m^3}; $\Delta P = 1.8 \times 10^7 \times 8.3 \times 10^{-5} =
\qty{1.5}{kPa} = \qty{11}{mmHg}$.

\textbf{6.} $P = \Delta P\,D_V = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, about
$1\%$ of the body's power.

\textbf{7.} $(1/0.6)^4 = 7.7$. At rest the vessels downstream dilate and
compensate; during effort the demand rises four- or fivefold and the
narrowed artery cannot deliver it: angina.

\textbf{8.} Five times the speed: $\mathrm{Re} \approx 6500$, turbulent --- the
murmurs a stethoscope hears during exercise.

\textbf{9.} In the slow, low-shear flow of small veins and capillaries
the red cells aggregate and the apparent \omterm{def:b2:viscous-flows:viscosity}{viscosity} rises (the opposite
in the fast aorta); all the pressure-drop estimates for small vessels
are lower bounds.

\textbf{10.} $\langle v\rangle = 3/1.13 = \qty{2.65}{m/s}$; $\mathrm{Re} = 850 \times 2.65 \times
1.2/0.02 = 1.35 \times 10^5$: turbulent.

\textbf{11.} $\lambda = 0.316/19.2 = 0.0165$; $\Delta P/L = (0.0165/1.2) \times \tfrac12 \times
850 \times 7.0 = \qty{41}{Pa/m} = \qty{41}{kPa/km}$; \qty{534}{bar} over the line.

\textbf{12.} $534/80 = 6.7$: at least seven stations.

\textbf{13.} $P = \Delta P\,D_V = 5.34 \times 10^7 \times 3 = \qty{160}{MW}$;
\qty{53}{MJ/m^3}, $0.15\%$ of the oil's energy.

\textbf{14.} $\Delta P = 32\eta L\langle v\rangle/D^2 = 32 \times 0.02 \times 1.3 \times 10^6
\times 2.65/1.44 = \qty{15}{bar}$: thirty-five times less. Turbulence is
expensive; a few parts per million of long polymers delay it and cut
the drop by tens of percent.

\textbf{15.} $\mathrm{Re} = 1350$: laminar; Poiseuille: $\Delta P = 32 \times 2 \times 1.3 \times
10^6 \times 2.65/1.44 = \qty{1500}{bar}$ --- impossible; the oil must be kept
warm, hence the insulation and heating.

\textbf{16.} $1.3 \times 10^6/2.65 = \qty{4.9e5}{s} = \qty{5.7}{days}$.

\textbf{17.} $\tau_w = (\Delta P/L)D/4 = 41 \times 0.3 = \qty{12}{Pa}$; $F = \tau_w\pi DL =
\qty{6e7}{N}$.

\textbf{18.} $\Delta P \propto \lambda\langle v\rangle^2/D \propto D^{-4.75}$: $1.2^{4.75} = 2.4$.

\textbf{19.} $v = 2R^2\Delta\rho\,g/9\eta = 2 \times 2.5 \times 10^{-9} \times 1800 \times 9.81/
0.18 = \qty{0.49}{mm/s}$; $\mathrm{Re} = 2 \times 10^{-3}$; \qty{41}{min} to cross
\qty{1.2}{m}. Turbulent fluctuations of a few centimetres per second
keep it suspended while the oil runs; in a stopped line it settles in
under an hour.

\textbf{20.} $v = 2 \times 2.5 \times 10^{-11} \times 1000 \times 9.81/1.62 \times 10^{-4} =
\qty{3.0}{mm/s}$; $\mathrm{Re} = 2 \times 10^{-3}$; \qty{1}{km} in $3.3 \times 10^5$ s, four
days: any updraft above a few millimetres per second holds it.

\textbf{21.} \qty{1.2}{m/s}, $\mathrm{Re} = 16$: Stokes overestimates by about a
factor two (real: \qty{0.7}{m/s}).

\textbf{22.} Stokes would give \qty{120}{m/s} and $\mathrm{Re} \sim 10^4$: absurd.
Quadratic drag, $\tfrac43\pi R^3\rho_wg = \tfrac12\rho v^2\pi R^2C_x$: $v = \sqrt{8R
\rho_wg/3\rho C_x} = \qty{6.6}{m/s}$ ($\mathrm{Re} = 900$, consistent); \qty{150}{s} from
\qty{1}{km}.

\textbf{23.} \qty{1}{cm/s} $>$ \qty{3}{mm/s}: the droplet rises; the drizzle drop
needs about \qty{1}{m/s}, which convective clouds provide.

\textbf{24.} $2\gamma/R$: \qty{29}{kPa}, \qty{1.4}{kPa}, \qty{144}{Pa}; \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamic
pressures} $\tfrac12\rho v^2$: $5 \times 10^{-6}$, $0.3$ and \qty{26}{Pa}. The ratio
reaches $0.2$ at \qty{2}{mm}; at \qty{5}{mm} ($v \approx \qty{9}{m/s}$, $\tfrac12\rho v^2 =
\qty{49}{Pa}$ against \qty{58}{Pa}) the aerodynamic pressure flattens the
drop and breaks it: no raindrop exceeds about \qty{5}{mm}.

\textbf{25.} Aorta $10^3$ (laminar, near transition); capillary $10^{-3}$
(Poiseuille); arteriole $10^{-2}$ (Poiseuille); warm pipeline $10^5$
(turbulent, $\lambda(\mathrm{Re})$); cold pipeline $10^3$ (Poiseuille); sand in
oil $10^{-3}$ (Stokes); cloud droplet $10^{-3}$ (Stokes); raindrop $10^3$
(quadratic drag, $C_x \approx 0.5$).
\end{solution}
