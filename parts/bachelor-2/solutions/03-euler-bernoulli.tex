\section*{Chapter \ref{ch:b2:euler-bernoulli} --- Perfect Fluids: Euler and Bernoulli}

\begin{solution}{exo:b2:euler-bernoulli:1}
$v_2 = 1.5 \times 4 = \qty{6.0}{m/s}$, $P_2 = 3.0 \times 10^5 - 500(36 - 2.25) =
\qty{2.83}{bar}$. At \qty{2}{cm}: $v = \qty{24}{m/s}$, $P = 3.0 \times 10^5 - 500
(576 - 2.25) = \qty{0.13}{bar}$ --- close to the vapour pressure. A negative
result means the assumed flow rate is impossible: the water cavitates
and the flow is throttled.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:2}
$v = \sqrt{2 \times 15000/0.74} = \qty{201}{m/s}$; indicated $\sqrt{2 \times 15000/
1.22} = \qty{157}{m/s}$. The indicated airspeed measures the \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamic pressure},
which is what the wing feels: the stall speed is a fixed \emph{indicated}
speed at any altitude.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:3}
Depth \qty{2.0}{m}: $v = \sqrt{2g \times 2} = \qty{6.3}{m/s}$; $D_V = 0.6 \times 10^{-4}
\times 6.3 = \qty{0.38}{L/s}$; fall time $\sqrt{2/g} = \qty{0.45}{s}$, range
\qty{2.8}{m}. Range $= 2\sqrt{h(H - h)}$ is symmetric in $h \leftrightarrow H -
h$: a hole at depth \qty{1.0}{m} (\qty{2}{m} above the floor) hits the same
point.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:4}

\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:6}
(a) Paraboloid $z = z_0 + \Omega^2r^2/2g$, $\Omega = 4\pi = \qty{12.6}{rad/s}$. (b)
$\Omega^2R^2/2g = 158 \times 0.0225/19.6 = \qty{0.18}{m}$. (c) The mean height of a
paraboloid over the disk is halfway: centre $0.20 - 0.09 = \qty{0.11}{m}$, rim
\qty{0.29}{m}. (d) Centre dry when $\Omega^2R^2/4g = h_0$: $\Omega = \sqrt{4gh_0}/R =
\qty{18.7}{rad/s} = 3.0$ turns per second. (e) A paraboloid focuses
parallel rays to one point, at $f = g/2\Omega^2$; liquid-mirror telescopes
spin mercury.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:7}
(a) $v = \sqrt{2g \times 3} = \qty{7.7}{m/s}$; $D_V = \pi \times 10^{-4} \times 7.7 =
\qty{2.4}{L/s}$. (b) $P = P_0 - \rho g(2) - \tfrac12\rho v^2 = 100 - 19.6 - 29.4 =
\qty{51}{kPa}$. (c) $P_0 - \rho gh_c - \tfrac12\rho v^2 \ge \qty{2.3}{kPa}$: $h_c \le
\qty{7.0}{m}$. (d) Air is compressible and light: no continuous liquid
column to transmit the pressure, the water on both sides just stays.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:9}
(a) $v_2 = \qty{2.5}{m/s}$; $\Delta P = \tfrac12 \times 1060 \times (6.25 - 0.25) =
\qty{3.2}{kPa}$, $P_2 = \qty{9.8}{kPa}$. (b) Still above \qty{1}{kPa}: no. (c) At
\qty{0.05}{cm^2}, $v = \qty{5}{m/s}$ and $\Delta P = \qty{13}{kPa}$: $P_2 \approx 0$, below

\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:10}
(a) $c_PT_0 = c_PT + v^2/2$ with $c_P = \gamma R/(\gamma - 1)M_{\text{mol}}$ and
$c^2 = \gamma RT/M_{\text{mol}}$: $v^2/2c_PT = \tfrac12(\gamma - 1)M^2$. (b) $M =
0.85$: $T_0 = 220 \times 1.145 = \qty{252}{K}$; $M = 2$: $220 \times 1.8 = \qty{396}{K}$;
$M = 25$: $220 \times 126 = \qty{28000}{K}$ --- meaningless: the air dissociates
and ionizes, $c_P$ is not constant (real stagnation temperatures are
several thousand kelvin). (c) $P_0/P = (1 + \tfrac{\gamma - 1}2M^2)^{\gamma/(\gamma
- 1)} \approx 1 + \tfrac{\gamma}2M^2 + \tfrac{\gamma}8M^4$, and $\gamma PM^2 = \rho v^2$:
$P_0 - P = \tfrac12\rho v^2(1 + M^2/4)$. Reading $v$ from $\sqrt{2(P_0 - P)/\rho}$
overestimates it by $\sqrt{1 + M^2/4} - 1 \approx 9\%$ at $M = 0.85$.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:11}
(a) From the surface ($v \approx 0$, $P_0$, height $h$) to the outlet ($v$, $P_0$,
height $0$), with $\int\partial_tv\,\dd s = \ell\,\dd v/\dd t$. (b) $\dd v/(V^2 -
v^2) = \dd t/2\ell$, $V^2 = 2gh$: $\operatorname{artanh}(v/V) = Vt/2\ell$. (c) $V =
\qty{9.9}{m/s}$; $\tanh x = 0.9$ at $x = 1.47$: $t = 2\ell x/V = \qty{15}{s}$. (d)
$\dd v/\dd t \approx \qty{-20}{m/s^2}$: $\Delta P = \rho\ell \times 20 = \qty{9.9}{bar}$.
Water is slightly compressible: a fast closure launches a pressure
wave, $\Delta P = \rho c\Delta v \approx \qty{140}{bar}$.
\end{solution}

\begin{solution}{exo:b2:euler-bernoulli:12}
(a) Steady, perfect, incompressible and irrotational: one constant
for the whole flow. (b) On the surface $P = P_0$: $\tfrac12\rho v^2 + \rho gz =
0$, $z = -\Gamma^2/8\pi^2gr^2$. (c) Radial Euler: $\partial_rP = \rho\Omega^2r$,
vertical hydrostatics: the surface satisfies $z(r) - z(a) = \Omega^2(r^2 -
a^2)/2g$. (d) Bath: $\Omega = \Gamma/2\pi a^2 = \qty{191}{rad/s}$; $z(a) = -\Omega^2a^2/
2g = \qty{-4.7}{cm}$, and the core adds another \qty{4.7}{cm}: \qty{9.3}{cm} deep.
Tornado: $\Delta P = \tfrac12\rho v_{\max}^2$ outside plus the same inside:
$\rho v_{\max}^2 = 1.2 \times 6400 = \qty{7.7}{kPa}$ below ambient at the centre.
\end{solution}

\begin{solution}{pb:b2:euler-bernoulli:1}
\textbf{1.} $\rho g \times 150 = \qty{14.7}{bar}$ above atmospheric; $\tfrac12\rho gH_d^2
\times \qty{1}{m} = \qty{1.1e8}{N}$.

\textbf{2.} $\rho gH = \qty{19.6}{bar}$ gauge, \qty{20.6}{bar} absolute.

\textbf{3.} $v = 60/7.07 = \qty{8.5}{m/s}$; $\tfrac12\rho v^2 = \qty{0.36}{bar}$; $P =
P_0 + \rho gH - \tfrac12\rho v^2 = \qty{20.3}{bar}$ absolute.

\textbf{4.} $v_j = \sqrt{2gH} = \qty{62.6}{m/s}$; $s = 60/62.6 = \qty{0.96}{m^2}$, $d =
\qty{1.1}{m}$.

\textbf{5.} $\tfrac12\rho D_Vv_j^2 = \tfrac12\rho D_V(2gH) = \rho gD_VH = \qty{118}{MW}$.

\textbf{6.} $m = \rho S\ell = \qty{2.8e6}{kg}$, $E_k = \tfrac12mv^2 = \qty{1.0e8}{J}$;
transit $\ell/v = \qty{47}{s}$.

\textbf{7.} $\rho gH = (P - P_0) + \tfrac12\rho v^2$: \qty{19.3}{bar} of pressure
and \qty{0.36}{bar} of kinetic energy; in the nozzle the pressure falls
to $P_0$ while $\tfrac12\rho v^2$ rises to $\rho gH$: just after it, $P = P_0$.

\textbf{8.} $\Delta h = \lambda(\ell/D)v^2/2g = 0.015 \times 133 \times 3.67 = \qty{7.3}{m}$,
$3.7\%$ of the power.

\textbf{9.} $v = \qty{19.1}{m/s}$, $v^2/2g = \qty{18.6}{m}$, $\Delta h = 0.015 \times 200 \times
18.6 = \qty{56}{m}$: $28\%$ lost --- the saving in steel is paid every
second in energy.

\textbf{10.} $v_t = 60/\pi = \qty{19.1}{m/s}$; $\Delta P = 500(19.1^2 - 8.49^2) =
\qty{146}{kPa}$.

\textbf{11.} $\Delta P = (\rho_{\text{Hg}} - \rho)g\,\Delta h$: $\Delta h = 146000/(12600
\times 9.81) = \qty{1.18}{m}$.

\textbf{12.} $\Delta P \propto v^2 \propto D_V^2$: $D_V \propto \sqrt{\Delta h}$; half the
flow gives a quarter, \qty{0.30}{m}.

\textbf{13.} A protruding tap faces or disturbs the flow and reads part
of the \omterm{thm:b2:euler-bernoulli:bernoulli}{dynamic pressure} (a Pitot), not the static pressure.

\textbf{14.} $P = P_0 - \rho g \times 6 - \tfrac12\rho v^2 = 100 - 59 - 36 = \qty{5}{kPa}$:
near the vapour pressure, dissolved air comes out and the water may
boil (cavitation), breaking the column. Lower the crest, or fit an air
valve and a vacuum pump, or reduce the flow.

\textbf{15.} $D_V/A = 60/2 \times 10^6 = \qty{3e-5}{m/s} = \qty{11}{cm/h}$.

\textbf{16.} $A\,\dd h/\dd t = -s\sqrt{2gh}$: $\sqrt h = \sqrt{h_0} - (s/A)\sqrt{g/2}\,t$.

\textbf{17.} $\Delta t = (A/s)\sqrt{2/g}(\sqrt{200} - \sqrt{190}) = 2.09 \times 10^6 \times
0.45 \times 0.36 = \qty{3.4e5}{s} = \qty{3.9}{days}$; at constant design flow
$10A/D_V = \qty{3.3e5}{s}$: nearly the same, the nozzle having been sized
for Torricelli's flow at $h \approx H$.

\textbf{18.} $\ell\,\dd v/\dd t = gH - v^2/2$: $v = V\tanh(Vt/2\ell)$, $V = \qty{62.6}{m/s}$,
time constant $2\ell/V = \qty{13}{s}$.

\textbf{19.} At $v = V/2$: $\dd v/\dd t = (gH - V^2/8)/\ell = \qty{3.7}{m/s^2}$;
$P = P_0 + \rho gH - \tfrac12\rho v^2 - \rho\ell\,\dd v/\dd t = 1.0 + 19.6 - 4.9 - 14.7
= \qty{1.0}{bar}$: most of the head is busy accelerating the column, so
the pressure at the bottom is far below its steady value.

\textbf{20.} $|\dd v/\dd t| = 8.49/10 = \qty{0.85}{m/s^2}$: $\Delta P = \rho\ell \times 0.85 =
\qty{3.4}{bar}$.

\textbf{21.} $\Delta P = 1000 \times 1400 \times 8.49 = \qty{119}{bar}$, six times the
static pressure. The wave's round trip is $2\ell/c = \qty{0.57}{s}$: a
closure faster than that is "sudden" and gets the full Joukowsky
over-pressure; slower closures get less.

\textbf{22.} $Av = A_s\,\dd z/\dd t$.

\textbf{23.} Euler along the tunnel, from the reservoir (pressure
$P_0 + \rho g\times$depth) to the foot of the shaft (pressure $P_0 + \rho g
(\text{depth} + z)$): $\rho L\,\dd v/\dd t = -\rho gz$; with $v = (A_s/A)\dot z$:
$\ddot z + (gA/LA_s)z = 0$; $T = 2\pi\sqrt{LA_s/gA} = 2\pi\sqrt{2000 \times 100/
68.7} = \qty{340}{s} \approx \qty{5.7}{min}$.

\textbf{24.} $\tfrac12\rho ALv^2 = \tfrac12\rho gA_sz_{\max}^2$ with $v = 60/7 =
\qty{8.6}{m/s}$: $z_{\max} = v\sqrt{AL/gA_s} = \qty{32}{m}$.

\textbf{25.} Dam: hydrostatics, $\rho gh$. Turbine, valves closed: the
same, $\rho gH$. Penstock in operation: Bernoulli, $\tfrac12\rho v^2$ taken
from the static head. Water hammer: compressibility, $\rho c\Delta v$.
Surge: unsteady Euler along the tunnel, an oscillation of period
$2\pi\sqrt{LA_s/gA}$.
\end{solution}
