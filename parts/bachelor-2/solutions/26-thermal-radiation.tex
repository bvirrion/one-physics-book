\begin{solution}{exo:b2:thermal-radiation:5}
(a) $\eu^x - 1 \approx x$: $u_\nu = (8\pi\nu^2/c^3)\,k_BT$ --- $k_BT$ per mode.
(b) $\int\nu^2\dd\nu$ diverges. (c) $\nu = k_BT/h = \qty{6e12}{Hz}$ (\qty{50}{\micro m}):
radio and microwaves at \qty{300}{K} are deep in the classical regime (a
resistor's noise is $k_BT$ per unit bandwidth). (d) The probability of
finding the quantum $h\nu$ excited at $T$: a Boltzmann factor.
\end{solution}

\begin{solution}{exo:b2:thermal-radiation:6}
(a) $x = 4.965$. (b) $hc/4.965k_B = \qty{2.90e-3}{m.K}$. (c) $x^3/(\eu^x - 1)$

\end{solution}

\begin{solution}{exo:b2:thermal-radiation:7}
(a) $u = (8\pi^5k_B^4/15h^3c^3)T^4$, $\sigma = cu/4T^4 = \qty{5.67e-8}{W/m^2/K^4}$.
(b) Isotropy and the $\cos\theta$ projection: $\int\cos\theta\,\dd\Omega/4\pi = 1/4$
over a hemisphere. (c) $n = 8\pi \times 2.404\,(k_BT/hc)^3 = 60.4\,(k_BT/hc)^3$;
mean energy $(\pi^4/15)/2.404 = 2.70\,k_BT$. (d) $5 \times 10^8$ and $400$
per cubic centimetre.
\end{solution}

\begin{solution}{exo:b2:thermal-radiation:8}

\end{solution}

\begin{solution}{pb:b2:thermal-radiation:1}
\textbf{1.} \qty{5800}{K}.

\textbf{2.} \qty{6.4e7}{W/m^2}; \qty{3.9e26}{W}.

\textbf{3.} $L/4\pi d^2 = \qty{1.37e3}{W/m^2}$; $u = S/c = \qty{4.6e-6}{J/m^3}$;
$P = S/c = \qty{4.6e-6}{Pa}$ if absorbed, twice if reflected.

\textbf{4.} $2.7k_BT = \qty{2.2e-19}{J}$ (\qty{1.35}{eV}); $6 \times 10^{21}$
photons per second per square metre.

\textbf{5.} $3.9 \times 10^{26} \times 1.4 \times 10^{17} = \qty{5.5e43}{J}$; $Mc^2 =
\qty{1.8e47}{J}$: three ten-thousandths of its mass.

\textbf{6.} \qty{1.2e17}{W}; $T_{\text{e}} = \qty{255}{K}$.

\textbf{7.} Intercepted on $\pi R^2$, emitted from $4\pi R^2$ thanks to
rotation and winds; the subsolar point alone: $((1 - A)S/\sigma)^{1/4} =
\qty{360}{K}$.

\textbf{8.} \qty{381}{K}; at night the stored heat ($\sim C\Delta T$) drains by
radiation with the time constant $C/4\sigma T^3 \approx \qty{1}{day}$, short
against the fortnight of darkness: the ground falls to where its
radiation matches the trickle from below, \qty{100}{K}.

\textbf{9.} \qty{10}{\micro m}, twenty times the Sun's.

\textbf{10.} $(0.65/0.70)^{1/4} = 0.982$: \qty{4.7}{K} lower, \qty{250}{K}.

\textbf{11.} Space: $\sigma T_{\text{a}}^4 = \sigma T_{\text{e}}^4$; layer: $\sigma
T_{\text{s}}^4 = 2\sigma T_{\text{a}}^4$; surface: $\sigma T_{\text{s}}^4 = \sigma
T_{\text{e}}^4 + \sigma T_{\text{a}}^4$. Hence $T_{\text{s}} = 2^{1/4}T_{\text{e}} = \qty{303}{K}$.

\textbf{12.} $T_{\text{s}}^4 = T_{\text{e}}^4/(1 - \varepsilon/2)$; $(288/255)^4 = 1.63$
gives $\varepsilon = 0.77$.

\textbf{13.} Water vapour, carbon dioxide, methane, ozone absorb in the
infrared (vibration bands: H$_2$O around \qty{6}{\micro m} and beyond
\qty{17}{\micro m}, CO$_2$ at \qty{15}{\micro m}); N$_2$ and O$_2$ do not (no
dipole); the window is \qty{8}{}--\qty{13}{\micro m}.

\textbf{14.} $T_{\text{a}}^4 = T_{\text{s}}^4/2$: $\varepsilon\sigma T_{\text{a}}^4 \approx
\qty{150}{W/m^2}$, against \qty{238}{W/m^2} of absorbed sunlight (a
multilayer atmosphere sends back more, some \qty{330}{W/m^2}).

\textbf{15.} $3.7/3.8 \approx \qty{1}{K}$.

\textbf{16.} \qty{2.1e8}{J/m^2/K}; $\tau \approx \qty{5.6e7}{s}$, two years.

\textbf{17.} They reflect sunlight and radiate infrared down; at night
only the second acts: cloudy nights are warmer.

\textbf{18.} \qty{49}{mm^2}; \qty{31}{cm}; coiled to fit and to cut
convective losses to the filling gas.

\textbf{19.} \qty{1.04}{\micro m}; \qty{4.8}{W}; $8\%$.

\textbf{20.} $R = V^2/P = \qty{880}{\ohm}$; cold \qty{88}{\ohm}: \qty{2.6}{A}

\textbf{21.} $49 \times (2800/3200)^4 = \qty{29}{mm^2}$; \qty{8.4}{W} visible; eight

\textbf{22.} Evaporated tungsten binds to the halogen and is returned

\textbf{23.} \qty{55}{W} of infrared and conduction through the gas; the

\textbf{24.} $C = \qty{2.8e-3}{J/K}$; with $P \propto T^4$, the time to halve $T$
is $7CT_0/3P \approx \qty{0.3}{s}$ --- and the glow follows the mains
with a \qty{0.04}{s} lag, which is why it hardly flickers.

\textbf{25.} Temperature fixes the spectrum's shape and peak
($\lambda_{\max}T$), and the total ($\sigma T^4$); filament, Sun and Earth
are three black bodies, at \qty{2800}{}, \qty{5800}{} and \qty{288}{K}, each
balancing what it receives against $\sigma T^4$.
\end{solution}
