\end{solution}

\begin{solution}{exo:b2:particle-diffusion:4}
$D \approx \qty{1.2e-5}{m^2/s}$; $N = (L/\ell)^2 = 2 \times 10^{14}$; $N\ell/v^*

\begin{solution}{exo:b2:particle-diffusion:5}
(a) $Dn'' = q$: $n = n_0 - q(a^2 - x^2)/2D$. (b) $n(0) \ge 0$: $a \le
\sqrt{2Dn_0/q}$. (c) \qty{0.28}{mm}. (d) Nenhuma célula pode estar a mais que
algumas centenas de micrômetros de um capilar (na prática $50$--\qty{100}{\micro m}

\end{solution}

\begin{solution}{exo:b2:particle-diffusion:6}
(a) $n = n_{\text{s}}a/r$, $\Phi = 4\pi D a n_{\text{s}} = \qty{3.6e-8}{mol/s}$.
(b) $1.95 \times 10^{-5}\,\unit{mol}$; \qty{540}{s}. (c) A casca em $1/r$ leva
$\sim a^2/D \approx \qty{2000}{s}$ para se formar e o grão encolhe; mexer
substitui a casca de espessura $\sim a$ por uma \omterm{prop:b2:viscous-flows:bl}{camada limite} $\delta \ll a$
e multiplica o fluxo por $a/\delta$. (d) $(rn')' = 0$: $n = A + B\ln r$,
que não pode se anular no infinito: não há estado estacionário --- a nuvem de um
cilindro continua crescendo (logaritmicamente).
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:7}
(a) $\dd/\dd t\int n = D[n']_{-\infty}^{\infty} = 0$; $\dd/\dd t\int x^2 n = D\int
x^2 n'' = 2D\int n = 2DN$. (b) $\partial_t[n(x,-t)] = -D\,\partial_x^2 n$: o
sinal se inverte. (c) Para $t < 0$ a ``largura'' $2Dt$ é negativa: não existe tal
estado --- o pulso não pode ser desespalhado. (d) $\dd/\dd t\int n^2 = 2D\int nn''
= -2D\int n'^2 \le 0$.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:11}
(a) $n = n_\infty(1 - a/r)$, $\Phi = 4\pi D a n_\infty$. (b) $3.8 \times
10^6$ por segundo. (c) Os receptores agem como condutâncias em paralelo e,
depois, em série com a casca esférica: $\Phi \approx 4\pi Dan_\infty
\cdot N_{\text{r}}s/(N_{\text{r}}s + \pi a)$; metade do máximo para
$N_{\text{r}}s = \pi a$: com $s = \qty{1}{nm}$, $N_{\text{r}} \approx
3000$, cobrindo $N_{\text{r}}s^2/4a^2 \approx 10^{-3}$ da superfície.
(d) Cada tipo de receptor precisa de uma área desprezível para uma captura
quase máxima: uma célula pode vigiar milhares de substâncias ao mesmo tempo.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:12}
(a) Diferença progressiva em $t$, diferença segunda centrada em $x$. (b)
$\alpha = 1/2$: $n_i^{k+1} = (n_{i-1}^k + n_{i+1}^k)/2$ --- cada partícula salta
para a esquerda ou para a direita com probabilidade $1/2$. (c) Para $n_i = (-1)^i$, $n^{k+1} =
(1 - 4\alpha)n^k$: $|1 - 4\alpha| > 1$ quando $\alpha > 1/2$. (d) $\Delta t \le
\Delta x^2/2D = \qty{3.3}{s}$; cerca de $1100$ passos.
\end{solution}

\begin{solution}{pb:b2:particle-diffusion:1}
\textbf{1.} $k_BT = \qty{0.118}{eV}$, $E_{\text{a}}/k_BT = 29.2$: $D =
\qty{1.6e-17}{m^2/s}$; a \qty{1000}{\celsius}: \qty{1.6e-18}{m^2/s} --- um
fator $10$.

\textbf{2.} A superfície refletora é tratada pela imagem especular do
pulso, o que dobra a amplitude em $x > 0$; $\int_0^\infty = (Q/\sqrt{\pi Dt})
\cdot \tfrac12\sqrt{4\pi Dt} = Q$.

\textbf{3.} $Dt = \qty{5.8e-14}{m^2}$; $\sqrt{\pi Dt} = \qty{0.43}{\micro m}$; $n_{\text{s}}
= \qty{2.3e24}{m^{-3}}$.

\textbf{4.} $\eu^{-x^2/4Dt} = 4.3 \times 10^{-4}$: $x_j^2 = 4Dt \times 7.75$, $x_j =
\qty{1.3}{\micro m}$.

\textbf{5.} $n_{\text{s}}$ cai pela metade, o logaritmo desce a $7.05$, $4Dt$
quadruplica: $x_j = \qty{2.6}{\micro m}$ --- menos que o dobro.

\textbf{6.} $\delta D/D = (E_{\text{a}}/k_BT)\,\delta T/T = 21\%$; $x_j \propto
\sqrt{Dt}$ aproximadamente, de modo que $1\%$ em $x_j$ exige $2\%$ em $D$: \qty{1}{K}.

\textbf{7.} $E_{\text{a}}/k_BT = 134$: $D \approx 10^{-62}\,\unit{m^2/s}$; um
espaçamento (\qty{0.25}{nm}) em $a^2/D \sim 10^{43}\,\unit{s}$: congelado.

\textbf{8.} $E_{\text{a}}/k_BT = 32.8$: $D = \qty{4.3e-19}{m^2/s}$; $\sqrt{Dt} =
\qty{28}{nm}$.

\textbf{9.} $n_0\operatorname{erfc}(x/2\sqrt{Dt})$; $n_B/n_0 = 5 \times 10^{-6}$:
$x \approx 6.4\sqrt{Dt} = \qty{0.18}{\micro m}$.

\textbf{10.} $2n_0\sqrt{Dt/\pi} = \qty{6.3e18}{m^{-2}}$: seis vezes $Q$ ---
a dose cresce como $\sqrt t$ e é fixada pelo tempo.

\textbf{11.} Temperatura baixa: $D$ pequeno, dose controlada pelo tempo,
rasa; temperatura alta: a dose fixa é levada fundo depressa.

\textbf{12.} \qty{0.18}{\micro m} contra \qty{1.3}{\micro m}: fina.

\textbf{13.} $4 \times 10^{18}$ moléculas; $n_{\text{a}} = \qty{2.4e25}{m^{-3}}$.

\textbf{14.} $\ell = \qty{37}{nm}$; $v^* = \qty{206}{m/s}$.

\textbf{15.} $D \approx \qty{2.5e-6}{m^2/s}$.

\textbf{16.} $L^2/D = 10^7\,\unit{s}$, meses; com deriva: \qty{50}{s}.

\textbf{17.} $n = 8N(4\pi Dt)^{-3/2}\eu^{-r^2/4Dt}$; máximo em $t^* = r^2/6D =
\qty{6.7e4}{s}$, dezoito horas.

\textbf{18.} $n_{\max} = 8N(4\pi Dt^*)^{-3/2}\eu^{-3/2} \approx \qty{2e18}{m^{-3}}$,
muito acima do limiar: sente-se o cheiro, mais cedo ou mais tarde.

\textbf{19.} $r^2 = 4Dt\ln 10^6 = \qty{12}{m^2}$: \qty{3.5}{m}.

\textbf{20.} $v^*t/\ell \approx 5 \times 10^{14}$.

\textbf{21.} $\langle x\rangle = 0$, $\langle x^2\rangle = N\ell^2 = \ell^2 t/\tau$;
$D = \ell^2/2\tau$.

\textbf{22.} $\binom{N}{N/2}/2^N$: $1/2$, $3/8$, $5/16$ --- decrescente
(como $1/\sqrt{\pi N/2}$): o caminhante se afasta como $\sqrt N$.

\textbf{23.} A nuvem se junta num ponto --- nunca visto; isso
viola a \omterm{thm:b2:particle-diffusion:equation}{equação de difusão}, não as leis da mecânica.

\textbf{24.} Cada colisão é reversível, mas o estado inicial (todas as
partículas juntas) é excepcional: quase toda história microscópica
a partir dele se espalha, e a inversa exigiria uma conspiração de todas as
velocidades. A média do passeio aleatório guarda só o que é típico e
descarta essa informação; a irreversibilidade é estatística.

\textbf{25.} $L \sim \sqrt{Dt}$, $t \sim L^2/D$; $D \sim \ell v^*/3$, ou seja,
o passo entre colisões: $10^{-5}$, $10^{-9}$, $10^{-17}\,\unit{m^2/s}$.
\end{solution}
