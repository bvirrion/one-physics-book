\begin{solution}{exo:b2:boltzmann-factor:6}

\begin{solution}{exo:b2:boltzmann-factor:10}
(a) $n = n_0\eu^{m\omega^2r^2/2k_BT}$. (b) $\eu^{\Delta m\omega^2r^2/2k_BT} = \eu^{0.24} =
1.27$. (c) $H = k_BT/m'g = \qty{16}{\micro m}$; contar as esferas em
várias alturas dá $k_B$, e daí $N_A = R/k_B$. (d) Os íons seguem
$\eu^{\mp e\varphi/k_BT}$ em torno de uma carga; linearizado na \omterm{prop:b2:maxwell-equations:potentials}{equação de Poisson}
isso dá $\varphi'' = \varphi/\lambda_{\text{D}}^2$: o potencial fica blindado
além de $\lambda_{\text{D}}$.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:11}
(a) $Z = 1/(1 - \eu^{-x})$, $\langle E\rangle = h\nu/(\eu^x - 1)$. (b) $k_BT$:
equipartição; $h\nu\eu^{-h\nu/k_BT}$: congelado. (c) $u_\nu = (8\pi\nu^2/c^3)\langle E
\rangle$; Rayleigh--Jeans dá $k_BT$ a cada modo. (d) Um modo não pode ser
excitado por menos que um quantum.
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:12}
(a) Apenas $x = \mu B/k_BT$; $\ln 2$ e $0$. (b) $x = 0.67$: $S/Nk_B = 0.51$, logo
$0.18\,k_B$ retirados por spin. (c) $B/T$ constante: \qty{10}{mK}. (d) O campo
próprio dos spins (militesla) substitui $B$ no fim e limita o resfriamento;
medir $M = n\mu\tanh(\mu B/k_BT)$ dá $T$.
\end{solution}

\begin{solution}{pb:b2:boltzmann-factor:1}
\textbf{1.} $\dd P/\dd z = -nmg = -(mg/k_BT)P$; $H = \qty{8.4}{km}$.

\textbf{2.} $\eu^{-mgz/k_BT}$: a energia potencial de uma molécula, a
temperatura do ar.

\textbf{3.} \qty{0.70}{}, \qty{0.52}{}, \qty{0.35}{bar}; $H\ln 2 = \qty{5.8}{km}$.

\textbf{4.} $P_0/g = \qty{1.03e4}{kg/m^2}$; $\times 4\pi R^2$: \qty{5.3e18}{kg}; $\rho_0H =
1.22 \times 8400 = \qty{1.03e4}{kg/m^2}$.

\textbf{5.} $5.3 \times 10^{18}/(29 \times 10^{-3}/N_A) = 1.1 \times 10^{44}$.

\textbf{6.} $H_{\text{O}_2} = 7.6$, $H_{\text{N}_2} = \qty{8.7}{km}$: a razão a \qty{50}{km}
seria $\eu^{-50/7.6 + 50/8.7} = 0.44$ do valor no solo; a turbulência mistura
mais depressa do que a difusão separa.

\textbf{7.} Menor: o ar mais frio é mais denso e a pressão cai mais depressa.

\textbf{8.} \qty{2.5e25}{m^{-3}}; $\eu^{-11.9}$: \qty{1.7e20}{m^{-3}} --- não há borda, apenas uma
convenção.

\textbf{9.} $11.2\sqrt{6370/6870} = \qty{10.8}{km/s}$.

\textbf{10.} Hélio \qty{2040}{}, \qty{2300}{m/s}; nitrogênio \qty{770}{}, \qty{870}{m/s}.

\textbf{11.} $x = 5.3$: $4 \times 10^{-12}$; nitrogênio $\eu^{-196}$: nenhum.

\textbf{12.} $\tfrac14 \times 10^{12} \times 2300 \times 4 \times 10^{-12} \times \tfrac12 \approx
10^3$ átomos por metro quadrado e por segundo; para a Terra inteira e num ano,
$2 \times 10^{25}$ átomos --- \qty{0.1}{kg}.

\textbf{13.} $5 \times 10^{-6} \times 1.1 \times 10^{44} = 5.5 \times 10^{38}$ átomos, \qty{3.7e12}{kg};
frente a \qty{0.1}{kg/yr} o tempo de residência seria absurdo.

\textbf{14.} O regime estacionário exige uma perda de \qty{3e6}{kg/yr}: um tempo
de residência de cerca de um milhão de anos, que é o valor aceito --- logo, a
estimativa térmica a \qty{1000}{K} é pequena demais: o escape é
dominado por episódios mais quentes e por processos não térmicos (iônicos).

\textbf{15.} $x = 3.75$: $\eu^{-14}$ em vez de $\eu^{-28}$ --- um fator $10^6$:
a perda é extraordinariamente sensível à temperatura da exosfera.

\textbf{16.} A Lua é quente e leve: tudo escapa em
tempo geológico; Titã é frio, com $x = 11$ para o nitrogênio.

\textbf{17.} $p_\pm = \eu^{\pm x}/2\cosh x$; $M = n\mu\tanh x$.

\textbf{18.} $x = \mu B/k_BT$: \qty{4.2}{A/m}, \qty{310}{A/m}, \qty{1.6e4}{A/m}
(saturação \qty{1.9e4}{A/m}); Curie para $x \ll 1$, ou seja, $T \gg \qty{70}{mK}$.

\textbf{19.} $M$ só depende de $T$ uma vez conhecido $B$; mais inclinada perto de
$x \approx 1$, $T \approx \mu B/k_B = \qty{70}{mK}$; na prática mede-se a
suscetibilidade com uma bobina.

\textbf{20.} $\langle E\rangle = -N\mu B\tanh x$; $C = Nk_Bx^2/\cosh^2x$, com pico em
$x = 1.2$: \qty{56}{mK}.

\textbf{21.} $Nk_B\ln 2$ e $0$: todos os spins se alinham com o campo.

\textbf{22.} \qty{10}{mK}; o campo próprio dos spins, de um militesla, substitui $B$
e fixa o piso.

\textbf{23.} $x = 1$ em $\mu_{\text{N}}B/k_B = \qty{0.37}{mK}$: Curie vale até um
milikelvin, e o termômetro é útil até microkelvins.

\textbf{24.} $\Delta\nu/\nu = \sqrt{8k_BT\ln2/mc^2} \propto \sqrt T$: a largura
de uma linha mede a temperatura do gás.

\textbf{25.} Atmosfera, gás que escapa, spins: $\eu^{-E/k_BT}$ com $E = mgz$,
$\tfrac12mv^2$, $\mp\mu B$.
\end{solution}
