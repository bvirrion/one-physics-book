\section*{Capítulo \ref{ch:b2:viscous-flows} --- Escoamentos viscosos}

\begin{solution}{exo:b2:viscous-flows:1}
$\tau = \eta U/h = 0.25 \times 2/2 \times 10^{-4} = \qty{2.5}{kPa}$; $F = \qty{1.25}{kN}$;
$P = Fv = \qty{2.5}{kW}$. Atrito seco: $0.3 \times 98 = \qty{29}{N}$ --- aqui o
filme fino, fortemente cisalhado, resiste muito mais; o arrasto viscoso cresce como
$U/h$, de modo que a lubrificação compensa a velocidades menores e com filmes mais espessos (e
nunca gripa).
\end{solution}

\begin{solution}{exo:b2:viscous-flows:3}
$D_V = \pi R^4\Delta P/8\eta L = \pi(2 \times 10^{-4})^4 \times 2 \times 10^4/(8 \times 10^{-3}
\times 0.03) = \qty{0.42}{mL/s}$; \qty{12}{s} para \qty{5}{mL}. $v_{\max} = \Delta PR^2/
4\eta L = \qty{6.7}{m/s}$, $\langle v\rangle = \qty{3.3}{m/s}$, $\mathrm{Re} = 2R\langle v
\rangle/\nu = 1300$: laminar. $\tau_w = \Delta PR/2L = \qty{67}{Pa}$. A \qty{0.8}{mm}:
dezesseis vezes a vazão, \qty{6.7}{mL/s} --- mas $\mathrm{Re} \approx 10^4$: o escoamento
torna-se turbulento e o ganho é menor.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:4}
$v_\infty = 2R^2\Delta\rho\,g/9\eta$. (a) \qty{1.3}{cm/s}, $\mathrm{Re} = 0.02$, \qty{75}{s}
por metro. (b) \qty{0.24}{mm/s}, $\mathrm{Re} = 3 \times 10^{-5}$, \qty{70}{min} por
metro. (c) \qty{1.2}{m/s}, mas $\mathrm{Re} = 16$: Stokes superestima (a celeridade
real é de cerca de \qty{0.7}{m/s}), cerca de um segundo por metro.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:5}
(a) $\langle v\rangle = 0.1/0.0707 = \qty{1.4}{m/s}$, $\mathrm{Re} = 4.2 \times 10^5$. (b)
$8\eta LD_V/\pi R^4 = \qty{500}{Pa}$. (c) $\lambda = 0.3/25.5 = 0.0118$, $\Delta P = 0.0118
\times 3333 \times 1000 = \qty{39}{kPa}$: $80$ vezes mais; $P = \Delta PD_V = \qty{3.9}{kW}$.
(d) Mesma celeridade, $\mathrm{Re} = 3 \times 10^5$, $\lambda = 0.0128$, $\Delta P = 0.0128 \times 4717
\times 1000 = \qty{60}{kPa}$, \qty{6}{kW}: mais parede por unidade de vazão, mais perda.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:6}
(a) $v = 2 \times 25 \times 10^{-12} \times 50 \times 9.81/(9 \times 10^{-3}) = \qty{2.7}{\micro m/s}$;
\qty{5}{cm} em $1.8 \times 10^4$ s, cinco horas. (b) \qty{2.7}{mm/s}, \qty{18}{s}. (c)
$\Omega = \sqrt{1000g/r} = \qty{313}{rad/s} = \qty{3000}{rpm}$. (d) $v \propto R^2$: as partículas

\end{solution}

\begin{solution}{exo:b2:viscous-flows:7}
(a) $F = 0.6 \times 144 \times 0.35 = \qty{30}{N}$, $+\qty{4}{N}$: $P = 34 \times 12 =
\qty{410}{W}$. (b) $47 + 4 = \qty{51}{N}$, \qty{770}{W}. (c) Celeridade relativa

Arrasto de $\qty{21}{N}$: \qty{300}{W}, com economia de \qty{110}{W}. (e) $mg\sin\alpha = \qty{39}{N} =

\end{solution}

\begin{solution}{exo:b2:viscous-flows:8}
(a) Golfe: $\tfrac12\rho C_xS = \qty{2.2e-4}{kg/m}$, $v_\infty = \sqrt{0.45/2.2 \times 10^{-4}}
= \qty{45}{m/s}$; paraquedista, $\sqrt{785/0.42} = \qty{43}{m/s}$; com paraquedas, $\sqrt{785/15} =
\qty{7.2}{m/s}$. (b) $m\dot v = mg - kv^2$, ou seja, $\dot v = g(1 - v^2/v_\infty^2)$,
cuja solução a partir do repouso é $v_\infty\tanh(gt/v_\infty)$. (c) $\tanh x = 0.9$ em
$x = 1.47$: $t = 1.47 \times 43/9.81 = \qty{6.5}{s}$; $z = (v_\infty^2/g)\ln\cosh x =
190 \times 0.83 = \qty{160}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:9}
(a) Taxa de cisalhamento $\Omega R/e = 0.5/10^{-3} = \qty{500}{s^{-1}}$; $\Gamma = \tau\cdot2\pi Rh
\cdot R$: $\tau = 0.02/(2\pi \times 2.5 \times 10^{-3} \times 0.1) = \qty{12.7}{Pa}$. (b) $\eta
= 12.7/500 = \qty{2.5e-2}{Pa.s}$. (c) $\mathrm{Re} = \rho Ue/\eta \approx 1000 \times 0.5 \times
10^{-3}/0.025 = 20$: laminar. (d) Uma folga estreita torna a taxa de cisalhamento uniforme
(a aproximação plana) e mantém o \omterm{def:b2:viscous-flows:reynolds}{escoamento laminar}.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:10}
$D_V = \qty{8.3e-5}{m^3/s}$. (a) $R_h = 8\eta L/\pi R^4 = 8 \times 3.5 \times 10^{-3} \times
0.4/(\pi \times 2.4 \times 10^{-8}) = \qty{1.5e5}{Pa.s/m^3}$, $\Delta P = \qty{12}{Pa}$. (b)
Um capilar, $\qty{3.5e16}{Pa.s/m^3}$; $10^{10}$ em paralelo: \qty{3.5e6}{Pa.s/m^3};
$\Delta P = \qty{290}{Pa}$ (\qty{2}{mmHg}). (c) Uma arteríola, $8 \times 3.5 \times 10^{-3}
\times 2 \times 10^{-3}/(\pi \times 10^{-20}) = \qty{1.8e15}{Pa.s/m^3}$; o conjunto:
\qty{1.8e7}{Pa.s/m^3}; $\Delta P = \qty{1.5}{kPa}$: as arteríolas dominam as
duas outras (é nos vasos pequenos, e não nos grandes, que a pressão
cai). (d) $P = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, cerca de $1\%$. (e)
$(1/1.1)^4 = 0.68$: um terço da resistência arteriolar removido ---
é assim que os vasodilatadores baixam a pressão arterial.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:11}
(a) $F = b\int_0^L\tau_w\,\dd x = 0.332\rho U^2b\sqrt{\nu/U}\int_0^L\dd x/\sqrt x =
0.664\rho U^2b\sqrt{\nu L/U} = 0.664\rho U^2bL/\sqrt{\mathrm{Re}_L}$. (b) $\mathrm{Re}_L =
3 \times 10^5$ (laminar); $\delta \approx 5\sqrt{\nu L/U} = \qty{9}{mm}$; $F = 0.664 \times 1000
\times 0.09 \times 1/548 = \qty{0.11}{N}$ por face, \qty{0.22}{N}. (c) $\mathrm{Re}_L = 10^7$:
a camada é turbulenta além dos primeiros centímetros, e a fórmula
laminar falha (o arrasto real é várias vezes maior). (d) $\mathrm{Re}_x =
10^9$: $\delta \approx 0.37 \times 100/63 = \qty{0.6}{m}$.
\end{solution}

\begin{solution}{exo:b2:viscous-flows:12}
(a) $0 = \rho g + \eta v''(y)$, $v(0) = 0$ (aderência), $v'(h) = 0$ (sem tensão na
superfície livre). (b) Integre duas vezes: $v = (\rho g/2\eta)(2hy - y^2)$; $q =
\int_0^hv\,\dd y = \rho gh^3/3\eta$. (c) $v_s = \rho gh^2/2\eta = 1200 \times 9.81 \times
4 \times 10^{-8}/1 = \qty{0.47}{mm/s}$; $q = \qty{6.3e-8}{m^2/s}$; \qty{28}{cm} em dez
minutos --- uma tinta newtoniana com essa espessura escorreria toda; tintas
reais afinam sob cisalhamento e são tixotrópicas, e escorrem mesmo assim quando grossas demais,
porque $v_s \propto h^2$. (d) $1400 \times 9.81 \times 4 \times 10^{-6}/20 = \qty{2.7}{mm/s}$.
\end{solution}

\begin{solution}{pb:b2:viscous-flows:1}
\textbf{1.} $S = \qty{4.9}{cm^2}$, $\langle v\rangle = 83/4.9 = \qty{0.17}{m/s}$; $\mathrm{Re}
= 1060 \times 0.17 \times 0.025/3.5 \times 10^{-3} = 1300$: laminar (os picos pulsáteis
chegam a $\sim 5000$).

\textbf{2.} $v_{\max} = \qty{0.34}{m/s}$; $\tau_w = 4\eta\langle v\rangle/R = \qty{0.19}{Pa}$.

\textbf{3.} $\mathrm{Re} = 1060 \times 3 \times 10^{-4} \times 8 \times 10^{-6}/3.5 \times 10^{-3} =
7 \times 10^{-4}$; $\Delta P = 8\eta L\langle v\rangle/R^2 = 8 \times 3.5 \times 10^{-3} \times 10^{-3}
\times 3 \times 10^{-4}/1.6 \times 10^{-11} = \qty{525}{Pa} = \qty{3.9}{mmHg}$.

\textbf{4.} $1/0.3 = \qty{3.3}{s}$: tempo suficiente para o oxigênio difundir os poucos
micrômetros até o tecido (\cref{ch:b2:particle-diffusion}).

\textbf{5.} Uma arteríola, $8\eta L/\pi R^4 = \qty{1.8e15}{Pa.s/m^3}$; $10^8$ em
paralelo: \qty{1.8e7}{Pa.s/m^3}; $\Delta P = 1.8 \times 10^7 \times 8.3 \times 10^{-5} =
\qty{1.5}{kPa} = \qty{11}{mmHg}$.

\textbf{6.} $P = \Delta P\,D_V = 13000 \times 8.3 \times 10^{-5} = \qty{1.1}{W}$, cerca de
$1\%$ da potência do corpo.

\textbf{7.} $(1/0.6)^4 = 7.7$. Em repouso os vasos a jusante se dilatam e
compensam; no esforço a demanda sobe quatro ou cinco vezes e a
artéria estreitada não consegue atendê-la: angina.

\textbf{8.} Cinco vezes a celeridade: $\mathrm{Re} \approx 6500$, turbulento --- são os
sopros que um estetoscópio ouve durante o exercício.

\textbf{9.} No escoamento lento e pouco cisalhado das vênulas e dos capilares
as hemácias se agregam e a \omterm{def:b2:viscous-flows:viscosity}{viscosidade} aparente sobe (o contrário
na aorta rápida); todas as estimativas de queda de pressão nos vasos pequenos
são limites inferiores.

\textbf{10.} $\langle v\rangle = 3/1.13 = \qty{2.65}{m/s}$; $\mathrm{Re} = 850 \times 2.65 \times
1.2/0.02 = 1.35 \times 10^5$: turbulento.

\textbf{11.} $\lambda = 0.316/19.2 = 0.0165$; $\Delta P/L = (0.0165/1.2) \times \tfrac12 \times
850 \times 7.0 = \qty{41}{Pa/m} = \qty{41}{kPa/km}$; \qty{534}{bar} ao longo da linha.

\textbf{12.} $534/80 = 6.7$: pelo menos sete estações.

\textbf{13.} $P = \Delta P\,D_V = 5.34 \times 10^7 \times 3 = \qty{160}{MW}$;
\qty{53}{MJ/m^3}, $0.15\%$ da energia do óleo.

\textbf{14.} $\Delta P = 32\eta L\langle v\rangle/D^2 = 32 \times 0.02 \times 1.3 \times 10^6
\times 2.65/1.44 = \qty{15}{bar}$: trinta e cinco vezes menos. A turbulência é
cara; algumas partes por milhão de polímeros longos a retardam e cortam
a queda em dezenas por cento.

\textbf{15.} $\mathrm{Re} = 1350$: laminar; por Poiseuille, $\Delta P = 32 \times 2 \times 1.3 \times
10^6 \times 2.65/1.44 = \qty{1500}{bar}$ --- impossível; o óleo precisa ser mantido
quente, daí o isolamento e o aquecimento.

\textbf{16.} $1.3 \times 10^6/2.65 = \qty{4.9e5}{s} = \qty{5.7}{dias}$.

\textbf{17.} $\tau_w = (\Delta P/L)D/4 = 41 \times 0.3 = \qty{12}{Pa}$; $F = \tau_w\pi DL =
\qty{6e7}{N}$.

\textbf{18.} $\Delta P \propto \lambda\langle v\rangle^2/D \propto D^{-4.75}$: $1.2^{4.75} = 2.4$.

\textbf{19.} $v = 2R^2\Delta\rho\,g/9\eta = 2 \times 2.5 \times 10^{-9} \times 1800 \times 9.81/
0.18 = \qty{0.49}{mm/s}$; $\mathrm{Re} = 2 \times 10^{-3}$; \qty{41}{min} para atravessar
\qty{1.2}{m}. Flutuações turbulentas de alguns centímetros por segundo
o mantêm em suspensão enquanto o óleo corre; numa linha parada ele decanta em
menos de uma hora.

\textbf{20.} $v = 2 \times 2.5 \times 10^{-11} \times 1000 \times 9.81/1.62 \times 10^{-4} =
\qty{3.0}{mm/s}$; $\mathrm{Re} = 2 \times 10^{-3}$; \qty{1}{km} em $3.3 \times 10^5$ s, quatro
dias: qualquer corrente ascendente acima de alguns milímetros por segundo a sustenta.

\textbf{21.} \qty{1.2}{m/s}, $\mathrm{Re} = 16$: Stokes superestima em cerca de um
fator dois (real: \qty{0.7}{m/s}).

\textbf{22.} Stokes daria \qty{120}{m/s} e $\mathrm{Re} \sim 10^4$: absurdo.
Arrasto quadrático, $\tfrac43\pi R^3\rho_wg = \tfrac12\rho v^2\pi R^2C_x$: $v = \sqrt{8R
\rho_wg/3\rho C_x} = \qty{6.6}{m/s}$ ($\mathrm{Re} = 900$, coerente); \qty{150}{s} desde
\qty{1}{km}.

\textbf{23.} \qty{1}{cm/s} $>$ \qty{3}{mm/s}: a gotícula sobe; a gota de garoa
precisa de cerca de \qty{1}{m/s}, que as nuvens convectivas fornecem.

\textbf{24.} $2\gamma/R$: \qty{29}{kPa}, \qty{1.4}{kPa}, \qty{144}{Pa}; pressões
dinâmicas $\tfrac12\rho v^2$: $5 \times 10^{-6}$, $0.3$ e \qty{26}{Pa}. A razão
chega a $0.2$ em \qty{2}{mm}; em \qty{5}{mm} ($v \approx \qty{9}{m/s}$, $\tfrac12\rho v^2 =
\qty{49}{Pa}$ contra \qty{58}{Pa}) a pressão aerodinâmica achata a
gota e a rompe: nenhuma gota de chuva passa de cerca de \qty{5}{mm}.

\textbf{25.} Aorta, $10^3$ (laminar, perto da transição); capilar, $10^{-3}$
(Poiseuille); arteríola, $10^{-2}$ (Poiseuille); oleoduto quente, $10^5$
(turbulento, $\lambda(\mathrm{Re})$); oleoduto frio, $10^3$ (Poiseuille); areia no
óleo, $10^{-3}$ (Stokes); gotícula de nuvem, $10^{-3}$ (Stokes); gota de chuva, $10^3$
(arrasto quadrático, $C_x \approx 0.5$).
\end{solution}
