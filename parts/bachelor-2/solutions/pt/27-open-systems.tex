\begin{solution}{exo:b2:open-systems:10}
(a) $\dd(mu)/\dd t = \dot m h_0$ integra-se em $mu = mh_0$: $c_vT = c_pT_0$,
$T = \gamma T_0$. (b) \qty{410}{K}. (c) O gás final é uma mistura do
gás original a $T_0$ e do gás que entra aquecido a $\gamma T_0$, logo
fica entre os dois. (d) O trabalho de escoamento aquece o gás; um enchimento lento num banho
de água deixa o calor sair, de modo que a pressão a \qty{20}{\celsius} seja a
desejada.
\end{solution}

\begin{solution}{pb:b2:open-systems:1}
\textbf{1.} Compressor (adiabático, $w = \Delta h$), condensador (isobárico, calor
para fora), \omterm{prop:b2:open-systems:devices}{estrangulamento} (isentálpico), evaporador (isobárico, calor para dentro).

\textbf{2.} $h_4 = \qty{256}{kJ/kg}$, $x_4 = 0.28$.

\textbf{3.} \qty{143}{kJ/kg}.

\textbf{4.} \qty{27}{} e \qty{34}{kJ/kg}; $h_2 = 433$; $T_2 \approx 45 + 7 \approx
\qty{52}{\celsius}$.

\textbf{5.} \qty{177}{kJ/kg}; $\mathrm{COP}_{\text{q}} = 5.2$, $\mathrm{COP}_{\text{f}} = 4.2$;
$5.2 = 4.2 + 1$.

\textbf{6.} $7.8$ e $6.8$; dois terços.

\textbf{7.} \qty{0.045}{kg/s}; \qty{1.5}{kW}, \qty{1.7}{kW} na tomada.

\textbf{8.} Compressor: $c_p\ln(325/318) \approx \qty{0.022}{kJ/kg/K}$;
\omterm{prop:b2:open-systems:devices}{estrangulamento}: $s_4 - s_3 = 1.204 - 1.19 = \qty{0.014}{kJ/kg/K}$ --- o compressor.

\textbf{9.} $313/55 = 5.7$; a razão de pressões quase dobra (mais trabalho
por quilograma) e a densidade do vapor cai pela metade (menos massa por curso):
a capacidade despenca justo quando a casa precisa de mais.

\textbf{10.} A serpentina fica abaixo de \qty{0}{\celsius} em ar úmido; a geada
a isola; o degelo inverte o ciclo por alguns minutos a cada hora e
custa alguns por cento.

\textbf{11.} $308/35 = 8.8$ contra $328/55 = 6.0$: quanto menor a
temperatura do condensador, melhor --- o piso aquecido.

\textbf{12.} \qty{1}{kWh} de combustível $\to$ \qty{0.4}{kWh} de eletricidade $\to$
\qty{1.6}{kWh} de calor, contra \qty{0.9}{kWh} da caldeira.

\textbf{13.} Bomba, caldeira (aquecimento isobárico, vaporização, superaquecimento),
turbina, condensador.

\textbf{14.} \qty{6}{kJ/kg}; $h_2 = 144$.

\textbf{15.} $3278 = 1069 + 1571 + 638$ (\unit{kJ/kg}).

\textbf{16.} $x = 0.81$, $h_{4s} = 2098$, $w_{\text{s}} = \qty{1324}{kJ/kg}$.

\textbf{17.} \qty{1125}{kJ/kg}, $h_4 = 2297$, $x_4 = 0.89$; um vapor mais úmido erode as
pás e reduz o rendimento.

\textbf{18.} $\eta = 1119/3278 = 0.34$; Carnot $0.60$; $\bar T = 3278/6.40 =
\qty{512}{K}$, o que dá $0.40$.

\textbf{19.} \qty{536}{kg/s}; \qty{1160}{MW}; \qty{28}{m^3/s} de água de rio,
ou \qty{480}{kg/s} evaporados.

\textbf{20.} \qty{0.65}{kJ/kg/K}, \qty{350}{kW/K}; \qty{200}{kJ/kg}, $18\%$ do
trabalho da turbina.

\textbf{21.} O segundo fornecimento de calor se dá a alta temperatura (um $\bar T$
maior), e a segunda expansão começa superaquecida e termina mais seca.

\textbf{22.} $1757\,\unit{MW} \times (1/512 - 1/1500) = \qty{2.3}{MW/K}$ --- seis vezes
a da turbina.

\textbf{23.} $1160\,\unit{MW} \times (1/293 - 1/306) = \qty{0.17}{MW/K}$.

\textbf{24.} Fornalha $\gg$ turbina $>$ condensador: daí temperaturas e pressões
de vapor mais altas, reaquecimento e regeneração; depois, pás melhores;
o condensador vem por último.

\textbf{25.} Desenhe o \omterm{def:b2:open-systems:cv}{volume de controle}; a massa que entra é igual à que sai; $\Delta h +
\Delta c^2/2 + g\Delta z = w_{\text{u}} + q$ com $h$ vindo de $c_p$, de tabelas ou
de $v\Delta P$; $\Delta s = \sum q/T + s_{\text{c}}$ com referências isentrópicas e
rendimentos; monte os componentes e compare com Carnot.
\end{solution}
