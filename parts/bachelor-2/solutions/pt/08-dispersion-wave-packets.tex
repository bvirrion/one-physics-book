\begin{solution}{exo:b2:dispersion-wave-packets:1}
(a) $k = \qty{0.063}{rad/m}$, $\omega = \sqrt{gk} = \qty{0.79}{rad/s}$, $T = \qty{8.0}{s}$;
$v_\varphi = \qty{12.5}{m/s}$, $v_g = \qty{6.3}{m/s}$. (b) $\omega = \qty{0.45}{rad/s}$, $k = \omega^2
/g = \qty{0.021}{rad/m}$, $\lambda = \qty{310}{m}$; $v_\varphi = gT/2\pi = \qty{22}{m/s}$, $v_g =
\qty{11}{m/s}$; $5 \times 10^6/11 = \qty{4.6e5}{s} = 5.3$ dias. (c) $\lambda = \qty{6}{m}$, $v_g
= \qty{1.6}{m/s}$: seriam necessárias cinco semanas, e as ondas curtas são amortecidas
pela \omterm{def:b2:viscous-flows:viscosity}{viscosidade} e quebradas pelo vento muito antes.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:2}
$\omega = \qty{1.26e8}{rad/s}$, $\omega_c = \qty{6.3e7}{rad/s}$: $k = \sqrt{\omega^2 - \omega_c^2}/c =
\qty{0.36}{rad/m}$, $v_\varphi = \qty{3.5e8}{m/s}$, $v_g = c^2k/\omega = \qty{2.6e8}{m/s}$,
produto $c^2$. A \qty{5}{MHz}, $\omega < \omega_c$: $\kappa = \sqrt{\omega_c^2 - \omega^2}/c =
\qty{0.18}{m^{-1}}$, profundidade \qty{5.5}{m}.
\end{solution}

\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:5}
(a) $\omega^2 = c^2k^2 + (EI/\mu)k^4$; $v_\varphi = \sqrt{c^2 + (EI/\mu)k^2}$; $v_g = (c^2k +
2(EI/\mu)k^3)/\omega$. (b) $c^2 = \qty{1.12e5}{m^2/s^2}$, $k = \qty{66}{rad/m}$, $(EI/\mu)k^2
= \qty{7.0e3}{m^2/s^2}$: $v_\varphi/c = \sqrt{1.063} = 1.031$, $+3.1\%$ --- os $53$ cents do
\cref{ch:b2:waves-on-strings}. (c) $v_g > v_\varphi$: anômala; os harmônicos
agudos correm à frente e o ataque de um beliscão vira um breve
gorjeio descendente.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:6}
(a) $-\mu\omega^2 + \iu\alpha\omega = -Tk^2$: $\underline k^2 = (\omega^2/c^2)(1 - \iu\alpha/\mu\omega)$.
(b) $\underline k \approx (\omega/c)(1 - \iu\alpha/2\mu\omega)$: $k' = \omega/c$, $k'' = \alpha/2\mu c =
\alpha/2Z$; $8.7k''$ dB/m. (c) $c = \qty{100}{m/s}$, $Z = \qty{0.5}{kg/s}$, $\mu\omega = 3.1 \gg
0.02$: $k'' = \qty{0.02}{m^{-1}}$, $\delta = \qty{50}{m}$, \qty{0.17}{dB/m}. (d) Não neste
limite ($k' = \omega/c$); quando $\alpha \gtrsim \mu\omega$ a raiz quadrada deixa de
se linearizar e $k'$ depende não linearmente de $\omega$: dispersiva (o
limite difusivo do cabo).
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:7}
(a) $v_\varphi = \sqrt{\gamma k/\rho}$, $v_g = \tfrac32v_\varphi$: o grupo ultrapassa as
cristas. (b) $v_\varphi^2 = g/k + \gamma k/\rho$ é mínima em $k_m = \sqrt{\rho g/\gamma}$:
$\lambda_m = 2\pi\sqrt{\gamma/\rho g} = \qty{1.7}{cm}$, $v_{\min} = (4g\gamma/\rho)^{1/4} =
\qty{23}{cm/s}$. (c) Uma esteira é um padrão estacionário em relação ao
barco, feito de ondas cuja \omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} iguala a celeridade dele; nenhuma é
mais lenta que \qty{23}{cm/s}. (d) Regime capilar, $v_g > v_\varphi$: as cristas
nascem na frente do anel e morrem atrás.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:8}
(a) $k = \qty{0.126}{rad/m}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-18}{m^2/s}$: $t_{\text{al}}
= 2.5 \times 10^5/18 = \qty{1.4e4}{s}$, quatro horas. (b) \qty{10}{ps}: $t_{\text{al}} =

\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:9}
(a) $\partial_x^2v = R\Gamma\,\partial_tv$, $D = 1/R\Gamma = \qty{1.1e12}{m^2/s}$. (b) $\underline k^2
= -\iu R\Gamma\omega$: $k' = k'' = \sqrt{R\Gamma\omega/2}$; a \qty{1}{Hz}: $k = \qty{1.7e-6}{m^{-1}}$,
$v_\varphi = \qty{3.7e6}{m/s}$, $\delta = \qty{600}{km}$; a \qty{10}{Hz}: $v_\varphi = \qty{1.2e7}{m/s}$,
$\delta = \qty{190}{km}$. (c) $R\Gamma L^2 = 9 \times 10^{-13} \times 9 \times 10^{12} = \qty{8}{s}$
por ponto: cerca de uma palavra por minuto. (d) $\Lambda = R\Gamma/G = \qty{9}{H/km}$ --- uma
indutância absurda para um cabo submarino; as bobinas de carga funcionavam em linhas
telefônicas terrestres, cujos $G$ e $R\Gamma$ eram diferentes.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:10}
(a) $\partial_te = \dot y(\mu\ddot y + Ky) + Ty'\dot y' = \dot yTy'' + Ty'\dot y' = \partial_x
(Ty'\dot y) = -\partial_x\mathcal P$. (b) $\langle e\rangle = \tfrac14A^2(\mu\omega^2 + Tk^2 + K) =
\tfrac12\mu\omega^2A^2$ (usando $\mu\omega^2 = Tk^2 + K$); $\langle\mathcal P\rangle = \tfrac12Tk\omega
A^2$. (c) Razão $Tk/\mu\omega = c^2k/\omega = v_g$. (d) $y = A\eu^{-\kappa x}\cos\omega t$:
$\mathcal P \propto \sin2\omega t$, de média nula --- nenhuma energia atravessa o meio.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:11}
(a) $\tanh kh \to 1$: $\omega^2 = gk$; com $kh \ll 1$: $\omega = \sqrt{gh}\,k$, celeridade $\sqrt{gh}$,
não dispersiva. (b) $kh = 0.13$: águas rasas; $c = \sqrt{9.81 \times 4000} = \qty{198}{m/s}$,
$T = \qty{17}{min}$; \qty{8000}{km} em \qty{11}{h}. (c) $\tfrac12\rho gA^2 = \qty{1.2}{kJ/m^2}$
sobre $200 \times 1000$ km$^2$: \qty{2.5e14}{J}. Green: $A \propto h^{-1/4}$, $(4000/10)^{1/4}
= 4.5$: \qty{2.2}{m}, antes de empinar e galgar a costa. (d) Em $h \approx \lambda/2 =
\qty{50}{m}$; a parte de uma crista que está em água mais rasa desacelera, de modo que a
crista gira até ficar paralela às curvas de nível do fundo --- refração.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:12}
(a) Em $x = \ell$: $v = v_i + v_r$, $i = (v_i - v_r)/Z_c$, $v = R_Li$: $r = (R_L -
Z_c)/(R_L + Z_c)$. (b) $0$, $-1$, $+1$, $0.5$. (c) A entrada da linha parece um
$Z_c$, de modo que o osciloscópio vê \qty{0.5}{V} em $t = 0$ e depois o eco $0.5r$ V em
$2\ell/c = \qty{1}{\micro s}$: nada, \qty{-0.5}{V}, \qty{+0.5}{V}, \qty{+0.25}{V}; o
gerador casado absorve o eco, e por isso não há um segundo. (d)
$\tfrac12 \times 2 \times 10^8 \times 1.8 \times 10^{-6} = \qty{180}{m}$.
\end{solution}

\begin{solution}{pb:b2:dispersion-wave-packets:1}
\textbf{1.} $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi = -\Gamma\partial_tv$; com
$\eu^{\iu(\omega t - kx)}$: $\iu kv = (\iu\omega\Lambda + R)i$, $\iu ki = \iu\omega\Gamma v$, donde
$k^2 = \omega\Gamma(\omega\Lambda - \iu R)$.

\textbf{2.} $\omega = R/\Lambda = \qty{4000}{rad/s}$, \qty{640}{Hz}: a alguns hertz $\Lambda\omega
\ll R$, regime difusivo.

\textbf{3.} $k' = k'' = \sqrt{R\Gamma\omega/2}$ com $R\Gamma = \qty{5e-13}{s/m^2}$; a \qty{1}{Hz}:
$\qty{1.25e-6}{m^{-1}}$, $v_\varphi = \qty{5e6}{m/s}$, $\delta = \qty{800}{km}$; a \qty{4}{Hz}:
$\qty{2.5e-6}{m^{-1}}$, $\qty{1e7}{m/s}$, \qty{400}{km}.

\textbf{4.} $8.7L/\delta$: \qty{38}{dB} a \qty{1}{Hz}, \qty{76}{dB} a \qty{4}{Hz}: só as
componentes mais lentas sobrevivem; um ponto nítido chega como uma corcova longa e baixa.

\textbf{5.} $R\Gamma L^2 = 5 \times 10^{-13} \times 1.2 \times 10^{13} = \qty{6}{s}$: um ponto a cada
seis segundos, uma palavra por minuto (o cabo real, com galvanômetros
de espelho sensíveis, conseguia algumas).

\textbf{6.} $\Lambda = R/\omega = \qty{80}{mH/km}$. Então $k^2 = \Gamma R\omega(1 - \iu)$, $k = \sqrt{\Gamma R\omega}
\,2^{1/4}\eu^{-\iu\pi/8}$: $k' = \qty{3.9e-6}{m^{-1}}$, $k'' = \qty{1.6e-6}{m^{-1}}$: $v_\varphi =
\qty{6.4e6}{m/s}$, $\delta = \qty{620}{km}$, \qty{49}{dB} ao longo do cabo, em vez de
\qty{76}{dB}.

\textbf{7.} Um ponto a cada \qty{6}{s} são cerca de \qty{0.2}{bit/s}: a fibra é
$10^{10}$ a $10^{11}$ vezes mais rápida.

\textbf{8.} $v_\varphi = gT/2\pi$, $v_g = gT/4\pi$.

\textbf{9.} \qty{18}{s}: $v_g = \qty{14.1}{m/s}$, $4 \times 10^6/14.1 = \qty{3.3}{dias}$;
\qty{6}{s}: \qty{4.7}{m/s}, \qty{9.9}{dias}: a ondulação dura mais de seis dias.

\textbf{10.} $t = D/v_g = 4\pi D/gT$, logo $1/T = gt/4\pi D$. $\Delta(1/T) = 1/12 - 1/16
= \qty{0.0208}{Hz}$ em \qty{86400}{s}: $g/4\pi D = \qty{2.4e-7}{s^{-2}}$, $D = \qty{3200}{km}$.

\textbf{11.} $\tfrac12\rho gA^2 = \qty{11}{kJ/m^2}$; a energia anda a $v_g = gT/4\pi =
\qty{9.4}{m/s}$: \qty{103}{kW} por metro de crista.

\textbf{12.} $10^3 \times 103 \times 0.3 = \qty{31}{MW}$: dez turbinas eólicas.

\textbf{13.} $\lambda = 2\pi U^2/g = \qty{16}{m}$. A energia de cada componente
viaja à metade de sua celeridade de fase, de modo que o padrão fica confinado a uma cunha
atrás do barco (semiângulo de \qty{19.5}{\degree}, seja qual for a celeridade dele).

\textbf{14.} Uma crista nasce atrás do grupo, ultrapassa-o
(as cristas andam a $2v_g$ em relação à água, e a $v_g$ em relação ao
grupo) e morre à frente. O grupo tem $120 \times 9.4 = \qty{1.1}{km}$
de comprimento, cinco comprimentos de onda ($\lambda = gT^2/2\pi = \qty{225}{m}$); dez cristas passam pela
boia, tendo cada uma vivido cerca de dois minutos.

\textbf{15.} $kh = 0.13$: águas rasas. $c = \qty{198}{m/s}$, $T = \lambda/c = \qty{17}{min}$,
\qty{8.4}{h}.

\textbf{16.} Não dispersivo: todas as componentes a \qty{198}{m/s}, chegando
juntas como um só pulso (a ondulação, a $\sim\qty{10}{m/s}$, separada por período,
levaria uma semana).

\textbf{17.} $\tfrac12\rho gA^2 = \qty{1.8}{kJ/m^2}$ sobre $500 \times 200$ km$^2$: \qty{1.8e14}{J},
$4\%$ de uma megatonelada.

\textbf{18.} $A = 0.6(4000/20)^{1/4} = \qty{2.3}{m}$; $c = \sqrt{gh} = \qty{14}{m/s}$, $\lambda =
cT = \qty{14}{km}$. A crista, em água mais funda que o cavado à sua frente,
anda mais depressa e o alcança: a frente empina até virar uma macaréu.

\textbf{19.} Uma elevação de \qty{0.6}{m} ao longo de um quarto de hora, com declive
de $10^{-5}$: nada que se sinta.

\textbf{20.} A celeridade média, $\sqrt{gh}$ para a profundidade média; como
$\sqrt h$ é côncava, os trechos rasos freiam a onda mais do que os fundos
a aceleram, e a média ao longo do trajeto fica abaixo de $\sqrt{g\bar h}$.

\textbf{21.} $\omega' = \tfrac12\sqrt{g/k}$, $\omega'' = -\tfrac14\sqrt{g/k^3}$; $k = \qty{0.063}{rad/m}$:
$\omega'' = \qty{-50}{m^2/s}$.

\textbf{22.} $t_{\text{al}} = 10^6/50 = \qty{2e4}{s}$, \qty{5.6}{h}; o pacote
andou $v_gt = 6.25 \times 2 \times 10^4 = \qty{125}{km}$.

\textbf{23.} Um trem longo contém uma banda estreita de $k$, cujas \omterm{prop:b2:dispersion-wave-packets:group}{velocidades
de grupo} quase coincidem: as componentes ficam juntas por mais tempo. O
meio do tsunami é não dispersivo: celeridades iguais, sem alargamento.

\textbf{24.} $\omega'' = \hbar/m = \qty{1.2e-4}{m^2/s}$: $t_{\text{al}} = 10^{-18}/1.2 \times
10^{-4} = \qty{9e-15}{s}$; $v_g = \hbar k/m = \qty{7.3e6}{m/s}$: \qty{60}{nm}.

\textbf{25.} Cabo: $k^2 = -\iu R\Gamma\omega$, $v_\varphi = \sqrt{2\omega/R\Gamma}$, sem
$v_g$ bem definida, atenua e borra. Ondulação: $\omega^2 = gk$, $v_\varphi = 2v_g$,
alarga. Tsunami: $\omega = \sqrt{gh}\,k$, $v_\varphi = v_g$, nenhum dos dois. Elétron:
$\omega = \hbar k^2/2m$, $v_g = 2v_\varphi$, alarga.
\end{solution}
