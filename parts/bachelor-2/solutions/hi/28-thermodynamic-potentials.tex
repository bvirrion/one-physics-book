\section*{अध्याय \ref{ch:b2:thermodynamic-potentials} --- ऊष्मागतिकीय विभव}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$, $G = nRT\ln P + g(T)$; $P = -\partial_VF = nRT/V$,
$V = \partial_PG = nRT/P$; $-\partial_TF = nR\ln V - f'(T)$, अर्थात् एंट्रॉपी
$nC_V\ln T + nR\ln V + \text{अचर}$ यदि $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$, $\Delta U = 0$, $\Delta S = \qty{19.1}{J/K}$।
उत्क्रमणीय: \qty{5.7}{kJ}, $S_{\text{सृजित}} = 0$; \qty{1}{bar} के सामने: \qty{0.9}{kJ},
$S_{\text{सृजित}} = 19.1 - 900/300 = \qty{16}{J/K}$; निर्वात: कुछ नहीं, $S_{\text{सृजित}} =
\qty{19.1}{J/K}$।
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$, जिसमें $\Delta h \approx L$, $\Delta s \approx L/T_{\text{गलन}}$।
\qty{-5}{\celsius} पर: $g_{\text{द्रव}} - g_{\text{हिम}} = +\qty{6.1}{kJ/kg}$, अर्थात् बर्फ़
स्थायी, द्रव अर्ध-स्थायी (अतिशीतन होता है); और $+\qty{5}{\celsius}$ पर:
\qty{-6.1}{kJ/kg}, द्रव स्थायी --- अतितापित बर्फ़ नहीं देखी जाती, क्योंकि उसकी
सतह पहले पिघल जाती है।
\end{solution}

\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:7}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(क) पदार्थ नीचे $\mu$ की ओर जाता है: $\mu_{\text{वा}} < \mu_{\text{द्रव}}$ जब $p <
p_{\text{संत}}$; और आर्द्रता $= p/p_{\text{संत}}$। (ख) $-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
वाष्प के पक्ष में। (ग) वाष्पन को केवल $p < p_{\text{संत}}$ चाहिए,
उबलना नहीं; और $100\%$ पर विभव बराबर हो जाते हैं। (घ) सबसे ठंडी
सतहों पर, जिनका $p_{\text{संत}}(T_{\text{सतह}})$ हवा के $p$ से नीचे गिर चुका हो
--- यानी वे जो रात के आकाश को विकिरित करती हैं।
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(क) मानक। (ख) $b = RT_{\text{क्रां}}/8P_{\text{क्रां}} = \qty{4.3e-5}{m^3/mol}$, $a =
27R^2T_{\text{क्रां}}^2/64P_{\text{क्रां}} = \qty{0.36}{Pa.m^6/mol^2}$। (ग) $\dd g = v\,\dd P$ नियत $T$ पर
और $g$ दोनों सिरों पर बराबर: अतः पाश भर $\int v\,\dd P = 0$। (घ) अस्थायी वहाँ जहाँ
$\partial_vP > 0$; और बीच में अर्ध-स्थायी: यानी
अतितापित द्रव (उछलना) और अतिसंतृप्त वाष्प (वह कुहरा जो
नाभिकों की बाट जोहता है, कुहरा-कक्ष)।
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}
(क) $\dd P/\dd T = L_{\text{मोलर}}P/RT^2$: $\ln(P/P_0) = -(L_{\text{मोलर}}/R)(1/T - 1/T_0)$।
(ख) \qty{28}{mbar}, \qty{23}{mbar} के सामने --- क्योंकि \qty{20}{\celsius} पर $L$ बड़ा है।
(ग) \qty{820}{Pa}, \qty{611}{Pa} के सामने। (घ) $\partial(\Delta g/T)/\partial T = -L/T^2$
और वक्र के अनुदिश $\Delta g = 0$ वही ढाल देते हैं।
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(क) झिल्ली के पार बराबर $\mu$: $\mu^\circ(P + \Pi) - RTx_{\text{घु}} =
\mu^\circ(P)$, अर्थात् $v_{\text{मोलर}}\Pi = RTx_{\text{घु}}$, $\Pi = cRT$। (ख) \qty{27}{bar}।
(ग) प्रति घन मीटर $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$; और असली संयंत्र
चार गुना अधिक। (घ) जल नीचे विभव की ओर बहता है: विलेय-भरी कोशिका में
भीतर, और नमकीन जल में उससे बाहर।
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(क) $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$। (ख) $k_BT/Na^2 \propto T$। (ग)
\qty{1.7e-13}{N}; $1.8 \times 10^{14}$। (घ) $E \sim nk_BT \approx \qty{0.4}{MPa}$।
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$; $\dd F = -S\dd T + f\dd L$।

\textbf{2.} $\dd F$ पर श्वार्ट्स।

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$: अर्थात् ऊर्जा
बदलती नहीं; और खिंचाव एंट्रॉपिक है।

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$; $Q = \qty{-1.5}{J}$; $W = +\qty{1.5}{J}$; और योग
शून्य।

\textbf{5.} $+\qty{0.4}{K}$।

\textbf{6.} $L - L_0 \propto 1/T$: \qty{9.1}{cm}, यानी \qty{9}{mm} का सिकुड़ाव; और
गरम किए गए आरे छोटे हो जाते हैं, रिम के द्रव्यमान-केंद्र की जगह खिसक जाती है, और पहिया
घूम पड़ता है।

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$; और एंट्रॉपी का दंड
$k_BT$ में चुकाया जाता है।

\textbf{8.} \qty{1.7e-13}{N}; $1.8 \times 10^{14}$ शृंखलाएँ।

\textbf{9.} उसका बल ऊर्जात्मक है ($\partial_LU \ne 0$); और गरम करना
बंधों को नरम कर देता है, अतः $f$ $T$ के साथ गिरता है।

\textbf{10.} ढालें $-s$, जिनमें $s_{\text{गैस}} > s_{\text{द्रव}} > s_{\text{हिम}}$; सबसे
नीची रेखा स्थायी प्रावस्था है; और कटान $T_{\text{गलन}}$ तथा $T_{\text{वाष्प}}$ पर।

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}।

\textbf{12.} $\qty{-1.3e7}{Pa/K}$; ऋणात्मक क्योंकि $v_{\text{द्रव}} < v_{\text{हिम}}$।

\textbf{13.} \qty{70}{bar}: \qty{-0.5}{K} --- अर्थात् \qty{-10}{\celsius} पर
कोई इस तरह स्केट नहीं करता; वह घर्षण और द्रव-जैसी सतही परत करती है।

\textbf{14.} \qty{270}{bar}, \qty{-2}{K}: और भूतापीय फ्लक्स के साथ
तली पिघल जाती है तथा चादर फिसलती है।

\textbf{15.} \qty{3.6}{kPa/K}; \qty{92}{\celsius}; लगभग \qty{127}{\celsius}
(असल में \qty{120}{})।

\textbf{16.} \qty{28}{mbar} (\qty{23}{}), \qty{820}{Pa} (\qty{611}{}): $L$ बढ़ता है जैसे-जैसे
$T$ गिरे; और त्रुटियाँ $20$--$30\%$।

\textbf{17.} वही $\Delta v$, अतः ढालों का अनुपात $L_{\text{ऊर्ध्व}}/L_{\text{वाष्प}}
= 1.15$: अर्थात् ऊर्ध्वपातन वक्र अधिक तीखा है --- त्रिक बिंदु पर कोई मोड़।

\textbf{18.} \qty{12}{kJ/kg}; बर्फ़ को नाभिकित होना पड़ता है, और कोई छोटा नाभिक
जितना पाता है उससे अधिक सतही ऊर्जा चुकाता है: अतः शुद्ध बूँदें बाट जोहती हैं,
\qty{-40}{\celsius} तक।

\textbf{19.} $\Delta v \to 0$, $L \to 0$, और उनका अनुपात परिमित: अतः वक्र ख़त्म हो जाता है;
और उसके परे कोई एक ही तरल, जो द्रव-जैसे से गैस-जैसे में सतत बदल जाता है।

\textbf{20.} $\dd g = v\,\dd P$ नियत $T$ पर; और $g$ क्षैतिज के दोनों सिरों पर
बराबर: $\int v\,\dd P = 0$, यानी बराबर क्षेत्रफल।

\textbf{21.} अतितापित-द्रव वाली शाखा: किसी चिकने प्याले में कोई बुलबुला-नाभिक
नहीं; और कोई हलचल उसे तुरंत उबाल देती है।

\textbf{22.} जमाइए, \qty{611}{Pa} से नीचे पंप कीजिए, और हलके से गरम कीजिए: पथ
ऊर्ध्वपातन वक्र काटता है, गलन वक्र कभी नहीं; और संरचना
बची रहती है।

\textbf{23.} $\mu_{\text{वा}}(p) < \mu_{\text{द्रव}}$ जब $p < p_{\text{संत}}$; और
संतृप्ति पर बराबर।

\textbf{24.} \qty{100}{} के बजाय \qty{120}{\celsius}: चार गुना तेज़।

\textbf{25.} $S$ विलगित के लिए, $F$ नियत $T,V$ के लिए, $G$ नियत $T,P$ के लिए;
वापस पाया कार्य $-\Delta F$, $-\Delta G$; और मैक्सवेल संबंध $S$ के
अवकलजों को अवस्था-समीकरण में बदल देते हैं।
\end{solution}
