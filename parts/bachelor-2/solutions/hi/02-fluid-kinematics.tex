\begin{solution}{exo:b2:fluid-kinematics:6}
(क) $\dd y/\dd x = (V/U)\cos\omega t$: ढाल
$(V/U)\cos\omega t$ की सरल रेखाएँ, सब समांतर, समय के साथ झुकती हुईं। (ख) $x = Ut$, $y =
(V/\omega)\sin\omega t$: अर्थात् ज्यावक्र $y = (V/\omega)\sin(\omega x/U)$। (ग)
$t = 0$ पर \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखाएँ} \qty{45}{\degree} पर हैं और \omterm{def:b2:fluid-kinematics:euler}{पथ-रेखा} मूल बिंदु से
\qty{45}{\degree} पर निकलती है; $t = \pi/2\omega$ पर \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखाएँ}
क्षैतिज हैं जबकि \omterm{def:b2:fluid-kinematics:euler}{पथ-रेखा} वही अचल ज्यावक्र है। वे इसलिए भिन्न हैं कि
प्रवाह अस्थायी है: \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} एक क्षण-चित्र है, \omterm{def:b2:fluid-kinematics:euler}{पथ-रेखा} एक इतिहास।
(घ) $\vect a = \partial_t\vect v = (0, -V\omega\sin\omega t)$ --- क्षेत्र
एकसमान है, अतः संवहनी पद लुप्त हो जाता है।
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:9}
(क) $\varphi = \tfrac12k(x^2 - y^2)$, $\Delta\varphi = k - k = 0$। (ख) $\dd x/
kx = \dd y/(-ky)$: $xy = \text{अचर}$; $\partial_y\psi = kx = v_x$, $-\partial_x\psi =
-ky = v_y$। (ग) $\vect a = (v_x\partial_x + v_y\partial_y)\vect v = k^2(x, y) =
\operatorname{\vect{grad}}\bigl(\tfrac12k^2(x^2 + y^2)\bigr) = \operatorname{\vect{grad}}
(v^2/2)$: अर्थात् स्थायी \omterm{def:b2:fluid-kinematics:vorticity}{अघूर्णी प्रवाह} में त्वरण प्रति एकांक द्रव्यमान
गतिज ऊर्जा की प्रवणता है; वह स्थगन बिंदु से दूर की ओर संकेत करता है ---
कण दीवार के पास $y$ के अनुदिश आते हुए मंदित होता है और $x$ के अनुदिश दूर जाते हुए त्वरित। (घ) $\dot x = kx$, $\dot y = -ky$:
$x = x_0\eu^{kt}$, $y = y_0\eu^{-kt}$ ($xy$ अचर)।
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:10}
(क) $\dd x/\dd t = x/(t + \tau)$ से $x = x_0(1 + t/\tau)$, $\ddot x = 0$;
\omterm{thm:b2:fluid-kinematics:material}{द्रव्य अवकलज} से: $\partial_tv + v\partial_xv = -x/(t + \tau)^2 + x/(t + \tau)^2
= 0$। (ख) $\partial_t\rho + \partial_x(\rho v) = \dot\rho + \rho/(t + \tau) = 0$: $\rho =
\rho_0\tau/(t + \tau)$। (ग) दो कणों का अंतराल $(1 + t/\tau)$ की तरह बढ़ता है
और उनके बीच का द्रव्यमान नियत है: $\rho \propto 1/(1 + t/\tau)$। (घ) $v =
Hx$, जहाँ $H = 1/(t + \tau)$: अर्थात् हबल नियम, मुक्त प्रसार (त्वरण
शून्य), और घनत्व मापक गुणक के व्युत्क्रम की तरह गिरता हुआ।
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:11}
(क) $\vect v = \operatorname{\vect{grad}}\varphi = (U + qx/2\pi r^2, qy/2\pi r^2)$;
स्रोत के लिए $v_r = q/2\pi r$, $\operatorname{div} = \frac1r\dd(rv_r)/\dd r
= 0$ और $\theta$ पर कोई निर्भरता नहीं, अतः $\omega_z = 0$; और ऐसे क्षेत्रों के योग
फिर से असंपीड्य तथा अघूर्णी होते हैं। (ख) ऋणात्मक $x$ अक्ष पर,
$U - q/2\pi r = 0$: $x_s = -q/2\pi U$। (ग) स्थगन बिंदु पर $\theta =
\pi$, $y = 0$: $\psi = q/2$; अतः \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} $Ur\sin\theta + q\theta/2\pi = q/2$,
अर्थात् $r\sin\theta = (q/2\pi U)(\pi - \theta)$ --- यह $x_s$ से निकलकर
प्रवाह की दिशा में खुलता वक्र है। (घ) $\theta \to 0$: $y \to q/2U$; $\theta \to 2\pi$ (निचली
शाखा): $y \to -q/2U$। उस \omterm{def:b2:fluid-kinematics:euler}{धारा-रेखा} के भीतर का तरल स्रोत से आता
है; उसके बाहर एकसमान धारा $q/U$ अनंतस्पर्शी चौड़ाई के अर्ध-पिंड के
चारों ओर बहती है।
\end{solution}

\begin{solution}{exo:b2:fluid-kinematics:12}
(क) $\operatorname{div}\vect v = \frac1r\partial_r(-q/2\pi) = 0$; $\omega_z =
\frac1r\partial_r(\Gamma/2\pi) = 0$। (ख) $\dd r/v_r = r\,\dd\theta/v_\theta$ से
$\dd r/r = -(q/\Gamma)\dd\theta$। (ग) $a_r = v_r\partial_rv_r - v_\theta^2/r =
-(q^2 + \Gamma^2)/4\pi^2r^3 = -v^2/r$; $a_\theta = v_r\partial_rv_\theta + v_rv_\theta/r
= 0$। (घ) $\dd r/\dd t = -q/2\pi r$: $t = \pi(r_0^2 - a^2)/q$; चक्कर
$= (\Gamma/2\pi q)\ln(r_0/a)$। (ङ) $t = \qty{520}{s} \approx \qty{9}{min}$; $10$
चक्कर।
\end{solution}

\begin{solution}{pb:b2:fluid-kinematics:1}
\textbf{1.} $r$ त्रिज्या के बेलन से होकर: $2\pi rh|v_r| = D_V$, अतः
$v_r = -q/2\pi r$, $q = D_V/h = \qty{1.5e-3}{m^2/s}$।

\textbf{2.} $rv_\theta = 0.5 \times 0.01 = \qty{5e-3}{m^2/s}$: अतः $v_\theta = \Gamma/2\pi r$,
जहाँ $\Gamma = \qty{3.1e-2}{m^2/s}$।

\textbf{3.} $\operatorname{div}\vect v = \frac1r\partial_r(rv_r) = 0$, $\omega_z =
\frac1r\partial_r(rv_\theta) = 0$।

\textbf{4.} $\dd r/r = -(q/\Gamma)\dd\theta$: $r = r_0\eu^{-q\theta/\Gamma}$; चक्कर
$= (\Gamma/2\pi q)\ln(r_0/a) = 3.3 \times 3.2 \approx 11$।

\textbf{5.} $t = \pi(r_0^2 - a^2)/q = \qty{520}{s} \approx \qty{9}{min}$; $v_\theta(a) =
\qty{0.25}{m/s}$।

\textbf{6.} $\vect a = -(v^2/r)\vect e_r$; $r = a$ पर: $v^2 = 0.25^2 + 0.012^2$,
$a = \qty{3.1}{m/s^2} \approx 0.3g$।

\textbf{7.} $\Omega_\oplus\sin\lambda\,r = 5 \times 10^{-5} \times 0.5 = \qty{2.5e-5}{m/s}$,
जो हिलाने से मिली चाल से चार सौ गुना कम है: अतः साधारण टब में
गोलार्ध कुछ भी तय नहीं करता (दिन भर स्थिर छोड़ी गई टंकी में करता है)।

\textbf{8.} $v_\theta = \Omega r$ ($r < a$), $\Omega a^2/r$ ($r > a$); $\Omega = 80/60
= \qty{1.33}{rad/s}$, $\Gamma = 2\pi av_{\max} = \qty{3.0e4}{m^2/s}$।

\textbf{9.} क्रोड में $\omega_z = 2\Omega = \qty{2.7}{s^{-1}}$, बाहर $0$;
$v_\theta$ रैखिक रूप से \qty{80}{m/s} तक चढ़ता है और फिर $1/r$ की तरह गिरता है; $\omega_z$
एक सीढ़ी है।

\textbf{10.} $a = v_{\max}^2/a = 6400/60 = \qty{107}{m/s^2} \approx 11g$, भीतर की ओर।

\textbf{11.} बाहर \omterm{def:b2:fluid-kinematics:vorticity}{भ्रमिलता} शून्य है: कोई छोटा अवयव अपने वृत्त पर
स्थानांतरित होता है पर अपने बारे में घूमता नहीं --- अर्थात् पंखी-चक्र
अपना दिक्-विन्यास बनाए रखता है। क्रोड में (ठोस घूर्णन) वह तरल के साथ
घूमता है, हर एक चक्कर पर एक बार।

\textbf{12.} $E_{\text{क्रोड}} = \int_0^a\tfrac12\rho\Omega^2r^2\,2\pi r\,\dd r =
\tfrac14\pi\rho\Omega^2a^4 = \qty{2.2e7}{J/m}$।

\textbf{13.} $E_{\text{बाहर}} = \int_a^R\tfrac12\rho(\Omega a^2/r)^2\,2\pi r\,\dd r =
\pi\rho\Omega^2a^4\ln(R/a) = 8.7 \times 10^7 \times 4.4 = \qty{3.8e8}{J/m}$; यहाँ
समाकल्य $\rho v^2r \propto 1/r$ है, इसी से लघुगणक आता है।

\textbf{14.} $\qty{4.0e8}{J/m} \times \qty{1000}{m} = \qty{4e11}{J}$: अर्थात् लगभग
\qty{100}{t} टीएनटी।

\textbf{15.} आगम $2\pi a \times 300 \times 5 = \qty{5.7e5}{m^3/s}$, और वह
$\pi a^2 = \qty{1.1e4}{m^2}$ से होकर: $v_z = \qty{50}{m/s}$।

\textbf{16.} $\omega = 2v_{\max}/a = \qty{2.5e-3}{s^{-1}} = 50f$; $\Gamma = 2\pi av_{\max}
= \qty{1.3e7}{m^2/s}$।

\textbf{17.} $r_0\cdot fr_0/2 = r(v_\theta + fr/2)$: $v_\theta = f(r_0^2 - r^2)/2r$;
\qty{100}{km} पर \qty{60}{m/s}, \qty{40}{km} पर \qty{155}{m/s}।

\textbf{18.} उत्तरी गोलार्ध में $f > 0$ होने से $v_\theta > 0$ मिलता है:

\textbf{21.} $\vect v(M, t) = -U\vect e_x + \vect v_0(M + Ut\,\vect e_x)$: अर्थात्
आकृति बह जाती है, और धरती-तंत्र में क्षेत्र अस्थायी है।

\end{solution}
