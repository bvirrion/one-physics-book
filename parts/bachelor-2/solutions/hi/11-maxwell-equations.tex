\begin{solution}{exo:b2:maxwell-equations:7}

$E$; और $U = 0$: कोई परवलय, जिसका शिखर $d/2$ पर है।
\end{solution}

\begin{solution}{exo:b2:maxwell-equations:10}

\end{solution}

\begin{solution}{pb:b2:maxwell-equations:1}
\textbf{1.} $Q = (I_0/\omega)\sin\omega t$, $E = Q/\varepsilon_0\pi R^2$: आयाम $I_0/\omega\varepsilon_0
\pi R^2 = \qty{5.7e5}{V/m}$।

\textbf{2.} $j_d = \varepsilon_0\partial_tE = I(t)/\pi R^2 = 32\cos\omega t$ A/m$^2$; और प्लेट से
उसका अभिवाह $I(t)$ है।

\textbf{3.} $2\pi rB = \mu_0\varepsilon_0\pi r^2\partial_tE$: $B = \mu_0I(t)r/2\pi R^2$; और $R$ पर:
$\mu_0I_0/2\pi R = \qty{2}{\micro T}$।

\textbf{4.} $r > R$ के लिए पूरी \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा} घिर जाती है: अतः $B =
\mu_0I/2\pi r$, अर्थात् तार का क्षेत्र; और $R$ पर संतत।

\textbf{5.} $(\operatorname{\vect{curl}}\vect E_1)_\theta = -\partial_rE_1 = -\partial_tB_\theta$, अतः $\partial_rE_1
= \mu_0\varepsilon_0(r/2)\ddot E$ और $E_1(r) - E_1(0) = \mu_0\varepsilon_0r^2\ddot E/4$; और
$\ddot E = -\omega^2E$ लेकर: $|E_1(R) - E_1(0)|/|E| = (\omega R/c)^2/4 = 1 \times 10^{-6}$।

\textbf{6.} उत्तम। $10\%$ के लिए $(\omega R/c)^2 = 0.4$ चाहिए, $f \approx \qty{300}{MHz}$ ---
अर्थात् कोई गुहा, अब संधारित्र नहीं।

\textbf{7.} $W_e = \tfrac12\varepsilon_0E_0^2\pi R^2d = \qty{4.5e-5}{J}$; $W_m = \int_0^R(\mu_0I_0r/
2\pi R^2)^2/2\mu_0\cdot2\pi rd\,\dd r = \mu_0I_0^2d/16\pi = \qty{2.5e-11}{J}$; और अनुपात $6 \times
10^{-7} \approx (\omega R/c)^2/8$: अर्थात् वही छोटा प्राचल।

\textbf{8.} $\oint\vect E\cdot\dd\vect l = 2\pi r_0E_\theta = -\dd\Phi/\dd t$।

\textbf{9.} $\dd p/\dd t = e|E_\theta| = (e/2\pi r_0)|\dd\Phi/\dd t|$; और प्रति चक्कर,
$2\pi r_0\cdot e|E_\theta| = e|\dd\Phi/\dd t|$।

\textbf{10.} $p = eB(r_0)r_0$ से $\dot p = er_0\dot B(r_0)$; और 9 के साथ बराबर करने पर:
$B(r_0) = \Phi/2\pi r_0^2 = \tfrac12\langle B\rangle$।

\textbf{11.} किसी एकसमान क्षेत्र के लिए $B(r_0) = \langle B\rangle$, अर्थात् दुगुना अधिक:
तब चुंबकीय बल संवेग से तेज़ बढ़ता है और कक्षा
सिकुड़ जाती है। ध्रुव ऐसे गढ़े जाते हैं कि क्षेत्र अक्ष के पास तेज़ हो
और कक्षा पर उसका आधा।

\textbf{12.} $p = eBr_0 = \qty{3.2e-20}{kg.m/s}$; $E = pc = \qty{9.6e-12}{J} = \qty{60}{MeV}$;
$\Phi = 2\pi r_0^2B = \qty{0.63}{Wb}$ \qty{5}{ms} में: अतः प्रति चक्कर \qty{126}{eV}, $4.8 \times
10^5$ चक्कर, \qty{1500}{km} --- और $c$ पर \qty{5}{ms} में: संगत।

\textbf{13.} चुंबक की विद्युत-आपूर्ति से, जो फ़ैराडे पद के माध्यम से
इलेक्ट्रॉनों को सौंपी जाती है: बदलता अभिवाह ही त्वरक क्षेत्र बनाता है।

\textbf{14.} किसी भारी लक्ष्य में मंदित होते दसियों \unit{MeV} के इलेक्ट्रॉन
भेदक X-किरणें देते हैं; और सघन रैखिक त्वरक, जिसकी मात्रा-दर ऊँची है,
उसकी जगह ले आया।

\textbf{15.} $\operatorname{div}\operatorname{\vect{curl}}\vect A = 0$; $\operatorname{\vect{curl}}(-\operatorname{\vect{grad}}
V - \partial_t\vect A) = -\partial_t\operatorname{\vect{curl}}\vect A = -\partial_t\vect B$।

\textbf{16.} $\frac1{r^2}(r^2V')' = -\rho_0/\varepsilon_0$: भीतर $V = \rho_0(3a^2 - r^2)/6\varepsilon_0$,
बाहर $\rho_0a^3/3\varepsilon_0r = Q/4\pi\varepsilon_0r$; और $E = \rho_0r/3\varepsilon_0$ तथा
$Q/4\pi\varepsilon_0r^2$।

\textbf{17.} $\tfrac12\int\rho V\,\dd\tau = 2\pi\rho_0^2(a^5/6 - a^5/30)/\varepsilon_0 = 4\pi\rho_0^2a^5/
15\varepsilon_0 = 3Q^2/20\pi\varepsilon_0a$; यूरेनियम: \qty{1.6e-10}{J} $\approx \qty{1000}{MeV}$ (और विखंडन
वे \qty{200}{MeV} छोड़ता है जिनसे दो छोटे गोले सस्ते पड़ते हैं)।

\textbf{18.} $V = -E(t)z$ (और उसमें कोई अचर)। $\vect A = A_z(r, t)\vect e_z$ लेकर,
$B_\theta = -\partial_rA_z = \mu_0Ir/2\pi R^2$ से $A_z = -\mu_0I(t)r^2/4\pi R^2$।

\textbf{19.} $-\partial_tA_z = \mu_0\dot Ir^2/4\pi R^2 = \mu_0\varepsilon_0r^2\ddot E/4$ (क्योंकि $\dot I =
\varepsilon_0\pi R^2\ddot E$): अर्थात् प्रश्न 5 का सुधार, किसी अचर तक।

\textbf{20.} $\operatorname{div}\vect A = \partial_zA_z = 0$, क्योंकि $A_z$ केवल $r$ पर
निर्भर है; और कोई भी $\vect A + \operatorname{\vect{grad}}\chi$।

\textbf{21.} $\sigma = \varepsilon_0E_0 = \qty{5}{\micro C/m^2}$; और धातु के भीतर $\vect E =
\vect 0$, अतः उछाल $\sigma/\varepsilon_0$ ठीक अंतराल का क्षेत्र है।

\textbf{22.} धारा $I(t)$ प्लेट को अक्ष से खिलाती है; और $r$ से आगे
अब तक न पहुँचाया भाग $I(1 - r^2/R^2)$ है, अतः $j_s = I(1 -
r^2/R^2)/2\pi r$। बाहरी फलक पर ऐंपियर तार का क्षेत्र $\mu_0I/
2\pi r$ देता है; भीतरी फलक पर $\mu_0Ir/2\pi R^2$; और अंतर, $\mu_0I(1 -
r^2/R^2)/2\pi r = \mu_0j_s$, वही सीमा-उछाल है।

\textbf{23.} प्लेटें: $R/\lambda = 3 \times 10^{-4}$, वैद्युत अर्ध-स्थायी। पाश: $7 \times 10^{-4}$,
चुंबकीय अर्ध-स्थायी। \qty{1}{GHz} पर: $R/\lambda = 0.3$ --- अर्थात् कोई गुहा और कोई ऐंटेना।

\textbf{24.} किसी मीटर भर के वलय के लिए $\lambda = \qty{3000}{km}$: अतः \omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन
धारा} नितांत नगण्य है; और यहाँ का अनिवार्य विद्युत क्षेत्र
फ़ैराडे का है, जिसे चुंबकीय अर्ध-स्थायी दशा पूरा रखती है।

\textbf{25.} I: ऐंपियर--मैक्सवेल (\omterm{thm:b2:maxwell-equations:maxwell}{विस्थापन धारा}), सुधार के लिए फ़ैराडे,
और ऊर्जा-घनत्व। II: समाकल फ़ैराडे, लोरेंत्ज़ बल।
III: विभव, प्वासों (गाउस)। IV: $\vect E$ और $\vect B$ के सीमा-उछाल;
और दशा के लिए आकार-से-तरंगदैर्घ्य का अनुपात।
\end{solution}
