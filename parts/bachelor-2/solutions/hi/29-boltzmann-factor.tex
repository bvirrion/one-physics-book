\begin{solution}{exo:b2:boltzmann-factor:2}

\begin{solution}{exo:b2:boltzmann-factor:5}
(क) \cref{prop:b2:boltzmann-factor:twolevel}। (ख) $\eu^{-x} < 1$ जब $T > 0$;
$N_2 > N_1$ को $T < 0$ चाहिए होता --- कोई जनसंख्या-प्रतिलोमन। (ग) $0$ और
$Nk_B\ln 2$। (घ) $x = 1.93$: $0.13$; \qty{77}{K} पर, $x = 7.5$: $5 \times 10^{-4}$।
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:6}
(क) $N\varepsilon/(\eu^x + 1)$ का अवकलज। (ख) $x = 2.40$, $C = 0.44\,Nk_B$।
(ग) $\propto x^2\eu^{-x} \to 0$: जमा हुआ; $\propto 1/T^2 \to 0$: दोनों स्तर भरे,
सोखने को कुछ बचा नहीं। (घ) $k_BT = 0.42\varepsilon$: \qty{0.5}{mK}; किसी
मिलिकेल्विन से नीचे नाभिकीय प्रचक्रण अधिकांश ऊष्मा-धारिता थामे रहते हैं और
हर शीतलन को धीमा कर देते हैं।
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:7}

\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:8}

\begin{solution}{exo:b2:boltzmann-factor:9}
(क) $f$ का खंडशः समाकलन कीजिए; अग्र पद $(2/\sqrt\pi)x\eu^{-x^2}$ है। (ख)
हीलियम $x = 5.5$: $5 \times 10^{-13}$; नाइट्रोजन $x = 14.5$: $\eu^{-211}$। (ग) $x =
4.9$: प्रति टक्कर-काल $10^{-10}$ --- गया; $10^{-17}$ को $x \approx 6.6$ चाहिए,
$T \lesssim \qty{230}{K}$। (घ) $x = 11$: $\eu^{-121}$; ठंडा पर्याप्त।
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:10}
(क) $n = n_0\eu^{m\omega^2r^2/2k_BT}$। (ख) $\eu^{\Delta m\omega^2r^2/2k_BT} = \eu^{0.24} =
1.27$। (ग) $H = k_BT/m'g = \qty{16}{\micro m}$; कई ऊँचाइयों पर गोले
गिनने से $k_B$ मिलता है, अतः $N_A = R/k_B$। (घ) आयन किसी आवेश के चारों ओर
$\eu^{\mp e\varphi/k_BT}$ मानते हैं; पॉइसों के समीकरण में रैखिकीकृत कर
यह $\varphi'' = \varphi/\lambda_{\text{D}}^2$ देता है: विभव $\lambda_{\text{D}}$ के परे
परिरक्षित हो जाता है।
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:11}
(क) $Z = 1/(1 - \eu^{-x})$, $\langle E\rangle = h\nu/(\eu^x - 1)$। (ख) $k_BT$:
समविभाजन; $h\nu\eu^{-h\nu/k_BT}$: जमा हुआ। (ग) $u_\nu = (8\pi\nu^2/c^3)\langle E
\rangle$; रैले--जीन्स हर मोड को $k_BT$ देता है। (घ) कोई मोड एक क्वांटम से
कम से उत्तेजित नहीं हो सकती।
\end{solution}

\begin{solution}{exo:b2:boltzmann-factor:12}
(क) केवल $x = \mu B/k_BT$; $\ln 2$ और $0$। (ख) $x = 0.67$: $S/Nk_B = 0.51$, अतः
प्रति प्रचक्रण $0.18\,k_B$ हटा। (ग) $B/T$ नियत: \qty{10}{mK}। (घ) प्रचक्रणों का
अपना क्षेत्र (मिलिटेस्ला) अंत में $B$ की जगह ले लेता है और शीतलन को सीमित करता है;
$M = n\mu\tanh(\mu B/k_BT)$ मापने से $T$ मिलता है।
\end{solution}

\begin{solution}{pb:b2:boltzmann-factor:1}
\textbf{1.} $\dd P/\dd z = -nmg = -(mg/k_BT)P$; $H = \qty{8.4}{km}$।

\textbf{2.} $\eu^{-mgz/k_BT}$: एक अणु की स्थितिज ऊर्जा, और
हवा का ताप।

\textbf{3.} \qty{0.70}{}, \qty{0.52}{}, \qty{0.35}{bar}; $H\ln 2 = \qty{5.8}{km}$।

\textbf{4.} $P_0/g = \qty{1.03e4}{kg/m^2}$; $\times 4\pi R^2$: \qty{5.3e18}{kg}; $\rho_0H =
1.22 \times 8400 = \qty{1.03e4}{kg/m^2}$।

\textbf{5.} $5.3 \times 10^{18}/(29 \times 10^{-3}/N_A) = 1.1 \times 10^{44}$।

\textbf{6.} $H_{\text{O}_2} = 7.6$, $H_{\text{N}_2} = \qty{8.7}{km}$: \qty{50}{km} पर अनुपात
अपने ज़मीनी मान का $\eu^{-50/7.6 + 50/8.7} = 0.44$ होता; विक्षोभ विसरण के
अलग करने से तेज़ मिलाता है।

\textbf{7.} नीचा: ठंडी हवा सघन है और दाब तेज़ी से गिरता है।

\textbf{8.} \qty{2.5e25}{m^{-3}}; $\eu^{-11.9}$: \qty{1.7e20}{m^{-3}} --- कोई देहरी नहीं, केवल कोई
रूढ़ि।

\textbf{9.} $11.2\sqrt{6370/6870} = \qty{10.8}{km/s}$।

\textbf{10.} हीलियम \qty{2040}{}, \qty{2300}{m/s}; नाइट्रोजन \qty{770}{}, \qty{870}{m/s}।

\textbf{11.} $x = 5.3$: $4 \times 10^{-12}$; नाइट्रोजन $\eu^{-196}$: कोई नहीं।

\textbf{12.} $\tfrac14 \times 10^{12} \times 2300 \times 4 \times 10^{-12} \times \tfrac12 \approx
10^3$ परमाणु प्रति वर्ग मीटर प्रति सेकंड; पूरी पृथ्वी पर और एक वर्ष में,
$2 \times 10^{25}$ परमाणु --- \qty{0.1}{kg}।

\textbf{13.} $5 \times 10^{-6} \times 1.1 \times 10^{44} = 5.5 \times 10^{38}$ परमाणु, \qty{3.7e12}{kg};
\qty{0.1}{kg/yr} के सामने निवास-काल बेतुका होता।

\textbf{14.} स्थायी अवस्था को \qty{3e6}{kg/yr} की हानि चाहिए: लगभग दस लाख
वर्ष का कोई निवास-काल, जो स्वीकृत मान है --- अतः \qty{1000}{K} पर
तापीय आकलन कहीं बहुत छोटा है: पलायन पर
अधिक गरम प्रसंगों और अ-तापीय (आयनिक) प्रक्रियाओं का बोलबाला है।

\textbf{15.} $x = 3.75$: $\eu^{-14}$, $\eu^{-28}$ के बजाय --- $10^6$ का कोई गुणक:
हानि बहिर्मंडलीय ताप के प्रति असाधारण रूप से संवेदी है।

\textbf{16.} चंद्रमा गरम और हलका है: भूवैज्ञानिक काल में सब कुछ
निकल जाता है; टाइटन ठंडा है, नाइट्रोजन के लिए $x = 11$।

\textbf{17.} $p_\pm = \eu^{\pm x}/2\cosh x$; $M = n\mu\tanh x$।

\textbf{18.} $x = \mu B/k_BT$: \qty{4.2}{A/m}, \qty{310}{A/m}, \qty{1.6e4}{A/m}
(संतृप्ति \qty{1.9e4}{A/m}); $x \ll 1$ के लिए क्यूरी, अर्थात् $T \gg \qty{70}{mK}$।

\textbf{19.} $M$ केवल $T$ पर निर्भर है, एक बार $B$ ज्ञात हो जाए; सबसे तीखा
$x \approx 1$, $T \approx \mu B/k_B = \qty{70}{mK}$ के पास; और प्रवृत्ति किसी कुंडली से
मापी जाती है।

\textbf{20.} $\langle E\rangle = -N\mu B\tanh x$; $C = Nk_Bx^2/\cosh^2x$, शिखर
$x = 1.2$ पर: \qty{56}{mK}।

\textbf{21.} $Nk_B\ln 2$ और $0$: सारे प्रचक्रण क्षेत्र के साथ पंक्तिबद्ध हो जाते हैं।

\textbf{22.} \qty{10}{mK}; प्रचक्रणों का मिलिटेस्ला का अपना क्षेत्र $B$ की जगह
ले लेता है और फ़र्श तय कर देता है।

\textbf{23.} $x = 1$ जब $\mu_{\text{N}}B/k_B = \qty{0.37}{mK}$: क्यूरी किसी
मिलिकेल्विन तक टिकता है, और तापमापी माइक्रोकेल्विन तक काम का है।

\textbf{24.} $\Delta\nu/\nu = \sqrt{8k_BT\ln2/mc^2} \propto \sqrt T$: किसी रेखा की
चौड़ाई गैस का ताप मापती है।

\textbf{25.} वायुमंडल, पलायन करती गैस, प्रचक्रण: $\eu^{-E/k_BT}$ जिसमें $E = mgz$,
$\tfrac12mv^2$, $\mp\mu B$।
\end{solution}
