\section*{الفصل \ref{ch:b2:waves-in-media} --- الأمواج الكهرمغناطيسية في البلازما والنواقل والعوازل}

\begin{solution}{exo:b2:waves-in-media:1}
$f_p \approx 9\sqrt n$ هرتز: أي \qty{2.8}{MHz} و \qty{9}{MHz} و \qty{9}{GHz} و \qty{2.6e15}{Hz}
(و $\lambda_p = \qty{115}{nm}$) و \qty{2.8}{kHz}. والتضمين السعوي عند \qty{1}{MHz}: لا يمرّ إلا عبر
الفضاء بين النجوم. والتضمين الترددي: يمرّ عبر الأيونوسفير لا عبر غلاف العودة. وتحديد المواقع:
عبر كل شيء إلا غلاف العودة. والضوء: عبر كل شيء إلا النحاس.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:5}
(أ) $1/\gamma\pi a^2 = \qty{5.3}{m\ohm/m}$. (ب) $\delta = \qty{65}{\micro m}$؛ و $R \approx 1/\gamma2\pi a\delta
= \qty{41}{m\ohm/m}$، أي $7.7$ أضعاف المستمرة. (ج) $\delta = \qty{6.5}{\micro m}$، و \qty{0.41}{\ohm/m}،

القشرة، وكل لفة تضيف مقاومة؛ وسلك ليتز يحزم خيوطًا كثيرة
معزولة أرقّ من $\delta$، فيوصّل النحاس كله.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:6}
$\delta_{\text{الألومنيوم}} = 11.9$ و $0.84$ و \qty{0.084}{mm}: ومنه $d/\delta = 0.08$ و $1.2$ و $12$: أي $0.7$

\end{solution}

\begin{solution}{exo:b2:waves-in-media:7}
(أ) $v_0 = eE_0/m\omega$؛ و $\tfrac12nm\langle v^2\rangle = ne^2E_0^2/4m\omega^2 = \tfrac14\varepsilon_0
E_0^2\,\omega_p^2/\omega^2$. (ب) $\tfrac14\varepsilon_0E_0^2$ و $\tfrac14\varepsilon_0E_0^2(1 - \omega_p^2/
\omega^2)$. (ج) والمجموع $\tfrac12\varepsilon_0E_0^2$ (فالقوس يساوي $1$)؛ و $\langle\Pi
\rangle = E_0^2k/2\mu_0\omega$، والنسبة $c^2k/\omega = v_g$. (د) $\tfrac14\cdot\tfrac14
/\tfrac12 = 1/8$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:8}
(أ) $1/v_g = (1/c)(1 - \omega_p^2/\omega^2)^{-1/2} \approx (1 + \omega_p^2/2\omega^2)/c$. (ب) $\Delta t
= (L\omega_p^2/2c)(1/\omega_1^2 - 1/\omega_2^2)$ مع $\omega_p^2 = ne^2/\varepsilon_0m$ و $\omega =
2\pi f$: فالثابت $e^2/8\pi^2\varepsilon_0mc = \qty{1.34e-7}{}$ بواحدات النظام الدولي. (ج) $nL = \qty{9.3e23}{m^{-2}}$،
و $(1/f_1^2 - 1/f_2^2) = \qty{5.7e-18}{s^2}$: ومنه $\Delta t = \qty{0.72}{s}$. (د) العمود $nL$
(أي مقياس التبدد)، ومنه البعد إذا عُرف نموذج $n$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:9}
(أ) $\operatorname{Im}\underline\chi = \omega_p^2\Gamma\omega/[(\omega_0^2 - \omega^2)^2 + \Gamma^2\omega^2]$ مع
$\omega_0^2 - \omega^2 \approx 2\omega_0(\omega_0 - \omega)$: ومنه $n'' = \tfrac12\operatorname{Im}\underline\chi$ هو
منحني لورنتس المذكور. (ب) $n''(\omega_0) = \omega_p^2/2\omega_0\Gamma$، و $\alpha = 2n''\omega_0/c =
\omega_p^2/\Gamma c$. (ج) $\omega_p^2 = \qty{3.2e20}{s^{-2}}$: ومنه $\alpha = \qty{1.8e4}{m^{-1}}$؛ و
$\ln10/\alpha = \qty{0.13}{mm}$. (د) $n' - 1 \propto (\omega_0 - \omega)/[(\omega_0 - \omega)^2 +
\Gamma^2/4]$: فهو موجب دون الخط، وسالب فوقه، ومتناقص ضمن $\pm\Gamma/2$
منه --- أي شاذّ هناك.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:10}
(أ) $\underline E = E_0\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$، و $\underline B = \underline k\underline E/\omega
= (1 - \iu)\underline E/\omega\delta$. (ب) $\langle\Pi_x\rangle = \tfrac12\operatorname{Re}(\underline E
\underline B^*)/\mu_0 = E_0^2/2\mu_0\omega\delta$ عند $x = 0$. (ج) $\int_0^\infty\tfrac12\gamma E_0^2
\eu^{-2x/\delta}\dd x = \gamma E_0^2\delta/4 = E_0^2/2\mu_0\omega\delta$ باستعمال $\gamma = 2/\mu_0\omega\delta^2$.
(د) $H_0 = \sqrt2E_0/\mu_0\omega\delta$، ومنه $\tfrac12R_sH_0^2 = E_0^2/\gamma\mu_0^2\omega^2\delta^3 =
E_0^2/2\mu_0\omega\delta$. والنحاس عند \qty{1}{GHz}: $R_s = 1/\gamma\delta = \qty{8}{m\ohm}$؛
و $\tfrac12 \times 8 \times 10^{-3} \times 10^4 = \qty{40}{W/m^2}$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:11}
(أ) $\omega_p = \qty{1.4e16}{rad/s}$، و $\lambda_p = 2\pi c/\omega_p = \qty{140}{nm}$. (ب) $\omega = \qty{3.8e15}{rad/s}
< \omega_p$: ومنه $1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{23}{nm}$، وانعكاس تام. (ج)
وما زال دون $\omega_p$: أي عاكس؛ ولا تأتي الشفافية إلا فيما وراء \qty{140}{nm} (وفي
الصوديوم، \qty{210}{nm}). (د) وتعطي التصادمات جزءًا حقيقيًّا للمقدار $\underline\gamma$ و
بضعة بالمئة من الامتصاص؛ ورنينات الإلكترونات المرتبطة في المرئي
(في الذهب والنحاس) تمتصّ الأزرق وتلوّن المعدن.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:12}
(أ) $\omega\tau = 0.12$: ومنه $\underline\varepsilon_r = 84 - 9.7\iu$؛ وعند \qty{20}{GHz}، $\omega\tau = 1$:
$45 - 40\iu$. (ب) $\underline n \approx 9.2 - 0.53\iu$؛ و $c/2n''\omega = \qty{1.8}{cm}$: فيطبخ الفرن

\end{solution}

\begin{solution}{pb:b2:waves-in-media:1}
\textbf{1.} $f_p = \qty{9}{MHz}$: فتحديد المواقع يمرّ، والموجة القصيرة تنعكس.

\textbf{2.} $k = (\omega/c)\sqrt{1 - \omega_p^2/\omega^2}$: ومنه $v_\varphi = \omega/k \approx c(1 + \omega_p^2/
2\omega^2)$، و $v_g = \dd\omega/\dd k = c^2k/\omega \approx c(1 - \omega_p^2/2\omega^2)$.

\textbf{3.} $\Delta t = \int(1/v_g - 1/c)\dd z = \int\omega_p^2\dd z/2c\omega^2 = e^2N_T/2\varepsilon_0
mc\omega^2 = (e^2/8\pi^2\varepsilon_0mc)N_T/f^2 = 1.34 \times 10^{-7} \times 2 \times 10^{17}/2.48 \times
10^{18} = \qty{11}{ns}$: أي \qty{3.2}{m}.

\textbf{4.} $N_T = (\Delta t_1 - \Delta t_2)/K(1/f_1^2 - 1/f_2^2)$؛ والفرق
\qty{7}{ns}: ومنه تحتاج نسبة $1\%$ إلى \qty{0.07}{ns}.

\textbf{5.} تفوق $v_\varphi$ السرعةَ $c$ بالمقدار الذي تقصّر به $v_g$: ف
يصل طور الحامل باكرًا بينما يصل التضمين متأخرًا. و
المستقبل الذي يعوّل على طور الحامل يطبّق التصحيح بالإشارة المعاكسة.

\textbf{6.} \qty{3.2}{m} نهارًا، و \qty{0.3}{m} ليلًا.

\textbf{7.} $f_p = \qty{0.9}{MHz}$ مع التصادمات: فتُمتصّ الأمواج المتوسطة والقصيرة
(أي انقطاع راديوي)؛ وأما تحديد المواقع عند \qty{1.5}{GHz} فيحسّ امتصاصًا
$\propto 1/\omega^2$، وهو مهمَل.

\textbf{8.} $f_1\Delta t = 1.575 \times 10^9 \times 1.1 \times 10^{-8} = 17$ دورة.

\textbf{9.} $\delta = \sqrt{2/\mu_0\gamma\omega} = \qty{10}{\micro m}$: فجدار بسماكة ميليمتر
مئة عمق قشرة.

\textbf{10.} $R_s = 1/\gamma\delta = \qty{0.1}{\ohm}$؛ و $\tfrac12R_sH_0^2 = \qty{45}{W/m^2}$: أي \qty{45}{W}،
و $6\%$.

\textbf{11.} $\eu^{-\pi \times 0.5} = 0.21$: أي \qty{-14}{dB} في السعة، و \qty{-27}{dB} في
الاستطاعة (والثقب الضئيل يقترن قليلًا أصلًا).

\textbf{12.} الضوء، $\lambda = \qty{0.5}{\micro m}$، أصغر ألفي مرة
من الثقوب فيمرّ؛ وأما الموجة الصغرية، $\lambda = \qty{12}{cm}$، فأكبر
مئة مرة.

\textbf{13.} $\underline n = 2 - 0.005\iu$؛ و $c/2n''\omega = \qty{1.9}{m}$: فلا يكاد الزجاج
يسخن.

\textbf{14.} $j \approx H_0/\delta = \qty{3e7}{A/m^2}$، و $j^2/\gamma = \qty{9e8}{W/m^3}$ في طبقة
سماكتها عشرة ميكرومترات: فتسخن الأطراف في ميلي ثوانٍ، وتبثّ إلكترونات،
وتؤيّن الهواء --- ومن هنا الشرر.

\textbf{15.} $\delta = \qty{18}{\micro m}$؛ والمقطع $2\pi a\delta = \qty{5.6e-8}{m^2}$ مقابل
$\pi a^2 = \qty{7.9e-7}{m^2}$: ومنه $R_{\text{متناوبة}} = \qty{0.30}{\ohm/m}$، أي أربعة عشر ضعف المستمرة.

\textbf{16.} $RI^2 = \qty{120}{W/m}$؛ والمساحة $\pi d = \qty{3.1e-3}{m^2}$ في المتر:
ومنه $\Delta T \approx \qty{1900}{K}$ --- فينصهر؛ ولهذا تكون هذه الوشائع أنابيب مبرَّدة بالماء.

\textbf{17.} $\delta = \qty{0.14}{mm}$: فيبقى التيار المحرَّض، وتبقى الحرارة،
في عُشر الميليمتر؛ ونبضة قصيرة يتبعها إخماد تقسّي
سطح الترس بينما يبقى لبّه صلدًا.

\textbf{18.} $R = 1/\gamma2\pi a\delta = \qty{0.10}{\ohm/m}$، أي أقل ثلاث مرات؛ و
الداخل لا يحمل شيئًا --- ومن هنا الأنبوب، وفيه المبرّد حيث كان
النحاس ليضيع.

\textbf{19.} $\delta = a$ عند $f = 1/\pi\mu_0\gamma a^2 = \qty{17}{Hz}$؛ ودونه يكون التيار
منتظمًا وتصحّ صيغة التيار المستمر.

\textbf{20.} $n^2 \approx 1 + \omega_p^2/\omega_0^2 + (\omega_p^2/\omega_0^2)(\lambda_0/\lambda)^2$: أي صيغة كوشي
مع $n(\infty)^2 - 1 = \omega_p^2/\omega_0^2 = 1.25$ من أجل $n(\infty) = 1.5$.

\textbf{21.} $n = 1.531$ و $1.510$، و $v = 1.96$ و \qty{1.99e8}{m/s}؛ و $n_g(550) =
1.550$؛ و $n_g(400) - n_g(700) = 0.063$: أي \qty{210}{ns} على الكيلومتر.

\textbf{22.} $\sin((A + D)/2) = 1.5165 \times 0.5$: ومنه $D = \qty{38.6}{\degree}$؛ و $\dd D = 2\sin
(A/2)\dd n/\cos((A + D)/2) = 1.53\,\dd n$: ومنه $\Delta D = \qty{1.85}{\degree}$، أي \qty{6.5}{cm} على \qty{2}{m}.

\textbf{23.} عند \qty{200}{nm} يقترب التواتر من الرنين و
ينمو $n''$: فيصير عاتمًا. ورنين ضعيف عند \qty{700}{nm} يمتصّ الأحمر:
فيكون الزجاج أخضر.

\textbf{24.} الأزرق أقرب إلى رنين فوق البنفسجي: فتستجيب الإلكترونات
المرتبطة أكثر، وتكبر $n$، ويكبر الانحراف معها.

\textbf{25.} الطبقة F عند \qty{1.5}{GHz}: $\underline\gamma$ تخيلية، وانتشار
بتأخّر مجموعة صغير. والفولاذ عند \qty{2.45}{GHz}: $\underline\gamma$ حقيقية، \omterm{thm:b2:waves-in-media:skin}{وأثر
قشري}، وانعكاس. والنحاس عند \qty{13.56}{MHz}: الأمر نفسه، بقشرة
\qty{18}{\micro m}. والزجاج في المرئي: $\iu\omega\varepsilon_0\chi$، وقرينة حقيقية، ووسط
شفاف متبدّد.
\end{solution}
