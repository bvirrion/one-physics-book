\begin{solution}{pb:b2:thermal-radiation:1}
\textbf{1.} \qty{5800}{K}.

\textbf{2.} \qty{6.4e7}{W/m^2}؛ و \qty{3.9e26}{W}.

\textbf{3.} $L/4\pi d^2 = \qty{1.37e3}{W/m^2}$؛ و $u = S/c = \qty{4.6e-6}{J/m^3}$؛
و $P = S/c = \qty{4.6e-6}{Pa}$ إذا امتُصّ، وضعف ذلك إذا انعكس.

\textbf{4.} $2.7k_BT = \qty{2.2e-19}{J}$ (أي \qty{1.35}{eV})؛ و $6 \times 10^{21}$
فوتونًا في الثانية في المتر المربع.

\textbf{5.} $3.9 \times 10^{26} \times 1.4 \times 10^{17} = \qty{5.5e43}{J}$؛ و $Mc^2 =
\qty{1.8e47}{J}$: أي ثلاثة أجزاء من عشرة آلاف من كتلتها.

\textbf{6.} \qty{1.2e17}{W}؛ و $T_{\text{e}} = \qty{255}{K}$.

\textbf{7.} تُعترَض على $\pi R^2$، وتُصدَر من $4\pi R^2$ بفضل
الدوران والرياح؛ وأما النقطة تحت الشمس وحدها: $((1 - A)S/\sigma)^{1/4} =
\qty{360}{K}$.

\textbf{8.} \qty{381}{K}؛ وليلًا تنزف الحرارة المخزونة ($\sim C\Delta T$)
بالإشعاع بثابت الزمن $C/4\sigma T^3 \approx \qty{1}{d}$، وهو قصير
في وجه أسبوعي الظلام: فتهبط الأرض إلى حيث يوازن
إشعاعها ما ينساب من تحت، أي \qty{100}{K}.

\textbf{9.} \qty{10}{\micro m}، أي عشرون ضعف قمة الشمس.

\textbf{10.} $(0.65/0.70)^{1/4} = 0.982$: أي أدنى بمقدار \qty{4.7}{K}، أي \qty{250}{K}.

\textbf{11.} الفضاء: $\sigma T_{\text{a}}^4 = \sigma T_{\text{e}}^4$؛ والطبقة: $\sigma
T_{\text{s}}^4 = 2\sigma T_{\text{a}}^4$؛ والسطح: $\sigma T_{\text{s}}^4 = \sigma
T_{\text{e}}^4 + \sigma T_{\text{a}}^4$. ومن ثَمّ $T_{\text{s}} = 2^{1/4}T_{\text{e}} = \qty{303}{K}$.

\textbf{12.} $T_{\text{s}}^4 = T_{\text{e}}^4/(1 - \varepsilon/2)$؛ و $(288/255)^4 = 1.63$
يعطي $\varepsilon = 0.77$.

\textbf{13.} بخار الماء وثنائي أكسيد الكربون والميثان والأوزون تمتصّ في
تحت الأحمر (بعصابات اهتزاز: H$_2$O حول \qty{6}{\micro m} وبعد
\qty{17}{\micro m}، و CO$_2$ عند \qty{15}{\micro m})؛ وأما N$_2$ و O$_2$ فلا (إذ لا
ثنائي قطب لهما)؛ والنافذة \qty{8}{}--\qty{13}{\micro m}.

\textbf{14.} $T_{\text{a}}^4 = T_{\text{s}}^4/2$: $\varepsilon\sigma T_{\text{a}}^4 \approx
\qty{150}{W/m^2}$، في وجه \qty{238}{W/m^2} من ضوء الشمس الممتصّ (وجوّ
متعدد الطبقات يعيد أكثر، نحو \qty{330}{W/m^2}).

\textbf{15.} $3.7/3.8 \approx \qty{1}{K}$.

\textbf{16.} \qty{2.1e8}{J/m^2/K}؛ و $\tau \approx \qty{5.6e7}{s}$، أي سنتان.

\textbf{17.} تعكس ضوء الشمس وتشعّ تحت الأحمر نازلًا؛ وليلًا
لا يعمل إلا الثاني: فالليالي الغائمة أدفأ.

\textbf{18.} \qty{49}{mm^2}؛ و \qty{31}{cm}؛ وهي ملفوفة لتتسع وللتقليل من
الضياعات الحملية إلى الغاز المالئ.

\textbf{19.} \qty{1.04}{\micro m}؛ و \qty{4.8}{W}؛ و $8\%$.

\textbf{20.} $R = V^2/P = \qty{880}{\ohm}$؛ وباردةً \qty{88}{\ohm}: أي \qty{2.6}{A}
بدل \qty{0.26}{A} --- فالمصابيح تنقطع عند التشغيل.

\textbf{21.} $49 \times (2800/3200)^4 = \qty{29}{mm^2}$؛ و \qty{8.4}{W} مرئية؛ وثماني
مضاعفات للتبخر، أي عمر أقصر $250$ مرة --- أي ساعات.

\textbf{22.} يرتبط التنغستن المتبخر بالهالوجين ويعود
إلى الشعيرة الساخنة بدل أن يسوّد الزجاج: فشعيرة أسخن
بعمر عادي، في غلاف كوارتز صغير.

\textbf{23.} \qty{55}{W} من تحت الأحمر والتوصيل عبر الغاز؛ والزجاج
يمتصّ جزءًا من تحت الأحمر.

\textbf{24.} $C = \qty{2.8e-3}{J/K}$؛ ومع $P \propto T^4$، يكون زمن انتصاف $T$
هو $7CT_0/3P \approx \qty{0.3}{s}$ --- والتوهج يتبع الشبكة
بتأخر \qty{0.04}{s}، ولهذا لا يكاد يومض.

\textbf{25.} تثبّت الحرارة شكل الطيف وقمته
($\lambda_{\max}T$)، والمجموع ($\sigma T^4$)؛ والشعيرة والشمس والأرض
ثلاثة أجسام سوداء، عند \qty{2800}{} و \qty{5800}{} و \qty{288}{K}، وكلٌّ منها
يوازن ما يتلقاه بالمقدار $\sigma T^4$.
\end{solution}
