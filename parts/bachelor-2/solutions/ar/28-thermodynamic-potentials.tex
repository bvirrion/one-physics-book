\section*{الفصل \ref{ch:b2:thermodynamic-potentials} --- الكمونات الترموديناميكية}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$، و $G = nRT\ln P + g(T)$؛ و $P = -\partial_VF = nRT/V$،
و $V = \partial_PG = nRT/P$؛ و $-\partial_TF = nR\ln V - f'(T)$، أي الإنتروبيا
$nC_V\ln T + nR\ln V + \text{ثابت}$ إذا كان $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$، و $\Delta U = 0$، و $\Delta S = \qty{19.1}{J/K}$.
وعكوسًا: \qty{5.7}{kJ}، و $S_{\text{c}} = 0$؛ وفي وجه \qty{1}{bar}: \qty{0.9}{kJ}،
و $S_{\text{c}} = 19.1 - 900/300 = \qty{16}{J/K}$؛ وفي الخلاء: لا شيء، و $S_{\text{c}} =
\qty{19.1}{J/K}$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$ مع $\Delta h \approx L$ و $\Delta s \approx L/T_{\text{انصهار}}$.
وعند \qty{-5}{\celsius}: $g_{\text{سائل}} - g_{\text{جليد}} = +\qty{6.1}{kJ/kg}$، فالجليد
مستقر، والسائل شبه مستقر (فالتبريد المفرط موجود)؛ وعند $+\qty{5}{\celsius}$:
\qty{-6.1}{kJ/kg}، فالسائل مستقر --- والجليد المفرط التسخين لا يُرصَد، إذ
ينصهر سطحه أولًا.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:4}

ينمو الميل).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(أ) تذهب المادة إلى $\mu$ الأدنى: $\mu_{\text{v}} < \mu_{\text{سائل}}$ حين $p <
p_{\text{إشباع}}$؛ والرطوبة $= p/p_{\text{إشباع}}$. (ب) و $-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
لمصلحة البخار. (ج) والتبخر لا يحتاج إلا إلى $p < p_{\text{إشباع}}$،
لا إلى الغليان؛ وعند $100\%$ يتساوى الكمونان. (د) وعلى أبرد
السطوح، التي هبط $p_{\text{إشباع}}(T_{\text{s}})$ لها دون $p$ في الهواء
--- أي التي تشعّ إلى سماء الليل.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(أ) قياسي. (ب) و $b = RT_{\text{c}}/8P_{\text{c}} = \qty{4.3e-5}{m^3/mol}$، و $a =
27R^2T_{\text{c}}^2/64P_{\text{c}} = \qty{0.36}{Pa.m^6/mol^2}$. (ج) و $\dd g = v\,\dd P$
عند $T$ ثابتة و $g$ متساوٍ عند الطرفين: $\int v\,\dd P = 0$ حول
الحلقة. (د) وغير مستقرة حيث $\partial_vP > 0$؛ وشبه مستقرة بينهما: أي
السائل المفرط التسخين (الذي يقفز) والبخار المفرط الإشباع (وهو ضباب
ينتظر النوى، وغرفة الغيوم).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}

(ب) و \qty{28}{mbar} في وجه \qty{23}{mbar} --- إذ $L$ أكبر عند \qty{20}{\celsius}.
(ج) و \qty{820}{Pa} في وجه \qty{611}{Pa}. (د) و $\partial(\Delta g/T)/\partial T = -L/T^2$

\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(أ) تساوي $\mu$ عبر الغشاء: $\mu^\circ(P + \Pi) - RTx_{\text{s}} =
\mu^\circ(P)$، أي $v_{\text{m}}\Pi = RTx_{\text{s}}$، و $\Pi = cRT$. (ب) و \qty{27}{bar}.
(ج) و $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$ في المتر المكعب؛ والمنشآت الحقيقية
أربعة أضعاف ذلك. (د) والماء يجري إلى الكمون الأدنى: أي إلى
الخلية الغنية بالمذاب، وخارجًا منها في الماء المالح.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(أ) $S = S_0 - k_BL^2/2Na^2$؛ و $f = k_BTL/Na^2$. (ب) و $k_BT/Na^2 \propto T$. (ج)
و \qty{1.7e-13}{N}؛ و $1.8 \times 10^{14}$. (د) و $E \sim nk_BT \approx \qty{0.4}{MPa}$.
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$؛ و $\dd F = -S\dd T + f\dd L$.

\textbf{2.} شوارتز على $\dd F$.

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$: فالطاقة
لا تتغير؛ والشدّ إنتروبي.

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$؛ و $Q = \qty{-1.5}{J}$؛ و $W = +\qty{1.5}{J}$؛ والمجموع
معدوم.

\textbf{5.} $+\qty{0.4}{K}$.

\textbf{6.} $L - L_0 \propto 1/T$: أي \qty{9.1}{cm}، وانكماش \qty{9}{mm}؛ فتقصر
الأضلاع المسخَّنة، وينزاح مركز كتلة الإطار، فيدور
الدولاب.

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$؛ و $f = k_BTL/Na^2$؛ وثمن الإنتروبيا
يُدفَع بالمقدار $k_BT$.

\textbf{8.} \qty{1.7e-13}{N}؛ و $1.8 \times 10^{14}$ سلسلة.

\textbf{9.} قوته طاقية ($\partial_LU \ne 0$)؛ والتسخين يليّن
الروابط، فتهبط $f$ مع $T$.

\textbf{10.} الميول $-s$، مع $s_{\text{غاز}} > s_{\text{سائل}} > s_{\text{جليد}}$؛ و
أدنى الخطوط هو الطور المستقر؛ والتقاطعان عند $T_{\text{انصهار}}$ و $T_{\text{تبخر}}$.

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}.

\textbf{12.} $\qty{-1.3e7}{Pa/K}$؛ وهو سالب لأن $v_{\text{سائل}} < v_{\text{جليد}}$.

\textbf{13.} \qty{70}{bar}: \qty{-0.5}{K} --- وليس هكذا يتزلج المرء عند
\qty{-10}{\celsius}؛ بل بالاحتكاك والطبقة السطحية الشبيهة بالسائل.

\textbf{14.} \qty{270}{bar}، و \qty{-2}{K}: ومع التدفق الجوفي، تنصهر
القاعدة وتنزلق الصفيحة.

\textbf{15.} \qty{3.6}{kPa/K}؛ و \qty{92}{\celsius}؛ ونحو \qty{127}{\celsius}
(و \qty{120}{} في الواقع).

\textbf{16.} \qty{28}{mbar} (و \qty{23}{})، و \qty{820}{Pa} (و \qty{611}{}): إذ تنمو $L$ كلما هبطت $T$؛
والأخطاء $20$--$30\%$.

\textbf{17.} $\Delta v$ نفسه، ومن ثَمّ نسبة الميلين $L_{\text{تسام}}/L_{\text{تبخر}}
= 1.15$: فمنحني التسامي أشدّ ميلًا --- أي كسرة عند النقطة الثلاثية.

\textbf{18.} \qty{12}{kJ/kg}؛ ويجب أن يتنوّى الجليد، والنواة الصغيرة تكلّف
من طاقة السطح أكثر مما تربح: فتنتظر القطيرات النقية، حتى
\qty{-40}{\celsius}.

\textbf{19.} $\Delta v \to 0$، و $L \to 0$، ونسبتهما منتهية: فينتهي المنحني؛
وفي ما وراءه مائع واحد يمرّ متصلًا من شبيه السائل إلى شبيه الغاز.

\textbf{20.} $\dd g = v\,\dd P$ عند $T$ ثابتة؛ و $g$ متساوٍ عند طرفي
الخط الأفقي: $\int v\,\dd P = 0$، أي مساحتان متساويتان.

\textbf{21.} فرع السائل المفرط التسخين: إذ لا نواة فقاعة في
كأس ملساء؛ وأدنى اضطراب يغليه دفعة واحدة.

\textbf{22.} جمّد، واضخخ دون \qty{611}{Pa}، وسخّن برفق: فيعبر المسار
منحني التسامي، ولا يعبر منحني الانصهار قط؛ فتُحفَظ
البنية.

\textbf{23.} $\mu_{\text{v}}(p) < \mu_{\text{سائل}}$ حين $p < p_{\text{إشباع}}$؛ ويتساويان عند
الإشباع.

\textbf{24.} \qty{120}{\celsius} بدل \qty{100}{}: أي أسرع أربع مرات.

\textbf{25.} $S$ للمعزول، و $F$ عند $T,V$ ثابتين، و $G$ عند $T,P$ ثابتين؛
و $-\Delta F$ و $-\Delta G$ هما العمل المستعاد؛ وعلاقات ماكسويل تحوّل
مشتقات $S$ إلى معادلة الحالة.
\end{solution}
