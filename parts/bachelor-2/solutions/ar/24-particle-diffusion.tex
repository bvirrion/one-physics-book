\begin{solution}{exo:b2:particle-diffusion:6}
(أ) $n = n_{\text{s}}a/r$، و $\Phi = 4\pi D a n_{\text{s}} = \qty{3.6e-8}{mol/s}$.
(ب) و $1.95 \times 10^{-5}\,\unit{mol}$؛ و \qty{540}{s}. (ج) وقشرة $1/r$ تحتاج إلى
$\sim a^2/D \approx \qty{2000}{s}$ لتتكوّن والحبيبة تتقلص؛ والتقليب
يستبدل بالقشرة التي سماكتها $\sim a$ طبقةً حدية $\delta \ll a$
ويضرب التدفق في $a/\delta$. (د) و $(rn')' = 0$: $n = A + B\ln r$،
وهو لا يمكن أن ينعدم في اللانهاية: فلا حالة مستقرة --- إذ تظل غيمة
الأسطوانة تكبر (لوغاريتميًّا).
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:7}
(أ) $\dd/\dd t\int n = D[n']_{-\infty}^{\infty} = 0$؛ و $\dd/\dd t\int x^2 n = D\int
x^2 n'' = 2D\int n = 2DN$. (ب) و $\partial_t[n(x,-t)] = -D\,\partial_x^2 n$: فتنقلب
الإشارة. (ج) ومن أجل $t < 0$ يكون ``العرض'' $2Dt$ سالبًا: فلا وجود لحالة
كهذه --- إذ لا يمكن ردّ النبضة عن انتشارها. (د) و $\dd/\dd t\int n^2 = 2D\int nn''
= -2D\int n'^2 \le 0$.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:11}
(أ) $n = n_\infty(1 - a/r)$، و $\Phi = 4\pi D a n_\infty$. (ب) و $3.8 \times
10^6$ في الثانية. (ج) وتعمل المستقبلات عمل مواصلات على التفرّع ثم
على التسلسل مع القشرة الكروية: $\Phi \approx 4\pi Dan_\infty
\cdot N_{\text{r}}s/(N_{\text{r}}s + \pi a)$؛ ونصف العظمى عند
$N_{\text{r}}s = \pi a$: فمع $s = \qty{1}{nm}$، يكون $N_{\text{r}} \approx
3000$، مغطّيًا $N_{\text{r}}s^2/4a^2 \approx 10^{-3}$ من السطح.
(د) فكل نوع من المستقبلات يحتاج إلى مساحة مهمَلة ليكون الالتقاط
شبه أعظمي: فتستطيع الخلية مراقبة آلاف المواد معًا.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:12}
(أ) فرق أمامي في $t$، وفرق ثانٍ مركزي في $x$. (ب) ومع
$\alpha = 1/2$: $n_i^{k+1} = (n_{i-1}^k + n_{i+1}^k)/2$ --- أي إن كل جسيم يقفز
يمينًا أو يسارًا باحتمال $1/2$. (ج) ومن أجل $n_i = (-1)^i$، $n^{k+1} =
(1 - 4\alpha)n^k$: و $|1 - 4\alpha| > 1$ حين $\alpha > 1/2$. (د) و $\Delta t \le
\Delta x^2/2D = \qty{3.3}{s}$؛ ونحو $1100$ خطوة.
\end{solution}

\begin{solution}{pb:b2:particle-diffusion:1}
\textbf{1.} $k_BT = \qty{0.118}{eV}$، و $E_{\text{a}}/k_BT = 29.2$: $D =
\qty{1.6e-17}{m^2/s}$؛ وعند \qty{1000}{\celsius}: \qty{1.6e-18}{m^2/s} --- أي
بعامل $10$.

\textbf{2.} يُعالَج السطح العاكس بصورة النبضة المرآوية،
فتتضاعف السعة على $x > 0$؛ و $\int_0^\infty = (Q/\sqrt{\pi Dt})
\cdot \tfrac12\sqrt{4\pi Dt} = Q$.

\textbf{3.} $Dt = \qty{5.8e-14}{m^2}$؛ و $\sqrt{\pi Dt} = \qty{0.43}{\micro m}$؛ و $n_{\text{s}}
= \qty{2.3e24}{m^{-3}}$.

\textbf{4.} $\eu^{-x^2/4Dt} = 4.3 \times 10^{-4}$: $x_j^2 = 4Dt \times 7.75$، و $x_j =
\qty{1.3}{\micro m}$.

\textbf{5.} ينتصف $n_{\text{s}}$، ويهبط اللوغاريتم إلى $7.05$، ويتضاعف $4Dt$
أربع مرات: $x_j = \qty{2.6}{\micro m}$ --- أي أقلّ من الضعف.

\textbf{6.} $\delta D/D = (E_{\text{a}}/k_BT)\,\delta T/T = 21\%$؛ و $x_j \propto
\sqrt{Dt}$ تقريبًا، فيحتاج $1\%$ على $x_j$ إلى $2\%$ على $D$: أي \qty{1}{K}.

\textbf{7.} $E_{\text{a}}/k_BT = 134$: $D \approx 10^{-62}\,\unit{m^2/s}$؛ \omterm{def:b2:maxwell-equations:operators}{وتباعد}
ذري واحد (\qty{0.25}{nm}) في $a^2/D \sim 10^{43}\,\unit{s}$: أي مجمَّد.

\textbf{8.} $E_{\text{a}}/k_BT = 32.8$: $D = \qty{4.3e-19}{m^2/s}$؛ و $\sqrt{Dt} =
\qty{28}{nm}$.

\textbf{9.} $n_0\operatorname{erfc}(x/2\sqrt{Dt})$؛ و $n_B/n_0 = 5 \times 10^{-6}$:
$x \approx 6.4\sqrt{Dt} = \qty{0.18}{\micro m}$.

\textbf{10.} $2n_0\sqrt{Dt/\pi} = \qty{6.3e18}{m^{-2}}$: أي ستة أضعاف $Q$ ---
فالجرعة تنمو وفق $\sqrt t$ ويحددها الزمن.

\textbf{11.} الحرارة المنخفضة: $D$ صغير، والجرعة يضبطها الزمن،
والعمق ضحل؛ والحرارة العالية: تُدفَع الجرعة الثابتة عميقًا وسريعًا.

\textbf{12.} \qty{0.18}{\micro m} في وجه \qty{1.3}{\micro m}: أي رقيقة.

\textbf{13.} $4 \times 10^{18}$ جزيء؛ و $n_{\text{a}} = \qty{2.4e25}{m^{-3}}$.

\textbf{14.} $\ell = \qty{37}{nm}$؛ و $v^* = \qty{206}{m/s}$.

\textbf{15.} $D \approx \qty{2.5e-6}{m^2/s}$.

\textbf{16.} $L^2/D = 10^7\,\unit{s}$، أي أشهر؛ ومع الانسياق: \qty{50}{s}.

\textbf{17.} $n = 8N(4\pi Dt)^{-3/2}\eu^{-r^2/4Dt}$؛ وهو أعظم عند $t^* = r^2/6D =
\qty{6.7e4}{s}$، أي ثماني عشرة ساعة.

\textbf{18.} $n_{\max} = 8N(4\pi Dt^*)^{-3/2}\eu^{-3/2} \approx \qty{2e18}{m^{-3}}$،
أي فوق العتبة بكثير: فيُشَمّ، في النهاية.

\textbf{19.} $r^2 = 4Dt\ln 10^6 = \qty{12}{m^2}$: أي \qty{3.5}{m}.

\textbf{20.} $v^*t/\ell \approx 5 \times 10^{14}$.

\textbf{21.} $\langle x\rangle = 0$، و $\langle x^2\rangle = N\ell^2 = \ell^2 t/\tau$؛
و $D = \ell^2/2\tau$.

\textbf{22.} $\binom{N}{N/2}/2^N$: $1/2$، $3/8$، $5/16$ --- أي متناقص
(وفق $1/\sqrt{\pi N/2}$): فالماشي يبتعد وفق $\sqrt N$.

\textbf{23.} تتجمع الغيمة في نقطة --- وهو ما لا يُرى قط؛ وهو
ينتهك \omterm{thm:b2:particle-diffusion:equation}{معادلة الانتشار}، لا قوانين الميكانيك.

\textbf{24.} كل تصادم عكوس، لكن الحالة الابتدائية (أي الجسيمات
كلها معًا) استثنائية: فكل تاريخ مجهري تقريبًا
منها ينتشر، ويحتاج العكس إلى تآمر السرعات
كلها. ومتوسط المسير العشوائي لا يبقي إلا ما هو نمطي
ويطرح تلك المعلومة؛ فاللاعكوسية إحصائية.

\textbf{25.} $L \sim \sqrt{Dt}$، و $t \sim L^2/D$؛ و $D \sim \ell v^*/3$، أي
الخطوة بين التصادمات: $10^{-5}$، $10^{-9}$، $10^{-17}\,\unit{m^2/s}$.
\end{solution}
