\begin{solution}{exo:b2:guided-waves:5}
(a) $\underline B_y = (k_g/\omega)\underline E$, $\underline B_z = (\iu/\omega)\partial_y\underline E = (\iu\pi/a\omega)
E_0\cos(\pi y/a)\eu^{\iu(\omega t - k_gz)}$: in quadrature. (b) At $y = 0$, $B_y = 0$ and
$\vect j_s = \vect e_y\wedge\vect B/\mu_0 = (B_z/\mu_0)\vect e_x$: along the field. (c)
$\langle\Pi_z\rangle = E_0^2k_g\sin^2(\pi y/a)/2\mu_0\omega$, power per unit width
$E_0^2k_ga/4\mu_0\omega$; $\langle\Pi_y\rangle \propto \operatorname{Re}(\underline E\,\iu\underline B_z^*)
= 0$. (d) Mean energy per unit length and width $\tfrac14\varepsilon_0E_0^2a$ (electric
and magnetic halves each $\tfrac18\varepsilon_0E_0^2a$); the ratio is $c^2k_g/\omega = v_g$.
\end{solution}

\begin{solution}{exo:b2:guided-waves:9}
(a) Points $(m, n, p)$ with $m^2 + n^2 + p^2 \le (2fL/c)^2$ fill an eighth
of a sphere: $\tfrac18\cdot\tfrac43\pi(2fL/c)^3$, times two polarizations:
$8\pi L^3f^3/3c^3$. (b) $8\pi f^2/c^3$. (c) $9 \times 10^{-7}$ per hertz (one per
megahertz) at \qty{1}{GHz}; $2 \times 10^5$ per hertz at \qty{500}{THz}. (d) Modes are
discrete when their spacing exceeds their width; in the oven, one mode
per \qty{10}{MHz} or so around \qty{2.45}{GHz} --- dozens within a few
percent.
\end{solution}

\begin{solution}{exo:b2:guided-waves:10}
(a) From $\operatorname{\vect{curl}}\vect E = -\partial_t\vect B$:
\[
\underline B_x = -\frac{k_g}\omega\underline E , \qquad
\underline B_z = \frac{\iu\pi}{a\omega}E_0\cos\frac{\pi x}a\,\eu^{\iu(\omega t - k_gz)} ,
\]
and $\partial_xB_x + \partial_zB_z = -(k_g\pi/a\omega)E_0\cos(\pi x/a) + (k_g\pi/a\omega)E_0
\cos(\pi x/a) = 0$. (b) $(\operatorname{\vect{curl}}\vect B)_y = \partial_zB_x - \partial_xB_z =
(\iu/\omega)(k_g^2 + \pi^2/a^2)\underline E = (\iu\omega/c^2)\underline E$: $k_g^2 = \omega^2/c^2 - \pi^2/a^2$.
(c) Broad walls: $\vect j_s = \pm(B_z\vect e_x - B_x\vect e_z)/\mu_0$ --- along $z$
as $\sin(\pi x/a)$, across as $\cos(\pi x/a)$ (zero at the centre); narrow
walls: along $y$ only. (d) A central longitudinal slot cuts only
transverse currents, which vanish there; a transverse slot cuts the
longitudinal ones and radiates: the slotted-waveguide antenna.
\end{solution}

\begin{solution}{exo:b2:guided-waves:11}
(a) $2a\sqrt{n_1^2 - n_2^2}/\lambda = 2 \times 50 \times 0.242/1.5 \approx 16$ per polarization.
(b) $a < \lambda/2\mathrm{NA} = \qty{3.1}{\micro m}$. (c) A round core with $\mathrm{NA} =
0.12$ is single-mode below $d = 2.405\lambda/\pi\mathrm{NA} = \qty{10}{\micro m}$: the
\qty{9}{\micro m} standard. (d) A grazing ray is always totally reflected: the
lowest mode exists at every frequency, its evanescent tails merely
spreading into the cladding.
\end{solution}

\begin{solution}{exo:b2:guided-waves:12}
(a) The off-axis ray spends its time in lower index, where light is
faster: the longer path is compensated. (b) $n(r)\cos\theta = n_1\cos\theta_0$
with $n \approx n_1(1 - \Delta r^2/a^2)$ and $\cos\theta \approx 1 - \theta^2/2$: $(\dd r/\dd z)^2 =
\theta_0^2 - 2\Delta r^2/a^2$, a harmonic oscillation $r = r_0\sin(\sqrt{2\Delta}\,z/a)$.
(c) $2\pi a/\sqrt{2\Delta} = \qty{1.1}{mm}$. (d) $n_1L\Delta^2/2c = \qty{0.25}{ns/km}$ against
\qty{49}{ns/km}: two hundred times better.
\end{solution}

\begin{solution}{pb:b2:guided-waves:1}
\textbf{1.} $c/2a = \qty{1.74}{GHz}$; TE$_{20}$ and TE$_{01}$: \qty{3.49}{GHz}: only
TE$_{10}$ at \qty{2.45}{GHz}.

\textbf{2.} $\sqrt{1 - (1.74/2.45)^2} = 0.70$: $\lambda_g = \qty{17.4}{cm}$, $v_\varphi = 1.42c$,
$v_g = 0.70c = \qty{2.1e8}{m/s}$.

\textbf{3.} $E_0^2 = 4P/\varepsilon_0c\,ab\sqrt{\cdot} = 3200/6.9 \times 10^{-6}$: $E_0 = \qty{21}{kV/m}$,
a hundredth of breakdown.

\textbf{4.} $j_s \approx E_0/\mu_0c = \qty{57}{A/m}$; $\delta = \qty{10}{\micro m}$, $R_s = \qty{0.1}{\ohm}$;
$\tfrac12R_sj_s^2 \times 2a = \qty{28}{W/m}$: $3.5\%$ per metre --- steel is lossy,
the guide is short.

\textbf{5.} $f = 1.5 \times 10^8\sqrt{11.1(m^2 + n^2) + 25p^2}$ within $\pm3\%$ of
\qty{2.45}{GHz}: $(1, 2, 3)$, $(2, 1, 3)$ at \qty{2.51}{GHz}, $(4, 0, 2)$, $(0, 4, 2)$,
$(4, 3, 0)$, $(3, 4, 0)$ at \qty{2.50}{GHz} --- about six.

\textbf{6.} Half-wavelengths $a/m$, $b/n$, $d/p$: $7$ to \qty{10}{cm} apart. The
turntable drags the food through maxima and minima; a stirrer (a
rotating metal fan) reshuffles the modes.

\textbf{7.} $f/Q = \qty{12}{MHz}$: the loaded resonances overlap into a
continuum; the magnetron always finds a mode to feed.

\textbf{8.} $W = QP/\omega = 3000 \times 800/1.54 \times 10^{10} = \qty{0.16}{J}$; $u = \qty{8.7}{J/m^3}$;
$E_0 \approx \sqrt{4u/\varepsilon_0} = \qty{2}{MV/m}$ --- near breakdown: arcs; the reflected
power goes back to the magnetron, protected (a little) by a dummy
load or circulator --- hence "never run it empty".

\textbf{9.} $0.2/2.1 \times 10^8 = \qty{1}{ns}$.

\textbf{10.} \qty{1.6}{kW} peak; between bursts the field rings down in
$Q/\omega = \qty{13}{ns}$: the cavity is empty most of the time.

\textbf{11.} $\ell \approx a/\pi = \qty{0.48}{mm}$; $\eu^{-1/0.48}$: \qty{-18}{dB} in amplitude,
\qty{-36}{dB} in power.

\textbf{12.} $\lambda/4 = \qty{3.1}{cm}$.

\textbf{13.} \qty{5}{mW/cm^2} $\times$ \qty{100}{cm^2} = \qty{0.5}{W}: $0.06\%$.

\textbf{14.} $f_1 \approx c/2d = \qty{3.75}{GHz} > \qty{2.45}{GHz}$: below cut-off; $\ell =
\qty{1.7}{cm}$, $\eu^{-30/1.7}$: \qty{-150}{dB}.

\textbf{15.} The wall currents must flow continuously into the mesh;
a gap in the bond is a slot that cuts them --- a slot antenna
radiating outward.

\textbf{16.} $\mathrm{NA} = \sqrt{1.465^2 - 1.460^2} = 0.12$, \qty{7}{\degree}: a narrow
cone, needing a laser and a lens.

\textbf{17.} $(n_1L/c)(n_1/n_2 - 1) = \qty{17}{ns/km}$; \qty{170}{ns} over \qty{10}{km}:
\qty{6}{Mbit/s}.

\textbf{18.} $\tfrac12(\pi d\,\mathrm{NA}/\lambda)^2 = 75$; for \qty{9}{\micro m}: $2.4$ --- the
single-mode regime ($V = \pi d\,\mathrm{NA}/\lambda = 2.2 < 2.4$).

\textbf{19.} $\Delta x_0 = (c/n)\tau = \qty{1}{cm}$: $t_{\text{sp}} = 10^{-4}/0.2 = \qty{0.5}{ms}$,
\qty{100}{km}: \qty{10}{Gbit/s} over that distance before compensation.

\textbf{20.} \qty{20}{dB}, a factor $100$: \qty{10}{\micro W}; $6000/80 = 75$
amplifiers.

\textbf{21.} $f = \qty{1.9e14}{Hz}$, $hf = \qty{0.8}{eV}$; $8 \times 10^{15}$ photons per
second; $8 \times 10^5$ per bit at the input, $8 \times 10^3$ after \qty{100}{km}.

\textbf{22.} The sum of the two losses is minimal near \qty{1.55}{\micro m}
(\qty{0.2}{dB/km}).

\textbf{23.} $R = ((1.465 - 1)/2.465)^2 = 3.6\%$: \qty{0.16}{dB} per face, twice
at a connector plus the gap's interference; gel or physical contact
removes the air.

\textbf{24.} On a gentle bend the ray still meets the boundary beyond
the critical angle; on a sharp one the incidence on the outer wall
falls below it and light leaks out.

\textbf{25.} Oven: metal walls, reflection with $r = -1$; bandwidth set
by the cavity's modes; losses in the steel skin. Fibre: \omterm{thm:b2:wave-interfaces:snell}{total internal
reflection} at a glass--glass boundary; bandwidth set by modal and
chromatic dispersion; losses by scattering and absorption, \qty{0.2}{dB/km}.
\end{solution}
