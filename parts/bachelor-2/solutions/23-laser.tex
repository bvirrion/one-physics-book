\begin{solution}{exo:b2:laser:5}
(a) See \cref{prop:b2:laser:einstein}. (b) \qty{600}{nm}: $h\nu/k_BT
= 80$, ratio $5 \times 10^{34}$; \qty{1}{cm}: $4.8 \times 10^{-3}$ ---

\begin{solution}{exo:b2:laser:6}
(a) $I_{\text{s}} = h\nu/\sigma\tau = \qty{1.0e5}{W/m^2}$. (b) $R =
g_0/\sigma\tau = \qty{3.3e22}{m^{-3}.s^{-1}}$. (c) $I = I_{\text{s}}(5 - 1) =
\qty{4.2e5}{W/m^2}$; $P_{\text{out}} = 0.01 \times 4.2 \times 10^5 \times
10^{-6} = \qty{4.2}{mW}$. (d) $I = I_{\text{s}}(\sigma R\tau/g_{\text{th}}
- 1) = (h\nu/g_{\text{th}})(R - g_{\text{th}}/\sigma\tau)$: linear in $R$
above $R_{\text{th}} = g_{\text{th}}/\sigma\tau$.
\end{solution}

\begin{solution}{exo:b2:laser:7}

\end{solution}

\begin{solution}{exo:b2:laser:11}
(a) Intensity $\times 1.04$ every \qty{2}{ns}. (b) $N = P\,(2L/c)/h\nu =
6.4 \times 10^8$ photons; $\ln(6.4 \times 10^8)/\ln 1.04 \approx 520$ round
trips, about \qty{1}{\micro s}. (c) Gain saturation; a mode below
threshold loses more per round trip than it gains and stays at the
spontaneous level. (d) The modes share one inversion; the strongest
saturates it and starves the others; drifts of length and gain change
the winner --- the output jumps from mode to mode.
\end{solution}

\begin{solution}{pb:b2:laser:1}
\textbf{1.} \qty{1.96}{eV}, \qty{4.74e14}{Hz}.

\textbf{2.} $\qty{0.05}{eV} \approx 1.5\,k_BT$: the thermal motion supplies
the small deficit; the transfer is nearly resonant, hence efficient.

\textbf{3.} $\eu^{-20.66/0.0345} = \eu^{-600} \approx 10^{-260}$: none; only the
discharge feeds the level.

\textbf{4.} The lower level empties ten times faster than the upper
one decays: it is always nearly empty --- a four-level scheme.

\textbf{5.} $v^* = \sqrt{2k_BT/m} = \qty{580}{m/s}$; $\nu v^*/c \approx \qty{0.9}{GHz}$;
the exact formula gives \qty{1.5}{GHz}.

\textbf{6.} $g_0 = \qty{0.09}{m^{-1}}$; $\eu^{0.027}$: $2.7\%$ per pass.

\textbf{7.} $g_{\text{th}} = -\ln(0.999 \times 0.99)/0.6 = \qty{0.018}{m^{-1}}$;
margin $5$.

\textbf{8.} \qty{500}{MHz}; three.

\textbf{9.} $P_{\text{in}} = \qty{1}{mW}/0.01 = \qty{100}{mW}$; \qty{1.3e5}{W/m^2}.

\textbf{10.} The gain saturates to $g_{\text{th}}$; the surplus becomes
output photons (and \omterm{def:b2:laser:processes}{spontaneous emission}, heat).

\textbf{11.} $\qty{1}{mW}/\qty{7.5}{W} = 1.3 \times 10^{-4}$: most of the
electron energy goes elsewhere than the helium metastable, and the
photon carries \qty{1.96}{eV} of the \qty{20.66}{eV} invested.

\textbf{12.} $2 \times 0.072 \times 0.3 = 0.043$; $N = 0.1 \times 2 \times 10^{-9}/3.14
\times 10^{-19} = 6.4 \times 10^8$; $20.3/0.043 \approx 470$ round trips;
\qty{0.9}{\micro s}.

\textbf{13.} $\delta\nu = \nu\,\delta L/L = \qty{1.6}{GHz}$: the mode crosses the
whole gain curve --- the \omterm{def:b2:laser:cavity}{laser} hops to another mode unless the length is
stabilised.

\textbf{14.} The $p$ polarization crosses a Brewster surface without
reflection; the $s$ polarization loses some $15\%$ per surface and
never reaches threshold: linearly polarised output at no cost.

\textbf{15.} $z_{\text{R}} = \qty{0.45}{m}$; $\theta = \qty{6.7e-4}{rad}$; $w(\qty{10}{m})
= \qty{6.7}{mm}$: \qty{13}{mm} across.

\textbf{16.} $2P/\pi w_0^2 = \qty{7}{kW/m^2}$: seven suns.

\textbf{17.} $w_0' = \qty{34}{\micro m}$; \qty{5.6e5}{W/m^2}.

\textbf{18.} $w_0' = \qty{11}{\micro m}$; \qty{5e6}{W/m^2}; the retina survives
\qty{1}{mW} for the \qty{0.25}{s} of the aversion reflex --- class 2: visible,
$\le \qty{1}{mW}$, safe because one blinks.

\textbf{19.} \qty{6.7e-5}{rad}; $w_0 = \qty{3}{mm}$, $z_{\text{R}} = \qty{45}{m}$,
$w(\qty{1}{km}) = \qty{67}{mm}$: \qty{13}{cm}.

\textbf{20.} $c/\Delta\nu$: \qty{300}{m}; \qty{30}{cm}.

\textbf{21.} $2.7 \times 10^{17}$; \qty{1}{GW}.

\textbf{22.} \qty{1.8}{km}; \omterm{def:b2:diffraction:huygens}{diffraction} alone: \qty{130}{m}.

\textbf{23.} $0.1/\pi(1800)^2 = 10^{-8}$; $2.6 \times 10^9$ photons.

\textbf{24.} $\lambda/d = \qty{1.3e-5}{rad}$, \qty{5}{km}; $(0.5/5000)^2 = 10^{-8}$:
$25$ photons, two or three detected.

\textbf{25.} \qty{1.5}{cm}; \qty{0.15}{mm}; the recession is measured within

\end{solution}
