\section*{Capítulo \ref{ch:b2:dispersion-wave-packets} --- Dispersión y paquetes de ondas}

\begin{solution}{exo:b2:dispersion-wave-packets:1}
(a) $k = \qty{0.063}{rad/m}$, $\omega = \sqrt{gk} = \qty{0.79}{rad/s}$, $T = \qty{8.0}{s}$;
$v_\varphi = \qty{12.5}{m/s}$, $v_g = \qty{6.3}{m/s}$. (b) $\omega = \qty{0.45}{rad/s}$, $k = \omega^2
/g = \qty{0.021}{rad/m}$, $\lambda = \qty{310}{m}$; $v_\varphi = gT/2\pi = \qty{22}{m/s}$, $v_g =
\qty{11}{m/s}$; $5 \times 10^6/11 = \qty{4.6e5}{s} = 5.3$ días. (c) $\lambda = \qty{6}{m}$, $v_g
= \qty{1.6}{m/s}$: necesitaría cinco semanas, y las ondas cortas las
amortigua la \omterm{def:b2:viscous-flows:viscosity}{viscosidad} y las rompe el viento mucho antes.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:2}
$\omega = \qty{1.26e8}{rad/s}$, $\omega_c = \qty{6.3e7}{rad/s}$: $k = \sqrt{\omega^2 - \omega_c^2}/c =
\qty{0.36}{rad/m}$, $v_\varphi = \qty{3.5e8}{m/s}$, $v_g = c^2k/\omega = \qty{2.6e8}{m/s}$,
con producto $c^2$. A \qty{5}{MHz}, $\omega < \omega_c$: $\kappa = \sqrt{\omega_c^2 - \omega^2}/c =
\qty{0.18}{m^{-1}}$, profundidad \qty{5.5}{m}.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:3}
Batido a \qty{4}{Hz}, periodo de la sonoridad \qty{0.25}{s}. Agua: $f = 0.125$ y
\qty{0.119}{Hz}; envolvente $\cos(\delta\omega t - \delta kx)$ con $\delta k = \qty{2.9e-3}{rad/m}$
(batidos separados \qty{1.1}{km}) y celeridad $\delta\omega/\delta k = \qty{6.4}{m/s}$ --- la
\omterm{prop:b2:dispersion-wave-packets:group}{velocidad de grupo} a \qty{105}{m}, $\tfrac12\sqrt{g\lambda/2\pi} = \qty{6.4}{m/s}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:4}
$\ln(b/a) = 1.20$: $\Lambda = \qty{0.24}{\micro H/m}$, $\Gamma = \qty{106}{pF/m}$, $c = 1/
\sqrt{\Lambda\Gamma} = \qty{2.0e8}{m/s}$, $Z_c = \qty{48}{\ohm}$, \qty{5.0}{ns/m}; \qty{10}{ns}
$\leftrightarrow$ \qty{2.0}{m}. Con aire: $\Gamma = \qty{46}{pF/m}$, $c = c_0$, $Z_c = \qty{72}{\ohm}$.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:5}
(a) $\omega^2 = c^2k^2 + (EI/\mu)k^4$; $v_\varphi = \sqrt{c^2 + (EI/\mu)k^2}$; $v_g = (c^2k +
2(EI/\mu)k^3)/\omega$. (b) $c^2 = \qty{1.12e5}{m^2/s^2}$, $k = \qty{66}{rad/m}$, $(EI/\mu)k^2
= \qty{7.0e3}{m^2/s^2}$: $v_\varphi/c = \sqrt{1.063} = 1.031$, $+3.1\%$ --- los $53$ cents de
\cref{ch:b2:waves-on-strings}. (c) $v_g > v_\varphi$: anómala; los armónicos
agudos se adelantan y el ataque de un pulsado se convierte en un breve
gorjeo descendente.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:6}
(a) $-\mu\omega^2 + \iu\alpha\omega = -Tk^2$: $\underline k^2 = (\omega^2/c^2)(1 - \iu\alpha/\mu\omega)$.
(b) $\underline k \approx (\omega/c)(1 - \iu\alpha/2\mu\omega)$: $k' = \omega/c$, $k'' = \alpha/2\mu c =
\alpha/2Z$; $8.7k''$ dB/m. (c) $c = \qty{100}{m/s}$, $Z = \qty{0.5}{kg/s}$, $\mu\omega = 3.1 \gg
0.02$: $k'' = \qty{0.02}{m^{-1}}$, $\delta = \qty{50}{m}$, \qty{0.17}{dB/m}. (d) No en este
límite ($k' = \omega/c$); cuando $\alpha \gtrsim \mu\omega$ la raíz ya no se
linealiza y $k'$ depende de $\omega$ de forma no lineal: dispersiva (el
límite difusivo del cable).
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:7}
(a) $v_\varphi = \sqrt{\gamma k/\rho}$, $v_g = \tfrac32v_\varphi$: el grupo adelanta a
las crestas. (b) $v_\varphi^2 = g/k + \gamma k/\rho$ es mínimo en $k_m = \sqrt{\rho g/\gamma}$:
$\lambda_m = 2\pi\sqrt{\gamma/\rho g} = \qty{1.7}{cm}$, $v_{\min} = (4g\gamma/\rho)^{1/4} =
\qty{23}{cm/s}$. (c) Una estela es un dibujo estacionario respecto de la
barca, formado por ondas cuya \omterm{def:b2:dispersion-wave-packets:relation}{velocidad de fase} iguala a su celeridad;
ninguna es más lenta que \qty{23}{cm/s}. (d) Régimen capilar, $v_g > v_\varphi$: las crestas
nacen en el frente del anillo y mueren en su parte trasera.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:8}
(a) $k = \qty{0.126}{rad/m}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-18}{m^2/s}$: $t_{\text{ens}}
= 2.5 \times 10^5/18 = \qty{1.4e4}{s}$, cuatro horas. (b) \qty{10}{ps}: $t_{\text{ens}} =
4 \times 10^{-6}/0.16 = \qty{25}{\micro s}$, \qty{5}{km} a \qty{2e8}{m/s}; \qty{1}{ns}: $0.04/0.16 =
\qty{0.25}{s}$, \qty{50000}{km}. Los pulsos cortos (los caudales binarios altos) se
ensanchan en unos kilómetros: los enlaces largos usan fibra compensadora o ritmos moderados.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:9}
(a) $\partial_x^2v = R\Gamma\,\partial_tv$, $D = 1/R\Gamma = \qty{1.1e12}{m^2/s}$. (b) $\underline k^2
= -\iu R\Gamma\omega$: $k' = k'' = \sqrt{R\Gamma\omega/2}$; a \qty{1}{Hz}: $k = \qty{1.7e-6}{m^{-1}}$,
$v_\varphi = \qty{3.7e6}{m/s}$, $\delta = \qty{600}{km}$; a \qty{10}{Hz}: $v_\varphi = \qty{1.2e7}{m/s}$,
$\delta = \qty{190}{km}$. (c) $R\Gamma L^2 = 9 \times 10^{-13} \times 9 \times 10^{12} = \qty{8}{s}$
por punto: alrededor de una palabra por minuto. (d) $\Lambda = R\Gamma/G = \qty{9}{H/km}$ --- una
inductancia absurda para un cable submarino; las bobinas de carga sí
servían en las líneas telefónicas terrestres, cuyos $G$ y $R\Gamma$ eran distintos.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:10}
(a) $\partial_te = \dot y(\mu\ddot y + Ky) + Ty'\dot y' = \dot yTy'' + Ty'\dot y' = \partial_x
(Ty'\dot y) = -\partial_x\mathcal P$. (b) $\langle e\rangle = \tfrac14A^2(\mu\omega^2 + Tk^2 + K) =
\tfrac12\mu\omega^2A^2$ (usando $\mu\omega^2 = Tk^2 + K$); $\langle\mathcal P\rangle = \tfrac12Tk\omega
A^2$. (c) Cociente $Tk/\mu\omega = c^2k/\omega = v_g$. (d) $y = A\eu^{-\kappa x}\cos\omega t$:
$\mathcal P \propto \sin2\omega t$, de media nula --- ninguna energía atraviesa el medio.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:11}
(a) $\tanh kh \to 1$: $\omega^2 = gk$; si $kh \ll 1$: $\omega = \sqrt{gh}\,k$, celeridad $\sqrt{gh}$,
no dispersiva. (b) $kh = 0.13$: aguas someras; $c = \sqrt{9.81 \times 4000} = \qty{198}{m/s}$,
$T = \qty{17}{min}$; \qty{8000}{km} en \qty{11}{h}. (c) $\tfrac12\rho gA^2 = \qty{1.2}{kJ/m^2}$
sobre $200 \times 1000$ km$^2$: \qty{2.5e14}{J}. Green: $A \propto h^{-1/4}$, $(4000/10)^{1/4}
= 4.5$: \qty{2.2}{m}, antes del empinamiento y del remonte. (d) A $h \approx \lambda/2 =
\qty{50}{m}$; la parte de la cresta que está en aguas menos profundas se
frena, de modo que la cresta gira hasta ponerse paralela a las curvas de
nivel del fondo --- refracción.
\end{solution}

\begin{solution}{exo:b2:dispersion-wave-packets:12}
(a) En $x = \ell$: $v = v_i + v_r$, $i = (v_i - v_r)/Z_c$, $v = R_Li$: $r = (R_L -
Z_c)/(R_L + Z_c)$. (b) $0$, $-1$, $+1$, $0.5$. (c) La entrada de la línea se ve como
$Z_c$, de modo que el osciloscopio ve \qty{0.5}{V} en $t = 0$ y luego el eco de $0.5r$ V en
$2\ell/c = \qty{1}{\micro s}$: nada, \qty{-0.5}{V}, \qty{+0.5}{V}, \qty{+0.25}{V}; el
generador adaptado absorbe el eco, así que no hay un segundo. (d)
$\tfrac12 \times 2 \times 10^8 \times 1.8 \times 10^{-6} = \qty{180}{m}$.
\end{solution}

\begin{solution}{pb:b2:dispersion-wave-packets:1}
\textbf{1.} $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi = -\Gamma\partial_tv$; con
$\eu^{\iu(\omega t - kx)}$: $\iu kv = (\iu\omega\Lambda + R)i$, $\iu ki = \iu\omega\Gamma v$, luego
$k^2 = \omega\Gamma(\omega\Lambda - \iu R)$.

\textbf{2.} $\omega = R/\Lambda = \qty{4000}{rad/s}$, \qty{640}{Hz}: a unos pocos hercios $\Lambda\omega
\ll R$, régimen difusivo.

\textbf{3.} $k' = k'' = \sqrt{R\Gamma\omega/2}$ con $R\Gamma = \qty{5e-13}{s/m^2}$; a \qty{1}{Hz}:
$\qty{1.25e-6}{m^{-1}}$, $v_\varphi = \qty{5e6}{m/s}$, $\delta = \qty{800}{km}$; a \qty{4}{Hz}:
$\qty{2.5e-6}{m^{-1}}$, $\qty{1e7}{m/s}$, \qty{400}{km}.

\textbf{4.} $8.7L/\delta$: \qty{38}{dB} a \qty{1}{Hz} y \qty{76}{dB} a \qty{4}{Hz}: solo
sobreviven las componentes más lentas; un punto seco llega como una joroba
larga y baja.

\textbf{5.} $R\Gamma L^2 = 5 \times 10^{-13} \times 1.2 \times 10^{13} = \qty{6}{s}$: un punto cada
seis segundos, una palabra por minuto (el cable real, con galvanómetros de
espejo sensibles, llegaba a unas pocas más).

\textbf{6.} $\Lambda = R/\omega = \qty{80}{mH/km}$. Entonces $k^2 = \Gamma R\omega(1 - \iu)$, $k = \sqrt{\Gamma R\omega}
\,2^{1/4}\eu^{-\iu\pi/8}$: $k' = \qty{3.9e-6}{m^{-1}}$, $k'' = \qty{1.6e-6}{m^{-1}}$: $v_\varphi =
\qty{6.4e6}{m/s}$, $\delta = \qty{620}{km}$, \qty{49}{dB} en todo el cable en lugar de
\qty{76}{dB}.

\textbf{7.} Un punto cada \qty{6}{s} son unos \qty{0.2}{bit/s}: la fibra es
de $10^{10}$ a $10^{11}$ veces más rápida.

\textbf{8.} $v_\varphi = gT/2\pi$, $v_g = gT/4\pi$.

\textbf{9.} \qty{18}{s}: $v_g = \qty{14.1}{m/s}$, $4 \times 10^6/14.1 = \qty{3.3}{d}$;
\qty{6}{s}: \qty{4.7}{m/s}, \qty{9.9}{d}: el episodio de mar de fondo dura más de seis días.

\textbf{10.} $t = D/v_g = 4\pi D/gT$, luego $1/T = gt/4\pi D$. $\Delta(1/T) = 1/12 - 1/16
= \qty{0.0208}{Hz}$ en \qty{86400}{s}: $g/4\pi D = \qty{2.4e-7}{s^{-2}}$, $D = \qty{3200}{km}$.

\textbf{11.} $\tfrac12\rho gA^2 = \qty{11}{kJ/m^2}$; la energía se mueve a $v_g = gT/4\pi =
\qty{9.4}{m/s}$: \qty{103}{kW} por metro de cresta.

\textbf{12.} $10^3 \times 103 \times 0.3 = \qty{31}{MW}$: diez aerogeneradores.

\textbf{13.} $\lambda = 2\pi U^2/g = \qty{16}{m}$. La energía de cada componente
viaja a la mitad de su celeridad de fase, de modo que el dibujo queda
confinado en una cuña tras la barca (de semiángulo \qty{19.5}{\degree}, sea
cual sea su celeridad).

\textbf{14.} Una cresta nace en la parte trasera del grupo, lo adelanta
(las crestas se mueven a $2v_g$ respecto del agua y a $v_g$ respecto del
grupo) y muere en el frente. El grupo mide $120 \times 9.4 = \qty{1.1}{km}$
de largo, cinco longitudes de onda ($\lambda = gT^2/2\pi = \qty{225}{m}$); pasan diez crestas por
la boya, cada una tras vivir unos dos minutos.

\textbf{15.} $kh = 0.13$: aguas someras. $c = \qty{198}{m/s}$, $T = \lambda/c = \qty{17}{min}$,
\qty{8.4}{h}.

\textbf{16.} No dispersiva: todas las componentes a \qty{198}{m/s}, que
llegan juntas como un único pulso (el mar de fondo, a $\sim\qty{10}{m/s}$ y
clasificado por periodos, tardaría una semana).

\textbf{17.} $\tfrac12\rho gA^2 = \qty{1.8}{kJ/m^2}$ sobre $500 \times 200$ km$^2$: \qty{1.8e14}{J},
el $4\%$ de una megatonelada.

\textbf{18.} $A = 0.6(4000/20)^{1/4} = \qty{2.3}{m}$; $c = \sqrt{gh} = \qty{14}{m/s}$, $\lambda =
cT = \qty{14}{km}$. La cresta, en aguas más profundas que el valle que
tiene delante, viaja más deprisa y lo alcanza: el frente se empina hasta
formar un muro.

\textbf{19.} Una subida de \qty{0.6}{m} a lo largo de un cuarto de hora,
con una pendiente de $10^{-5}$: nada perceptible.

\textbf{20.} La celeridad media, $\sqrt{gh}$ para la profundidad media;
como $\sqrt h$ es cóncava, los tramos someros frenan la onda más de lo que
la aceleran los profundos, y el promedio a lo largo del camino queda por
debajo de $\sqrt{g\bar h}$.

\textbf{21.} $\omega' = \tfrac12\sqrt{g/k}$, $\omega'' = -\tfrac14\sqrt{g/k^3}$; $k = \qty{0.063}{rad/m}$:
$\omega'' = \qty{-50}{m^2/s}$.

\textbf{22.} $t_{\text{ens}} = 10^6/50 = \qty{2e4}{s}$, \qty{5.6}{h}; el paquete
ha recorrido $v_gt = 6.25 \times 2 \times 10^4 = \qty{125}{km}$.

\textbf{23.} Un tren largo contiene una banda estrecha de $k$, cuyas
\omterm{prop:b2:dispersion-wave-packets:group}{velocidades de grupo} casi coinciden: las componentes siguen juntas más
tiempo. El medio del tsunami es no dispersivo: celeridades iguales, sin
ensanchamiento.

\textbf{24.} $\omega'' = \hbar/m = \qty{1.2e-4}{m^2/s}$: $t_{\text{ens}} = 10^{-18}/1.2 \times
10^{-4} = \qty{9e-15}{s}$; $v_g = \hbar k/m = \qty{7.3e6}{m/s}$: \qty{60}{nm}.

\textbf{25.} Cable: $k^2 = -\iu R\Gamma\omega$, $v_\varphi = \sqrt{2\omega/R\Gamma}$, sin
$v_g$ limpia, atenúa y difumina. Mar de fondo: $\omega^2 = gk$, $v_\varphi = 2v_g$,
se ensancha. Tsunami: $\omega = \sqrt{gh}\,k$, $v_\varphi = v_g$, ninguna de las
dos cosas. Electrón: $\omega = \hbar k^2/2m$, $v_g = 2v_\varphi$, se ensancha.
\end{solution}
