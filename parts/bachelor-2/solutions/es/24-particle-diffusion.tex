\begin{solution}{exo:b2:particle-diffusion:4}

\begin{solution}{exo:b2:particle-diffusion:5}

\begin{solution}{exo:b2:particle-diffusion:6}
(a) $n = n_{\text{s}}a/r$, $\Phi = 4\pi D a n_{\text{s}} = \qty{3.6e-8}{mol/s}$.
(b) $1.95 \times 10^{-5}\,\unit{mol}$; \qty{540}{s}. (c) La capa en $1/r$ tarda
$\sim a^2/D \approx \qty{2000}{s}$ en formarse y el grano encoge; remover
sustituye la capa de espesor $\sim a$ por una \omterm{prop:b2:viscous-flows:bl}{capa límite} $\delta \ll a$
y multiplica el flujo por $a/\delta$. (d) $(rn')' = 0$: $n = A + B\ln r$,
que no puede anularse en el infinito: no hay estado estacionario --- la nube de un
cilindro sigue creciendo (logarítmicamente).
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:7}
(a) $\dd/\dd t\int n = D[n']_{-\infty}^{\infty} = 0$; $\dd/\dd t\int x^2 n = D\int
x^2 n'' = 2D\int n = 2DN$. (b) $\partial_t[n(x,-t)] = -D\,\partial_x^2 n$: el
signo se invierte. (c) Para $t < 0$ la ``anchura'' $2Dt$ es negativa: no existe tal
estado --- el pulso no puede desextenderse. (d) $\dd/\dd t\int n^2 = 2D\int nn''
= -2D\int n'^2 \le 0$.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:11}
(a) $n = n_\infty(1 - a/r)$, $\Phi = 4\pi D a n_\infty$. (b) $3.8 \times
10^6$ por segundo. (c) Los receptores actúan como conductancias en paralelo y
luego en serie con la capa esférica: $\Phi \approx 4\pi Dan_\infty
\cdot N_{\text{r}}s/(N_{\text{r}}s + \pi a)$; la mitad del máximo para
$N_{\text{r}}s = \pi a$: con $s = \qty{1}{nm}$, $N_{\text{r}} \approx
3000$, que cubren $N_{\text{r}}s^2/4a^2 \approx 10^{-3}$ de la superficie.
(d) Cada tipo de receptor necesita un área despreciable para una captura casi
máxima: una célula puede vigilar miles de sustancias a la vez.
\end{solution}

\begin{solution}{exo:b2:particle-diffusion:12}
(a) Diferencia hacia delante en $t$ y diferencia segunda centrada en $x$. (b)
$\alpha = 1/2$: $n_i^{k+1} = (n_{i-1}^k + n_{i+1}^k)/2$ --- cada partícula salta
a izquierda o a derecha con probabilidad $1/2$. (c) Para $n_i = (-1)^i$, $n^{k+1} =
(1 - 4\alpha)n^k$: $|1 - 4\alpha| > 1$ cuando $\alpha > 1/2$. (d) $\Delta t \le
\Delta x^2/2D = \qty{3.3}{s}$; unos $1100$ pasos.
\end{solution}

\begin{solution}{pb:b2:particle-diffusion:1}
\textbf{1.} $k_BT = \qty{0.118}{eV}$, $E_{\text{a}}/k_BT = 29.2$: $D =
\qty{1.6e-17}{m^2/s}$; a \qty{1000}{\celsius}: \qty{1.6e-18}{m^2/s} --- un
factor $10$.

\textbf{2.} La superficie reflectante se trata con la imagen especular del
pulso, que duplica la amplitud en $x > 0$; $\int_0^\infty = (Q/\sqrt{\pi Dt})
\cdot \tfrac12\sqrt{4\pi Dt} = Q$.

\textbf{3.} $Dt = \qty{5.8e-14}{m^2}$; $\sqrt{\pi Dt} = \qty{0.43}{\micro m}$; $n_{\text{s}}
= \qty{2.3e24}{m^{-3}}$.

\textbf{4.} $\eu^{-x^2/4Dt} = 4.3 \times 10^{-4}$: $x_j^2 = 4Dt \times 7.75$, $x_j =
\qty{1.3}{\micro m}$.

\textbf{5.} $n_{\text{s}}$ se reduce a la mitad, el logaritmo baja a $7.05$ y $4Dt$
se cuadruplica: $x_j = \qty{2.6}{\micro m}$ --- menos del doble.

\textbf{6.} $\delta D/D = (E_{\text{a}}/k_BT)\,\delta T/T = 21\%$; $x_j \propto
\sqrt{Dt}$ aproximadamente, así que un $1\%$ en $x_j$ exige un $2\%$ en $D$: \qty{1}{K}.

\textbf{7.} $E_{\text{a}}/k_BT = 134$: $D \approx 10^{-62}\,\unit{m^2/s}$; una
distancia atómica (\qty{0.25}{nm}) en $a^2/D \sim 10^{43}\,\unit{s}$: congelado.

\textbf{8.} $E_{\text{a}}/k_BT = 32.8$: $D = \qty{4.3e-19}{m^2/s}$; $\sqrt{Dt} =
\qty{28}{nm}$.

\textbf{9.} $n_0\operatorname{erfc}(x/2\sqrt{Dt})$; $n_B/n_0 = 5 \times 10^{-6}$:
$x \approx 6.4\sqrt{Dt} = \qty{0.18}{\micro m}$.

\textbf{10.} $2n_0\sqrt{Dt/\pi} = \qty{6.3e18}{m^{-2}}$: seis veces $Q$ ---
la dosis crece como $\sqrt t$ y la fija el tiempo.

\textbf{11.} Temperatura baja: $D$ pequeño, dosis controlada por el tiempo,
poco profunda; temperatura alta: la dosis fija se lleva deprisa a lo hondo.

\textbf{12.} \qty{0.18}{\micro m} frente a \qty{1.3}{\micro m}: fina.

\textbf{13.} $4 \times 10^{18}$ moléculas; $n_{\text{a}} = \qty{2.4e25}{m^{-3}}$.

\textbf{14.} $\ell = \qty{37}{nm}$; $v^* = \qty{206}{m/s}$.

\textbf{15.} $D \approx \qty{2.5e-6}{m^2/s}$.

\textbf{16.} $L^2/D = 10^7\,\unit{s}$, meses; con deriva: \qty{50}{s}.

\textbf{17.} $n = 8N(4\pi Dt)^{-3/2}\eu^{-r^2/4Dt}$; máxima en $t^* = r^2/6D =
\qty{6.7e4}{s}$, dieciocho horas.

\textbf{18.} $n_{\max} = 8N(4\pi Dt^*)^{-3/2}\eu^{-3/2} \approx \qty{2e18}{m^{-3}}$,
muy por encima del umbral: se huele, al final.

\textbf{19.} $r^2 = 4Dt\ln 10^6 = \qty{12}{m^2}$: \qty{3.5}{m}.

\textbf{20.} $v^*t/\ell \approx 5 \times 10^{14}$.

\textbf{21.} $\langle x\rangle = 0$, $\langle x^2\rangle = N\ell^2 = \ell^2 t/\tau$;
$D = \ell^2/2\tau$.

\textbf{22.} $\binom{N}{N/2}/2^N$: $1/2$, $3/8$, $5/16$ --- decreciente
(como $1/\sqrt{\pi N/2}$): el caminante se aleja como $\sqrt N$.

\textbf{23.} La nube se reúne en un punto --- jamás visto; viola
la \omterm{thm:b2:particle-diffusion:equation}{ecuación de difusión}, no las leyes de la mecánica.

\textbf{24.} Cada colisión es reversible, pero el estado inicial (todas las
partículas juntas) es excepcional: casi toda historia microscópica
que parta de él se extiende, y la inversa exige una conspiración de todas las
velocidades. El promedio del camino aleatorio conserva solo lo típico y
descarta esa información; la irreversibilidad es estadística.

\textbf{25.} $L \sim \sqrt{Dt}$, $t \sim L^2/D$; $D \sim \ell v^*/3$, es decir,
el paso entre colisiones: $10^{-5}$, $10^{-9}$, $10^{-17}\,\unit{m^2/s}$.
\end{solution}
