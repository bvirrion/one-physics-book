\begin{solution}{pb:b2:open-systems:1}
\textbf{1.} Compresor (adiabático, $w = \Delta h$), condensador (isobárico, calor
cedido), \omterm{prop:b2:open-systems:devices}{estrangulamiento} (isentálpico), evaporador (isobárico, calor absorbido).

\textbf{2.} $h_4 = \qty{256}{kJ/kg}$, $x_4 = 0.28$.

\textbf{3.} \qty{143}{kJ/kg}.

\textbf{4.} \qty{27}{} y \qty{34}{kJ/kg}; $h_2 = 433$; $T_2 \approx 45 + 7 \approx
\qty{52}{\celsius}$.

\textbf{5.} \qty{177}{kJ/kg}; $\mathrm{COP}_{\text{cal}} = 5.2$, $\mathrm{COP}_{\text{fr}} = 4.2$;
$5.2 = 4.2 + 1$.

\textbf{6.} $7.8$ y $6.8$; dos tercios.

\textbf{7.} \qty{0.045}{kg/s}; \qty{1.5}{kW} y \qty{1.7}{kW} en el enchufe.

\textbf{8.} Compresor: $c_p\ln(325/318) \approx \qty{0.022}{kJ/kg/K}$;
\omterm{prop:b2:open-systems:devices}{estrangulamiento}: $s_4 - s_3 = 1.204 - 1.19 = \qty{0.014}{kJ/kg/K}$ --- el compresor.

\textbf{9.} $313/55 = 5.7$; la relación de presiones casi se dobla (más trabajo
por kilogramo) y la densidad del vapor se reduce a la mitad (menos masa por embolada):
la capacidad cae justo cuando la casa necesita más.

\textbf{10.} La batería está por debajo de \qty{0}{\celsius} en aire húmedo; la escarcha
la aísla; el desescarche invierte el ciclo unos minutos cada hora y
cuesta un pequeño porcentaje.

\textbf{11.} $308/35 = 8.8$ frente a $328/55 = 6.0$: cuanto menor es la
temperatura del condensador, mejor --- el suelo radiante.

\textbf{12.} \qty{1}{kWh} de combustible $\to$ \qty{0.4}{kWh} de electricidad $\to$
\qty{1.6}{kWh} de calor, frente a \qty{0.9}{kWh} de la caldera.

\textbf{13.} Bomba, caldera (calentamiento isobárico, vaporización, sobrecalentamiento),
turbina y condensador.

\textbf{14.} \qty{6}{kJ/kg}; $h_2 = 144$.

\textbf{15.} $3278 = 1069 + 1571 + 638$ (\unit{kJ/kg}).

\textbf{16.} $x = 0.81$, $h_{4s} = 2098$, $w_{\text{s}} = \qty{1324}{kJ/kg}$.

\textbf{18.} $\eta = 1119/3278 = 0.34$; Carnot $0.60$; $\bar T = 3278/6.40 =

\textbf{19.} \qty{536}{kg/s}; \qty{1160}{MW}; \qty{28}{m^3/s} de agua de río,

\textbf{21.} El segundo aporte de calor es a temperatura alta (mayor

\textbf{22.} $1757\,\unit{MW} \times (1/512 - 1/1500) = \qty{2.3}{MW/K}$ --- seis veces

álabes; el condensador viene el último.

\textbf{25.} Dibújese el \omterm{def:b2:open-systems:cv}{volumen de control}; la masa que entra iguala a la que sale; $\Delta h +
\Delta c^2/2 + g\Delta z = w_{\text{u}} + q$ con $h$ a partir de $c_p$, de tablas o de
$v\Delta P$; $\Delta s = \sum q/T + s_{\text{c}}$ con referencias isentrópicas y
rendimientos; móntense los componentes y compárese con Carnot.
\end{solution}
