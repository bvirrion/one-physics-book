\begin{solution}{exo:b2:waves-in-media:1}

\begin{solution}{exo:b2:waves-in-media:4}

\begin{solution}{exo:b2:waves-in-media:5}
(a) $1/\gamma\pi a^2 = \qty{5.3}{m\ohm/m}$. (b) $\delta = \qty{65}{\micro m}$; $R \approx 1/\gamma2\pi a\delta
= \qty{41}{m\ohm/m}$, $7.7$ veces la de continua. (c) $\delta = \qty{6.5}{\micro m}$, \qty{0.41}{\ohm/m},
$77$ veces. (d) $R \propto \sqrt f$ crece tan deprisa como $L\omega$ en cuanto domina
el \omterm{thm:b2:waves-in-media:skin}{efecto pelicular}, y cada vuelta añade resistencia; el hilo de Litz agrupa muchos
hilos aislados más finos que $\delta$, de modo que conduce todo el cobre.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:6}

\begin{solution}{exo:b2:waves-in-media:7}
(a) $v_0 = eE_0/m\omega$; $\tfrac12nm\langle v^2\rangle = ne^2E_0^2/4m\omega^2 = \tfrac14\varepsilon_0
E_0^2\,\omega_p^2/\omega^2$. (b) $\tfrac14\varepsilon_0E_0^2$ y $\tfrac14\varepsilon_0E_0^2(1 - \omega_p^2/
\omega^2)$. (c) La suma vale $\tfrac12\varepsilon_0E_0^2$ (el paréntesis es $1$); $\langle\Pi
\rangle = E_0^2k/2\mu_0\omega$, y el cociente es $c^2k/\omega = v_g$. (d) $\tfrac14\cdot\tfrac14
/\tfrac12 = 1/8$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:8}
(a) $1/v_g = (1/c)(1 - \omega_p^2/\omega^2)^{-1/2} \approx (1 + \omega_p^2/2\omega^2)/c$. (b) $\Delta t
= (L\omega_p^2/2c)(1/\omega_1^2 - 1/\omega_2^2)$ con $\omega_p^2 = ne^2/\varepsilon_0m$ y $\omega =
2\pi f$: constante $e^2/8\pi^2\varepsilon_0mc = \qty{1.34e-7}{}$ (SI). (c) $nL = \qty{9.3e23}{m^{-2}}$,
$(1/f_1^2 - 1/f_2^2) = \qty{5.7e-18}{s^2}$: $\Delta t = \qty{0.72}{s}$. (d) La columna $nL$
(la medida de dispersión) y, con ella, la distancia para un modelo de $n$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:9}
(a) $\operatorname{Im}\underline\chi = \omega_p^2\Gamma\omega/[(\omega_0^2 - \omega^2)^2 + \Gamma^2\omega^2]$ con
$\omega_0^2 - \omega^2 \approx 2\omega_0(\omega_0 - \omega)$: $n'' = \tfrac12\operatorname{Im}\underline\chi$ es
la lorentziana enunciada. (b) $n''(\omega_0) = \omega_p^2/2\omega_0\Gamma$, $\alpha = 2n''\omega_0/c =
\omega_p^2/\Gamma c$. (c) $\omega_p^2 = \qty{3.2e20}{s^{-2}}$: $\alpha = \qty{1.8e4}{m^{-1}}$;
$\ln10/\alpha = \qty{0.13}{mm}$. (d) $n' - 1 \propto (\omega_0 - \omega)/[(\omega_0 - \omega)^2 +
\Gamma^2/4]$: positiva por debajo, negativa por encima y decreciente dentro de $\pm\Gamma/2$
en torno a la raya --- anómala ahí.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:10}
(a) $\underline E = E_0\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$, $\underline B = \underline k\underline E/\omega
= (1 - \iu)\underline E/\omega\delta$. (b) $\langle\Pi_x\rangle = \tfrac12\operatorname{Re}(\underline E
\underline B^*)/\mu_0 = E_0^2/2\mu_0\omega\delta$ en $x = 0$. (c) $\int_0^\infty\tfrac12\gamma E_0^2
\eu^{-2x/\delta}\dd x = \gamma E_0^2\delta/4 = E_0^2/2\mu_0\omega\delta$ usando $\gamma = 2/\mu_0\omega\delta^2$.
(d) $H_0 = \sqrt2E_0/\mu_0\omega\delta$, luego $\tfrac12R_sH_0^2 = E_0^2/\gamma\mu_0^2\omega^2\delta^3 =
E_0^2/2\mu_0\omega\delta$. Cobre a \qty{1}{GHz}: $R_s = 1/\gamma\delta = \qty{8}{m\ohm}$;
$\tfrac12 \times 8 \times 10^{-3} \times 10^4 = \qty{40}{W/m^2}$.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:11}
(a) $\omega_p = \qty{1.4e16}{rad/s}$, $\lambda_p = 2\pi c/\omega_p = \qty{140}{nm}$. (b) $\omega = \qty{3.8e15}{rad/s}
< \omega_p$: $1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{23}{nm}$, reflexión total. (c)
Sigue por debajo de $\omega_p$: refleja; la transparencia solo llega más allá de \qty{140}{nm} (en
el sodio, \qty{210}{nm}). (d) Las colisiones dan una parte real a $\underline\gamma$ y
un pequeño porcentaje de absorción; las resonancias de electrones ligados en el visible
(oro, cobre) absorben el azul y colorean el metal.
\end{solution}

\begin{solution}{exo:b2:waves-in-media:12}
(a) $\omega\tau = 0.12$: $\underline\varepsilon_r = 84 - 9.7\iu$; a \qty{20}{GHz}, $\omega\tau = 1$:
$45 - 40\iu$. (b) $\underline n \approx 9.2 - 0.53\iu$; $c/2n''\omega = \qty{1.8}{cm}$: el horno
cocina los dos centímetros exteriores y la conducción hace el resto. (c) $\tfrac12 \times

\end{solution}

\begin{solution}{pb:b2:waves-in-media:1}
\textbf{1.} $f_p = \qty{9}{MHz}$: el GPS pasa, la onda corta se refleja.

\textbf{2.} $k = (\omega/c)\sqrt{1 - \omega_p^2/\omega^2}$: $v_\varphi = \omega/k \approx c(1 + \omega_p^2/
2\omega^2)$, $v_g = \dd\omega/\dd k = c^2k/\omega \approx c(1 - \omega_p^2/2\omega^2)$.

\textbf{3.} $\Delta t = \int(1/v_g - 1/c)\dd z = \int\omega_p^2\dd z/2c\omega^2 = e^2N_T/2\varepsilon_0
mc\omega^2 = (e^2/8\pi^2\varepsilon_0mc)N_T/f^2 = 1.34 \times 10^{-7} \times 2 \times 10^{17}/2.48 \times
10^{18} = \qty{11}{ns}$: \qty{3.2}{m}.

\textbf{4.} $N_T = (\Delta t_1 - \Delta t_2)/K(1/f_1^2 - 1/f_2^2)$; la diferencia vale
\qty{7}{ns}: el $1\%$ exige \qty{0.07}{ns}.

\textbf{5.} $v_\varphi$ supera a $c$ tanto como $v_g$ se queda corta: la
fase de la portadora llega pronto mientras que la modulación llega tarde. Un
receptor de fase de portadora aplica la corrección con el signo opuesto.

\textbf{6.} \qty{3.2}{m} de día, \qty{0.3}{m} de noche.

\textbf{7.} $f_p = \qty{0.9}{MHz}$ con colisiones: las ondas medias y cortas se
absorben (un apagón de radio); el GPS a \qty{1.5}{GHz} sufre una absorción
$\propto 1/\omega^2$, despreciable.

\textbf{8.} $f_1\Delta t = 1.575 \times 10^9 \times 1.1 \times 10^{-8} = 17$ ciclos.

\textbf{9.} $\delta = \sqrt{2/\mu_0\gamma\omega} = \qty{10}{\micro m}$: una pared de un milímetro es
cien \omterm{def:b2:dispersion-wave-packets:complex}{profundidades de penetración}.

\textbf{10.} $R_s = 1/\gamma\delta = \qty{0.1}{\ohm}$; $\tfrac12R_sH_0^2 = \qty{45}{W/m^2}$: \qty{45}{W},
el $6\%$.

\textbf{11.} $\eu^{-\pi \times 0.5} = 0.21$: \qty{-14}{dB} en amplitud y \qty{-27}{dB} en
potencia (y además el agujerito acopla poco de entrada).

\textbf{12.} La luz, con $\lambda = \qty{0.5}{\micro m}$, es dos mil veces menor
que los agujeros y pasa; la microonda, con $\lambda = \qty{12}{cm}$, es cien
veces mayor.

\textbf{13.} $\underline n = 2 - 0.005\iu$; $c/2n''\omega = \qty{1.9}{m}$: el vidrio apenas
se calienta.

\textbf{14.} $j \approx H_0/\delta = \qty{3e7}{A/m^2}$, $j^2/\gamma = \qty{9e8}{W/m^3}$ en una
capa de diez micrómetros: las puntas se calientan en milisegundos, emiten electrones,
ionizan el aire --- chispas.

\textbf{15.} $\delta = \qty{18}{\micro m}$; sección $2\pi a\delta = \qty{5.6e-8}{m^2}$ frente a
$\pi a^2 = \qty{7.9e-7}{m^2}$: $R_{\text{ca}} = \qty{0.30}{\ohm/m}$, catorce veces la de continua.

\textbf{16.} $RI^2 = \qty{120}{W/m}$; superficie $\pi d = \qty{3.1e-3}{m^2}$ por metro:
$\Delta T \approx \qty{1900}{K}$ --- se fundiría; esas bobinas son tubos refrigerados por agua.

\textbf{17.} $\delta = \qty{0.14}{mm}$: la corriente inducida, y el calor, se quedan
en una décima de milímetro; un pulso corto seguido de un temple endurece
la superficie de un engranaje mientras el núcleo sigue tenaz.

\textbf{18.} $R = 1/\gamma2\pi a\delta = \qty{0.10}{\ohm/m}$, tres veces menos; el
interior no lleva nada --- de ahí el tubo, con el refrigerante donde el
cobre se desperdiciaría.

\textbf{19.} $\delta = a$ para $f = 1/\pi\mu_0\gamma a^2 = \qty{17}{Hz}$; por debajo, la corriente es
uniforme y vale la fórmula de continua.

\textbf{20.} $n^2 \approx 1 + \omega_p^2/\omega_0^2 + (\omega_p^2/\omega_0^2)(\lambda_0/\lambda)^2$: la forma de Cauchy
con $n(\infty)^2 - 1 = \omega_p^2/\omega_0^2 = 1.25$ para $n(\infty) = 1.5$.

\textbf{21.} $n = 1.531$ y $1.510$, $v = 1.96$ y \qty{1.99e8}{m/s}; $n_g(550) =
1.550$; $n_g(400) - n_g(700) = 0.063$: \qty{210}{ns} en un kilómetro.

\textbf{22.} $\sin((A + D)/2) = 1.5165 \times 0.5$: $D = \qty{38.6}{\degree}$; $\dd D = 2\sin
(A/2)\dd n/\cos((A + D)/2) = 1.53\,\dd n$: $\Delta D = \qty{1.85}{\degree}$, \qty{6.5}{cm} a \qty{2}{m}.

\textbf{23.} A \qty{200}{nm} la frecuencia se acerca a la resonancia y
$n''$ crece: opaco. Una resonancia débil a \qty{700}{nm} absorbe el rojo:
vidrio verde.

\textbf{24.} El azul está más cerca de la resonancia ultravioleta: los electrones
ligados responden más, $n$ es mayor y la desviación también.

\textbf{25.} Capa F a \qty{1.5}{GHz}: $\underline\gamma$ imaginaria, propagación
con un pequeño retardo de grupo. Acero a \qty{2.45}{GHz}: $\underline\gamma$ real, \omterm{thm:b2:waves-in-media:skin}{efecto
pelicular}, reflexión. Cobre a \qty{13.56}{MHz}: lo mismo, con una piel de
\qty{18}{\micro m}. Vidrio en el visible: $\iu\omega\varepsilon_0\chi$, un índice real, un
\omterm{def:b2:dispersion-wave-packets:relation}{medio dispersivo} transparente.
\end{solution}
