\section*{Chapter \ref{ch:b2:thermodynamic-potentials} --- Thermodynamic Potentials}

\begin{solution}{exo:b2:thermodynamic-potentials:1}
$F = -nRT\ln V + f(T)$, $G = nRT\ln P + g(T)$; $P = -\partial_VF = nRT/V$,
$V = \partial_PG = nRT/P$; $-\partial_TF = nR\ln V - f'(T)$, the entropy
$nC_V\ln T + nR\ln V + \text{const}$ if $f' = -nC_V\ln T + \dots$
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:2}
$\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$, $\Delta U = 0$, $\Delta S = \qty{19.1}{J/K}$.
Reversible: \qty{5.7}{kJ}, $S_{\text{c}} = 0$; against \qty{1}{bar}: \qty{0.9}{kJ},
$S_{\text{c}} = 19.1 - 900/300 = \qty{16}{J/K}$; vacuum: nothing, $S_{\text{c}} =
\qty{19.1}{J/K}$.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:3}
$\Delta g = \Delta h - T\Delta s$ with $\Delta h \approx L$, $\Delta s \approx L/T_{\text{fus}}$.
At \qty{-5}{\celsius}: $g_{\text{liq}} - g_{\text{ice}} = +\qty{6.1}{kJ/kg}$, ice
stable, liquid metastable (supercooling exists); at $+\qty{5}{\celsius}$:
\qty{-6.1}{kJ/kg}, liquid stable --- superheated ice is not observed, its
surface melts first.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:4}

\qty{-0.5}{K}; \qty{0.7}{bar}: \qty{92}{\celsius}; \qty{2}{bar}: $\approx
\qty{127}{\celsius}$ by the local slope (\qty{120}{\celsius} in fact: the
slope grows).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:7}

\begin{solution}{exo:b2:thermodynamic-potentials:8}
(a) Matter goes to the lower $\mu$: $\mu_{\text{v}} < \mu_{\text{liq}}$ when $p <
p_{\text{sat}}$; humidity $= p/p_{\text{sat}}$. (b) $-RT\ln 0.5 = \qty{1.7}{kJ/mol}$
in favour of the vapour. (c) Evaporation needs only $p < p_{\text{sat}}$,
not boiling; at $100\%$ the potentials are equal. (d) On the coldest
surfaces, whose $p_{\text{sat}}(T_{\text{s}})$ has dropped below the air's $p$
--- those radiating to the night sky.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:9}
(a) Standard. (b) $b = RT_{\text{c}}/8P_{\text{c}} = \qty{4.3e-5}{m^3/mol}$, $a =
27R^2T_{\text{c}}^2/64P_{\text{c}} = \qty{0.36}{Pa.m^6/mol^2}$. (c) $\dd g = v\,\dd P$
at fixed $T$ and $g$ equal at the two ends: $\int v\,\dd P = 0$ around the
loop. (d) Unstable where $\partial_vP > 0$; metastable between: the
superheated liquid (bumping) and the supersaturated vapour (fog that
waits for nuclei, cloud chamber).
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:10}
(a) $\dd P/\dd T = L_{\text{m}}P/RT^2$: $\ln(P/P_0) = -(L_{\text{m}}/R)(1/T - 1/T_0)$.
(b) \qty{28}{mbar} against \qty{23}{mbar} --- $L$ is larger at \qty{20}{\celsius}.
(c) \qty{820}{Pa} against \qty{611}{Pa}. (d) $\partial(\Delta g/T)/\partial T = -L/T^2$
and $\Delta g = 0$ along the curve give the same slope.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:11}
(a) Equal $\mu$ across the membrane: $\mu^\circ(P + \Pi) - RTx_{\text{s}} =
\mu^\circ(P)$, i.e. $v_{\text{m}}\Pi = RTx_{\text{s}}$, $\Pi = cRT$. (b) \qty{27}{bar}.
(c) $\Pi V = \qty{2.7}{MJ} = \qty{0.75}{kWh}$ per cubic metre; real plants
four times more. (d) Water flows to the lower potential: into the
solute-rich cell, out of it in brine.
\end{solution}

\begin{solution}{exo:b2:thermodynamic-potentials:12}
(a) $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$. (b) $k_BT/Na^2 \propto T$. (c)
\qty{1.7e-13}{N}; $1.8 \times 10^{14}$. (d) $E \sim nk_BT \approx \qty{0.4}{MPa}$.
\end{solution}

\begin{solution}{pb:b2:thermodynamic-potentials:1}
\textbf{1.} $\dd U = T\dd S + f\dd L$; $\dd F = -S\dd T + f\dd L$.

\textbf{2.} Schwarz on $\dd F$.

\textbf{3.} $f - T\partial_Tf = CT(L - L_0) - CT(L - L_0) = 0$: the energy
does not change; the pull is entropic.

\textbf{4.} $\Delta S = \qty{-5e-3}{J/K}$; $Q = \qty{-1.5}{J}$; $W = +\qty{1.5}{J}$; sum
zero.

\textbf{5.} $+\qty{0.4}{K}$.

\textbf{6.} $L - L_0 \propto 1/T$: \qty{9.1}{cm}, a contraction of \qty{9}{mm}; the
heated spokes shorten, the rim's centre of mass shifts, the wheel
turns.

\textbf{7.} $S = S_0 - k_BL^2/2Na^2$; $f = k_BTL/Na^2$; the entropy penalty is
paid in $k_BT$.

\textbf{8.} \qty{1.7e-13}{N}; $1.8 \times 10^{14}$ chains.

\textbf{9.} Its force is energetic ($\partial_LU \ne 0$); warming softens
the bonds, so $f$ falls with $T$.

\textbf{10.} Slopes $-s$, with $s_{\text{gas}} > s_{\text{liq}} > s_{\text{ice}}$; the
lowest line is the stable phase; crossings at $T_{\text{fus}}$ and $T_{\text{vap}}$.

\textbf{11.} \cref{thm:b2:thermodynamic-potentials:coexistence}.

\textbf{12.} $\qty{-1.3e7}{Pa/K}$; negative because $v_{\text{liq}} < v_{\text{ice}}$.

\textbf{13.} \qty{70}{bar}: \qty{-0.5}{K} --- not how one skates at
\qty{-10}{\celsius}; friction and the liquid-like surface layer do it.

\textbf{14.} \qty{270}{bar}, \qty{-2}{K}: with the geothermal flux, the
base melts and the sheet slides.

\textbf{15.} \qty{3.6}{kPa/K}; \qty{92}{\celsius}; about \qty{127}{\celsius}
(\qty{120}{} in fact).

\textbf{16.} \qty{28}{mbar} (\qty{23}{}), \qty{820}{Pa} (\qty{611}{}): $L$ grows as $T$
falls; errors of $20$--$30\%$.

\textbf{17.} Same $\Delta v$, so the ratio of slopes is $L_{\text{sub}}/L_{\text{vap}}
= 1.15$: the sublimation curve is steeper --- a kink at the triple point.

\textbf{18.} \qty{12}{kJ/kg}; ice must nucleate, and a small nucleus costs
more surface energy than it gains: pure droplets wait, down to
\qty{-40}{\celsius}.

\textbf{19.} $\Delta v \to 0$, $L \to 0$, their ratio finite: the curve ends;
beyond, one fluid passing continuously from liquid-like to gas-like.

\textbf{20.} $\dd g = v\,\dd P$ at fixed $T$; $g$ equal at the two ends of
the horizontal: $\int v\,\dd P = 0$, equal areas.

\textbf{21.} The superheated-liquid branch: no bubble nucleus in a
smooth cup; a disturbance boils it at once.

\textbf{22.} Freeze, pump below \qty{611}{Pa}, warm gently: the path
crosses the sublimation curve, never the fusion curve; the structure
is preserved.

\textbf{23.} $\mu_{\text{v}}(p) < \mu_{\text{liq}}$ when $p < p_{\text{sat}}$; equal at
saturation.

\textbf{24.} \qty{120}{\celsius} instead of \qty{100}{}: four times faster.

\textbf{25.} $S$ for isolated, $F$ for fixed $T,V$, $G$ for fixed $T,P$;
$-\Delta F$, $-\Delta G$ the recoverable work; the Maxwell relations turn
derivatives of $S$ into the equation of state.
\end{solution}
