\chapter{Gelombang Elektromagnetik Bidang dan Polarisasi}\label{ch:b2:plane-waves-polarization}

Miringkanlah kepala Anda sambil mengenakan kacamata surya berpolarisasi dan
silau di jalan yang basah datang dan hilang; putarlah kacamatanya di depan
layar telepon dan layarnya menjadi hitam. Cahaya yang mencapai mata Anda
bukan hanya gelombang berfrekuensi dan berintensitas: ia mempunyai
\emph{arah getar}, dan setiap polarisator, layar, lensa bioskop tiga
dimensi dan penyambung serat optik bermain dengannya. Bab ini menyelesaikan
\omterm{thm:b2:maxwell-equations:maxwell}{persamaan Maxwell} dalam ruang kosong --- gelombang bidang, dengan medan
listrik dan medan magnetnya yang transversal dan terkunci saling tegak lurus
--- meninjau spektrum yang dibentangkan gelombang itu, lalu mengkaji
polarisasinya: bagaimana melukiskannya, bagaimana memilihnya dengan sebuah
polarisator (\omterm{prop:b2:plane-waves-polarization:malus}{hukum Malus}), dan bagaimana mengubahnya dengan sebuah keping
birefringen.

\section{Gelombang bidang dalam hampa}

\begin{theorem}[Persamaan gelombang; gelombang bidang harmonik yang berjalan]\label{thm:b2:plane-waves-polarization:wave}
\index{gelombang elektromagnetik}\index{gelombang bidang (elektromagnetik)}\index{ketransversalan}
Dalam hampa, bebas muatan dan arus, \omterm{thm:b2:maxwell-equations:maxwell}{persamaan Maxwell} memberi
\[
\Delta\vect E = \frac1{c^2}\frac{\partial^2\vect E}{\partial t^2} , \qquad
\Delta\vect B = \frac1{c^2}\frac{\partial^2\vect B}{\partial t^2} , \qquad
c = \frac1{\sqrt{\varepsilon_0\mu_0}} = \qty{2.998e8}{m/s} .
\]
\emph{Gelombang bidang harmonik yang berjalan} (gelombang PPH) $\underline{\vect E} =
\underline{\vect E}_0\,\eu^{\iu(\omega t - \vect k\cdot\vect r)}$ merupakan penyelesaiannya asalkan
$\omega = ck$ (tanpa dispersi), dan \omterm{thm:b2:maxwell-equations:maxwell}{persamaan Maxwell} lalu menuntut
\[
\vect k\cdot\vect E = 0 , \qquad
\vect B = \frac{\vect k\wedge\vect E}\omega = \frac{\vect u\wedge\vect E}c ,
\]
dengan $\vect u = \vect k/k$ yaitu arah rambatnya: gelombangnya bersifat
\emph{transversal}, $(\vect E, \vect B, \vect u)$ merupakan triad ortogonal
tangan kanan, $\vect E$ dan $\vect B$ sefase dan $B = E/c$. Intensitasnya
$I = \varepsilon_0cE_0^2/2$ dan ia mengemban \omterm{thm:b2:flow-balances:momentum}{fluks momentum} $I/c$
(\cref{ch:b2:poynting-vector}).
\end{theorem}

\begin{proof}
\omterm{thm:b2:waves-on-strings:equation}{Persamaan gelombangnya} telah diturunkan dalam \cref{exo:b2:maxwell-equations:11}.
Bagi $\eu^{\iu(\omega t - \vect k\cdot\vect r)}$, $\partial_t \to \iu\omega$ dan $\vect\nabla \to
-\iu\vect k$: $\Delta \to -k^2$, jadi $k^2 = \omega^2/c^2$; Maxwell--Gauss memberi $-\iu
\vect k\cdot\underline{\vect E} = 0$, Maxwell--fluks $\vect k\cdot\underline{\vect B} = 0$,
Maxwell--Faraday $-\iu\vect k\wedge\underline{\vect E} = -\iu\omega\underline{\vect B}$, yang darinya
$\underline{\vect B} = \vect k\wedge\underline{\vect E}/\omega$, bermodulus $kE/\omega = E/c$ dan
tegak lurus keduanya; Maxwell--Ampère lalu terpenuhi dengan sendirinya.
\end{proof}

\begin{remark}[Spektrumnya]\label{rem:b2:plane-waves-polarization:spectrum}
\index{spektrum elektromagnetik}
Satu persamaan, satu kelajuan, dan dua puluh orde besar frekuensi.
Radio: \qty{30}{kHz}--\qty{300}{MHz} ($\lambda$ dari \qty{10}{km} sampai \qty{1}{m}),
antena dan rangkaian (\cref{ch:b2:dipole-radiation}). Gelombang mikro:
\qty{300}{MHz}--\qty{300}{GHz}, radar, oven, telepon seluler, latar
kosmik. Inframerah: sampai \qty{400}{THz} (\qty{750}{nm}), yaitu radiasi
termal segala benda di sekeliling kita (\cref{ch:b2:thermal-radiation}).
Cahaya tampak: \qty{400}{THz}--\qty{750}{THz}, \qty{750}{nm} (merah) sampai \qty{400}{nm}
(ungu) --- satu oktaf, dengan foton $1.6$ sampai \qty{3.1}{eV}. Ultraungu sampai
\qty{10}{nm}; sinar-X sampai \qty{10}{pm} (\qty{100}{keV}), dari kulit atom dalam
dan elektron yang direm; sinar gama di luar itu, dari inti. Hanya sumber
dan pengindranya yang berbeda: gelombangnya sama.
\end{remark}

\section{Polarisasi}

\begin{definition}[Keadaan polarisasi]\label{def:b2:plane-waves-polarization:states}
\index{polarisasi (cahaya)}\index{polarisasi linear}\index{polarisasi melingkar}\index{polarisasi elips}\index{cahaya alami}
Bagi gelombang PPH sepanjang $z$ medannya dalam bidang $z =$ tetap adalah
\[
\vect E = E_{0x}\cos(\omega t - kz)\,\vect e_x + E_{0y}\cos(\omega t - kz - \varphi)\,\vect e_y ,
\]
dan ujung $\vect E$ melukiskan, pada umumnya, sebuah elips: \emph{polarisasi
elips}. Dua hal khususnya: $\varphi = 0$ atau $\pi$, polarisasi
\emph{linear} sepanjang arah tetap bersudut $\arctan(\pm E_{0y}/E_{0x})$
dari $x$; $E_{0x} = E_{0y}$ dan $\varphi = \pm\pi/2$, polarisasi
\emph{melingkar}, dengan $\vect E$ bermodulus tetap yang berputar pada $\omega$ (ke kanan atau
ke kiri menurut arah putarannya, dilihat menghadap gelombang yang
datang). Cahaya \emph{alami} (tak terpolarisasi), dari sebuah lampu atau Matahari, adalah
rentetan kereta gelombang pendek dengan polarisasi acak yang berubah
cepat: tak ada arah yang disukai secara rerata. Polarisasi apa pun
terurai menjadi dua komponen linear (atau dua komponen melingkar).
\end{definition}

\begin{proof}
Dengan $X = E_x/E_{0x}$, $Y = E_y/E_{0y}$: $X = \cos\theta$, $Y = \cos(\theta - \varphi)$, jadi
$X^2 + Y^2 - 2XY\cos\varphi = \sin^2\varphi$, sebuah elips, yang merosot menjadi
garis $Y = \pm X$ untuk $\varphi = 0, \pi$ dan menjadi lingkaran untuk $E_{0x} = E_{0y}$,
$\varphi = \pm\pi/2$.
\end{proof}

\begin{proposition}[Polarisator dan hukum Malus]\label{prop:b2:plane-waves-polarization:malus}
\index{polarisator}\index{hukum Malus}\index{penganalisis}
Sebuah \emph{polarisator} meneruskan komponen $\vect E$ sepanjang sumbu
penerusannya $\vect p$ lalu menyerap yang lain (sehelai lembaran molekul
panjang yang terjajar, yang menghantar sepanjang panjangnya --- polarisasi
yang diteruskan tegak lurus molekulnya). Cahaya terpolarisasi linear
berintensitas $I_0$ yang arahnya membuat sudut $\theta$ dengan
$\vect p$ muncul terpolarisasi linear sepanjang $\vect p$ dengan
\[
I = I_0\cos^2\theta \qquad\text{(Malus)} ;
\]
\omterm{def:b2:plane-waves-polarization:states}{cahaya alami} muncul dengan $I_0/2$, terpolarisasi sepanjang $\vect p$. Dua
polarisator bersilang tak meneruskan apa pun; satu lagi yang disisipkan pada
\qty{45}{\degree} di antaranya meneruskan $I_0/8$.
\end{proposition}

\begin{proof}
Amplitudo yang diteruskan adalah $E_0\cos\theta$ dan $I \propto E_0^2$. \omterm{def:b2:plane-waves-polarization:states}{Cahaya
alami}: rerata $\cos^2\theta$ atas semua $\theta$, yaitu $\tfrac12$. Tiga polarisator:
$\tfrac12\cdot\cos^2 45^\circ\cdot\cos^2 45^\circ = \tfrac18$ --- polarisator
tengahnya tidak sekadar melemahkan, ia \emph{memutar} polarisasinya.
\end{proof}

\begin{example}[Kacamata surya, layar, fotografi]\label{ex:b2:plane-waves-polarization:uses}
Cahaya yang dipantulkan pada sudut menyerempet dari air atau aspal sebagian besar
terpolarisasi mendatar (sudut Brewster, \cref{ch:b2:wave-interfaces}):
kacamata surya bersumbu penerusan tegak memangkas silaunya dan menyimpan
selebihnya. Layar kristal cair memancarkan cahaya terpolarisasi: menembus
polarisator pada \qty{90}{\degree} ia menjadi hitam. Tapis polarisasi seorang
juru foto menggelapkan langitnya (cahaya terhambur sebagian terpolarisasi,
\cref{ch:b2:dipole-radiation}) lalu menyingkirkan pantulan dari kaca.
\end{example}

\section{Birefringensi dan keping gelombang}

\begin{proposition}[Keping gelombang]\label{prop:b2:plane-waves-polarization:plates}
\index{birefringensi}\index{keping gelombang}\index{keping setengah gelombang}\index{keping seperempat gelombang}
\index{sumbu cepat}\index{sumbu lambat}
Dalam kristal \emph{birefringen} (kalsit, kuarsa, mika; juga lembaran
plastik yang teregang) sebuah gelombang yang merambat sepanjang arah tertentu
terbelah menjadi dua \omterm{def:b2:plane-waves-polarization:states}{polarisasi linear}, sepanjang sumbu \emph{cepat} dan
sumbu \emph{lambat} kepingnya, yang berjalan dengan indeks berbeda $n_f <
n_s$ (diterima tanpa bukti --- kristalnya menanggapi secara berbeda sepanjang sumbunya). Sebuah
keping setebal $e$ menunda komponen lambatnya sebesar fase
\[
\delta = \frac{2\pi}\lambda(n_s - n_f)\,e .
\]
Sebuah \emph{\omterm{prop:b2:plane-waves-polarization:plates}{keping setengah gelombang}} ($\delta = \pi$) mengubah \omterm{def:b2:plane-waves-polarization:states}{polarisasi linear} bersudut
$\alpha$ dari \omterm{prop:b2:plane-waves-polarization:plates}{sumbu cepatnya} menjadi \omterm{def:b2:plane-waves-polarization:states}{polarisasi linear} pada $-\alpha$: ia
memutarnya sebesar $2\alpha$. Sebuah \emph{\omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang}} ($\delta = \pi/2$) mengubah
\omterm{def:b2:plane-waves-polarization:states}{polarisasi linear} pada \qty{45}{\degree} dari sumbunya menjadi cahaya melingkar,
dan cahaya melingkar kembali menjadi linear; pada sudut lain, menjadi cahaya
elips yang sumbunya sepanjang sumbu kepingnya.
\end{proposition}

\begin{proof}
Tulislah medannya $E_0(\cos\alpha\,\vect e_f + \sin\alpha\,\vect e_s)\cos\omega t$;
sesudah kepingnya, $E_0[\cos\alpha\cos\omega t\,\vect e_f + \sin\alpha\cos(\omega t - \delta)
\vect e_s]$. Untuk $\delta = \pi$ komponen $\vect e_s$-nya berubah tanda: arahnya
$(\cos\alpha, -\sin\alpha)$. Untuk $\delta = \pi/2$ dan $\alpha = 45^\circ$: $(E_0/\sqrt2)(\cos
\omega t, \sin\omega t)$, sebuah lingkaran.
\end{proof}

\begin{example}[Di mana keping dipakai]\label{ex:b2:plane-waves-polarization:plateuses}
Sebuah \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} di antara polarisator dan cermin membuat sebuah
\emph{isolator optik}: cahayanya keluar melingkar, kembali
melingkar dengan arah putar berlawanan, diubah kepingnya menjadi cahaya
linear pada \qty{90}{\degree} dari polarisatornya lalu terserap --- tak ada
pantulan yang kembali ke lasernya. Kedua citra mata sebuah film tiga dimensi
diproyeksikan dalam \omterm{def:b2:plane-waves-polarization:states}{polarisasi melingkar} yang berlawanan lalu disortir oleh
lapisan \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} dan polarisator kacamatanya, yang tetap bekerja
ketika Anda memiringkan kepala (polarisator linear tidak akan). Sebuah penggaris
bening di antara polarisator bersilang memperlihatkan rumbai berwarna: tegangan membuat
plastik menjadi birefringen, dan insinyur membaca tegangan sebuah model dari
polanya.
\end{example}

\begin{method}[Mengikuti sebuah polarisasi menembus optika]\label{met:b2:plane-waves-polarization:method}
(1) Pilihlah sumbunya; tuliskanlah medan datangnya sebagai dua komponen dengan
beda fasenya. (2) Sebuah polarisator: simpanlah proyeksinya pada
sumbunya (amplitudonya $\times\cos$, intensitasnya $\times\cos^2$; \omterm{def:b2:plane-waves-polarization:states}{cahaya alami}
$\times\tfrac12$). (3) Sebuah keping: proyeksikanlah pada \omterm{prop:b2:plane-waves-polarization:plates}{sumbu cepat} dan lambatnya, tundalah
yang lambat sebesar $\delta$, lalu gabungkanlah kembali. (4) Bacalah hasilnya: fase $0$ atau $\pi$
berarti linear, $\pm\pi/2$ dengan amplitudo sama berarti melingkar. (5) Untuk
menguji berkas yang tak diketahui: putarlah sebuah polarisator (intensitasnya berubah: bagian linear
atau elipsnya), lalu tambahkanlah \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} (yang melingkar menjadi
linear dan dapat dipadamkan).
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:plane-waves-polarization:1}
Sebuah pemancar FM pada \qty{100}{MHz} memberi, di sebuah penerima, intensitas
\qty{1.0}{\micro W/m^2}. Tentukanlah panjang gelombang, bilangan gelombang, periodenya; amplitudo $E_0$ dan
$B_0$; rapat energinya; ggl yang terimbas dalam antena \qty{1}{m} yang terjajar dengan
$\vect E$; fluks fotonnya (tiap meter persegi dan sekon).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:plane-waves-polarization:2}
Tentukanlah frekuensi dan energi foton (dalam eV) bagi $\lambda = \qty{1}{km}$, \qty{10}{cm},
\qty{10}{\micro m}, \qty{550}{nm}, \qty{10}{nm}, \qty{0.1}{nm}, \qty{1}{pm}; namailah
pita masing-masingnya; panjang gelombang sinar gama \qty{1}{MeV} dan panjang gelombang latar
kosmik \qty{2.7}{K} (puncaknya dekat \qty{160}{GHz}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:plane-waves-polarization:3}
Cahaya terpolarisasi linear berintensitas $I_0$ jatuh pada polarisator pada
\qty{30}{\degree}, lalu yang kedua pada \qty{60}{\degree} dari yang pertama, lalu yang
ketiga pada \qty{90}{\degree} dari yang pertama: intensitasnya sesudah masing-masing. Hal yang sama dengan
\omterm{def:b2:plane-waves-polarization:states}{cahaya alami}. Singkirkanlah yang tengah: apa yang berubah?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:plane-waves-polarization:4}
Periksalah bahwa $\vect E = E_0\cos(\omega t - kz)\,\vect e_x$, $\vect B = (E_0/c)\cos(\omega t
- kz)\,\vect e_y$ memenuhi keempat \omterm{thm:b2:maxwell-equations:maxwell}{persamaan Maxwell} dalam hampa ketika $\omega =
kc$. Persamaan mana yang menetapkan arah $\vect B$, mana yang menetapkan
besarnya? Apa yang terjadi jika orang mencoba $\vect E$ sepanjang $\vect e_z$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:plane-waves-polarization:5}
\emph{Cahaya melingkar.} (a) Tuliskanlah medan sebuah gelombang terpolarisasi
melingkar sepanjang $z$ lalu tunjukkanlah bahwa $|\vect E|$ tetap sementara arahnya
berputar pada $\omega$. (b) Tunjukkanlah bahwa jumlah gelombang melingkar kanan dan kiri
yang beramplitudo sama terpolarisasi linear, sepanjang arah yang ditetapkan
beda fasenya. (c) Menembus sebuah polarisator, apa yang diberikan cahaya melingkar,
dan bergantungkah intensitasnya pada sudut polarisatornya? (d) Lalu bagaimana
orang dapat membedakan cahaya melingkar dari \omterm{def:b2:plane-waves-polarization:states}{cahaya alami}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:plane-waves-polarization:6}
Kuarsa mempunyai $n_s - n_f = 0.0091$ pada \qty{589}{nm}. (a) Tebal sebuah
\omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} berorde terendah; dan \omterm{prop:b2:plane-waves-polarization:plates}{keping setengah gelombang}. (b) Sebuah
\omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} bagi \qty{589}{nm} yang dipakai pada \qty{450}{nm}: tunda fasenya
(abaikanlah perubahan indeksnya); apa yang keluar bagi cahaya linear
pada \qty{45}{\degree}? (c) Mengapa keping ``berorde ganda'' (setebal $e +
m\lambda/(n_s - n_f)$) lebih peka terhadap panjang gelombang dan suhu? (d) Mengapa
tebal sebuah keping harus dikendalikan sampai lebih baik daripada satu mikrometer?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:plane-waves-polarization:7}
\omterm{prop:b2:plane-waves-polarization:plates}{Keping setengah gelombang}. (a) Tunjukkanlah bahwa cahaya linear bersudut $\alpha$ dari
\omterm{prop:b2:plane-waves-polarization:plates}{sumbu cepatnya} keluar linear pada $-\alpha$. (b) Karena itu keping yang diputar sebesar $\beta$
memutar polarisasinya sebesar $2\beta$: putaran keping berapa yang mengubah
polarisasi tegak menjadi mendatar? (c) Apa yang dilakukan \omterm{prop:b2:plane-waves-polarization:plates}{keping setengah gelombang} pada
cahaya melingkar? (d) Di antara polarisator bersilang, sebuah \omterm{prop:b2:plane-waves-polarization:plates}{keping setengah gelombang}
diputar perlahan: sketsalah intensitas yang diteruskan terhadap sudutnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:plane-waves-polarization:8}
Sebanyak $N$ polarisator ditaruh berturut-turut, masing-masing terputar sebesar
$\pi/2N$ dari yang sebelumnya, yang terakhir pada \qty{90}{\degree} dari yang
pertama. Tentukanlah penerusan cahaya terpolarisasi linear yang terjajar dengan yang pertama
untuk $N = 1, 2, 5, 10, 100$; limit $N \to \infty$ (pakailah $\cos^{2N}(\pi/2N) \approx
1 - \pi^2/4N$). Berilah tanggapan: sebuah polarisasi dapat diputar \qty{90}{\degree}
tanpa rugi --- oleh apa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:plane-waves-polarization:9}
\emph{Polarisasi sebagian.} Sebuah polarisator yang diputar di depan sebuah berkas
memberi intensitas maksimum $I_{\max}$ dan minimum $I_{\min}$. (a) Modelkanlah
berkasnya sebagai \omterm{def:b2:plane-waves-polarization:states}{cahaya alami} $I_n$ ditambah cahaya terpolarisasi linear $I_p$: nyatakanlah
$I_{\max}$ dan $I_{\min}$; derajat polarisasinya $p = I_p/(I_n + I_p) =
(I_{\max} - I_{\min})/(I_{\max} + I_{\min})$. (b) Langit biru pada \qty{90}{\degree} dari
Matahari: $I_{\max}/I_{\min} = 5$: berapa $p$? (c) Cahaya yang dipantulkan dari sebuah danau dekat
sudut Brewster, $p = 0.9$: seberapa banyak yang dapat disingkirkan kacamata surya berpolarisasi?
(d) Dapatkah sebuah polarisator sendirian membedakan cahaya yang sebagian terpolarisasi
linear dari yang sebagian melingkar? Apa yang diperlukan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:plane-waves-polarization:10}
\emph{Dua berkas bersilang.} Dua gelombang PPH beramplitudo dan berfrekuensi
sama, keduanya terpolarisasi sepanjang $\vect e_y$, merambat dalam bidang $xz$
pada sudut $\pm\theta$ dari $z$. (a) Tuliskanlah medan totalnya lalu tunjukkanlah bahwa ia sebuah
gelombang yang berjalan sepanjang $z$ yang amplitudonya termodulasi sepanjang $x$ sebagai
$\cos(kx\sin\theta)$. (b) Jarak antara bidang gelapnya (rumbai
interferensi); angkanya bagi $\lambda = \qty{633}{nm}$, $\theta = \qty{5}{\degree}$, dan bagi
$\theta = \qty{90}{\degree}$ (berlawanan arah). (c) \omterm{def:b2:dispersion-wave-packets:relation}{Kecepatan fase}
polanya sepanjang $z$; bandingkanlah dengan $c$; adakah sesuatu yang berjalan lebih cepat daripada
cahaya? (d) Hitunglah \omterm{thm:b2:poynting-vector:poynting}{vektor Poynting} yang dirata-ratakan waktu lalu tunjukkanlah bahwa
komponen $x$-nya lenyap: energinya mengalir sepanjang $z$ dalam bidang
terangnya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:plane-waves-polarization:11}
\emph{Piksel kristal cair.} Sebuah sel nematik terpuntir di antara
polarisator bersilang memutar polarisasi cahaya sebesar \qty{90}{\degree}
ketika tak ada tegangan yang dikenakan (penjajaran molekulnya terpuntir menembus
selnya dan polarisasinya mengikutinya --- diterima tanpa bukti), dan sama sekali tidak
ketika beberapa volt meluruskannya. (a) Penerusan dalam kedua keadaannya: yang
mana yang terang? (b) Sel yang sungguhan memutar sebesar $90^\circ \pm 5^\circ$ pada panjang gelombang
rancangannya: nisbah kontras $I_{\text{terang}}/I_{\text{gelap}}$ jika keadaan
gelapnya bocor lewat galat sisa \qty{5}{\degree}. (c) Mengapa sel yang
dirancang bagi \qty{550}{nm} membocorkan sedikit biru dan merah dalam keadaan gelapnya
(pikirkanlah putarannya sebagai tumpukan banyak penunda tipis serupa
setengah gelombang)? (d) Sebuah piksel warna memakai tiga sel semacam itu dengan tapis; mengapa
layarnya menjadi hitam menembus sebuah polarisator pada sudut tertentu, dan apa
yang diberitahukan sudut itu kepada Anda?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:plane-waves-polarization:12}
\emph{Aktivitas optik.} Dalam larutan gula kedua \omterm{def:b2:plane-waves-polarization:states}{polarisasi
melingkarnya} berjalan dengan indeks yang sedikit berbeda $n_L$ dan $n_R$.
(a) Uraikanlah sebuah \omterm{def:b2:plane-waves-polarization:states}{polarisasi linear} menjadi komponen melingkar lalu tunjukkanlah
bahwa sesudah panjang $\ell$ polarisasinya tetap linear tetapi terputar
sebesar $\theta = \pi(n_L - n_R)\ell/\lambda$. (b) Sebuah tabung \qty{1}{dm} berisi larutan
sukrosa \qty{0.10}{g/mL} memutar cahaya natrium sebesar \qty{6.65}{\degree}: berapa $n_L - n_R$? (c)
Sebuah sakarimeter mengukur $\theta$ sampai \qty{0.01}{\degree}: ketelitian pada
kepekatannya. (d) Efek yang sama dihasilkan dalam kaca oleh medan magnet
sepanjang berkasnya (efek Faraday, $\theta = VB\ell$ dengan $V \approx
\qty{4}{rad.T^{-1}.m^{-1}}$): mengapa sebuah pemutar Faraday, tidak seperti tabung
gula, tidak meniadakan putarannya ketika cahayanya dikirim balik, dan bagaimana
itu membuatnya menjadi sebuah isolator?
\end{exercise}

\section{Soal: Gelombang, layar dan layar surya}

\begin{problem}[{Soal akhir pekan --- sebuah berkas laser yang dibongkar menjadi
medannya, sebuah layar yang melukis dengan polarisasi, dan sebuah wahana antariksa yang
berlayar di atas cahaya}]
\label{pb:b2:plane-waves-polarization:1}

\medskip
\noindent\textbf{Bagian I --- Berkasnya.}
Sebuah laser helium--neon memancarkan \qty{10}{mW} pada \qty{632.8}{nm} dalam berkas
bergaris tengah \qty{1.0}{mm}, terpolarisasi linear sepanjang $x$, berjalan sepanjang
$z$.
\begin{enumerate}
  \item Frekuensi, bilangan gelombang, energi fotonnya; foton yang dipancarkan tiap
        sekon.
  \item Intensitasnya; amplitudo $E_0$ dan $B_0$; tuliskanlah $\vect E(z, t)$ dan
        $\vect B(z, t)$.
  \item Rapat energinya (rerata) dan energi yang termuat dalam \qty{1}{m}
        berkasnya; berapa panjang berkas yang dipancarkan dalam \qty{1}{ns}, dan berapa
        panjang gelombang yang dimuatnya?
  \item \omterm{thm:b2:poynting-vector:poynting}{Vektor Poynting}: nilai rerata dan ayunannya; momentum
        yang diemban tiap sekon; gaya pada sasaran hitam dan pada cermin.
  \item Bandingkanlah $E_0$ dengan medan yang mengikat elektron dalam sebuah
        atom hidrogen (\qty{5e11}{V/m}) dan $B_0$ dengan medan Bumi
        (\qty{50}{\micro T}).
  \item \omterm{prop:b2:poynting-vector:wave}{Tekanan radiasi} pada sebuah penghenti berkas hitam, dalam pascal.
  \item Sebuah elektron bebas yang ditaruh dalam berkasnya: amplitudo ayunan
        kecepatannya dalam medan listriknya (abaikanlah gaya magnetiknya
        lebih dulu); nisbah gaya magnetik terhadap gaya listrik padanya ---
        mengapa gaya magnetiknya dapat diabaikan bagi materi dalam cahaya
        biasa?
  \item Tuliskanlah medan berkas yang sama seandainya ia terpolarisasi
        melingkar; apa yang berubah pada intensitasnya, pada \omterm{thm:b2:poynting-vector:poynting}{vektor Poyntingnya}
        dan pada kebergantungan waktunya?
  \item Berkasnya menembus sebuah polarisator yang sumbunya membuat
        \qty{30}{\degree} dengan $x$: daya yang diteruskan; lalu menembus polarisator
        kedua yang bersilang dengan yang pertama: dayanya; sisipkanlah di antaranya
        yang ketiga pada \qty{45}{\degree} dari yang pertama: dayanya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Layarnya.}
Sebuah layar kristal cair: lampu latar (\omterm{def:b2:plane-waves-polarization:states}{cahaya alami}), sebuah polarisator, sebuah
sel, sebuah polarisator bersilang. Tanpa tegangan selnya memutar
polarisasinya sebesar \qty{90}{\degree}; dengan tegangan ia tak melakukan apa pun.
\begin{enumerate}[resume]
  \item Bagian intensitas lampu latarnya yang diteruskan dalam
        keadaan terang (polarisator sempurna); dalam keadaan gelap.
  \item Sebuah tegangan antara membuat selnya berperilaku sebagai penunda
        berfase $\delta$ antara sumbunya, yang berada pada \qty{45}{\degree} terhadap
        polarisatornya (tanpa putaran): tunjukkanlah bahwa penerusannya
        $\sin^2(\delta/2)$ --- yaitu skala kelabunya.
  \item Dilihat menembus sebuah polarisator yang diputar ke \qty{90}{\degree} dari
        polarisator keluaran layarnya, apa yang terlihat? Pada \qty{45}{\degree}?
  \item Sebuah \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} yang dilekatkan pada layarnya pada \qty{45}{\degree} terhadap
        polarisatornya membuat keluarannya melingkar: apa untungnya bagi penonton
        yang mengenakan kacamata surya berpolarisasi?
  \item Polarisator yang sempurna akan meloloskan separuh lampu latarnya dalam
        keadaan terang; lembaran yang sungguhan meneruskan $85\%$ dari yang ideal masing-masing, dan
        tapis warna sebuah piksel meloloskan sepertiga cahayanya:
        bagian daya lampu latarnya yang mencapai penontonnya dalam warna putih;
        sebutkanlah dua tempat yang di sana sebagian besar cahayanya hilang.
  \item Lembaran bersilang yang sungguhan membocorkan $0.1\%$ cahayanya: nisbah kontras
        layarnya (terang dibagi gelap) dalam ruang gelap.
  \item ``Polarisator melingkar'' para juru foto adalah sebuah polarisator linear
        yang diikuti \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} pada \qty{45}{\degree}: apa yang
        diterima kameranya, dan mengapa itu lebih baik bagi pembelah berkas
        yang peka polarisasi pada fokus otomatisnya daripada cahaya
        linear?
  \item Mengapa kacamata bioskop tiga dimensi memakai \omterm{def:b2:plane-waves-polarization:states}{polarisasi melingkar} alih-alih
        \omterm{def:b2:plane-waves-polarization:states}{polarisasi linear}? Apa yang terjadi pada rembesan silang antara kedua matanya
        jika penontonnya memiringkan kepalanya \qty{20}{\degree} dengan kacamata
        linear (intensitas yang bocor ke mata yang salah)?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Layar suryanya.}
Sebuah layar surya seluas $S = \qty{1000}{m^2}$ dan bermassa total $m = \qty{50}{kg}$,
yang memantul sempurna, pada $D = \qty{1.0}{au} = \qty{1.5e11}{m}$ dari Matahari
($I = \qty{1.36}{kW/m^2}$; gravitasi Matahari \qty{5.9e-3}{m/s^2} di sana).
\begin{enumerate}[resume]
  \item Gaya radiasi ketika layarnya menghadap Matahari; percepatannya;
        nisbahnya terhadap gravitasi matahari.
  \item Layarnya dimiringkan sebesar $\theta$ dari menghadap Matahari: tunjukkanlah bahwa
        gayanya normal terhadap layarnya dan sebanding dengan $\cos^2\theta$
        (daya yang tercegat $\times\cos\theta$, perubahan momentum sepanjang
        normalnya $\times\cos\theta$).
  \item Untuk menyusur spiral ke luar layarnya menjaga sebuah komponen gaya sepanjang
        kecepatan orbitnya: untuk $\theta = \qty{35}{\degree}$, tentukanlah percepatan
        tangensialnya; kelajuan yang diperoleh tiap tahun; bandingkanlah dengan kelajuan
        orbit Bumi (\qty{30}{km/s}).
  \item Mengapa layar yang sama, dekat Merkurius ($D = \qty{0.39}{au}$), memperoleh
        gaya $6.6$ kali lipat, dan mengapa nisbahnya terhadap gravitasi
        tak berubah?
  \item Selaput layarnya adalah plastik beralumunium setebal \qty{2}{\micro m}: berapa
        suhunya dalam cahaya matahari jika ia menyerap $10\%$ lalu memancar dari
        kedua mukanya sebagai benda hitam ($\sigma = \qty{5.67e-8}{W.m^{-2}.K^{-4}}$).
  \item Waktu yang diperlukan untuk memperoleh \qty{1}{km/s} sambil menghadap Matahari; apa yang akan
        dicapai layar hitam (yang menyerap) berukuran sama?
  \item Periksalah momentum cahayanya dengan foton: jumlah foton yang
        menumbuk layarnya tiap sekon (rata-rata \qty{2}{eV}) dan
        momentum $hf/c$ masing-masingnya; dapatkanlah kembali gayanya.
  \item Rangkumlah: ketiga besaran yang diemban gelombangnya (energi,
        momentum, polarisasi) dan perangkat soal ini yang memanfaatkan
        masing-masingnya.
\end{enumerate}
\end{problem}
