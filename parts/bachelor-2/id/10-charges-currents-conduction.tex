\chapter{Muatan, Arus dan Penghantaran}\label{ch:b2:charges-currents-conduction}

Tekanlah sakelarnya dan lampunya menyala seketika; namun elektron dalam
kawat tembaganya menghanyut menujunya sepersekian milimeter tiap sekon,
dan akan menuntut berjam-jam untuk tiba. Sebuah sambaran petir mengemban tiga puluh
ribu ampere menembus saluran selebar sebuah jari; saraf

hukum bahwa muatan tak pernah diciptakan atau dimusnahkan, lalu menurunkan hukum
Ohm dari desak-desakan elektron dalam sebuah logam: gambaran mikroskopik
di balik resistor, sekring dan pengindra Hall.

\section{Rapat muatan dan rapat arus}

\begin{definition}[Rapat muatan; rapat arus]\label{def:b2:charges-currents-conduction:densities}
\index{rapat muatan}\index{rapat arus}\index{kuat arus (fluks j)}
Pada \omterm{def:b2:fluid-kinematics:particle}{skala mesoskopik}, materi mengemban \emph{rapat muatan volume}
$\rho(M, t)$ (\unit{C/m^3}): muatan $\rho\,\dd\tau$ dalam $\dd\tau$. Sebuah permukaan
dapat mengemban \emph{rapat permukaan} $\sigma$ (\unit{C/m^2}), sebuah kawat rapat
garis $\lambda$ (\unit{C/m}). Jika pengemban jenis $i$ (bermuatan
$q_i$, berkerapatan jumlah $n_i$) bergerak dengan kecepatan rata-rata $\vect v_i$,
\emph{rapat arus}nya adalah
\[
\vect j = \sum_in_iq_i\,\vect v_i \qquad (\unit{A/m^2}) ,
\]
dan \emph{kuat arus} menembus permukaan terorientasi $S$ adalah fluks
$I = \iint_S\vect j\cdot\vect n\,\dd S$: muatan yang menyeberangi $S$ tiap satuan
waktu. Untuk satu jenis saja, $\vect j = \rho_m\vect v$ dengan $\rho_m = nq$ yaitu
rapat muatan yang dapat bergerak.
\end{definition}

\begin{proof}
Dalam $\dd t$ pengemban jenis $i$ yang menyeberangi $\dd S$ adalah yang berada dalam
tabung serong beralas $\dd S$ dan bergaris pelukis $\vect v_i\dd t$: jumlahnya
$n_i\vect v_i\cdot\vect n\,\dd S\,\dd t$, muatannya $q_i$ kali itu.
\end{proof}

\begin{example}[Betapa lambat elektron menghanyut]\label{ex:b2:charges-currents-conduction:drift}
\index{kecepatan hanyut}
Tembaga mempunyai satu elektron hantaran tiap atom: $n = \rho_{\text{Cu}}N_A/M =
8900 \times 6.02 \times 10^{23}/0.0635 = \qty{8.5e28}{m^{-3}}$. Arus
\qty{10}{A} dalam kawat \qty{1.5}{mm^2} berarti $j = \qty{6.7e6}{A/m^2}$, dan
\emph{\omterm{ex:b2:charges-currents-conduction:drift}{kecepatan hanyut}}nya adalah $v = j/ne = 6.7 \times 10^6/(8.5 \times 10^{28} \times 1.6 \times
10^{-19}) = \qty{0.5}{mm/s}$ --- dua jam tiap meter. Lampunya menyala
seketika sebab \emph{medan} yang mendorong elektronnya terbangun
sepanjang seluruh kawatnya dalam hitungan nanosekon (ia berjalan hampir
dengan kelajuan cahaya, seperti diperlihatkan kabel \cref{ch:b2:dispersion-wave-packets}):
semua elektronnya berangkat bersama, seperti air dalam selang yang penuh.
\end{example}

\section{Kekekalan muatan}

\begin{theorem}[Kekekalan muatan lokal]\label{thm:b2:charges-currents-conduction:conservation}
\index{kekekalan muatan (lokal)}\index{persamaan kontinuitas (muatan)}
Di setiap titik,
\[
\frac{\partial\rho}{\partial t} + \operatorname{div}\vect j = 0 ;
\]
diintegralkan atas volume tetap $V$ yang dibatasi permukaan tertutup $\Sigma$,
$\dd Q_V/\dd t = -I_{\text{luar}}$: muatan di dalamnya hanya berubah sebanyak yang
menyeberangi batasnya. Dalam \emph{rezim tunak} ($\partial_t\rho = 0$),
$\operatorname{div}\vect j = 0$: fluks $\vect j$ kekal sepanjang sebuah tabung
arus, kuat arus yang sama menyeberangi tiap penampang sebuah kawat, dan
arus yang memasuki sebuah simpul sama dengan yang meninggalkannya (hukum simpul Kirchhoff).
\end{theorem}

\begin{proof}
Kata demi kata sama dengan neraca massa \cref{thm:b2:fluid-kinematics:continuity}
dengan $\rho$ yaitu \omterm{def:b2:charges-currents-conduction:densities}{rapat muatannya} dan $\vect j$ arusnya: keluaran total
muatan menembus keenam sisi sebuah kotak tetap, tiap satuan volume, adalah
$\operatorname{div}\vect j$, dan tak ada muatan yang diciptakan. Bentuk integralnya menjumlahkan
kotaknya (sisi dalamnya saling meniadakan). Untuk sebuah tabung antara dua penampang
$S_1$, $S_2$ tanpa fluks menembus sisinya, $I_1 = I_2$; sebuah simpul adalah
volume kecil dengan beberapa kawat: $\sum I_{\text{masuk}} = \sum I_{\text{luar}}$.
\end{proof}

\begin{remark}[Di mana arusnya berhenti: kapasitor]\label{rem:b2:charges-currents-conduction:capacitor}
Sebuah kawat yang menyuapi keping kapasitor adalah tabung arus yang berakhir:
fluks $\vect j$ ke dalam kepingnya adalah $I = \dd Q/\dd t$, dan muatan menumpuk
--- $\operatorname{div}\vect j \ne 0$ di sana, rezimnya tidak tunak.
Namun arus $I$ yang sama mengalir dalam kawat di sisi yang lain: sesuatu
harus mengemban imbangannya menyeberangi celahnya. \cref{ch:b2:maxwell-equations}
menamainya --- arus pergeseran --- lalu menjadikan kekekalan
muatan sebagai batu kunci persamaan Maxwell.
\end{remark}

\section{Model Drude dan hukum Ohm}

\begin{proposition}[Model Drude; hukum Ohm lokal]\label{prop:b2:charges-currents-conduction:drude}
\index{model Drude}\index{hukum Ohm (lokal)}\index{konduktivitas}\index{hambatan jenis}\index{mobilitas}
Modelkanlah elektron hantaran sebuah logam sebagai partikel bebas (massa $m$,
muatan $-e$, kerapatan $n$) yang merasakan medan $\vect E$ dan, rata-rata,
kehilangan momentum hanyutnya karena tumbukan dengan kisinya tiap $\tau$
sekon --- yaitu \omterm{def:b2:rigid-body-mechanics:contact}{gaya gesek} $-m\vect v/\tau$ pada kecepatan rata-ratanya:
\[
m\frac{\dd\vect v}{\dd t} = -e\vect E - \frac m\tau\vect v .
\]
Dalam medan tetap hanyutannya menetap, dalam beberapa $\tau$, pada $\vect v = -e\tau
\vect E/m$, dan rapat arusnya adalah
\[
\vect j = \gamma\,\vect E , \qquad \gamma = \frac{ne^2\tau}{m} :
\]
yaitu \emph{hukum Ohm lokal}, dengan $\gamma$ yaitu \emph{konduktivitas}
(\unit{S/m}; kebalikannya $\rho_e = 1/\gamma$ adalah \emph{\omterm{prop:b2:charges-currents-conduction:drude}{hambatan jenis}},
\unit{\ohm.m}). Nisbah $\mu = e\tau/m$ antara kelajuan hanyutnya dan medannya adalah
\emph{mobilitas} (\unit{m^2/(V.s)}). Untuk penghantar sepanjang $L$ dan
berpenampang seragam $S$ yang mengemban $\vect j$ seragam, mengintegralkan $E = j/\gamma$
sepanjang panjangnya memberi $U = EL = (L/\gamma S)I$: yaitu hambatan $R = L/\gamma S
= \rho_eL/S$ jilid Tahun ke-1.
\end{proposition}

\begin{proof}
Dengan $\vect v = -e\tau\vect E/m$, $\vect j = -ne\vect v = (ne^2\tau/m)\vect E$.
Peralihannya $\vect v(t) = \vect v_\infty(1 - \eu^{-t/\tau})$ berlangsung $\tau \sim
\qty{1e-14}{s}$, seketika pada setiap skala waktu rangkaian. Hambatannya
menyusul dari $\int\vect E\cdot\dd\vect l$ antara kedua ujungnya.
\end{proof}

\begin{example}[Tembaga, dari konduktivitas terukurnya]\label{ex:b2:charges-currents-conduction:copper}
Tembaga: $\gamma = \qty{6.0e7}{S/m}$, $n = \qty{8.5e28}{m^{-3}}$: $\tau = m\gamma/ne^2 =
9.1 \times 10^{-31} \times 6 \times 10^7/(8.5 \times 10^{28} \times 2.56 \times 10^{-38}) = \qty{2.5e-14}{s}$
--- dua puluh lima femtosekon di antara tumbukan. Kelajuan acak
elektron dalam logam sekitar \qty{1.6e6}{m/s} (sebuah gejala kuantum, bukan
kelajuan termal $\sqrt{3k_BT/m} \approx \qty{1e5}{m/s}$), jadi jalan bebas rata-ratanya $\ell
= v\tau \approx \qty{40}{nm}$, yaitu seratus jarak atom: elektronnya tidak
bertumbukan dengan atomnya sendiri (kristal yang sempurna tembus pandang bagi
mereka) melainkan dengan cacat dan getaran termalnya --- itulah sebabnya
$\gamma$ turun ketika sebuah logam dipanaskan atau dipaduk. Mobilitasnya $e\tau/m =
\qty{4.4e-3}{m^2/(V.s)}$: \qty{4}{mm/s} tiap volt tiap meter.
\end{example}

\begin{proposition}[Hukum Ohm pada frekuensi tinggi]\label{prop:b2:charges-currents-conduction:acdrude}
\index{konduktivitas kompleks}
Untuk medan sinusoidal $\underline{\vect E}\,\eu^{\iu\omega t}$ persamaan Drude
memberi $\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$ dengan
\[
\underline\gamma(\omega) = \frac{\gamma_0}{1 + \iu\omega\tau} , \qquad \gamma_0 = \frac{ne^2\tau}{m} :
\]
konduktivitasnya nyata dan sama dengan nilai searahnya selama $\omega\tau
\ll 1$, yakni sampai sekitar \qty{1e13}{Hz} bagi tembaga --- di sepanjang seluruh
jangkauan radio dan gelombang mikro; dalam inframerah dan cahaya tampak ($\omega\tau \gg 1$)
ia menjadi khayal, $\underline\gamma \approx ne^2/\iu m\omega$: elektronnya
berayun bebas, berbeda fase dengan medannya, dan logamnya berperilaku
sebagai plasma \cref{ch:b2:waves-in-media}.
\end{proposition}

\begin{proof}
$\iu\omega m\underline{\vect v} = -e\underline{\vect E} - m\underline{\vect v}/\tau$, yang darinya
$\underline{\vect v} = -e\tau\underline{\vect E}/m(1 + \iu\omega\tau)$ dan $\underline{\vect j}
= -ne\underline{\vect v}$.
\end{proof}

\begin{remark}[Penghantar, semikonduktor, elektrolit]\label{rem:b2:charges-currents-conduction:materials}
\index{semikonduktor}\index{elektrolit}
Hukum yang sama $\vect j = \sum n_iq_i\mu_i\vect E$ berlaku bagi setiap penghantar;
yang berubah adalah $n$ dan $\mu$. Logam: $n \sim 10^{29}$, $\gamma \sim 10^7$
S/m. Semikonduktor: $n \sim 10^{16}$--$10^{23}$, yang ditetapkan pendopingan dan naik
tajam dengan suhu, dengan dua macam pengemban (elektron dan
lubang) --- dari situlah termistor dan dioda. Elektrolit: ion bertanda
dua-duanya, $\mu \sim 10^{-8}$--$10^{-7}$ m$^2$/(V s); air laut $\gamma \approx \qty{5}{S/m}$,
air keran $10^{-2}$, air murni $10^{-5}$. Isolator (kaca, polimer):
$\gamma \lesssim 10^{-12}$ S/m. Efek Hall jilid Tahun ke-1, dalam
bentuk lokal $\vect E_H = -\vect j\wedge\vect B/nq$, mengukur $n$ dan tanda
$q$: dalam aluminium atau seng pengembannya ternyata positif ---
lubang --- dan tak ada model klasik yang menjelaskannya.
\end{remark}

\section{Energi: hukum Joule lokal}

\begin{proposition}[Hukum Joule lokal]\label{prop:b2:charges-currents-conduction:joule}
\index{hukum Joule (lokal)}
Medan listriknya menyerahkan kepada pengembannya, tiap satuan volume dan waktu,
daya $\vect j\cdot\vect E$; dalam penghantar ohmik ini sama dengan
\[
p = \vect j\cdot\vect E = \gamma E^2 = \frac{j^2}\gamma \ \ (\unit{W/m^3}) ,
\]
seluruhnya berubah menjadi kalor oleh tumbukannya (energi kinetik hanyutnya
dapat diabaikan dan tetap). Diintegralkan atas sebuah kawat, $\int j^2/\gamma\,\dd\tau
= (L/\gamma S)I^2 = RI^2$.
\end{proposition}

\begin{proof}
Gaya $q\vect E$ pada tiap pengemban melakukan usaha dengan laju $q\vect E\cdot\vect v$;
dijumlahkan atas $n\,\dd\tau$ pengembannya: $nq\vect v\cdot\vect E\,\dd\tau = \vect j\cdot
\vect E\,\dd\tau$ (gaya magnetiknya tak melakukan usaha). Dalam keadaan tunak
Drude daya ini sama dengan yang dilesapkan suku gesekannya, yakni
dipindahkan ke kisinya.
\end{proof}

\begin{example}[Sebuah kawat yang memanas]\label{ex:b2:charges-currents-conduction:heating}
Kawat \qty{1.5}{mm^2} pada \qty{10}{A}: $p = j^2/\gamma = (6.7 \times 10^6)^2/6 \times 10^7 =
\qty{7.4e5}{W/m^3}$, yakni \qty{1.1}{W} tiap meter --- mudah dibuang udara
di sekelilingnya. Dibiarkan sendiri (tanpa pendinginan), tembaga ($c = \qty{385}{J/(kg.K)}$,
$\rho = \qty{8900}{kg/m^3}$) akan menghangat dengan $p/\rho c = \qty{0.2}{K/s}$. Pada \qty{100}{A}
dalam kawat yang sama lajunya seratus kali lebih besar, \qty{20}{K/s}: ia
mencair dalam semenit --- yaitu asas sekring, yang kawat tipisnya
dirancang agar mencair lebih dulu.
\end{example}

\begin{proposition}[Relaksasi muatan dalam penghantar]\label{prop:b2:charges-currents-conduction:relaxation}
\index{waktu relaksasi (penghantar)}
Dalam penghantar ohmik \omterm{def:b2:charges-currents-conduction:densities}{rapat muatan} volumenya meluruh sebagai $\rho(t) =
\rho_0\eu^{-t/\tau_r}$ dengan
\[
\tau_r = \frac{\varepsilon_0}{\gamma} :
\]
\qty{1.5e-19}{s} bagi tembaga, \qty{2e-12}{s} bagi air laut, \qty{10}{s} bagi kaca.
Sampai frekuensi berorde $1/\tau_r$ sebuah penghantar tak mengemban \emph{muatan
volume} sama sekali: kelebihan muatan mana pun berlari ke permukaannya. Kata
``penghantar'' karena itu bergantung pada skala waktunya: air laut adalah penghantar
bagi gelombang radio dan isolator bagi cahaya tampak.
\end{proposition}

\begin{proof}
Kekekalan muatan dengan $\vect j = \gamma\vect E$ dan hukum Gauss dalam bentuk
lokal, $\operatorname{div}\vect E = \rho/\varepsilon_0$ (\cref{ch:b2:maxwell-equations}):
$\partial_t\rho = -\gamma\operatorname{div}\vect E = -(\gamma/\varepsilon_0)\rho$.
\end{proof}

\begin{method}[Dari hukum lokal ke hukum integral]\label{met:b2:charges-currents-conduction:method}
Untuk mencari hambatan penghantar berbentuk sembarang: (1) pakailah
simetrinya untuk menuliskan $\vect j$ (tunak: fluksnya kekal, jadi dalam
sebuah tabung $j \times$ penampangnya $= I$); (2) $\vect E = \vect j/\gamma$; (3) integralkanlah
$\vect E$ sepanjang sebuah garis dari satu elektrode ke yang lain: $U = \int\vect E\cdot
\dd\vect l$, dan $R = U/I$; (4) periksalah dengan daya Joulenya $\int j^2/\gamma\,
\dd\tau = RI^2$. Langkah yang sama memberi kebocoran sebuah kabel, hambatan
tanah di sekeliling sebuah penangkal petir, atau hambatan sebuah sentuhan kerucut.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:charges-currents-conduction:4}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:5}
Sebuah kapasitor berkeping $S$ diisi lewat sebuah kawat oleh $I(t)$. (a) Terapkanlah
hukum kekekalan integralnya pada permukaan tertutup yang melingkupi satu keping:
kaitkanlah $I$ dan muatan kepingnya $Q$. (b) Mengapa $\operatorname{div}\vect j \ne 0$
di kepingnya? (c) Sebuah elektrode bola berjari-jari $a$ dalam medium
penghantar memancarkan arus radial $I$: tentukanlah $\vect j(r)$; periksalah $\operatorname{div}
\vect j = 0$ untuk $r > a$ (pakailah $\operatorname{div}\vect j = \frac1{r^2}\partial_r(r^2j_r)$
bagi medan radial). (d) Di manakah, kalau begitu, kekekalan muatannya
``patah''?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:6}
\emph{Kebocoran sebuah \omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial}.} Isolator di antara kedua penghantarnya
(jari-jari $a = \qty{1}{mm}$, $b = \qty{3.5}{mm}$) mempunyai $\gamma = \qty{1e-13}{S/m}$; sebuah
tegangan $U$ dikenakan di antara keduanya, sebuah arus bocor radial $I$ mengalir
tiap panjang $L$. (a) $\vect j(r)$ dan $\vect E(r)$. (b) Tunjukkanlah bahwa
hambatan sepanjang $L$ adalah $R = \ln(b/a)/2\pi\gamma L$; nilainya tiap
kilometer. (c) Arus bocor dan dayanya tiap kilometer pada \qty{1}{kV}.
(d) Bandingkanlah dengan kapasitansi tiap panjangnya $2\pi\varepsilon_0\varepsilon_r/\ln(b/a)$:
tunjukkanlah $RC = \varepsilon_0\varepsilon_r/\gamma$ lalu tafsirkanlah dengan waktu relaksasinya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:7}
\omterm{prop:b2:charges-currents-conduction:acdrude}{Konduktivitas kompleks} tembaga ($\tau = \qty{2.5e-14}{s}$). (a) $\omega\tau$ pada
\qty{50}{Hz}, \qty{10}{GHz} (gelombang mikro), \qty{30}{THz} (inframerah), \qty{5e14}{Hz}
(cahaya hijau). (b) Modulus dan fase $\underline\gamma/\gamma_0$ dalam tiap halnya.
(c) Sampai frekuensi berapa hukum Ohm baik sampai $1\%$? (d) Dalam cahaya tampak,
tuliskanlah $\underline\gamma$ lalu tunjukkanlah bahwa $\underline{\vect j}$ berada dalam kuadratur dengan
$\underline{\vect E}$: berapa daya Joule reratanya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:8}
Sebuah kawat sekring tembaga \qty{0.20}{mm} (tembaga: $\gamma = \qty{6.0e7}{S/m}$, $\rho =
\qty{8900}{kg/m^3}$, $c = \qty{385}{J/(kg.K)}$, mencair pada \qty{1085}{\degreeCelsius},

kalor yang lolos (abaikanlah kenaikan hambatan jenisnya dengan suhu). (c)
Hal yang sama pada \qty{30}{A}. (d) Kenyataannya kawatnya kehilangan kalor ke udara: bagaimana
itu menetapkan arus terala sebuah sekring, dan mengapa sekring dikemas dalam
selongsong berisi pasir?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:charges-currents-conduction:9}
\emph{Relaksasi.} (a) Turunkanlah $\partial_t\rho = -(\gamma/\varepsilon_0)\rho$ dari kekekalan
muatan, hukum Ohm dan $\operatorname{div}\vect E = \rho/\varepsilon_0$. (b)
Tentukanlah $\tau_r$ bagi tembaga, air laut ($\gamma = \qty{5}{S/m}$), air keran
(\qty{5e-2}{S/m}), kaca (\qty{1e-12}{S/m}), PTFE (\qty{1e-16}{S/m}). (c)
Sampai frekuensi berapa masing-masingnya ``penghantar yang baik'' (muatannya berelaksasi dalam
satu periode)? (d) Sebatang kaca bermuatan menahan muatannya bermenit-menit: selarask ah
itu?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:10}
\emph{Pembumian.} Sebuah elektrode setengah bola berjari-jari $a = \qty{0.50}{m}$
terkubur rata dengan permukaan tanah berkonduktivitas $\gamma =
\qty{0.010}{S/m}$; sebuah arus $I$ memasukinya lalu menyebar secara radial ke
separuh ruangnya. (a) $\vect j(r)$, $\vect E(r)$, potensial $V(r)$ dengan $V(\infty) = 0$.
(b) Hambatan pembumiannya, $R = 1/2\pi\gamma a$; nilainya. (c) Sebuah
sambaran petir \qty{20}{kA} menghantam batangnya: potensial batangnya;
potensial di $r = \qty{10}{m}$ dan \qty{10.7}{m}; ``tegangan langkah'' antara
kedua kaki seseorang yang berdiri di sana (kakinya terpisah \qty{0.7}{m}, secara radial).
(d) Mengapa ternak, yang keempat kakinya berjauhan, lebih sering terbunuh sambaran
di dekatnya daripada manusia?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:11}
\emph{Pengindra Hall.} Sebuah pita setebal $t$ dan selebar $w$ mengemban $I$
sepanjang $x$ dalam medan $B$ sepanjang $z$. (a) Dari $\vect E_H = -\vect j\wedge\vect B
/nq$, tunjukkanlah bahwa tegangan Hall menyeberangi lebarnya adalah $V_H = IB/nqt$. (b)
Pita tembaga, $t = \qty{0.10}{mm}$, \qty{1}{A}, \qty{1}{T}: berapa $V_H$? (c) Sebuah
semikonduktor terdoping dengan $n = \qty{1e22}{m^{-3}}$, geometri sama: $V_H$; mengapa
pengindra memakai semikonduktor. (d) Sebuah pengindra dengan $n = \qty{1e22}{m^{-3}}$, $t =
\qty{20}{\micro m}$, yang disuapi \qty{1}{mA}, membaca \qty{10}{\micro V}: medan yang terukur.
Apa yang diberitahukan tanda $V_H$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:charges-currents-conduction:12}
\emph{Elektrolit.} Dalam air, ion Na$^+$ dan Cl$^-$ mempunyai mobilitas
$\mu_+ = \qty{5.2e-8}{m^2/(V.s)}$ dan $\mu_- = \qty{7.9e-8}{m^2/(V.s)}$. (a)
Tunjukkanlah bahwa $\gamma = ne(\mu_+ + \mu_-)$ bagi $n$ pasang ion tiap satuan volume.
(b) Air laut, \qty{0.5}{mol/L} NaCl: berapa $\gamma$? Bandingkanlah dengan
\qty{5}{S/m} yang terukur. (c) Kelajuan hanyutnya pada \qty{10}{V/m}; ion mana yang mengemban lebih banyak
arusnya? (d) Sebuah kolom air laut, \qty{1}{cm^2} kali \qty{10}{cm}, mengemban
\qty{1}{A}: tegangan, daya, dan waktu untuk menghangatkannya \qty{10}{K} ($c =
\qty{4.0}{kJ/(kg.K)}$); mengapa gel elektroforesis didinginkan.
\end{exercise}

\section{Soal: Kawat tembaga, dari atom sampai jaringan listrik}

\begin{problem}[{Soal akhir pekan --- satu meter kawat rumah tangga yang diperiksa
pada empat skala: elektronnya, hukum lokalnya, rangkaiannya, dan
saluran dayanya}]
\label{pb:b2:charges-currents-conduction:1}

Tembaga: massa molar \qty{63.5}{g/mol}, massa jenis \qty{8900}{kg/m^3}, satu
elektron hantaran tiap atom, konduktivitas $\gamma = \qty{6.0e7}{S/m}$ pada
\qty{20}{\degreeCelsius}, \omterm{prop:b2:charges-currents-conduction:drude}{hambatan jenis} yang naik $\rho_e(T) = \rho_e(20)[1 +
\alpha(T - 20)]$ dengan $\alpha = \qty{4.0e-3}{K^{-1}}$; $c = \qty{385}{J/(kg.K)}$. Kawat:
penampang $S = \qty{2.5}{mm^2}$, panjang \qty{1}{m}, arus \qty{16}{A}.

\medskip
\noindent\textbf{Bagian I --- Elektronnya.}
\begin{enumerate}
  \item Kerapatan jumlah $n$ elektron hantarannya.
  \item \omterm{def:b2:charges-currents-conduction:densities}{Rapat arus} dan \omterm{ex:b2:charges-currents-conduction:drift}{kecepatan hanyutnya}; berapa lama sebuah elektron
        perlu untuk menyeberangi satu meternya.
  \item $\tau$ Drude dan mobilitasnya; jalan bebas rata-rata dengan kelajuan
        acak \qty{1.6e6}{m/s}; berapa jarak atom (\qty{0.26}{nm})
        nilai itu, dan apa katanya tentang dengan apa elektronnya
        bertumbukan?
  \item Getaran termalnya tumbuh dengan $T$, jadi kira-kira $\tau \propto 1/T$:
        periksalah terhadap $\alpha$ dekat suhu ruang ($\dd\rho_e/\rho_e =
        \dd T/T$ akan memberi $\alpha = 1/293$).
  \item Medan listrik dalam kawatnya; tegangan total sepanjang meternya;
        energi kinetik hanyutnya tiap elektron dibandingkan dengan $k_BT$ ---
        terganggukah gas elektronnya oleh arusnya?
  \item Jumlah elektron yang menyeberangi sebuah penampang kawatnya tiap sekon.
  \item Muatan elektron hantaran dalam satu meter kawatnya; gaya
        yang dikerahkan medannya pada semuanya sekaligus; bandingkanlah dengan
        berat kawatnya. Ke mana gaya itu berakhir (pikirkanlah
        tumbukannya)?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Hukum lokalnya.}
\begin{enumerate}[resume]
  \item Daya Joule tiap satuan volume dan tiap meter.
  \item Kawatnya, di udara, kehilangan kalor dengan laju $h(T - T_a)$ tiap satuan
        permukaan dengan $h = \qty{10}{W/(m^2.K)}$ (konveksi): kenaikan suhu
        setimbangnya; periksalah apakah koreksi hambatan jenisnya
        berarti.
  \item Kawat yang sama terkubur dalam penyekat yang hanya melewatkan $h =
        \qty{2}{W/(m^2.K)}$: kenaikan suhu barunya, dengan koreksi
        hambatan jenisnya disertakan (selesaikanlah neracanya).
  \item Dibiarkan tanpa pendinginan apa pun, laju kenaikan suhunya pada \qty{16}{A}
        dan pada \qty{160}{A} (sebuah korsleting); waktu untuk mencapai titik
        leburnya (\qty{1085}{\degreeCelsius}) dalam hal yang kedua; apa yang melindungi
        instalasinya?
  \item Pada \qty{50}{Hz}, $\omega\tau = ?$ Terpengaruhkah hukum Ohmnya? (Efek
        kulit \cref{ch:b2:waves-in-media} adalah perkara lain.)
  \item Tuliskanlah waktu relaksasi $\varepsilon_0/\gamma$ tembaganya; apa siratnya
        bagi \omterm{def:b2:charges-currents-conduction:densities}{rapat muatan} di dalam kawatnya, dan di manakah
        muatan yang menghasilkan medan $E$ di dalam kawatnya berada?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Kekekalan muatan dalam rangkaiannya.}
Kawatnya menyuapi sebuah kapasitor $C = \qty{100}{\micro F}$ lewat sebuah sakelar;
arusnya $I(t) = I_0\eu^{-t/\tau_c}$ dengan $I_0 = \qty{16}{A}$, $\tau_c = \qty{1}{ms}$.
\begin{enumerate}[resume]
  \item Muatan yang mencapai kepingnya; tegangan akhirnya.
  \item Terapkanlah $\dd Q/\dd t = -I_{\text{luar}}$ pada permukaan tertutup di sekeliling
        kepingnya: sisi mana yang mengemban arus hantaran, sisi mana yang tidak?
        Besaran apa karena itu yang harus berubah dalam celahnya?
  \item Pada $t = \qty{0.5}{ms}$, laju perubahan muatan kepingnya; jika
        kepingnya seluas \qty{1}{cm^2} dan berjarak \qty{0.1}{mm} di udara, laju
        perubahan medan di antaranya.
  \item Tunjukkanlah bahwa $\varepsilon_0\,\dd E/\dd t \times$ luas kepingnya sama dengan arus
        dalam kawatnya --- yaitu besaran yang ditambahkan
        \cref{ch:b2:maxwell-equations} pada hukum Ampère.
  \item Kawatnya dipotong dan kedua ujungnya berjarak \qty{1}{mm} di udara:
        arus berapa yang mengalir (udara $\gamma \approx \qty{1e-14}{S/m}$)? Pada
        medan berapa udaranya tembus listrik (\qty{3}{MV/m}): tegangan berapa menyeberangi
        celahnya yang membuat sebuah bunga api?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Salurannya.}
Sebuah saluran \qty{230}{V} dari tembaga \qty{10}{mm^2}, sepanjang \qty{500}{m} (pergi dan pulang:
\qty{1}{km} kawat), menyuapi sebuah peternakan yang menarik \qty{40}{A}.
\begin{enumerate}[resume]
  \item Hambatan, jatuh tegangan, dan daya yang hilang dalam salurannya; bagian
        daya yang terserahkan.
  \item Daya yang sama pada \qty{20}{kV} dengan sebuah transformator di tiap ujungnya:
        arus, jatuh tegangan dan ruginya; simpulkanlah mengapa penyaluran dilakukan pada
        tegangan tinggi.
  \item Aluminium ($\gamma = \qty{3.6e7}{S/m}$, \qty{2700}{kg/m^3}) dipakai pada
        menara: penampang bagi hambatan yang sama dengan \qty{10}{mm^2}
        tembaga; massa tiap kilometer bagi keduanya; mengapa aluminium menang di
        atas sana dan tembaga menang di dalam dinding.
  \item Sebuah pengindra Hall yang dijepitkan mengelilingi salurannya membaca medannya: bagi
        sebuah tang ampere dengan $n = \qty{1e22}{m^{-3}}$, $t = \qty{0.1}{mm}$, yang disuapi
        \qty{5}{mA} dalam medan \qty{0.1}{T} yang dipusatkan sebuah teras besi,
        berapa tegangan Hallnya.
  \item Sesudah beban seharian yang panas kawatnya berada pada \qty{80}{\degreeCelsius}:
        hambatannya ketika itu, dan rugi tambahannya.
  \item Petir menghantam salurannya: \qty{20}{kA} selama \qty{100}{\micro s}. Energi
        yang dilesapkan dalam \qty{1}{km} kawatnya; kenaikan suhunya jika tak ada yang
        lolos; rapat arusnya --- bandingkanlah dengan Bagian I.
  \item Rangkumlah: kelima besaran ($n$, $\tau$, $\gamma$, $j$, $E$) yang
        melukiskan kawatnya pada tiap skala, dan hukum yang mengaitkan tiap
        pasangnya.
\end{enumerate}
\end{problem}
