\chapter{Sumur Potensial, Penghalang dan Penerowongan}\label{ch:b2:potential-wells-tunneling}

Sebuah bola dalam sebuah mangkuk menggelinding bolak-balik dengan energi berapa pun yang
diberikan padanya; sebuah elektron dalam sebuah atom, sebuah nukleon dalam sebuah inti, sebuah elektron dalam
sebuah hablur seukuran nanometer hanya dapat mempunyai energi tertentu, dan yang terendah
di antaranya bukan nol. Sebuah bola tak dapat menyeberangi sebuah bukit yang lebih tinggi daripada yang dibolehkan
energinya; sebuah partikel $\alpha$ meninggalkan sebuah inti menembus sebuah dinding Coulomb yang
tak akan pernah dapat didakinya, jarum sebuah mikroskop terowongan payar menggambar
sebuah arus menyeberangi sebuah celah hampa, dan atom nitrogen amonia
berayun menembus bidang ketiga hidrogennya dua puluh empat miliar
kali tiap sekon. Bab ini menyelesaikan persamaan Schrödinger dalam
potensial yang paling sederhana --- sumur berdinding takhingga dan berhingga, sebuah tangga, sebuah
penghalang, sebuah sumur ganda --- lalu menemukan di dalamnya dua kenyataan kuantum besar
yang tak dapat dihasilkan mekanika klasik: \emph{kuantisasi}
energi terikatnya, dan \emph{penerowongan} menembus daerah yang terlarang secara klasik,
dengan kepekaan eksponensialnya terhadap lebar dan tinggi
penghalangnya, itulah sebabnya ia dapat melayani sebagai penggaris pikometer dan itulah sebabnya
umur radioaktif merentangi tiga puluh orde besaran.

\section{Sumur takhingga}

\begin{proposition}[Sumur segi empat takhingga]\label{prop:b2:potential-wells-tunneling:infinite}
Sebuah partikel yang terkurung dalam $0 < x < L$ oleh dinding yang tak tertembus ($V = 0$
di dalamnya, $V = \infty$ di luarnya, sehingga $\varphi = 0$ di dindingnya) mempunyai
\omterm{thm:b2:schrodinger-wave-functions:stationary}{keadaan tegar} dan energi
\[
\varphi_n(x) = \sqrt{\frac{2}{L}}\sin\frac{n\pi x}{L}, \qquad
E_n = \frac{n^2h^2}{8mL^2} = n^2E_1, \qquad n = 1, 2, 3, \dots
\]
Energinya \emph{terkuantumkan}\index{kuantisasi energi}, tumbuh
sebagai $n^2$, dan yang terendahnya bukan nol: \emph{energi
pengurungan}\index{energi pengurungan} $E_1 = h^2/8mL^2$ adalah harga
pelokalannya ($\Delta x \sim L$ memaksa $\Delta p \sim h/2L$). Fungsi
$\varphi_n$-nya saling ortogonal, $\int\varphi_n\varphi_m\dd x = \delta_{nm}$, dan sebarang
keadaan sumurnya adalah sebuah tumpang tindih darinya.
\end{proposition}

\begin{proof}
Di dalamnya, $\varphi'' = -k^2\varphi$ dengan $k^2 = 2mE/\hbar^2$: $\varphi = A\sin kx
+ B\cos kx$; $\varphi(0) = 0$ mematikan $B$, $\varphi(L) = 0$ menuntut $kL = n\pi$
--- yaitu \omterm{prop:b2:waves-on-strings:modes}{gelombang berdiri} sebuah dawai, dengan $E = \hbar^2k^2/2m$.
Penormalannya memberi $A = \sqrt{2/L}$; keortogonalannya adalah keortogonalan
sinusnya.
\end{proof}

\begin{example}[Orde besaran, dan warna]\label{ex:b2:potential-wells-tunneling:orders}
$E_1 = h^2/8mL^2$: sebuah elektron dalam \qty{1}{nm}, \qty{0.38}{eV}; dalam
\qty{0.1}{nm} (sebuah atom), \qty{38}{eV} --- yaitu skala energi atomik; sebuah
nukleon dalam \qty{5}{fm}, \qty{8}{MeV} --- yaitu skala energi inti; sebuah
kelereng dalam sebuah kotak, $10^{-63}\,\unit{J}$ --- tak seorang pun akan memperhatikannya. Sebuah
nanohablur semikonduktor selebar beberapa nanometer (sebuah \emph{titik
kuantum}) mengurung elektronnya: \omterm{prop:b2:potential-wells-tunneling:infinite}{energi pengurungannya} menambahi
senjang hablurnya, sehingga makin kecil titiknya, makin biru cahaya yang dipancarkannya ---
bahan yang sama berpijar merah pada \qty{6}{nm} dan hijau pada \qty{3}{nm},
yang ditala oleh ukurannya saja.
\end{example}

\section{Sumur berhingga}

\begin{proposition}[Sumur segi empat berhingga]\label{prop:b2:potential-wells-tunneling:finite}
Bagi $V = 0$ dalam $|x| < a$ dan $V = V_0$ di luarnya, sebuah keadaan ber-$0 < E < V_0$
berayun di dalamnya ($k = \sqrt{2mE}/\hbar$) lalu meluruh di luarnya ($\kappa =
\sqrt{2m(V_0 - E)}/\hbar$): $\varphi \propto \eu^{-\kappa|x|}$ di luar dindingnya.
Menyesuaikan $\varphi$ dan $\varphi'$ di $x = \pm a$ memberi, bagi keadaan genap dan
ganjilnya,
\[
k\tan ka = \kappa \quad\text{(genap)}, \qquad
-k\cot ka = \kappa \quad\text{(ganjil)}, \qquad\text{dan}\quad
(ka)^2 + (\kappa a)^2 = \frac{2mV_0a^2}{\hbar^2} \equiv R^2 .
\]
Secara grafis, penyelesaiannya adalah perpotongan kurva $\eta =
\xi\tan\xi$, $\eta = -\xi\cot\xi$ dengan lingkaran $\xi^2 + \eta^2 = R^2$
($\xi = ka$, $\eta = \kappa a$): \emph{selalu ada sekurangnya satu keadaan
terikat} (yaitu keadaan dasar yang genap), dan seluruhnya $1 + \lfloor 2R/\pi\rfloor$.
Partikelnya \emph{menembus} daerah terlarangnya sedalam
$1/\kappa$, dan arasnya terletak di bawah aras sumur takhingga yang
selebar sama.
\end{proposition}

\begin{proof}
Penyelesaian genapnya adalah $A\cos kx$ di dalamnya dan $B\eu^{-\kappa|x|}$ di luarnya;
kesinambungan $\varphi'/\varphi$ di $x = a$ memberi $-k\tan ka = -\kappa$. Sama
bagi yang ganjil dengan $\sin$. Lingkarannya adalah definisi $k$ dan
$\kappa$. Tiap cabang $\xi\tan\xi$ yang bermula di $\xi = n\pi$ dan
$-\xi\cot\xi$ yang bermula di $(n + \tfrac12)\pi$ menjumpai lingkarannya sekali jika ia
bermula di dalamnya: maka cacahnya.
\end{proof}

\begin{example}[Sebuah elektron dalam sumur senanometer]\label{ex:b2:potential-wells-tunneling:finite}
$V_0 = \qty{1}{eV}$, lebarnya $2a = \qty{1}{nm}$: $R = a\sqrt{2mV_0}/\hbar = 0.5 \times
10^{-9} \times 5.1 \times 10^9 = 2.6$, sehingga ada $1 + \lfloor 2R/\pi\rfloor = 2$ keadaan
terikat; keadaan dasarnya terletak dekat \qty{0.26}{eV} alih-alih \qty{0.38}{eV}
sumur takhingganya, lalu merembes \qty{0.2}{nm} ke dalam dindingnya
($1/\kappa = \hbar/\sqrt{2m(V_0 - E)}$). Sumur semacam itu, yang ditumbuhkan sebagai lapisan
semikonduktor setebal beberapa nanometer, adalah jantung dioda
laser \cref{ch:b2:laser}.
\end{example}

\section{Tangga dan penghalang: penerowongan}

\begin{proposition}[Tangga potensial]\label{prop:b2:potential-wells-tunneling:step}
Sebuah partikel berenergi $E$ yang datang dari $x < 0$ ke tangga $V = V_0$
bagi $x > 0$:
\begin{itemize}
\item $E > V_0$: $\varphi = \eu^{\iu k_1x} + r\eu^{-\iu k_1x}$ di kirinya, $t\eu^{\iu
      k_2x}$ di kanannya, dengan $k_{1,2} = \sqrt{2m(E - V_{0,\text{kiri/kanan}})}/
      \hbar$; kesinambungan $\varphi$ dan $\varphi'$ memberi $r = (k_1 - k_2)/(k_1
      + k_2)$ dan peluang pemantulannya (nisbah arusnya) $R = r^2$,
      $T = 1 - R = 4k_1k_2/(k_1 + k_2)^2$ --- sebuah partikel \emph{dapat
      terpantul oleh sebuah tangga yang energinya cukup untuk didaki}, tepat seperti sebuah gelombang pada
      sebuah dawai oleh sebuah perubahan impedansi (\cref{ch:b2:wave-interfaces});
\item $E < V_0$: di kanannya $\varphi = t\eu^{-\kappa x}$, $\kappa = \sqrt{2m(V_0
      - E)}/\hbar$ --- sebuah \emph{gelombang evanesen}\index{gelombang evanesen
      (kuantum)}; $|r| = 1$ (pemantulan total, dengan sebuah geseran fase), tak ada
      arus yang mengalir ke dalam tangganya, tetapi partikelnya ditemukan di sana dengan sebuah
      peluang yang meluruh sepanjang $1/\kappa$.
\end{itemize}
\end{proposition}

\begin{proof}
Tuliskanlah persamaan kontinuitasnya di $x = 0$ lalu selesaikanlah; bagi arusnya
pakailah $j = (|A|^2 - |B|^2)\hbar k/m$ (\cref{ch:b2:schrodinger-wave-functions}).
Bagi $E < V_0$, $r = (k - \iu\kappa)/(k + \iu\kappa)$, yang bermodulus satu.
\end{proof}

\begin{theorem}[Penerowongan menembus sebuah penghalang]\label{thm:b2:potential-wells-tunneling:tunnel}
Sebuah partikel berenergi $E < V_0$ yang menjumpai sebuah penghalang segi empat setinggi
$V_0$ dan selebar $a$ diteruskan dengan peluang
\[
T = \Big[1 + \frac{V_0^2\sinh^2(\kappa a)}{4E(V_0 - E)}\Big]^{-1}
\ \approx\ 16\,\frac{E}{V_0}\Big(1 - \frac{E}{V_0}\Big)\eu^{-2\kappa a}
\quad (\kappa a \gg 1), \qquad
\kappa = \frac{\sqrt{2m(V_0 - E)}}{\hbar} .
\]
Inilah \emph{efek terowongan}\index{efek terowongan}: yang mustahil
secara klasik, dan secara kuantum kecil secara eksponensial --- serta
peka secara eksponensial terhadap lebarnya $a$, tingginya $V_0 - E$ dan
massanya $m$. Bagi sebuah elektron \qty{1}{eV} di bawah puncaknya, $\kappa = \qty{5.1}{nm^{-1}}$:
$T \sim 10^{-2}$ bagi $a = \qty{0.5}{nm}$, $10^{-4}$ bagi \qty{1}{nm}, $10^{-9}$ bagi
\qty{2}{nm}; bagi sebuah proton, $\kappa$ menjadi $43$ kali lebih besar dan tak ada yang lolos
pada lebar ini.
\end{theorem}

\begin{proof}
$\varphi = \eu^{\iu kx} + r\eu^{-\iu kx}$ bagi $x < 0$, $A\eu^{\kappa x} + B\eu^{-\kappa x}$
di dalamnya, $t\eu^{\iu kx}$ bagi $x > a$; empat syarat kesinambungan (bagi $\varphi$
dan $\varphi'$ di $0$ dan $a$) bagi empat yang tak diketahui; menyingkirkan $A$, $B$, $r$
memberi $1/|t|^2 = 1 + (k^2 + \kappa^2)^2\sinh^2(\kappa a)/4k^2\kappa^2$, yang merupakan
rumusnya dengan $k^2 = 2mE/\hbar^2$, $\kappa^2 = 2m(V_0 - E)/\hbar^2$. Bagi
$\kappa a \gg 1$, $\sinh\kappa a \approx \eu^{\kappa a}/2$. Bagi sebuah penghalang berbentuk
sembarang eksponennya menjadi $2\int\kappa(x)\dd x$ menyeberangi
daerah terlarangnya (yaitu faktor Gamow, yang diterima tanpa bukti).
\end{proof}

\begin{example}[Mikroskop terowongan payar]\label{ex:b2:potential-wells-tunneling:stm}
Sebuah ujung logam yang tajam didekatkan sampai $d \approx \qty{0.5}{nm}$ dari sebuah
permukaan yang menghantar lalu sebuah tegangan kecil diterapkan: elektronnya menerowong menyeberangi
celah hampanya, yang tinggi penghalangnya adalah fungsi kerjanya $\Phi \approx
\qty{4.5}{eV}$, sehingga $\kappa = \sqrt{2m\Phi}/\hbar = \qty{1.1e10}{m^{-1}}$ dan
arusnya $I \propto \eu^{-2\kappa d}$ berubah sebesar faktor $\eu^{2.2} \approx
10$ tiap \qty{0.1}{nm}. Sebuah simpal \omterm{def:b2:feedback-oscillators:loop}{umpan balik} menggerakkan ujungnya untuk menjaga $I$
tetap, dan tinggi ujungnya, yang direkam ketika ia memayar, menggambar permukaannya
dengan resolusi tegak sepikometer --- dan secara mendatar atom demi
atom, karena atom terakhir ujungnya, yang paling dekat, mengemban sebagian besar
arusnya (Binnig dan Rohrer, 1981).
\end{example}

\begin{example}[Peluruhan alfa]\label{ex:b2:potential-wells-tunneling:alpha}
Sebuah partikel $\alpha$ \qty{5}{MeV} di dalam sebuah inti berat menghadapi
penghalang Coulomb muatan sisanya $Ze$, yang berpuluh MeV tingginya di
permukaan intinya dan terentang sampai jari-jari $b = 2Ze^2/4\pi\varepsilon_0E
\approx \qty{50}{fm}$ yang di sana $V = E$. Faktor Gamownya $2\int\kappa\,\dd r$
berorde $80$: $T \sim \eu^{-80} \sim 10^{-35}$. Partikelnya menumbuk dindingnya
sekitar $10^{21}$ kali tiap sekon, sehingga ia lolos dengan laju $\sim 10^{-14}
\,\unit{s^{-1}}$: yaitu umur sejuta tahun. Karena eksponennya
berubah sebagai $Z/\sqrt E$, sebuah perubahan $\qty{2}{MeV}$ pada $E$ menggeser umurnya
sebesar dua puluh orde besaran --- yaitu hukum Geiger--Nuttall, dari
mikrosekon sampai umur alam semesta (Gamow, 1928: yaitu penerapan
pertama penerowongan).
\end{example}

\section{Sumur ganda}

\begin{proposition}[Pembelahan terowongan]\label{prop:b2:potential-wells-tunneling:double}
Dua sumur serupa yang dipisahkan sebuah penghalang: jika penghalangnya
tak tertembus tiap sumurnya akan mempunyai aras dasar yang sama $E_0$, dua kali.
Penerowongan mengopel keduanya dan kedua keadaan terendah sumur gandanya
adalah gabungan \emph{setangkup} dan \emph{taksetangkup},
$\varphi_\pm \approx (\varphi_{\text{L}} \pm \varphi_{\text{R}})/\sqrt2$, berenergi
$E_0 \mp \Delta/2$: arasnya \emph{terbelah} sebesar $\Delta \propto
\eu^{-\kappa a}$ (yaitu amplitudo penerowongannya, bukan kuadratnya). Sebuah partikel
yang ditaruh dalam sumur kirinya, $\psi = (\varphi_+ + \varphi_-)/\sqrt2$, tidak
tegar: ia berayun antara sumurnya pada frekuensi
$\nu = \Delta/h$ --- yaitu \emph{ayunan penerowongan}, yaitu layangan dua aras
\cref{ch:b2:schrodinger-wave-functions}.
\end{proposition}

\begin{proof}
Gabungan setangkupnya tak mempunyai simpul di bawah penghalangnya dan berkelengkungan
lebih rendah, maka berenergi lebih rendah; yang taksetangkup mempunyai sebuah simpul lalu
terletak lebih tinggi; pembelahannya sebanding dengan tumpang tindih fungsi
yang terlokalkan di bawah penghalangnya, $\eu^{-\kappa a}$. Layangannya adalah
tumpang tindih dua aras yang sudah dihitung.
\end{proof}

menerowong bolak-balik pada frekuensi $\Delta/h$.}
\end{omfigure}

\begin{example}[Amonia]\label{ex:b2:potential-wells-tunneling:ammonia}
Dalam NH$_3$ atom nitrogennya duduk pada satu sisi bidang ketiga
hidrogennya atau pada sisi lainnya --- yaitu sebuah sumur ganda berpenghalang sekitar
\qty{0.25}{eV}. Pembelahan terowongan aras dasarnya adalah $\Delta =
\qty{1e-4}{eV}$: $\nu = \Delta/h = \qty{24}{GHz}$, yaitu frekuensi
\emph{inversi}-nya, yaitu transisi maser pertama dan jam atom
pertama (1949). Gantilah hidrogennya dengan deuterium dan molekul yang lebih berat itu
menerowong lebih sedikit: \qty{1.6}{GHz}. Dalam piramida yang lebih berat (PH$_3$, AsH$_3$)
pembelahannya runtuh sampai kilohertz dan di bawahnya: molekulnya tinggal pada
sisinya bertahun --- itulah sebabnya molekul kidal dan kinan ada
sama sekali, dan itulah sebabnya gula tak merasemat di rak.
\end{example}

{\small Charles Townes dengan maser pertama (1954): sebuah berkas molekul amonia, yang dipilah sebuah medan listrik ke dalam keadaan inversi atasnya, menguatkan transisi \qty{24}{GHz}-nya dalam sebuah rongga --- yaitu piranti pertama \omterm{def:b2:laser:processes}{pancaran terangsang}, enam tahun sebelum lasernya.}
\end{omfigure}

\begin{method}[Sumur dan penghalang]\label{meth:b2:potential-wells-tunneling:method}
(1) Tuliskanlah $\varphi$ daerah demi daerah: $\eu^{\pm\iu kx}$ di tempat $E > V$,
$\eu^{\pm\kappa x}$ di tempat $E < V$, dengan $k, \kappa = \sqrt{2m|E - V|}/\hbar$.
(2) Sesuaikanlah $\varphi$ dan $\varphi'$ di tiap batasnya (pada sebuah dinding takhingga,
$\varphi = 0$). (3) Keadaan terikat: penyesuaiannya berpenyelesaian hanya bagi
$E$ yang diskret --- cacahlah secara grafis. (4) Hamburan: arusnya
$(|A|^2 - |B|^2)\hbar k/m$ memberi $R$ dan $T$. (5) Penerowongan: $T \sim
\eu^{-2\kappa a}$; orde besarannya dahulu, prafaktornya kemudian. (6)
Sumur ganda: pembelahannya $\propto\eu^{-\kappa a}$, layangannya pada $\Delta/h$.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:1}
Sumur takhingga: sebuah elektron dalam \qty{1}{nm} (tentukanlah $E_1$, $E_2$, $E_3$, panjang gelombang
foton $2 \to 1$-nya); sebuah elektron dalam \qty{0.1}{nm}; sebuah nukleon dalam \qty{5}{fm};
sebuah kelereng \qty{1}{g} dalam \qty{10}{cm} ($E_1$, dan bilangan kuantumnya bagi sebuah
kelajuan \qty{1}{cm/s}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:2}
Titik kuantum: sebuah nanohablur bersenjang \qty{1.74}{eV} mengurung sebuah
pasangan elektron--lubang yang \omterm{prop:b2:potential-wells-tunneling:infinite}{energi pengurungannya} $h^2/8m^*L^2$ dengan
$m^* = 0.1\,m_{\text{e}}$. Tentukanlah panjang gelombang pancarannya bagi $L = 6$, $4$, $3$,
\qty{2.5}{nm}; warna yang mana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:3}
Penerowongan: sebuah elektron ber-$V_0 - E = \qty{4.5}{eV}$: tentukanlah $\kappa$; $\eu^{-2\kappa a}$
bagi $a = 0.3$, $0.5$, $1$, \qty{2}{nm}; faktornya tiap \qty{0.1}{nm}; hal yang sama bagi
sebuah proton pada $a = \qty{0.3}{nm}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:4}
Tangga: sebuah elektron \qty{2}{eV} ke sebuah tangga \qty{1}{eV}: tentukanlah $k_1$, $k_2$,
$R$, $T$. Hal yang sama bagi sebuah tangga turun \qty{1}{eV}. Mengapa ini mustahil
secara klasik, dan apa analoginya bagi cahaya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:5}
\emph{Sumur takhingga secara rinci.} (a) Turunkanlah arasnya lalu normalkanlah.
(b) Tunjukkanlah keortogonalannya. (c) Bagi $n = 1$: tentukanlah $\langle x\rangle$, $\Delta x =
L\sqrt{1/12 - 1/2\pi^2}$, $\langle p\rangle = 0$, $\Delta p = \pi\hbar/L$; periksalah
Heisenbergnya. (d) Sketsalah $|\varphi_n|^2$ bagi $n$ yang besar lalu bandingkanlah dengan
peluang klasik menemukan sebuah partikel yang memantul-mantul.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:6}
\emph{Sumur berhingga.} (a) Turunkanlah $k\tan ka = \kappa$ bagi keadaan genapnya.
(b) Tunjukkanlah secara grafis bahwa selalu ada sebuah keadaan terikat lalu cacahlah bagi
$R = 1$, $4$, $10$. (c) Elektron, $V_0 = \qty{1}{eV}$, $2a = \qty{1}{nm}$:
tentukanlah $R$, jumlah keadaannya, kedalaman tembus keadaan dasarnya (ambillah
$E \approx \qty{0.26}{eV}$). (d) Apa yang terjadi pada jumlah keadaan terikatnya
ketika $V_0 \to \infty$, dan pada energinya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:7}
\emph{Tangga, $E < V_0$.} (a) Tuliskanlah $\varphi$ di kedua sisinya lalu carilah $r =
(k - \iu\kappa)/(k + \iu\kappa)$. (b) Tunjukkanlah $|r| = 1$ lalu hitunglah fase
$r$-nya. (c) Tunjukkanlah bahwa arusnya lenyap bagi $x > 0$ meskipun $|\varphi|^2 \ne
0$ di sana. (d) Elektron \qty{1}{eV} di bawah puncaknya: berapa kedalaman tembusnya;
bandingkanlah dengan \omterm{def:b2:dispersion-wave-packets:complex}{gelombang evanesen} \omterm{thm:b2:wave-interfaces:snell}{pemantulan dalam total}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:8}
\emph{Penghalangnya.} (a) Tuliskanlah keempat syarat kesinambungannya lalu
turunkanlah $T$. (b) Nilailah secara eksak dan dengan hampirannya bagi $E =
V_0/2$, $\kappa a = 3$; dan bagi $\kappa a = 1$. (c) Bagi $E > V_0$ tunjukkanlah $T = [1 +
V_0^2\sin^2(k_2a)/4E(E - V_0)]^{-1}$ lalu carilah energi penerusan
sempurnanya. (d) Tafsirkanlah (pikirkanlah \omterm{prop:b2:wave-interfaces:optics}{lapisan antipantul}
\cref{ch:b2:wave-interfaces}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:9}
\emph{Gamow.} Bagi penghalang Coulomb dari jari-jari intinya $R_{\text{n}}$
sampai $b = 2Ze^2/4\pi\varepsilon_0E$, eksponennya adalah $G = 2\int\kappa\,\dd r =
(2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b} - \sqrt{(R_{\text{n}}/b)(1 -
R_{\text{n}}/b)}]$. (a) $Z = 90$, $E = \qty{5}{MeV}$, $R_{\text{n}} = \qty{8}{fm}$:
tentukanlah $b$, $G$, $T$. (b) Frekuensi tumbukannya $v/2R_{\text{n}}$ dan umurnya. (c)
Ulangilah bagi $E = 4$ dan \qty{6}{MeV}: berapa nisbah umurnya. (d) Mengapa
inti memancarkan partikel $\alpha$ alih-alih proton atau $^{12}$C?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:10}
\emph{Amonia.} Pembelahannya $\Delta = \qty{1e-4}{eV}$. (a) Tentukanlah frekuensi inversi
dan periode ayunan penerowongannya. (b) Atom N-nya
disiapkan pada satu sisi: tuliskanlah $\psi(t)$ dan peluang menemukannya
pada sisi lainnya. (c) ND$_3$ menerowong pada \qty{1.6}{GHz}: turunkanlah $\kappa a$
dari nisbah pembelahannya jika $\Delta \propto \eu^{-\kappa a}$ dan $\kappa
\propto\sqrt m$ ($m_{\text{D}} = 2m_{\text{H}}$, penghalangnya tak berubah). (d) Mengapa
sebuah molekul kiral dapat tetap kidal bertahun?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:11}
\emph{STM-nya.} $\Phi = \qty{4.5}{eV}$, $d = \qty{0.5}{nm}$, panjarnya \qty{0.1}{V},
$I = \qty{1}{nA}$. (a) Tentukanlah $\kappa$, faktor $I$ tiap \qty{0.1}{nm},
resolusi tingginya jika $I$ terukur sampai $1\%$. (b) Elektron tiap sekon
dalam arusnya. (c) Atom puncak ujungnya \qty{0.25}{nm} lebih dekat daripada
tetangganya (yang \qty{0.25}{nm} jauhnya secara mendatar, sehingga $d' = \sqrt{d^2 + 0.25^2}$):
berapa bagian arus yang diembannya. (d) Sebuah kerangka baja \qty{1}{cm} memuai
sebesar $\alpha L\Delta T$ dengan $\alpha = \qty{1e-5}{K^{-1}}$: kemantapan suhu berapa
yang dituntut sepikometer, dan bagaimana rancangannya menyiasatinya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:12}
\emph{Sebuah laser sumur kuantum.} Sebuah lapisan GaAs \qty{10}{nm} (senjangnya
\qty{1.42}{eV}, $m_{\text{e}}^* = 0.067\,m_{\text{e}}$, $m_{\text{h}}^* = 0.45\,m_{\text{e}}$)
di antara penghalang yang \qty{0.3}{eV} lebih tinggi. (a) Tentukanlah \omterm{prop:b2:potential-wells-tunneling:infinite}{energi pengurungan} dasar
elektron dan lubangnya (sumur takhingga). (b) Energi foton dan panjang gelombang
lasernya. (c) Lebarnya bagi \qty{780}{nm}. (d) Dengan penghalang yang berhingga,
berapa banyak keadaan elektron terikat yang dimuat sumur \qty{10}{nm}-nya?
\end{exercise}

\section{Soal: STM, partikel alfa dan maser amonia}

\begin{problem}[{Soal akhir pekan --- tiga terowongan}]\label{pb:b2:potential-wells-tunneling:1}
Data: $\hbar = \qty{1.05e-34}{J.s}$, $h = \qty{6.63e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $m_\alpha = \qty{6.64e-27}{kg}$, $e^2/4\pi\varepsilon_0 =
\qty{1.44}{MeV.fm}$, \qty{1}{eV} $= \qty{1.60e-19}{J}$.

\textbf{Bagian I --- Mikroskop terowongan payarnya.}
Ujung dan cuplikannya dari wolfram ($\Phi = \qty{4.5}{eV}$); celahnya $d$; panjarnya
\qty{50}{mV}; arusnya adalah $I = I_0\eu^{-2\kappa d}$.
\begin{enumerate}
  \item Tentukanlah $\kappa$ bagi elektron pada aras Fermi yang menghadapi
        penghalang hampanya $\Phi$.
  \item Faktor perubahan $I$ bagi $\Delta d = \qty{0.1}{nm}$; bagi
        \qty{1}{pm}.
  \item \omterm{def:b2:feedback-oscillators:loop}{Umpan baliknya} menahan $I$ tetap sampai $1\%$: berapa derau tingginya.
  \item $I = \qty{1}{nA}$: berapa elektron tiap sekon; waktu rerata antara dua
        elektron, dibandingkan dengan ``waktu'' penerowongannya $\sim\hbar/\Phi$.
  \item Atom puncaknya dan tetangganya yang \qty{0.25}{nm} jauhnya secara mendatar,
        $d = \qty{0.5}{nm}$: berapa bagian atom puncaknya; mengapa satu atom saja
        cukup untuk mencitrakan atom.
  \item Sebuah atom teradsorpsi setinggi \qty{0.1}{nm}: berapa perubahan $I$ dalam
        ragam tinggi tetapnya; mengapa ragam arus tetapnya lebih disukai.
  \item Sebuah kerangka logam \qty{2}{cm}, $\alpha = \qty{1e-5}{K^{-1}}$: berapa perubahan
        suhu yang menggerakkan ujungnya sebesar \qty{1}{pm}; bagaimana instrumennya
        menyiasatinya (kesetangkupan, kecepatan, bahan bermuai rendah)?
  \item Dengan panjar \qty{1}{V} penghalangnya tak lagi segi empat:
        sketsalah lalu katakanlah apakah $I$ tumbuh lebih cepat atau lebih lambat daripada
        linear dengan $V$.
  \item Pada sebuah bercak teroksidasi fungsi kerjanya \qty{3}{eV} alih-alih
        \qty{4.5}{eV}: seberapa jauh ujungnya menarik diri dalam
        ragam arus tetap di atas sebuah permukaan yang rata sempurna? Lalu apa
        yang dicitrakan STM-nya?
\end{enumerate}

\textbf{Bagian II --- Peluruhan alfa sebuah inti $Z = 92$, $A = 238$.}
Partikel $\alpha$-nya ($E = \qty{4.2}{MeV}$) bergerak dalam sebuah inti berjari-jari
$R_{\text{n}} = \qty{8}{fm}$; di luarnya, energi Coulombnya adalah $V(r) =
2(Z - 2)e^2/4\pi\varepsilon_0r$.
\begin{enumerate}[resume]
  \item Tinggi penghalangnya di $R_{\text{n}}$; titik balik klasiknya
        $b$.
  \item Kelajuan $\alpha$-nya di dalamnya dan frekuensi tumbukannya pada
        dindingnya.
  \item Eksponen Gamownya $G = (2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b}
        - \sqrt{(R_{\text{n}}/b)(1 - R_{\text{n}}/b)}]$ dan $T = \eu^{-G}$.
  \item Laju peluruhan dan waktu paruhnya; bandingkanlah dengan $4.5 \times
        10^9$ tahun yang terukur.
  \item Hal yang sama dengan $E = \qty{5.2}{MeV}$ (sebuah isotop lain): berapa waktu paruhnya;
        dan kepekaan Geiger--Nuttallnya.
  \item Dengan mengambil $R_{\text{n}} = \qty{9}{fm}$ alih-alih \qty{8}{fm}: apa efeknya pada
        waktu paruhnya; ulaslah ketelitian taksiran semacam itu.
  \item Mengapa penerowongan sebuah proton (bermuatan $1$, bermassa $1/4$)
        tak teramati dari inti yang sama, dan begitu pula penerowongan sebuah $^{12}$C
        (bermuatan $6$)?
  \item Partikel $\alpha$-nya muncul dengan \qty{4.2}{MeV} penuh meskipun ia
        ``lewat di bawah'' sebuah dinding \qty{30}{MeV}: kekalkah energinya?
\end{enumerate}

\textbf{Bagian III --- Maser amonianya.}
Atom N-nya menerowong menembus bidang H$_3$-nya; aras dasarnya terbelah
sebesar $\Delta = \qty{9.8e-5}{eV}$.
\begin{enumerate}[resume]
  \item Frekuensi dan panjang gelombang transisi inversinya.
  \item Tuliskanlah keadaan setangkup dan taksetangkupnya dalam
        ``N atas'' dan ``N bawah''; yang mana yang lebih rendah, dan mengapa?
  \item Sebuah molekul yang disiapkan ``N atas'': tentukanlah $\psi(t)$, peluang
        ``bawah'' pada saat $t$, dan periodenya.
  \item Nisbah Boltzmann kedua arasnya pada \qty{300}{K}: dapatkah
        populasi termalnya memberi sebuah maser? Bagaimana pembalikannya diperoleh
        (sebuah medan listrik pemilah keadaan)?
  \item Energi fotonnya dalam joule; berapa molekul yang harus memancar
        tiap sekon bagi sebuah keluaran \qty{1e-10}{W}.
  \item ND$_3$ menerowong pada \qty{1.6}{GHz}: turunkanlah $\kappa a$ bagi NH$_3$
        dengan menganggap $\Delta \propto \eu^{-\kappa a}$ dengan $\kappa \propto \sqrt{m}$.
  \item Masernya adalah jam atom yang pertama: mengapa sebuah frekuensi
        penerowongan merupakan jam yang baik dan efek apa yang menggesernya?
  \item Ringkaslah: kuantisasi dari pengurungan, penerowongan dari
        keevanesenan, dan apa yang menetapkan eksponennya.
\end{enumerate}
\end{problem}
