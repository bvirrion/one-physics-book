\chapter{Interferensi Dua Gelombang}\label{ch:b2:two-wave-interference}

Sebuah gelembung sabun terbuat dari cairan yang tak berwarna, namun ia berkilau
berwarna; selaput minyak di jalan yang basah menunjukkan pita pelangi yang sama;
sayap seekor capung berkilat hijau dan lembayung. Tak satu pun warna ini
berasal dari pigmen: semuanya berasal dari dua salinan gelombang cahaya yang sama,
yang terpantul pada kedua muka sebuah selaput tipis, tiba dengan tunda lalu
saling menguatkan atau meniadakan --- yaitu \emph{interferensi}
dua gelombang. Young menjadikannya bukti bahwa cahaya adalah gelombang pada 1801,
dengan dua celah dan sebatang lilin; bab ini memaparkan rumusnya,
geometri rumbai Young, kedua syaratnya --- koherensi waktu dan
\omterm{prop:b2:two-wave-interference:spatial}{koherensi ruang} --- yang di bawahnya rumbainya bertahan, dan warna
selaput tipisnya.

\section{Rumus interferensinya}

\begin{theorem}[Interferensi dua gelombang]\label{thm:b2:two-wave-interference:formula}
\index{interferensi}\index{beda lintasan}\index{orde interferensi}\index{kontras (rumbai)}
Dua \omterm{def:b2:scalar-light-model:scalar}{gelombang monokromatik} koheren berintensitas $I_1$, $I_2$ dan berbeda
fase $\Delta\varphi$ di sebuah titik $M$ memberi di sana intensitas
\[
I(M) = I_1 + I_2 + 2\sqrt{I_1I_2}\,\cos\Delta\varphi(M) , \qquad
\Delta\varphi = \frac{2\pi}{\lambda_0}\,\delta(M) ,
\]
dengan $\delta$ yaitu \emph{\omterm{thm:b2:two-wave-interference:formula}{beda lintasan}} optik antara kedua
rute dari sumbernya ke $M$. Intensitasnya maksimal ($I_1 + I_2 +
2\sqrt{I_1I_2}$) di tempat $\delta = p\lambda_0$, dengan $p$ bulat --- yaitu \emph{\omterm{thm:b2:two-wave-interference:formula}{orde
interferensi}}-nya --- dan minimum ($I_1 + I_2 - 2\sqrt{I_1I_2}$) di tempat $\delta = (p +
\tfrac12)\lambda_0$. \emph{Kontras} (kenampakan) rumbainya adalah
\[
V = \frac{I_{\max} - I_{\min}}{I_{\max} + I_{\min}} = \frac{2\sqrt{I_1I_2}}{I_1 + I_2} ,
\]
yang sama dengan $1$ bagi intensitas yang sama, yang minimumnya gelap sempurna:
$I = 2I_0(1 + \cos\Delta\varphi) = 4I_0\cos^2(\Delta\varphi/2)$.
\end{theorem}

\begin{proof}
$s = a_1\cos(\omega t - \varphi_1) + a_2\cos(\omega t - \varphi_2)$; $\langle s^2\rangle$ seperti dalam
\cref{ch:b2:scalar-light-model}, dengan $\Delta\varphi$ yang tetap selama
pendeteksiannya; $I_i = \tfrac12Ka_i^2$. Beda fase dua gelombang
yang meninggalkan sumbernya bersama-sama lalu menempuh \omterm{def:b2:scalar-light-model:path}{lintasan optik} $L_1$, $L_2$ adalah
$2\pi(L_2 - L_1)/\lambda_0$ (ditambah $\pi$ bagi tiap pemantulan yang membalikkan
tandanya, \cref{ch:b2:wave-interfaces}).
\end{proof}

\begin{remark}[Ketiga syaratnya]\label{rem:b2:two-wave-interference:conditions}
\index{koherensi (ruang)}\index{koherensi (waktu)}
Rumbainya ada hanya jika (1) kedua gelombangnya berasal dari \emph{sumber
yang sama} (dua salinan satu getaran, jika tidak $\Delta\varphi$ acak); (2)
\omterm{thm:b2:two-wave-interference:formula}{beda lintasannya} lebih kecil daripada \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensinya}, $|\delta| < \ell_c$
(\emph{koherensi waktu}: di luarnya salinannya milik \omterm{prop:b2:scalar-light-model:coherence}{deret
gelombang} yang berbeda); (3) sumbernya cukup kecil sehingga titiknya yang berbeda
tak menghasilkan sistem rumbai tergeser yang saling menghapuskan
(\emph{\omterm{prop:b2:two-wave-interference:spatial}{koherensi ruang}}, di bawah). Dan syarat keempat, yang biasanya terpenuhi: kedua
gelombangnya tak boleh berpolarisasi tegak lurus (keduanya akan bertambah
dalam intensitas, seperti senternya).
\end{remark}

\section{Celah Young}

\begin{proposition}[Pembelahan muka gelombang: percobaan Young]\label{prop:b2:two-wave-interference:young}
\index{celah Young}\index{jarak antarrumbai}\index{pembelahan muka gelombang}
Sebuah sumber titik $S$ (atau sebuah celah) menyinari dua celah sempit yang sejajar
$S_1$, $S_2$ yang terpisah jarak $a$, yang berjarak sama dari $S$; sebuah layar berdiri pada
jarak $D \gg a$. Di titik layarnya pada ketinggian $x$ (sejajar
$a$) \omterm{thm:b2:two-wave-interference:formula}{beda lintasannya} adalah
\[
\delta = S_2M - S_1M \approx \frac{ax}D ,
\]
dan layarnya menunjukkan rumbai lurus yang berjarak sama, terang di $x_p =
p\lambda D/a$, dengan \emph{\omterm{prop:b2:two-wave-interference:young}{jarak antarrumbai}}
\[
i = \frac{\lambda D}a .
\]
Rumbainya ada di mana pun di belakang celahnya (keduanya \emph{tak
terlokalkan}); dengan sebuah lensa berjarak fokus $f$ sesudah celahnya, rumbainya
terlihat pada bidang fokusnya dengan $i = \lambda f/a$ (rumbai ``di tak hingga'').
Intensitasnya adalah $4I_0\cos^2(\pi ax/\lambda D)$.
\end{proposition}

\begin{proof}
$S_2M^2 - S_1M^2 = [(x + a/2)^2 - (x - a/2)^2] = 2ax$ dan $S_2M + S_1M \approx 2D$,
jadi $\delta \approx 2ax/2D$. Dua orde berselisih $\lambda$, yakni sebesar $\Delta x = \lambda D/a$.
Dengan sebuah lensa kedua gelombang yang meninggalkan celahnya pada sudut $\theta$ dari
sumbunya mempunyai \omterm{thm:b2:two-wave-interference:formula}{beda lintasan} $a\sin\theta \approx a\theta$ lalu berjumpa di $x = f\theta$.
\end{proof}

\begin{example}[Angka, dan cahaya putih]\label{ex:b2:two-wave-interference:numbers}
$a = \qty{0.5}{mm}$, $D = \qty{2}{m}$, $\lambda = \qty{600}{nm}$: $i = \qty{2.4}{mm}$ --- cukup
besar untuk dilihat, itulah sebabnya \omterm{prop:b2:two-wave-interference:young}{celah Young} harus berdekatan dan layarnya
jauh. Dengan cahaya putih tiap warna membuat rumbainya sendiri, semuanya
berimpit di tengahnya ($\delta = 0$): sebuah rumbai pusat yang putih, lalu
rumbai berwarna ketika ordenya menyimpang ($i$ merahnya $1.8$ kali
lembayungnya), dan sesudah beberapa orde sebuah medan keputihan yang seragam ---
\omterm{prop:b2:scalar-light-model:coherence}{panjang koherensi} cahaya putih adalah semikrometer, dua atau tiga
panjang gelombang. Sebuah spektrometer yang diarahkan ke medan ``putih'' itu masih melihat
pita gelap pada panjang gelombang yang $\delta = (p + \tfrac12)\lambda$: sebuah
\emph{spektrum berkanal}.
\end{example}

\begin{proposition}[Koherensi ruang: sumbernya harus kecil]\label{prop:b2:two-wave-interference:spatial}
\index{koherensi ruang}\index{sumber luas}
Tiap titik sebuah \omterm{prop:b2:two-wave-interference:spatial}{sumber luas} menghasilkan sistem rumbainya sendiri; sebuah
titik sumber yang tergeser melintang sebesar $b$ pada jarak $L$ di depan
celahnya menggeser rumbainya sebesar $x = bD/L$ (\omterm{thm:b2:two-wave-interference:formula}{beda lintasannya} bertambah
$ab/L$). Sebuah sumber selebar $b$ menyebarkan rumbainya sepanjang satu \omterm{prop:b2:two-wave-interference:young}{jarak antarrumbai},
dan kontrasnya lenyap, ketika
\[
\frac{ab}L \approx \lambda , \qquad\text{yakni ketika celahnya melihat sumbernya di bawah sudut } \frac bL \gtrsim \frac\lambda a .
\]
Kedua celahnya harus berada di dalam \emph{luasan koherensi} cahayanya ---
yaitu daerah yang di dalamnya gelombang dari sumbernya berfase
terdefinisi baik, selebar $\lambda L/b$. Cahaya matahari ($b/L = \qty{0.5}{\degree} = \qty{9}{mrad}$)
memberi lebar koherensi $\lambda/0.009 \approx \qty{60}{\micro m}$: Young memerlukan sebuah
lubang jarum untuk melebarkannya.
\end{proposition}

\begin{proof}
Bagi sebuah titik sumber pada ketinggian $b$, lintasan tambahannya $SS_2 - SS_1 \approx -ab/L$
menambahi $ax/D$: polanya bergeser sebesar $x = bD/L$. Mengintegralkan
pola $\cos^2$ yang tergeser itu pada sumber selebar $b$ memberi kontras
$|\operatorname{sinc}(\pi ab/\lambda L)|$, yang nol pada $ab/L = \lambda$.
\end{proof}

\section{Selaput tipis: pembelahan amplitudo}

\begin{proposition}[Interferensi dalam sebuah selaput tipis]\label{prop:b2:two-wave-interference:film}
\index{interferensi selaput tipis}\index{pembelahan amplitudo}\index{rumbai kemiringan sama}\index{rumbai ketebalan sama}
Sebuah selaput berindeks $n$ dan setebal $e$ dalam udara, yang disinari pada datang $i$
(sudut biasnya $r$), mengembalikan dua gelombang yang terpantul pada kedua mukanya,
dengan \omterm{thm:b2:two-wave-interference:formula}{beda lintasan}
\[
\delta = 2ne\cos r \ (+\tfrac{\lambda_0}2) ,
\]
dengan setengah panjang gelombang tambahannya mencacah pembalikan tanda pemantulan
pada medium yang lebih rapat di muka pertamanya dan tidak di muka kedua
(\cref{ch:b2:wave-interfaces}). Rumbainya terang bagi $2ne\cos r =
(p + \tfrac12)\lambda_0$, gelap bagi $2ne\cos r = p\lambda_0$ --- selaput yang tebalnya
lenyap tak memantulkan \emph{apa pun} (puncak hitam sebuah selaput sabun
yang meniris). Bagi selaput yang tebalnya seragam rumbainya hanya bergantung pada $i$
lalu terlihat di tak hingga (cincin \emph{kemiringan sama}); bagi
sebuah baji yang tebalnya berubah lambat yang dipandang pada datang hampir tegak lurus rumbainya
mengikuti kontur $e$ yang sama lalu terlokalkan pada selaputnya
(rumbai \emph{ketebalan sama}): sebuah baji udara bersudut $\alpha$ menunjukkan
rumbai lurus berjarak $\lambda_0/2\alpha$; sebuah lensa berjari-jari $R$ di atas sebuah kaca rata memberi
\emph{cincin Newton}, yang gelap di $r_p = \sqrt{p\lambda_0R}$.
\end{proposition}

\begin{proof}
Geometrinya: gelombang yang dibiaskan di muka pertamanya menyeberangi selaputnya dua kali
($2e/\cos r$ dalam kacanya, \omterm{def:b2:scalar-light-model:path}{lintasan optiknya} $2ne/\cos r$) sementara gelombang yang terpantul
di muka pertamanya maju $2e\tan r\sin i$ dalam udara; selisihnya adalah
$2ne/\cos r - 2e\tan r\sin i = 2ne\cos r$ dengan $\sin i = n\sin r$. Bagi
bajinya, $e = \alpha x$ dan rumbai gelap berturut-turutnya di $2\alpha x = p\lambda$; bagi
lensanya, $e \approx r^2/2R$.
\end{proof}

\begin{example}[Sabun, minyak, capung]\label{ex:b2:two-wave-interference:colours}
Sebuah selaput sabun ($n = 1.33$) setebal \qty{300}{nm} yang dipandang pada datang tegak lurus:
terang bagi $2ne = (p + \tfrac12)\lambda$, yakni $\lambda = 798/(p + \tfrac12)$ nm: \qty{532}{nm}
(hijau, $p = 1$) di dalam cahaya tampak, dan sebuah pita gelap pada \qty{400}{nm};
selaputnya tampak hijau. Ketika ia meniris lalu menipis, warnanya menjelajahi
runtunannya, berakhir hitam. Sebuah selaput minyak (\qty{400}{nm}, $n = 1.45$) di atas air
($1.33$): pantulan keduanya, ke arah indeks yang lebih rendah, tak membalikkan
tandanya, jadi lagi-lagi $\delta = 2ne + \lambda/2$ dan selaputnya terang pada $2ne =
(p + \tfrac12)\lambda$: \qty{464}{nm} ($p = 2$), biru. Kilau warna sayap
kupu-kupu, cangkang kumbang atau sebuah cakram padat adalah interferensi
sejenis (milik cakramnya, dari banyak gelombang: \cref{ch:b2:gratings}),
itulah sebabnya semuanya berubah dengan sudutnya --- $\delta$ bergantung pada $\cos r$ ---
sementara warna sebuah pigmen tidak.
\end{example}

\begin{method}[Menyelesaikan sebuah soal interferensi]\label{met:b2:two-wave-interference:method}
(1) Kenalilah kedua rute dari sumbernya ke $M$ lalu hitunglah
beda \omterm{def:b2:scalar-light-model:path}{lintasan optiknya} $\delta(M)$, dengan menambahkan $\lambda/2$ bagi tiap pemantulan
yang membalikkan tanda. (2) $I = I_1 + I_2 + 2\sqrt{I_1I_2}\cos(2\pi\delta/\lambda)$;
terang di $\delta = p\lambda$. (3) Jarak rumbainya dari $\delta(M + \Delta M) - \delta(M) =
\lambda$. (4) Periksalah koherensi waktunya ($\delta_{\max} < \ell_c$, jumlah
rumbai yang tampak $\sim\lambda/\Delta\lambda$) dan \omterm{prop:b2:two-wave-interference:spatial}{koherensi ruangnya} (sudut sumbernya
$< \lambda/a$). (5) Katakanlah di mana rumbainya terlokalkan (di mana pun bagi dua
sumber titik; pada selaputnya bagi \omterm{prop:b2:two-wave-interference:film}{rumbai ketebalan sama}; di tak hingga
bagi yang kemiringan sama).
\end{method}

\omterm{prop:b2:two-wave-interference:young}{Celah Young} yang terpisah $a = \qty{0.40}{mm}$, layar di $D = \qty{1.5}{m}$, $\lambda =
\qty{633}{nm}$: tentukanlah \omterm{prop:b2:two-wave-interference:young}{jarak antarrumbainya}; orde rumbai di $x = \qty{1.0}{cm}$; kedudukan
rumbai gelap kesepuluh; \omterm{prop:b2:two-wave-interference:young}{jarak antarrumbainya} dengan celah yang terpisah \qty{2}{mm},
dan dalam air ($n = 1.33$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:two-wave-interference:2}
Dua gelombang berintensitas $I_0$ dan $I_0/4$ berinterferensi: tentukanlah $I_{\max}$, $I_{\min}$,
kontrasnya. Hal yang sama bagi $I_0$ dan $I_0/100$: mengapa rumbainya masih tampak
dengan berkas yang begitu tak sama? Salah satu celah pasangan Young ditutupi sebuah
tapis kelabu yang meneruskan $50\%$: berapa kontras barunya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:two-wave-interference:3}
Sebuah selaput sabun setebal \qty{250}{nm} ($n = 1.33$) yang dipandang pada datang tegak lurus:
tentukanlah panjang gelombang tampak yang paling banyak dan paling sedikit terpantul; warnanya. Sebuah selaput
\qty{1.0}{\micro m}: berapa banyak panjang gelombang terang dalam cahaya tampak, dan apa
yang dilihat mata?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:two-wave-interference:4}
Cincin Newton antara sebuah lensa berjari-jari $R = \qty{2.0}{m}$ dan sebuah kaca rata, dalam
cahaya natrium: tentukanlah jari-jari ketiga cincin gelap pertamanya; jumlah cincin
dalam jari-jari \qty{1}{cm}; terang atau gelapkah pusatnya, dan mengapa?
Apa yang berubah jika celahnya diisi air?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:two-wave-interference:5}
\emph{Cahaya putih.} \omterm{prop:b2:two-wave-interference:young}{Celah Young} dengan cahaya putih ($400$--\qty{700}{nm}),
$a = \qty{0.5}{mm}$, $D = \qty{1}{m}$. (a) Tentukanlah \omterm{prop:b2:two-wave-interference:young}{jarak antarrumbai} bagi lembayung dan merahnya;
kedudukan rumbai terang merah kedua dan lembayung ketiga. (b)
Di luar orde berapa rumbai merah dan lembayung orde berturut-turutnya
bertumpang tindih? (c) Di $x = \qty{5}{mm}$, panjang gelombang mana yang hilang
(spektrum berkanal)? (d) Jelaskanlah runtunan warna yang terlihat dari
tengahnya ke luar.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:two-wave-interference:6}
\emph{Ukuran sumber.} Sebuah lampu natrium menyinari \omterm{prop:b2:two-wave-interference:young}{celah Young} ($a = \qty{1.0}{mm}$)
lewat sebuah celah sumber di $L = \qty{40}{cm}$. (a) Tentukanlah lebar maksimum
celah sumbernya bagi kontras yang baik (ambillah $ab/L < \lambda/4$). (b) Celah sumbernya
dilebarkan menjadi \qty{0.5}{mm}: tentukanlah kontrasnya (pakailah $|\operatorname{sinc}(\pi ab/\lambda L)|$).
(c) Mengapa orang tak dapat sekadar memakai lampu telanjangnya (selebar \qty{1}{cm})? (d)
Matahari sebagai sumber, selebar \qty{9}{mrad}: berapa pemisahan celah maksimum yang
memberi rumbai.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:two-wave-interference:7}
\emph{Cermin Lloyd.} Sebuah sumber titik pada ketinggian $h = \qty{0.2}{mm}$ di atas sebuah
cermin bidang, layar di $D = \qty{1.0}{m}$, $\lambda = \qty{500}{nm}$: gelombang langsungnya
berinterferensi dengan yang terpantul cerminnya (sebuah sumber maya di
$-h$). (a) Tentukanlah \omterm{prop:b2:two-wave-interference:young}{jarak antarrumbainya}. (b) Rumbai di tepi cerminnya ($x = 0$)
gelap: apa yang dikatakannya tentang pemantulannya? (c) Berapa jumlah rumbai
yang tampak jika \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensi} sumbernya \qty{50}{\micro m}. (d) Mengapa
susunan ini merupakan ``interferometer laut'' ahli astronomi radio bagi
sebuah bintang yang terbit?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:two-wave-interference:8}
\emph{Baji udara.} Dua keping kaca sepanjang \qty{10}{cm} bersentuhan di satu ujungnya lalu
terpisah di ujung lainnya oleh sehelai rambut berdiameter $d$; dalam cahaya natrium
tercacah $42$ rumbai gelap. (a) Tentukanlah $d$. (b) Jarak rumbainya. (c) Celahnya
diisi air: berapa jumlah rumbainya. (d) Kepingnya ditekan
sedikit di ujung jauhnya: apa yang dilakukan rumbainya, dan ketelitian $d$ berapa
yang diberikan pencacahan seperempat rumbai?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:two-wave-interference:9}
\emph{Minyak di atas air.} Sebuah selaput minyak ($n = 1.45$) setebal $e$ mengapung
di atas air ($n = 1.33$). (a) Yang mana di antara kedua pantulannya yang membalikkan
tandanya? Berapa \omterm{thm:b2:two-wave-interference:formula}{beda lintasannya}. (b) Bagi $e = \qty{400}{nm}$ pada datang tegak lurus:
tentukanlah panjang gelombang pantul yang menguat dan yang tertiadakan; warnanya. (c) Selaput
paling tipis yang tampak lembayung (\qty{420}{nm}). (d) Mengapa warnanya bergeser ke arah
biru ketika selaputnya dipandang miring? (e) Apa yang berubah bagi selaput
yang indeksnya terletak \emph{di antara} indeks kedua mediumnya (sebuah lapisan pada
kaca)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:two-wave-interference:10}
\emph{Sebuah kaca objek pada satu celah.} Sebuah keping kaca setebal $e = \qty{20}{\micro m}$
dan berindeks $1.5$ menutupi celah $S_1$ sebuah pasangan Young ($a = \qty{1}{mm}$, $D =
\qty{2}{m}$, $\lambda = \qty{500}{nm}$). (a) Tentukanlah \omterm{def:b2:scalar-light-model:path}{lintasan optik} tambahannya; sebesar berapa rumbai
dan ke arah mana polanya bergeser? (b) Dengan cahaya putih,
ke mana rumbai pusat putih (yang kini tergeser) pergi --- masihkah ia
tampak? \omterm{prop:b2:scalar-light-model:coherence}{Panjang koherensinya} \qty{1.5}{\micro m}. (c) Pakailah pola cahaya putihnya
untuk mengukur $e$: bagaimana? (d) Kepingnya dimiringkan: geserannya membesar
--- tunjukkanlah bahwa sampai orde pertama ia membesar sebagai $e(n - 1)\theta^2/2 \times (1 + 1/n)$
lalu taksirlah nilainya bagi \qty{5}{\degree}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:two-wave-interference:11}
\emph{Interferometer bintang.} Dua cermin yang terpisah jarak $a$
memungut cahaya sebuah bintang lalu membawanya berinterferensi. Bintangnya sebuah
cakram seragam berdiameter sudut $\theta$; tiap titiknya menggeser
rumbainya, dan kontrasnya lenyap bagi $a = 1.22\lambda/\theta$ (diterima tanpa bukti, yaitu
versi cakram dari $ab/L = \lambda$). (a) Betelgeuse, $\theta = \qty{0.047}{''}$
(sekon busur), $\lambda = \qty{570}{nm}$: tentukanlah garis dasar yang padanya rumbainya
lenyap (Michelson, 1920, Mount Wilson). (b) Mengapa sebuah teleskop
tunggal \qty{2.5}{m} tak dapat memisahkan cakramnya (batasnya $1.22\lambda/d$)?
(c) Interferometer modern menautkan teleskop yang terpisah \qty{100}{m}: berapa
diameter terkecil yang terukur. (d) Mengapa kedua lintasannya harus sama sampai di dalam
\omterm{prop:b2:scalar-light-model:coherence}{panjang koherensinya}, dan berapa panjang itu bagi sebuah tapis \qty{10}{nm}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:two-wave-interference:12}
\emph{Cincin kemiringan sama.} Sebuah keping sejajar setebal $e =
\qty{2.0}{mm}$ dan berindeks $1.5$ disinari sumber natrium yang lebar;
cahaya pantulnya difokuskan sebuah lensa ber-$f = \qty{50}{cm}$. (a) Tunjukkanlah bahwa
cincinnya berada pada sudut $i_p$ dengan $2ne\cos r_p = (p + \tfrac12)\lambda$ (terang)
dan bahwa, dekat pusatnya, $\cos r \approx 1 - i^2/2n^2$. (b) Tentukanlah orde di
pusatnya; terangkah pusatnya? (c) Jari-jari ketiga cincin terang
pertama pada bidang fokusnya. (d) Mengapa cincin ini terlihat dengan sebuah
\omterm{prop:b2:two-wave-interference:spatial}{sumber luas} sementara rumbai Young menuntut yang sempit? (Pelokalan
di tak hingga.)
\end{exercise}

\section{Soal: Cermin Fresnel dan sebuah selaput minyak}

\begin{problem}[{Soal akhir pekan --- sebuah interferometer tahun 1816 dan
sebuah pelangi di genangan: dua cara membelah satu gelombang dan kedua
koherensi yang membatasi keduanya}]
\label{pb:b2:two-wave-interference:1}

\medskip
\noindent\textbf{Bagian I --- Cermin Fresnel.}
Dua cermin bidang yang bersentuhan sepanjang sebuah garis membentuk sudut yang sangat kecil $\alpha =
\qty{5.0e-3}{rad}$; sebuah sumber celah $S$, yang sejajar garis itu, berada pada $d =
\qty{20}{cm}$ darinya, dan layarnya pada $D = \qty{1.5}{m}$ di luarnya. $\lambda =
\qty{589}{nm}$ (natrium).
\begin{enumerate}
  \item Tunjukkanlah bahwa kedua cerminnya memberi dua citra maya $S_1$, $S_2$ dari
        $S$ pada sebuah lingkaran berjari-jari $d$ yang berpusat pada garis sentuhnya,
        yang terpisah $a = 2\alpha d$; berapa nilainya.
  \item Mengapa gelombang dari $S_1$ dan $S_2$ berinterferensi, sedangkan gelombang
        dua lampu terpisah tidak?
  \item \omterm{prop:b2:two-wave-interference:young}{Jarak antarrumbai} di layarnya (jarak dari sumbernya ke
        layarnya $\approx d + D$); jumlah rumbai di seluruh daerah selebar
        \qty{5}{mm} tempat kedua berkas pantulnya bertumpang tindih.
  \item \omterm{thm:b2:two-wave-interference:formula}{Orde interferensi} di tepi daerah itu; adakah ia di bawah
        batas yang ditetapkan \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensi} lampunya (\qty{2}{cm})?
  \item Celah sumbernya selebar \qty{20}{\micro m}: berapa geseran rumbai
        yang dihasilkan kedua tepinya (tiap titik sumbernya dicitrakan menjadi dua
        titik maya; sistem rumbainya bergeser sebesar $bD'/d$ dengan $D' =
        d + D$); terjagakah kontrasnya (bandingkanlah dengan $i$)?
  \item Celahnya dilebarkan menjadi \qty{0.1}{mm}: tentukanlah kontrasnya, dengan memakai
        $|\operatorname{sinc}(\pi ab/\lambda d)|$.
  \item Kedua cerminnya memantulkan $90\%$ dan $80\%$ cahayanya: berapa kontras
        rumbainya.
  \item Sebuah keping kaca setebal \qty{10}{\micro m} ($n = 1.5$) ditaruh di
        depan salah satu cerminnya: berapa geseran polanya, dalam rumbai; dapatkah ia
        diikuti dengan cahaya natrium? Dengan cahaya putih?
  \item Gantilah lampu natriumnya dengan cahaya putih: lukiskanlah layarnya;
        berapa banyak rumbai berwarna di tiap sisi pusatnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Selaput minyaknya.}
Sebuah selaput minyak ($n = 1.45$) menyebar di jalan yang basah ($n_{\text{air}} = 1.33$),
yang disinari langit mendung dan dipandang pada datang hampir tegak lurus.
\begin{enumerate}[resume]
  \item Pemantulan di kedua mukanya: mana yang membalikkan tandanya? Berapa \omterm{thm:b2:two-wave-interference:formula}{beda
        lintasannya}; syarat bagi sebuah warna yang terang.
  \item Tebal daerah paling tipis yang tampak merah ($\lambda =
        \qty{650}{nm}$); hijau (\qty{550}{nm}); lembayung (\qty{420}{nm}).
  \item Sebuah daerah setebal \qty{600}{nm}: panjang gelombang tampak mana yang
        menguat, mana yang tertiadakan; warnanya.
  \item Mengapa selaputnya menunjukkan pita warna (apa yang berubah sepanjang
        jalannya), dan mengapa pitanya mengikuti kontur tebalnya?
  \item Selaputnya menipis lewat penguapan: lukiskanlah perubahan runtunan
        warnanya di sebuah titik tertentu; apa yang terlihat ketika $e \to 0$
        di sini --- terang atau gelap? Bandingkanlah dengan selaput sabun dalam udara.
  \item Puncak sebuah selaput sabun yang meniris berubah hitam tepat sebelum ia
        pecah: berapa tebalnya di sana (beberapa nanometer) dan mengapa hitam.
  \item Dipandang pada datang \qty{60}{\degree}: tentukanlah sudut biasnya dalam minyaknya,
        dan geseran panjang gelombang terang pertanyaan 11.
  \item Di luar tebal berapa warnanya memudar menjadi kelabu yang
        seragam, bagi cahaya siang yang \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensinya} sekitar \qty{1}{\micro m}?
  \item Cahaya dari langitnya tak terpolarisasi dan tiba dari banyak
        arah: mengapa polanya tak terhapus sebagaimana \omterm{prop:b2:two-wave-interference:spatial}{sumber luas}
        menghapus rumbai Young? (Pelokalan pada selaputnya.)
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Dua koherensi.}
\begin{enumerate}[resume]
  \item Koherensi waktu: bagi cermin Fresnel dengan cahaya natrium,
        berapa orde maksimum yang tampak; dengan sebuah laser ($\ell_c = \qty{10}{m}$).
  \item \omterm{prop:b2:two-wave-interference:spatial}{Koherensi ruang}: berapa lebar sumber maksimum bagi cerminnya
        (dari $ab/d < \lambda/4$) dan bagi \omterm{prop:b2:two-wave-interference:young}{celah Young} dengan $a = \qty{1}{mm}$ di
        $L = \qty{40}{cm}$.
  \item Jelaskanlah mengapa selaput tipisnya memaklumi \omterm{prop:b2:two-wave-interference:spatial}{sumber luas} sementara
        \omterm{prop:b2:two-wave-interference:young}{celah Young} tidak: sketsalah kedua sinar selaputnya bagi dua
        titik sumber yang berbeda lalu tunjukkanlah bahwa keduanya masih berjumpa pada selaputnya.
  \item Garis natriumnya sendiri selebar \qty{0.02}{nm}: berapa banyak rumbai
        yang paling banyak dapat ditunjukkan cerminnya (pakailah $N \approx \lambda/\Delta\lambda$)?
  \item Seorang mahasiswa mengganti lampu natriumnya dengan sebuah LED hijau ($\Delta\lambda =
        \qty{30}{nm}$): berapa jumlah rumbai yang tampak dengan cerminnya.
  \item Dengan sebuah laser rumbainya terentang di seluruh daerah tumpang tindihnya
        dengan kontras penuh tetapi layarnya ``berbercak'': mengapa?
  \item Dalam satu tabel, bagi cerminnya dan selaputnya: bagaimana gelombangnya
        dibelah, di mana rumbainya terlokalkan, apa yang membatasi
        jumlahnya, dan apa yang dilakukan sumber yang lebih lebar.
\end{enumerate}
\end{problem}
