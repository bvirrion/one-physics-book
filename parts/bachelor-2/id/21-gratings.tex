\chapter{Interferensi Banyak Gelombang dan Kisi}\label{ch:b2:gratings}

Miringkanlah sebuah cakram padat di bawah sebuah lampu dan ia melemparkan pelangi: pilinan
lubang mungilnya, yang terpisah satu setengah mikrometer, adalah sebuah kisi --- beribu
jalur pemantul mungil, yang masing-masingnya mengirimkan kembali salinan gelombangnya, semuanya
berinterferensi. Dua gelombang memberi rumbai $\cos^2$ yang lembut pada bab
terdahulu; seribu gelombang memberi rumbai yang seribu kali lebih sempit, cukup tajam
untuk memisahkan dua panjang gelombang yang berselisih satu bagian dalam seratus
ribu. Itulah instrumen yang dengannya susunan
Matahari dan kelajuan galaksi terbaca, dan yang duduk dalam setiap
spektrometer. Bab ini menjumlahkan $N$ gelombang, menurunkan \omterm{prop:b2:gratings:equation}{persamaan
kisinya}, dispersi dan daya urainya, lalu menjumpai piranti banyak gelombang
yang lain, yaitu rongga Fabry--Pérot yang akan menjadi resonator
laser.

\section{Interferensi $N$ gelombang}

\begin{theorem}[$N$ gelombang sama dalam kemajuan fase aritmetika]\label{thm:b2:gratings:nwaves}
\index{interferensi banyak gelombang}\index{maksimum utama}\index{maksimum sekunder}\index{fasor}
$N$ gelombang koheren beramplitudo sama $a$, yang masing-masingnya tertinggal dari yang sebelumnya sebesar
fase $\varphi$, bertumpang tindih menjadi sebuah gelombang beramplitudo $A(\varphi)$ dan
berintensitas
\[
A = a\,\frac{\sin(N\varphi/2)}{\sin(\varphi/2)} , \qquad
I(\varphi) = I_0\,\frac{\sin^2(N\varphi/2)}{\sin^2(\varphi/2)} :
\]
\emph{\omterm{thm:b2:gratings:nwaves}{maksimum utama}} berintensitas $N^2I_0$ di tempat $\varphi = 2\pi p$ (semua
gelombangnya sefase), berlebar setengah sudut $2\pi/N$ dalam $\varphi$ (nol pertamanya
berada di $\varphi = 2\pi p \pm 2\pi/N$), yang dipisahkan $N - 2$ \emph{\omterm{thm:b2:gratings:nwaves}{maksimum
sekunder}} yang tak lebih tinggi daripada sekitar $4.5\%$ \omterm{thm:b2:gratings:nwaves}{maksimum utamanya} bagi $N$
yang besar. Makin banyak gelombangnya, makin tajam dan terang puncaknya: lebarnya
menyusut sebagai $1/N$, tingginya tumbuh sebagai $N^2$, energinya (hasil kali keduanya)
sebagai $N$.
\end{theorem}

\begin{proof}
Amplitudo kompleksnya $a\eu^{\iu k\varphi}$, $k = 0, \dots, N - 1$: sebuah deret
ukur, $a(1 - \eu^{\iu N\varphi})/(1 - \eu^{\iu\varphi}) = a\eu^{\iu(N - 1)\varphi/2}\sin(N\varphi/2)
/\sin(\varphi/2)$. Nolnya di tempat $\sin(N\varphi/2) = 0$ tetapi $\sin(\varphi/2) \ne 0$: $\varphi = 2\pi
m/N$, dengan $m$ bukan kelipatan $N$. Dalam gambaran fasornya $N$ anak panahnya
menutup menjadi sebuah poligon beraturan di nolnya dan berjajar pada \omterm{thm:b2:gratings:nwaves}{maksimum
utamanya}.
\end{proof}

\section{Kisi difraksi}

\begin{proposition}[Persamaan kisi]\label{prop:b2:gratings:equation}
\index{kisi difraksi}\index{persamaan kisi}\index{orde (kisi)}\index{dispersi sudut}
Sebuah \emph{kisi} adalah sehimpunan $N$ celah sejajar yang serupa dan berjarak sama
(atau alur) berperiode $a$. Sebuah gelombang bidang yang datang pada sudut $\theta_0$
dari normalnya dipancarkan ulang tiap celahnya; dalam arah $\theta$
gelombang dari celah tetangganya mempunyai \omterm{thm:b2:two-wave-interference:formula}{beda lintasan} $\delta = a(\sin\theta
- \sin\theta_0)$, dan semuanya menguat dalam arah
\[
a\,(\sin\theta_p - \sin\theta_0) = p\lambda , \qquad p = 0, \pm1, \pm2, \dots
\]
--- yaitu \emph{orde} kisinya. Orde nolnya adalah berkas yang tak
menyimpang bagi setiap panjang gelombang; tiap orde lainnya menyebarkan panjang gelombangnya,
dengan \emph{\omterm{prop:b2:gratings:equation}{dispersi sudut}}
\[
\frac{\dd\theta}{\dd\lambda} = \frac{p}{a\cos\theta} ,
\]
yang lebih besar bagi kisi yang lebih halus dan orde yang lebih tinggi; orde maksimumnya adalah
$|p| \le a(1 + |\sin\theta_0|)/\lambda$. Sebuah \emph{kisi pantul} (alur yang digurat
pada sebuah cermin) menuruti persamaan yang sama dengan sudut pantulnya.
\end{proposition}

\begin{proof}
Fase antara celah yang berturut-turut adalah $\varphi = 2\pi\delta/\lambda$; \omterm{thm:b2:gratings:nwaves}{maksimum
utama} \cref{thm:b2:gratings:nwaves} berada di $\delta = p\lambda$.
Turunkanlah $a\sin\theta = p\lambda + a\sin\theta_0$ pada $p$, $\theta_0$ tetap.
\end{proof}

\begin{example}[Sebuah kisi ruang kelas dan sebuah CD]\label{ex:b2:gratings:examples}
Sebuah kisi $600$ garis tiap milimeter ($a = \qty{1.67}{\micro m}$) pada datang
tegak lurus mengirimkan cahaya \qty{500}{nm} ke $\sin\theta = 0.3$, $\theta = \qty{17.5}{\degree}$ dalam
orde pertama, $\qty{36.9}{\degree}$ dalam orde kedua, dan tak mempunyai orde ketiga
($3 \times 0.3 > 1$); spektrum orde pertamanya berjalan dari \qty{13.9}{\degree}
(\qty{400}{nm}) sampai \qty{24.8}{\degree} (\qty{700}{nm}). Sebuah cakram padat, $a =
\qty{1.6}{\micro m}$, adalah kisi semacam itu dalam pantulan --- dari situlah pelanginya,
dan kenyataan bahwa sebuah DVD ($a = \qty{0.74}{\micro m}$) menyebarkannya lebih lebar.
Merah orde kedua (\qty{700}{nm} pada $\sin\theta = 0.84$) bertumpang tindih dengan lembayung
orde ketiga: ordenya bertumpang tindih di luar yang pertama, sebuah gangguan yang disembuhkan dengan tapis
atau sebuah prisma.
\end{example}

\section{Daya urai}

\begin{proposition}[Daya urai sebuah kisi]\label{prop:b2:gratings:resolution}
\index{daya urai (kisi)}\index{kriteria Rayleigh (spektral)}
Dua panjang gelombang yang berdekatan $\lambda$ dan $\lambda + \Delta\lambda$ tepat terurai dalam orde $p$
ketika \omterm{thm:b2:gratings:nwaves}{maksimum utama} yang satu jatuh pada nol pertama yang lain
(kriteria Rayleigh), yang terjadi bagi
\[
\frac\lambda{\Delta\lambda} = pN :
\]
\emph{daya urai}-nya adalah ordenya dikalikan jumlah garis yang
tersinari. Ia dapat pula dituliskan $W(\sin\theta - \sin\theta_0)/\lambda$ dengan $W = Na$ yaitu
lebar kisinya: yaitu \omterm{thm:b2:two-wave-interference:formula}{beda lintasan} terbesar menyeberangi kisinya,
dalam panjang gelombang --- itulah sebabnya spektrograf besar memakai kisi selebar berpuluh
sentimeter lalu bekerja pada sudut yang menyerempet.
\end{proposition}

\begin{proof}
Maksimum orde $p$ bagi $\lambda + \Delta\lambda$ berada di $\varphi = 2\pi p$ bagi panjang
gelombang itu, yakni di $\delta = p(\lambda + \Delta\lambda)$; nol pertama maksimum
bagi $\lambda$ berada di $\delta = p\lambda + \lambda/N$. Menyamakannya: $p\Delta\lambda = \lambda/N$. Lalu $pN =
Na(\sin\theta - \sin\theta_0)/\lambda$.
\end{proof}

\begin{example}[Memisahkan garis natrium]\label{ex:b2:gratings:sodium}
Dubletnya, $\Delta\lambda = \qty{0.6}{nm}$ pada \qty{589}{nm}, menuntut $R = 1000$: yaitu
seribu garis dalam orde pertama, \qty{2}{mm} sebuah kisi \qty{600}{garis/mm}.
Lebar garisnya sendiri, \qty{0.02}{nm}, menuntut $R = 30\,000$: sebuah kisi \qty{5}{cm}
dalam orde pertama, atau \qty{2.5}{cm} dalam orde kedua --- dan sebuah kisi
\qty{10}{cm} ber-\qty{1200}{garis/mm} dalam orde kedua mencapai
$240\,000$, yang mengurai \qty{0.0025}{nm}: yaitu lebar sebuah garis yang dilebarkan
gerak termal atomnya, dan geseran Doppler sebuah bintang yang bergerak
dengan \qty{1}{km/s}.
\end{example}

\section{Rongga Fabry--Pérot}

\begin{proposition}[Interferensi banyak gelombang lewat pembelahan amplitudo]\label{prop:b2:gratings:fabryperot}
\index{interferometer Fabry--Pérot}\index{fungsi Airy}\index{kehalusan}\index{jangkauan spektral bebas}
Dua cermin sejajar yang memantul sebagian (reflektansi intensitas $R$)
yang terpisah jarak $L$ meneruskan sebuah gelombang pada datang tegak lurus hanya sejauh
gelombang yang telah memantul $0, 1, 2, \dots$ kali di antaranya bertambah
sefase: dengan $\varphi = 4\pi L/\lambda$ yaitu fase pulang perginya, intensitas
terusannya adalah (\omterm{prop:b2:gratings:fabryperot}{fungsi Airy})
\[
\frac{I_t}{I_0} = \frac1{1 + F\sin^2(\varphi/2)} , \qquad F = \frac{4R}{(1 - R)^2} :
\]
puncak tajam berpenerusan penuh di $2L = p\lambda$ (yaitu \emph{resonansi}
rongganya, yang berjarak dalam frekuensi sebesar \emph{\omterm{prop:b2:gratings:fabryperot}{jangkauan spektral bebas}}
$c/2L$), berlebar nisbi $1/\mathcal F$ dengan \emph{kehalusan} $\mathcal F = \pi
\sqrt R/(1 - R)$ --- $30$ bagi $R = 0.9$, $300$ bagi $R = 0.99$. Ia menandingi kisinya
dalam daya urai ($p\mathcal F$ dengan $p = 2L/\lambda \sim 10^5$, maka $R \sim
10^7$) dan, ditutup pada dirinya sendiri, menjadi resonator laser
(\cref{ch:b2:laser}).
\end{proposition}

\begin{proof}
Tiap perjalanan pulang perginya mengalikan amplitudonya dengan $R\eu^{\iu\varphi}$ (sampai pada sebuah
fase tetap); amplitudo terusannya adalah $t^2a\sum(R\eu^{\iu\varphi})^k =
t^2a/(1 - R\eu^{\iu\varphi})$ dengan $t^2 = 1 - R$; kuadrat modulusnya adalah $(1 - R)^2/
(1 - 2R\cos\varphi + R^2) = (1 - R)^2/[(1 - R)^2 + 4R\sin^2(\varphi/2)]$. Puncaknya di
$\varphi = 2\pi p$; setengah maksimumnya di tempat $F\sin^2(\varphi/2) = 1$, yakni $\Delta\varphi \approx
4/\sqrt F = 2(1 - R)/\sqrt R$, melawan $2\pi$ di antara puncaknya.
\end{proof}

\begin{method}[Perhitungan kisi]\label{met:b2:gratings:method}
(1) Periodenya $a = 1/(\text{garis tiap mm})$; \omterm{prop:b2:gratings:equation}{persamaan kisinya} dengan
tanda sudutnya. (2) Daftarkanlah orde yang ada ($|\sin\theta| \le 1$)
lalu periksalah tumpang tindihnya ($p\lambda_{\max}$ terhadap $(p + 1)\lambda_{\min}$). (3)
Dispersinya $p/a\cos\theta$, yang diubah menjadi kedudukan pada detektornya dengan
jarak fokusnya: $\dd x = f\,\dd\theta$. (4) Daya urainya $pN$ dengan $N$ yaitu
garis yang sungguh tersinari; bandingkanlah $\lambda/R$ dengan struktur yang hendak
diurai. (5) Bagi instrumen yang sungguhan tambahkanlah lebar celah masuknya (ia
mengaburkan garisnya) dan piksel detektornya.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:gratings:1}
Sebuah kisi $500$ garis tiap milimeter pada datang tegak lurus, $\lambda =
\qty{550}{nm}$: tentukanlah sudut ordenya; orde tertingginya; lebar sudut
spektrum tampak orde pertamanya ($400$--\qty{700}{nm}); kisi yang sama pada
datang \qty{30}{\degree}: ordenya di tiap sisinya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:2}
Dublet natrium dengan kisi $600$-garis/mm dan sebuah lensa ber-$f =
\qty{50}{cm}$: tentukanlah pemisahan sudut dan lurus kedua garisnya pada
detektornya dalam orde pertama dan kedua; ukuran piksel yang diperlukan untuk melihat
keduanya terpisah.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:3}
Daya urai sebuah kisi \qty{2}{cm} ber-$600$ garis/mm dalam orde $1$, $2$,
$3$; $\Delta\lambda$ terkecil yang terurai pada \qty{550}{nm}; terpisahkankah dublet
natriumnya? Dan garis H$\alpha$ hidrogen dari kembaran deuteriumnya (terpisah
\qty{0.18}{nm})?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:gratings:4}
Sebuah cakram padat ($a = \qty{1.6}{\micro m}$) dan sebuah DVD ($a = \qty{0.74}{\micro m}$) yang disinari pada
datang tegak lurus: tentukanlah sudut lembayung dan merah orde pertamanya; orde mana yang
ada bagi masing-masingnya; mengapa CD-nya menunjukkan dua pelangi dan DVD-nya satu?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:5}
\emph{Pola $N$ celah.} (a) Tunjukkanlah bahwa antara dua \omterm{thm:b2:gratings:nwaves}{maksimum utama}
terdapat $N - 1$ nol dan $N - 2$ \omterm{thm:b2:gratings:nwaves}{maksimum sekunder}. (b) Bagi $N$ yang besar,
tentukanlah intensitas \omterm{thm:b2:gratings:nwaves}{maksimum sekunder} pertama relatif terhadap \omterm{thm:b2:gratings:nwaves}{maksimum
utamanya} (pada $N\varphi/2 \approx 3\pi/2$): sekitar $1/22$. (c) Sketsalah polanya bagi
$N = 3$ dan $N = 6$ lalu berikanlah tinggi \omterm{thm:b2:gratings:nwaves}{maksimum sekundernya} bagi
$N = 3$. (d) Berapa bagian energi totalnya yang berada dalam \omterm{thm:b2:gratings:nwaves}{maksimum utamanya} bagi
$N$ yang besar (bandingkanlah integral puncaknya, berlebar $\propto 1/N$, bertinggi
$\propto N^2$, dengan sisanya).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:6}
\emph{Orde yang bertumpang tindih.} (a) Tunjukkanlah bahwa orde $p$ dari $\lambda_{\max}$ bertumpang tindih dengan
orde $p + 1$ dari $\lambda_{\min}$ ketika $p\lambda_{\max} > (p + 1)\lambda_{\min}$; bagi $400$--\qty{700}{nm},
sejak orde yang mana? (b) \omterm{prop:b2:gratings:fabryperot}{Jangkauan spektral bebas} dalam orde $p$: $\Delta\lambda_{\text{FSR}}
= \lambda/p$. (c) Sebuah spektrograf bekerja dalam orde $20$ dekat \qty{500}{nm} (sebuah
kisi echelle): tentukanlah jangkauan spektral bebasnya; bagaimana tumpang tindihnya disingkirkan
(penyebar silang)? (d) \omterm{prop:b2:gratings:fabryperot}{Jangkauan spektral bebas} Fabry--Pérotnya adalah $c/2L$:
tuliskanlah ia dalam panjang gelombang lalu bandingkanlah dengan milik kisinya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:7}
\emph{Kisi berkilas.} Sebuah kisi pantul mempunyai alurnya yang dimiringkan sebesar
sudut kilas $\beta$ sehingga tiap fasetnya memantul secara cermin ke dalam
arah yang di sana orde $p$ diinginkan. (a) Bagi tatanan Littrow
($\theta = \theta_0 = \beta$): tunjukkanlah $2a\sin\beta = p\lambda$; berapa sudut kilas bagi
\qty{500}{nm} orde pertama dengan $1200$ garis/mm. (b) Mengapa kilasnya mengirimkan sebagian besar
cahayanya ke dalam satu orde (pikirkanlah selubung difraksi satu
faset, \cref{ch:b2:diffraction})? (c) Kisi yang sama dikilaskan
bagi \qty{1}{\micro m}: dalam orde yang mana ia mangkus pada \qty{500}{nm}? (d)
Berapa daya urai sebuah kisi \qty{10}{cm} dalam Littrow pada $\beta = \qty{64}{\degree}$,
$\lambda = \qty{500}{nm}$ (pakailah $R = 2W\sin\beta/\lambda$).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:8}
\emph{Celahnya.} Celah masuk sebuah spektrometer selebar $s$ dicitrakan
pada detektornya dengan perbesaran $1$; kisinya ($600$ garis/mm,
\qty{5}{cm}, orde $1$, $f = \qty{50}{cm}$) menyebarkan. (a) Tentukanlah lebar sebuah garis pada
detektornya karena celahnya saja bagi $s = \qty{50}{\micro m}$, dalam nm. (b)
Lebarnya karena daya urai kisinya. (c) Mana yang menguasai; berapa lebar celah
yang padanya keduanya sama. (d) Mengapa tidak menutup celahnya lebih rapat lagi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:gratings:9}
\emph{Fabry--Pérot.} Cermin ber-$R = 0.95$, $L = \qty{5}{mm}$, $\lambda = \qty{600}{nm}$.
(a) Tentukanlah orde $p$, \omterm{prop:b2:gratings:fabryperot}{jangkauan spektral bebas} dalam frekuensi dan dalam panjang gelombang. (b)
Kehalusannya; lebar sebuah puncak penerusan dalam frekuensi dan panjang gelombang. (c)
Daya urainya $p\mathcal F$; bandingkanlah dengan kisi \qty{10}{cm}. (d) Mengapa
sebuah Fabry--Pérot selalu dipakai dengan sebuah pratapis atau sebuah kisi?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:10}
\emph{Batas sebuah kisi.} (a) Tunjukkanlah bahwa $R = pN = W(\sin\theta - \sin\theta_0)
/\lambda \le 2W/\lambda$: daya urainya tak dapat melampaui lebar
kisinya dalam panjang gelombang, dikalikan dua. (b) Tafsirkanlah: yaitu \omterm{thm:b2:two-wave-interference:formula}{beda lintasan}
terbesar antara sinar ekstremnya. (c) Sebuah kisi \qty{30}{cm} pada
\qty{500}{nm}: tentukanlah $R$ pamungkasnya; kelajuan terkecil yang terurai lewat geseran Doppler
($\Delta\lambda/\lambda = v/c$). (d) Mengapa ahli astronomi toh mencapai \qty{1}{m/s}
(pikirkanlah mengukur kedudukan sebuah garis sampai sepersekian lebarnya,
dengan banyak garis)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:11}
\emph{Bragg.} Sebuah hablur adalah kisi tiga dimensi: sinar-X yang
dipantulkan bidang atom yang berturut-turut yang terpisah jarak $d$, pada
sudut serempet $\theta$, berinterferensi membangun ketika $2d\sin\theta = p\lambda$
(Bragg). (a) Turunkanlah ia dari \omterm{thm:b2:two-wave-interference:formula}{beda lintasan} antara dua bidang.
(b) Radiasi K$\alpha$ tembaga (\qty{0.154}{nm}) pada garam batu ($d = \qty{0.282}{nm}$):
tentukanlah sudut Braggnya. (c) Mengapa panjang gelombang tampak tak memberi pemantulan
Bragg dari hablur, dan apa yang dikatakan kebalikannya (tak ada difraksi
sinar-X dari kaca) tentang kaca? (d) Mengapa harus $\lambda < 2d$?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:gratings:12}
\emph{Larik berfase.} $N$ antena serupa dalam satu garis, berjarak $a$, disuapi
dengan amplitudo yang sama dan tangga fase $\psi$ antara tetangganya. (a)
Tunjukkanlah bahwa medan terpancar dalam arah $\theta$ adalah medan
jumlah $N$ gelombangnya dengan $\varphi = 2\pi a\sin\theta/\lambda - \psi$: berkas utamanya menunjuk
ke tempat $\sin\theta = \lambda\psi/2\pi a$. (b) Lebar setengah berkas utamanya, $\Delta\theta
\approx \lambda/Na\cos\theta$. (c) Mengapa harus $a \le \lambda/2$ untuk menghindari berkas utama kedua
(sebuah cuping kisi) ketika mengemudi? (d) Sebuah radar dengan $64$ elemen pada
$\lambda/2$, $\lambda = \qty{3}{cm}$: berapa lebar berkasnya; waktu untuk memindai $90^\circ$ dalam langkah selebar satu
berkas jika fasenya beralih dalam \qty{1}{\micro s}.
\end{exercise}

\section{Soal: Sebuah spektrograf bagi sebuah bintang}

\begin{problem}[{Soal akhir pekan --- merancang spektrograf yang
membaca cahaya sebuah bintang: kisinya, detektornya, garis
hidrogennya, dan goyangan sebuah planet}]
\label{pb:b2:gratings:1}

Sebuah spektrograf: celah masuk, kolimator berjarak fokus $f_1 =
\qty{1.0}{m}$, sebuah kisi pantul $1200$ garis/mm dan berlebar $W =
\qty{12}{cm}$, sebuah kamera ber-$f_2 = \qty{1.0}{m}$, sebuah detektor berpiksel \qty{15}{\micro m}.
Ia bekerja dalam orde pertama, di sekitar \qty{550}{nm}, dengan
kisinya pada datang \qty{20}{\degree}.

\medskip
\noindent\textbf{Bagian I --- Kisinya.}
\begin{enumerate}
  \item Periode kisinya; jumlah garis yang tersinari jika berkasnya menutupi
        seluruh lebarnya.
  \item Sudut difraksi \qty{550}{nm} dalam orde pertama; sudut
        \qty{400}{nm} dan \qty{700}{nm}; lebar sudut spektrum
        tampaknya.
  \item \omterm{prop:b2:gratings:equation}{Dispersi sudutnya} pada \qty{550}{nm}; dispersi lurusnya pada
        detektornya dalam nm tiap mm dan nm tiap piksel.
  \item Daya urainya; $\Delta\lambda$ terkecil yang terurai pada \qty{550}{nm}; dalam
        piksel.
  \item Jatuhkah lembayung orde kedua ke atas cahaya tampak orde
        pertama? Tapis mana yang menyingkirkannya?
  \item Panjang detektor yang diperlukan bagi seluruh spektrum tampaknya dalam
        orde pertama.
  \item Kisinya dikilaskan pada \qty{17}{\degree}: bagi panjang gelombang yang mana ia
        paling mangkus dalam orde pertama pada datang ini (pakailah
        taksiran Littrow $2a\sin\beta = \lambda$)?
  \item Kisinya mengembalikan $80\%$ cahayanya pada panjang gelombang kilasnya
        dan teleskop ditambah optiknya $40\%$: berapa bagian foton bintangnya
        yang mencapai detektornya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Garisnya.}
\begin{enumerate}[resume]
  \item Garis Balmer hidrogen bintangnya: H$\alpha$ \qty{656.3}{nm},
        H$\beta$ \qty{486.1}{nm}, H$\gamma$ \qty{434.0}{nm}: tentukanlah kedudukannya pada
        detektornya relatif terhadap \qty{550}{nm}.
  \item H$\alpha$ dari sebuah atom deuterium \qty{0.18}{nm} lebih pendek: terpisahkah?
        Sebesar berapa piksel?
  \item Dublet natrium dalam spektrum bintangnya: berapa pemisahannya dalam
        piksel; dan lebar tiap garisnya jika atmosfer bintangnya
        melebarkannya menjadi \qty{0.05}{nm}.
  \item Celah masuknya selebar \qty{100}{\micro m}, yang dicitrakan pada
        perbesaran $f_2/f_1$: berapa lebarnya pada detektornya dalam piksel dan
        dalam nm; terbatasi celah atau terbatasi kisikah spektrografnya?
  \item Cahaya dari teleskopnya datang sebagai cakram \qty{1}{''} di
        langit, yang oleh teleskopnya ($f = \qty{20}{m}$) dijadikan selebar \qty{100}{\micro m}
        di celahnya: berapa bagian cahaya yang hilang jika celahnya
        dipersempit menjadi \qty{50}{\micro m} (ambillah cakramnya seragam).
  \item Mengapa celah yang lebih sempit memberi garis yang lebih tajam tetapi spektrum
        yang lebih berderau?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Bintangnya bergerak.}
\begin{enumerate}[resume]
  \item Sebuah bintang menjauh dengan \qty{30}{km/s}: berapa geseran Doppler H$\alpha$ dalam nm
        dan dalam piksel.
  \item Gerak orbit Bumi menambahkan $\pm\qty{30}{km/s}$ sepanjang tahunnya:
        mengapa ia harus dikurangkan, dan sampai ketelitian berapa jika orang menginginkan
        \qty{100}{m/s}?
  \item Sebuah planet membuat bintangnya bergoyang dengan \qty{10}{m/s}: berapa geserannya dalam nm dan
        dalam piksel; adakah ia di bawah resolusinya? Jelaskanlah bagaimana mengukur
        pusat sebuah garis sampai seperseratus lebarnya, sepanjang
        seribu garis, memulihkannya (perolehan statistik $\sqrt{N}$).
  \item Suhu instrumennya berubah sebesar \qty{1}{K}, dan
        kisinya (kaca, muai \qty{1e-5}{K^{-1}}) memuai: berapa geseran
        garisnya dalam piksel; kemantapan apa yang diperlukan bagi \qty{10}{m/s}?
  \item Sebuah lampu tera (torium--argon) menaruh beratus garis yang diketahui
        pada detektor yang sama: jelaskanlah perannya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Pilihan lain.}
\begin{enumerate}[resume]
  \item Lebar kisi yang sama yang digurat pada $300$ garis/mm dan dipakai dalam
        orde keempat (dengan sebuah penyebar silang untuk memilah ordenya):
        tentukanlah sudut difraksinya, daya urainya, dispersinya dan jangkauan
        spektral bebasnya pada \qty{550}{nm}; apa yang berubah, apa yang tidak?
  \item Sebuah etalon Fabry--Pérot ($L = \qty{1}{mm}$, $R = 0.9$) yang ditaruh sebelum
        celahnya: tentukanlah jangkauan spektral bebasnya dalam nm, kehalusannya, lebar puncaknya; seperti apa
        sisir puncak penerusannya pada detektornya,
        dan untuk apa ia dipakai?
  \item Sebuah prisma \qty{60}{\degree} dalam kaca ber-$\dd n/\dd\lambda = \qty{-1e-4}{nm^{-1}}$
        mempunyai \omterm{prop:b2:gratings:equation}{dispersi sudut} $\dd D/\dd\lambda \approx 2\,\dd n/\dd\lambda$ dekat
        simpangan minimumnya: bandingkanlah dengan milik kisinya; mengapa kisi
        menggantikan prisma?
  \item Mengapa spektrograf astronomi yang terbesar dibangun dengan
        kisi selebar semeter lalu dipakai pada sudut yang besar (batas
        $2W/\lambda$)?
  \item Sebuah bintang bermagnitudo 8 menyerahkan \qty{1e4}{} foton tiap sekon dan
        tiap nanometer di teleskopnya: berapa foton tiap piksel tiap sekon
        pada detektornya, dan nisbah sinyal terhadap derau sebuah pajanan
        \qty{100}{s} (derau foton saja).
  \item Ringkaslah: ketiga angka yang melukiskan sebuah spektrograf
        (dispersi, daya urai, \omterm{prop:b2:gratings:fabryperot}{jangkauan spektral bebas}) dan apa yang
        menetapkan masing-masingnya.
\end{enumerate}
\end{problem}
