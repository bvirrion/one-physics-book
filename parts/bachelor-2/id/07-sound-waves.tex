\chapter{Gelombang Bunyi dalam Fluida}\label{ch:b2:sound-waves}

Hitunglah sekon antara kilat dan guruhnya lalu bagilah dengan tiga:
itulah jaraknya dalam kilometer. Bunyi menyeberangi udara pada
sepertiga kilometer tiap sekon, air pada satu setengah kilometer, dan
sebuah pemindai ultrasonik mengukur waktu gema dari dalam tubuh sampai
sepersejuta sekon untuk melukis gambarnya. Bab ini menerapkan
mekanika fluida \cref{ch:b2:euler-bernoulli,ch:b2:fluid-kinematics}
pada fluida diam yang diganggu riak tekanan kecil, lalu menemukan
\omterm{thm:b2:waves-on-strings:equation}{persamaan d'Alembert} lagi: \omterm{thm:b2:sound-waves:equation}{kelajuan bunyi}, apa yang mengemban
energinya, seberapa keras itu keras, bagaimana bunyi menyebar dalam ruang, dan mengapa sirene
berubah nadanya ketika ia lewat.

\section{Hampiran akustik}

\begin{definition}[Gangguan akustik]\label{def:b2:sound-waves:approximation}
\index{hampiran akustik}\index{tekanan lebih}
Sebuah fluida yang diam, bertekanan seragam $P_0$ dan bermassa jenis $\rho_0$,
diganggu: $P = P_0 + p(M, t)$, $\rho = \rho_0 + \rho_1(M, t)$, dengan kecepatan
$\vect v(M, t)$. \emph{Hampiran akustik} hanya menyimpan orde
pertama dalam besaran kecilnya $p$ (yaitu \emph{tekanan lebih}, atau tekanan
akustik), $\rho_1$ dan $\vect v$: $|p| \ll P_0$, $|\rho_1| \ll \rho_0$, $v \ll c$,
dan perubahan keadaan yang dialami tiap \omterm{def:b2:fluid-kinematics:particle}{partikel fluidanya} begitu cepat
sehingga ia tak bertukar kalor: ia bersifat \emph{isentropik}, dengan
kemampatan $\chi_S = \frac1\rho\bigl(\frac{\partial\rho}{\partial P}\bigr)_S$.
Gravitasi diabaikan.
\end{definition}

\begin{theorem}[Gelombang bunyi]\label{thm:b2:sound-waves:equation}
\index{kelajuan bunyi}\index{persamaan gelombang (akustika)}
Sampai orde pertama, \omterm{thm:b2:euler-bernoulli:euler}{persamaan Euler}, kekekalan massa dan hukum
isentropiknya berbunyi
\[
\rho_0\frac{\partial\vect v}{\partial t} = -\operatorname{\vect{grad}}p , \qquad
\frac{\partial\rho_1}{\partial t} + \rho_0\operatorname{div}\vect v = 0 , \qquad
\rho_1 = \rho_0\chi_Sp ,
\]
dan bersama-sama keduanya memberi \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombang} bagi \omterm{def:b2:sound-waves:approximation}{tekanan lebihnya},
\[
\Delta p = \frac1{c^2}\frac{\partial^2p}{\partial t^2} , \qquad
c = \frac1{\sqrt{\rho_0\chi_S}} ,
\]
(begitu pula bagi $\rho_1$ dan bagi $\vect v$). Untuk gas sempurna, $\chi_S =
1/\gamma P_0$ dan
\[
c = \sqrt{\frac{\gamma P_0}{\rho_0}} = \sqrt{\frac{\gamma RT}{M}} :
\]
\qty{343}{m/s} di udara pada \qty{20}{\degreeCelsius}, naik \qty{0.6}{m/s}
tiap kelvin; \qty{1000}{m/s} dalam helium; \qty{1480}{m/s} dalam air ($\chi_S =
\qty{4.5e-10}{Pa^{-1}}$).
\end{theorem}

\begin{proof}
\omterm{def:b2:fluid-kinematics:euler}{Euler}: suku konvektifnya $\rho(\vect v\cdot\operatorname{\vect{grad}})\vect v$
berorde dua, $\rho\,\partial_t\vect v \approx \rho_0\,\partial_t\vect v$, dan
$\operatorname{\vect{grad}}P = \operatorname{\vect{grad}}p$. Massanya: $\partial_t\rho +
\operatorname{div}(\rho\vect v) \approx \partial_t\rho_1 + \rho_0\operatorname{div}\vect v$.
Hukum isentropiknya: $\dd\rho = \rho\chi_S\,\dd P$ sampai orde pertama. Ambillah
divergensi persamaan pertamanya dan turunan waktu persamaan keduanya:
$\rho_0\partial_t\operatorname{div}\vect v = -\Delta p = -\partial_t^2\rho_1 = -\rho_0\chi_S
\partial_t^2p$. Bagi gas sempurnanya, $PV^\gamma$ yang tetap sepanjang perubahan
keadaan isentropik memberi $\dd\rho/\rho = \dd P/\gamma P$, sehingga $\chi_S = 1/\gamma P$,
dan $P_0/\rho_0 = RT/M$.
\end{proof}

\begin{remark}[Mengapa isentropik]\label{rem:b2:sound-waves:laplace}
Newton menghitung \omterm{thm:b2:sound-waves:equation}{kelajuan bunyi} dengan kemampatan isotermal,
$\sqrt{P_0/\rho_0} = \qty{290}{m/s}$, $15\%$ terlalu rendah; Laplace melihat bahwa
mampatannya terlalu cepat bagi kalor untuk mengalir antara puncak yang hangat dan
lembah yang dingin --- selama satu periode, kalor berdifusi beberapa mikrometer
(\cref{ch:b2:heat-conduction}), tak berarti dibandingkan panjang gelombangnya ---
sehingga perubahan keadaannya adiabatik dan terbalikkan, dan $\gamma$ muncul.
\omterm{thm:b2:sound-waves:equation}{Kelajuan bunyi} karena itu merupakan pengukuran langsung atas $\gamma$: $1.40$ bagi
udara, $1.67$ bagi argon.
\end{remark}

\section{Gelombang bidang, impedansi, intensitas}

\begin{proposition}[Gelombang bunyi bidang berjalan]\label{prop:b2:sound-waves:plane}
\index{impedansi akustik}\index{intensitas akustik}\index{rapat energi (akustik)}
Untuk gelombang bidang $p = f(x - ct)$ yang berjalan menuju $+x$, kecepatan
fluidanya searah $x$, sefase dengan \omterm{def:b2:sound-waves:approximation}{tekanan lebihnya}, dan
\[
p = Z\,v , \qquad Z = \rho_0c = \sqrt{\rho_0/\chi_S} ,
\]
dengan $Z$ yaitu \emph{\omterm{prop:b2:sound-waves:plane}{impedansi akustik}} mediumnya
(\unit{kg.m^{-2}.s^{-1}}): \qty{410}{kg.m^{-2}.s^{-1}} bagi udara, \qty{1.5e6}{kg.m^{-2}.s^{-1}}
bagi air, $\sim\qty{1.6e6}{kg.m^{-2}.s^{-1}}$ bagi jaringan lunak,
\qty{4e7}{kg.m^{-2}.s^{-1}} bagi baja. Gelombangnya mengemban rapat energi dan
intensitas (daya tiap satuan luas, menuju $+x$)
\[
e = \tfrac12\rho_0v^2 + \tfrac12\chi_Sp^2 , \qquad I = p\,v ,
\]
yang menuruti $\partial_te + \partial_xI = 0$; bagi gelombang berjalan kedua
paruh $e$ sama besar dan $I = p^2/Z = Zv^2$. Gelombang sinusoidal
beramplitudo tekanan $p_m$ mempunyai intensitas rerata $\langle I\rangle = p_m^2/2Z$.
\end{proposition}

\begin{proof}
Bagi $p = f(x - ct)$, $\rho_0\partial_tv = -\partial_xp = f'/c\cdot c\dots$: tepatnya,
$\rho_0\partial_tv = -f'(x - ct)$, dan $v = f(x - ct)/\rho_0c$ memenuhinya. Suku
potensial $e$ adalah usaha yang tersimpan dalam memampatkan fluidanya
(mampatan isentropik satu satuan volume sebesar $\dd V/V = -\chi_S\,\dd p$
menyimpan $\int p\,\chi_S\dd p = \tfrac12\chi_Sp^2$); $I$ adalah laju usaha
gaya tekanan $pS$ pada fluida di depannya, yang bergerak dengan $v$, tiap satuan luas.
Neracanya: $\partial_te = \rho_0v\partial_tv + \chi_Sp\partial_tp = -v\partial_xp - p\partial_xv =
-\partial_x(pv)$ menurut kedua persamaan terlinearkannya.
\end{proof}

\begin{definition}[Taraf bunyi]\label{def:b2:sound-waves:decibel}
\index{desibel}\index{taraf bunyi}\index{ambang pendengaran}
Yang disebut \emph{taraf bunyi} sebuah gelombang berintensitas rerata $I$ adalah
\[
L = 10\log_{10}\frac I{I_0} \ \text{dB} , \qquad I_0 = \qty{1e-12}{W/m^2}
\]
--- yaitu ambang pendengaran pada \qty{1}{kHz}, yang di udara bersesuaian dengan
amplitudo tekanan $p_0 \approx \qty{2e-5}{Pa}$ (rms) $= 2 \times 10^{-10}$
atmosfer. Melipatduakan intensitasnya menambah \qty{3}{dB}; sepuluh kali, \qty{10}{dB};
seratus kali amplitudo tekanannya, \qty{40}{dB}. Ruang yang tenang
\qty{30}{dB}, percakapan \qty{60}{dB}, jalan yang ramai \qty{80}{dB}, konser rok
\qty{110}{dB}, ambang nyeri \qty{120}{dB} ($I = \qty{1}{W/m^2}$, $p_m = \qty{29}{Pa}$).
\end{definition}

\begin{example}[Betapa kecilnya sebuah bunyi]\label{ex:b2:sound-waves:small}
Pada ambangnya, \qty{1}{kHz}: $p_m = \qty{2.8e-5}{Pa}$, $v_m = p_m/Z =
\qty{7e-8}{m/s}$, dan amplitudo simpangannya $\xi_m = v_m/\omega =
\qty{1e-11}{m}$ --- sepersepuluh garis tengah sebuah atom: gendang telinga mengindra
gerak yang lebih kecil daripada sebuah atom. Pada \qty{120}{dB}, $\xi_m = \qty{11}{\micro m}$,
masih tak kasatmata, dan $p_m/P_0 = 3 \times 10^{-4}$: bahkan bunyi yang paling keras pun
merupakan gangguan yang amat kecil, itulah sebabnya teori linearnya bekerja begitu baik.
\end{example}

\section{Gelombang bola}

\begin{proposition}[Gelombang bola dan hukum kuadrat terbalik]\label{prop:b2:sound-waves:spherical}
\index{gelombang bola}\index{hukum kuadrat terbalik (bunyi)}
Sebuah sumber kecil yang memancar sama ke segala arah menghasilkan
\omterm{prop:b2:sound-waves:spherical}{gelombang bola}
\[
p(r, t) = \frac{A}{r}\,f\Bigl(t - \frac rc\Bigr) ,
\]
yang amplitudonya turun sebagai $1/r$ dan intensitasnya turun sebagai $1/r^2$:
bagi sumber berdaya akustik $\mathcal P$, $I = \mathcal P/4\pi r^2$, yakni
$-\qty{6}{dB}$ setiap kali jaraknya berlipat dua. Jauh dari sumbernya
gelombangnya bersifat bidang secara setempat, dengan $v = p/Z$ yang radial.
\end{proposition}

\begin{proof}
Dalam koordinat bola, bagi fungsi yang hanya bergantung pada $r$, $\Delta p =
\frac1r\frac{\partial^2(rp)}{\partial r^2}$ (diterima tanpa bukti; \cref{ch:b2:maxwell-equations}),
sehingga $u = rp$ menuruti \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombang} satu dimensi $\partial_r^2u = \partial_t^2u/c^2$,
yang darinya $u = Af(t - r/c)$ bagi gelombang yang keluar. Daya yang menyeberangi
bola berjari-jari $r$ adalah $4\pi r^2I$, tetap jika tak ada yang terserap.
\end{proof}

\begin{example}[Sebuah sirene, sebuah percakapan]\label{ex:b2:sound-waves:siren}
Sirene \qty{10}{W}: $I = 10/4\pi r^2$, \qty{0.8}{W/m^2} pada \qty{1}{m} (\qty{119}{dB},
ambang nyeri), \qty{79}{dB} pada \qty{100}{m}, \qty{59}{dB} pada \qty{1}{km} --- masih terdengar di seluruh
kota. Sebuah suara memancarkan sekitar \qty{10}{\micro W}: \qty{60}{dB} pada \qty{1}{m},
dan \qty{40}{dB} pada \qty{10}{m}. Udara juga menyerap bunyi (lebih banyak pada frekuensi
tinggi, beberapa \omterm{def:b2:sound-waves:decibel}{desibel} tiap seratus meter pada \qty{10}{kHz}), itulah sebabnya
guruh yang jauh menggemuruh rendah.
\end{example}

\section{Efek Doppler}

\begin{theorem}[Efek Doppler]\label{thm:b2:sound-waves:doppler}
\index{efek Doppler}
Sebuah sumber memancar pada frekuensi $f$ dan bergerak dengan kelajuan $v_s$ sepanjang garis
yang menghubungkannya dengan seorang pengamat yang bergerak dengan $v_o$, kedua kelajuannya dihitung
positif ketika keduanya saling mendekat; bunyinya berjalan dengan $c$ dalam
udara yang diam. Pengamatnya menerima frekuensi
\[
f' = f\,\frac{c + v_o}{c - v_s} .
\]
Sumber yang bergerak merapatkan muka gelombangnya di depannya ($\lambda' = (c - v_s)/f$)
dan merenggangkannya di belakangnya; pengamat yang bergerak menjumpai muka gelombangnya dengan
laju $(c + v_o)/\lambda$. Untuk $v_s, v_o \ll c$ keduanya memberi $\Delta f/f \approx
(v_s + v_o)/c$.
\end{theorem}

\begin{proof}
Sumber yang bergerak: muka gelombang yang dipancarkan pada $t$ dan $t + 1/f$ terpisah
sebesar $c/f - v_s/f$ searah geraknya, jadi panjang gelombang di depannya
$\lambda' = (c - v_s)/f$ dan frekuensi yang diterima pengamat yang diam adalah
$c/\lambda'$. Pengamat yang bergerak: panjang gelombangnya $\lambda = c/f$ tetapi mukanya lewat
dengan kelajuan nisbi $c + v_o$. Padukanlah.
\end{proof}

\begin{example}[Sirene, pistol radar, sel darah merah]\label{ex:b2:sound-waves:dopplerex}
Sirene ambulans \qty{700}{Hz} yang lewat pada \qty{25}{m/s}: \qty{755}{Hz}
saat mendekat, \qty{651}{Hz} saat menjauh --- turun seperenam oktaf,
lekuk ``ni-nau'' yang akrab itu. Sebuah pistol radar polisi mengukur efek
yang sama pada gelombang radio \qty{24}{GHz} yang dipantulkan sebuah mobil (gelombangnya
tergeser dua kali, oleh mobil yang menerima lalu memancarkan ulang): $\Delta f = 2fv/c =
\qty{4.8}{kHz}$ pada \qty{30}{m/s}. Sebuah pemeriksa ultrasonik \qty{5}{MHz} yang diarahkan ke
sebuah arteri mendengar sel darahnya pada $\Delta f = 2fv\cos\theta/c \approx \qty{1.6}{kHz}$
bagi \qty{0.5}{m/s} pada \qty{60}{\degree}: sebuah siulan terdengar yang nadanya adalah
kelajuan darahnya. Dan garis spektrum sebuah galaksi yang menjauh tergeser
merah sebesar $\Delta\lambda/\lambda = v/c$ --- rumusnya berlaku bagi cahaya pada kelajuan rendah,
dengan koreksi relativistik di luar itu.
\end{example}

\begin{remark}[Supersonik]\label{rem:b2:sound-waves:mach}
\index{kerucut Mach}\index{ledakan sonik}
Ketika $v_s > c$ rumusnya runtuh ($f' < 0$ di depan): tak ada bunyi yang mendahului
sumbernya. Muka gelombang yang dipancarkan sepanjang lintasannya mempunyai selubung
kerucut bersetengah sudut $\theta$ dengan $\sin\theta = c/v_s$ (\emph{\omterm{rem:b2:sound-waves:mach}{kerucut Mach}});
lompatan tekanan yang diemban kerucutnya adalah \omterm{rem:b2:sound-waves:mach}{ledakan sonik}, yang mencapai
seorang pendengar di tanah hanya sesudah pesawatnya lewat --- pada
Mach $2$ dan ketinggian \qty{10}{km}, $\theta = \qty{30}{\degree}$ dan dentumannya
tiba $10\,\text{km}/\tan\theta/v_s \approx \qty{25}{s}$ sesudah pesawatnya
lewat di atas kepala. Letupan peluru dan ujung sebuah cambuk adalah kerucut yang sama.
\end{remark}

\begin{method}[Orde besar dalam akustika]\label{met:b2:sound-waves:method}
Diberikan \omterm{def:b2:sound-waves:decibel}{taraf bunyi} $L$: $I = I_0\,10^{L/10}$; lalu $p_m = \sqrt{2ZI}$,
$v_m = p_m/Z$, $\xi_m = v_m/\omega$; daya sebuah sumber adalah $4\pi r^2I$ jika ia
memancar ke segala arah (dua kali lebih kecil di atas lantai keras, yang
memantulkan ke separuh ruang). Beralih dari $r_1$ ke $r_2$ mengubah $L$ sebesar
$20\log_{10}(r_1/r_2)$. Menambahkan $n$ sumber tak koheren yang sama menambah
$10\log_{10}n$; dua sumber koheren yang sefase pada sebuah titik menambah \qty{6}{dB}.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:1}
Tentukanlah \omterm{thm:b2:sound-waves:equation}{kelajuan bunyi} di udara ($\gamma = 1.40$, $M = \qty{29}{g/mol}$) pada
\qty{0}{\degreeCelsius}, \qty{20}{\degreeCelsius} dan pada \qty{-50}{\degreeCelsius}
(ketinggian \qty{10}{km}); dalam helium ($\gamma = 1.67$, $M = \qty{4}{g/mol}$) dan
karbon dioksida ($\gamma = 1.30$, $M = \qty{44}{g/mol}$) pada \qty{20}{\degreeCelsius}.
Mengapa suara terdengar melengking sesudah menghirup helium?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:2}
Sebuah percakapan pada \qty{60}{dB}, \qty{500}{Hz}: tentukanlah intensitas, amplitudo
tekanan, amplitudo kecepatan dan amplitudo simpangannya di udara
($Z = \qty{410}{kg.m^{-2}.s^{-1}}$). Hal yang sama bagi \qty{120}{dB}. Taraf dua
orang yang berbicara sekaligus; taraf seratus orang.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:3}
Sebuah pengeras suara memancarkan \qty{1.0}{W} bunyi sama ke segala arah.
Tentukanlah intensitas dan tarafnya pada \qty{1}{m}, \qty{10}{m}, \qty{100}{m}; jarak ketika
tarafnya turun menjadi \qty{60}{dB}; daya yang diperlukan bagi \qty{100}{dB} pada \qty{30}{m}
(sebuah konser terbuka).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:sound-waves:4}
Guruh terdengar \qty{4.5}{s} sesudah kilatnya: jaraknya. Terompet sebuah kapal
bergema dari tebing sesudah \qty{3.0}{s}: jaraknya. Sonar: gema dari
dasar laut kembali sesudah \qty{2.4}{s} ($c = \qty{1500}{m/s}$): kedalamannya. Gema
ultrasonik \qty{5}{MHz} kembali sesudah \qty{130}{\micro s} dalam jaringan
(\qty{1540}{m/s}): kedalamannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:5}
Terompet kereta pada \qty{500}{Hz}; $c = \qty{340}{m/s}$. (a) Frekuensi yang didengar
seseorang di peron ketika keretanya mendekat pada \qty{40}{m/s}, dan
sesudah ia lewat. (b) Orangnya berada dalam kereta yang bergerak menuju
terompetnya pada \qty{40}{m/s}, dengan terompetnya diam; lalu keduanya bergerak saling
mendekat pada \qty{40}{m/s}: bandingkanlah dengan kelajuan nisbi \qty{80}{m/s} dan
satu benda yang diam. (c) Frekuensi yang didengar masinis kereta terompetnya
dari gema pada dinding di depannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:6}
Bagi amplitudo tekanan \qty{1.0}{Pa} pada \qty{1}{kHz}, hitunglah
amplitudo kecepatan dan simpangannya, intensitas dan tarafnya di
udara, dalam air ($Z = \qty{1.5e6}{kg.m^{-2}.s^{-1}}$) dan dalam baja
($\qty{4e7}{kg.m^{-2}.s^{-1}}$). Tekanan yang sama, intensitas yang amat berbeda:
berilah tanggapan. (Taraf di bawah air dinyatakan relatif terhadap \qty{1}{\micro Pa}: berapa
\qty{1}{Pa} pada skala itu?)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:7}
(a) Kelajuan isotermal Newton $\sqrt{P_0/\rho_0}$ bagi udara pada
\qty{20}{\degreeCelsius} ($\rho_0 = \qty{1.20}{kg/m^3}$) dan koreksi Laplacenya.
(b) Yang terukur: \qty{343}{m/s} di udara, \qty{323}{m/s} dalam argon ($M = \qty{40}{g/mol}$)
pada \qty{20}{\degreeCelsius}: simpulkanlah $\gamma$ bagi masing-masingnya; tafsirkanlah dengan
jumlah derajat kebebasannya. (c) Perkirakanlah sejauh mana kalor berdifusi dalam
satu periode pada \qty{1}{kHz} (difusivitas termal udara
\qty{2e-5}{m^2/s}) lalu bandingkanlah dengan panjang gelombangnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:8}
\emph{Pipa organa.} Dalam sebuah pipa \omterm{def:b2:sound-waves:approximation}{tekanan lebihnya} nol di ujung terbuka
dan kecepatannya nol di ujung tertutup. (a) \omterm{prop:b2:waves-on-strings:modes}{Gelombang berdiri} dalam pipa
yang terbuka di kedua ujungnya: tunjukkanlah bahwa ragamnya $f_n = nc/2L$; dalam pipa yang tertutup di
satu ujungnya, $f_n = (2n - 1)c/4L$. (b) Panjangnya bagi nada dasar \qty{440}{Hz}
dalam tiap halnya. (c) Harmonik mana yang dimuat tiap pipanya? (d) Ujung
terbuka yang sungguhan terletak sekitar $0.6$ jari-jari di luar pipanya (koreksi ujung): untuk
pipa \qty{4}{cm}, tentukanlah koreksinya pada $L$ dan pada $f$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:sound-waves:9}
Sebuah sirene listrik \qty{2.0}{kW} mengubah $20\%$ dayanya menjadi bunyi
yang dipancarkan ke separuh ruang (ia berada di atas atap). (a) Intensitas, taraf
dan amplitudo tekanannya pada \qty{100}{m}. (b) Jarak ambang nyerinya
(\qty{120}{dB}); jarak kontur \qty{70}{dB} tempat ia berhenti membuat waspada.
(c) Serapan udara menambahkan \qty{1}{dB} tiap \qty{100}{m} pada frekuensinya:
jarak kontur \qty{70}{dB} sekarang (selesaikanlah secara numerik). (d) Berapa banyak
sirene semacam itu yang meliputi sebuah kota berjari-jari \qty{10}{km}?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:10}
\emph{Energi gelombang bunyi berdiri.} Dalam tabung tertutup sepanjang $L$
\omterm{def:b2:sound-waves:approximation}{tekanan lebihnya} adalah $p = p_m\cos(kx)\cos(\omega t)$, $k = n\pi/L$. (a) Carilah $v(x,
t)$ dari \omterm{thm:b2:euler-bernoulli:euler}{persamaan Euler} yang terlinearkan. (b) Rapat energi kinetik dan
potensialnya; tunjukkanlah bahwa keduanya berayun dalam kuadratur menurut waktu dan
tergeser seperempat panjang gelombang dalam ruang. (c) Energi total dalam
tabungnya (berpenampang $S$); periksalah bahwa ia tetap. (d) Intensitas reratanya: nol
di mana-mana --- namun energinya berpindah: lukiskanlah bagaimana.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:11}
\emph{\omterm{rem:b2:sound-waves:mach}{Ledakan sonik}.} Sebuah pesawat terbang mendatar dengan $v = 1.6\,c$ pada
ketinggian $h = \qty{12}{km}$, dengan $c = \qty{300}{m/s}$ di sana. (a) Turunkanlah sudut
Mach dari selubung muka \omterm{prop:b2:sound-waves:spherical}{gelombang bolanya}. (b) Waktu antara
lewatnya di atas kepala dan dentumannya di tanah (abaikanlah perubahan
$c$ dengan ketinggian). (c) Lebar jalur di tanah yang mendengar
dentumannya ketika pesawatnya lewat, jika jejak kerucutnya terdengar dalam
$\pm\qty{30}{km}$ dari lintasannya: mengapa penerbangan supersonik di atas daratan
dilarang? (d) Sebuah peluru senapan pada \qty{800}{m/s}: sudut Machnya; apa yang didengar seorang pengamat
yang berdiri \qty{10}{m} dari lintasannya, dan dalam urutan apa.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:sound-waves:12}
\emph{Doppler ultrasonik.} Sebuah pemeriksa memancar pada $f = \qty{5.0}{MHz}$ ke dalam jaringan
($c = \qty{1540}{m/s}$); sel darah merahnya bergerak dengan $v$ sepanjang pembuluh yang membuat
sudut $\theta$ dengan berkasnya. (a) Frekuensi yang diterima sebuah sel (pengamat
yang bergerak). (b) Ia memancarkan ulang (menghamburkan) frekuensi itu sebagai sumber yang bergerak:
frekuensinya kembali di pemeriksanya; tunjukkanlah $\Delta f = 2fv\cos\theta/c$ untuk $v \ll c$. (c)
$\Delta f$ untuk $v = \qty{0.50}{m/s}$, $\theta = \qty{60}{\degree}$; untuk $\theta = \qty{90}{\degree}$
--- apa yang dilakukan operatornya? (d) Pemeriksanya dapat memisahkan \qty{50}{Hz}:
resolusi kecepatannya. Mengapa sinyal Dopplernya dibuat terdengar alih-alih
hanya ditampilkan?
\end{exercise}

\section{Soal: Dari guruh sampai ekograf}

\begin{problem}[{Soal akhir pekan --- bunyi di udara terbuka, dalam sebuah aula
konser, di dalam tubuh dan dari sirene yang lewat}]
\label{pb:b2:sound-waves:1}

Udara: $\gamma = 1.40$, $M = \qty{29}{g/mol}$, $Z = \qty{410}{kg.m^{-2}.s^{-1}}$ pada
\qty{15}{\degreeCelsius}; $R = \qty{8.31}{J/(mol.K)}$.

\medskip
\noindent\textbf{Bagian I --- Badai.}
\begin{enumerate}
  \item \omterm{thm:b2:sound-waves:equation}{Kelajuan bunyi} pada \qty{15}{\degreeCelsius} dan pada \qty{30}{\degreeCelsius};
        benarkanlah aturan ``tiga sekon tiap kilometer''.
  \item Guruhnya tiba \qty{4.0}{s} sesudah kilatnya: jarak
        sambarannya. Guruhnya berlangsung beberapa sekon walaupun kilatnya
        seketika: mengapa?
  \item Dekat sambarannya tarafnya mencapai \qty{120}{dB}: amplitudo tekanan
        dan amplitudo kecepatan udaranya.
  \item Energi yang diterima tiap meter persegi dinding selama guruh
        \qty{2}{s} pada \qty{120}{dB}.
  \item Pada malam yang cerah tanahnya mendingin dan udaranya lebih dingin di bawah
        daripada di atas. Sinar bunyi menuruti hukum Snellius antar lapisan,
        $\sin i/c =$ tetap: sinar yang diluncurkan ke atas membelok turun atau naik?
        Mengapa bunyi jauh terdengar begitu jelas pada malam hari dan di atas air?
  \item Pada siang hari, kebalikannya: seorang pendengar pada \qty{3}{km} tak mendengar apa pun.
        Perkirakanlah sudut belokan sebuah sinar mendatar sesudah
        \qty{3}{km} jika $c$ turun \qty{1}{\%} tiap \qty{500}{m} ketinggian
        (perlakukanlah belokannya sebagai seragam: sinarnya sebuah lingkaran).

  \item Udara menyerap frekuensi tinggi jauh lebih banyak daripada yang rendah: mengapa
        guruh yang jauh menggemuruh sementara sambaran yang dekat meletup?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Konser.}
Sebuah pengeras suara menerima \qty{100}{W} daya listrik lalu mengubah
$2\%$ darinya menjadi bunyi, yang dipancarkan sama ke separuh ruang di depannya.
\begin{enumerate}[resume]
  \item Daya akustiknya; intensitas dan tarafnya pada \qty{10}{m}.
  \item Amplitudo tekanannya di sana; amplitudo kecepatan dan simpangannya
        pada \qty{100}{Hz}.
  \item Tarafnya pada \qty{1}{m} (amankah itu?) dan jarak ketika musiknya
        turun menjadi \qty{60}{dB}.
  \item Sepuluh pengeras suara serupa tersebar di sekeliling panggung, tidak
        sefase: tarafnya pada \qty{10}{m}. Dua pengeras suara yang disuapi sefase, pada
        jarak yang sama dari seorang pendengar pada sumbunya: tarafnya di sana.
  \item Aulanya (bervolume \qty{10000}{m^3}) mempunyai \qty{2000}{m^2} dinding dan penonton yang menyerap $30\%$
        bunyi yang menumbuknya: perkirakanlah energi akustik yang tersimpan dalam
        aulanya pada kesetimbangan dan waktu dengungnya (waktu bagi
        tarafnya untuk turun \qty{60}{dB} sesudah musiknya berhenti), dengan memakai
        neraca antara daya yang dipancarkan dan daya yang diserap
        ($I_{\text{dinding}} \approx ce/4$ bagi medan yang membaur, diterima tanpa bukti).
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Ekograf.}
Jaringan lunak: $c = \qty{1540}{m/s}$, $Z = \qty{1.6e6}{kg.m^{-2}.s^{-1}}$; tulang
$Z = \qty{7e6}{kg.m^{-2}.s^{-1}}$; udara $\qty{400}{kg.m^{-2}.s^{-1}}$. Pada
datang tegak lurus, amplitudo tekanan yang dipantulkan pada sebuah antarmuka adalah $r = (Z_2 -
Z_1)/(Z_2 + Z_1)$ kali amplitudo datangnya (diturunkan dalam
\cref{ch:b2:wave-interfaces}); jaringan melemahkan bunyi sekitar
\qty{0.5}{dB} tiap sentimeter dan tiap megahertz.
\begin{enumerate}[resume]
  \item Panjang gelombang dalam jaringan pada \qty{3}{MHz}, \qty{5}{MHz}, \qty{10}{MHz};
        detail terkecil yang dapat dipisahkan masing-masingnya sekitar satu panjang gelombang.
  \item Tunda gema dari sebuah organ sedalam \qty{12}{cm}; laju maksimum pengiriman
        pulsa tanpa mengacaukan gemanya; waktu untuk membangun
        sebuah citra $200$ baris.
  \item Bagian tekanan, dan bagian energi, yang dipantulkan pada
        antarmuka jaringan--udara; pada antarmuka jaringan--tulang; antara lemak
        ($Z = \qty{1.38e6}{kg.m^{-2}.s^{-1}}$) dan otot
        ($\qty{1.70e6}{kg.m^{-2}.s^{-1}}$). Mengapa gelnya, dan mengapa tulang melemparkan
        bayang-bayang?
  \item Pada \qty{5}{MHz} berkasnya menyeberangi \qty{2}{cm} lemak sebelum menjumpai
        otot: bagian energi yang dipancarkan yang kembali ke
        pemeriksanya dari antarmuka itu (pemantulan dan pelemahan pulang
        pergi), dalam \omterm{def:b2:sound-waves:decibel}{desibel}.
  \item Pelemahan pulang pergi dalam \omterm{def:b2:sound-waves:decibel}{desibel} bagi organ pertanyaan 12
        pada ketiga frekuensinya; jika pemindainya sanggup menangani
        \qty{80}{dB} rugi, frekuensi mana yang bekerja? Nyatakanlah imbal balik antara resolusi
        dan kedalamannya.
  \item Pemeriksanya memancarkan pulsa \qty{1}{\micro s} pada \qty{5}{MHz}: berapa
        panjang gelombang panjang sebuah pulsanya, dan bagaimana itu membatasi resolusi kedalamannya?
  \item Moda Doppler pada \qty{5}{MHz} pada sebuah arteri berkelajuan \qty{0.40}{m/s}, dengan berkasnya pada
        \qty{60}{\degree}: pergeserannya; apa yang didengar operatornya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Sirene.}
\begin{enumerate}[resume]
  \item Sebuah ambulans lewat pada \qty{90}{km/h} dengan sirene \qty{700}{Hz}:
        frekuensi yang terdengar saat mendekat dan menjauh; selang
        di antara keduanya dalam semiton (satu semiton adalah nisbah $2^{1/12}$).
  \item Pendengarnya berada dalam mobil yang melaju \qty{15}{m/s} menuju
        ambulans yang mendekat: frekuensi yang terdengar.
  \item Pada saat manakah seorang pendengar di trotoar mendengar tepat
        \qty{700}{Hz}?
  \item Muka gelombang ambulans di depannya: panjang gelombangnya; \omterm{thm:b2:sound-waves:equation}{kelajuan
        bunyinya} relatif terhadap ambulansnya di depan dan di belakang.
  \item Sebuah pistol radar \qty{24}{GHz} membaca pergeseran \qty{3.2}{kHz} dari sebuah mobil:
        kelajuan mobilnya (gelombangnya tergeser dua kali).
  \item Rangkumlah: keenam kelajuan soal ini (bunyi di udara, dalam
        jaringan, ambulansnya, mobilnya, sel darah merahnya, cahaya) serta
        satu rumus yang menangani semuanya.
\end{enumerate}
\end{problem}
