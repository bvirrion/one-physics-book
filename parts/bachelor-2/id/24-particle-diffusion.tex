\chapter{Difusi Partikel}\label{ch:b2:particle-diffusion}

Jatuhkanlah sebutir hablur zat warna ke dalam air yang tenang lalu amatilah: sebuah awan
berwarna terbentuk di sekitarnya, tumbuh, melembut di tepinya, lalu menyebar --- dalam
semenit sepanjang semilimeter, dalam sejam sesentimeter, dalam sepekan sepanjang
gelasnya. Tak ada yang mendorong zat warnanya; ia terbawa desak-desakan tak henti
molekulnya, yang mengirimkan tiap partikelnya pada sebuah jalan acak
dan, secara rerata, dari tempat yang banyak ke tempat yang sedikit.
Inilah \emph{difusi}, yang paling lambat dan paling semesta di antara pengangkutan:
ia menyuapi setiap sel dengan oksigen, mendopingkan boron pada setiap transistor,
mengeraskan baja dengan karbon, dan meloloskan sebuah parfum menyeberangi sebuah ruangan --- meskipun,
sebagaimana akan kita lihat, tidak dalam waktu yang mungkin orang duga. Bab ini memberi
difusinya hukumnya (Fick), persamaannya (dari sebuah imbangan partikel),
penyelesaian ciri dan skala waktunya ($L \sim \sqrt{Dt}$), dan
asal mikroskopisnya dalam jalan acak --- yang juga menjelaskan mengapa
\omterm{thm:b2:particle-diffusion:equation}{persamaan difusinya}, tak seperti setiap persamaan mekanika, mengetahui
arah waktu. Bab berikutnya akan memakai ulang semuanya bagi kalor.

\section{Rapat cacah partikel, fluks dan hukum Fick}

\begin{definition}[Rapat cacah dan arus partikel]\label{def:b2:particle-diffusion:flux}
Bagi sebuah spesies partikel (molekul, ion, atom dalam zat padat)
\emph{rapat cacah}\index{rapat cacah} $n(M,t)$ adalah jumlah
partikel tiap satuan volume di sekitar $M$ (dalam \unit{m^{-3}}; konsentrasi
molarnya adalah $c = n/N_A$). \emph{Rapat arus
partikel}\index{rapat arus partikel} $\vect{j}_N$ adalah vektor
yang jumlah partikel yang menyeberangi sebuah elemen permukaan berarah
$\dd\vect S$ dalam $\dd t$ adalah $\vect{j}_N\cdot\dd\vect S\,\dd t$ (dalam
\unit{m^{-2}.s^{-1}}): \emph{fluks partikel} menembus sebuah permukaan $S$ adalah
$\Phi_N = \iint_S \vect{j}_N\cdot\dd\vect S$, yaitu jumlah partikel tiap
sekon menembus $S$. Bagi partikel yang terbawa sebuah fluida yang bergerak dengan $\vect v$,
$\vect{j}_N = n\vect v$ (konveksi); difusi adalah pengangkutan yang
tersisa dalam fluida yang diam.
\end{definition}

\begin{theorem}[Hukum Fick]\label{thm:b2:particle-diffusion:fick}
Dalam medium yang diam, yang kerapatannya tak seragam, muncul sebuah arus
partikel, yang sebanding dan berlawanan dengan \omterm{def:b2:maxwell-equations:operators}{gradien}
kerapatannya:
\[
\vect{j}_N = -D\,\vect{\operatorname{grad}}\,n ,
\]
dengan \emph{koefisien difusi}\index{koefisien difusi}
$D > 0$ (dalam \unit{m^2/s}) bergantung pada spesies yang berdifusi, pada
mediumnya dan pada suhunya. Partikelnya pergi \emph{menuruni} \omterm{def:b2:maxwell-equations:operators}{gradien}
kerapatannya, dari daerah yang lebih rapat ke yang lebih jarang, dengan laju yang ditetapkan $D$.
\end{theorem}

\begin{proof}
Fenomenologis (sebuah hukum pengalaman, seperti hukum Ohm): linear dalam
\omterm{def:b2:maxwell-equations:operators}{gradiennya} bagi \omterm{def:b2:maxwell-equations:operators}{gradien} yang kecil, isotropik dalam medium isotropik, dan
dengan tanda yang dipaksakan pengalaman. Model jalan acak
\cref{sec:b2:particle-diffusion:microscopic} menurunkannya, dan $D$ bersamanya,
dari gerak molekulnya.
\end{proof}

\begin{example}[Orde besaran $D$]\label{ex:b2:particle-diffusion:orders}
Gas: $D \sim \qty{1e-5}{m^2/s}$ (uap air dalam udara
\qty{2.5e-5}{m^2/s}, sebuah molekul parfum \qty{5e-6}{m^2/s}). Cairan:
$D \sim \qty{1e-9}{m^2/s}$ (oksigen dalam air \qty{2e-9}{m^2/s}, gula
\qty{5e-10}{m^2/s}, sebuah protein \qty{1e-10}{m^2/s}). Zat padat: mungil dan
naik curam dengan suhunya, $D = D_0\,\eu^{-E_{\text{a}}/k_BT}$
(boron dalam silikon: \qty{1.5e-17}{m^2/s} pada \qty{1100}{\celsius},
kecil tak terukur pada suhu ruang --- itulah sebabnya sebuah transistor,
sekali dibuat, menjaga profil dopingnya berpuluh tahun). Empat orde
besaran dari gas ke cairan, delapan atau lebih dari cairan ke zat padat.
\end{example}

\section{Imbangan partikel dan persamaan difusi}

\begin{theorem}[Imbangan partikel setempat]\label{thm:b2:particle-diffusion:balance}
Jika partikelnya tak diciptakan maupun dimusnahkan,
\[
\frac{\partial n}{\partial t} + \operatorname{div}\vect{j}_N = 0 ;
\]
dengan sebuah sumber yang menciptakan $\sigma$ partikel tiap satuan volume dan waktu
(sebuah reaksi kimia, sebuah serapan), $\partial_t n +
\operatorname{div}\vect{j}_N = \sigma$. Dalam satu dimensi (kerapatan dan
arusnya hanya bergantung pada $x$): $\partial_t n + \partial_x j_N = 0$.
\end{theorem}

\begin{proof}
Ambillah lempeng antara $x$ dan $x + \dd x$, berpenampang $S$: ia memuat
$n\,S\,\dd x$ partikel; dalam $\dd t$, $j_N(x)S\,\dd t$ masuk menembus muka
kirinya dan $j_N(x + \dd x)S\,\dd t$ pergi menembus muka kanannya, jadi
$\partial_t n\,S\,\dd x\,\dd t = -(j_N(x+\dd x) - j_N(x))S\,\dd t =
-\partial_x j_N\,\dd x\,S\,\dd t$. Dalam tiga dimensi pencacahan yang sama pada
sebuah kotak kecil memberi \omterm{def:b2:maxwell-equations:operators}{divergensinya} (yaitu fluks yang keluar dari sebuah permukaan tertutup tiap
satuan volume, \cref{ch:b2:maxwell-equations}), atau langsung: bagi sebarang
volume tetap $V$, $\dd/\dd t \iiint_V n\,\dd\tau = -\iint_{\partial V}
\vect{j}_N\cdot\dd\vect S = -\iiint_V \operatorname{div}\vect{j}_N\,\dd\tau$
lewat teorema Ostrogradski.
\end{proof}

\begin{theorem}[Persamaan difusi]\label{thm:b2:particle-diffusion:equation}
Dengan menggabungkan hukum Fick dan imbangannya, bagi $D$ yang seragam,
\[
\frac{\partial n}{\partial t} = D\,\Delta n + \sigma ,
\qquad\text{dalam satu dimensi}\quad
\frac{\partial n}{\partial t} = D\,\frac{\partial^2 n}{\partial x^2} + \sigma .
\]
\emph{Persamaan difusi}\index{persamaan difusi} ini linear
(penyelesaiannya bertumpang tindih), berorde pertama dalam waktu dan orde kedua dalam ruang
--- bukan sebuah \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombang}: ia tak mempunyai kelajuan rambat, melainkan sebuah
hubungan ciri antara panjang dan waktu,
\[
L \sim \sqrt{D t}, \qquad t \sim \frac{L^2}{D} :
\]
berdifusi sepanjang dua kali jaraknya memakan waktu empat kali lebih lama.
\end{theorem}

\begin{proof}
$\partial_t n = -\operatorname{div}(-D\,\vect{\operatorname{grad}}\,n) +
\sigma = D\operatorname{div}\vect{\operatorname{grad}}\,n + \sigma = D\Delta
n + \sigma$. Bagi skalanya, bandingkanlah $n/t$ dengan $Dn/L^2$.
\end{proof}

\begin{remark}[Difusi bersifat takterbalikkan]\label{rem:b2:particle-diffusion:irreversible}
Ubahlah $t$ menjadi $-t$ dalam \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombangnya}, $\partial_t^2 u = c^2
\partial_x^2 u$, dan ia tak berubah: sebuah film gelombang yang diputar mundur adalah
sebuah gelombang yang mungkin. Lakukanlah hal yang sama dalam \omterm{thm:b2:particle-diffusion:equation}{persamaan difusinya} dan tanda
ruas kirinya berbalik: $n(x,-t)$ \emph{bukan} sebuah penyelesaian. Sebuah awan yang
menyebar itu wajar; sebuah awan yang secara spontan berkumpul menjadi sebuah hablur
tidak --- difusi mempunyai anak panah waktu, yaitu anak panah hukum kedua
(\cref{ch:b2:heat-conduction} menghitung entropi yang diciptakannya). Persamaannya
juga menunjukkan bahwa difusi memuluskan: di tempat $n$ setempat merupakan sebuah
maksimum ($\partial_x^2 n < 0$) ia menurun, di tempat ia sebuah minimum ia
menaik.
\end{remark}

\begin{example}[Berapa lama waktunya?]\label{ex:b2:particle-diffusion:times}
$t \sim L^2/D$. Oksigen menyeberangi sebuah sel, $L = \qty{10}{\micro m}$ dalam air:
$10^{-10}/2 \times 10^{-9} = \qty{0.05}{s}$ --- difusi menyuapi sebuah sel dengan
mudah; menyeberangi \qty{1}{mm} jaringan: \qty{500}{s} --- terlalu lambat, itulah
sebabnya tak ada makhluk hidup yang lebih tebal daripada sepersekian milimeter tanpa
pembuluh darah. Gula menyeberangi secangkir teh yang tak teraduk, $L = \qty{5}{cm}$:
$2.5 \times 10^{-3}/5 \times 10^{-10} = 5 \times 10^6\,\unit{s}$, dua bulan
--- aduklah. Sebuah parfum menyeberangi sebuah ruangan lewat difusi saja, \qty{5}{m} dalam
udara: $25/10^{-5} = \qty{3e6}{s}$, sebulan; Anda menciumnya dalam semenit
karena udaranya bergerak: \emph{konveksi} yang mengemban, difusi hanya
menuntaskan milimeter terakhirnya.
\end{example}

\section{Penyelesaian cirinya}

\begin{proposition}[Rezim tegar: membran]\label{prop:b2:particle-diffusion:membrane}
Antara dua tandon yang ditahan pada $n_1$ dan $n_2$, menyeberangi sebuah membran
setebal $e$ dan berluas $S$, kerapatan tegarnya linear,
\[
n(x) = n_1 + (n_2 - n_1)\frac{x}{e}, \qquad
j_N = D\,\frac{n_1 - n_2}{e}, \qquad
\Phi_N = \frac{n_1 - n_2}{R_{\text{d}}}, \quad R_{\text{d}} = \frac{e}{DS}:
\]
membrannya mempunyai \emph{hambatan difusif} $e/DS$, tepat sebagaimana sebuah kawat
mempunyai $\ell/\gamma S$ --- hambatan yang terseri bertambah, dan sebuah membran tipis
ber-$D$ kecil masih dapat menguasai.
\end{proposition}

\begin{proof}
Tegar, satu dimensi, tanpa sumber: $\partial_x^2 n = 0$, jadi $n$
afin; $j_N = -D\,\dd n/\dd x$ seragam.
\end{proof}

\begin{proposition}[Rezim tegar: geometri bola]\label{prop:b2:particle-diffusion:sphere}
Di sekitar sebuah bola berjari-jari $a$ yang permukaannya ditahan pada $n_0$, dalam sebuah
medium yang $n \to 0$ jauh di sana, kerapatan tegarnya adalah
\[
n(r) = n_0\,\frac{a}{r}, \qquad
\Phi_N = 4\pi D a\,n_0 :
\]
arus total yang dilepaskan (atau diserap, dengan tandanya dibalik) tumbuh
dengan \emph{jari-jari}, bukan permukaan, bolanya --- yaitu
geometri sebuah lubuk difusif yang sama dengan geometri sebuah kapasitansi elektrostatik.
\end{proposition}

\begin{proof}
$\Delta n = \frac{1}{r^2}\frac{\dd}{\dd r}(r^2\,\dd n/\dd r) = 0$ memberi
$n = A + B/r$; syarat batasnya menetapkan $A = 0$, $B = n_0 a$; lalu
$\Phi_N = -4\pi r^2 D\,\dd n/\dd r = 4\pi D n_0 a$ pada setiap $r$ (tanpa
penimbunan dalam rezim tegarnya).
\end{proof}

\begin{proposition}[Gauss yang menyebar]\label{prop:b2:particle-diffusion:gaussian}
$N$ partikel tiap satuan luas yang dilepaskan pada $t = 0$ di bidang $x = 0$ sebuah
medium takhingga menyebar sebagai
\[
n(x,t) = \frac{N}{\sqrt{4\pi D t}}\,\exp\Big(-\frac{x^2}{4Dt}\Big) :
\]
sebuah Gauss bersimpangan baku $\sigma(t) = \sqrt{2Dt}$, berluas tetap
$N$ (partikelnya kekal) dan berpuncak menurun $\propto
1/\sqrt t$. Dalam tiga dimensi, $N$ partikel yang dilepaskan di sebuah titik
menyebar sebagai $n = N\,(4\pi Dt)^{-3/2}\exp(-r^2/4Dt)$, dengan $\langle r^2
\rangle = 6Dt$. Sebuah pulsa partikel yang dilepaskan di \emph{permukaan}
sebuah setengah ruang (yang tak dapat mereka tinggalkan) memberi dua kali ungkapan di atas
bagi $x > 0$.
\end{proposition}

\begin{proof}
Sisipkanlah: dengan $u = x^2/4Dt$, $\partial_t n = n\,(-1/2t + u/t)$ dan
$D\,\partial_x^2 n = D\,n\,(-1/2Dt + x^2/4D^2t^2) = n\,(-1/2t + u/t)$;
sama. Penormalannya $\int n\,\dd x = N$ menyusul dari $\int
\eu^{-x^2/4Dt}\dd x = \sqrt{4\pi Dt}$, dan $\int x^2 n\,\dd x/N = 2Dt$.
Ketunggalannya (mengingat pulsa awalnya) diterima tanpa bukti.
\end{proof}

\begin{proposition}[Konsentrasi permukaan yang tetap]\label{prop:b2:particle-diffusion:erfc}
Sebuah setengah ruang $x > 0$ yang mula-mula kosong, yang permukaannya ditahan pada
kerapatan $n_0$ sejak $t = 0$ (sebuah gas yang bersentuhan dengan sebuah zat padat yang
melarutkannya), terisi sebagai
\[
n(x,t) = n_0\,\operatorname{erfc}\Big(\frac{x}{2\sqrt{Dt}}\Big),
\qquad
\operatorname{erfc}(u) = \frac{2}{\sqrt\pi}\int_u^{\infty}\eu^{-s^2}\dd s,
\]
yaitu \emph{fungsi galat komplementer}, yang turun dari $1$ pada $u =
0$ menjadi $0.16$ pada $u = 1$ dan $0.005$ pada $u = 2$: kedalaman tembusnya
lagi-lagi $\sim 2\sqrt{Dt}$, dan jumlah total partikel yang terserap tiap
satuan luas adalah $2n_0\sqrt{Dt/\pi}$.
\end{proposition}

\begin{proof}
Carilah $n = f(u)$ dengan $u = x/2\sqrt{Dt}$: persamaannya menjadi
$f'' + 2uf' = 0$, jadi $f' \propto \eu^{-u^2}$ dan $f$ adalah sebuah fungsi
galat; syarat $f(0) = n_0$, $f(\infty) = 0$ memilih
$\operatorname{erfc}$. Jumlah yang terserapnya adalah $\int_0^\infty n\,\dd x =
n_0\,2\sqrt{Dt}\int_0^\infty\operatorname{erfc}(u)\dd u = 2n_0\sqrt{Dt/\pi}$.
\end{proof}

\begin{example}[Mengeraskan baja]\label{ex:b2:particle-diffusion:carburising}
Sebuah suku baja yang ditahan pada \qty{900}{\celsius} dalam gas kaya karbon menyerap
karbon di permukaannya; dengan $D = \qty{5e-12}{m^2/s}$ bagi karbon dalam besi
panas, empat jam memberi $2\sqrt{Dt} = \qty{0.5}{mm}$: sebuah kulit keras setengah
milimeter pada inti yang liat --- yaitu \emph{pengerasan kulit} roda gigi dan
\omterm{def:b2:rigid-body-mechanics:pivot}{bantalan}, yang diatur waktunya dengan profil erfcnya.
\end{example}

\section{Gambaran mikroskopisnya: jalan acak}\label{sec:b2:particle-diffusion:microscopic}

\begin{proposition}[Jalan acak dan koefisien difusi]\label{prop:b2:particle-diffusion:walk}
Sebuah partikel yang membuat sebuah langkah sepanjang $\ell$ dalam arah acak
tiap $\tau$ (sebuah molekul di antara dua tumbukan) mempunyai, sesudah $N =
t/\tau$ langkah, sebuah simpangan kuadrat rerata
\[
\langle x^2\rangle = N\ell^2 = \frac{\ell^2}{\tau}\,t
\quad\text{(satu dimensi)}, \qquad
\langle r^2\rangle = 3\langle x^2\rangle \quad\text{(tiga dimensi)}.
\]
Dengan membandingkannya dengan penyelesaian Gaussnya, $\langle x^2\rangle = 2Dt$:
\[
D = \frac{\ell^2}{2\tau} \quad\text{(1D)}, \qquad
D = \frac{\ell^2}{6\tau} = \frac{\ell\,v^*}{6}\ \sim\ \frac{\ell\,v^*}{3}
\quad\text{(3D, } v^* = \ell/\tau\text{; teori kinetik memberi } \tfrac13) .
\]
Jarak pejalannya tumbuh sebagai $\sqrt t$, bukan $t$: untuk pergi dua kali lebih jauh
ia memerlukan empat kali lebih banyak langkah.
\end{proposition}

\begin{proof}
$x = \sum_i \epsilon_i\ell$ dengan $\epsilon_i = \pm1$ yang saling bebas (1D):
$\langle x^2\rangle = \sum_{i,j}\langle\epsilon_i\epsilon_j\rangle\ell^2 =
N\ell^2$, dengan suku silangnya rata-ratanya nol. Sebaran $x$
bagi $N$ yang besar cenderung ke sebuah Gauss (yaitu teorema limit pusat, yang
buktinya milik kuliah peluang) --- itulah sebabnya
hukum makroskopisnya adalah \omterm{thm:b2:particle-diffusion:equation}{persamaan difusinya}; hukum Fick sendiri menyusul
dari mencacah pejalan yang menyeberangi sebuah bidang dari kedua sisinya: $j_N \approx
-\tfrac12\ell v^*\,\partial_x n$ (rerata yang lebih cermat memberi $1/3$ dalam
tiga dimensi).
\end{proof}

\begin{example}[Dari molekul ke $D$]\label{ex:b2:particle-diffusion:kinetic}
Udara pada suhu ruang: jalan bebas reratanya $\ell \approx \qty{70}{nm}$,
kelajuan reratanya $v^* \approx \qty{500}{m/s}$: $D \approx \ell v^*/3 =
\qty{1.2e-5}{m^2/s}$, seperti yang terukur. Untuk berdifusi \qty{1}{m} sebuah molekul
memerlukan $N = (L/\ell)^2 = 2 \times 10^{14}$ tumbukan, yakni $N\ell/v^*
\approx \qty{3e4}{s}$ --- delapan jam bagi sebuah perjalanan yang akan ditempuhnya dalam
\qty{2}{ms} jika ia terbang lurus. Dalam sebuah cairan, sebuah bola berjari-jari $a$
yang dihantam-hantam molekulnya menuruti hubungan Stokes--Einstein $D =
k_BT/6\pi\eta a$ (Einstein 1905): bagi gula, $a \approx \qty{0.4}{nm}$
dalam air, $D = \qty{5e-10}{m^2/s}$; bagi sebutir \qty{1}{\micro m},
$\qty{4e-13}{m^2/s}$, yaitu simpangan semikrometer tiap sekon ---
yaitu gerak Brown yang diukur Perrin untuk mencacah bilangan Avogadro. Dalam
sebuah zat padat sebuah atom melompat ke situs tetangganya hanya ketika sebuah fluktuasi
termal memasok energi aktivasinya: $D = D_0\,\eu^{-E_{\text{a}}/k_BT}$,
berlipat dua tiap beberapa puluh kelvin dekat \qty{1000}{\celsius}.
\end{example}

\begin{method}[Taksiran difusi]\label{meth:b2:particle-diffusion:method}
(1) Kenalilah $D$ (gas $10^{-5}$, cairan $10^{-9}$, zat padat Arrhenius).
(2) Waktu atau jaraknya: $t \sim L^2/D$, $L \sim \sqrt{Dt}$ (dengan
$\sqrt{2Dt}$ yang eksak bagi lebar Gaussnya, $2\sqrt{Dt}$ bagi kedalaman erfcnya).
(3) Tegar: linear dalam sebuah lempeng ($R_{\text{d}} = e/DS$), $1/r$ di sekitar sebuah
bola ($\Phi = 4\pi D a\,\Delta n$), $\ln r$ di sekitar sebuah silinder. (4)
Peralihan: Gauss bagi sebuah pulsa, erfc bagi sebuah permukaan yang ditahan; tumpangkanlah.
(5) Tanyakanlah apakah konveksi tidak menguasai. (6) Periksa mikroskopisnya:
$D \sim \ell v^*/3$.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:1}
Tentukanlah waktu difusi $L^2/D$: sebuah gula batu di dasar sebuah cangkir \qty{5}{cm}

sel \qty{10}{\micro m} ($D = \qty{2e-9}{m^2/s}$). Mana di antara semuanya yang sungguh
dilakukan difusi?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:2}
Sebuah membran setebal \qty{1}{\micro m} dan berluas \qty{1}{cm^2}, $D =
\qty{1e-11}{m^2/s}$ di dalamnya, memisahkan sebuah larutan pada \qty{1}{mol/m^3}
dari air murni. Tentukanlah rapat arusnya, fluks molarnya dan jumlah molekul
tiap sekon; hambatan difusifnya; dan dengan dua membran semacam itu terseri.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:3}
Sebuah lapisan tipis zat warna ($N = \qty{1e20}{m^{-2}}$) dilepaskan dalam air,
$D = \qty{1e-9}{m^2/s}$. Tentukanlah lebarnya $\sqrt{2Dt}$ dan kerapatan puncaknya sesudah
\qty{1}{s}, \qty{1}{h}, \qty{1}{hari}. Kapan puncaknya turun menjadi
$1\%$ nilainya pada \qty{1}{s}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:particle-diffusion:4}
Molekul udara: $\ell = \qty{70}{nm}$, $v^* = \qty{500}{m/s}$. Taksirlah
$D$; jumlah tumbukan dan waktu untuk berdifusi \qty{1}{m}; bandingkanlah dengan
penerbangan lurusnya. Hal yang sama bagi sebuah molekul dalam air ($\ell \approx
\qty{0.1}{nm}$, $v^* \approx \qty{500}{m/s}$): benarkah taksirannya, dan
mengapa tidak tepat?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:5}
\emph{Oksigen dalam sebuah jaringan.} Sebuah lempeng jaringan setebal $2a$ mengonsumsi
oksigen dengan laju seragam $q$ (tiap satuan volume); kedua mukanya
ditahan pada $n_0$. (a) Tuliskanlah persamaan tegarnya dengan lubuknya lalu selesaikanlah.
(b) Syarat agar oksigennya mencapai pusatnya. (c) $D = \qty{2e-9}{m^2/s}$,
$n_0 = \qty{0.2}{mol/m^3}$, $q = \qty{0.01}{mol/m^3/s}$: berapa $a$ maksimumnya. (d)
Simpulkanlah tentang jarak antara pembuluh kapilernya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:6}
\emph{Sebutir yang melarut.} Sebuah bola gula berjari-jari $a = \qty{1}{mm}$
dalam air yang tenang menjaga permukaannya pada kerapatan jenuhnya $n_{\text{s}}
= \qty{5800}{mol/m^3}$; jauh di sana airnya murni; $D =
\qty{5e-10}{m^2/s}$. (a) Tentukanlah profil tegar dan arus totalnya. (b)
Jumlah mol dalam butirnya (kerapatan \qty{1590}{kg/m^3}, massa molar
\qty{342}{g/mol}) dan waktu pelarutannya (ambillah arusnya tetap).
(c) Mengapa waktu yang sebenarnya lebih lama, dan mengapa pengadukan begitu menolong?
(d) Tunjukkanlah bahwa bagi sebuah silinder yang sangat panjang profil tegarnya
logaritmik dan soalnya tak berpenyelesaian dalam medium takhingga.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:7}
\emph{Kekekalan dan ketakterbalikan.} (a) Dari persamaannya, tunjukkanlah
bahwa $\int n\,\dd x$ tetap dan bahwa $\dd\langle x^2\rangle/\dd t =
2D$ (integralkanlah secara bagian demi bagian, $n \to 0$ di takhingga). (b) Tunjukkanlah bahwa
$n(x,-t)$ tak memenuhi persamaannya. (c) Menjadi apa penyelesaian
Gaussnya bagi $t < 0$? (d) Tunjukkanlah bahwa $\int n^2\,\dd x$ hanya dapat
menurun: difusi meratakan.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:8}
\emph{Karburisasi.} Karbon dalam besi panas, $D = \qty{5e-12}{m^2/s}$ pada
\qty{900}{\celsius}; permukaannya ditahan pada $n_0$. (a) Periksalah bahwa $n_0
\operatorname{erfc}(x/2\sqrt{Dt})$ menyelesaikan persamaannya. (b) Pada kedalaman
berapa $n = 0.1\,n_0$ sesudah \qty{4}{h} ($\operatorname{erfc}(1.16) = 0.1$).
(c) Waktu bagi kedalaman dua kalinya. (d) Pada \qty{950}{\celsius}, $D$ menjadi $2.5$
kali lebih besar: berapa waktu yang dihemat.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:particle-diffusion:9}
\emph{Stokes--Einstein.} $D = k_BT/6\pi\eta a$, air ber-$\eta =
\qty{1e-3}{Pa.s}$, \qty{300}{K}. (a) Tentukanlah $D$ bagi $a = \qty{1}{\micro m}$ dan
simpangan akar kuadrat reratanya dalam \qty{1}{s}, \qty{1}{min}. (b) Berapa $D$
bagi sebuah protein, $a = \qty{3}{nm}$. (c) Gula mempunyai $D = \qty{5e-10}{m^2/s}$:
berapa jari-jari efektifnya. (d) Bagaimana mengukur $\langle x^2\rangle$ sebutir
di bawah sebuah mikroskop memberi bilangan Avogadro?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:10}
\emph{Tunda sebuah membran.} Sebuah membran setebal $e$, yang mula-mula
kosong, dipaparkan pada satu mukanya kepada $n_1$ pada $t = 0$, muka lainnya ditahan
pada $0$. (a) Berapa fluks tegar akhirnya? (b) Berikanlah alasan bahwa fluks
pada muka jauhnya naik sepanjang waktu $\sim e^2/D$ (tunda eksaknya
$e^2/6D$). (c) Sebuah tempelan obat dengan $e = \qty{20}{\micro m}$, $D =
\qty{1e-13}{m^2/s}$ menembus lapisan luar kulitnya: berapa waktu tundanya. (d) Mengapa
tundanya berguna untuk mengukur $D$ dan fluks tegarnya untuk mengukur
$D \times$ kelarutannya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:11}
\emph{Penyerap sempurna.} Sebuah bola berjari-jari $a$ menyerap setiap
partikel yang menyentuhnya, dalam sebuah medium pada $n_\infty$ jauh di sana. (a)
Tentukanlah profil tegar dan arus total yang tertangkapnya. (b) Sebuah bakteri, $a =
\qty{1}{\micro m}$, dalam larutan gula pada $n_\infty = \qty{6e20}{m^{-3}}$,
$D = \qty{5e-10}{m^2/s}$: berapa molekul yang tertangkap tiap sekon. (c)
Permukaannya diliputi $N_{\text{r}}$ reseptor penyerap kecil berjari-jari
$s$, sisanya memantulkan: tiap reseptor sendiri menangkap $\approx
4Ds n_\infty$ (sebuah cakram); tunjukkanlah bahwa seluruh selnya menangkap nyaris
maksimumnya begitu $N_{\text{r}} s \gg a$, yakni dengan sepersekian kecil
permukaannya terliput. (d) Ulaslah: mengapa sel dapat menanggung beribu
reseptor yang berbeda.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:particle-diffusion:12}
\emph{Menyelesaikannya pada sebuah petak.} Bagilah ruangnya menjadi sel berukuran $\Delta x$
dan waktunya menjadi langkah $\Delta t$; tuliskanlah $n_i^{k+1} = n_i^k + \alpha\,(n_{i+1}^k
- 2n_i^k + n_{i-1}^k)$ dengan $\alpha = D\Delta t/\Delta x^2$. (a) Benarkanlah
ia dari persamaannya. (b) Tafsirkanlah ia sebagai sebuah jalan acak ketika $\alpha =
1/2$. (c) Tunjukkanlah bahwa bagi $\alpha > 1/2$ sebuah kerapatan yang berselang-seling $+,-,+,-$
dari sel ke sel tumbuh: ketakstabilan. (d) Bagi profil boron
soal di bawah ($D = \qty{1.5e-17}{m^2/s}$, $\Delta x = \qty{10}{nm}$),
tentukanlah $\Delta t$ mantap terbesarnya dan jumlah langkahnya bagi satu jam.
\end{exercise}

\qty{7.6e-5}{m^2/s}$, $E_{\text{a}} = \qty{3.46}{eV}$; $k_B =
\qty{8.62e-5}{eV/K}$. Sebuah dosis $Q = \qty{1e18}{m^{-2}}$ boron telah
diendapkan dalam sebuah lapisan yang sangat tipis di permukaan sebuah wafer yang
doping latarnya $n_B = \qty{1e21}{m^{-3}}$; ia lalu dipanaskan
(``didorong masuk'') pada \qty{1100}{\celsius} selama satu jam. Permukaannya
memantulkan boronnya (tak ada yang lolos).
\begin{enumerate}
  \item Hitunglah $D$ pada \qty{1100}{\celsius} dan pada \qty{1000}{\celsius}.
        Sebesar faktor berapa ia berubah sepanjang \qty{100}{K} ini?
  \item Mengapa profil sesudah pendorongan masuknya adalah $n(x,t) = (Q/\sqrt{\pi
        Dt})\exp(-x^2/4Dt)$ dan bukan Gauss
        \cref{prop:b2:particle-diffusion:gaussian}? Periksalah bahwa
        integralnya pada $x > 0$ adalah $Q$.
  \item Konsentrasi permukaannya sesudah satu jam.
  \item \emph{Kedalaman sambungan} $x_j$ adalah tempat $n = n_B$: hitunglah.
  \item Bagaimana $x_j$ berubah jika pendorongan masuknya berlangsung empat jam?
        (Hati-hati: konsentrasi permukaannya berubah juga.)
  \item Perubahan nisbi $D$ bagi galat suhu \qty{10}{K};
        kendali suhu yang diperlukan agar $x_j$ tepat sampai \qty{1}{\%}.
  \item Pada suhu ruang, berapa $D$: taksirlah waktu bagi profilnya untuk
        beranjak sejarak satu jarak atom, lalu simpulkanlah.
\end{enumerate}

\textbf{Bagian II --- Pradeposisi.}
Dosisnya sendiri dimasukkan pada \qty{950}{\celsius} dari sebuah gas yang menahan
permukaannya pada batas kelarutannya $n_0 = \qty{2e26}{m^{-3}}$ selama
\qty{30}{min}.
\begin{enumerate}[resume]
  \item Tentukanlah $D$ pada \qty{950}{\celsius} dan $\sqrt{Dt}$ bagi \qty{30}{min}.
  \item Profil di akhir pradeposisinya; kedalaman yang padanya $n =
        n_B$ ($\operatorname{erfc}(3.2) \approx 6 \times 10^{-6}$).
  \item Dosis yang dimasukkan; bandingkanlah dengan $Q$ Bagian I.
  \item Mengapa orang mempradeposisi pada suhu yang lebih rendah lalu mendorong masuk pada
        suhu yang lebih tinggi?
  \item Benarkanlah bahwa selama pendorongan masuknya lapisan yang terpradeposisi dapat
        diperlakukan sebagai tipis takhingga.
\end{enumerate}

\textbf{Bagian III --- Sebuah parfum dalam sebuah ruangan.}
Setetes parfum (\qty{1}{mg}, massa molar \qty{150}{g/mol}) menguap
seketika di sudut sebuah ruangan yang tenang pada \qty{300}{K}, \qty{1}{bar}.
Ambillah diameter tumbukan molekul parfumnya dengan udara sebagai $d =
\qty{0.5}{nm}$ dan \omterm{def:b2:particle-diffusion:flux}{rapat cacah} udaranya $n_{\text{a}} = P/k_BT$.
\begin{enumerate}[resume]
  \item Jumlah molekul yang dilepaskan; \omterm{def:b2:particle-diffusion:flux}{rapat cacah} udaranya.
  \item Jalan bebas rerata $\ell = 1/(\sqrt2\,\pi d^2 n_{\text{a}})$ molekul
        parfumnya, dan kelajuan reratanya $v^* = \sqrt{8RT/\pi M}$.
  \item Taksirlah $D = \ell v^*/3$.
  \item Waktu untuk berdifusi \qty{5}{m}; dan jika udaranya hanyut pada
        \qty{0.1}{m/s}?
  \item Tuliskanlah Gauss tiga dimensinya bagi awannya (sebuah sudut:
        dindingnya memantulkan, kalikanlah dengan $8$). Pada sebuah titik \qty{1}{m} jauhnya,
        pada waktu berapa konsentrasinya maksimal?
  \item \omterm{def:b2:particle-diffusion:flux}{Rapat cacah} maksimumnya di sana; bandingkanlah dengan ambang
        pengindraan \qty{1e13}{m^{-3}}.
  \item Tepi awannya: pada jarak berapa kerapatannya $10^{-6}$
        nilai pusatnya sesudah satu hari?
  \item Jumlah tumbukan yang dialami sebuah molekul parfum dalam sehari.
\end{enumerate}

\textbf{Bagian IV --- Pejalan dan anak panah waktu.}
\begin{enumerate}[resume]
  \item Seorang pejalan membuat $N$ langkah $\pm\ell$ pada sebuah garis: tentukanlah simpangan
        rerata dan simpangan kuadrat reratanya; tunjukkanlah bahwa $\langle x^2\rangle = 2Dt$
        mengenali $D = \ell^2/2\tau$.
  \item Peluang bahwa pejalannya kembali tepat ke awalnya sesudah
        $N = 2$, $4$, $6$ langkah; ulaslah kecenderungannya.
  \item Filmkanlah sebuah awan yang berdifusi lalu putarlah filmnya mundur: apa yang Anda
        lihat, dan persamaan mana yang dilanggar?
  \item Jelaskanlah dalam satu paragraf bagaimana gerak molekul yang terbalikkan
        menghasilkan \omterm{thm:b2:particle-diffusion:equation}{persamaan difusi} yang takterbalikkan.
  \item Ringkaslah: skala panjangnya, skala waktunya, dan satu
        besaran yang membedakan sebuah gas, sebuah cairan dan sebuah zat padat panas.
\end{enumerate}
\end{problem}
