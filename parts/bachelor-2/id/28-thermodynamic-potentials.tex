\chapter{Potensial Termodinamika}\label{ch:b2:thermodynamic-potentials}

Regangkanlah sebuah karet gelang dengan cepat lalu sentuhkanlah ke bibir Anda: ia hangat;
biarkanlah ia mengerut dan ia sejuk. Gantungkanlah sebuah beban padanya lalu hangatkanlah dengan sebuah
pengering rambut: ia \emph{memendek}. Tak ada pegas baja yang berperilaku begitu, dan
alasannya adalah bahwa tegangan sebuah karet gelang bukanlah soal energi melainkan soal
entropi --- rantainya yang teregang mempunyai lebih sedikit cara untuk menata dirinya.
Untuk menalar tentang hal semacam itu orang perlu mengetahui apa yang cenderung dilakukan sebuah sistem pada
suhu tetap, atau pada suhu dan tekanan tetap, dan
berapa banyak kerja yang dapat ditarik darinya: pertanyaan yang dijawab energi dalam dan
entropi hanya dengan kikuk, karena keduanya berbicara dalam bahasa sebuah
sistem terkucil. Bab ini memperkenalkan kedua fungsi yang dibangun bagi
kondisi laboratoriumnya, yaitu \emph{\omterm{def:b2:thermodynamic-potentials:FG}{energi bebas} Helmholtz} $F = U -
TS$ dan \emph{\omterm{def:b2:thermodynamic-potentials:FG}{entalpi bebas} Gibbs} $G = H - TS$, menunjukkan bahwa keduanya
mengukur kerja yang dapat dipulihkan dan menurun menuju kesetimbangannya, menarik
dari diferensialnya hubungan Maxwell yang menautkan besaran yang
tak terpikirkan siapa pun berkaitan, lalu menerapkan semuanya pada
kesetimbangan dua fase, yang di sana $G$ memutuskan fase mana yang menang lalu
mengembalikan rumus Clapeyron.

\section{Kesamaan dasar dan peubahnya}

\begin{proposition}[Kesamaan dasar]\label{prop:b2:thermodynamic-potentials:identity}
Bagi sebuah sistem tertutup bersusunan tetap, energi dalamnya,
yang dipandang sebagai fungsi entropi dan volumenya, memenuhi
\[
\dd U = T\,\dd S - P\,\dd V, \qquad
T = \Big(\frac{\partial U}{\partial S}\Big)_V, \quad
P = -\Big(\frac{\partial U}{\partial V}\Big)_S ,
\]
dan \omterm{thm:b2:open-systems:firstlaw}{entalpinya} $H = U + PV$ memenuhi $\dd H = T\,\dd S + V\,\dd P$.
Tiap fungsinya, dalam \emph{peubah alami}\index{peubah
alami}-nya --- $(S,V)$ bagi $U$, $(S,P)$ bagi $H$ --- memuat seluruh
termodinamika sistemnya: turunan parsialnya memberi peubah
keadaan lainnya. Hukum kedua memilih kesetimbangan sebuah sistem
terkucil ($U$, $V$ tetap) sebagai keadaan yang $S$-nya maksimum.
\end{proposition}

\begin{proof}
Hukum pertama bagi sebuah transformasi terbalikkan, $\dd U = \delta Q_{\text{balik}}
+ \delta W_{\text{balik}} = T\,\dd S - P\,\dd V$, hanya melibatkan fungsi
keadaan dan karena itu berlaku bagi sebarang perubahan infinitesimal antara
keadaan setimbang. Tambahkanlah $\dd(PV)$ bagi $H$.
\end{proof}

\begin{remark}[Mengapa fungsi baru]\label{rem:b2:thermodynamic-potentials:why}
Percobaan tak dilakukan pada entropi tetap. Labu seorang kimiawan duduk pada
suhu ruangannya dan tekanan atmosfernya; sebuah
torak dalam sebuah termostat bekerja pada $T$ tetap. Orang menginginkan fungsi yang
\omterm{prop:b2:thermodynamic-potentials:identity}{peubah alaminya} $(T,V)$ dan $(T,P)$, dan yang memainkan, di bawah kondisi
itu, peran yang dimainkan $S$ bagi sebuah sistem terkucil: memberitahukan
ke arah mana sistemnya berkembang dan berapa banyak kerja yang dapat diberikannya. Sebuah
\emph{transformasi Legendre} --- mengurangkan $TS$ untuk menukar $S$ dengan $T$ sebagai
peubah bebasnya --- membangun keduanya.
\end{remark}

\section{Energi bebas dan entalpi bebas}

\begin{definition}[Energi bebas Helmholtz, entalpi bebas Gibbs]\label{def:b2:thermodynamic-potentials:FG}
\[
F = U - TS \quad\text{(\emph{energi bebas}\index{energi bebas})}, \qquad
G = H - TS = U + PV - TS \quad\text{(\emph{entalpi bebas}\index{entalpi bebas})}.
\]
Keduanya fungsi keadaan yang ekstensif, yang diukur dalam joule.
\end{definition}

\begin{theorem}[Diferensial dan besaran turunannya]\label{thm:b2:thermodynamic-potentials:dFdG}
\[
\dd F = -S\,\dd T - P\,\dd V, \qquad
\dd G = -S\,\dd T + V\,\dd P ;
\]
maka, dalam \omterm{prop:b2:thermodynamic-potentials:identity}{peubah alami} $(T,V)$ milik $F$ dan $(T,P)$ milik $G$,
\[
S = -\Big(\frac{\partial F}{\partial T}\Big)_V, \quad
P = -\Big(\frac{\partial F}{\partial V}\Big)_T, \qquad
S = -\Big(\frac{\partial G}{\partial T}\Big)_P, \quad
V = \Big(\frac{\partial G}{\partial P}\Big)_T ,
\]
dan (Gibbs--Helmholtz) $U = F - T(\partial F/\partial T)_V =
-T^2\,\partial(F/T)/\partial T$, $H = -T^2\,\partial(G/T)/\partial T$.
Mengetahui $F(T,V)$ atau $G(T,P)$ berarti mengetahui segalanya: persamaan
keadaannya, entropinya, energinya, kapasitas kalornya.
\end{theorem}

\begin{proof}
$\dd F = \dd U - T\dd S - S\dd T = -S\dd T - P\dd V$; demikian pula bagi $G$
dari $\dd H$. Bacalah turunan parsialnya; bagi Gibbs--Helmholtz,
$F - T\,\partial_TF = F + TS = U$ dan $\partial_T(F/T) = (T\partial_TF -
F)/T^2 = -U/T^2$.
\end{proof}

\begin{example}[Gas sempurna]\label{ex:b2:thermodynamic-potentials:perfect}
Bagi $n$ mol, $U = nc_{V}T + U_0$ dan $S = nc_V\ln T + nR\ln V + S_0$
memberi $F(T,V) = -nRT\ln V + f(T)$: lalu $P = -\partial_VF = nRT/V$
--- persamaan keadaannya keluar dari $F$ --- dan $S = -\partial_TF$
mengembalikan entropinya. Dalam peubah $(T,P)$, $G(T,P) = nRT\ln P +
g(T)$, sehingga $V = \partial_PG = nRT/P$, dan \omterm{def:b2:thermodynamic-potentials:FG}{entalpi bebas} \emph{molar}-nya
$\mu(T,P) = \mu^\circ(T) + RT\ln(P/P^\circ)$ --- yaitu \emph{potensial
kimia}\index{potensial kimia} gasnya, yang tumbuh dengan
tekanannya: gas mengalir dari $\mu$ tinggi ke yang rendah, di sini dari tekanan tinggi ke
tekanan rendah.
\end{example}

\section{Kerja maksimum dan arah perkembangannya}

\begin{theorem}[Transformasi monotermal]\label{thm:b2:thermodynamic-potentials:monothermal}
Sebuah sistem yang mempertukarkan kalor dengan sebuah termostat tunggal pada $T_0$, dan yang
suhu awal dan akhirnya sama dengan $T_0$, menerima selama sebarang
transformasi kerja
\[
W \ge \Delta F , \qquad\text{yakni}\qquad W_{\text{pulih}} = -W \le -\Delta F :
\]
penurunan $F$ adalah \emph{kerja maksimum yang dapat dipulihkan} dari
sistemnya pada suhu tetap (yang dicapai bagi transformasi yang terbalikkan) ---
dari situlah energi ``bebas''nya: bagian $U$ yang dapat diubah menjadi kerja,
sisanya, $TS$, terikat. Jika lagi pula transformasinya monobarik
(tekanan $P_0$ di luarnya, tekanan awal dan akhirnya $P_0$), kerja
\emph{selain kerja atmosfernya} (\omterm{def:b2:open-systems:cv}{kerja berguna}, misalnya listrik)
memenuhi $W_{\text{u}} \ge \Delta G$: $-\Delta G$ adalah \omterm{def:b2:open-systems:cv}{kerja berguna} maksimum
yang dapat dipulihkan pada $T$ dan $P$ tetap (kerja sebuah sel bahan bakar, sebuah
baterai).
\end{theorem}

\begin{proof}
Hukum pertama: $\Delta U = W + Q$. Hukum kedua dengan satu termostat: $\Delta S
= Q/T_0 + S_{\text{c}}$, $S_{\text{c}} \ge 0$, jadi $Q \le T_0\Delta S$. Maka
$W = \Delta U - Q \ge \Delta U - T_0\Delta S = \Delta(U - T_0S) = \Delta F$, dengan memakai
$T = T_0$ di kedua ujungnya. Dengan kerja atmosfernya $-P_0\Delta V$
dipisahkan, $W = W_{\text{u}} - P_0\Delta V$ dan $W_{\text{u}} \ge \Delta U + P_0
\Delta V - T_0\Delta S = \Delta G$.
\end{proof}

\begin{theorem}[Kriteria perkembangan dan kesetimbangan]\label{thm:b2:thermodynamic-potentials:criteria}
Sebuah sistem yang ditahan pada $T$ dan $V$ tetap dan tak menerima kerja hanya dapat
berkembang sedemikian sehingga \emph{$F$ menurun}; kesetimbangannya adalah keadaan yang
$F$-nya minimum. Sebuah sistem yang ditahan pada $T$ dan $P$ tetap dan tak menerima \omterm{def:b2:open-systems:cv}{kerja
berguna} hanya dapat berkembang sedemikian sehingga \emph{$G$ menurun}; kesetimbangannya adalah
keadaan yang $G$-nya minimum. (Secara lebih umum, bagi sebuah sistem yang bersentuhan dengan
sebuah sekeliling pada $T_0$, $P_0$, besaran $U - T_0S + P_0V$ menurun.)
\end{theorem}

\begin{proof}
Ambillah $W = 0$ (berturut-turut $W_{\text{u}} = 0$) dalam teorema sebelumnya: $\Delta F
\le 0$ (berturut-turut $\Delta G \le 0$), dengan kesamaan bagi perkembangan yang terbalikkan, yakni
setimbang.
\end{proof}

\begin{example}[Pemuaian isotermal, yang nyata dan yang ideal]\label{ex:b2:thermodynamic-potentials:expansion}
Satu mol gas pada \qty{300}{K} memuai dari \qty{1}{L} ke \qty{10}{L} dalam
sentuhan dengan termostatnya: $\Delta F = -RT\ln 10 = \qty{-5.7}{kJ}$. Sebuah
pemuaian terbalikkan memulihkan \qty{5.7}{kJ}; sebuah pemuaian melawan
atmosfernya pada \qty{1}{bar} hanya memulihkan $P_0\Delta V = \qty{0.9}{kJ}$; sebuah
pemuaian bebas ke dalam hampa, tak ada apa pun. \qty{5.7}{kJ}-nya adalah maksimum
yang dapat diberikan gasnya dan termostatnya bagi perubahan keadaan ini, betapa pun
cerdik mesinnya.
\end{example}

\section{Hubungan Maxwell}

\begin{theorem}[Hubungan Maxwell]\label{thm:b2:thermodynamic-potentials:maxwell}
Karena $\dd F$ dan $\dd G$ merupakan diferensial eksak (teorema Schwarz
tentang turunan kedua campurannya),
\[
\Big(\frac{\partial S}{\partial V}\Big)_T = \Big(\frac{\partial P}{\partial T}\Big)_V,
\qquad
\Big(\frac{\partial S}{\partial P}\Big)_T = -\Big(\frac{\partial V}{\partial T}\Big)_P .
\]
Keduanya menyatakan yang tak terukur (bagaimana entropinya berubah dengan volume atau
tekanan pada suhu tetap) lewat persamaan keadaannya. Dua
akibatnya:
\[
\Big(\frac{\partial U}{\partial V}\Big)_T = T\Big(\frac{\partial P}{\partial T}\Big)_V - P,
\qquad
C_P - C_V = T\Big(\frac{\partial P}{\partial T}\Big)_V\Big(\frac{\partial V}{\partial T}\Big)_P
= \frac{TV\alpha^2}{\kappa_T} ,
\]
dengan $\alpha = V^{-1}(\partial V/\partial T)_P$ yaitu koefisien muai dan
$\kappa_T = -V^{-1}(\partial V/\partial P)_T$ kemampatan isotermalnya.
\end{theorem}

\begin{proof}
$\partial^2F/\partial T\partial V$ yang dihitung dalam kedua urutannya: $-\partial_VS =
-\partial_TP$. Sama bagi $G$. Lalu $\dd U = T\dd S - P\dd V$ pada $T$ tetap
memberi $(\partial_VU)_T = T(\partial_VS)_T - P$; dan $C_P - C_V$ menyusul
dari menuliskan $S(T,V(T,P))$: $(\partial_TS)_P = (\partial_TS)_V +
(\partial_VS)_T(\partial_TV)_P$, yang dikalikan $T$.
\end{proof}

\begin{example}[Apa yang diungkapkan hubungannya]\label{ex:b2:thermodynamic-potentials:reveal}
Bagi sebuah gas sempurna, $T(\partial_TP)_V - P = 0$: energinya tak bergantung
pada volumenya (hukum Joule yang dipulihkan) dan $C_P - C_V = nR$. Bagi sebuah gas van
der Waals, $(\partial_VU)_T = a/V_{\text{m}}^2$: yaitu tekanan dalam
tarikannya. Bagi air cair, $(\partial_TP)_V = \alpha/\kappa_T = 2.1
\times 10^{-4}/4.6 \times 10^{-10} = \qty{4.6e5}{Pa/K}$: memanaskan air dalam sebuah
bejana tegar yang penuh sebesar satu kelvin menaikkan tekanannya sebesar \qty{4.6}{bar}
(jangan pernah menyegel sebuah botol yang penuh), dan $(\partial_VU)_T \approx \qty{1400}{bar}$ ---
yaitu kepaduan cairannya; namun $c_P - c_V = Tv\alpha^2/\kappa_T \approx
\qty{30}{J/kg/K}$, kurang daripada $1\%$ dari $c_P$: bagi materi mampat kedua
kapasitas kalornya nyaris berimpit.
\end{example}

\begin{proposition}[Keelastikan entropik karet]\label{prop:b2:thermodynamic-potentials:rubber}
Bagi sebuah gelang sepanjang $L$ di bawah tegangan $f$, $\dd U = T\dd S + f\dd L$
dan $\dd F = -S\dd T + f\dd L$, sehingga $(\partial S/\partial L)_T =
-(\partial f/\partial T)_L$ dan $(\partial U/\partial L)_T = f - T(\partial f/
\partial T)_L$. Karet menuruti, sampai hampiran yang baik, $f = T\,\varphi(L)$
dengan $\varphi > 0$ bagi $L > L_0$: lalu $(\partial U/\partial L)_T = 0$ ---
energinya tak berubah saat diregangkan --- dan $(\partial S/\partial L)_T =
-\varphi(L) < 0$: tegangannya bersifat \emph{entropik}, $f = -T(\partial S/\partial
L)_T$, gelangnya menarik kembali karena rantainya yang teregang mempunyai lebih sedikit
konfigurasi. Akibatnya: diregangkan secara isotermal ia melepaskan kalor
($Q = T\Delta S < 0$); diregangkan secara adiabatik ia menghangat; dipanaskan di bawah
beban tetap ia mengerut ($f/T$ tetap berarti $\varphi(L)$ tetap, jadi pada
$f$ tetap, $\varphi$ harus turun ketika $T$ naik).
\end{proposition}

\begin{proof}
Hubungan Maxwell dari $\dd F$; sisanya penyulihan. Asal
mikroskopisnya adalah jalan acak \cref{ch:b2:particle-diffusion}:
sebuah rantai $N$ mata sepanjang $a$ mempunyai jarak ujung ke ujungnya yang
tersebar sebagai sebuah Gauss bervariansi $Na^2$, sehingga jumlah
konfigurasi pada rentangan $L$ adalah $\Omega \propto \eu^{-L^2/2Na^2}$, $S = S_0 -
k_BL^2/2Na^2$ dan $f = k_BTL/Na^2$: sebuah pegas yang kekakuannya
\emph{sebanding dengan suhunya}.
\end{proof}

\section{Kesetimbangan antara dua fase}

\begin{theorem}[Syarat hidup berdampingan]\label{thm:b2:thermodynamic-potentials:coexistence}
Sebuah zat murni pada $T$ dan $P$ tetap, yang terbelah antara dua fase bermassa
$m_1$, $m_2$ dengan \omterm{def:b2:thermodynamic-potentials:FG}{entalpi bebas} tiap satuan massa $g_1(T,P)$,
$g_2(T,P)$, mempunyai $G = m_1g_1 + m_2g_2$; pada kesetimbangannya $G$ minimum, jadi
\begin{itemize}
\item jika $g_1 < g_2$ seluruh materinya masuk ke fase 1 (fase yang
      mantap adalah yang $g$-nya \emph{lebih rendah});
\item kedua fasenya hanya hidup berdampingan di tempat
      \[
      g_1(T,P) = g_2(T,P) ,
      \]
      sebuah hubungan antara $T$ dan $P$: yaitu kurva hidup berdampingan diagram
      fasenya.
\end{itemize}
Sepanjang kurva itu (Clapeyron),
\[
\frac{\dd P}{\dd T} = \frac{s_2 - s_1}{v_2 - v_1} = \frac{L_{1\to2}}{T\,(v_2 - v_1)} .
\]
\end{theorem}

\begin{proof}
$\dd G = (g_1 - g_2)\dd m_1$ pada $T$, $P$ tetap (dengan $m_1 + m_2$ tetap):
$G$ menurun dengan memindahkan massa ke fase yang $g$-nya lebih rendah, sampai fase
itu sendirian atau sampai $g_1 = g_2$. Turunkanlah $g_1 = g_2$ sepanjang
kurvanya dengan $\dd g = -s\,\dd T + v\,\dd P$ bagi tiap fasenya: $-s_1\dd T +
v_1\dd P = -s_2\dd T + v_2\dd P$; dan $s_2 - s_1 = L/T$ karena transisinya
pada $T$, $P$ tetap terbalikkan. Jilid Tahun 1 memperoleh rumus ini
dari sebuah daur Carnot; di sini ia jatuh keluar dari $G$.
\end{proof}

Peleburan: $L = \qty{334}{kJ/kg}$, $v_{\text{cair}} - v_{\text{es}} = -\qty{9.1e-5}{m^3/kg}$:

tekanan, tetapi \qty{130}{bar} tiap kelvin: \qty{70}{bar} seorang peluncur es menurunkan
titik leburnya hanya setengah derajat (peluncuran es bekerja lewat gesekan dan

\qty{0}{\celsius}, air cair mempunyai $g_{\text{cair}} - g_{\text{es}} \approx

awan tetap cair sampai \qty{-40}{\celsius} karena kekurangan inti.
\end{example}

\begin{remark}[Van der Waals dan konstruksi Maxwell]\label{rem:b2:thermodynamic-potentials:maxwellconstruction}
Persamaan van der Waals $(P + a/v^2)(v - b) = RT$ memberi, di bawah suhu
kritisnya $T_{\text{c}} = 8a/27Rb$, isoterm bersimpal. Fluida
yang sungguhan tak menuruti simpalnya: pada $P_{\text{jenuh}}(T)$ ia melompat dari
cabang cairannya ke cabang uapnya, dan $P_{\text{jenuh}}$ ditetapkan $g_{\text{cair}}
= g_{\text{uap}}$, yakni $\int_{\text{cair}}^{\text{uap}}v\,\dd P = 0$ sepanjang
simpalnya: garis mendatarnya memotong \emph{luas yang sama}. Bagian
simpalnya yang $(\partial P/\partial v)_T > 0$ takstabil secara mekanis;
bagian antara keduanya dan garisnya metastabil --- yaitu cairan lewat panas
yang ``meletup'' dalam sebuah gelombang mikro, dan uap lewat jenuh
kamar awannya.
\end{remark}

\begin{method}[Memakai potensialnya]\label{meth:b2:thermodynamic-potentials:method}
(1) Kenalilah kendalanya: terkucil $\to$ $S$ maksimum; $T,V$ tetap $\to$
$F$ minimum; $T,P$ tetap $\to$ $G$ minimum. (2) Kerja maksimumnya: $-\Delta F$ pada $T$
tetap, $-\Delta G$ \omterm{def:b2:open-systems:cv}{kerja berguna} pada $T,P$ tetap. (3) Memerlukan sebuah turunan
$S$ pada $T$ tetap? Pakailah sebuah hubungan Maxwell dan persamaan keadaannya.
(4) Dua fase: bandingkanlah $g$-nya; hidup berdampingannya $g_1 = g_2$; Clapeyron bagi
kemiringannya; tanda $\Delta v$ bagi tanda kemiringannya. (5) \omterm{ex:b2:thermodynamic-potentials:perfect}{Potensial
kimianya} $\mu = g$ tiap mol: materi mengalir menuju $\mu$ yang lebih rendah.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:1}
Bagi $n$ mol gas sempurna, tuliskanlah $F(T,V)$ dan $G(T,P)$ sampai pada
fungsi $T$ saja; pulihkanlah persamaan keadaannya dari masing-masingnya; tunjukkanlah
bahwa $S = -\partial_TF$ memberi entropi yang telah diketahui.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:2}
Satu mol gas, \qty{300}{K}, dari \qty{1}{L} ke \qty{10}{L} dalam sebuah
termostat: tentukanlah $\Delta F$, $\Delta U$, $\Delta S$; kerja yang dipulihkan jika terbalikkan;
jika melawan \qty{1}{bar}; jika ke dalam hampa; dan entropi yang tercipta pada tiap
halnya.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:3}
Air dan es pada \qty{1}{bar}: tunjukkanlah bahwa $g_{\text{cair}} - g_{\text{es}} \approx
(L/T_{\text{lebur}})(T_{\text{lebur}} - T)$ dekat \qty{0}{\celsius} lalu hitunglah ia
pada \qty{-5}{} dan $+\qty{5}{\celsius}$ ($L = \qty{334}{kJ/kg}$). Fase mana yang
mantap pada tiap halnya; mustahilkah yang lainnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermodynamic-potentials:4}
Kemiringan Clapeyron air: peleburannya ($\Delta v = -\qty{9.1e-5}{m^3/kg}$) dan
penguapannya pada \qty{100}{\celsius} ($L = \qty{2.26}{MJ/kg}$, $\Delta v =
\qty{1.67}{m^3/kg}$). Tentukanlah titik leburnya di bawah \qty{70}{bar}; titik didihnya pada
\qty{0.7}{bar} dan pada \qty{2}{bar}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:5}
\emph{Maxwell.} (a) Turunkanlah kedua hubungan Maxwell babnya
dan kedua hubungan dari $\dd U$ dan $\dd H$. (b) Tunjukkanlah $(\partial U/\partial V)_T =
T(\partial P/\partial T)_V - P$; nilailah ia bagi sebuah gas sempurna dan sebuah gas van
der Waals. (c) Air cair: $\alpha = \qty{2.1e-4}{K^{-1}}$, $\kappa_T =
\qty{4.6e-10}{Pa^{-1}}$: tentukanlah kenaikan tekanannya tiap kelvin dalam sebuah bejana tegar yang penuh,
dan $(\partial U/\partial V)_T$. (d) Sebuah botol kaca yang penuh dan tersegel dihangatkan
sebesar \qty{10}{K}: ulaslah.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:6}
\emph{$C_P - C_V$.} (a) Turunkanlah $C_P - C_V = TV\alpha^2/\kappa_T$. (b)
Bagi gas sempurna. (c) Bagi air pada \qty{300}{K}; bagi tembaga ($\alpha = \qty{5e-5}{K^{-1}}$,
$\kappa_T = \qty{7e-12}{Pa^{-1}}$, $\rho = \qty{8900}{kg/m^3}$); bandingkanlah dengan
$c_P$ (\qty{4180}{} dan \qty{385}{J/kg/K}). (d) Mengapa orang dapat berbicara tentang
``sang'' kapasitas kalor sebuah zat padat?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:7}
\emph{Karet.} Sebuah gelang menuruti $f = CT(L - L_0)$ dengan $C = \qty{1}{N/m/K}$.
(a) Tentukanlah $(\partial S/\partial L)_T$ dan $(\partial U/\partial L)_T$. (b) Perubahan
entropinya, kalornya dan kerjanya ketika diregangkan secara isotermal pada \qty{300}{K} dari
$L_0$ ke $L_0 + \qty{0.1}{m}$. (c) Diregangkan secara adiabatik sebagai gantinya (kapasitas
kalornya \qty{4}{J/K}): berapa perubahan suhunya. (d) Digantungi sebuah beban tetap
lalu dihangatkan sebesar \qty{30}{K}: berapa perubahan nisbi $L - L_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:8}
\emph{Penguapan sebuah genangan.} \omterm{ex:b2:thermodynamic-potentials:perfect}{Potensial kimia} uap air
pada tekanan parsial $p$ adalah $\mu_{\text{v}} = \mu_{\text{v}}^\circ(T) +
RT\ln(p/p^\circ)$, dan sama dengan milik cairannya ketika $p = p_{\text{jenuh}}(T)$.
(a) Tunjukkanlah bahwa cairannya menguap ketika $p < p_{\text{jenuh}}$ dan uapnya
mengembun ketika $p > p_{\text{jenuh}}$; definisikanlah kelembapan nisbinya. (b)
Tentukanlah $\mu_{\text{cair}} - \mu_{\text{v}}$ pada kelembapan $50\%$, \qty{293}{K}. (c) Mengapa
cucian mengering bahkan pada \qty{20}{\celsius}, dan tak pernah pada kelembapan $100\%$?
(d) Embun: pada permukaan yang mana ia terbentuk lebih dahulu, dan mengapa?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermodynamic-potentials:9}
\emph{Van der Waals.} $(P + a/v^2)(v - b) = RT$ (molar). (a) Carilah
titik kritisnya ($\partial_vP = \partial_v^2P = 0$): $v_{\text{c}} = 3b$,
$T_{\text{c}} = 8a/27Rb$, $P_{\text{c}} = a/27b^2$. (b) Karbon dioksida,
$T_{\text{c}} = \qty{304}{K}$, $P_{\text{c}} = \qty{73.8}{bar}$: tentukanlah $a$ dan $b$. (c)
Bernalarkanlah aturan luas sama Maxwell dari $g$. (d) Bagian simpal yang mana
yang takstabil, yang mana yang metastabil, dan gejala apa yang menunjukkan yang
metastabil?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:10}
\emph{Clausius--Clapeyron yang diintegralkan.} Bagi penguapannya, abaikanlah $v_{\text{cair}}$
lalu perlakukanlah uapnya sebagai sempurna. (a) Tunjukkanlah $\dd\ln P/\dd T = L_{\text{m}}/RT^2$
($L_{\text{m}}$ molar) lalu integralkanlah bagi $L_{\text{m}}$ tetap. (b) Air,
$L_{\text{m}} = \qty{40.7}{kJ/mol}$: tentukanlah $P_{\text{jenuh}}$ pada \qty{20}{\celsius} dari
\qty{1}{bar} pada \qty{100}{\celsius}; bandingkanlah dengan \qty{23}{mbar} yang terukur.
(c) Ramalkanlah tekanan titik tripelnya (\qty{0.01}{\celsius}) lalu bandingkanlah
dengan \qty{611}{Pa}. (d) Tunjukkanlah bahwa hukum yang sama menyusul dari Gibbs--Helmholtz
yang diterapkan pada $\Delta g$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:11}
\emph{Osmosis.} \omterm{ex:b2:thermodynamic-potentials:perfect}{Potensial kimia} sebuah pelarut diturunkan sebuah
zat terlarut yang encer: $\mu = \mu^\circ - RTx_{\text{s}}$ ($x_{\text{s}}$ yaitu fraksi
mol zat terlarutnya). Sebuah membran hanya meloloskan pelarutnya; pelarut murni di
satu sisi, larutan di sisi lain, $T$ yang sama. (a) Tunjukkanlah bahwa kesetimbangannya
menuntut sebuah tekanan tambahan $\Pi$ pada larutannya dengan $\Pi v_{\text{m}} =
RTx_{\text{s}}$, yakni $\Pi = cRT$ (van 't Hoff). (b) Air laut, $c \approx
\qty{1.1}{mol/L}$ partikel terlarut, \qty{293}{K}: tentukanlah $\Pi$. (c) Kerja minimum
untuk mengambil satu meter kubik air tawar; bandingkanlah dengan pabrik yang sungguhan
(\qty{3}{kWh/m^3}). (d) Mengapa sel tumbuhan pecah dalam air murni dan
mengerut dalam air garam?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermodynamic-potentials:12}
\emph{Rantainya.} Sebuah rantai polimer $N$ mata sepanjang $a$: jumlah
konfigurasi dengan vektor ujung ke ujung sepanjang $L$ sepanjang
gelangnya adalah $\Omega \propto \eu^{-L^2/2Na^2}$ (yaitu jalan acaknya). (a) Tentukanlah $S(L)$
dan, pada $U$ tetap, tegangannya $f = -T\,\partial S/\partial L$. (b) Kekakuan
rantainya dan kebergantungan suhunya. (c) $N = 1000$, $a =
\qty{0.5}{nm}$, \qty{300}{K}, $L = \qty{10}{nm}$: tentukanlah $f$; berapa banyak rantai yang
terparalel memberi \qty{30}{N}? (d) Taksirlah \omterm{prop:b2:waves-on-strings:chain}{modulus Young} sebuah karet
dengan $10^{26}$ rantai tiap meter kubik, masing-masingnya ber-$N = 100$.
\end{exercise}

\section{Soal: Karet gelang dan diagram fase air}

\begin{problem}[{Soal akhir pekan --- entropi yang menarik, dan fase
yang bersaing}]\label{pb:b2:thermodynamic-potentials:1}
\textbf{Bagian I --- Karet gelangnya.}
Sebuah gelang bermassa \qty{2}{g} (kapasitas kalornya \qty{4}{J/K}) mempunyai tegangan
$f = CT(L - L_0)$ dengan $C = \qty{1}{N/m/K}$; pada \qty{300}{K} dan $L - L_0 =
\qty{0.1}{m}$, $f = \qty{30}{N}$.
\begin{enumerate}
  \item Tuliskanlah $\dd U$ dan $\dd F$ bagi gelangnya ($\dd U = T\dd S + f\dd L$).
  \item Turunkanlah hubungan Maxwellnya $(\partial S/\partial L)_T = -(\partial
        f/\partial T)_L$.
  \item Tunjukkanlah bahwa $(\partial U/\partial L)_T = 0$ bagi gelang ini: apa
        artinya?
  \item Peregangan isotermal dari $L_0$ ke $L_0 + \qty{0.1}{m}$ pada
        \qty{300}{K}: tentukanlah $\Delta S$, kalor $Q$, kerja $W$; periksalah $\Delta U = 0$.
  \item Peregangan yang sama yang dilakukan secara adiabatik: berapa perubahan suhunya.
  \item Gelangnya menahan sebuah beban tetap; ia dihangatkan sebesar \qty{30}{K}: berapa
        $L - L_0$ barunya. Lukiskanlah ``mesin karet''-nya (sebuah roda berjeruji
        karet yang dipanaskan pada satu sisinya).
  \item Sebuah rantai $N$ mata sepanjang $a$ mempunyai $\Omega \propto \eu^{-L^2/2Na^2}$:
        tentukanlah $S(L)$ dan $f(L,T)$; mengapa kekakuannya tumbuh dengan $T$?
  \item $N = 1000$, $a = \qty{0.5}{nm}$, $L = \qty{10}{nm}$, \qty{300}{K}: tentukanlah $f$
        bagi satu rantai dan jumlah rantai bagi \qty{30}{N}.
  \item Mengapa sebuah pegas baja berperilaku sebaliknya (tegangannya turun
        dengan $T$ pada panjang tetap)?
\end{enumerate}

\textbf{Bagian II --- Diagram fase air.}
Data: $L_{\text{lebur}} = \qty{334}{kJ/kg}$, $v_{\text{es}} = \qty{1.091e-3}{m^3/kg}$,
$v_{\text{cair}} = \qty{1.000e-3}{m^3/kg}$; $L_{\text{uap}} = \qty{2.26}{MJ/kg}$ pada
\qty{100}{\celsius}, $v_{\text{uap}} = \qty{1.67}{m^3/kg}$; $L_{\text{m}} = \qty{40.7}{kJ/mol}$;
titik tripelnya \qty{0.01}{\celsius}, \qty{611}{Pa}; titik kritisnya \qty{647}{K},
\qty{221}{bar}.
\begin{enumerate}[resume]
  \item Sketsalah $g(T)$ bagi es, cairan dan uapnya pada \qty{1}{bar}; bernalarkanlah
        urutan kemiringannya lalu tempatkanlah transisinya.
  \item Turunkanlah rumus Clapeyron dari $g_1 = g_2$.
  \item Kemiringan kurva peleburannya; tandanya, dan mengapa ia tak lazim.
  \item Seorang peluncur es: \qty{700}{N} pada \qty{1}{cm^2}: berapa perubahan titik
        leburnya; simpulkanlah tentang penjelasan populer peluncuran esnya.
  \item Di bawah \qty{3}{km} es ($\rho = \qty{917}{kg/m^3}$): berapa tekanan dan
        titik leburnya; apa akibatnya bagi lapisan esnya.
  \item Kemiringan kurva penguapannya pada \qty{100}{\celsius}; titik didihnya
        pada \qty{0.7}{bar} dan pada \qty{2}{bar}.
  \item Integralkanlah Clausius--Clapeyron dengan uap yang sempurna dan
        $L_{\text{m}}$ tetap; tentukanlah $P_{\text{jenuh}}$ pada \qty{20}{\celsius} dan pada
        titik tripelnya; ulaslah ketelitiannya.
  \item Pada titik tripelnya, bandingkanlah kemiringan kurva sublimasi dan
        penguapannya ($L_{\text{sub}} = L_{\text{lebur}} + L_{\text{uap}}$).
  \item Air lewat dingin pada \qty{-10}{\celsius}: tentukanlah $g_{\text{cair}} - g_{\text{es}}$;
        mengapa tetes awan dapat tetap cair sampai \qty{-40}{\celsius}?
  \item Apa yang terjadi pada $\Delta v$, $L$ dan kemiringannya di titik
        kritisnya; apa itu sebuah fluida superkritis?
\end{enumerate}

\textbf{Bagian III --- Terapan.}
\begin{enumerate}[resume]
  \item Jelaskanlah konstruksi luas sama Maxwell pada sebuah isoterm van der
        Waals dari kesinambungan $g$.
  \item Air lewat panas dalam sebuah gelombang mikro ``meletup'': cabang isoterm
        yang mana, dan mengapa berbahaya?
  \item Pengeringan beku: lukiskanlah lintasannya pada diagram $(P,T)$ dan
        mengapa airnya pergi tanpa melebur.
  \item Mengapa sepiring air menguap pada \qty{20}{\celsius} dalam udara kering
        lalu berhenti dalam udara jenuh --- dalam hal \omterm{ex:b2:thermodynamic-potentials:perfect}{potensial kimianya}?
  \item Sebuah panci presto pada \qty{2}{bar}: mengapa makanannya matang jauh
        lebih cepat (laju kimianya kira-kira berlipat dua tiap
        \qty{10}{K})?
  \item Ringkaslah: potensial mana bagi kendala yang mana, dan apa yang
        dibeli hubungan Maxwellnya.
\end{enumerate}
\end{problem}
