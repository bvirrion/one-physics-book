\chapter{Radiasi Dwikutub dan Hamburan}\label{ch:b2:dipole-radiation}

Mengapa langit biru, dan mengapa ia paling biru pada sudut siku-siku terhadap Matahari
--- lalu terpolarisasi di sana, seperti yang diketahui lebah dan juru foto? Mengapa sebuah
tiang radio harus setinggi berpuluh meter, dan mengapa ia tak memancarkan
apa pun lurus ke atas? Kedua pertanyaannya berjawaban sama: sebuah
muatan listrik yang berayun memancarkan \omterm{thm:b2:plane-waves-polarization:wave}{gelombang elektromagnetik}, dengan
daya yang tumbuh sebagai pangkat empat frekuensinya dan pola
yang lenyap sepanjang sumbu ayunannya. Bab ini menyatakan
medan \emph{\omterm{thm:b2:dipole-radiation:field}{dwikutub berayun}}, sistem pemancar yang paling sederhana,
lalu menarik darinya antena dan radiasi sebuah muatan yang dipercepat,
lalu menerapkannya pada molekul udara, yang masing-masingnya sebuah antena mungil
yang memancarkan ulang cahaya matahari: \omterm{prop:b2:dipole-radiation:rayleigh}{hamburan Rayleigh}.

\section{Medan sebuah dwikutub berayun}

\begin{theorem}[Medan radiasi sebuah dwikutub listrik]\label{thm:b2:dipole-radiation:field}
\index{dwikutub berayun}\index{zona radiasi}\index{penundaan}
Sebuah dwikutub bermomen $\vect p(t) = p_0\cos\omega t\,\vect e_z$, berukuran $a$, di
titik asal, memancar. Dalam \emph{\omterm{thm:b2:dipole-radiation:field}{zona radiasi}} --- pada jarak $r \gg
\lambda = 2\pi c/\omega \gg a$ --- medannya, dalam koordinat bola $(r,
\theta, \varphi)$ terhadap sumbu dwikutubnya, adalah
\[
\vect E(M, t) = -\frac{\mu_0\,\omega^2p_0}{4\pi}\,\frac{\sin\theta}r\,\cos\bigl(\omega(t - r/c)\bigr)\,\vect e_\theta ,
\qquad
\vect B = \frac{\vect e_r\wedge\vect E}c :
\]
sebuah \omterm{prop:b2:sound-waves:spherical}{gelombang bola}, yang setempat bidang --- $(\vect E, \vect B, \vect e_r)$ sebuah
triad tangan kanan dengan $B = E/c$ --- \emph{tertunda} sebesar waktu tempuh
$r/c$, beramplitudo yang turun sebagai $1/r$, sebanding dengan percepatan
$\ddot p = -\omega^2p$ dwikutubnya dan dengan $\sin\theta$: maksimal dalam
bidang khatulistiwanya, nol sepanjang sumbunya; terpolarisasi dalam bidang
yang memuat sumbunya.
\end{theorem}
\admitted

\begin{remark}[Membaca rumusnya]\label{rem:b2:dipole-radiation:reading}
Penyelesaian eksak \omterm{thm:b2:maxwell-equations:maxwell}{persamaan Maxwell} (dari potensial
tertunda, di luar jilid ini) memuat juga suku dalam $1/r^2$ dan
$1/r^3$; yang terakhir adalah medan dwikutub kuasi-statik jilid Tahun 1,
$\vect E \propto p/4\pi\varepsilon_0r^3$, yang menguasai dalam \emph{zona
dekat} $r \ll \lambda$; pada $r \sim \lambda/2\pi$ keduanya sebanding, di luarnya
hanya suku $1/r$ yang bertahan. Tiap faktornya mempunyai alasan. $1/r$:
daya yang menembus sebuah bola, $\propto E^2r^2$, tak boleh bergantung pada $r$. $\omega^2p_0$:
medan sebuah muatan yang diam atau bergerak beraturan tidak memancar; yang
memancar adalah \emph{percepatan}-nya. $\sin\theta$: seorang pengamat pada
sumbunya melihat muatannya datang dan pergi sepanjang garis pandangnya, tanpa
percepatan melintang; dalam bidang khatulistiwanya seluruh
percepatannya melintang. Penundaannya: medan di $M$ pada saat $t$
mencerminkan apa yang dilakukan dwikutubnya pada $t - r/c$.
\end{remark}

\section{Daya terpancar}

\begin{proposition}[Daya sebuah dwikutub; rumus Larmor]\label{prop:b2:dipole-radiation:power}
\index{daya terpancar (dwikutub)}\index{rumus Larmor}\index{pola radiasi}
\omterm{thm:b2:poynting-vector:poynting}{Vektor Poynting} rerata medan dwikutubnya bersifat radial,
\[
\langle\vect\Pi\rangle = \frac{\mu_0\,\omega^4p_0^2}{32\pi^2c}\,\frac{\sin^2\theta}{r^2}\,\vect e_r ,
\]
dan daya rerata total yang dipancarkan adalah
\[
\mathcal P = \frac{\mu_0\,\omega^4p_0^2}{12\pi c} = \frac{p_0^2\omega^4}{12\pi\varepsilon_0c^3} .
\]
Bagi sebuah muatan tunggal $q$ berpercepatan $a$ (tak relativistik),
daya sesaatnya adalah $\mathcal P = q^2a^2/6\pi\varepsilon_0c^3$ (Larmor).
\end{proposition}

\begin{proof}
$\langle\Pi\rangle = \langle E^2\rangle/\mu_0c$ dengan $\langle\cos^2\rangle = \tfrac12$;
integralkanlah di seluruh bolanya:
\[
\int\sin^2\theta\,r^2\dd\Omega = 2\pi r^2\int_0^\pi\sin^3\theta\,\dd\theta = \tfrac83\pi r^2 ,
\qquad
\mathcal P = \frac{\mu_0\omega^4p_0^2}{32\pi^2c}\cdot\frac{8\pi}3 = \frac{\mu_0\omega^4p_0^2}{12\pi c} .
\]
Larmor: $p = qz$, jadi $\ddot p = qa$;
dengan mengganti reratanya $\tfrac12\omega^4p_0^2 = \langle\ddot p^2\rangle$ oleh
yang sesaat $q^2a^2$ diperoleh $\mathcal P = \mu_0q^2a^2/6\pi c$.
\end{proof}

\begin{example}[Sebuah antena dan sebuah kawat]\label{ex:b2:dipole-radiation:antenna}
\index{antena}\index{hambatan radiasi}\index{dwikutub setengah gelombang}
Sebuah kawat pendek sepanjang $\ell \ll \lambda$ yang mengemban arus $I_0\cos\omega t$ adalah
dwikutub dengan $\dot p = I_0\ell\cos\omega t$, yakni $p_0 = I_0\ell/\omega$: ia memancarkan
$\mathcal P = \mu_0\omega^2I_0^2\ell^2/12\pi c = \tfrac12R_{\text{rad}}I_0^2$ dengan
\emph{\omterm{ex:b2:dipole-radiation:antenna}{hambatan radiasi}} $R_{\text{rad}} = \dfrac{2\pi}3\sqrt{\dfrac{\mu_0}{\varepsilon_0}}
\Bigl(\dfrac\ell\lambda\Bigr)^2 \approx 790\,(\ell/\lambda)^2\ \Omega$ --- dayanya meninggalkan
rangkaiannya seakan-akan menembus sebuah resistor, tanpa memanaskan apa pun. Sebuah kawat
\qty{1}{m} pada \qty{1}{MHz} ($\lambda = \qty{300}{m}$): $R_{\text{rad}} = \qty{9}{m\ohm}$, lebih kecil daripada
hambatan tembaganya, sebuah antena yang buruk; pada \qty{100}{MHz}: \qty{88}{\ohm} (rumus
dwikutub pendeknya diregangkan di sini), sebuah antena yang baik. \emph{\omterm{ex:b2:dipole-radiation:antenna}{Dwikutub setengah
gelombang}}, $\ell = \lambda/2$ dengan sebaran arus sinusoidal, mempunyai
$R_{\text{rad}} = \qty{73}{\ohm}$ (diterima tanpa bukti) dan pola yang dekat dengan
dwikutub dasarnya --- yaitu antena baku, dari tiang radio sampai
telinga kelinci pada televisi tua. Sebuah kawat jala-jala \qty{1}{m} pada \qty{50}{Hz}:
$R_{\text{rad}} \approx \qty{2e-11}{\ohm}$ --- rangkaian kuasi-tunak
\cref{ch:b2:maxwell-equations} tak memancarkan apa pun.
\end{example}

\begin{example}[Sebuah elektron yang memancar]\label{ex:b2:dipole-radiation:electron}
Dalam tabung sinar-X sebuah elektron \qty{50}{keV} ($v = \qty{1.2e8}{m/s}$) berhenti
dalam \qty{1}{\micro m} wolfram: $a \sim v^2/2d = \qty{7e21}{m/s^2}$, daya Larmornya
$\qty{1.2e-4}{W}$ selama $\qty{1.6e-14}{s}$: \qty{2e-18}{J}, sekitar $0.02\%$
energinya dipancarkan sebagai ``radiasi pengereman'' (bremsstrahlung)
tabungnya --- sisanya kalor, dan anodanya didinginkan. Elektron klasik
sebuah atom hidrogen, yang dipercepat sebesar $v^2/r = \qty{9e22}{m/s^2}$,
akan memancarkan \qty{5e-8}{W} lalu berpilin ke dalam intinya dalam \qty{1e-11}{s}:
fisika klasik melarang atom, dan \cref{ch:b2:schrodinger-wave-functions}
menyelamatkannya.
\end{example}

\section{Hamburan Rayleigh: langit biru}

\begin{proposition}[Hamburan oleh elektron terikat]\label{prop:b2:dipole-radiation:rayleigh}
\index{hamburan Rayleigh}\index{penampang lintang hamburan}\index{hamburan Thomson}
Sebuah gelombang beramplitudo $E_0$ dan berfrekuensi $\omega$ jatuh pada sebuah atom yang dimodelkan
sebagai elektron terikat elastik beresonansi $\omega_0 \gg \omega$
(\cref{ch:b2:waves-in-media}): elektronnya berayun dengan amplitudo
$eE_0/m\omega_0^2$, atomnya menjadi dwikutub $p_0 = e^2E_0/m\omega_0^2$ yang sefase
dengan medannya, lalu memancarkan ulang daya $\mathcal P = p_0^2\omega^4/12\pi
\varepsilon_0c^3$. Nisbah daya itu terhadap intensitas datangnya adalah
\emph{\omterm{prop:b2:dipole-radiation:rayleigh}{penampang lintang hamburannya}}
\[
\sigma = \frac{\mathcal P}{\varepsilon_0cE_0^2/2} = \frac{8\pi}3\,r_e^2\Bigl(\frac\omega{\omega_0}\Bigr)^4 ,
\qquad r_e = \frac{e^2}{4\pi\varepsilon_0mc^2} = \qty{2.8e-15}{m} :
\]
hukum Rayleigh, $\sigma \propto \omega^4 \propto 1/\lambda^4$. Bagi elektron \emph{bebas}
($\omega_0 = 0$) penampang lintangnya adalah $\sigma_T = 8\pi r_e^2/3 =
\qty{6.65e-29}{m^2}$, tak bergantung frekuensi (\omterm{prop:b2:dipole-radiation:rayleigh}{hamburan Thomson}).
\end{proposition}

\begin{proof}
Dari \omterm{prop:b2:waves-in-media:lorentz}{model Lorentz} dengan $\omega \ll \omega_0$ dan peredaman yang dapat diabaikan,
$\underline{\vect r} = -e\underline{\vect E}/m\omega_0^2$; sisipkanlah $p_0$ ke dalam dayanya lalu
bagilah dengan $I = \varepsilon_0cE_0^2/2$; $r_e$ menghimpun tetapannya. Bagi elektron
bebasnya, $\vect r = e\vect E/m\omega^2$ (berlawanan fase), $p_0 = e^2E_0/m\omega^2$, dan
$\omega^4$-nya saling meniadakan.
\end{proof}

\begin{example}[Langit biru, matahari terbenam merah, awan putih]\label{ex:b2:dipole-radiation:sky}
Cahaya lembayung ($\qty{400}{nm}$) terhambur $(700/400)^4 = 9.4$ kali lebih banyak
daripada merah: langitnya, yang disinari cahaya matahari yang terhambur, biru (bukan lembayung:
Matahari memancarkan lebih sedikit lembayung dan mata melihatnya dengan buruk). Pada matahari terbenam cahaya
langsungnya menyeberangi udara empat puluh kali lebih banyak daripada pada tengah hari lalu kehilangan birunya kepada
langit tempat lain: ia memerah. Tiap molekul udara menghamburkan dengan
$\sigma \approx \qty{5e-31}{m^2}$ pada \qty{550}{nm}; dengan $n = \qty{2.5e25}{m^{-3}}$
jalan bebas reratanya $1/n\sigma$ adalah \qty{80}{km}: sepersepuluh cahaya hijaunya
terhambur dalam perjalanannya turun menembus atmosfer, sepertiga
lembayungnya. Butir awan, yang jauh lebih besar daripada $\lambda$, bukan penghambur
Rayleigh: mereka menghamburkan semua warna sama saja --- putih.
\end{example}

\begin{proposition}[Polarisasi lewat hamburan]\label{prop:b2:dipole-radiation:polarization}
\index{polarisasi lewat hamburan}
\omterm{def:b2:plane-waves-polarization:states}{Cahaya alami} yang berjalan sepanjang $x$ mengimbaskan dwikutub dalam bidang $yz$.
Seorang pengamat yang memandang sepanjang $y$ (pada \qty{90}{\degree} dari berkasnya) tak menerima
radiasi dari komponen dwikutub sepanjang $y$ (garis pandangnya)
dan menerima radiasi penuh dari yang sepanjang $z$: cahaya yang terhambur itu
\emph{terpolarisasi linear} sepanjang $z$, tegak lurus bidang
yang memuat berkas dan garis pandangnya. Pada sudut lain ia
terpolarisasi sebagian, dengan derajat $(1 - \cos^2\chi)/(1 + \cos^2\chi)$ bagi
sudut hamburan $\chi$.
\end{proposition}

\begin{proof}
\omterm{def:b2:plane-waves-polarization:states}{Cahaya alami} yang datang mempunyai $\vect E$ dengan komponen rerata yang sama
sepanjang $y$ dan $z$; dwikutub sepanjang $y$ tak memancar ke arah $y$ ($\sin
\theta = 0$), yang sepanjang $z$ memancar $\propto\sin\theta = 1$. Pada sudut $\chi$
sumbangan komponen $y$-nya berkurang sebesar $\cos^2\chi$ dalam intensitas.
\end{proof}

\begin{remark}[Mengapa cahaya birunya sampai kepada kita sama sekali]\label{rem:b2:dipole-radiation:why}
Dalam medium yang seragam sempurna gelombang yang dihamburkan molekul
tetangganya akan berinterferensi memusnah ke setiap arah kecuali ke depan
(indeks biasnya adalah gelombang yang terhambur ke depan itu). Yang
meninggalkan intensitas hambur bersih, sebanding dengan jumlah
molekulnya seperti yang dihitung di atas bagi satu molekul, adalah \emph{fluktuasi} kerapatan udaranya
--- molekulnya tak berada pada kisi (Einstein, 1910). Keacakan
yang sama membuat cahaya langitnya tak koheren dari titik ke titik.
\end{remark}

\begin{method}[Taksiran radiasi]\label{met:b2:dipole-radiation:method}
(1) Kenalilah dwikutubnya: $p_0 = q\,\times$ amplitudo, atau $I_0\ell/\omega$ bagi sebuah
kawat. (2) Medan pada jarak $r$: $E \approx \mu_0\omega^2p_0\sin\theta/4\pi r$; dayanya:
$\mu_0\omega^4p_0^2/12\pi c$. (3) Bagi muatan yang dipercepat pakailah Larmor; bagi
sebuah antena pakailah \omterm{ex:b2:dipole-radiation:antenna}{hambatan radiasinya}. (4) Bagi hamburan, hitunglah
dwikutub imbasnya dari tanggapan mediumnya (\omterm{prop:b2:waves-in-media:lorentz}{model Lorentz}) lalu bagilah
daya terpancarnya dengan intensitas datangnya. (5) Ingatlah nolnya:
tak ada apa pun sepanjang sumbu ayunannya, maka polarisasinya pada
\qty{90}{\degree} dan \omterm{prop:b2:wave-interfaces:brewster}{sudut Brewster} \cref{ch:b2:wave-interfaces}.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:1}
Sebuah antena pendek \qty{1.0}{m} mengemban \qty{2.0}{A} (amplitudo) pada
\qty{100}{MHz}. Tentukanlah amplitudo dwikutubnya $p_0$; amplitudo medannya pada \qty{1}{km} dalam
bidang khatulistiwanya dan pada \qty{30}{\degree} dari sumbunya; daya terpancarnya;
\omterm{ex:b2:dipole-radiation:antenna}{hambatan radiasinya}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:2}
Tentukanlah \omterm{ex:b2:dipole-radiation:antenna}{hambatan radiasi} sebuah kawat \qty{1}{m} pada \qty{100}{kHz}, \qty{1}{MHz},
\qty{10}{MHz}, \qty{100}{MHz}; bandingkanlah dengan hambatan tembaganya
(berdiameter \qty{1}{mm}, dengan \omterm{thm:b2:waves-in-media:skin}{efek kulit} pada frekuensi yang lebih tinggi,
$\delta = \qty{0.2}{mm}$ pada \qty{100}{kHz}); tentukanlah efisiensinya (yang terpancar dibagi total)
pada tiap halnya. Mengapa pemancar gelombang panjang mempunyai tiang setinggi beratus
meter?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:3}
Tentukanlah nisbah \omterm{prop:b2:dipole-radiation:rayleigh}{hamburan Rayleigh} cahaya \qty{400}{nm}, \qty{550}{nm} dan
\qty{700}{nm}; dan nisbah bagi panjang gelombang radar \qty{3}{cm} dan \qty{10}{cm} oleh
tetes hujan (seandainya cukup kecil --- sebenarnya tidak begitu); mengapa
radar cuaca memakai panjang gelombang pendek untuk melihat hujan dan yang panjang untuk melihat
menembusnya?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dipole-radiation:4}
Tentukanlah derajat polarisasi cahaya langit pada \qty{30}{\degree}, \qty{60}{\degree},
\qty{90}{\degree}, \qty{120}{\degree} dari Matahari; ke bagian langit yang mana seorang
juru foto harus mengarahkan polarisatornya untuk mendapatkan biru yang paling gelap, dan apa yang
terjadi ketika Matahari tepat di atas kepala? Mengapa maksimum yang sebenarnya hanya sekitar
$75\%$ (pikirkanlah hamburan ganda dan apa yang dipantulkan tanahnya)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:5}
(a) Integralkanlah \omterm{thm:b2:poynting-vector:poynting}{vektor Poynting} dwikutubnya di seluruh sebuah bola lalu
periksalah $\mathcal P = \mu_0\omega^4p_0^2/12\pi c$. (b) Berapa bagian dayanya yang
dipancarkan dalam \qty{30}{\degree} dari bidang khatulistiwanya. (c) Sudut
setengah daya: pada $\theta$ berapa intensitasnya separuh maksimumnya? (d) ``Gain''
dwikutubnya, yaitu intensitas maksimum dibagi rerata isotropiknya:
tunjukkanlah bahwa nilainya $1.5$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:6}
\emph{Dekat dan jauh.} (a) Tuliskanlah medan kuasi-statik sebuah dwikutub pada
$\theta = 90^\circ$, $E_{\text{ns}} = p/4\pi\varepsilon_0r^3$, dan yang radiatif; pada
jarak $r_0$ berapa amplitudonya sama? Nyatakanlah $r_0$ dalam
$\lambda$. (b) Bagi antena \qty{1}{MHz}, berapa $r_0$; medan mana yang dirasakan sebuah penerima
pada \qty{10}{m}, pada \qty{10}{km}? (c) Sebuah telepon genggam pada \qty{1}{GHz} yang dipegang
\qty{2}{cm} dari kepala: zona dekat atau jauh? (d) Mengapa medan dekatnya
tak membawa daya bersih pergi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:7}
\emph{Thomson dan korona.} (a) Hitunglah $\sigma_T$. (b) Korona matahari
mempunyai $n_e \approx \qty{1e14}{m^{-3}}$ sepanjang $10^9$ m: berapa bagian cahaya
fotosfernya yang terhambur ke arah kita; mengapa koronanya putih dan hanya tampak
selama gerhana? (c) Sinar-X \qty{10}{keV} pada sebuah atom karbon ($6$
elektron, berenergi ikat di bawah \qty{300}{eV}): mengapa elektronnya
menghambur seakan-akan bebas, dan berapa penampang lintang atomnya? (d) Mengapa
citra sinar-X medis tulang berbeda dari daging terutama lewat
serapan, bukan \omterm{prop:b2:dipole-radiation:rayleigh}{hamburan Thomson}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:8}
\emph{Matahari terbenam.} Kedalaman optik atmosfer bagi \omterm{prop:b2:dipole-radiation:rayleigh}{hamburan
Rayleigh} di zenit adalah $\tau(\lambda) = 0.1\,(550\,\text{nm}/\lambda)^4$ (cahaya
langsungnya dilemahkan sebesar $\eu^{-\tau}$). (a) Tentukanlah penerusannya di
zenit bagi \qty{400}{nm}, \qty{550}{nm}, \qty{700}{nm}. (b) Pada matahari terbenam lintasannya
$38$ kali lebih panjang: tentukanlah penerusannya; nisbah merah/birunya. (c) Mengapa
Bulan berubah merah selama gerhana bulan? (d) Mengapa langit dekat
cakrawala keputihan alih-alih biru pekat?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dipole-radiation:9}
\emph{Dua antena.} Dua dwikutub tegak yang terpisah jarak $d$ sepanjang
$x$ disuapi dengan amplitudo yang sama. (a) Sefase, $d = \lambda/2$: ke arah
mendatar yang mana medan jauhnya bertambah, ke arah mana meniadakan?
(larik sisi lebar). (b) Berlawanan fase, $d = \lambda/2$: hal yang sama (tembak ujung).
(c) Dalam kuadratur, $d = \lambda/4$: tunjukkanlah bahwa lariknya memancar hanya ke satu
sisi. (d) Mengapa pemancar siaran gelombang menengah memakai beberapa tiang, dan
bagaimana sebuah radar larik berfase mengemudikan berkasnya tanpa bergerak?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:10}
\emph{Peredaman radiasi.} Elektron Lorentz (\cref{ch:b2:waves-in-media})
yang berayun pada $\omega_0$ dengan amplitudo $x_0$ memancar menurut rumus Larmor.
(a) Tentukanlah daya terpancar reratanya; energi osilatornya $\tfrac12m\omega_0^2x_0^2$;
tunjukkanlah bahwa energinya meluruh dengan laju $\Gamma = e^2\omega_0^2/6\pi\varepsilon_0mc^3$.
(b) Nilai $\Gamma$ bagi garis natrium (\qty{589}{nm}): bandingkanlah dengan
$\Gamma$ yang dipakai dalam \cref{exo:b2:waves-in-media:9}. (c) Umur $1/\Gamma$ dan
lebar alami garisnya, $\Delta\lambda = \lambda^2\Gamma/2\pi c$. (d) Faktor kualitas
$\omega_0/\Gamma$ osilator atomiknya.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:11}
\emph{Tiangnya.} Sebuah pemancar gelombang menengah memancarkan \qty{50}{kW} pada \qty{1}{MHz} dari sebuah
tiang tegak setinggi $\lambda/4$ di atas tanah yang menghantar (yang berlaku
sebagai cermin, sehingga tiangnya menjadi separuh \omterm{ex:b2:dipole-radiation:antenna}{dwikutub setengah gelombang}: $R_{\text{rad}} \approx
\qty{36}{\ohm}$, dan dayanya masuk hanya ke setengah ruang atasnya). (a)
Tentukanlah tingginya; arus di kakinya. (b) Amplitudo medan pada \qty{10}{km} dalam
bidang mendatarnya (pakailah pola dwikutubnya, $\mathcal P$ di seluruh
setengah bola atasnya $= \int\langle\Pi\rangle\dd S$; ambillah $\langle\Pi\rangle = 1.5\,
\mathcal P/2\pi r^2$ di cakrawalanya sebagai taksiran yang layak). (c) Ggl pada
sebuah cambuk penerima \qty{1}{m} di sana. (d) Mengapa konduktivitas tanahnya
berarti, dan mengapa kawat tembaga radial ditanam di bawah tiangnya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dipole-radiation:12}
\emph{Hamburan dan indeksnya.} Bagi gas encer gelombang yang terhambur ke
depan dari semua molekulnya bertambah secara koheren lalu membangun indeks biasnya,
$n - 1 = N\alpha/2\varepsilon_0$ dengan $\alpha = e^2/m\omega_0^2$ yaitu keterpolarannya;
penampang lintang Rayleighnya adalah $\sigma = (8\pi/3)r_e^2(\omega/\omega_0)^4$. (a) Nyatakanlah
$\sigma$ dalam $n$, $N$ dan $\lambda$: $\sigma = 8\pi^3(n^2 - 1)^2/3N^2\lambda^4$. (b)
Bagi udara ($n - 1 = 2.9 \times 10^{-4}$, $N = \qty{2.5e25}{m^{-3}}$) pada \qty{550}{nm}:
tentukanlah $\sigma$ dan jalan bebas reratanya $1/N\sigma$. (c) Kedalaman optik tegak
atmosfernya (tinggi setaranya \qty{8}{km}); bandingkanlah dengan $0.1$
\cref{exo:b2:dipole-radiation:8}. (d) Mengapa hubungan ini (Rayleigh,
1899) memberi nilai bilangan Avogadro?
\end{exercise}

\section{Soal: Tiang, langit dan elektron}

\begin{problem}[{Soal akhir pekan --- tiga antena: sebuah tiang
siaran, sebuah molekul udara dalam cahaya matahari, dan sebuah elektron bebas}]
\label{pb:b2:dipole-radiation:1}

\medskip
\noindent\textbf{Bagian I --- Tiangnya.}
Sebuah pemancar gelombang menengah pada $f = \qty{1.0}{MHz}$ memancarkan $\mathcal P = \qty{100}{kW}$
dari sebuah tiang seperempat gelombang di atas tanah yang menghantar ($R_{\text{rad}} = \qty{36}{\ohm}$,
berpola dwikutub tegak di seluruh setengah ruang atasnya, seperti dalam
\cref{exo:b2:dipole-radiation:11}).
\begin{enumerate}
  \item Panjang gelombang dan tinggi tiangnya; amplitudo arus di
        kakinya; tegangan pada $R_{\text{rad}}$.
  \item Amplitudo momen dwikutub setaranya, dengan memperlakukan tiangnya sebagai
        dwikutub pendek dengan $p_0 = I_0\ell_{\text{ef}}/\omega$, $\ell_{\text{ef}} \approx
        \lambda/2\pi$ bagi tiang seperempat gelombangnya (diterima tanpa bukti).
  \item Intensitas rerata pada \qty{10}{km} dan \qty{100}{km} dalam bidang
        mendatarnya (ambillah pola setengah bola atasnya, yang maksimal di
        cakrawalanya, $\langle\Pi\rangle = 1.5\mathcal P/2\pi r^2$); amplitudo medannya.
  \item Ggl pada cambuk tegak \qty{1}{m} pada \qty{100}{km}; pada sebuah antena
        batang ferit (kumparan berlilitan $N = 100$, \qty{1}{cm^2}, permeabilitas
        efektif $100$) yang menghadap $\vect B$ gelombangnya: bandingkanlah.
  \item Daya yang diterima sebuah penerima tersesuaikan berantena
        berluas efektif $A = 1.5\lambda^2/4\pi$ (milik dwikutubnya) pada \qty{100}{km}.
  \item Penghantar tiangnya mempunyai hambatan rugi total
        \qty{2}{\ohm}: tentukanlah efisiensinya; daya yang hilang sebagai kalor.
  \item Adakah sebuah penerima pada \qty{10}{km} berada dalam \omterm{thm:b2:dipole-radiation:field}{zona radiasi}? Dan yang pada
        \qty{100}{m} dari tiangnya?
  \item Mengapa sinyal gelombang menengah mencapai lebih jauh pada malam hari? (Ingatlah
        lapisan D \cref{ch:b2:waves-in-media}.)
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Langitnya.}
Udara di permukaan laut: $N = \qty{2.5e25}{m^{-3}}$, penampang lintang Rayleighnya $\sigma =
\qty{4.6e-31}{m^2}\,(550\,\text{nm}/\lambda)^4$; cahaya matahari \qty{1.0}{kW/m^2} di
tanahnya; tebal setara atmosfernya pada kerapatan permukaan lautnya adalah
\qty{8}{km}.
\begin{enumerate}[resume]
  \item Penampang lintangnya pada \qty{450}{nm} dan \qty{650}{nm}; nisbahnya.
  \item Jalan bebas rerata ketiga warnanya; kedalaman optik tegaknya
        $\tau = N\sigma H$; bagian tiap warnanya yang terhambur keluar dari
        berkas langsungnya di zenit.
  \item Daya yang terhambur tiap meter kubik udara di tanahnya oleh
        cahaya hijaunya (ambillah \qty{300}{W/m^2} cahaya matahari di sekitar
        \qty{550}{nm}); ke dalam sudut ruang berapa, dengan pola apa?
  \item Sebuah kolom udara berpenampang \qty{1}{m^2} yang dipandang pada \qty{90}{\degree} dari
        Matahari: taksirlah intensitas cahaya langit yang tiba pada
        pengamatnya dari kolom itu (daya yang terhambur ke arahnya tiap
        satuan sudut ruang, sepanjang \qty{8}{km}), lalu bandingkanlah dengan
        Matahari langsungnya; berordekah kecerahan langitnya dengan benar
        (sekitar \qty{1e-5}{} dari Matahari)?
  \item Derajat polarisasi cahaya dari kolom itu; mengapa
        nilainya lebih rendah dalam kenyataan, dan bagaimana sebagian serangga memakainya?
  \item Pada matahari terbenam lintasannya $38$ atmosfer: tentukanlah bagian yang diteruskan
        dari biru dan merahnya; warna Mataharinya; mengapa langit di atas kepala tetap
        biru.
  \item Mengapa langit hitam di Bulan bahkan pada siang bolong?
  \item Di Mars udaranya seratus kali lebih tipis tetapi penuh debu
        halus (butir semikrometer): mengapa langit Mars
        berwarna karamel dan matahari terbenamnya biru?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Elektronnya.}
\begin{enumerate}[resume]
  \item Penampang lintang Thomson dari $r_e$.
  \item Dalam tabung sinar-X sebuah elektron \qty{100}{keV} ($v = \qty{1.6e8}{m/s}$,
        perlakukanlah secara tak relativistik) dihentikan sepanjang \qty{2}{\micro m} dalam
        wolfram: tentukanlah percepatannya, daya Larmornya, lamanya, energi
        yang terpancar dan bagiannya dari energi kinetiknya.
  \item Panjang gelombang terpendek yang dapat dipancarkan tabungnya (sebuah foton yang mengambil
        seluruh energi elektronnya).
  \item Tabungnya berjalan pada \qty{20}{mA}: tentukanlah daya terpancarnya; kalor dalam anodanya;
        mengapa anodanya berputar.
  \item Sebuah elektron klasik yang mengelilingi sebuah proton pada jari-jari Bohr
        (\qty{5.3e-11}{m}, kelajuan \qty{2.2e6}{m/s}): tentukanlah percepatannya, daya
        Larmornya, dan waktu untuk memancarkan habis \qty{13.6}{eV}-nya --- yaitu
        umur atom klasiknya.
  \item Atom yang sungguhan bertahan selamanya: sebutkanlah apa yang terlewat fisika klasik
        di sini, dan apa yang akan dikatakan \cref{ch:b2:schrodinger-wave-functions}
        tentang sebuah keadaan tegar.
  \item Sebuah elektron dalam sinkrotron berjari-jari \qty{100}{m} pada $v \approx c$:
        rumus Larmor tak relativistiknya memberi daya berapa? (Daya
        yang sebenarnya $\gamma^4$ kali lebih besar, dengan $\gamma \sim 10^4$: yaitu
        sumber cahaya sinkrotron.)
  \item Elektron yang sama dalam keadaan diam dalam \qty{1}{kW/m^2} cahaya matahari:
        tentukanlah daya yang dipancarkannya ulang (Thomson); jumlah elektron semacam itu
        yang diperlukan untuk menghamburkan satu watt.
  \item Ringkaslah ketiga antenanya dalam sebuah tabel: momen dwikutub,
        frekuensi, daya, pola.
\end{enumerate}
\end{problem}
