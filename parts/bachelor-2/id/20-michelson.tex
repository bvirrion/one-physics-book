\chapter{Interferometer Michelson}\label{ch:b2:michelson}

Pada 1887 Michelson dan Morley membelah sebuah berkas cahaya menjadi dua, mengirimkan
paruhnya sepanjang lengan yang saling tegak lurus, menggabungkannya kembali, lalu mengamati
rumbainya untuk mencari geseran yang seharusnya dihasilkan gerak Bumi menembus eter:
tak ada yang beranjak, dan fisika berubah. Pada 2015 instrumen yang sama
--- dua lengan, yang kini sepanjang empat kilometer dan dilipat menjadi
seribu --- merekam sebuah gelombang gravitasi yang mengubah panjangnya
sebesar seperseribu diameter sebuah proton. Di antara kedua tanggal itu
\omterm{def:b2:michelson:instrument}{interferometer Michelson} mengukur meter dalam panjang gelombang, memisahkan
garis spektrum, dan menjadi perkakas baku optika presisi.
Bab ini melukiskannya dalam kedua tatanannya --- baji udara
dan keping sejajar --- lalu menunjukkan bagaimana rumbainya mengukur panjang,
indeks dan spektrum.

\section{Instrumennya}

\begin{definition}[Interferometer Michelson]\label{def:b2:michelson:instrument}
\index{interferometer Michelson}\index{pembelah berkas}\index{keping pengimbang}
Sebuah \emph{pembelah berkas} $Sp$ --- sebuah keping kaca bermuka setengah memantul
--- pada \qty{45}{\degree} membagi berkas datangnya menjadi dua: satu pergi ke
cermin $M_1$, satunya ke $M_2$, keduanya tegak lurus berkasnya; sesudah
pemantulannya masing-masing kembali ke $Sp$, yang di sana separuhnya dikirimkan ke arah
pengamatnya, tempat kedua paruhnya berinterferensi. Sebuah \emph{keping pengimbang}
$C$, yang serupa dengan pembelahnya dan sejajar dengannya, ditaruh pada lengan
yang jika tidak hanya akan menyeberangi kaca pembelahnya sekali, sehingga kedua
berkasnya menyeberangi tebal kaca yang sama: \omterm{thm:b2:two-wave-interference:formula}{beda lintasannya} lalu menjadi
\omterm{thm:b2:two-wave-interference:formula}{beda lintasan} udara murni, yang sama bagi setiap panjang gelombang --- yang
dituntut rumbai cahaya putih. Jarak cerminnya dari $Sp$ adalah $d_1$
dan $d_2$; satu cerminnya terpasang pada sebuah sekrup mikrometer dan pada sekrup
kemiringan.
\end{definition}

\begin{proposition}[Susunan setara]\label{prop:b2:michelson:equivalent}
\index{baji udara (Michelson)}\index{keping udara (Michelson)}
Dilihat dari pengamatnya, interferometernya setara dengan cermin
$M_2$ dan citra $M_1'$ dari $M_1$ dalam pembelahnya: sebuah lapisan udara tipis
di antara dua permukaan pemantul, setebal $e = |d_1 - d_2|$, yang kedua
pantulannya berinterferensi (\omterm{prop:b2:two-wave-interference:film}{pembelahan amplitudo}):
\begin{itemize}
  \item $M_1' \parallel M_2$: sebuah \emph{keping udara} setebal $e$ --- \omterm{prop:b2:michelson:rings}{cincin
        kemiringan sama}, yang terlokalkan di tak hingga, $\delta = 2e\cos i$;
  \item $M_1'$ yang dimiringkan sedikit sebesar $\alpha$ relatif terhadap $M_2$: sebuah \emph{baji udara}
        --- \omterm{prop:b2:two-wave-interference:film}{rumbai ketebalan sama} yang lurus, yang terlokalkan pada
        cerminnya (dekat bajinya), $\delta = 2\alpha x$ pada datang tegak lurus,
        berjarak $\lambda/2\alpha$.
\end{itemize}
Pada \emph{sentuhan} ($e = 0$, $\alpha = 0$) medannya seragam, dan bagi cahaya
putih di sanalah satu-satunya tempat warnanya selaras: ``rumbai putih''-nya.
\end{proposition}

\begin{proof}
Melipat lengan $M_1$ ke atas lengan $M_2$ lewat pemantulan dalam $Sp$ memetakan
kedua berkasnya ke satu garis dengan dua cermin yang terpisah $e$; kedua gelombang
yang kembali itu adalah yang terpantul kedua muka lapisan udaranya (tanpa
$\lambda/2$ tambahan: keduanya memantul pada logam, dan pembelahnya memperlakukannya
setangkup, sampai pada sebuah fase tetap yang menggeser seluruh polanya).
Hasil selaput tipis \cref{ch:b2:two-wave-interference} lalu berlaku
dengan $n = 1$.
\end{proof}

\section{Kedua sistem rumbainya}

\begin{proposition}[Cincin keping udara]\label{prop:b2:michelson:rings}
\index{cincin kemiringan sama}\index{rumbai Haidinger}
Dengan $M_1' \parallel M_2$ dan sebuah \omterm{prop:b2:two-wave-interference:spatial}{sumber luas}, pengamatnya (atau sebuah lensa
berjarak fokus $f$ dan sebuah layar pada bidang fokusnya) melihat cincin
sepusat; cincin pada sudut $i$ mempunyai orde $p = 2e\cos i/\lambda$; cincinnya
terang di tempat $p$ bulat. Pusatnya berorde tertinggi, $p_0 =
2e/\lambda$; dekat pusatnya, $\cos i \approx 1 - i^2/2$, sehingga cincin terang ke-$k$ dari
pusatnya (dihitung dari yang pertama, pada $\varepsilon = p_0 - \lfloor p_0\rfloor$
di bawah orde pusatnya) berjari-jari
\[
r_k = f\,i_k = f\sqrt{\frac{(k - 1 + \varepsilon)\lambda}e} :
\]
cincinnya lebih rapat bagi $e$ yang lebih besar (lebih banyak cincin yang lebih tipis), lalu merenggang ketika $e \to 0$
sampai, pada sentuhannya, menjadi satu medan yang seragam. Menggeser sebuah cermin sebesar $\lambda/2$ mengubah
$p_0$ sebesar satu: sebuah cincin lahir di (atau ditelan) pusatnya bagi tiap
setengah panjang gelombang geseran.
\end{proposition}

\begin{proof}
$\delta = 2e\cos i$ bagi keping sejajarnya (\cref{ch:b2:two-wave-interference},
$n = 1$, tanpa $\lambda/2$ tambahan bagi dua pemantulan logam); $2e(1 - i^2/2) =
(p_0 - k + 1 - \varepsilon)\lambda$ dengan $2e = p_0\lambda$ memberi $ei^2 = (k - 1 + \varepsilon)\lambda$.
\end{proof}

\begin{proposition}[Rumbai baji udara]\label{prop:b2:michelson:wedge}
\index{rumbai ketebalan sama (Michelson)}\index{rumbai Fizeau}
Dengan $M_1'$ yang dimiringkan sebesar sudut kecil $\alpha$ dan datang hampir tegak lurus,
\omterm{thm:b2:two-wave-interference:formula}{beda lintasan} di titik cerminnya pada jarak $x$ dari
garis sentuhnya adalah $\delta = 2\alpha x$: rumbai lurus yang sejajar garis itu,
berjarak $\lambda/2\alpha$, yang terlokalkan pada cerminnya (rumbainya dipandang dengan memfokus
pada $M_2$, atau diproyeksikan pada sebuah layar dengan lensa yang menyekawankan $M_2$ dan
layarnya). Rumbai orde nolnya terletak pada garis sentuhnya, yang sama bagi
setiap panjang gelombang: dengan cahaya putih ia putih (atau hitam, menurut
fase pembelahnya), yang diapit beberapa rumbai berwarna --- yaitu
tanda tangan yang dengannya sentuhannya ditemukan.
\end{proposition}

\begin{proof}
\omterm{prop:b2:two-wave-interference:film}{Rumbai ketebalan sama} sebuah baji udara, $e(x) = \alpha x$. Pelokalannya:
bagi sebuah \omterm{prop:b2:two-wave-interference:spatial}{sumber luas} sinar yang menembus sebuah titik tertentu bajinya
bergabung kembali di sana, dengan $\delta$ yang sama pada datang hampir tegak lurus.
\end{proof}

\section{Mengukur dengan Michelson}

\begin{proposition}[Panjang, indeks, spektrum]\label{prop:b2:michelson:measure}
\index{spektroskopi transformasi Fourier}\index{spektrum berkanal}
Mencacah $N$ rumbai yang melewati pusatnya sementara sebuah cermin bergerak sebesar $\Delta e$
memberi $\Delta e = N\lambda/2$: panjang dalam panjang gelombang (sebuah sekrup mikrometer
yang tertera sampai \qty{0.1}{\micro m}; meter itu sendiri diukur begitu). Sebuah
keping bening setebal $\ell$ dan berindeks $n$ yang disisipkan pada satu lengan
menambahkan $2(n - 1)\ell$ pada $\delta$ lalu menggeser polanya sebesar $2(n - 1)\ell/\lambda$
rumbai: yaitu indeksnya (dan perubahannya dengan tekanan atau suhu,
dalam sebuah sel gas). Dengan sebuah dublet berpanjang gelombang $\lambda$ dan $\lambda + \Delta\lambda$, kedua
sistem rumbainya kehilangan langkah lalu selaras lagi ketika $e$ membesar, dengan
kontrasnya lenyap tiap
\[
\Delta e = \frac{\lambda^2}{2\Delta\lambda} :
\]
yaitu pembelahan garis yang berdekatan. Secara lebih umum kontras sebagai
fungsi $e$ memetakan spektrum sumbernya (kontrasnya
meluruh sepanjang $e \sim \ell_c/2$): merekam intensitas di pusatnya sementara
$e$ memindai adalah \emph{\omterm{prop:b2:michelson:measure}{spektroskopi transformasi Fourier}}, yaitu metode
spektrometer inframerah.
\end{proposition}

\begin{proof}
Orde di pusatnya adalah $2e/\lambda$; tiap garisnya mempunyai ordenya sendiri. Kedua
sistemnya berimpit lagi ketika $2e/\lambda - 2e/(\lambda + \Delta\lambda) = 1$, yakni $2e\Delta\lambda
/\lambda^2 = 1$; keduanya berlawanan di pertengahannya. Pernyataan umumnya adalah
rumus kontras \cref{exo:b2:scalar-light-model:10}.
\end{proof}

\begin{example}[Natrium, dan meter]\label{ex:b2:michelson:sodium}
Dublet natriumnya ($\Delta\lambda = \qty{0.6}{nm}$ pada \qty{589}{nm}): cincinnya kabur lalu
menajam tiap $\Delta e = 589^2/(2 \times 0.6)$ nm $= \qty{0.29}{mm}$ --- sebuah pengukuran
klasik $\Delta\lambda$ sampai beberapa persen dengan sebuah sekrup. Michelson (1892)
mencacah rumbai garis merah kadmium sepanjang meter
bakunya: $1\,553\,163.5$ panjang gelombang, tautan pertama antara
meter dan sebuah atom; kini meter didefinisikan lewat $c$ dan
sekon, dan interferometer memeriksa balok ukur dalam produksi sampai
nanometer.
\end{example}

\begin{remark}[Lengan yang tak mengukur apa pun, dan yang mengukur sebuah gelombang]\label{rem:b2:michelson:history}
\index{percobaan Michelson--Morley}\index{detektor gelombang gravitasi}
Michelson dan Morley mengharapkan kelajuan Bumi $v$ menembus
eter mengubah waktu pulang pergi sepanjang lengan yang sejajar dengannya sebesar
$(L/c)(v/c)^2$ relatif terhadap lengan yang tegak lurus, yaitu geseran rumbai sebesar
$2L(v/c)^2/\lambda \approx 0.4$ rumbai bagi $L = \qty{11}{m}$ dan $v = \qty{30}{km/s}$;
memutar radasnya seharusnya mengayunkan polanya sebesar itu.
Mereka dapat melihat $0.01$ rumbai; mereka tak melihat apa pun. \omterm{rem:b2:michelson:history}{Detektor gelombang
gravitasi} masa kini adalah Michelson berlengan \qty{4}{km}, dengan cahaya yang memantul-mantul
sekitar tiga ratus kali di tiap lengannya, dan sebuah laser yang cukup mantap untuk membaca
perubahan panjang lengan $\Delta L/L \sim 10^{-21}$ --- $10^{-18}$ m, seperseribu
sebuah proton --- terhadap derau gerak seismik dan derau foton
itu sendiri.
\end{remark}

\begin{method}[Memakai sebuah Michelson]\label{met:b2:michelson:method}
(1) Kenalilah tatanannya: keping sejajar (cincin di tak hingga;
amatilah dengan sebuah lensa atau mata di tak hingga) atau baji (rumbai pada
cerminnya; fokuslah pada $M_2$). (2) Tuliskanlah $\delta = 2e\cos i$ atau $2\alpha x$ dan
ordenya. (3) Terjemahkanlah tiap pengukurannya menjadi rumbai: $\Delta e = N\lambda/2$,
$2(n - 1)\ell$ bagi sebuah keping, $\lambda^2/2\Delta\lambda$ bagi sebuah dublet. (4) Pakailah
kedudukan sentuhnya ($e = 0$, rumbai putih) sebagai acuan mutlaknya;
perhatikanlah bahwa jari-jari cincinnya \emph{menyusut} ketika $e$ membesar. (5) Batas
koherensinya: cincinnya tampak selama $2e < \ell_c$; bajinya menuntut
datang hampir tegak lurus.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:michelson:1}
Sebuah Michelson dalam cahaya natrium ($\lambda = \qty{589}{nm}$): cerminnya digeser sebesar
\qty{0.10}{mm}: tentukanlah jumlah cincin yang melewati pusatnya; geseran
yang bersesuaian dengan $250$ rumbai; ketelitian pada $e$ jika sepersepuluh rumbai
dapat dibaca.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:2}
Keping sejajar setebal $e = \qty{0.50}{mm}$, lensa $f = \qty{1.0}{m}$, $\lambda =
\qty{589}{nm}$: tentukanlah orde di pusatnya; terangkah pusatnya? Jari-jari
ketiga cincin terang pertamanya; apa yang terjadi pada cincinnya ketika $e$ dikurangi
menjadi \qty{0.05}{mm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:3}
Baji udara dengan $\alpha = \qty{1.0e-4}{rad}$, $\lambda = \qty{633}{nm}$: tentukanlah jarak rumbainya;
jumlah rumbai di seluruh \qty{2}{cm} cerminnya; sudut bajinya
dilipatduakan: apa yang terjadi? Bagaimana sentuhannya ditemukan dengan cahaya putih?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:4}
Sebuah keping kaca setebal \qty{0.10}{mm} disisipkan pada satu lengannya: tentukanlah geseran rumbainya bagi
$n = 1.52$ pada \qty{589}{nm}; geserannya terukur sampai $0.2$ rumbai: berapa ketelitian
pada $n$. Mengapa pengukuran yang sama tak dapat dilakukan dengan cahaya putih
jika kepingnya setebal \qty{1}{cm}?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:5}
\emph{Dublet natrium.} (a) Tentukanlah tebal $e$ antara dua kali lenyapnya
cincin yang berturut-turut, bagi $\Delta\lambda = \qty{0.597}{nm}$. (b) Seorang mahasiswa
mengukur $17$ kali lenyap sepanjang \qty{5.00}{mm}: berapa $\Delta\lambda$ dan ketelitiannya.
(c) Orde kedua garisnya di $e = \qty{0.29}{mm}$: periksalah bahwa keduanya berselisih
$\tfrac12$. (d) Mengapa kontrasnya tak pernah tepat nol (kedua garisnya berintensitas
dengan nisbah $2 : 1$): berapa kontras minimumnya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:6}
\emph{Indeks udara.} Sebuah sel \qty{10.0}{cm} pada satu lengannya dihampakan, lalu
diisi udara perlahan pada \qty{1}{bar}; $94$ rumbai lewat ($\lambda = \qty{589}{nm}$).
(a) Tentukanlah $n - 1$ bagi udaranya. (b) Rumbai yang lewat bagi perubahan tekanan \qty{1}{\%};
koefisien suhunya jika $n - 1 \propto \rho$. (c) Mengapa ini
berarti bagi kemantapan interferometer mana pun dalam udara? (d) Ketelitian pada
$n - 1$ dari sepersepuluh rumbai.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:7}
\emph{\omterm{prop:b2:scalar-light-model:coherence}{Panjang koherensi}.} Dengan lampu raksa bertekanan rendah (garis hijau,
\qty{546}{nm}, berlebar \qty{0.01}{nm}) cincinnya memudar ketika $e$ membesar. (a) Tentukanlah \omterm{prop:b2:scalar-light-model:coherence}{panjang
koherensinya}; tebal $e$ yang padanya kontrasnya menjadi dapat diabaikan. (b) Hal yang sama
dengan lampu bertekanan tinggi ($\Delta\lambda = \qty{1}{nm}$). (c) Dengan sebuah laser He--Ne
($\Delta\nu = \qty{1}{MHz}$): masih dapatkah cincinnya terlihat dengan $e = \qty{10}{m}$? (d) Apa
yang membatasi rumbai cahaya putih sampai $\pm2$ orde di sekitar sentuhannya?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:8}
\emph{\omterm{def:b2:michelson:instrument}{Keping pengimbang}.} Tanpa $C$, satu berkasnya menyeberangi kaca
pembelahnya ($e_g = \qty{5}{mm}$, $n = 1.52$) tiga kali dan satunya sekali. (a)
Tentukanlah \omterm{def:b2:scalar-light-model:path}{lintasan optik} tambahannya; dapatkah ia diimbangi dengan menggeser sebuah cermin bagi
cahaya monokromatik? (b) Mengapa tidak bagi cahaya putih (dispersinya: $n$
berubah sebesar $0.01$ di seluruh cahaya tampak)? (c) Toleransi berapa pada
kesejajaran $C$ dan $Sp$ yang menjaga rumbai putihnya (ambillah beda tebal
sisanya $< \qty{0.3}{\micro m}$)? (d) Pembelah modern berupa
pelikel tipis atau kubus: mengapa $C$ lalu tak diperlukan?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:9}
\emph{Michelson--Morley.} Lengan sepanjang $L = \qty{11}{m}$ (dilipat), $\lambda = \qty{550}{nm}$,
kelajuan orbit Bumi $v = \qty{30}{km/s}$. (a) Dalam teori eternya perjalanan
pulang pergi sepanjang lengan yang sejajar $\vect v$ memakan $2L/c(1 - v^2/c^2)$
dan yang tegak lurus $2L/c\sqrt{1 - v^2/c^2}$: tentukanlah selisihnya, sampai orde
terendah. (b) Geseran rumbai yang diharapkan ketika radasnya diputar \qty{90}{\degree}
(peran lengannya bertukar). (c) Mereka dapat mendeteksi $0.01$ rumbai: berapa batas
atas $v$. (d) Apa yang dikatakan sebuah hasil nol, dan apa yang disimpulkan Einstein?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:10}
\emph{Spektroskopi Fourier.} Intensitas di pusat cincinnya
adalah $I(e) = \int S(\nu)[1 + \cos(4\pi\nu e/c)]\dd\nu$ bagi sumber berspektrum
$S(\nu)$. (a) Bagi satu garis tunggal: tentukanlah $I(e)$; periodenya dalam $e$. (b) Bagi sebuah garis
Gauss berlebar $\Delta\nu$: tunjukkanlah bahwa rumbainya berselubung Gauss berlebar
$\sim c/2\Delta\nu$ dalam $e$ (terimalah bahwa transformasi Fourier sebuah Gauss
adalah Gauss). (c) Resolusinya: dua garis yang terpisah $\Delta\nu$ terpisahkan jika
pemindaiannya mencapai $e_{\max} \approx c/2\Delta\nu$; bagi $e_{\max} = \qty{10}{cm}$, berapa
daya urainya $\lambda/\Delta\lambda$ pada \qty{1}{\micro m}. (d) Mengapa metode ini
lebih disukai dalam inframerah (sinyal, keunggulan ``multipleks'')?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:11}
\emph{\omterm{rem:b2:michelson:history}{Detektor gelombang gravitasi}.} Lengan $L = \qty{4}{km}$, cahayanya didaur ulang
menjadi $N = 300$ perjalanan pulang pergi, $\lambda = \qty{1064}{nm}$, gelombangnya mengubah lengannya sebesar
$\Delta L/L = 10^{-21}$ dalam arah berlawanan. (a) Tentukanlah perubahan \omterm{thm:b2:two-wave-interference:formula}{beda lintasannya};
geseran fasenya. (b) Instrumennya ditahan pada sebuah rumbai gelap lalu mengukur
intensitas kecilnya $I = I_0\sin^2(\Delta\varphi/2)$: berapa perubahan intensitas bagi
sinyalnya. (c) Dengan \qty{100}{W} cahaya yang beredar pada \qty{1064}{nm}, tentukanlah foton
tiap sekonnya; derau fotonnya dalam \qty{1}{ms} ($\sqrt N$) terhadap sinyalnya:
terdeteksikah ia? Apa yang dibeli daya yang lebih besar? (d) Mengapa lengannya
berkilometer, dan mengapa cerminnya digantung?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:12}
\emph{Pelokalan.} (a) Tunjukkanlah bahwa bagi bajinya kedua sinar yang meninggalkan sebuah
titik $P$ pada $M_2$ menuju seorang pengamat di tak hingga (datang $i$) mempunyai
\omterm{thm:b2:two-wave-interference:formula}{beda lintasan} $2e(P)\cos i$ ditambah suku $\propto\alpha i$ yang lenyap bagi
$i = 0$: rumbainya berada pada cerminnya hanya pada datang tegak lurus. (b)
Sebuah \omterm{prop:b2:two-wave-interference:spatial}{sumber luas} menyinari bajinya dengan datang sampai
$i_{\max}$: apa yang harus dipenuhi $i_{\max}$ agar rumbainya tetap tajam
($2e\,i_{\max}^2/2 < \lambda/4$ di tepi medannya, $e = \qty{20}{\micro m}$)? (c)
Bagi keping sejajarnya cincinnya tak beranjak ketika matanya bergerak:
mengapa? (d) Seorang mahasiswa melihat rumbai yang hanyut ketika ia menggerakkan kepalanya:
dalam tatanan yang mana ia berada, dan apa yang harus ia setel?
\end{exercise}

\section{Soal: Dublet natrium dan indeks udara}

\begin{problem}[{Soal akhir pekan --- sebuah sesi pengukuran lengkap pada
sebuah interferometer Michelson: menemukan sentuhan, membaca panjang, membelah
garis kuningnya, menimbang udaranya}]
\label{pb:b2:michelson:1}

Interferometernya mempunyai sebuah sekrup mikrometer yang terbaca sampai \qty{1}{\micro m};
lampu natriumnya memberi $\lambda_1 = \qty{589.0}{nm}$ dan $\lambda_2 = \qty{589.6}{nm}$ dengan
intensitas $2 : 1$; sebuah lampu putih dan sebuah laser He--Ne (\qty{632.8}{nm}) juga
tersedia.

\medskip
\noindent\textbf{Bagian I --- Menyiapkannya.}
\begin{enumerate}
  \item Dengan lasernya, mahasiswanya melihat dua bintik dari kedua lengannya pada
        sebuah kartu; ia menumpangkannya dengan sekrup kemiringannya lalu melihat
        rumbai lurus: tatanan yang mana ini, dan apa yang dikatakan
        jaraknya tentang kemiringan sisanya (jaraknya \qty{2}{mm})?
  \item Dengan memutar sekrup kemiringannya, rumbainya melengkung lalu menjadi
        cincin: mengapa cincinnya muncul, dan di mana $M_1'$ sekarang?
  \item Ia menggeser cerminnya lalu mencacah $158$ cincin yang ditelan
        pusatnya: berapa geserannya; ke arah mana $e$ berubah?
  \item Ketika ia melanjutkannya, cincinnya merenggang dan bintik pusatnya membesar
        sampai memenuhi medannya: apa yang telah ia capai? Lampu natriumnya
        menggantikan lasernya dan ia tak melihat apa pun yang istimewa: mengapa tidak, dan
        mengapa cahaya putih menunjukkan sebuah rumbai pusat yang terang yang diapit
        rumbai berwarna di titik itu?
  \item Ia memiringkan $M_1$ sangat sedikit pada sentuhannya lalu melihat, dalam cahaya
        putih, rumbai berwarna yang lurus: jelaskanlah, lalu katakanlah berapa banyak
        yang dapat ia harapkan di tiap sisi rumbai putihnya.
  \item Bintik lasernya selebar \qty{10}{mm} dan ia menginginkan sepuluh cincin
        dalam medan sebuah lensa ber-$f = \qty{20}{cm}$ ($\lambda = \qty{632.8}{nm}$):
        tebal $e$ yang diperlukan (pakailah jari-jari cincin ke-$k$).
  \item Mengapa \omterm{def:b2:michelson:instrument}{keping pengimbangnya} harus sejajar dengan pembelahnya,
        dan apa yang akan ia lihat dalam cahaya putih jika keping itu disingkirkan?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Dubletnya.}
\begin{enumerate}[resume]
  \item Beranjak dari sentuhannya dengan lampu natriumnya, ia menaikkan $e$:
        kontras cincinnya turun, lenyap, lalu kembali. Tebal pada
        kali lenyap yang pertama; pada yang ke-$n$.
  \item Ia merekam $12$ kali lenyap antara $e = \qty{0.15}{mm}$ dan
        $e = \qty{3.35}{mm}$ (yang pertama dan terakhir terbaca pada sekrupnya):
        berapa $\Delta\lambda$; ketelitiannya jika tiap pembacaannya baik sampai \qty{5}{\micro m}.
  \item Orde $p_1$ dan $p_2$ kedua garisnya pada kali lenyap yang
        pertama; pada kembalinya kontras penuh yang pertama.
  \item Kontras minimum pada sebuah kali lenyap, mengingat nisbah
        intensitas $2 : 1$ itu (rumbai garis yang lebih lemah tetap tinggal).
  \item Berapa kali lenyap yang dapat ia cacah sebelum lebar alami
        garisnya (\qty{0.02}{nm} masing-masing) menghapus segalanya?
  \item Apa yang akan ditunjukkan sebuah lampu dengan tiga garis berjarak sama?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Udaranya.}
Sebuah sel sepanjang $\ell = \qty{5.00}{cm}$ berjendela kaca ditaruh pada satu
lengannya lalu dipompa turun; udaranya dimasukkan perlahan sementara rumbai lasernya
dicacah.
\begin{enumerate}[resume]
  \item Beda \omterm{def:b2:scalar-light-model:path}{lintasan optik} yang tercipta oleh mengisi selnya sampai
        \qty{1}{atm} jika $n - 1 = \qty{2.9e-4}{}$; berapa jumlah rumbainya.
  \item Ia mencacah $46$ rumbai: berapa nilai $n - 1$ miliknya.
  \item Ia hanya memasukkan \qty{0.20}{bar}: berapa rumbai yang diharapkan ($n - 1 \propto$
        tekanan).
  \item Ruangannya menghangat sebesar \qty{3}{K} selama pengukurannya, dan
        \qty{40}{cm} lintasan udara terbuka di tiap lengannya berubah indeks sebesar
        $-\qty{1e-6}{}$ tiap kelvin: berapa hanyutan rumbainya; masalahkah ia bagi
        Bagian II, bagi Bagian III?
  \item Mengapa jendela selnya tak menggeser pengukurannya (jendelanya
        ada di sana selama seluruh pencacahannya)?
  \item Percobaannya diulangi dengan CO$_2$ ($n - 1 = \qty{4.5e-4}{}$):
        berapa rumbainya bagi sel yang penuh; bagaimana gasnya dapat dikenali dari
        cacahnya saja?
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Batasnya.}
\begin{enumerate}[resume]
  \item Sekrupnya terbaca sampai \qty{1}{\micro m}: berapa ketelitian sebuah panjang
        yang diukur lewat pencacahan rumbai sepanjang \qty{1}{cm}, melawan lewat sekrupnya
        saja.
  \item Sebuah getaran beramplitudo \qty{50}{nm} pada \qty{100}{Hz} mengguncang satu
        cerminnya: berapa gerak rumbainya; apa yang dilihat mata, apa yang dilihat sebuah
        kamera pada $25$ bingkai tiap sekon?
  \item Frekuensi lasernya hanyut sebesar \qty{1e-8}{} dalam nilai nisbi
        selama sebuah pencacahan sepanjang $e = \qty{1}{cm}$: berapa galatnya dalam rumbai.
  \item Sebuah balok ukur \qty{10}{mm} disertifikasi sampai $\lambda/20$ lewat
        interferometri: berapa ketelitian mutlak dan nisbinya.
  \item Mengapa Michelson merupakan instrumen pilihan bagi panjang, dan
        kisi (bab berikutnya) bagi spektrum?
  \item Ringkaslah keempat besaran yang terukur (sebuah panjang, sebuah pembelahan
        dublet, sebuah indeks, sebuah \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensi}) dan pengamatan
        rumbai yang memberi masing-masingnya.
\end{enumerate}
\end{problem}
