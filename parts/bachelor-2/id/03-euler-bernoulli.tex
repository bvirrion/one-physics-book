\chapter{Fluida Sempurna: Euler dan Bernoulli}\label{ch:b2:euler-bernoulli}

Peganglah selembar kertas pada tepinya lalu tiuplah di atasnya: kertas
itu terangkat. Bukalah keran pancuran mandi dan tirainya condong ke
dalam. Sebuah sayap menahan tiga ratus ton karena udara mengalir lebih
cepat di atas permukaannya daripada di bawahnya. Ketiganya adalah satu
pernyataan yang sama: \emph{di tempat sebuah fluida bergerak lebih cepat,
tekanannya lebih rendah}. Bab ini menuliskan hukum kedua Newton bagi
\omterm{def:b2:fluid-kinematics:particle}{partikel fluida} dalam model yang paling sederhana --- fluida
\emph{sempurna}, yang merasakan tekanan tetapi tanpa gesekan --- lalu
menarik darinya teorema Bernoulli, neraca energi sebuah \omterm{def:b2:fluid-kinematics:euler}{garis arus},
yang penerapannya membentang dari meteran aliran dalam pipa sampai
penunjuk kelajuan sebuah pesawat terbang.

\section{Persamaan Euler}

\begin{proposition}[Gaya tekanan pada sebuah partikel fluida]\label{prop:b2:euler-bernoulli:pressure}
\index{gaya tekanan (rapat volume)}
Resultan gaya tekanan yang dikerahkan fluida di sekelilingnya pada
sebuah partikel bervolume $\dd\tau$ adalah
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
tekanan bekerja sebagai gaya volume dengan rapat $-\operatorname{\vect{grad}}P$,
yang mendorong dari tekanan tinggi menuju tekanan rendah.
\end{proposition}

\begin{proof}
Pada kotak \cref{thm:b2:fluid-kinematics:continuity}, sisi di
$x$ dan di $x + \dd x$ menerima $+P(x)\dd y\dd z$ dan $-P(x + \dd x)\dd y\dd z$
sepanjang $x$: totalnya $-\partial_xP\,\dd x\dd y\dd z$; begitu pula bagi $y$ dan $z$.
\end{proof}

\begin{definition}[Fluida sempurna]\label{def:b2:euler-bernoulli:perfect}
\index{fluida sempurna}
Yang disebut \emph{fluida sempurna} adalah fluida yang satu-satunya gaya
sentuh antara partikel bertetangganya adalah tekanan --- normal terhadap
setiap permukaan, tanpa tegangan tangensial (kental). Ia sebuah
idealisasi yang sahih, seperti dijelaskan bab berikutnya, jauh dari
dinding dan pada bilangan Reynolds tinggi: air dalam pipa jauh dari
dindingnya, udara di sekeliling sayap di luar lapisan batasnya yang
tipis, gelombang di laut.
\end{definition}

\begin{theorem}[Persamaan Euler]\label{thm:b2:euler-bernoulli:euler}
\index{persamaan Euler}
Dalam kerangka Galilei, sebuah \omterm{def:b2:euler-bernoulli:perfect}{fluida sempurna} bermassa jenis $\rho$
yang dikenai gravitasi menuruti, di setiap titik,
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
dan pada ruas kanannya ditambahkan rapat gaya volume lain mana pun
$\vect f_v$ (listrik, atau inersia dalam kerangka non-Galilei). Dalam
keadaan diam ia menyusut menjadi hukum hidrostatika
$\operatorname{\vect{grad}}P = \rho\vect g$ jilid Tahun ke-1.
\end{theorem}

\begin{proof}
Hukum kedua Newton bagi partikel bermassa $\rho\,\dd\tau$: percepatannya
adalah \omterm{thm:b2:fluid-kinematics:material}{turunan partikelnya}
(\cref{thm:b2:fluid-kinematics:material}), gayanya adalah resultan
tekanan dan beratnya $\rho\vect g\,\dd\tau$; bagilah dengan $\dd\tau$.
\end{proof}

\begin{example}[Sebuah tangki yang dipercepat]\label{ex:b2:euler-bernoulli:tank}
Air dalam tangki yang diangkut truk berpercepatan $\vect a$ berada dalam
keadaan diam dalam kerangka truknya; di sana, \omterm{def:b2:fluid-kinematics:euler}{Euler} dengan gaya inersia $-\rho\vect a$
memberi $\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$: tekanannya
tumbuh sepanjang ``gravitasi efektif'' $\vect g - \vect a$, permukaan
bebasnya (sebuah isobar) tegak lurus terhadapnya dan miring ke belakang
sebesar sudut $\arctan(a/g)$ --- \qty{11}{\degree} untuk $a = \qty{2}{m/s^2}$. Dalam
ember yang berputar dengan $\Omega$, gaya sentrifugal $\rho\Omega^2r\,\vect e_r$
memberi $P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$ dan permukaan bebasnya
adalah paraboloida $z = \Omega^2r^2/2g$.
\end{example}

\section{Teorema Bernoulli}

\begin{theorem}[Bernoulli]\label{thm:b2:euler-bernoulli:bernoulli}
\index{teorema Bernoulli}\index{tekanan stagnasi}\index{tekanan dinamik}
Untuk fluida \emph{sempurna} dan \emph{tak termampatkan} bermassa jenis
seragam dalam aliran \emph{tunak}, dalam kerangka Galilei dengan gravitasi seragam, besaran
\[
P + \tfrac12\rho v^2 + \rho gz
\]
tetap \emph{sepanjang tiap \omterm{def:b2:fluid-kinematics:euler}{garis arus}}. Jika alirannya juga
\emph{irotasional}, tetapannya sama di seluruh fluidanya. Suku
$\tfrac12\rho v^2$ adalah \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan dinamik}}; $P + \tfrac12\rho v^2$
adalah \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan stagnasi}}, yaitu tekanan yang akan dicapai fluidanya
seandainya ia dihentikan sepanjang garis arusnya.
\end{theorem}

\begin{proof}
\omterm{def:b2:fluid-kinematics:euler}{Euler} tunak dengan $(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$ dan $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$, dengan $\rho$ seragam:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
Ruas kanannya tegak lurus $\vect v$, jadi gradien besaran Bernoullinya
tak berkomponen sepanjang garis arusnya: besaran itu tetap sepanjang
garis itu. Jika $\vect\omega = \vect 0$ gradiennya lenyap di mana-mana.
\end{proof}

\begin{remark}[Apa yang dikatakan Bernoulli dan apa yang tidak]\label{rem:b2:euler-bernoulli:meaning}
Dibagi $\rho$, teoremanya berbunyi: energi mekanik tiap satuan massa,
$v^2/2 + gz$, ditambah $P/\rho$ kekal sepanjang lintasan sebuah partikel.
Suku tekanannya adalah usaha yang dilakukan \emph{pada} partikelnya oleh
tekanan fluida di belakangnya, dikurangi usaha yang dilakukannya pada
fluida di depannya: teorema energi bagi partikel yang didorong menembus
medan tekanan tanpa gesekan. Empat syaratnya: \omterm{def:b2:euler-bernoulli:perfect}{fluida sempurna} (tanpa
rugi kekentalan), tunak, tak termampatkan, dan \emph{sepanjang sebuah
\omterm{def:b2:fluid-kinematics:euler}{garis arus}}. Membandingkan dua titik pada \omterm{def:b2:fluid-kinematics:euler}{garis arus} yang berbeda hanya
boleh dalam \omterm{def:b2:fluid-kinematics:vorticity}{aliran irotasional}. Teoremanya tidak mengatakan
``udara cepat mengisap'': ia mengatakan bahwa partikel yang telah
\emph{dipercepat} (oleh penurunan tekanan) kini bertekanan lebih rendah
--- sebabnya medan tekanannya, kelajuannya adalah akibatnya.
\end{remark}

\begin{remark}[Fluida termampatkan]\label{rem:b2:euler-bernoulli:compressible}
Bagi gas massa jenisnya berubah-ubah, tetapi jika alirannya tunak,
sempurna dan isentropik bukti yang sama memberi kekekalan $h + v^2/2 + gz$
sepanjang \omterm{def:b2:fluid-kinematics:euler}{garis arus}, dengan $h$ entalpi tiap satuan massa ($h = c_PT$
untuk gas sempurna): bentuk yang dipakai bagi corong dan turbin dalam
\cref{ch:b2:open-systems}. Pada kelajuan yang jauh di bawah kelajuan bunyi
($v \lesssim \qty{100}{m/s}$ di udara) massa jenisnya berubah kurang dari $5\%$
dan teorema tak termampatkannya cukup teliti.
\end{remark}

\section{Penerapan}

\begin{proposition}[Meteran Venturi]\label{prop:b2:euler-bernoulli:venturi}
\index{efek Venturi}\index{meteran Venturi}
Dalam pipa mendatar yang menyempit dari penampang $S_1$ ke $S_2 < S_1$,
kelajuannya naik ($v_2 = v_1S_1/S_2$, kekekalan massa) dan tekanannya
turun:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
Mengukur beda tekanannya dengan manometer memberi debitnya: sebuah
\emph{\omterm{prop:b2:euler-bernoulli:venturi}{meteran Venturi}}. Efek yang sama menggerakkan karburator, penyemprot
cat, saluran udara pembakar Bunsen dan ``isapan'' kereta yang melintas.
\end{proposition}

\begin{proof}
Bernoulli sepanjang \omterm{def:b2:fluid-kinematics:euler}{garis arus} tengah pada ketinggian tetap, dengan $v_2 =
v_1S_1/S_2$; selesaikanlah untuk $v_1$.
\end{proof}

\begin{proposition}[Tabung Pitot]\label{prop:b2:euler-bernoulli:pitot}
\index{tabung Pitot}
Sebuah tabung yang menghadap alirannya, tertutup di ujung jauhnya,
menghentikan fluidanya di mulutnya (sebuah \emph{titik stagnasi}); lubang
kedua di sisinya, sejajar alirannya, membaca tekanan statiknya $P$.
Selisihnya adalah \omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan dinamiknya}:
\[
P_{\text{stag}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{stag}} - P)/\rho} .
\]
Setiap pesawat terbang mengukur kelajuan udaranya dengan cara ini.
\end{proposition}

\begin{proof}
Bernoulli sepanjang \omterm{def:b2:fluid-kinematics:euler}{garis arus} yang berakhir di titik stagnasinya, tempat
$v = 0$; lubang sisinya tidak mengganggu alirannya, yang lewat dengan $v$
dan $P$ yang tak terganggu.
\end{proof}

\begin{proposition}[Rumus Torricelli]\label{prop:b2:euler-bernoulli:torricelli}
\index{rumus Torricelli}\index{pancaran keluar}
Sebuah tangki terbuka yang besar terkuras lewat lubang kecil pada
kedalaman $h$ di bawah permukaan bebasnya; pancarannya berangkat dengan
\[
v = \sqrt{2gh} ,
\]
yaitu kelajuan jatuh bebas dari permukaannya. Debitnya adalah $D_V \approx
\alpha s\sqrt{2gh}$ dengan $s$ luas lubangnya dan $\alpha \approx 0.6$ yaitu
\emph{koefisien penyempitan} lubang bertepi tajam (garis arusnya
terus memusat melewati lubangnya).
\end{proposition}

\begin{proof}
Bernoulli dari permukaan bebasnya (diam, sebab tangkinya besar:
$v_{\text{perm}} = v\,s/S \ll v$; tekanan $P_0$) ke pancarannya (tekanan $P_0$
tepat di luar lubangnya): $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$. Alirannya
kuasi-tunak: $h$ berubah perlahan.
\end{proof}

\begin{example}[Angka]\label{ex:b2:euler-bernoulli:numbers}
Sebuah Venturi \qty{5}{cm} yang menyempit menjadi \qty{2.5}{cm} pada pipa air, dengan
$\Delta h = \qty{20}{cm}$ air (\qty{2.0}{kPa}): $v_1 = \sqrt{2 \times 2000/
(1000 \times 15)} = \qty{0.52}{m/s}$, $D_V = \qty{1.0}{L/s}$. Sebuah pesawat penumpang di
\qty{11}{km} ($\rho = \qty{0.36}{kg/m^3}$) pada \qty{250}{m/s}: \omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan dinamiknya}
\qty{11}{kPa}, $11\%$ tekanan sekitarnya --- \omterm{prop:b2:euler-bernoulli:pitot}{tabung Pitotnya} tetap
bekerja, dengan koreksi kemampatan. Sebuah menara air \qty{40}{m}
di atas kerannya: \qty{4}{bar} ketika tertutup, pancaran \qty{28}{m/s} jika dibuka
lebar (kenyataannya jauh lebih kecil: pipanya tidak sempurna).
\end{example}

\begin{proposition}[Gaya angkat pada sebuah sayap]\label{prop:b2:euler-bernoulli:lift}
\index{gaya angkat}\index{teorema Kutta--Joukowski}\index{efek Magnus}
Dalam aliran mengelilingi sebuah sayap, udaranya lewat lebih cepat di
permukaan atasnya daripada di bawahnya; tekanannya karena itu lebih
rendah di atas daripada di bawah, dan resultannya adalah \emph{\omterm{prop:b2:euler-bernoulli:lift}{gaya
angkat}} $F_L =
\tfrac12\rho v^2SC_L$ ($S$ luas sayapnya, $C_L \approx 0.3$--$1.5$ koefisien
\omterm{prop:b2:euler-bernoulli:lift}{gaya angkat}, bergantung pada profil dan sudut serangnya). Beda
kelajuannya setara dengan \emph{sirkulasi} $\Gamma$ kecepatan
mengelilingi profilnya, dan \omterm{prop:b2:euler-bernoulli:lift}{gaya angkat} tiap satuan bentang adalah $F_L' =
\rho v\Gamma$ (Kutta--Joukowski, diterima tanpa bukti). Bola yang berputar
menyeret udara ikut berkeliling dan terangkat ke samping dengan cara yang
sama: \emph{\omterm{prop:b2:euler-bernoulli:lift}{efek Magnus}} pada bola tenis yang diiris atau tendangan bebas
yang melengkung.
\end{proposition}

\begin{proof}
Bernoulli pada \omterm{def:b2:fluid-kinematics:euler}{garis arus} tepat di atas dan di bawahnya, yang berangkat
dari keadaan hulu yang sama: $P_{\text{bawah}} - P_{\text{atas}} = \tfrac12\rho
(v_{\text{atas}}^2 - v_{\text{bawah}}^2)$; integralkanlah atas talinya. Mengapa
aliran atasnya lebih cepat --- tepi belakangnya yang tajam memaksa titik
stagnasi belakang berada di sana dan menetapkan sirkulasinya (syarat
Kutta) --- berada di luar jangkauan model \omterm{def:b2:euler-bernoulli:perfect}{fluida sempurna} sendirian; itu
memerlukan lapisan batas bab berikutnya.
\end{proof}

\begin{proposition}[Sebuah aliran tak tunak: tabung U]\label{prop:b2:euler-bernoulli:utube}
\index{osilasi tabung U}\index{Bernoulli tak tunak}
Sebuah kolom zat cair sepanjang total $L$ mengisi tabung U berpenampang
seragam; disimpangkan sejauh $z$ dari kesetimbangannya, ia berosilasi menurut
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
seperti bandul sepanjang $L/2$. Secara lebih umum, untuk \omterm{def:b2:fluid-kinematics:vorticity}{aliran
irotasional} yang tak tunak berpotensial $\varphi$, \omterm{def:b2:fluid-kinematics:euler}{Euler} terintegralkan menjadi
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$, dengan tetapan
yang sama di seluruh fluidanya (\omterm{prop:b2:euler-bernoulli:utube}{Bernoulli tak tunak}).
\end{proposition}

\begin{proof}
Kelajuan $v = \dot z$ seragam sepanjang tabungnya (tak termampatkan,
penampang seragam). Proyeksikanlah \omterm{def:b2:fluid-kinematics:euler}{Euler} pada sumbu tabungnya lalu
integralkanlah sepanjang kolomnya dari satu permukaan bebas ke yang lain:
suku tekanannya saling meniadakan ($P_0$ di kedua ujungnya),
$\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$, suku konvektifnya
$\rho\,\Delta(v^2/2)$ lenyap (kelajuan sama di kedua ujungnya), dan suku
gravitasinya memberi $\rho g\cdot2z$. Jadi $\rho L\ddot z = -2\rho gz$.
Pernyataan umumnya menyusul dari $\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ untuk $\vect v = \operatorname{\vect{grad}}\varphi$.
\end{proof}

\begin{example}[Menghidupkan sebuah pipa]\label{ex:b2:euler-bernoulli:start}
Sebuah pipa mendatar sepanjang $\ell$ menghubungkan waduk bertinggi
tekan tetap $h$ ke ujung terbuka; katupnya dibuka pada $t = 0$. Dengan $v$ seragam
sepanjang pipanya, \omterm{def:b2:fluid-kinematics:euler}{Euler} yang diintegralkan dari permukaan waduknya ke
lubang keluarnya memberi $\ell\,\dd v/\dd t = gh - \tfrac12v^2$, yang penyelesaiannya
$v = V\tanh(Vt/2\ell)$ dengan $V = \sqrt{2gh}$: alirannya mencapai kelajuan
Torricelli dengan tetapan waktu $2\ell/V$ --- \qty{13}{s} bagi pipa pesat \qty{400}{m}
di bawah tinggi tekan \qty{200}{m}.
\end{example}

\begin{method}[Memakai Bernoulli]\label{met:b2:euler-bernoulli:method}
(1) Periksalah keempat syaratnya; putuskanlah apakah alirannya
irotasional (aliran hulu seragam mengelilingi rintangan: ya; aliran
dalam pipa dengan profil kecepatan: tidak --- tinggallah pada satu \omterm{def:b2:fluid-kinematics:euler}{garis
arus}). (2) Pilihlah dua titik pada satu \omterm{def:b2:fluid-kinematics:euler}{garis arus} yang sebanyak mungkin
besarannya diketahui: permukaan bebas ($P = P_0$, $v \approx 0$ jika
penampangnya besar), pancaran di udara terbuka ($P = P_0$), titik
stagnasi ($v = 0$). (3) Tambahkanlah kekekalan massa
$vS =$ tetap untuk mengaitkan kelajuannya. (4) Untuk aliran tak tunak,
integralkanlah \omterm{def:b2:fluid-kinematics:euler}{Euler} sepanjang garisnya sebagai gantinya.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:1}
Air mengalir dalam pipa mendatar berdiameter \qty{8}{cm} dengan \qty{1.5}{m/s}
dan tekanan \qty{3.0}{bar}. Ia masuk ke penampang \qty{4}{cm}: kelajuan
dan tekanannya di sana; lalu ke penampang \qty{2}{cm}: tekanannya --- apa
yang terjadi jika hasilnya keluar negatif?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:2}
\omterm{prop:b2:euler-bernoulli:pitot}{Tabung Pitot} sebuah pesawat pada \qty{5}{km} ($\rho = \qty{0.74}{kg/m^3}$)
membaca \omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan dinamik} \qty{15}{kPa}. Tentukanlah kelajuan udara
sejatinya. Penunjuk kelajuannya dikalibrasi bagi massa jenis di
permukaan laut (\qty{1.22}{kg/m^3}): ``kelajuan tertunjuk'' berapa yang
diperlihatkannya, dan mengapa penerbang justru merasa selisihnya berguna?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:3}
Sebuah tangki berisi air sedalam \qty{3.0}{m}; lubang \qty{1.0}{cm^2}
dilubangi \qty{1.0}{m} di atas lantainya. Tentukanlah kelajuan keluarnya;
debitnya (penyempitan $0.6$); di mana pancarannya menumbuk lantainya?
Pada kedalaman berapa lubang kedua harus dilubangi agar mencapai titik
yang sama?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:4}
Angin \qty{30}{m/s} berembus di atas atap datar seluas \qty{100}{m^2};
udara di bawah atapnya diam pada tekanan atmosfer ($\rho = \qty{1.2}{kg/m^3}$).
Tentukanlah beda tekanan dan gaya total pada atapnya; bandingkanlah
dengan berat atapnya (\qty{50}{kg/m^2}). Ke arah mana ia terdorong?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:5}
Sebuah tangki persegi panjang tertutup sepanjang \qty{2.0}{m}, setengah penuh air, terletak di atas
sebuah truk. (a) Dipercepat dengan \qty{3.0}{m/s^2}: sudut permukaan bebasnya,
dan selisih tinggi air antara dinding belakang dan dinding depannya.
(b) Beda tekanan antara kedua sudut dasarnya. (c)
Direm dengan \qty{6}{m/s^2} --- apakah airnya mencapai tutupnya \qty{0.8}{m} di atas
dasarnya? (d) Truk yang sama menikung pada tikungan berjari-jari \qty{40}{m} dengan
\qty{15}{m/s}: kemiringan permukaannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
Sebuah ember berjari-jari \qty{15}{cm} berisi air sedalam \qty{20}{cm} diputar
terhadap sumbunya dengan \qty{2.0}{putaran/s}. (a) Bentuk dan persamaan permukaan
bebasnya. (b) Beda tinggi antara pelek dan pusatnya. (c) Karena volumenya
kekal, tinggi airnya di pusat dan di peleknya. (d) Pada
laju putar berapa pusat dasarnya menjadi kering? (e) Mengapa
paraboloida yang sama menjadi cermin teleskop yang sempurna (genangan
raksa yang diputar)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}
\emph{Sifon.} Sebuah tabung membawa air dari tangki melewati puncak \qty{2.0}{m}
di atas permukaan bebasnya ke lubang keluar \qty{3.0}{m} di bawahnya. (a) Kelajuan
keluarnya dan, untuk tabung \qty{2}{cm}, debitnya. (b) Tekanan di puncaknya. (c)
Tinggi maksimum puncaknya agar sifonnya bekerja (tekanan uap airnya,
\qty{2.3}{kPa}, tidak boleh tercapai). (d) Mengapa sifonnya
tidak bekerja jika tabungnya penuh udara?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}
Sebuah pesawat ringan \qty{1000}{kg} menjelajah dengan \qty{60}{m/s} di permukaan laut dengan
sayap seluas \qty{15}{m^2}. (a) Beda tekanan rata-rata antara kedua
sisi sayapnya; koefisien \omterm{prop:b2:euler-bernoulli:lift}{gaya angkatnya}. (b) Jika kelajuan di bawah sayapnya
\qty{58}{m/s}, berapakah kelajuan di atasnya (ambillah keadaan hulu yang
sama)? (c) Sirkulasi mengelilingi profilnya dari rumus
Kutta--Joukowski, untuk bentang \qty{10}{m}. (d) Kelajuan macet jika
$C_L$ tak dapat melampaui $1.4$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{Stenosis.} Darah ($\rho = \qty{1060}{kg/m^3}$) mengalir dengan \qty{0.50}{m/s}
dalam arteri berpenampang \qty{0.50}{cm^2} pada \qty{13}{kPa} di atas tekanan
atmosfer. Sebuah plak menyempitkannya menjadi \qty{0.10}{cm^2}. (a) Kelajuan dan tekanan
dalam stenosisnya. (b) Jaringan di luar arterinya bertekanan sekitar
\qty{1}{kPa}: dapatkah arterinya kolaps? (c) Jika ia menyempit, kelajuannya
naik lebih jauh: jelaskanlah gejala lepas kendalinya dan mengapa penyempitan semacam itu dapat mengibas.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{Suhu stagnasi.} Untuk gas sempurna dalam aliran isentropik
tunak, $c_PT + v^2/2$ kekal sepanjang sebuah \omterm{def:b2:fluid-kinematics:euler}{garis arus}. (a) Tunjukkanlah bahwa
udara yang dihentikan di hidung sebuah pesawat mencapai $T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$, dengan $M = v/c$ bilangan Mach dan $c =
\sqrt{\gamma RT/M_{\text{mol}}}$ kelajuan bunyinya. (b) Suhu hidung
sebuah pesawat penumpang pada $M = 0.85$ di udara \qty{220}{K}; sebuah pesawat
supersonik pada $M = 2$; sebuah kapsul yang masuk kembali pada $M = 25$ (masihkah rumusnya
bermakna?). (c) Tunjukkanlah bahwa rumus Pitot tak termampatkannya
melebih-lebihkan kelajuannya pada $M = 0.85$, dan kira-kira sebanyak apa (uraikanlah
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ sampai orde kedua dalam $M^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{Peralihan saat dihidupkan.} Sebuah pipa sepanjang $\ell = \qty{50}{m}$ berpenampang
seragam menghubungkan waduk besar (tinggi tekan $h = \qty{5.0}{m}$ di atas
lubang keluarnya) ke sebuah katup di ujung terbukanya. Katupnya dibuka pada $t = 0$. (a)
Integralkanlah \omterm{def:b2:fluid-kinematics:euler}{Euler} sepanjang pipanya untuk memperoleh $\ell\,\dd v/\dd t = gh - v^2/2$.
(b) Selesaikanlah: $v = V\tanh(Vt/2\ell)$ dengan $V = \sqrt{2gh}$. (c) Waktu untuk
mencapai $90\%$ kelajuan akhirnya. (d) Katupnya kini ditutup mendadak dalam
\qty{0.5}{s} dari aliran penuh: perkirakanlah perlambatan rata-ratanya dan
tekanan lebihnya di katup dari $\rho\ell\,\dd v/\dd t$ --- dan mengapa penutupan
nyata menghasilkan jauh lebih besar (bab-bab berikutnya).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{Permukaan bebas sebuah vorteks.} Air berputar sebagai \omterm{ex:b2:fluid-kinematics:planeflows}{vorteks titik}
\cref{ch:b2:fluid-kinematics}, $v_\theta = \Gamma/2\pi r$, irotasional,
dengan permukaan bebas di $z = 0$ jauh dari sana. (a) Mengapa Bernoulli boleh diterapkan
antara dua titik mana pun? (b) Bentuk permukaan bebasnya $z(r)$. (c)
Di dalam inti Rankine ($r < a$, $v_\theta = \Omega r$, berotasi) pakailah
\omterm{def:b2:fluid-kinematics:euler}{Euler} dalam kerangka berputar atau langsung: tunjukkanlah $z = z(a) +
\Omega^2(r^2 - a^2)/2g$. (d) Kedalaman total corongnya bagi vorteks bak mandi
dengan $\Gamma = \qty{0.03}{m^2/s}$ dan $a = \qty{5}{mm}$; bagi puting beliung
($\Gamma = \qty{3e4}{m^2/s}$, $a = \qty{60}{m}$, di udara: nyatakanlah
hasilnya sebagai penurunan tekanan di pusatnya).
\end{exercise}

\section{Soal: Bendungan, pipa pesat dan turbin}

\begin{problem}[{Soal akhir pekan --- sebuah pembangkit listrik tenaga air
dari waduknya sampai pancarannya: tekanan, kelajuan, daya, rugi-rugi dan
palu air}]
\label{pb:b2:euler-bernoulli:1}

Sebuah waduk menampung air sedalam \qty{150}{m} di balik bendungan; permukaan bebasnya
berada $H = \qty{200}{m}$ di atas corong sebuah turbin. Sebuah pipa pesat (pipa
baja) berdiameter $D = \qty{3.0}{m}$ dan sepanjang $\ell = \qty{400}{m}$
membawa debit rancangan $D_V = \qty{60}{m^3/s}$ turun ke rumah turbinnya,
tempat sebuah corong mengubahnya menjadi pancaran bebas pada tekanan atmosfer $P_0 =
\qty{1.0}{bar}$ yang menghantam mangkuk sebuah roda Pelton. $\rho =
\qty{1000}{kg/m^3}$, $g = \qty{9.81}{m/s^2}$; fluidanya sempurna kecuali
dinyatakan lain.

\medskip
\noindent\textbf{Bagian I --- Dalam keadaan diam dan pada debit rancangan.}
\begin{enumerate}
  \item Tekanan di dasar bendungannya, dan gaya pada jalur tegak
        selebar \qty{1}{m} pada bendungannya dari puncak sampai dasar.
  \item Dengan katup turbinnya tertutup, tekanan dalam pipa pesatnya di
        rumah turbin.
  \item Kelajuan dalam pipa pesatnya pada debit rancangan; \omterm{thm:b2:euler-bernoulli:bernoulli}{tekanan
        dinamiknya} di sana; tekanan di ujung pipa pesat pada rumah
        turbinnya (sebelum corongnya), lewat Bernoulli dari permukaan
        waduknya.
  \item Kelajuan pancaran bebasnya; penampang dan diameter corongnya.
  \item Daya kinetik yang diemban pancarannya; tunjukkanlah bahwa ia
        sama dengan $\rho gD_VH$ lalu hitunglah.
  \item Massa dan energi kinetik air yang termuat dalam pipa pesatnya
        pada debit rancangan; waktu singgah sebuah partikel menuruni
        pipanya.
  \item Tunjukkanlah bahwa tinggi tekan $H$ terbagi, pada debit
        rancangan, antara tekanan di dasar pipa pesatnya dan energi
        kinetik airnya, dan bahwa corongnya mengubah yang pertama
        menjadi yang kedua: berapa tekanan tepat sesudah corongnya?
  \item \omterm{def:b2:euler-bernoulli:perfect}{Fluida sempurna} tak mengenal rugi tinggi tekan, fluida nyata
        mengenalnya: ruginya dalam sebuah pipa adalah
        $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ dengan $\lambda
        \approx 0.015$ bagi baja licin. Rugi tinggi tekannya dalam meter
        air, dan bagian daya yang hilang.
  \item Seorang insinyur mengusulkan pipa pesat \qty{2}{m} untuk
        menghemat baja. Ulangilah pertanyaan 7; berilah tanggapan.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Mengukur debitnya.}
Sebuah Venturi dipasang di dalam pipa pesatnya, dengan leher \qty{2.0}{m}.
\begin{enumerate}[resume]
  \item Kelajuan di lehernya dan penurunan tekanan antara pipanya dan
        lehernya pada debit rancangan.
  \item Penurunannya dibaca pada manometer raksa ($\rho_{\text{Hg}} =
        \qty{13600}{kg/m^3}$): beda tingginya.
  \item Tunjukkanlah bahwa debitnya sebanding dengan akar kuadrat
        bacaan manometernya, lalu berikanlah bacaannya pada separuh
        debit rancangan.
  \item Mengapa lubang tekanannya harus rata dengan dindingnya dan tidak
        menjulur ke dalam alirannya?
  \item Sepenggal pipanya melintas, karena alasan topografi, di atas
        sebuah punggungan \qty{6}{m} \emph{di atas} permukaan waduknya.
        Tekanannya di sana pada debit rancangan; apa yang terjadi, dan
        apa yang harus dilakukan perancangnya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Keadaan tak tunak.}
Waduknya berluas permukaan $A = \qty{2.0}{km^2}$.
\begin{enumerate}[resume]
  \item Laju turunnya muka waduknya pada debit rancangan, dalam
        sentimeter tiap jam.
  \item Turbinnya dihentikan dan corongnya (penampang dari pertanyaan 4)
        dibiarkan terbuka ke udara: waduknya terkuras oleh gravitasi.
        Tuliskanlah neraca massanya lalu carilah mukanya $h(t)$ (aliran
        Torricelli kuasi-tunak).
  \item Waktu bagi mukanya untuk turun \qty{10}{m}; bandingkanlah dengan
        waktunya pada debit rancangan yang tetap.
  \item Menghidupkan pembangkitnya: pipa pesatnya penuh, corongnya
        dibuka pada $t = 0$. Dengan mengambil kelajuannya seragam
        sepanjang pipa pesatnya dan mengabaikan bentuk corongnya,
        integralkanlah \omterm{def:b2:fluid-kinematics:euler}{Euler} sepanjang pipanya untuk memperoleh
        $\ell\,\dd v/\dd t = gH - v^2/2$, selesaikanlah, lalu berikanlah
        tetapan waktunya.
  \item Selama penghidupan ini, pada saat $v$ separuh nilai akhirnya,
        berapakah tekanan di dasar pipa pesatnya? Jelaskanlah tanda
        selisihnya terhadap nilai tunaknya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Menutup katupnya.}
\begin{enumerate}[resume]
  \item Katup di turbinnya ditutup dalam $\tau = \qty{10}{s}$; andaikanlah
        air dalam pipa pesatnya diperlambat secara seragam. Dengan
        memakai \omterm{def:b2:fluid-kinematics:euler}{Euler} sepanjang pipanya, perkirakanlah tekanan lebih di
        katupnya, dalam bar.
  \item Bagi penutupan cepat airnya tidak tak termampatkan: gelombang
        tekanan menjalar naik menyusuri pipanya dengan $c = \qty{1400}{m/s}$
        (kelajuan bunyi dalam air di dalam pipa baja ---
        \cref{ch:b2:sound-waves}), dan tekanan lebihnya adalah
        $\Delta P = \rho c\,\Delta v$ (Joukowsky). Nilainya bagi
        penghentian mendadak dari debit rancangan; bandingkanlah dengan
        tekanan statiknya. Berapa lama gelombangnya mencapai waduk dan
        kembali, dan apa artinya itu bagi makna kata ``cepat''?
  \item Untuk melindungi pipa pesatnya sebuah \emph{tangki peredam} ---
        sumuran tegak terbuka berpenampang $A_s = \qty{100}{m^2}$ ---
        disambungkan di tempat terowongan mendatar sepanjang
        $L = \qty{2000}{m}$ berpenampang $A =
        \qty{7.0}{m^2}$ dari waduknya bertemu dengan pipa pesatnya.
        Sesudah penutupan mendadak, air dalam terowongannya terus
        bergerak dan muka $z$ dalam sumurannya naik. Tuliskanlah
        kekekalan massa antara terowongan dan sumurannya.
  \item Integralkanlah \omterm{def:b2:fluid-kinematics:euler}{Euler} sepanjang terowongannya (\omterm{def:b2:euler-bernoulli:perfect}{fluida sempurna})
        untuk menunjukkan bahwa $z$ menuruti
        $\ddot z + \dfrac{gA}{LA_s}z = 0$; periode osilasinya.
  \item Kenaikan muka maksimum dalam sumurannya sesudah penutupan dari
        debit rancangan (argumen energi: energi kinetik air terowongannya
        menjadi energi potensial dalam sumurannya).
  \item Rangkumlah: kelima tekanan yang dijumpai dalam pembangkit ini
        (statik di bendungannya, statik di turbinnya, dinamik dalam
        pipanya, palu air, dan lonjakan) serta hukum yang memberi
        masing-masingnya.
\end{enumerate}
\end{problem}
