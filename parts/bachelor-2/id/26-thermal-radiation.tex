\section{Radiasi termal dan benda hitam}

\begin{definition}[Radiasi termal, absorptivitas, benda hitam]\label{def:b2:thermal-radiation:blackbody}
Setiap benda bersuhu $T$ memancarkan radiasi elektromagnetik dari
keriuhan termal muatannya: yaitu \emph{radiasi
termal}\index{radiasi termal}-nya. Permukaannya juga menerima
radiasi dari sekelilingnya lalu menyerap sebagian $\alpha$ darinya
(yaitu \emph{absorptivitas}, antara $0$ dan $1$, umumnya bergantung pada
panjang gelombangnya), sambil memantulkan atau meneruskan sisanya. Sebuah \emph{benda
hitam}\index{benda hitam} menyerap segalanya ($\alpha = 1$ pada setiap
panjang gelombang). Sebuah lubang kecil pada dinding sebuah rongga tertutup adalah
benda hitam yang praktis: radiasi yang memasukinya terserap di dindingnya
sebelum ia dapat menemukan jalan keluarnya.
\end{definition}

\begin{proposition}[Radiasi kesetimbangan dan hukum Kirchhoff]\label{prop:b2:thermal-radiation:kirchhoff}
Di dalam sebuah rongga tertutup yang dindingnya berada pada $T$, radiasinya berada dalam
kesetimbangan dengan dindingnya dan bersifat \emph{semesta}: isotropik,
tak terpolarisasi, dengan rapat energi spektral $u_\nu(T)$ (energi tiap satuan
volume dan satuan frekuensi) yang hanya bergantung pada $T$ dan $\nu$, bukan pada
bahan atau bentuk rongganya. Radiasi yang meninggalkan sebuah lubang kecil pada
rongga semacam itu adalah \emph{radiasi \omterm{def:b2:thermal-radiation:blackbody}{benda hitam}} pada $T$. Sebuah permukaan yang
menyerap bagian $\alpha_\nu$ radiasi yang diterimanya pada
frekuensi $\nu$ memancarkan, pada frekuensi itu, bagian
$\varepsilon_\nu = \alpha_\nu$ dari yang akan dipancarkan sebuah \omterm{def:b2:thermal-radiation:blackbody}{benda hitam} pada suhu
yang sama (yaitu \emph{emisivitas}\index{emisivitas}-nya):
\emph{penyerap yang baik adalah pemancar yang baik}.
\end{proposition}

\begin{proof}
Kesemestaannya: hubungkanlah dua rongga pada $T$ yang sama lewat sebuah lubang dengan sebuah
tapis yang meloloskan satu frekuensi; jika kerapatannya berbeda, energi akan
mengalir dari yang satu ke yang lain pada suhu yang sama, yang melanggar hukum
kedua. Kirchhoff: taruhlah permukaannya dalam rongganya pada $T$-nya sendiri; dalam
kesetimbangan ia harus memancarkan pada tiap frekuensinya tepat sebanyak yang diserapnya,
$\varepsilon_\nu u_\nu = \alpha_\nu u_\nu$.
\end{proof}

\section{Hukum Planck}

\begin{theorem}[Hukum Planck]\label{thm:b2:thermal-radiation:planck}
Rapat energi spektral radiasi \omterm{def:b2:thermal-radiation:blackbody}{benda hitam} pada suhu
$T$ adalah
\[
u_\nu(T) = \frac{8\pi h\nu^3}{c^3}\,\frac{1}{\eu^{h\nu/k_BT} - 1},
\]
dan daya yang dipancarkan sebuah permukaan hitam tiap satuan luas dan tiap satuan
panjang gelombang (yaitu \emph{eksitansi spektral}\index{eksitansi spektral}-nya) adalah
\[
M_\lambda(T) = \frac{2\pi hc^2}{\lambda^5}\,\frac{1}{\eu^{hc/\lambda k_BT} - 1}.
\]
Dua limitnya: bagi $h\nu \ll k_BT$, $u_\nu \approx 8\pi\nu^2 k_BT/c^3$
(Rayleigh--Jeans: tiap ragam rongganya mengemban energi klasik
$k_BT$ --- sebuah hukum yang berintegral takhingga, yaitu ``malapetaka
ultraungu''); bagi $h\nu \gg k_BT$, $u_\nu \approx (8\pi h\nu^3/c^3)
\eu^{-h\nu/k_BT}$ (Wien: yaitu faktor Boltzmann sebuah foton berenergi
$h\nu$, \cref{ch:b2:boltzmann-factor}).
\end{theorem}

\begin{proof}
Diterima tanpa bukti: pencacahan ragam rongganya ($8\pi\nu^2/c^3$ tiap satuan
volume dan frekuensi, dua polarisasi) milik jilid Tahun 3,
dan energi rerata sebuah ragamnya, $h\nu/(\eu^{h\nu/k_BT} - 1)$ alih-alih
$k_BT$ klasiknya, adalah hipotesis kuantumnya: energi dipertukarkan dengan
ragamnya dalam kuantum $h\nu$, sehingga ragam yang $h\nu \gg k_BT$
membeku. Hubungan $M_\lambda\,\dd\lambda = (c/4)\,u_\nu\,\dd\nu$ adalah
faktor geometri yang dihitung dalam \cref{thm:b2:thermal-radiation:stefan}.
\end{proof}

\begin{proposition}[Hukum pergeseran Wien]\label{prop:b2:thermal-radiation:wien}
\omterm{thm:b2:thermal-radiation:planck}{Eksitansi spektral} $M_\lambda$ memuncak pada
\[
\lambda_{\max}\,T = \frac{hc}{4.965\,k_B} = \qty{2.90e-3}{m.K} .
\]
Sebuah benda pada \qty{300}{K} memancar terutama dekat \qty{10}{\micro m}; Matahari
(\qty{5800}{K}) dekat \qty{0.5}{\micro m}, dalam hijaunya, yang di sana matanya
paling peka --- bukan kebetulan.
\end{proposition}

\begin{proof}
Dengan $x = hc/\lambda k_BT$, $M_\lambda \propto x^5/(\eu^x - 1)$; $\dd/\dd x = 0$
memberi $5(\eu^x - 1) = x\eu^x$, yakni $x = 5(1 - \eu^{-x})$, yang akar
taknolnya adalah $x = 4.965$.
\end{proof}

\begin{theorem}[Hukum Stefan--Boltzmann]\label{thm:b2:thermal-radiation:stefan}
Sebuah permukaan hitam pada $T$ memancarkan, tiap satuan luas, daya total
\[
M = \int_0^\infty M_\lambda\,\dd\lambda = \sigma T^4, \qquad
\sigma = \frac{2\pi^5k_B^4}{15h^3c^2} = \qty{5.67e-8}{W/m^2/K^4} ;
\]
rapat energi radiasi kesetimbangannya adalah $u = (4\sigma/c)T^4$
dan tekanannya $P = u/3$. Sebuah permukaan kelabu beremisivitas $\varepsilon$
memancarkan $\varepsilon\sigma T^4$.
\end{theorem}

\begin{proof}
Integralkanlah rapat Planck: $u = \int u_\nu\,\dd\nu = (8\pi h/c^3)(k_BT/h)^4
\int_0^\infty x^3\,\dd x/(\eu^x - 1)$ dan integralnya adalah $\pi^4/15$
(diterima tanpa bukti dari analisis): $u = (8\pi^5k_B^4/15h^3c^3)\,T^4$. Daya
yang meninggalkan sebuah lubang berluas satuan pada dinding rongganya adalah $cu/4$:
radiasinya isotropik, foton yang bergerak ke arah lubangnya dalam
sudut ruang $\dd\Omega$ pada sudut $\theta$ dari normalnya mengemban
fluks $c\,u\,(\dd\Omega/4\pi)\cos\theta$, dan $\int\cos\theta\,\dd\Omega/4\pi$
pada setengah bola keluarnya sama dengan $1/4$. Maka $M = cu/4 = \sigma T^4$.
Tekanannya, seperti bagi sebarang gas isotropik partikel yang bergerak dengan $c$, adalah
$u/3$.
\end{proof}

\begin{example}[Orde besaran]\label{ex:b2:thermal-radiation:orders}
Pada \qty{300}{K}: $\sigma T^4 = \qty{460}{W/m^2}$ --- seorang manusia seluas
\qty{1.7}{m^2} memancarkan \qty{800}{W}, tetapi menerima hampir sebanyak itu dari
dindingnya pada \qty{293}{K}; rugi bersihnya, $\sigma A(T^4 - T_0^4) \approx
\qty{100}{W}$, adalah seluruh keluaran metabolisme tubuhnya, itulah sebabnya orang
memerlukan pakaian dalam sebuah ruangan yang terasa sejuk. Permukaan Matahari: $\sigma
(5800)^4 = \qty{6.4e7}{W/m^2}$, totalnya $3.9 \times 10^{26}\,\unit{W}$. Filamen
sebuah bola lampu pada \qty{2800}{K}: \qty{3.5e6}{W/m^2}, \qty{60}{W} dari setengah
sentimeter persegi wolfram. Latar belakang kosmik, \qty{2.7}{K}:
puncaknya pada \qty{1}{mm}, $\qty{3e-6}{W/m^2}$, dan $400$ foton tiap sentimeter
kubik alam semesta.
\end{example}

\section{Pertukaran radiatif}

\begin{proposition}[Pertukaran bersih dan koefisien radiatif]\label{prop:b2:thermal-radiation:exchange}
Sebuah benda kelabu kecil (beremisivitas $\varepsilon$, berluas $S$, bersuhu $T$)
di dalam sekeliling yang luas pada $T_0$ kehilangan daya bersih
\[
P = \varepsilon\sigma S\,(T^4 - T_0^4)
\ \approx\ 4\varepsilon\sigma T_0^3\,S\,(T - T_0) = h_{\text{r}}S(T - T_0)
\quad\text{bagi } |T - T_0| \ll T_0 ,
\]
dengan $h_{\text{r}} = 4\varepsilon\sigma T_0^3 \approx \qty{6}{W/m^2/K}$ pada
suhu ruang --- sebanding dengan konveksi alami: dalam sebuah ruangan,
radiasi mengemban sekitar separuh kalor yang meninggalkan sebuah tubuh atau sebuah radiator.
Antara dua keping hitam sejajar yang luas rapat fluks bersihnya adalah
$\sigma(T_1^4 - T_2^4)$; bagi keping kelabu beremisivitas $\varepsilon_1$,
$\varepsilon_2$ ia menjadi $\sigma(T_1^4 - T_2^4)/(1/\varepsilon_1 + 1/\varepsilon_2
- 1)$, dan tiap \emph{perisai radiasi} tipis yang disisipkan membaginya
lebih jauh --- yaitu asas termos hampa dan kerajang emas pada
satelit.
\end{proposition}

\begin{proof}
Bendanya memancarkan $\varepsilon\sigma T^4$ lalu menyerap bagian $\alpha =
\varepsilon$ radiasi hitam $\sigma T_0^4$ yang mengisi kurungannya
(Kirchhoff). Linearkanlah $T^4 - T_0^4 \approx 4T_0^3(T - T_0)$. Bagi
keping kelabunya, jumlahkanlah pantulan gandanya (sebuah deret ukur).
\end{proof}

\begin{example}[Beku di bawah langit cerah, dan kamera termal]\label{ex:b2:thermal-radiation:frost}
Pada malam yang cerah langitnya memancar seperti sebuah benda pada sekitar \qty{250}{K}
(yaitu atmosfer atas yang kering), sementara udara dekat tanahnya berada pada
\qty{278}{K}. Sehelai daun atau atap sebuah mobil ($\varepsilon = 0.95$) mengimbangkan
rugi radiatifnya ke langitnya terhadap konveksi dari udaranya ($h =
\qty{5}{W/m^2/K}$): $\varepsilon\sigma(T^4 - 250^4) = h(278 - T)$ memberi $T
\approx \qty{266}{K}$ --- beku pada \qty{-7}{\celsius} di rumputnya sementara
termometernya membaca $+5$. Sebuah awan (yang memancar pada \qty{275}{K}) mengangkat $T$
menjadi \qty{277}{K}: tak ada beku. Sebuah kamera termal mengukur radiansi dekat
\qty{10}{\micro m} lalu mengubahnya menjadi sebuah suhu dengan menganggap
$\varepsilon \approx 0.95$; sebuah logam yang dipoles ($\varepsilon = 0.1$) kebanyakan
memantulkan ruangannya lalu tampak pada suhu ruangannya berapa pun suhunya
sendiri --- dari situlah pita hitam yang ditempelkan ahli termografi pada suku yang mengkilat.
\end{example}

\section{Matahari, Bumi dan rumah kaca}

\begin{proposition}[Tetapan surya dan suhu efektif]\label{prop:b2:thermal-radiation:earth}
Matahari (berjari-jari $R_\odot$, bersuhu permukaan $T_\odot$) menyerahkan pada
jarak $d$ rapat fluks $S = \sigma T_\odot^4 (R_\odot/d)^2 =
\qty{1361}{W/m^2}$ di atas atmosfer Bumi (yaitu \emph{tetapan
surya}\index{tetapan surya}). Sebuah planet beralbedo $A$ (yaitu bagian
yang terpantul) menyerap $(1 - A)S\pi R^2$ lalu, dalam keadaan tunaknya, memancarkannya
dari seluruh permukaannya $4\pi R^2$ sebagai sebuah \omterm{def:b2:thermal-radiation:blackbody}{benda hitam} pada \emph{suhu
efektif}\index{suhu efektif}-nya
\[
T_{\text{e}} = \Big(\frac{(1 - A)S}{4\sigma}\Big)^{1/4} = \qty{255}{K}
\quad\text{bagi Bumi } (A = 0.30) .
\]
\end{proposition}

\begin{proof}
Kekekalan energi: yang terserap $=$ yang terpancar, dengan \omterm{prop:b2:laser:gain}{penampang lintang} $\pi R^2$
bagi berkas yang tercegat, dan bola penuhnya bagi pancarannya (siang dan
malam, yang disebarkan rotasi dan sirkulasinya).
\end{proof}

\begin{proposition}[Model rumah kaca satu lapis]\label{prop:b2:thermal-radiation:greenhouse}
Kelilingilah planetnya dengan sebuah lapisan atmosfer tipis yang bening
bagi cahaya matahari tetapi menyerap semua inframerah yang dipancarkan permukaannya, lalu
memancar sebagai sebuah \omterm{def:b2:thermal-radiation:blackbody}{benda hitam} pada $T_{\text{a}}$ baik ke atas maupun ke bawah. Dalam
keadaan tunaknya,
\[
T_{\text{a}} = T_{\text{e}}, \qquad
T_{\text{s}} = 2^{1/4}\,T_{\text{e}} = \qty{303}{K} .
\]
Jika lapisannya hanya menyerap bagian $\varepsilon$ inframerahnya,
$T_{\text{s}}^4 = T_{\text{e}}^4/(1 - \varepsilon/2)$; \qty{288}{K} yang
teramati bersesuaian dengan $\varepsilon \approx 0.78$. Permukaannya lebih
hangat daripada \omterm{prop:b2:thermal-radiation:earth}{suhu efektifnya} karena ia menerima, di samping
Matahari, inframerah yang dikirimkan atmosfernya kembali ke bawah --- yaitu
\emph{efek rumah kaca}\index{efek rumah kaca}, \qty{33}{K} di Bumi,
yang tanpanya samudranya akan membeku.
\end{proposition}

\begin{proof}
Imbangan lapisannya: menyerap $\sigma T_{\text{s}}^4$, memancarkan $2\sigma
T_{\text{a}}^4$. Imbangan planetnya yang dilihat dari antariksa: memancarkan $\sigma
T_{\text{a}}^4 = (1 - A)S/4 = \sigma T_{\text{e}}^4$. Imbangan
permukaannya: menerima $(1 - A)S/4 + \sigma T_{\text{a}}^4 = 2\sigma T_{\text{e}}^4$,
memancarkan $\sigma T_{\text{s}}^4$. Bagi lapisan yang sebagian, gantilah
inframerah yang terserap dan terpancarnya dengan $\varepsilon\sigma T_{\text{s}}^4$ dan
$\varepsilon\sigma T_{\text{a}}^4$ lalu biarkanlah bagian $1 - \varepsilon$
radiasi permukaannya lolos langsung.
\end{proof}

\begin{remark}[Mengapa rumah kacanya selektif]\label{rem:b2:thermal-radiation:selective}
Cahaya matahari memuncak pada \qty{0.5}{\micro m}, radiasi Bumi pada
\qty{10}{\micro m}: kedua spektrumnya nyaris tak bertumpang tindih, sehingga sebuah gas dapat
bening bagi yang pertama dan menyerap bagi yang kedua. Nitrogen dan
oksigen tak menyerap keduanya; uap air, karbon dioksida, metana dan ozon
mempunyai pita getaran dalam inframerah dan merupakan gas rumah kacanya.
Jendela antara \qty{8}{} dan \qty{13}{\micro m} adalah yang ditembus pandang kamera termal,
yang dirujuk \qty{250}{K} langit malamnya, dan yang
dimanfaatkan cat pendingin radiatif. Kaca sebuah rumah kaca yang sungguhan melakukan
hal yang sama (bening bagi \qty{0.5}{\micro m}, legap bagi \qty{10}{\micro m}),
tetapi sebagian besarnya ia menghalangi konveksi --- namanya sedikit tak adil.
\end{remark}

\begin{method}[Taksiran radiasi]\label{meth:b2:thermal-radiation:method}
(1) $\lambda_{\max} = \qty{2900}{\micro m.K}/T$ memberitahukan di mana spektrumnya
terletak; (2) $\varepsilon\sigma T^4$ tiap satuan luas; (3) pertukarannya: yang terpancar
dikurangi yang terserap, yang dilinearkan menjadi $h_{\text{r}} = 4\varepsilon\sigma T^3$ dekat
suhu ruang; (4) sebuah planet: $(1 - A)S/4 = \sigma T_{\text{e}}^4$, lalu
lapisannya; (5) bandingkanlah dengan hantaran dan konveksi --- radiasi
menang pada suhu tinggi ($T^4$) dan dalam hampa; (6) perhatikanlah emisivitasnya:
logam mengkilat \qty{0.05}{}, cat, kulit, air, tanah $\approx 0.9$--$0.98$
dalam inframerah berapa pun warna tampaknya.
\end{method}

(\qty{310}{K}), filamen sebuah bola lampu (\qty{2800}{K}), latar belakang kosmik

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:2}
Dari $T_\odot = \qty{5772}{K}$, $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.496e11}{m}$:
tentukanlah \omterm{prop:b2:thermal-radiation:earth}{tetapan suryanya}, luminositas Mataharinya, daya yang dicegat
Bumi ($R = \qty{6.37e6}{m}$), dan hal yang sama dibagi $8 \times 10^9$
orang. Bandingkanlah dengan konsumsi daya dunianya (\qty{2e13}{W}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:3}
Seorang manusia: kulitnya pada \qty{306}{K}, $\varepsilon = 0.98$, \qty{1.7}{m^2}, dalam sebuah
ruangan pada \qty{293}{K}. Tentukanlah daya terpancarnya, daya terserapnya, rugi bersihnya;
koefisien radiatifnya $h_{\text{r}}$; bandingkanlah dengan \qty{100}{W}
metabolismenya lalu simpulkanlah tentang pakaian dan tentang sebuah ruangan \qty{20}{\celsius}
berdinding dingin.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:4}
Sebuah bola lampu \qty{60}{W}: filamen wolfram pada \qty{2800}{K}, $\varepsilon = 0.35$.
Tentukanlah luas pemancarnya; panjang sebuah kawat \qty{50}{\micro m} berluas itu;
sekitar $8\%$ dayanya tampak: berapa daya cahayanya; dan sebuah LED yang memberi
cahaya yang sama pada efisiensi $30\%$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:5}
\emph{Kedua limitnya.} (a) Uraikanlah $u_\nu$ Planck bagi $h\nu \ll k_BT$ lalu
kenalilah energi tiap ragamnya. (b) Tunjukkanlah bahwa hukum Rayleigh--Jeans
memberi energi total yang takhingga. (c) Pada \qty{300}{K}, tentukanlah frekuensi yang
$h\nu = k_BT$; simpulkanlah bahwa \omterm{def:b2:thermal-radiation:blackbody}{radiasi termal} radio dan gelombang mikro
bersifat klasik. (d) Dalam limit Wiennya, tafsirkanlah $\eu^{-h\nu/k_BT}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:6}
\emph{Wien.} (a) Turunkanlah $x = 5(1 - \eu^{-x})$ lalu selesaikanlah secara numerik
sampai tiga angka. (b) Berapa nilai $\lambda_{\max}T$. (c) Tunjukkanlah bahwa $u_\nu$
memuncak pada $h\nu = 2.82\,k_BT$, dan bahwa ini tak bersesuaian dengan
$c/\lambda_{\max}$: mengapa tidak? (d) Berapa suhu sebuah bintang yang spektrumnya
memuncak pada \qty{290}{nm}; warnanya bagi mata.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:7}
\emph{Stefan.} (a) Kerjakanlah pengintegralan hukum Planck, mengingat
$\int_0^\infty x^3\dd x/(\eu^x - 1) = \pi^4/15$, dan nilai $\sigma$-nya.
(b) Bernalarkanlah faktor $c/4$ antara kerapatannya dan eksitansinya. (c)
Mengingat $\int_0^\infty x^2\dd x/(\eu^x - 1) = 2.404$, tentukanlah jumlah foton
tiap satuan volume pada $T$, dan energi foton reratanya dalam satuan $k_BT$.
(d) Berapa foton tiap sentimeter kubik pada \qty{300}{K} dan pada \qty{2.73}{K}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:8}
\emph{Perisai.} Dua keping hitam yang luas pada $T_1$, $T_2$. (a) Tentukanlah fluks bersihnya.
(b) Sehelai lembaran hitam tipis ditaruh di antaranya: berapa suhunya dan
fluks barunya. (c) Dengan $n$ lembar. (d) Keping kelabu beremisivitas $\varepsilon$:
tunjukkanlah bahwa fluksnya adalah $\sigma(T_1^4 - T_2^4)/(2/\varepsilon - 1)$ lalu hitunglah
ia bagi permukaan yang dipoles ($\varepsilon = 0.05$) pada \qty{300}{K} dan
\qty{77}{K} --- yaitu termos hampanya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:9}
\emph{Beku malam.} Permukaan ber-$\varepsilon = 0.95$, langit pada \qty{250}{K}, udara pada
\qty{278}{K} dengan $h = \qty{5}{W/m^2/K}$. (a) Tuliskanlah imbangannya lalu selesaikanlah
bagi suhu permukaannya. (b) Di bawah sebuah awan pada \qty{275}{K}. (c) Mengapa
angin mencegah beku? (d) Mengapa sebuah mobil yang diparkir di bawah sebuah pohon terlepas
darinya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:10}
\emph{Planet.} (a) Tentukanlah \omterm{prop:b2:thermal-radiation:earth}{suhu efektif} Venus ($S = \qty{2600}{W/m^2}$,
$A = 0.75$), Bumi, Mars ($S = \qty{590}{W/m^2}$, $A = 0.25$). (b) Suhu
permukaannya adalah \qty{737}{K}, \qty{288}{K}, \qty{215}{K}:
berapa penghangatan rumah kaca masing-masingnya. (c) Dalam model dengan $n$ lapisan legap,
tunjukkanlah $T_{\text{s}}^4 = (n + 1)T_{\text{e}}^4$; berapa lapisan bagi Venus?
(d) Bulan: $A = 0.12$, tanpa atmosfer, rotasi lambat --- berapa suhu
titik subsuryanya, dan mengapa sisi malamnya turun sampai \qty{100}{K}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:11}
\emph{Kepekaan iklim tanpa \omterm{def:b2:feedback-oscillators:loop}{umpan balik}.} (a) Linearkanlah
imbangan planetnya: sebuah paksaan tambahan kecil $\Delta F$ (dalam \unit{W/m^2}) di
puncak atmosfernya menaikkan \omterm{prop:b2:thermal-radiation:earth}{suhu efektifnya} sebesar $\Delta T
= \Delta F/4\sigma T_{\text{e}}^3$. Hitunglah $4\sigma T_{\text{e}}^3$. (b)
Melipatduakan karbon dioksidanya adalah sebuah paksaan sekitar \qty{3.7}{W/m^2}:
berapa $\Delta T$ dalam model ini. (c) Lapisan campur samudranya (\qty{50}{m}
air) menyimpan kalornya: berapa tetapan waktu tanggapannya. (d) Mengapa \omterm{def:b2:feedback-oscillators:loop}{umpan balik}
uap air dan albedo esnya membuat kepekaan yang sebenarnya lebih besar
daripada (b)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:12}
\emph{Pirometri.} (a) Dalam limit Wiennya, tunjukkanlah bahwa nisbah
radiansi pada dua panjang gelombang, $\lambda_1$ dan $\lambda_2$, memberi $T$
tanpa bergantung pada \omterm{prop:b2:thermal-radiation:kirchhoff}{emisivitas} kelabu. (b) Sebatang baja: $M_{0.65}/M_{0.9}
= 0.10$: berapa suhunya. (c) Sebuah kamera termal pada \qty{10}{\micro m} menganggap
$\varepsilon = 0.95$; ia memandang aluminium yang dipoles ($\varepsilon = 0.1$)
pada \qty{350}{K} dalam sebuah ruangan pada \qty{293}{K}: dalam rezim yang terlinearkan,
suhu berapa yang dilaporkannya? (d) Mengapa sebuah permukaan bercat di samping
aluminiumnya, pada \qty{350}{K} yang sama, terbaca dengan benar?
\end{exercise}

\section{Soal: Anggaran energi Bumi dan bola lampu pijar}

\begin{problem}[{Soal akhir pekan --- benda hitam dari Matahari sampai
filamennya}]\label{pb:b2:thermal-radiation:1}
Data: $\sigma = \qty{5.67e-8}{W/m^2/K^4}$, $\lambda_{\max}T = \qty{2.90e-3}{m.K}$,
$h = \qty{6.63e-34}{J.s}$, $k_B = \qty{1.38e-23}{J/K}$, $c = \qty{3.00e8}{m/s}$;
Matahari: $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.50e11}{m}$; Bumi: $R =
\qty{6.37e6}{m}$, albedonya $0.30$.

\textbf{Bagian I --- Matahari sebagai \omterm{def:b2:thermal-radiation:blackbody}{benda hitam}.}
\begin{enumerate}
  \item Spektrum suryanya memuncak pada \qty{0.50}{\micro m}: berapa suhu
        permukaannya.
  \item Eksitansi permukaannya dan luminositas totalnya.
  \item \omterm{prop:b2:thermal-radiation:earth}{Tetapan surya} di Bumi; rapat energi dan tekanan
        cahaya matahari di sana.
  \item Energi foton rerata ($\approx 2.7\,k_BT$) dan jumlah foton surya
        tiap sekon tiap meter persegi di Bumi.
  \item Matahari telah memancar dengan laju ini selama $4.5 \times 10^9$ tahun:
        berapa energi yang terpancar; bandingkanlah dengan $Mc^2$ bagi $M = \qty{2e30}{kg}$.
\end{enumerate}

\textbf{Bagian II --- Bumi tanpa atmosfer.}
\begin{enumerate}[resume]
  \item Daya yang diserap Bumi; berapa \omterm{prop:b2:thermal-radiation:earth}{suhu efektifnya}.
  \item Mengapa imbangannya membagi dengan $4$? Berapa suhunya
        di titik subsurya sebuah Bumi tanpa udara yang tak berotasi
        ($A = 0.30$)?
  \item Bulan ($A = 0.12$): berapa suhu subsuryanya; dan mengapa sisi
        malamnya turun sampai \qty{100}{K} (pikirkanlah kalor yang tersimpan dalam
        regolitnya selama siang dua pekannya, \qty{1e6}{J/m^2/K} efektif).
  \item Panjang gelombang puncak pancaran Bumi; bandingkanlah dengan
        milik Mataharinya.
  \item Jika lebih banyak es atau awan menaikkan albedonya menjadi $0.35$, seberapa besar
        \omterm{prop:b2:thermal-radiation:earth}{suhu efektifnya} akan turun?
\end{enumerate}

\textbf{Bagian III --- Rumah kacanya.}
\begin{enumerate}[resume]
  \item Tuliskanlah ketiga imbangannya (antariksa, lapisan, permukaan)
        model satu lapisnya lalu peroleh $T_{\text{s}} = 2^{1/4}T_{\text{e}}$.
  \item Dengan sebuah lapisan yang menyerap bagian $\varepsilon$
        inframerahnya: tentukanlah $T_{\text{s}}(\varepsilon)$; nilai $\varepsilon$
        yang memberi \qty{288}{K}.
  \item Gas mana yang menyerap, mana yang tidak, dan dalam pita panjang
        gelombang yang mana; apa itu jendela atmosfernya?
  \item Inframerah yang dikirimkan atmosfernya kembali ke tanahnya,
        dalam \unit{W/m^2}, dalam model $\varepsilon = 0.78$; bandingkanlah dengan
        cahaya matahari yang terserapnya.
  \item Sebuah paksaan \qty{3.7}{W/m^2} (karbon dioksida yang berlipat dua):
        berapa perubahan suhunya $\Delta T = \Delta F/4\sigma T_{\text{e}}^3$
        tanpa \omterm{def:b2:feedback-oscillators:loop}{umpan balik}.
  \item Kapasitas kalor sebuah lapisan campur samudra \qty{50}{m} tiap meter
        persegi dan tetapan waktu tanggapannya.
  \item Mengapa awan sekaligus mendinginkan (albedo) dan menghangatkan (inframerah)
        permukaannya; efek mana yang menang pada malam hari?
\end{enumerate}

\textbf{Bagian IV --- Bola lampu pijarnya.}
Sebuah bola lampu \qty{60}{W} mempunyai filamen wolfram pada \qty{2800}{K},
$\varepsilon = 0.35$; bagian daya sebuah \omterm{def:b2:thermal-radiation:blackbody}{benda hitam} yang jatuh dalam
cahaya tampaknya ($0.4$--\qty{0.7}{\micro m}) adalah $0.08$ pada \qty{2800}{K} dan $0.14$
pada \qty{3200}{K}.
\begin{enumerate}[resume]
  \item Luas pemancar filamennya; panjang sebuah kawat \qty{50}{\micro m}
        yang mempunyainya; mengapa ia digulung?
  \item Panjang gelombang puncaknya; daya tampaknya; efisiensi cahayanya.
  \item Hambatan filamennya dalam kerja pada \qty{230}{V};
        \omterm{prop:b2:charges-currents-conduction:drude}{hambatan jenis} wolframnya sekitar sepuluh kali lebih kecil pada suhu
        ruang: berapa arusnya pada penyalaan, dan apa akibatnya.
  \item Pada \qty{3200}{K}: berapa luas yang diperlukan bagi \qty{60}{W} yang sama dan
        daya tampaknya; mengapa tidak menjalankan setiap bola lampu pada \qty{3200}{K}
        (laju penguapan wolframnya kira-kira berlipat dua tiap
        \qty{50}{K})?
  \item Apa yang dilakukan daur halogennya, dan mengapa ia membolehkan filamen yang
        lebih panas?
  \item Ke mana sisa \qty{60}{W}-nya pergi, dan mengapa kaca bola
        lampunya mencapai \qty{150}{\celsius}?
  \item Waktu pendinginan filamennya setelah dimatikan: taksirlah dari
        massanya (\qty{20}{mg}, $c = \qty{140}{J/kg/K}$) dan daya terpancarnya.
  \item Ringkaslah: apa yang ditetapkan suhu sebuah \omterm{def:b2:thermal-radiation:blackbody}{benda hitam}, dan apa
        yang dimiliki bersama oleh filamennya, Mataharinya dan Buminya.
\end{enumerate}
\end{problem}
