\chapter{Laser: Pancaran Terangsang dan Berkas Gauss}\label{ch:b2:laser}

Sebuah titik merah pada layar kuliah, pembaca kode batang di kasir,
berkas yang membaca sebuah cakram, mengelas bodi mobil, mengemban internet
di bawah samudra, mengoreksi sebuah kornea dan mengukur jarak ke
Bulan sampai semilimeter: tiap satunya adalah sebuah \emph{\omterm{def:b2:laser:cavity}{laser}}, sebuah sumber
cahaya yang tak seperti nyala api atau lampu mana pun. Cahayanya berwarna tunggal sampai satu bagian
dalam semiliar, tetap dalam berkas bermilimeter di seluruh ruangan, dan dapat
difokuskan ke sebuah bintik selebar beberapa panjang gelombang, yang di sana iradiansinya melampaui
iradiansi permukaan Matahari sejuta kali. Tak satu pun ini berasal dari
sejenis atom baru: semuanya berasal dari sebuah proses yang dikenali Einstein pada 1917
--- \emph{\omterm{def:b2:laser:processes}{pancaran terangsang}}, yaitu pemancaran sebuah foton yang serupa dengan
foton yang sudah ada --- dan dari menata materi sehingga proses ini
menang atas \omterm{def:b2:laser:processes}{serapan}, lalu menaruh penguatnya di antara dua cermin
sehingga ia menyuapi dirinya sendiri. Bab ini membangun \omterm{def:b2:laser:cavity}{lasernya} dalam tiga langkah:
bagaimana cahaya dan populasi atom bertukar energi (\omterm{def:b2:laser:processes}{koefisien
Einstein}), kapan pertukarannya menguatkan (\omterm{prop:b2:laser:gain}{pembalikan populasi} dan
penguatan), dan kapan penguatnya menjadi sebuah osilator (rongganya, ambangnya
dan ragamnya). Ia berakhir dengan bentuk berkas yang keluar,
yaitu \omterm{prop:b2:laser:gaussian}{berkas Gauss}, yang \omterm{prop:b2:laser:gaussian}{pinggang} dan \omterm{def:b2:maxwell-equations:operators}{divergensinya} memutuskan apa yang dapat dilakukan sebuah \omterm{def:b2:laser:cavity}{laser}
pada jarak jauh dan di sebuah fokus.

\section{Cahaya dan materi: ketiga prosesnya}

\begin{definition}[Serapan, pancaran spontan dan terangsang]\label{def:b2:laser:processes}
Tinjaulah atom dengan dua aras energi $E_1 < E_2$, berpopulasi (rapat
cacah) $N_1$, $N_2$, dan cahaya berfrekuensi $\nu$ dengan
$h\nu = E_2 - E_1$ dan rapat energi spektral $u(\nu)$ (energi tiap satuan
volume tiap satuan frekuensi). Tiga proses mempertukarkan energi:
\begin{itemize}
\item \emph{serapan}\index{serapan (cahaya)}: sebuah atom dalam
      $E_1$ menyerap sebuah foton lalu naik ke $E_2$, dengan laju
      $B_{12}\,u(\nu)\,N_1$ tiap satuan volume dan waktu;
\item \emph{pancaran spontan}\index{pancaran spontan}: sebuah atom
      dalam $E_2$ jatuh ke $E_1$ dengan sendirinya, sambil memancarkan sebuah foton dalam arah
      acak dengan fase acak, dengan laju $A_{21}N_2$ ---
      $1/A_{21} = \tau$ adalah umur arasnya;
\item \emph{pancaran terangsang}\index{pancaran terangsang}: sebuah foton
      berfrekuensi $\nu$ yang melewati sebuah atom dalam $E_2$ mengimbasnya untuk jatuh,
      sambil memancarkan sebuah foton kedua yang \emph{serupa} dengan yang pertama (sama
      frekuensi, arah, fase dan polarisasinya), dengan laju
      $B_{21}\,u(\nu)\,N_2$.
\end{itemize}
$A_{21}$, $B_{12}$, $B_{21}$ adalah \emph{koefisien
Einstein}\index{koefisien Einstein} transisinya.
\end{definition}

\begin{proposition}[Hubungan Einstein]\label{prop:b2:laser:einstein}
Bagi aras yang tak berdegenerasi,
\[
B_{12} = B_{21} \equiv B, \qquad
\frac{A_{21}}{B_{21}} = \frac{8\pi h\nu^3}{c^3}.
\]
\omterm{def:b2:laser:processes}{Pancaran terangsang} dan \omterm{def:b2:laser:processes}{serapan} mempunyai koefisien yang sama;
laju spontannya tumbuh sebagai $\nu^3$ relatif terhadap laju terangsangnya.
\end{proposition}

\begin{proof}
Taruhlah atomnya dalam kesetimbangan dengan radiasi bersuhu $T$. Dua
kenyataan dipinjam dari bab kemudian: populasi dua aras
pada kesetimbangan bernisbah $N_2/N_1 = \eu^{-h\nu/k_BT}$ (yaitu
faktor Boltzmann, \cref{ch:b2:boltzmann-factor}), dan radiasi
kesetimbangannya mempunyai rapat spektral Planck
$u(\nu) = (8\pi h\nu^3/c^3)/(\eu^{h\nu/k_BT} - 1)$
(\cref{ch:b2:thermal-radiation}). Pada kesetimbangan populasi arasnya
tunak: $B_{12}uN_1 = A_{21}N_2 + B_{21}uN_2$, maka
\[
u = \frac{A_{21}}{B_{12}\,N_1/N_2 - B_{21}}
  = \frac{A_{21}}{B_{12}\eu^{h\nu/k_BT} - B_{21}}.
\]
Ini harus sama dengan ungkapan Planck pada setiap $T$: limit $T \to \infty$
(yang $u \to \infty$ di sana) memaksa $B_{12} = B_{21}$, dan
perbandingannya lalu memberi $A_{21}/B_{21} = 8\pi h\nu^3/c^3$.
Koefisiennya adalah sifat atomnya, jadi hubungannya berlaku di luar
kesetimbangan juga.
\end{proof}

\begin{remark}[Mengapa lampu tidak menguatkan]\label{rem:b2:laser:lamps}
Dalam sebuah gas pada \qty{300}{K}, bagi sebuah transisi tampak ($h\nu \approx
\qty{2}{eV}$, $k_BT \approx \qty{0.025}{eV}$), $N_2/N_1 = \eu^{-80} \approx
10^{-35}$: pada dasarnya tak ada atom yang tereksitasi, dan cahayanya hanya terserap. Sebuah
lucutan atau nyala api mengisi $E_2$, tetapi masih dengan $N_2 < N_1$, dan
tiap foton yang terpancar kalah banyak dibandingkan \omterm{def:b2:laser:processes}{serapannya}; lagi pula nisbah
\omterm{def:b2:laser:processes}{pancaran spontan} terhadap terangsang pada kesetimbangan,
$A_{21}/(B_{21}u) = \eu^{h\nu/k_BT} - 1$, sangat besar secara astronomis dalam
cahaya tampak --- cahaya sebuah lampu bersifat spontan: fasenya acak, arahnya
serba, dan lebarnya lebar penuh garisnya. Hanya pada frekuensi radio dan gelombang mikro
($h\nu \ll k_BT$) \omterm{def:b2:laser:processes}{pancaran terangsang} bersaing
secara alami; piranti pertama sejenis ini memang maser, pada
\qty{24}{GHz} (1954), sebelum \omterm{def:b2:laser:cavity}{laser} optiknya (1960).
\end{remark}

\section{Penguatan: pembalikan populasi dan gain}

\begin{proposition}[Penguatan sebuah medium]\label{prop:b2:laser:gain}
Sebuah berkas berintensitas $I$ pada frekuensi $\nu$ yang menyeberangi mediumnya
memperoleh energi lewat \omterm{def:b2:laser:processes}{pancaran terangsang} lalu kehilangannya lewat \omterm{def:b2:laser:processes}{serapan}
(\omterm{def:b2:laser:processes}{pancaran spontannya} pergi ke segala arah dan nyaris tak menyumbang pada
berkasnya). Sepanjang $\dd z$,
\[
\dd I = \sigma\,(N_2 - N_1)\,I\,\dd z, \qquad
I(z) = I(0)\,\eu^{g z}, \qquad g = \sigma(N_2 - N_1),
\]
dengan $\sigma$ (sebuah luas, yaitu \emph{penampang lintang}\index{penampang lintang (sebuah transisi)} transisinya, $\sigma \propto B\,h\nu/c$) mengukur
kekuatan transisinya. Mediumnya \emph{menguatkan} ($g > 0$)
hanya jika
\[
N_2 > N_1 :
\]
yaitu sebuah \emph{pembalikan populasi}\index{pembalikan populasi}. Tanpanya
($N_2 < N_1$, yaitu hal setiap medium pada kesetimbangan) berkasnya
terserap.
\end{proposition}

\begin{proof}
Tiap satuan volume dan waktu, $B u (N_2 - N_1)$ transisi menambahkan (atau menyingkirkan)
sebuah foton $h\nu$ pada berkasnya; dengan $u = I/c$ (tiap satuan frekuensi, sepanjang
lebar garisnya), daya yang diperoleh tiap satuan volume adalah
$B h\nu (N_2 - N_1) I/c$, dan ini adalah $\dd I/\dd z$; definisikanlah $\sigma =
Bh\nu/c$ (dikalikan faktor bentuk garisnya).
\end{proof}

\begin{remark}[Mengapa dua aras tak dapat dibalik]\label{rem:b2:laser:twolevel}
Pompalah sebuah sistem dua aras sekeras yang Anda suka dengan cahaya pada $\nu$:
pompanya menyerap ketika $N_1 > N_2$ lalu merangsang pancaran ketika $N_2 > N_1$,
dan populasinya paling banter cenderung ke kesamaan, $N_2 = N_1$, yang di sana
mediumnya bening tetapi tak memperoleh apa pun. Sebuah pembalikan menuntut sekurangnya
tiga aras: pompalah dari aras dasarnya ke sebuah aras berumur pendek yang
meluruh cepat ke aras \omterm{def:b2:laser:cavity}{laser} atasnya, yang \emph{berumur panjang}
(metastabil) dan karena itu menyimpan populasi. Dalam skema
\emph{tiga aras}\index{laser tiga aras} (rubi, \omterm{def:b2:laser:cavity}{laser}
pertama) aras \omterm{def:b2:laser:cavity}{laser} bawahnya adalah keadaan dasarnya, sehingga lebih dari separuh
atomnya harus diangkat sebelum pembalikannya: pompanya kejam. Dalam skema
\emph{empat aras}\index{laser empat aras} (He--Ne, Nd:YAG, kebanyakan
\omterm{def:b2:laser:cavity}{laser}) aras \omterm{def:b2:laser:cavity}{laser} bawahnya adalah sebuah aras tereksitasi yang cepat kosong
ke keadaan dasarnya; ia nyaris kosong setiap saat, dan populasi kecil
di aras atasnya sudah merupakan sebuah pembalikan --- ambangnya
rendah.
\end{remark}

\begin{proposition}[Penjenuhan penguatannya]\label{prop:b2:laser:saturation}
Pembalikannya dipertahankan sebuah pompa yang memasok atom ke aras
atasnya dengan laju $R$ tiap satuan volume, melawan peluruhan $1/\tau$ dan
\omterm{def:b2:laser:processes}{pancaran terangsang} $\sigma I (N_2 - N_1)/h\nu$. Dalam sebuah medium empat aras
($N_1 \approx 0$) keadaan tunaknya adalah
\[
N_2 = \frac{R\tau}{1 + I/I_{\text{s}}}, \qquad
g = \frac{g_0}{1 + I/I_{\text{s}}}, \qquad
I_{\text{s}} = \frac{h\nu}{\sigma\tau} :
\]
\emph{penguatan sinyal kecil} $g_0 = \sigma R\tau$ dikurangi berkasnya
sendiri begitu $I$ mendekati \emph{intensitas
jenuh}\index{intensitas jenuh} $I_{\text{s}}$ --- penguatnya
taklinear, dan ketaklinearan inilah yang akan menetapkan daya
\omterm{def:b2:laser:cavity}{lasernya}.
\end{proposition}

\begin{proof}
$\dd N_2/\dd t = R - N_2/\tau - \sigma I N_2/h\nu = 0$.
\end{proof}

\section{Osilatornya: rongga, ambang dan ragam}

\begin{definition}[Rongga laser]\label{def:b2:laser:cavity}
Sebuah \emph{laser}\index{laser} adalah sebuah medium penguat sepanjang $\ell$
yang ditaruh di antara dua cermin (reflektansi $R_1$, $R_2$) yang terpisah jarak $L$:
yaitu sebuah rongga Fabry--Pérot (\cref{ch:b2:gratings}). Satu cerminnya, yaitu
\emph{pengopel keluaran}, meneruskan sebagian ($T_2 = 1 - R_2$ beberapa
persen) lalu meloloskan berkasnya keluar. Cahaya yang mengelilingi rongganya
dikuatkan dua kali oleh mediumnya lalu dilemahkan oleh cerminnya dan oleh
rugi lainnya (hamburan, \omterm{def:b2:laser:processes}{serapan}, \omterm{def:b2:diffraction:huygens}{difraksi}).
\end{definition}

\begin{theorem}[Syarat berayun]\label{thm:b2:laser:threshold}
Sebuah gelombang beramplitudo kompleks $\underline s$ memperbanyak dirinya sesudah satu
perjalanan pulang pergi jika
\[
\underline s\,\cdot\, \eu^{g\ell}\sqrt{R_1}\,\eu^{g\ell}\sqrt{R_2}\,
\eu^{-\alpha\cdot 2L}\,\eu^{2\iu kL} = \underline s
\]
($\alpha$: rugi lainnya tiap satuan panjang). Maka dua syarat.
\begin{itemize}
\item \emph{Amplitudo}: $R_1R_2\,\eu^{2(g\ell - \alpha L)} \ge 1$, yakni
      penguatannya harus mencapai \emph{ambang}\index{ambang (laser)}
      \[
      g_{\text{th}} = \frac{1}{\ell}\Big(\alpha L - \tfrac12\ln(R_1R_2)\Big)
      \approx \frac{1}{2\ell}\,(T_2 + \text{rugi tiap perjalanan pulang pergi}) ;
      \]
\item \emph{Fase}: $2kL = 2\pi p$, yakni $\nu_p = p\,c/2L$ --- \omterm{def:b2:laser:cavity}{lasernya}
      hanya berayun pada \emph{ragam membujur}\index{ragam membujur}
      rongganya, yang berjarak $c/2L$, dan di antaranya hanya pada
      yang terletak di bawah kurva penguatannya yang $g(\nu) \ge g_{\text{th}}$ di sana.
\end{itemize}
\end{theorem}

\begin{proof}
Amplitudo dan fase faktor pulang perginya harus keduanya sepele;
hampirannya memakai $-\ln R \approx 1 - R = T$ bagi $T \ll 1$.
\end{proof}

\begin{proposition}[Keadaan tunak di atas ambang]\label{prop:b2:laser:steady}
Beranjak dari \omterm{def:b2:laser:processes}{pancaran spontan} sebuah atom tunggal, setiap ragam yang
$g_0 > g_{\text{th}}$ tumbuh eksponensial, perjalanan demi perjalanan
pulang pergi; ketika intensitasnya tumbuh, penguatannya menjenuh
(\cref{prop:b2:laser:saturation}) sampai ia tepat sama dengan ruginya:
\[
g(I) = g_{\text{th}} \quad\Longrightarrow\quad
I_{\text{rongga}} = I_{\text{s}}\Big(\frac{g_0}{g_{\text{th}}} - 1\Big),
\qquad P_{\text{kel}} = T_2\,I_{\text{rongga}}\,A .
\]
Penguatannya \emph{terjepit} pada nilai ambangnya; tiap atom tambahan
yang dipompa di atas ambangnya menjadi sebuah foton keluaran, sehingga daya keluarannya
tumbuh linear dengan pompanya di atas pompa ambangnya.
\end{proposition}

\begin{remark}[Lasernya adalah osilator umpan balik]\label{rem:b2:laser:feedback}
Bandingkanlah dengan \omterm{prop:b2:feedback-oscillators:wien}{osilator jembatan Wien}
\cref{ch:b2:feedback-oscillators}: sebuah penguat (mediumnya yang terbalik),
sebuah \omterm{def:b2:feedback-oscillators:loop}{umpan balik} yang selektif frekuensi (rongganya, yang hanya meloloskan ragamnya),
sebuah syarat mula (\omterm{def:b2:feedback-oscillators:loop}{penguatan simpal} $> 1$, yakni $g_0 > g_{\text{th}}$),
awal dari derau (di sini \omterm{def:b2:laser:processes}{pancaran spontan}), dan sebuah amplitudo yang ditetapkan
ketaklinearan penguatnya (penjenuhan penguatan). \omterm{def:b2:laser:cavity}{Lasernya} adalah
anggota optik keluarga itu.
\end{remark}

\begin{example}[Laser helium--neon]\label{ex:b2:laser:hene}
Sebuah tabung kaca sepanjang \qty{30}{cm}, berliang \qty{1}{mm}, yang memuat helium dan
neon pada sekitar seperseribu atmosfer, yang diseberangi sebuah lucutan
beberapa miliampere. Elektronnya mengeksitasi helium ke sebuah aras metastabil pada
\qty{20.6}{eV}, yang menyerahkan energinya lewat tumbukan ke sebuah aras neon pada
ketinggian yang hampir tepat sama; dari sana neonnya meluruh ke sebuah aras yang lebih rendah
(yang cepat kosong ke keadaan dasarnya --- empat aras) sambil memancar pada
\qty{632.8}{nm}. Penguatan sinyal kecilnya mungil, beberapa persen tiap
lintasan, sehingga cerminnya harus sangat baik ($R_1 = 0.999$, $R_2 = 0.99$)
dan tabungnya bersih; garis penguatannya melebar Doppler menjadi
$\Delta\nu \approx \qty{1.5}{GHz}$, ragamnya terpisah $c/2L = \qty{500}{MHz}$,
sehingga dua atau tiga ragam berayun sekaligus. Keluarannya \qty{1}{mW} bagi
beberapa watt lucutan: efisiensinya $10^{-4}$; tetapi dengan panjang gelombang yang
terdefinisi sampai $10^{-6}$ dan sebuah berkas yang tetap selebar \qty{1}{mm} menyeberangi
laboratoriumnya.
\end{example}

\section{Berkas Gauss}

\begin{proposition}[Berkas Gauss]\label{prop:b2:laser:gaussian}
Berkas yang dipancarkan sebuah rongga mantap bercermin lengkung adalah (dalam
ragam melintang dasarnya) sebuah \emph{berkas Gauss}\index{berkas
Gauss}: pada jarak $z$ dari penampang tersempitnya (yaitu
\emph{pinggang}\index{pinggang (sebuah berkas)}, berjari-jari $w_0$) profil
intensitasnya adalah
\[
I(r,z) = \frac{2P}{\pi w(z)^2}\,\exp\Big(-\frac{2r^2}{w(z)^2}\Big),
\qquad
w(z) = w_0\sqrt{1 + \Big(\frac{z}{z_{\text{R}}}\Big)^2},
\qquad
z_{\text{R}} = \frac{\pi w_0^2}{\lambda},
\]
dengan $P$ yaitu daya totalnya dan $z_{\text{R}}$ \emph{panjang
Rayleigh}\index{panjang Rayleigh}: sepanjang $\pm z_{\text{R}}$ berkasnya tetap
dalam $\sqrt2$ \omterm{prop:b2:laser:gaussian}{pinggangnya}, dan jauh di luarnya ia menyebar dengan
setengah sudut
\[
\theta = \frac{\lambda}{\pi w_0} .
\]
Berkasnya adalah penyelesaian \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombang} dalam hampiran
paraksial; kita menerima bentuknya lalu memakainya. \omterm{def:b2:scalar-light-model:path}{Muka gelombangnya} bidang di
\omterm{prop:b2:laser:gaussian}{pinggangnya} dan bola jauh di sana.
\end{proposition}

\begin{proof}
Diterima tanpa bukti pada aras ini (perhitungannya adalah perambatan optika
Fourier sebuah sebaran apertur Gauss); sifat kuncinya adalah
yang sudah diberikan \omterm{def:b2:diffraction:huygens}{difraksi} dalam \cref{ch:b2:diffraction}: sebuah
apertur berukuran $w_0$ menyebar sepanjang $\lambda/w_0$, di sini dengan faktor
eksak $1/\pi$.
\end{proof}

\begin{proposition}[Memfokuskan dan melebarkan sebuah berkas]\label{prop:b2:laser:focus}
Sebuah \omterm{prop:b2:laser:gaussian}{berkas Gauss} berjari-jari $w$ (jauh lebih besar daripada medan jauhnya
di lensanya, yakni nyaris terkolimasi) yang jatuh pada sebuah lensa berjarak
fokus $f$ difokuskan menjadi sebuah \omterm{prop:b2:laser:gaussian}{pinggang}
\[
w_0' = \frac{\lambda f}{\pi w}
\]
di fokusnya, dengan \omterm{prop:b2:laser:gaussian}{panjang Rayleigh} $\pi w_0'^2/\lambda$. Iradiansi
puncaknya di sana adalah $2P/\pi w_0'^2$. Sebaliknya sebuah teleskop afokal
berperbesaran $M$ mengubah sebuah berkas berjari-jari $w$ menjadi berjari-jari $Mw$
yang \omterm{def:b2:maxwell-equations:operators}{divergensinya} $M$ kali lebih kecil: untuk mengirimkan sebuah berkas jauh, buatlah ia
lebar dahulu.
\end{proposition}

\begin{proof}
Lensanya mengubah \omterm{def:b2:scalar-light-model:path}{muka gelombang} bidangnya menjadi sebuah bola yang memusat di $f$;
\omterm{def:b2:diffraction:huygens}{difraksi} sebuah apertur berjari-jari $w$ memberi sebaran sudut
$\lambda/\pi w$, maka bintik fokusnya $f\lambda/\pi w$ --- selaras dengan
$\theta = \lambda/\pi w_0'$ yang dibaca terbalik.
\end{proof}

\begin{example}[Memotong, membaca, menunjuk]\label{ex:b2:laser:focus}
Sebuah \omterm{def:b2:laser:cavity}{laser} karbon dioksida \qty{1}{kW} pada \qty{10.6}{\micro m}, berkasnya berjari-jari
\qty{10}{mm}, yang difokuskan $f = \qty{100}{mm}$: $w_0' = \qty{34}{\micro m}$,
iradiansi puncaknya $2P/\pi w_0'^2 = \qty{5e11}{W/m^2}$ --- baja mendidih.
Sebuah penunjuk \qty{1}{mW}, $w_0 = \qty{0.5}{mm}$ pada \qty{633}{nm}: $\theta =
\qty{0.4}{mrad}$, sebuah bintik \qty{4}{cm} pada \qty{50}{m}, sebuah iradiansi
\qty{2.5}{kW/m^2} di pintu keluarnya --- di atas cahaya matahari, itulah sebabnya bahkan
semiliwatt pun tak boleh memasuki sebuah mata: lensa matanya akan memfokuskannya ke sebuah
bintik \qty{10}{\micro m} pada retinanya pada berjuta watt tiap meter
persegi. Sebuah pembaca cakram pada \qty{650}{nm} dengan sebuah lensa berapertur numerik
$0.6$ memfokus sampai sekitar $\lambda/2\mathrm{NA} \approx \qty{0.5}{\micro m}$,
seukuran sebuah lubang mungilnya.
\end{example}

\begin{remark}[Apa yang membuat cahaya laser istimewa]\label{rem:b2:laser:special}
\emph{Keterarahan}: satu ragam melintang, \omterm{def:b2:maxwell-equations:operators}{divergensi} pada batas
\omterm{def:b2:diffraction:huygens}{difraksinya}. \emph{Kemonokromatikan}: satu atau beberapa ragam rongga, masing-masingnya
lebih sempit daripada semegahertz --- \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensi} bermeter sampai berkilometer
(\cref{ch:b2:scalar-light-model}). \emph{\omterm{prop:b2:two-wave-interference:spatial}{Koherensi ruang}}: seluruh
berkasnya adalah satu gelombang, sehingga ia berinterferensi dengan dirinya sendiri di mana pun (holografi,
interferometri). \emph{Kecerahan}: semiliwatt dalam berkas yang terbatas \omterm{def:b2:diffraction:huygens}{difraksi}
menenggelamkan Matahari tiap satuan sudut ruang dan lebar pita. \emph{Daya}
dalam pulsa: sejoule dalam senanosekon adalah segigawatt, dalam sefemtosekon
sepetawatt. Dan, karena berkasnya adalah keluaran sebuah osilator, ia dapat
dimodulasi pada gigahertz --- yaitu pembawa \omterm{prop:b2:guided-waves:fibre}{serat optiknya}
(\cref{ch:b2:guided-waves}).
\end{remark}

\begin{method}[Taksiran laser]\label{meth:b2:laser:method}
(1) Energi fotonnya $h\nu = hc/\lambda$ dan laju fotonnya $P/h\nu$.
(2) Ambangnya: $g_{\text{th}}\ell \approx \tfrac12(T_2 + \text{rugi})$;
bandingkanlah dengan penguatan sinyal kecil $g_0\ell$ mediumnya.
(3) Ragamnya: $c/2L$ terhadap lebar penguatannya; cacahlah ragam yang berayun.
(4) Dayanya: penguatannya terjepit pada ambangnya; keluarannya $\propto$ (pompa $-$
pompa ambangnya). (5) Berkasnya: $z_{\text{R}} = \pi w_0^2/\lambda$, $\theta =
\lambda/\pi w_0$, bintik fokusnya $\lambda f/\pi w$, iradiansi puncaknya
$2P/\pi w_0^2$. (6) Keselamatan: bandingkanlah iradiansi retinanya dengan
milik Matahari.
\end{method}

\section{Soal: Sebuah laser helium--neon, dari tabungnya sampai ke Bulan}

\begin{problem}[{Soal akhir pekan --- sebuah laser yang dibangun, dicirikan dan
dipakai}]\label{pb:b2:laser:1}
\omterm{def:b2:laser:cavity}{Lasernya} di bangku: tabung kaca $L = \qty{30}{cm}$, diameter liangnya
\qty{1}{mm}, campuran helium--neon, lucutan \qty{5}{mA} di bawah
\qty{1.5}{kV}; cermin $R_1 = 0.999$, $R_2 = 0.99$; panjang gelombangnya
\qty{632.8}{nm}; keluarannya \qty{1}{mW}. Datanya: aras metastabil He terletak
pada \qty{20.61}{eV}, aras \omterm{def:b2:laser:cavity}{laser} atas Ne pada \qty{20.66}{eV} dengan
umur \qty{100}{ns}, aras \omterm{def:b2:laser:cavity}{laser} bawahnya pada \qty{18.70}{eV} dengan
umur \qty{10}{ns}; suhu gasnya \qty{400}{K}; massa molar neon
\qty{20}{g/mol}; \omterm{prop:b2:laser:gain}{penampang lintangnya} $\sigma = \qty{3e-17}{m^2}$; $h =
\qty{6.63e-34}{J.s}$, $k_B = \qty{1.38e-23}{J/K}$, $k_BT$ pada \qty{400}{K}
$= \qty{0.0345}{eV}$.

\textbf{Bagian I --- Mediumnya.}
\begin{enumerate}
  \item Energi foton dan frekuensi garis \omterm{def:b2:laser:cavity}{lasernya}.
  \item Aras He dan Ne berselisih \qty{0.05}{eV}: bandingkanlah dengan
        $k_BT$ lalu jelaskanlah mengapa alih tumbukan He $\to$ Ne
        mangkus.
  \item Nisbah Boltzmann aras atas Ne terhadap keadaan dasarnya pada
        \qty{400}{K} tanpa lucutannya: ulaslah.
  \item Jelaskanlah, dari kedua umurnya, mengapa sebuah pembalikan antara
        aras Ne-nya mudah dipertahankan begitu aras atasnya disuapi.
  \item Lebar Doppler garisnya: kelajuan neon yang paling mungkin pada
        \qty{400}{K} dan $\Delta\nu \approx \nu\,v/c$ dalam orde
        besaran (tepatnya $\Delta\nu = 2\nu\sqrt{2k_BT\ln2/Mc^2}$).
  \item Dengan $\Delta N = N_2 - N_1 = \qty{3e15}{m^{-3}}$: berapa koefisien
        penguatan sinyal kecilnya dan penguatan tiap lintasan sepanjang \qty{30}{cm}.
\end{enumerate}

\textbf{Bagian II --- Rongganya.}
\begin{enumerate}[resume]
  \item Koefisien penguatan ambangnya; berapa marginnya terhadap pertanyaan 6.
  \item Jarak ragamnya; jumlah ragam yang dapat berayun.
  \item Daya dalam rongganya dan iradiansi dalam rongganya di liangnya.
  \item Apa yang menetapkan penguatannya dalam keadaan tunak, dan ke mana energi
        pompa yang berlebih pergi?
  \item Daya listrik yang masuk dan daya optik yang keluar: berapa efisiensinya; dua
        alasan mengapa ia begitu rendah.
  \item Awal mulanya: penguatan bersih tiap perjalanan pulang pergi $2(g_0 - g_{\text{th}})L$,
        cacah foton dalam rongganya pada keadaan tunak, jumlah
        perjalanan pulang pergi dari satu foton, dan waktu mulanya.
  \item Tabungnya menghangat lalu memanjang sebesar \qty{1}{\micro m}: berapa geseran
        frekuensi sebuah ragamnya; apa akibatnya bagi sebuah \omterm{def:b2:laser:cavity}{laser} ragam tunggal.
  \item Tabungnya ditutup jendela pada \omterm{prop:b2:wave-interfaces:brewster}{sudut Brewster}: apa yang dilakukan
        ini pada polarisasi keluarannya, dan mengapa ia tak
        menambah rugi?
\end{enumerate}

\textbf{Bagian III --- Berkasnya.}
\begin{enumerate}[resume]
  \item Rongganya menghasilkan sebuah \omterm{prop:b2:laser:gaussian}{pinggang} $w_0 = \qty{0.3}{mm}$: berapa \omterm{prop:b2:laser:gaussian}{panjang
        Rayleighnya}, \omterm{def:b2:maxwell-equations:operators}{divergensinya}, diameter berkasnya \qty{10}{m} jauhnya.
  \item Iradiansi puncaknya di pintu keluarnya; bandingkanlah dengan cahaya matahari
        (\qty{1}{kW/m^2}).
  \item Sebuah lensa ber-$f = \qty{50}{mm}$: berapa \omterm{prop:b2:laser:gaussian}{pinggang} dan iradiansi puncaknya di
        fokusnya.
  \item Berkasnya memasuki sebuah mata (jarak fokusnya \qty{17}{mm}): berapa bintik retinanya
        dan iradiansinya; mengapa sebuah kedipan tak terlalu lambat bagi \qty{1}{mW},
        dan apa arti ``kelas 2''.
  \item Sebuah pelebar berkas $\times 10$: berapa \omterm{def:b2:maxwell-equations:operators}{divergensi} baru dan diameter bintiknya
        pada \qty{1}{km}.
  \item Berapa \omterm{prop:b2:scalar-light-model:coherence}{panjang koherensinya} jika \omterm{def:b2:laser:cavity}{lasernya} berjalan pada satu ragam tunggal berlebar
        \qty{1}{MHz}; jika ia berjalan pada tiga ragam yang merentangi \qty{1}{GHz}.
\end{enumerate}

\textbf{Bagian IV --- Ke Bulan.}
Sebuah stasiun pengukur jarak menembakkan pulsa \qty{100}{mJ} sepanjang \qty{100}{ps} pada
\qty{532}{nm} menembus sebuah teleskop \qty{1}{m}; atmosfernya menyebarkan
berkas keluarnya menjadi \qty{1}{''} ($\qty{4.8e-6}{rad}$).
\begin{enumerate}[resume]
  \item Foton tiap pulsa dan daya puncaknya.
  \item Jari-jari bintiknya di Bulan (\qty{3.8e8}{m}); bandingkanlah dengan
        bintik yang terbatasi \omterm{def:b2:diffraction:huygens}{difraksinya}.
  \item Sebuah larik pemantul \qty{0.1}{m^2} mencegat berapa bagian
        pulsanya, berapa foton?
  \item Kubus sudut \qty{4}{cm} lariknya mengembalikan cahayanya dengan
        sebaran $\lambda/d$: berapa jari-jari bintiknya di Bumi, bagian yang tertangkap
        teleskopnya, foton yang terdeteksi tiap pulsa (ambillah efisiensi optik
        keseluruhan $0.1$).
  \item Waktu pulang perginya sekitar \qty{2.56}{s}, yang terukur sampai
        \qty{100}{ps}: berapa ketelitian jaraknya tiap pulsa; sesudah $10^4$
        pantulan; bandingkanlah dengan \qty{3.8}{cm} tiap tahun yang dengannya
        Bulan menjauh.
\end{enumerate}
\end{problem}
