\chapter{Difraksi Fraunhofer dan Penapisan Ruang}\label{ch:b2:diffraction}

Menyipitlah ke arah lampu jalan menembus celah antara dua jari dan
lampunya merentang menjadi sepita rumbai; pandanglah sebuah bintang menembus
teleskop terhalus dan Anda melihat bukan sebuah titik melainkan sebuah cakram kecil yang bercincin
lingkaran samar. Cahaya yang menembus sebuah bukaan menyebar --- makin
sempit bukaannya, makin lebar sebarannya --- dan tak ada lensa yang dapat memfokuskannya
lebih rapat daripada yang dibolehkan sebaran ini. Inilah \emph{\omterm{def:b2:diffraction:huygens}{difraksi}}, alasan
sebuah mikroskop tak dapat melihat sebuah virus, daya urai sebuah teleskop tumbuh
dengan diameternya, dan sebuah berkas laser tak dapat tetap sejajar. Bab ini
menghitung pola \omterm{def:b2:diffraction:huygens}{difraksi} sebuah apertur dalam medan jauh,
batas yang ditetapkannya pada setiap instrumen optik, caranya bergabung dengan
interferensi, dan gagasan --- milik Abbe --- yang mengubah bidang fokus sebuah
lensa menjadi tempat sebuah citra dapat ditapis.

\section{Asas Huygens--Fresnel dan rezim Fraunhofer}

\begin{definition}[Asas Huygens--Fresnel]\label{def:b2:diffraction:huygens}
\index{asas Huygens--Fresnel}\index{difraksi}\index{difraksi Fraunhofer}
Setiap titik sebuah \omterm{def:b2:scalar-light-model:path}{muka gelombang} (atau sebuah apertur yang disinari sebuah gelombang) berlaku sebagai
sumber sekunder yang memancarkan anak \omterm{prop:b2:sound-waves:spherical}{gelombang bola}, sefase dengan gelombangnya
di sana; gelombang di luarnya adalah jumlah koheren anak gelombangnya. Bagi sebuah
apertur $\Sigma$ yang disinari gelombang bidang beramplitudo $a$, amplitudo
yang mencapai sebuah titik jauh adalah $\underline s(M) = K\iint_\Sigma\eu^{-\iu\varphi(P, M)}\dd S$, dengan
$\varphi(P, M)$ yaitu fase lintasan dari titik apertur $P$ ke $M$
($K$ sebuah tetapan). Difraksi \emph{Fraunhofer} (medan jauh) adalah hal
yang $M$-nya berada di tak hingga --- atau pada bidang fokus sebuah lensa --- sehingga
sinar dari semua titik apertur menuju $M$ sejajar, dalam
arah $\vect u$, dan
\[
\varphi(P, M) = \varphi_0 - \frac{2\pi}\lambda\,\vect{OP}\cdot\vect u , \qquad
\underline s(\vect u) \propto \iint_\Sigma\eu^{\iu(2\pi/\lambda)\vect{OP}\cdot\vect u}\dd S .
\]
Polanya diamati di tak hingga, atau di fokus $F'$ sebuah lensa,
yang di sana arah $\vect u$ bersudut kecil $(\alpha, \beta)$ dipetakan ke
titik $(f\alpha, f\beta)$.
\end{definition}
\admitted

\begin{remark}[Kapan cukup jauh?]\label{rem:b2:diffraction:fresnelnumber}
\index{bilangan Fresnel}
Bagi sebuah apertur berukuran $D$ yang dipandang pada jarak $L$ tanpa lensa,
sinarnya cukup sejajar ketika suku kuadratik
\cref{exo:b2:scalar-light-model:12} dapat diabaikan, $D^2/L \ll \lambda$ ---
yaitu \emph{\omterm{rem:b2:diffraction:fresnelnumber}{bilangan Fresnel}} $D^2/\lambda L \ll 1$. Sebuah celah \qty{0.1}{mm} pada
\qty{500}{nm} menuntut $L \gg \qty{2}{cm}$; sebuah apertur \qty{1}{cm}, $L \gg \qty{200}{m}$ ---
dari situlah lensanya, yang membawa tak hingga ke bidang fokusnya. Lebih dekat
(\omterm{def:b2:diffraction:huygens}{difraksi} Fresnel) polanya lebih rumit; bayang-bayang sebuah
tepi lurus, misalnya, mempunyai rumbai sepanjang sisi terangnya.
\end{remark}

\section{Celah tunggal dan apertur lingkaran}

\begin{proposition}[Difraksi oleh sebuah celah]\label{prop:b2:diffraction:slit}
\index{difraksi celah tunggal}\index{fungsi sinc}
Sebuah celah selebar $b$ (yang panjang sepanjang $y$) yang disinari pada datang tegak lurus mengirimkan dalam
arah $\theta$ (dalam bidang yang tegak lurus celahnya)
intensitas
\[
I(\theta) = I_0\,\operatorname{sinc}^2\Bigl(\frac{\pi b\sin\theta}\lambda\Bigr) , \qquad \operatorname{sinc}x = \frac{\sin x}x :
\]
sebuah maksimum pusat yang terang berlebar setengah sudut $\lambda/b$ (nol pertamanya di
$\sin\theta = \pm\lambda/b$), yang memuat $90\%$ cahayanya, dan \omterm{thm:b2:gratings:nwaves}{maksimum sekunder}
di sekitar $\sin\theta = \pm1.43\lambda/b$, $\pm2.46\lambda/b$, \dots, berintensitas
nisbi $4.7\%$, $1.6\%$, \dots --- makin sempit celahnya, makin lebar
polanya. Pada bidang fokus sebuah lensa berjarak fokus $f$
maksimum pusatnya selebar $2\lambda f/b$.
\end{proposition}

\begin{proof}
$\underline s(\theta) \propto \int_{-b/2}^{b/2}\eu^{\iu kx\sin\theta}\dd x = b\,\operatorname{sinc}(kb\sin\theta/2)$
dengan $k = 2\pi/\lambda$; kuadratkanlah. Nolnya di tempat $b\sin\theta = m\lambda$, $m \ne 0$: anak
gelombang dari kedua paruh celahnya lalu meniadakan berpasangan.
\end{proof}

\begin{proposition}[Apertur lingkaran; cakram Airy]\label{prop:b2:diffraction:airy}
\index{cakram Airy}\index{kriteria Rayleigh (resolusi)}\index{daya urai (instrumen)}
Sebuah apertur lingkaran berdiameter $D$ memberi sebuah cakram terang pusat, yaitu
\emph{\omterm{prop:b2:diffraction:airy}{cakram Airy}}, yang dikelilingi cincin samar; cincin gelap pertamanya berada di
\[
\sin\theta = 1.22\,\frac\lambda D ,
\]
dan cakramnya memuat $84\%$ cahayanya. Setiap instrumen dengan
pupil masuk berdiameter $D$ --- mata, kamera, teleskop, objektif
mikroskop --- mengubah sebuah sumber titik menjadi \omterm{prop:b2:diffraction:airy}{cakram Airy} berjari-jari sudut
itu; dua titik \emph{terurai} (kriteria Rayleigh) jika
pemisahan sudutnya melampaui $1.22\lambda/D$: yaitu daya urai
instrumennya.
\end{proposition}
\admitted

\begin{example}[Mata, teleskop, mikroskop]\label{ex:b2:diffraction:instruments}
Mata ($D = \qty{3}{mm}$, $\lambda = \qty{550}{nm}$): $1.22\lambda/D = \qty{2.2e-4}{rad}$, sekitar
\qty{45}{''} --- sebuah rincian \qty{1}{mm} pada \qty{4}{m}, yang memang sekitar
ketajaman sebuah mata yang baik: evolusi mencocokkan sel retinanya dengan
batas \omterm{def:b2:diffraction:huygens}{difraksinya}. Sebuah teleskop \qty{2.4}{m}: \qty{0.06}{''}, seribu kali
lebih baik, jika atmosfernya membolehkannya (dari tanah tidak:
\qty{1}{''} ``kaburan atmosfer'', dari situlah teleskop antariksa dan optika adaptif). Sebuah
piringan radio \qty{100}{m} pada $\lambda = \qty{21}{cm}$: \qty{9}{'}, lebih buruk daripada mata;
interferometer lintas benua memulihkan resolusinya. Sebuah
objektif mikroskop bersudut apertur $\alpha$ mengurai $0.61\lambda/n\sin\alpha$
--- sekitar \qty{0.2}{\micro m} dalam cahaya tampak, berapa pun perbesarannya: yaitu
batas yang mendorong mikroskopi ke elektron dan ke sinar-X.
\end{example}

\section{Difraksi dan interferensi bersama-sama}

\begin{proposition}[Celah Young berlebar berhingga; selubung kisinya]\label{prop:b2:diffraction:combined}
\index{selubung (difraksi)}\index{orde yang hilang}
Dua celah selebar $b$ yang pusatnya terpisah $a$ memberi
\[
I(\theta) = 4I_0\,\operatorname{sinc}^2\Bigl(\frac{\pi b\sin\theta}\lambda\Bigr)\cos^2\Bigl(\frac{\pi a\sin\theta}\lambda\Bigr) :
\]
rumbai Young (berjarak $\lambda/a$ dalam $\sin\theta$) di bawah selubung
celah tunggalnya (berlebar $2\lambda/b$); $a/b$ rumbai mengisi cuping pusatnya, dan
rumbai di $a\sin\theta = p\lambda$ yang jatuh pada sebuah nol selubungnya
($b\sin\theta = m\lambda$) \emph{hilang}. Sebuah kisi $N$ celah selebar
$b$ juga mempunyai \omterm{thm:b2:gratings:nwaves}{maksimum utamanya} yang termodulasi selubung yang sama ---
yang dimiringkan kilas \cref{ch:b2:gratings} ke arah orde yang dipakai.
\end{proposition}

\begin{proof}
Integral aperturnya terbelah menjadi jumlah pada kedua celahnya dari integral
yang sama yang tergeser sebesar $\pm a/2$: $\underline s \propto b\,\operatorname{sinc}(\pi b\sin\theta/\lambda)
\times2\cos(\pi a\sin\theta/\lambda)$. Bagi $N$ celah faktor keduanya adalah
jumlah $N$ gelombangnya.
\end{proof}

\section{Bidang Fourier dan penapisan ruang}

\begin{proposition}[Lensa sebagai penganalisis Fourier]\label{prop:b2:diffraction:fourier}
\index{bidang Fourier}\index{frekuensi ruang}\index{penapisan ruang}\index{teori Abbe}
Sebuah transparansi bertransmitans $t(x, y)$ pada bidang fokus depan sebuah
lensa, yang disinari sebuah gelombang bidang, menghasilkan pada bidang fokus belakangnya
amplitudo
\[
\underline s(X, Y) \propto \iint t(x, y)\,\eu^{-\iu2\pi(xX + yY)/\lambda f}\dd x\,\dd y
\]
--- yaitu transformasi Fourier dua dimensinya, dengan tiap titik $(X, Y)$
\emph{\omterm{prop:b2:diffraction:fourier}{bidang Fourier}} memungut cahaya yang terdifraksi pada sudut
$(X/f, Y/f)$, yakni \emph{\omterm{prop:b2:diffraction:fourier}{frekuensi ruang}} $(X/\lambda f, Y/\lambda f)$
objeknya: rincian halus (frekuensi tinggi) jauh dari sumbunya, yang kasar
dekat sumbunya. Sebuah lensa kedua membentuk ulang citranya dari bidang itu; apa pun
yang dihalangi atau diubah di sana tertapis keluar dari citranya
(\emph{\omterm{prop:b2:diffraction:fourier}{penapisan ruang}}, \omterm{prop:b2:diffraction:fourier}{teori Abbe} tentang mikroskop): sebuah lubang jarum
pada sumbunya hanya menyimpan frekuensi rendahnya (pemulusan, ``pembersihan'' sebuah
berkas laser); sebuah penghadang pada sumbunya menyingkirkan latar belakang yang seragam lalu
menyimpan tepinya (medan gelap, strioskopi); sebuah \omterm{prop:b2:plane-waves-polarization:plates}{keping seperempat gelombang} pada
sumbunya mengubah objek fase, yang jika tidak tak tampak, menjadi kontras intensitas
(kontras fase, Zernike).
\end{proposition}

\begin{proof}[Pembenaran]
Dalam integral Fraunhofernya, $\vect{OP}\cdot\vect u = x\alpha + y\beta$ dengan
arahnya dipetakan ke $(X, Y) = (f\alpha, f\beta)$; transmitansnya membobot
tiap sumber sekundernya. (Transformasi Fourier itu sendiri dipelajari dalam
jilid matematika Tahun 3; di sini ia hanya nama integral ini.)
Bagi objek berkala berperiode $d$ transformasinya adalah himpunan orde
kisinya di $X = p\lambda f/d$: citranya dapat menunjukkan periodenya hanya jika sekurangnya
orde $\pm1$-nya lolos menembus lensanya --- yaitu syarat Abbe, yang merupakan
batas resolusi mikroskopnya lagi.
\end{proof}

\begin{example}[Membersihkan sebuah berkas, melihat yang tak tampak]\label{ex:b2:diffraction:filtering}
Sebuah berkas laser mengemban debu dan riak: fokuskanlah ia menembus sebuah lubang jarum
beberapa mikrometer --- seukuran \omterm{prop:b2:diffraction:airy}{cakram Airy} pusat berkas
Gauss yang bersih --- dan segala yang lain, yang terletak di luarnya, terhalangi;
berkas yang dikolimasi ulang itu mulus: yaitu sebuah \emph{tapis ruang}. Sebuah sel
hidup itu bening; ia hanya mengubah fase cahayanya. Menghalangi
bintik sesumbunya membuat tepinya terang di atas hitam; keping Zernike
menggeser cahaya yang tak terdifraksi sebesar seperempat gelombang sehingga ia berinterferensi
dengan cahaya terdifraksinya dalam intensitas: bagian dalam selnya muncul,
dan biologi mendapat mikroskopnya (hadiah Nobel 1953).
\end{example}

\begin{method}[Taksiran difraksi]\label{met:b2:diffraction:method}
(1) Sebaran sudutnya $\sim\lambda/b$ bagi apertur berukuran $b$ ($1.22\lambda/D$ bagi
sebuah lingkaran); ukuran polanya pada jarak $L$ atau jarak fokus $f$ adalah
$\lambda L/b$, $\lambda f/b$. (2) Periksalah syarat Fraunhofernya $b^2/\lambda L \ll 1$ atau
pakailah sebuah lensa. (3) Gabungkanlah: selubung \omterm{def:b2:diffraction:huygens}{difraksi} $\times$ rumbai
interferensi. (4) Resolusinya: dua titik menuntut $\Delta\theta > 1.22\lambda/D$; sebuah objek
berkala menuntut orde pertamanya masuk ke pupilnya. (5) Gambaran \omterm{prop:b2:diffraction:fourier}{bidang
Fouriernya}: apa yang harus dihalangi untuk menyingkirkan rincian yang mana.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:diffraction:1}
Sebuah celah \qty{0.10}{mm} yang disinari sebuah laser He--Ne (\qty{633}{nm}); layar di
\qty{2.0}{m}. Tentukanlah lebar maksimum pusatnya; kedudukan kedua \omterm{thm:b2:gratings:nwaves}{maksimum
sekunder} pertamanya; \omterm{rem:b2:diffraction:fresnelnumber}{bilangan Fresnelnya} --- cukup jauhkah layarnya? Celah
yang sama dengan sebuah lensa ber-$f = \qty{50}{cm}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffraction:2}
Tentukanlah daya urai (Rayleigh): mata (\qty{3}{mm}, \qty{550}{nm}); sebuah
teleskop amatir \qty{10}{cm}; cermin Hubble \qty{2.4}{m}; sebuah piringan radio
\qty{64}{m} pada \qty{6}{cm}; sebuah lensa kamera \qty{50}{mm} pada $f/8$ (pupilnya \qty{6}{mm}):
berapa ukuran \omterm{prop:b2:diffraction:airy}{cakram Airy} pada pengindranya, dibandingkan dengan piksel \qty{4}{\micro m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffraction:3}
Dua celah selebar $b = \qty{0.05}{mm}$, yang pusatnya terpisah $a = \qty{0.25}{mm}$, $\lambda =
\qty{500}{nm}$, lensa $f = \qty{1}{m}$: tentukanlah jarak rumbainya; lebar selubung
pusatnya; jumlah rumbai di dalamnya; orde mana yang hilang?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:diffraction:4}
Sebuah berkas laser berdiameter \qty{2}{mm} pada \qty{633}{nm} merambat ke Bulan
(\qty{3.8e8}{m}): tentukanlah sudut \omterm{def:b2:diffraction:huygens}{difraksinya} $\sim1.22\lambda/D$, diameter bintiknya;
dengan sebuah teleskop \qty{3.5}{m} sebagai pemancarnya; mengapa pantulan pengukuran jarak bulan begitu
redup (retroreflektornya, \qty{1}{m}, mengirimkan cahayanya kembali dengan \omterm{def:b2:diffraction:huygens}{difraksinya}
sendiri)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffraction:5}
\emph{Celahnya secara rinci.} (a) Kerjakanlah integral Fraunhofer bagi
celahnya lalu dapatkanlah $\operatorname{sinc}^2$. (b) Tunjukkanlah bahwa \omterm{thm:b2:gratings:nwaves}{maksimum sekundernya}
berada di $\tan x = x$ ($x = \pi b\sin\theta/\lambda$), yang pertama dekat $x = 1.43\pi$, lalu
hitunglah intensitas nisbinya. (c) Berapa bagian energi dalam
cuping pusatnya (terimalah $\int_{-\pi}^\pi\operatorname{sinc}^2 = 0.903\pi$ dan
$\int_{-\infty}^\infty\operatorname{sinc}^2 = \pi$). (d) Celahnya disinari pada datang
$\theta_0$: tunjukkanlah bahwa polanya sama, yang berpusat pada arah $\theta_0$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffraction:6}
\emph{Mikroskop.} Sebuah objektif berapertur numerik $n\sin\alpha = 1.4$
(rendam minyak) pada \qty{500}{nm}: tentukanlah jarak terurai terkecilnya $0.61\lambda/n\sin
\alpha$; dengan cahaya biru (\qty{400}{nm}); dalam ultraungu (\qty{250}{nm}); mengapa
sebuah lensa mata $1000\times$ tak menolong (batas mata sendiri pada \qty{25}{cm} adalah
\qty{0.1}{mm})? Apa yang diubah mikroskop elektron ($\lambda = \qty{4}{pm}$ pada
\qty{100}{keV})?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffraction:7}
\emph{Segi empat dan lingkaran.} (a) Sebuah apertur segi empat $b \times h$:
tunjukkanlah bahwa polanya adalah $\operatorname{sinc}^2(\pi b\sin\theta_x/\lambda)\operatorname{sinc}^2(\pi h
\sin\theta_y/\lambda)$: sebuah salib, yang lebih lebar sepanjang sisi yang lebih sempit. (b) Sebuah bujur sangkar
\qty{0.2}{mm} dengan $f = \qty{1}{m}$: berapa ukuran bujur sangkar pusatnya. (c) Bagi sebuah
lingkaran selebar sama nol pertamanya berada di $1.22\lambda/D$ alih-alih $\lambda/b$:
mengapa lebih jauh ke luar (pikirkanlah cahayanya yang terpusat dekat pusat sebuah
cakram)? (d) Cermin sekunder sebuah teleskop ditahan empat bilah: apa yang
ditambahkannya pada citra sebuah bintang terang?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffraction:8}
\emph{Abbe.} Sebuah kisi berperiode $d = \qty{2}{\micro m}$ diamati dengan sebuah
objektif mikroskop bersudut apertur $\alpha$ dalam udara, $\lambda = \qty{550}{nm}$.
(a) Tentukanlah sudut orde $\pm1$, $\pm2$-nya. (b) $\sin\alpha$ minimum agar citranya
menunjukkan periodenya (orde $\pm1$ harus masuk). (c) Dengan $\sin\alpha = 0.4$: apa
yang ditunjukkan citranya --- sebuah kisi yang setiakah ia? (d) Penyinaran
miring meloloskan orde nolnya dan \emph{satu} orde pertamanya:
berapa $\sin\alpha$ minimumnya lalu, dan berapa perolehannya.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:diffraction:9}
\emph{Tapis ruang.} Sebuah berkas laser \qty{2}{mm} ($\lambda = \qty{633}{nm}$)
difokuskan sebuah lensa ber-$f = \qty{20}{mm}$ menembus sebuah lubang jarum. (a) Tentukanlah diameter
\omterm{prop:b2:diffraction:airy}{cakram Airy} pusatnya di fokusnya; diameter lubang jarum untuk meloloskannya
(ambillah $1.5\times$). (b) Sebutir debu \qty{50}{\micro m} pada berkasnya
mendifraksikan cahaya pada sudut berapa, dan di mana cahaya itu mendarat pada bidang
fokusnya --- menembus lubang jarumnya atau tidak? (c) Berapa daya yang hilang jika berkasnya
Gauss dan lubang jarumnya meloloskan $99\%$-nya; mengapa berkas yang muncul
mulus? (d) Mengapa lubang jarumnya harus terpusat sampai beberapa
mikrometer?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffraction:10}
\emph{Babinet.} Dua tabir komplementer (sebuah celah, dan sebuah jalur
legap selebar sama) disinari sebuah gelombang bidang. (a) Tunjukkanlah bahwa, jauh
dari arah berkas datangnya, pola Fraunhofernya
serupa (jumlah kedua amplitudonya adalah amplitudo gelombang yang
tak terhalangi, yang nol di luar sumbunya). (b) Tentukanlah pola sehelai rambut
berdiameter \qty{80}{\micro m} pada \qty{2}{m} dengan sebuah laser: jarak rumbainya, dan
diameter rambutnya dari situ --- sebuah pengukuran baku. (c) Apa yang terlihat
pada sumbunya sendiri? (d) Mengapa tetes kecil dalam awan tipis menghasilkan sebuah
korona cincin berwarna di sekeliling Bulan, dan apa yang diberikan jari-jari
cincinnya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffraction:11}
\emph{Medan gelap dan kontras fase.} Sebuah objek bening mempunyai
$t(x) = \eu^{\iu\varphi(x)} \approx 1 + \iu\varphi(x)$ dengan $|\varphi| \ll 1$. (a) Tentukanlah intensitas
citra biasanya: tunjukkanlah bahwa ia seragam sampai orde pertama --- tak tampak. (b)
Dalam \omterm{prop:b2:diffraction:fourier}{bidang Fouriernya} ``1''-nya adalah bintik sesumbunya dan $\iu\varphi$ cahaya
terdifraksi di sekitarnya: jika sebuah penghadang menghalangi bintik sesumbunya, intensitas
citranya $\propto\varphi^2$ --- medan gelap; kelemahannya. (c) Jika sebaliknya
bintik sesumbunya ditunda sebesar $\pi/2$ (dikalikan $\iu$): citranya $\propto|1 +
\varphi|^2 \approx 1 + 2\varphi$ --- kontras fase, yang linear dalam $\varphi$. (d) Mengapa
keping Zernike sering juga menyerap (ia melemahkan bintik sesumbunya)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:diffraction:12}
\emph{Resolusi citra celah sebuah spektrograf dan kisinya.} Sebuah
kisi selebar $W$ adalah apertur spektrografnya: tunjukkanlah bahwa
pola \omterm{def:b2:diffraction:huygens}{difraksi} apertur itu, $\operatorname{sinc}^2(\pi W\sin\theta/\lambda)$
(bagi berkas yang pergi pada $\theta \approx 0$), mempunyai lebar setengah sudut
$\lambda/W$, dan bahwa yang diubah menjadi panjang gelombang lewat dispersinya $p/a\cos
\theta$ ini adalah $\Delta\lambda = \lambda/pN$ --- yaitu daya urai
\cref{ch:b2:gratings}, yang dilihat sebagai \omterm{def:b2:diffraction:huygens}{difraksi}. (b) Sebuah teleskop
berdiameter $D$ dan sebuah kisi selebar $W$: kisinya harus lebih lebar daripada
$D\,f_{\text{coll}}/f_{\text{tel}}$: mengapa? (c) Jelaskanlah dalam satu kalimat mengapa
setiap ``daya urai'' dalam optika adalah $\sim$(ukuran aperturnya)/$\lambda$.
(d) Apa yang ditambahkan celahnya, dan kapan ia menguasai?
\end{exercise}

\section{Soal: Apa yang dapat diurai}

\begin{problem}[{Soal akhir pekan --- batas difraksi pada empat
skala: mata, sebuah teleskop, sebuah mikroskop, dan berkas laser yang
harus dibersihkan sebelum dipakai}]
\label{pb:b2:diffraction:1}

\medskip
\noindent\textbf{Bagian I --- Mata.}
Pupilnya $D = \qty{3}{mm}$ pada siang hari, \qty{7}{mm} pada malam hari; $\lambda = \qty{550}{nm}$;
sel kerucut retinanya terpisah \qty{2}{\micro m} di fovea, jarak fokus
matanya \qty{17}{mm}.
\begin{enumerate}
  \item Jari-jari sudut \omterm{prop:b2:diffraction:airy}{cakram Airynya} pada siang hari; jari-jari lurusnya pada
        retinanya; bandingkanlah dengan jarak sel kerucutnya --- cocokkah retinanya
        dengan optikanya?
  \item Pemisahan terkecil dua titik yang terurai pada \qty{25}{cm}; pada
        \qty{10}{m}; dapatkah Anda membaca teks \qty{1}{mm} sepanjang lengan?
  \item Pada malam hari pupilnya membuka menjadi \qty{7}{mm}: berapa batas \omterm{def:b2:diffraction:huygens}{difraksinya} lalu;
        mengapa penglihatannya toh memburuk (aberasi
        lensanya pada apertur penuh, sel batang yang lebih kasar daripada sel kerucut)?
  \item Dua lampu depan mobil yang terpisah \qty{1.5}{m}: pada jarak berapa keduanya melebur
        menjadi satu lampu.
  \item Sebuah lubang \qty{1}{mm} yang ditahan di depan mata: apa yang dilakukan \omterm{def:b2:diffraction:huygens}{difraksi},
        dan mengapa sebuah lubang jarum toh menolong mata yang rabun jauh
        (kedalaman medan)?
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Teleskopnya.}
Sebuah teleskop \qty{1}{m}, $f = \qty{10}{m}$, dengan sebuah kamera berpiksel \qty{10}{\micro m};
$\lambda = \qty{550}{nm}$.
\begin{enumerate}[resume]
  \item Tentukanlah batas \omterm{def:b2:diffraction:huygens}{difraksinya} dalam sekon busur; jari-jari Airynya pada
        detektornya; piksel tiap \omterm{prop:b2:diffraction:airy}{cakram Airynya}.
  \item Atmosfernya mengaburkan citranya menjadi \qty{1}{''}: berapa banyak resolusi
        teleskopnya yang hilang? Diameter teleskop yang
        batas \omterm{def:b2:diffraction:huygens}{difraksinya} sama dengan kaburan atmosfernya.
  \item Optika adaptif mengoreksi atmosfernya pada \qty{2.2}{\micro m}:
        berapa batas \omterm{def:b2:diffraction:huygens}{difraksinya} di sana; mengapa lebih mudah dalam inframerah?
  \item Sebuah bintang ganda yang komponennya terpisah \qty{0.15}{''}: teruraikah ia dengan
        optika adaptif pada \qty{2.2}{\micro m}? Pada \qty{550}{nm} dari antariksa?
  \item Bulan berjarak \qty{3.8e5}{km}: berapa kawah terkecil yang dapat diurai
        teleskop ini (\omterm{def:b2:diffraction:huygens}{difraksi} saja); dan jejak sebuah pendarat
        bulan (\qty{4}{m})?
  \item Daya kumpul cahaya teleskopnya relatif terhadap
        mata malam (\qty{7}{mm}); apa yang dibelinya, dan apa yang tidak.
  \item Cahaya sebuah bintang menembus keempat bilah yang menahan
        cermin sekundernya: lukiskanlah citranya.
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Mikroskopnya.}
Objektif berapertur numerik $\mathrm{NA} = n\sin\alpha = 0.9$ (kering) atau $1.4$
(minyak), $\lambda = \qty{550}{nm}$; jarak yang terurai adalah $0.61\lambda/\mathrm{NA}$.
\begin{enumerate}[resume]
  \item Tentukanlah jarak terurai dengan tiap objektifnya; jumlah pasangan garis
        tiap milimeter.
  \item Sebuah bakteri \qty{1}{\micro m} berstruktur dalam pada
        \qty{0.2}{\micro m}: apa yang terlihat dengan masing-masingnya? Sebuah virus \qty{0.1}{\micro m}?
  \item Abbe: sebuah struktur berkala berperiode $d$ hanya tercitrakan jika
        orde pertamanya di $\sin\theta = \lambda/d$ masuk ke objektifnya: berapa $d$ minimum
        bagi $\mathrm{NA} = 1.4$; bandingkanlah dengan $0.61\lambda/\mathrm{NA}$.
  \item Mengapa minyak antara kaca objek dan objektifnya memperbaiki
        resolusinya?
  \item Ultraungu pada \qty{250}{nm} dan sebuah mikroskop elektron pada \qty{4}{pm}:
        tentukanlah jarak teruraikannya dalam rumus yang sama (bagi elektron
        NA praktisnya \qty{0.01}{}): bandingkanlah.
  \item Sebuah sel itu bening: dalam medan terang citranya kosong.
        Jelaskanlah apa yang dilakukan keping kontras fase pada
        \omterm{prop:b2:diffraction:fourier}{bidang Fourier} objektifnya, dalam dua kalimat.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Berkasnya.}
Sebuah laser pada \qty{532}{nm}, berkasnya berdiameter \qty{1.5}{mm}, hendak dilebarkan menjadi
\qty{30}{mm} lalu dibersihkan.
\begin{enumerate}[resume]
  \item Tentukanlah setengah sudut \omterm{def:b2:diffraction:huygens}{difraksi} berkas mentahnya ($\sim1.22\lambda/D$);
        diameternya sesudah \qty{100}{m} tanpa optika.
  \item Sebuah lensa ber-$f_1 = \qty{10}{mm}$ memfokuskannya: berapa diameter bintik fokusnya;
        diameter lubang jarum yang dipilih ($1.5\times$ bintiknya).
  \item Sebuah lensa kedua ber-$f_2$ mengolimasi ulang pada \qty{30}{mm}: berapa $f_2$; berapa
        sudut \omterm{def:b2:diffraction:huygens}{difraksi} barunya; diameternya sesudah \qty{1}{km}.
  \item Debu \qty{100}{\micro m} pada lensa pertamanya mendifraksi pada sudut berapa;
        di mana cahaya itu jatuh pada bidang lubang jarumnya, dan apa
        yang terjadi padanya?
  \item Lubang jarumnya meloloskan $99\%$ sebuah berkas Gauss: berapa daya yang hilang, dan
        ke mana ia pergi.
  \item Mengapa berkas yang telah dilebarkan harus \emph{lebih besar} untuk tetap sejajar
        (kaitkanlah $\lambda/D$ dengan jarak yang sepanjangnya berkasnya berlipat dua,
        $\sim D^2/\lambda$)?
  \item Ringkaslah keempat batas soal ini dalam satu tabel: apertur,
        panjang gelombang, $\lambda/D$, apa yang dibatasinya.
\end{enumerate}
\end{problem}
