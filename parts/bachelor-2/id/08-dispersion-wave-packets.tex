\chapter{Dispersi dan Paket Gelombang}\label{ch:b2:dispersion-wave-packets}

Jatuhkanlah sebutir batu ke kolam lalu amatilah cincin riaknya: puncak
baru lahir di tepi dalamnya, berlari keluar menembus cincinnya, lalu mati
di tepi luarnya --- cincinnya sebagai keseluruhan bergerak lebih lambat
daripada puncak di dalamnya. Duduklah di pantai sesudah badai yang jauh
dan alun yang panjang dan lambat tiba sehari sebelum riak yang pendek: lautnya
menyortir gelombang menurut panjang gelombangnya. Kabel telegraf lintas
Atlantik yang pertama, pada 1858, mengubah titik dan garis yang tajam
menjadi noda yang menuntut waktu bermenit-menit untuk dibaca. Dalam ketiga hal itu
kelajuan sebuah gelombang bergantung pada frekuensinya: mediumnya bersifat
\emph{dispersif}. Bab ini memperkenalkan \omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersi},
\omterm{def:b2:dispersion-wave-packets:complex}{bilangan gelombang kompleks} yang mengemban perambatan sekaligus serapan,
\omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup} yang dengannya sebuah paket --- beserta energinya, dan
informasinya --- sungguh-sungguh berjalan, dan saluran yang dilalui telegrafi,
televisi dan internet: \omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial}.

\section{Hubungan dispersi}

\begin{definition}[Hubungan dispersi; kecepatan fase]\label{def:b2:dispersion-wave-packets:relation}
\index{hubungan dispersi}\index{kecepatan fase}\index{medium dispersif}
Medium yang linear, homogen dan tak berubah menurut waktu menerima gelombang
bidang monokromatik $\underline s = A\,\eu^{\iu(\omega t - kx)}$ (notasi
kompleks; sinyal fisisnya adalah bagian nyatanya) asalkan $\omega$ dan $k$
memenuhi \emph{hubungan dispersi} mediumnya, $\mathcal D(\omega, k)
= 0$, yang diselesaikan sebagai $k(\omega)$ atau $\omega(k)$. Untuk $k$ nyata gelombangnya berjalan
tanpa berubah bentuk dengan \emph{kecepatan fase}
\[
v_\varphi = \frac\omega k .
\]
Mediumnya bersifat \emph{tak dispersif} apabila $v_\varphi$ tak bergantung pada
$\omega$ (maka $\omega = ck$ dan setiap sinyal merambat tanpa berubah bentuk --- yaitu
\omterm{thm:b2:waves-on-strings:equation}{persamaan d'Alembert}) dan \emph{dispersif} pada hal lainnya.
\end{definition}

\begin{proof}
Menyisipkan $\eu^{\iu(\omega t - kx)}$ ke dalam persamaan linear berkoefisien
tetap mengubah tiap $\partial_t$ menjadi $\iu\omega$ dan tiap $\partial_x$ menjadi
$-\iu k$, sehingga menyisakan hubungan aljabar. Tali, gelombang bunyi dan
batangnya memberi $\omega^2 = c^2k^2$.
\end{proof}

\begin{example}[Dispersif dan tak dispersif]\label{ex:b2:dispersion-wave-packets:examples}
\index{persamaan Klein--Gordon}\index{gelombang air dalam}
(i) Sebuah tali di atas alas elastik (gaya pemulih $-Ky$ tiap satuan panjang,
atau rantai bandul), $\mu\partial_t^2y = T\partial_x^2y - Ky$: $\omega^2 = \omega_c^2
+ c^2k^2$ dengan $\omega_c = \sqrt{K/\mu}$ --- yaitu hubungan
\emph{Klein--Gordon}, yang sama dengan hubungan sebuah plasma dan sebuah pandu gelombang
(\cref{ch:b2:waves-in-media,ch:b2:guided-waves}): $v_\varphi = c/\sqrt{1 -
\omega_c^2/\omega^2} > c$, dan tak ada $k$ nyata di bawah \emph{ambang} $\omega_c$.
(ii) Gelombang gravitasi air dalam: $\omega^2 = gk$ (diterima tanpa bukti), $v_\varphi =
\sqrt{g/k} = \sqrt{g\lambda/2\pi}$: gelombang panjang lebih cepat --- \qty{12.5}{m/s} bagi
\qty{100}{m}, \qty{40}{m/s} bagi \qty{1}{km}. (iii) \omterm{prop:b2:waves-on-strings:chain}{Rantai atomnya}, $\omega =
2\omega_0|\sin(ka/2)|$ (\cref{ch:b2:waves-on-strings}). (iv) Cahaya dalam kaca,
$k = n(\omega)\omega/c$, itulah sebabnya sebuah prisma menguraikan spektrum.
\end{example}

\begin{definition}[Bilangan gelombang kompleks: perambatan dan pelemahan]\label{def:b2:dispersion-wave-packets:complex}
\index{bilangan gelombang kompleks}\index{pelemahan}\index{gelombang evanesen}\index{kedalaman kulit}
Ketika \omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersinya} memberi, bagi $\omega$ nyata, sebuah
$\underline k = k' - \iu k''$ yang kompleks, gelombangnya adalah
\[
\underline s = A\,\eu^{-k''x}\,\eu^{\iu(\omega t - k'x)} :
\]
ia merambat dengan $v_\varphi = \omega/k'$ dan amplitudonya meluruh sebagai
$\eu^{-x/\delta}$ dengan \emph{panjang pelemahan} $\delta = 1/k''$ (ia harus
meluruh searah rambatnya: $k'$ dan $k''$ bertanda sama).
Jika $\underline k$ murni khayal, $\underline k = -\iu\kappa$, gelombangnya bersifat
\emph{evanesen}: $A\,\eu^{-\kappa x}\eu^{\iu\omega t}$, sebuah ayunan berdiri
yang amplitudonya mati sepanjang jarak $1/\kappa$ tanpa merambatkan
apa pun --- yaitu hal medium Klein--Gordon di bawah ambangnya,
$\kappa = \sqrt{\omega_c^2 - \omega^2}/c$, dan hal sebuah logam pada frekuensi optik.
\end{definition}

\begin{example}[Tali teredam]\label{ex:b2:dispersion-wave-packets:damped}
Sebuah tali dalam fluida kental, $\mu\partial_t^2y + \alpha\partial_ty = T\partial_x^2y$:
$\underline k^2 = (\omega^2 - \iu\alpha\omega/\mu)/c^2$. Untuk peredaman lemah ($\alpha \ll \mu
\omega$), $\underline k \approx \dfrac\omega c\Bigl(1 - \iu\dfrac{\alpha}{2\mu\omega}\Bigr)$: gelombangnya
menjaga kelajuannya dan kehilangan amplitudo sepanjang $\delta = 2\mu c/\alpha = 2Z/\alpha$,
tak bergantung pada frekuensi --- sebuah rugi \emph{pelesap} tanpa dispersi,
\qty{8.7}{dB} tiap panjang $\delta$.
\end{example}

\section{Paket gelombang dan kecepatan grup}

\begin{proposition}[Layangan; kecepatan grup]\label{prop:b2:dispersion-wave-packets:group}
\index{kecepatan grup}\index{paket gelombang}\index{layangan}
Dua gelombang beramplitudo sama dan berfrekuensi bertetangga $\omega \pm
\delta\omega$, berbilangan gelombang $k \pm \delta k$, tertumpang menjadi
\[
s = 2A\cos(\delta\omega\,t - \delta k\,x)\cos(\omega t - kx) :
\]
sebuah pembawa pada $(\omega, k)$ yang bergerak dengan $v_\varphi = \omega/k$, yang termodulasi selubung
lambat yang bergerak dengan $\delta\omega/\delta k$. Sebuah \emph{\omterm{prop:b2:dispersion-wave-packets:group}{paket gelombang}} ---
tumpangan gelombang monokromatik yang bilangan gelombangnya berada dalam pita sempit
di sekitar $k_0$ --- berjalan, sebagai keseluruhan, dengan \emph{\omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup}}
\[
v_g = \frac{\dd\omega}{\dd k}\Bigr|_{k_0} ,
\]
yaitu kecepatan energinya dan kecepatan informasi yang diembannya.
Dalam medium tak dispersif $v_g = v_\varphi = c$; pada hal lainnya paketnya
berubah bentuk selama berjalan (ia \emph{menyebar}), makin cepat makin besar
$\dd^2\omega/\dd k^2$ dan makin sempit pelengkap spektrumnya, yaitu
lebar ruangnya.
\end{proposition}

\begin{proof}
Rumus jumlah ke hasil kali memberi layangannya. Untuk sebuah paket, tuliskanlah $s =
\int a(k)\eu^{\iu(\omega(k)t - kx)}\dd k$ (sebuah tumpangan Fourier ---
integral Fourier dipelajari dalam jilid matematika Tahun ke-3; di sini hanya
tafsirannya sebagai jumlah sinusoida yang sinambung yang dipakai), dengan
$a(k)$ yang memuncak di $k_0$. Uraikanlah $\omega(k) \approx \omega_0 + v_g(k - k_0)$: maka
$s \approx \eu^{\iu(\omega_0t - k_0x)}\int a(k)\eu^{\iu(k - k_0)(v_gt - x)}\dd k$, yaitu
pembawa dikali selubung yang hanya merupakan fungsi $x - v_gt$ ---
selubungnya bergerak dengan $v_g$. Suku berikutnya, $\tfrac12\omega''(k_0)(k - k_0)^2t$,
mendefasekan komponennya menurut waktu lalu menyebarkan selubungnya. Energi
sebuah paket terletak di tempat selubungnya berada; begitu pula setiap sinyal.
\end{proof}

\begin{example}[Air dalam: puncaknya mendahului grupnya]\label{ex:b2:dispersion-wave-packets:deepwater}
$\omega = \sqrt{gk}$: $v_g = \tfrac12\sqrt{g/k} = \tfrac12v_\varphi$. Sebuah grup
alun bergerak dengan separuh kelajuan puncaknya: sambil mengamati rentetan gelombang,
Anda melihat puncak muncul di belakang grupnya, berjalan menembusnya lalu
lenyap di depannya --- persis cincin di kolam itu. Sebuah badai \qty{3000}{km}
jauhnya mengirimkan alun \qty{15}{s}-nya ($\lambda = gT^2/2\pi = \qty{350}{m}$, $v_g =
gT/4\pi = \qty{11.7}{m/s}$) dalam tiga hari, gelombang \qty{8}{s}-nya ($v_g =
\qty{6.2}{m/s}$) dalam lima setengah hari: dari waktu tibanya
periode yang berbeda, ahli oseanografi menentukan letak badainya.
\end{example}

\begin{example}[Klein--Gordon: lebih cepat daripada cahaya, dan tidak]\label{ex:b2:dispersion-wave-packets:superluminal}
Untuk $\omega^2 = \omega_c^2 + c^2k^2$: $v_\varphi = \omega/k$ dan $v_g = c^2k/\omega$, jadi
$v_\varphi v_g = c^2$: \omterm{def:b2:dispersion-wave-packets:relation}{kecepatan fasenya} melampaui $c$ (dalam ionosfer, puncak
gelombang \qty{10}{MHz} bergerak lebih cepat daripada cahaya) tetapi \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan
grupnya}, yang mengemban sinyalnya, tinggal di bawahnya. \omterm{def:b2:dispersion-wave-packets:relation}{Kecepatan fase}
tidak mengangkut energi maupun informasi --- puncaknya seperti
titik cahaya mercusuar yang menyapu awan yang jauh.
\end{example}

\begin{remark}[Penyebaran]\label{rem:b2:dispersion-wave-packets:spreading}
Sebuah paket berlebar ruang $\Delta x$ memuat bilangan gelombang sepanjang $\Delta k \sim
1/\Delta x$ (kebalikan Fourier, diterima tanpa bukti); \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup}
komponennya berbeda sebesar $\Delta v_g \approx |\omega''|\Delta k$, jadi sesudah waktu $t$ ia
telah menyebar sebesar $|\omega''|\Delta k\,t$: lebarnya berlipat dua sesudah $t_{\text{seb}}
\sim \Delta x^2/|\omega''|$. Sebuah paket berisi sepuluh \omterm{ex:b2:dispersion-wave-packets:examples}{gelombang air dalam} \qty{100}{m}
($\Delta x \approx \qty{1}{km}$, $\omega'' = -\tfrac14\sqrt{g/k^3} = \qty{-50}{m^2/s}$)
berlipat dua dalam sekitar enam jam; sebuah pulsa cahaya \qty{1}{ns} dalam
serat optik menyebar berpuluh pikosekon tiap kilometer --- yaitu
batas laju bit sambungan panjang. \omterm{prop:b2:dispersion-wave-packets:group}{Paket gelombang} kuantum
\cref{ch:b2:schrodinger-wave-functions} menyebar karena sebab yang sama.
\end{remark}

\section{Kabel koaksial}

\begin{proposition}[Persamaan telegrafis; saluran tanpa rugi]\label{prop:b2:dispersion-wave-packets:coax}
\index{kabel koaksial}\index{persamaan telegrafis}
\index{impedansi ciri}\index{saluran transmisi}
Sebuah \omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial} (atau saluran dua penghantar mana pun) mempunyai, tiap satuan panjang,
induktansi $\Lambda$ dan kapasitansi $\Gamma$. Tegangan $v(x, t)$ antara
kedua penghantarnya dan arus $i(x, t)$ dalam penghantar dalamnya menuruti
\emph{\omterm{prop:b2:dispersion-wave-packets:coax}{persamaan telegrafis}}
\[
\frac{\partial v}{\partial x} = -\Lambda\frac{\partial i}{\partial t} , \qquad
\frac{\partial i}{\partial x} = -\Gamma\frac{\partial v}{\partial t} ,
\]
sehingga memberi \omterm{thm:b2:waves-on-strings:equation}{persamaan d'Alembert} bagi $v$ dan $i$ dengan
\[
c = \frac1{\sqrt{\Lambda\Gamma}} , \qquad
v = Z_c\,i \ \text{bagi gelombang menuju } +x, \quad Z_c = \sqrt{\frac\Lambda\Gamma} ,
\]
dengan $Z_c$ yaitu \emph{\omterm{prop:b2:dispersion-wave-packets:coax}{impedansi ciri}} salurannya. Untuk
\omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial} berjari-jari $a < b$ yang terisi dielektrik berpermitivitas
nisbi $\varepsilon_r$, $\Lambda = \dfrac{\mu_0}{2\pi}\ln\dfrac ba$, $\Gamma =
\dfrac{2\pi\varepsilon_0\varepsilon_r}{\ln(b/a)}$, jadi $c = c_0/\sqrt{\varepsilon_r}$ (sekitar
\qty{2e8}{m/s}, dua pertiga kelajuan cahaya) dan $Z_c = \dfrac{60\,\Omega}
{\sqrt{\varepsilon_r}}\ln\dfrac ba$: \qty{50}{\ohm} atau \qty{75}{\ohm} bagi kabel yang lazim.
\end{proposition}

\begin{proof}
Modelkanlah iris $[x, x + \dd x]$ sebagai induktansi seri $\Lambda\dd x$ dan
kapasitansi paralel $\Gamma\dd x$: jatuh tegangan sepanjang irisnya adalah
$\Lambda\dd x\,\partial_ti$, dan arus yang hilang ke kapasitansinya adalah $\Gamma\dd x
\,\partial_tv$. Turunkanlah bersilang. Untuk $v = f(x - ct)$, persamaan pertamanya
memberi $i = f(x - ct)/\Lambda c = v/\sqrt{\Lambda/\Gamma}$. Nilai $\Lambda$ dan
$\Gamma$ adalah nilai kapasitor tabung dan induktor koaksial
yang dihitung dalam jilid Tahun ke-1.
\end{proof}

\begin{proposition}[Saluran berugi; syarat tanpa distorsi]\label{prop:b2:dispersion-wave-packets:lossy}
\index{syarat Heaviside}\index{hukum kuadrat Kelvin}
Dengan hambatan seri $R$ dan konduktansi paralel $G$ tiap satuan
panjang, persamaannya menjadi $\partial_xv = -\Lambda\partial_ti - Ri$, $\partial_xi =
-\Gamma\partial_tv - Gv$, dan \omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersinya} $\underline k^2 = (\Lambda\omega -
\iu R)(\Gamma\omega - \iu G)$: salurannya dispersif dan melemahkan. Dua
limitnya: (i) $R/\Lambda = G/\Gamma$ (\omterm{prop:b2:dispersion-wave-packets:lossy}{syarat Heaviside}, yang dicapai dengan menambahkan
induktansi): $\underline k = \omega\sqrt{\Lambda\Gamma} - \iu\sqrt{RG}$, setiap frekuensi
berjalan dengan kelajuan yang sama dan \omterm{def:b2:dispersion-wave-packets:complex}{pelemahan} yang sama --- salurannya
\emph{melemahkan} tetapi tidak \emph{mencacatkan}. (ii) $\Lambda \approx 0$, $G \approx 0$
(kabel bawah laut yang pertama): $\partial_x^2v = R\Gamma\,\partial_tv$, sebuah
persamaan \emph{difusi} --- sebuah pulsa tidak merambat melainkan memudar,
mencapai jarak $L$ sesudah waktu $\sim R\Gamma L^2$ (\omterm{prop:b2:dispersion-wave-packets:lossy}{hukum kuadrat
Kelvin}), seratus kali lebih lama bagi kabel yang sepuluh kali lebih panjang.
\end{proposition}

\begin{proof}
Sisipkanlah $\eu^{\iu(\omega t - kx)}$: $-\iu kv = -(\iu\omega\Lambda + R)i$, $-\iu ki = -(\iu
\omega\Gamma + G)v$; kalikanlah. Di bawah \omterm{prop:b2:dispersion-wave-packets:lossy}{syarat Heaviside} hasil kalinya menjadi
kuadrat sempurna. Limit difusinya membuang suku $\iu\omega\Lambda$ dan $G$.
\end{proof}

\begin{example}[Tiga kabel]\label{ex:b2:dispersion-wave-packets:cables}
Kabel lintas Atlantik 1858: sebatang kawat tembaga dalam getah perca,
$R \approx \qty{3}{\ohm/km}$, $\Gamma \approx \qty{0.3}{\micro F/km}$, $L = \qty{3000}{km}$:
$R\Gamma = \qty{9e-10}{s/m^2}$ dan $R\Gamma L^2 \approx \qty{8}{s}$ tiap titik: satu kata tiap
menit, dan upaya para operatornya untuk memaksakan sinyalnya
dengan \qty{2}{kV} merusak isolasinya dalam beberapa pekan. Sebuah saluran telepon
yang dibebani kumparan tiap \qty{2}{km} untuk memenuhi \omterm{prop:b2:dispersion-wave-packets:lossy}{syarat Heaviside}
(1900): tutur kata yang jelas sepanjang ratusan kilometer. Sebuah \omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial}
\qty{50}{\ohm} modern: $c = \qty{2e8}{m/s}$, $\Lambda = \qty{0.25}{\micro H/m}$, $\Gamma =
\qty{100}{pF/m}$, dan sebuah \omterm{def:b2:dispersion-wave-packets:complex}{pelemahan}, akibat efek kulit dalam
penghantarnya (\cref{ch:b2:waves-in-media}), yang tumbuh sebagai $\sqrt f$: beberapa
\omterm{def:b2:sound-waves:decibel}{desibel} tiap seratus meter pada \qty{100}{MHz}.
\end{example}

\begin{method}[Membaca sebuah hubungan dispersi]\label{met:b2:dispersion-wave-packets:method}
(1) Tuliskanlah \omterm{thm:b2:waves-on-strings:equation}{persamaan gelombang} mediumnya dalam notasi kompleks lalu
selesaikanlah bagi $\underline k(\omega)$. (2) $k$ nyata: hitunglah $v_\varphi = \omega/k$ dan
$v_g = \dd\omega/\dd k$ (atau $1/(\dd k/\dd\omega)$); bandingkanlah: dispersif atau tidak,
normal ($v_g < v_\varphi$) atau anomali. (3) $k$ kompleks: belahlah menjadi
perambatan ($k'$) dan \omterm{def:b2:dispersion-wave-packets:complex}{pelemahan} ($k''$, dengan panjang $1/k''$); murni
khayal berarti evanesen, tanpa pengangkutan. (4) Sebuah frekuensi ambang
memisahkan pita yang merambat dari pita yang evanesen. (5) Bagi sebuah sinyal,
bernalarlah pada selubungnya dengan $v_g$, jangan pernah pada puncaknya.
\end{method}

\section{Latihan}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:1}
\omterm{ex:b2:dispersion-wave-packets:examples}{Gelombang air dalam}, $\omega^2 = gk$. (a) \omterm{def:b2:dispersion-wave-packets:relation}{Kecepatan fase}, \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup} dan

(c) Mengapa riak berperiode \qty{2}{s} sama sekali tak sampai ke sana?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:2}

dan $c = \qty{3e8}{m/s}$. Untuk $f = \qty{20}{MHz}$: tentukanlah $k$, $v_\varphi$, $v_g$, dan

\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:3}

periode perubahan kenyaringannya. Dua gelombang air berpanjang gelombang
\qty{100}{m} dan \qty{110}{m}: frekuensinya, panjang gelombang dan kelajuan
selubungnya; bandingkanlah dengan $v_g$ pada \qty{105}{m}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:dispersion-wave-packets:4}
Sebuah \omterm{prop:b2:dispersion-wave-packets:coax}{kabel koaksial}: penghantar dalam berdiameter \qty{0.9}{mm}, penghantar luar
\qty{3.0}{mm}, polietilena ($\varepsilon_r = 2.3$). Tentukanlah $\Lambda$, $\Gamma$, kelajuan gelombangnya,
\omterm{prop:b2:dispersion-wave-packets:coax}{impedansi cirinya}, tundanya tiap meter; panjang kabel yang setara
dengan tunda \qty{10}{ns}. Kabel yang sama dengan udara alih-alih polietilena.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:5}
\emph{Tali yang kaku.} Sebuah dawai piano yang sungguhan menuruti $\mu\partial_t^2y = T\partial_x^2y
- EI\,\partial_x^4y$ ($EI$ kekakuan lenturnya). (a) \omterm{def:b2:dispersion-wave-packets:relation}{Hubungan dispersinya},
\omterm{def:b2:dispersion-wave-packets:relation}{kecepatan fase} dan grupnya. (b) Bagi dawai A$_4$
\cref{ch:b2:waves-on-strings} ($T = \qty{685}{N}$, $\mu = \qty{6.1}{g/m}$, $EI =
\qty{9.8e-3}{N.m^2}$), tentukanlah kenaikan nisbi $v_\varphi$ bagi harmonik
kedelapannya ($k = 8\pi/L$, $L = \qty{0.38}{m}$); periksalah terhadap ketakharmonikan
yang ditemukan di sana. (c) Dispersinya normal atau anomali? Seperti apa rupa
petikan yang tajam sesudah beberapa perjalanan pulang pergi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:6}
\emph{Tali teredam.} $\mu\partial_t^2y + \alpha\partial_ty = T\partial_x^2y$. (a) Turunkanlah
\omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersinya}. (b) Untuk $\alpha \ll \mu\omega$, tunjukkanlah $k' \approx \omega/c$ dan
$k'' \approx \alpha/2Z$; \omterm{def:b2:dispersion-wave-packets:complex}{pelemahannya} dalam \omterm{def:b2:sound-waves:decibel}{desibel} tiap meter. (c) Angka: $\mu =
\qty{5}{g/m}$, $T = \qty{50}{N}$, $\alpha = \qty{0.02}{kg.m^{-1}.s^{-1}}$, $f =
\qty{100}{Hz}$: periksalah hampirannya lalu berikanlah $\delta$. (d) Dispersifkah tali
teredamnya? Dalam rezim $\alpha$ yang mana ia akan menjadi dispersif?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:7}
\emph{Riak.} Gelombang air yang pendek diatur \omterm{rem:b2:viscous-flows:capillarity}{tegangan permukaan}: $\omega^2 =
\gamma k^3/\rho$ ($\gamma = \qty{0.072}{N/m}$); hubungan lengkapnya adalah $\omega^2 = gk +
\gamma k^3/\rho$. (a) $v_\varphi$ dan $v_g$ bagi gelombang kapiler murni; mana yang
lebih besar? (b) Tunjukkanlah bahwa \omterm{def:b2:dispersion-wave-packets:relation}{kecepatan fase} hubungan lengkapnya
minimum pada $\lambda_m = 2\pi\sqrt{\gamma/\rho g}$ lalu hitunglah $\lambda_m$ dan $v_{\min}$.
(c) Sebuah perahu yang lebih lambat daripada $v_{\min}$ tak membuat jejak gelombang: mengapa? (d) Tetes hujan di
kolam membuat cincin yang puncaknya tertinggal di belakang cincinnya: dalam rezim yang mana?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:8}
\emph{Penyebaran.} Sebuah paket berlebar awal $\Delta x_0$ melipatduakan lebarnya
sesudah $t_{\text{seb}} \approx \Delta x_0^2/|\omega''(k_0)|$. (a) Sebuah grup \omterm{ex:b2:dispersion-wave-packets:examples}{gelombang air
dalam}, $\lambda = \qty{50}{m}$, $\Delta x_0 = \qty{500}{m}$. (b) Sebuah pulsa cahaya dalam
serat kaca: $\omega'' \approx \qty{-0.16}{m^2/s}$; sebuah pulsa \qty{10}{ps} ($\Delta x_0 =
\qty{2}{mm}$): waktu penyebarannya dan jarak yang ditempuhnya sementara itu; hal yang sama
bagi pulsa \qty{1}{ns}. Mengapa sambungan optik yang panjang memakai pulsa yang lebih panjang atau
serat pengimbang dispersi?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:dispersion-wave-packets:9}
\emph{Kabel bawah laut.} $R = \qty{3}{\ohm/km}$, $\Gamma = \qty{0.3}{\micro F/km}$,
$\Lambda$ dan $G$ dapat diabaikan. (a) Tunjukkanlah bahwa $v$ menuruti persamaan difusi lalu
berikanlah difusivitasnya. (b) \omterm{def:b2:dispersion-wave-packets:relation}{Hubungan dispersinya} $\underline k(\omega)$; \omterm{def:b2:dispersion-wave-packets:relation}{kecepatan
fase} dan panjang \omterm{def:b2:dispersion-wave-packets:complex}{pelemahannya} pada \qty{1}{Hz} dan pada \qty{10}{Hz}. (c)
Waktu bagi sebuah sinyal untuk ``tiba'' sepanjang \qty{3000}{km} ($\sim R\Gamma L^2$); kata tiap
menit. (d) Obat Heaviside: induktansi tiap kilometer yang diperlukan bagi
$R/\Lambda = G/\Gamma$ jika $G = \qty{1e-7}{S/km}$; mengapa sukar ditambahkan?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:10}
\emph{Energi berjalan dengan \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup}.} Tali di atas alas elastik:
$\mu\partial_t^2y = T\partial_x^2y - Ky$, dengan gelombang $y = A\cos(\omega t - kx)$ dan $\omega^2 =
\omega_c^2 + c^2k^2$. (a) Rapat energinya $e = \tfrac12\mu(\partial_ty)^2 + \tfrac12T
(\partial_xy)^2 + \tfrac12Ky^2$ dan fluksnya $\mathcal P = -T\partial_xy\,\partial_ty$: periksalah
$\partial_te + \partial_x\mathcal P = 0$. (b) Rerata waktunya $\langle e\rangle$ dan
$\langle\mathcal P\rangle$ bagi gelombangnya. (c) Tunjukkanlah bahwa $\langle\mathcal P\rangle/
\langle e\rangle = c^2k/\omega = v_g$, bukan $v_\varphi$. (d) Apa yang terjadi pada
fluks energinya di bawah ambangnya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:11}
\emph{Alun dan tsunami.} Gelombang air berpanjang gelombang $\lambda$ pada kedalaman $h$
menuruti $\omega^2 = gk\tanh kh$ (diterima tanpa bukti). (a) Dapatkanlah kembali limit air dalam dan
limit air dangkal ($kh \ll 1$); kelajuan gelombang air dangkal; dispersifkah
keduanya? (b) Sebuah tsunami berpanjang gelombang $\lambda = \qty{200}{km}$ di atas \qty{4}{km} samudra:
limit yang mana, kelajuannya, periodenya; waktu untuk menyeberangi \qty{8000}{km}. (c) Amplitudonya
di lepas pantai \qty{0.5}{m}: energi tiap satuan panjang puncaknya
(terimalah $E = \tfrac12\rho gA^2$ tiap satuan luas) bagi muka sepanjang \qty{1000}{km}; apa yang
terjadi ketika kedalamannya turun menjadi \qty{10}{m}, jika fluks energinya $Ev_g$
kekal (hukum pendangkalan, $A \propto h^{-1/4}$)? (d) Sebuah alun \qty{100}{m}
yang tiba di pantai: pada kedalaman berapa ia mulai merasakan dasarnya,
dan mengapa puncaknya berbelok sejajar garis pantainya?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:dispersion-wave-packets:12}
\emph{Sebuah pulsa pada saluran.} Sebuah kabel \qty{50}{\ohm} sepanjang $\ell = \qty{100}{m}$
($c = \qty{2e8}{m/s}$) disuapi di $x = 0$ oleh sebuah generator berhambatan
dalam \qty{50}{\ohm} dan diakhiri di $x = \ell$ oleh beban $R_L$. (a) Tunjukkanlah
bahwa gelombang yang tiba di ujungnya dipantulkan dengan koefisien
tegangan $r = (R_L - Z_c)/(R_L + Z_c)$ (tuliskanlah $v$ dan $i$ sebagai gelombang datang
ditambah gelombang pantul lalu terapkanlah hukum Ohm pada bebannya). (b) Nilainya bagi
$R_L = 50$, $0$, $\infty$, \qty{150}{\ohm}. (c) Sebuah pulsa \qty{1}{ns} sebesar \qty{1}{V}
dikirimkan: apa yang diperlihatkan osiloskop di generatornya, dan kapan, bagi tiap
bebannya; mengapa \qty{50}{\ohm} generatornya penting. (d) Sebuah kerusakan pada kabel
yang terkubur mengembalikan gema sesudah \qty{1.8}{\micro s}: di manakah ia? (Reflektometri
kawasan waktu.)
\end{exercise}

\section{Soal: Kabel dan laut}

\begin{problem}[{Soal akhir pekan --- dispersi yang bekerja: kabel telegraf
yang menodakan titiknya, alun yang mengumumkan sebuah badai,
tsunami yang menyeberangi samudra, dan paket yang menyebar}]
\label{pb:b2:dispersion-wave-packets:1}

\medskip
\noindent\textbf{Bagian I --- Kabel telegraf.}
Kabel lintas Atlantik 1866: $R = \qty{2.0}{\ohm/km}$, $\Gamma = \qty{0.25}{\micro F/km}$,
$\Lambda = \qty{0.5}{mH/km}$, $G$ dapat diabaikan, panjangnya $L = \qty{3500}{km}$.
\begin{enumerate}
  \item Tuliskanlah persamaan telegrafisnya dengan $R$ lalu turunkanlah
        \omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersinya} $\underline k^2 = \Gamma\omega(\Lambda\omega - \iu R)$.
  \item Pada frekuensi berapa $\Lambda\omega$ sama dengan $R$? Simpulkanlah bahwa untuk
        frekuensi telegraf (beberapa hertz) salurannya berada dalam
        rezim difusif.
  \item Dalam rezim itu, $\underline k = \sqrt{-\iu R\Gamma\omega}$: berikanlah $k'$ dan $k''$
        (ingat $\sqrt{-\iu} = (1 - \iu)/\sqrt2$); \omterm{def:b2:dispersion-wave-packets:relation}{kecepatan fase} dan
        panjang \omterm{def:b2:dispersion-wave-packets:complex}{pelemahannya} pada \qty{1}{Hz}; pada \qty{4}{Hz}.
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Pelemahannya} dalam \omterm{def:b2:sound-waves:decibel}{desibel} sepanjang seluruh kabelnya pada \qty{1}{Hz} dan
        pada \qty{4}{Hz}. Frekuensi mana yang bertahan? Apa akibatnya bagi sebuah
        titik yang tajam?
  \item Taksiran Kelvin atas waktu lintasnya, $R\Gamma L^2$; laju praktis
        kabelnya dalam kata tiap menit jika satu kata terdiri atas sekitar
        $10$ titik dan garis dan titiknya tak boleh bertindihan.
  \item Heaviside mengusulkan menaikkan $\Lambda$: nilai berapa yang membuat $\Lambda\omega = R$
        pada \qty{4}{Hz}, dan berapa kelajuan serta \omterm{def:b2:dispersion-wave-packets:complex}{pelemahannya} nanti
        (ambillah $G = 0$, sehingga syarat $R/\Lambda = G/\Gamma$ tak dapat dipenuhi
        secara tepat --- pakailah hubungan umumnya pada frekuensi itu)?
  \item Sebuah serat optik modern mengemban \qty{1e10}{bit/s}: bandingkanlah dengan
        pertanyaan 5, dalam pangkat sepuluh.
\end{enumerate}

\medskip
\noindent\textbf{Bagian II --- Alun.}
Air dalam: $\omega^2 = gk$.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Kecepatan fase} dan \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grup} sebagai fungsi periode $T$.
  \item Sebuah badai \qty{4000}{km} dari sebuah pantai membangkitkan gelombang berperiode
        dari \qty{6}{s} sampai \qty{18}{s}: waktu tiba tiap ujungnya;
        durasi peristiwa alunnya.
  \item Tunjukkanlah bahwa di pantai berjarak $D$ periode yang tiba menuruti
        $1/T = gt/4\pi D$, dengan $t$ dihitung dari badainya: $1/T$ tumbuh
        linear menurut waktu. Sebuah pelampung mencatat $T = \qty{16}{s}$ pada suatu tengah hari
        dan $T = \qty{12}{s}$ tepat \qty{24}{h} kemudian: jarak
        badainya.
  \item Energi tiap satuan luas sebuah alun beramplitudo $A$ adalah $\tfrac12\rho
        gA^2$; daya yang menyeberangi garis \qty{1}{m} puncaknya, untuk $A =
        \qty{1.5}{m}$, $T = \qty{12}{s}$. (Pakailah \omterm{prop:b2:dispersion-wave-packets:group}{kecepatan grupnya} --- mengapa?)
  \item Sebuah ladang energi gelombang bermuka \qty{1}{km} dengan efisiensi $30\%$:
        daya listriknya; bandingkanlah dengan sebuah turbin angin (\qty{3}{MW}).
  \item Sebuah perahu pada \qty{5}{m/s} di atas air dalam meninggalkan jejak gelombang tak bergerak
        di belakangnya: panjang gelombang manakah yang \omterm{def:b2:dispersion-wave-packets:relation}{kecepatan fasenya} sama dengan
        kelajuan perahunya sehingga ia mengikutinya? Mengapa polanya terseret dalam
        bentuk V, lebih sempit daripada yang disiratkan penyebaran puncaknya sendiri
        (pikirkanlah $v_g = v_\varphi/2$)?
  \item Sambil mengamati satu grup alun dari sebuah tebing, lukiskanlah apa yang dilakukan
        satu puncak selama grupnya melewati sebuah
        pelampung tetap; berapa puncak yang dimuat sebuah grup berdurasi
        \qty{2}{min} pada $T = \qty{12}{s}$, dan berapa ``masa hidup'' puncak
        yang diwakilinya?
\end{enumerate}

\medskip
\noindent\textbf{Bagian III --- Tsunami.}
$\omega^2 = gk\tanh kh$; kedalaman Pasifik $h = \qty{4.0}{km}$.
\begin{enumerate}[resume]
  \item Sebuah gempa mengangkat sepetak dasar laut seluas \qty{100}{km} setinggi
        \qty{2}{m}; gelombangnya mempunyai $\lambda \approx \qty{200}{km}$: periksalah bahwa ia
        gelombang air dangkal lalu berikanlah kelajuannya, periodenya dan waktu untuk
        mencapai pantai \qty{6000}{km} jauhnya.
  \item Dispersifkah ia? Bandingkanlah dengan alun Bagian II: apakah ia
        tiba sebelum atau sesudah alunnya akan tiba? Mana yang tiba sebagai
        pulsa tunggal?
  \item Tingginya di lepas pantai \qty{0.6}{m}: energi tiap meter persegi;
        energi total sebuah muka sepanjang \qty{500}{km} yang tersebar sepanjang satu
        panjang gelombang; bandingkanlah dengan ledakan \qty{1}{Mt} (\qty{4.2e15}{J}).
  \item Hukum pendangkalan: tingginya ketika kedalamannya turun menjadi \qty{20}{m} dekat
        pantai (fluks energi $\tfrac12\rho gA^2v_g$ kekal, $v_g = \sqrt{gh}$);
        kelajuan dan panjang gelombangnya di sana; mengapa gelombangnya lalu menjadi curam lalu
        pecah seperti sebuah dinding.
  \item Sebuah kapal di laut tak menyadari tsunaminya lewat: mengapa?
  \item Alat ukur pasang surut yang terpisah \qty{6000}{km} mencatat gelombangnya: apa yang
        diberikan waktu tempuhnya, dan mengapa kelajuan terukurnya sedikit
        lebih kecil daripada $\sqrt{gh}$ di atas samudra yang kedalamannya berubah-ubah.
\end{enumerate}

\medskip
\noindent\textbf{Bagian IV --- Paket yang menyebar.}
\begin{enumerate}[resume]
  \item Bagi air dalam, hitunglah $\omega''(k) = \dd^2\omega/\dd k^2$ dan nilainya
        bagi $\lambda = \qty{100}{m}$.
  \item Sebuah paket berisi \qty{10}{} gelombang semacam itu ($\Delta x_0 \approx \qty{1}{km}$)
        dilepaskan: waktu penyebarannya $\Delta x_0^2/|\omega''|$; jarak yang ditempuh
        paketnya sementara itu.
  \item Jelaskanlah dengan kata-kata mengapa paket berspektrum lebih sempit
        (rentetan lebih panjang) menyebar lebih lambat, dan mengapa tsunami
        Bagian III hampir tidak menyebar sama sekali.
  \item Taksiran yang sama bagi sebuah elektron berpanjang gelombang \qty{0.1}{nm}
        yang terlokalkan dalam \qty{1}{nm}, dengan $\omega = \hbar k^2/2m$ ($\hbar =
        \qty{1.05e-34}{J.s}$, $m = \qty{9.1e-31}{kg}$): waktu penyebaran dan
        jarak yang ditempuhnya --- perjumpaan pertama dengan \omterm{prop:b2:dispersion-wave-packets:group}{paket gelombang}
        kuantum \cref{ch:b2:schrodinger-wave-functions}.
  \item Rangkumlah dalam sebuah tabel: bagi kabelnya, alunnya, tsunaminya dan
        elektronnya, \omterm{def:b2:dispersion-wave-packets:relation}{hubungan dispersinya}, $v_\varphi$, $v_g$ dan
        apakah mediumnya menyebarkan atau sekadar melemahkan.
\end{enumerate}
\end{problem}
