\chapter{O interferômetro de Michelson}\label{ch:b2:michelson}

Em 1887 Michelson e Morley dividiram um feixe de luz em dois, enviaram
as metades por braços perpendiculares, recombinaram-nas e vigiaram as
franjas à espera do deslocamento que o movimento da Terra através do
éter deveria produzir: nada se moveu, e a física mudou. Em 2015 o mesmo
instrumento --- dois braços, agora com quatro quilômetros e dobrados até
mil --- registrou uma onda gravitacional que mudou seus comprimentos de
um milésimo do diâmetro de um próton. Entre essas duas datas o
\omterm{def:b2:michelson:instrument}{interferômetro de Michelson} mediu o metro em comprimentos de onda,
resolveu linhas espectrais e se tornou a ferramenta padrão da óptica de
precisão. Este capítulo o descreve em suas duas configurações --- a
cunha de ar e a lâmina de faces paralelas --- e mostra como suas franjas
medem comprimentos, índices e espectros.

\section{O instrumento}

\begin{definition}[O interferômetro de Michelson]\label{def:b2:michelson:instrument}
\index{interferômetro de Michelson}\index{divisor de feixe}\index{lâmina compensadora}
Um \emph{divisor de feixe} $Sp$ --- uma lâmina de vidro com uma face semirrefletora
--- a \qty{45}{\degree} divide um feixe incidente em dois: um vai para o
espelho $M_1$, o outro para $M_2$, ambos perpendiculares a seus feixes; após
a reflexão cada um volta a $Sp$, onde metade dele é enviada para o
observador, onde as duas metades interferem. Uma \emph{lâmina compensadora}
$C$, idêntica ao divisor e paralela a ele, é colocada no braço
que de outro modo atravessaria o vidro do divisor uma só vez, de modo que os dois
feixes atravessem a mesma espessura de vidro: a \omterm{thm:b2:two-wave-interference:formula}{diferença de caminho} é então uma
pura diferença de percursos no ar, a mesma para todo comprimento de onda --- o que
as franjas em luz branca exigem. As distâncias dos espelhos a $Sp$ são $d_1$
e $d_2$; um dos espelhos é montado sobre um parafuso micrométrico e sobre parafusos
de inclinação.
\end{definition}

\begin{proposition}[Montagem equivalente]\label{prop:b2:michelson:equivalent}
\index{cunha de ar (Michelson)}\index{lâmina de ar (Michelson)}
Visto do observador, o interferômetro é equivalente ao espelho
$M_2$ e à imagem $M_1'$ de $M_1$ no divisor: uma fina camada de ar
entre duas superfícies refletoras, de espessura $e = |d_1 - d_2|$, cujas
duas reflexões interferem (\omterm{prop:b2:two-wave-interference:film}{divisão da amplitude}):
\begin{itemize}
  \item $M_1' \parallel M_2$: uma \emph{lâmina de ar} de espessura $e$ --- \omterm{prop:b2:michelson:rings}{anéis de
        igual inclinação}, localizados no infinito, $\delta = 2e\cos i$;
  \item $M_1'$ ligeiramente inclinado de $\alpha$ em relação a $M_2$: uma \emph{cunha de ar}
        --- franjas retilíneas de igual espessura, localizadas sobre os
        espelhos (perto da cunha), $\delta = 2\alpha x$ sob incidência normal,
        espaçamento $\lambda/2\alpha$.
\end{itemize}
No \emph{contato} ($e = 0$, $\alpha = 0$) o campo é uniforme e, em luz
branca, é o único lugar onde as cores concordam: a "franja branca".
\end{proposition}

\begin{proof}
Dobrar o braço de $M_1$ sobre o braço de $M_2$ por reflexão em $Sp$ leva os
dois feixes sobre uma mesma reta com dois espelhos separados de $e$; as duas ondas
que voltam são as refletidas pelas duas faces da camada de ar (sem
$\lambda/2$ extra: ambas se refletem em metal, e o divisor as trata
simetricamente, a menos de uma fase constante que desloca todo o padrão).
Os resultados sobre películas finas do \cref{ch:b2:two-wave-interference} aplicam-se então
com $n = 1$.
\end{proof}

\section{Os dois sistemas de franjas}

\begin{proposition}[Os anéis da lâmina de ar]\label{prop:b2:michelson:rings}
\index{anéis de igual inclinação}\index{franjas de Haidinger}
Com $M_1' \parallel M_2$ e uma \omterm{prop:b2:two-wave-interference:spatial}{fonte extensa}, o observador (ou uma lente
de distância focal $f$ e um anteparo em seu plano focal) vê anéis
concêntricos; o anel sob o ângulo $i$ tem a ordem $p = 2e\cos i/\lambda$; anéis
claros onde $p$ é inteiro. O centro tem a ordem mais alta, $p_0 =
2e/\lambda$; perto dele, $\cos i \approx 1 - i^2/2$, de modo que o $k$-ésimo anel claro a partir do
centro (contado desde o primeiro, situado $\varepsilon = p_0 - \lfloor p_0\rfloor$
abaixo da ordem do centro) tem o raio
\[
r_k = f\,i_k = f\sqrt{\frac{(k - 1 + \varepsilon)\lambda}e} :
\]
anéis mais apertados quando $e$ é maior (mais anéis, mais finos), que se espalham quando $e \to 0$
até que, no contato, resta um campo uniforme. Deslocar um espelho de $\lambda/2$ muda
$p_0$ de uma unidade: nasce um anel no centro (ou ele é engolido por ele) a cada
meio comprimento de onda de deslocamento.
\end{proposition}

\begin{proof}
$\delta = 2e\cos i$ para a lâmina de faces paralelas (\cref{ch:b2:two-wave-interference},
$n = 1$, sem $\lambda/2$ extra para duas reflexões metálicas); $2e(1 - i^2/2) =
(p_0 - k + 1 - \varepsilon)\lambda$ com $2e = p_0\lambda$ dá $ei^2 = (k - 1 + \varepsilon)\lambda$.
\end{proof}

\begin{proposition}[As franjas da cunha de ar]\label{prop:b2:michelson:wedge}
\index{franjas de igual espessura (Michelson)}\index{franjas de Fizeau}
Com $M_1'$ inclinado de um pequeno ângulo $\alpha$ e sob incidência quase normal, a
\omterm{thm:b2:two-wave-interference:formula}{diferença de caminho} no ponto dos espelhos à distância $x$ da
linha de contato vale $\delta = 2\alpha x$: franjas retilíneas paralelas a essa linha,
espaçadas de $\lambda/2\alpha$, localizadas sobre os espelhos (elas são vistas focalizando
$M_2$, ou projetadas num anteparo com uma lente que conjuga $M_2$ e o
anteparo). A franja de ordem zero fica sobre a linha de contato, a mesma para
todo comprimento de onda: em luz branca ela é branca (ou preta, conforme
as fases do divisor), ladeada por algumas franjas coloridas --- a
assinatura pela qual se encontra o contato.
\end{proposition}

\begin{proof}
\omterm{prop:b2:two-wave-interference:film}{Franjas de igual espessura} de uma cunha de ar, $e(x) = \alpha x$. Localização:
para uma \omterm{prop:b2:two-wave-interference:spatial}{fonte extensa} os raios que passam por um dado ponto da cunha
se recombinam ali, com o mesmo $\delta$ sob incidência quase normal.
\end{proof}

\section{Medir com o Michelson}

\begin{proposition}[Comprimentos, índices, espectros]\label{prop:b2:michelson:measure}
\index{espectroscopia por transformada de Fourier}\index{espectro canelado}
Contar $N$ franjas que passam pelo centro enquanto um espelho se desloca de $\Delta e$
dá $\Delta e = N\lambda/2$: comprimentos em comprimentos de onda (um parafuso micrométrico
calibrado a \qty{0.1}{\micro m}; o próprio metro foi assim medido). Uma
lâmina transparente de espessura $\ell$ e índice $n$ inserida num dos braços
acrescenta $2(n - 1)\ell$ a $\delta$ e desloca o padrão de $2(n - 1)\ell/\lambda$
franjas: índices (e suas variações com a pressão ou a temperatura,
numa cuba de gás). Com um dubleto de comprimentos de onda $\lambda$ e $\lambda + \Delta\lambda$, os
dois sistemas de franjas saem de passo e voltam a ele à medida que $e$ cresce, o
contraste se anulando a cada
\[
\Delta e = \frac{\lambda^2}{2\Delta\lambda} :
\]
a separação de linhas próximas. Mais geralmente, o contraste em
função de $e$ desenha o espectro da fonte (o contraste
decai ao longo de $e \sim \ell_c/2$): registrar a intensidade no centro enquanto
$e$ varre é a \emph{\omterm{prop:b2:michelson:measure}{espectroscopia por transformada de Fourier}}, o método dos
espectrômetros de infravermelho.
\end{proposition}

\begin{proof}
A ordem no centro é $2e/\lambda$; cada linha tem a sua. Os dois
sistemas voltam a coincidir quando $2e/\lambda - 2e/(\lambda + \Delta\lambda) = 1$, ou seja, $2e\Delta\lambda
/\lambda^2 = 1$; eles estão em oposição no meio do caminho. O enunciado geral é
a fórmula do contraste do \cref{exo:b2:scalar-light-model:10}.
\end{proof}

\begin{example}[O sódio, e o metro]\label{ex:b2:michelson:sodium}
O dubleto do sódio ($\Delta\lambda = \qty{0.6}{nm}$ em \qty{589}{nm}): os anéis se embaçam e
voltam a ficar nítidos a cada $\Delta e = 589^2/(2 \times 0.6)$ nm $= \qty{0.29}{mm}$ --- uma medida
clássica de $\Delta\lambda$ com alguns por cento de precisão, feita com um parafuso. Michelson (1892)
contou as franjas da linha vermelha do cádmio ao longo do
metro padrão: $1\,553\,163.5$ comprimentos de onda, o primeiro laço entre o
metro e um átomo; hoje o metro é definido por $c$ e pelo
segundo, e interferômetros verificam blocos-padrão na produção com
precisão de nanômetros.
\end{example}

\begin{remark}[Os braços que não mediram nada, e os que mediram uma onda]\label{rem:b2:michelson:history}
\index{experiência de Michelson--Morley}\index{detector de ondas gravitacionais}
Michelson e Morley esperavam que a velocidade $v$ da Terra através do
éter mudasse o tempo de ida e volta ao longo do braço paralelo a ela de
$(L/c)(v/c)^2$ em relação ao braço perpendicular, um deslocamento de franjas de
$2L(v/c)^2/\lambda \approx 0.4$ franja para $L = \qty{11}{m}$ e $v = \qty{30}{km/s}$;
girar o aparelho deveria ter feito o padrão oscilar de outro tanto.
Eles conseguiam ver $0.01$ franja; não viram nada. Os \omterm{rem:b2:michelson:history}{detectores de ondas
gravitacionais} de hoje são Michelsons com braços de \qty{4}{km}, com a luz ricocheteando
umas trezentas vezes em cada um, e um laser estável o bastante para ler uma
variação de comprimento de braço $\Delta L/L \sim 10^{-21}$ --- $10^{-18}$ m, um milésimo
de um próton --- contra o ruído dos movimentos sísmicos e dos próprios
fótons.
\end{remark}

\begin{method}[Usar um Michelson]\label{met:b2:michelson:method}
(1) Identifique a configuração: lâmina de faces paralelas (anéis no infinito;
observe com uma lente ou com o olho no infinito) ou cunha (franjas sobre os
espelhos; focalize $M_2$). (2) Escreva $\delta = 2e\cos i$ ou $2\alpha x$ e a
ordem. (3) Traduza cada medida em franjas: $\Delta e = N\lambda/2$,
$2(n - 1)\ell$ para uma lâmina, $\lambda^2/2\Delta\lambda$ para um dubleto. (4) Use a
posição de contato ($e = 0$, franja branca) como referência absoluta;
note que os raios dos anéis \emph{diminuem} quando $e$ cresce. (5) Limites
de coerência: os anéis são visíveis enquanto $2e < \ell_c$; a cunha exige
incidência quase normal.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:michelson:1}
Um Michelson em luz de sódio ($\lambda = \qty{589}{nm}$): o espelho é deslocado de
\qty{0.10}{mm}: número de anéis que passam pelo centro; deslocamento
correspondente a $250$ franjas; precisão sobre $e$ se se pode ler um décimo de
franja.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:2}
Lâmina de faces paralelas de espessura $e = \qty{0.50}{mm}$, lente $f = \qty{1.0}{m}$, $\lambda =
\qty{589}{nm}$: ordem no centro; o centro é claro? Raios dos
três primeiros anéis claros; o que acontece com os anéis quando $e$ é reduzido
a \qty{0.05}{mm}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:3}
Cunha de ar com $\alpha = \qty{1.0e-4}{rad}$, $\lambda = \qty{633}{nm}$: espaçamento das franjas;
número de franjas ao longo de \qty{2}{cm} de espelho; o ângulo da cunha é
dobrado: o que acontece? Como se encontra o contato com luz branca?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:michelson:4}
Uma lâmina de vidro de \qty{0.10}{mm} de espessura é inserida num dos braços: deslocamento das franjas para
$n = 1.52$ em \qty{589}{nm}; o deslocamento é medido com $0.2$ franja de precisão: precisão
sobre $n$. Por que a mesma medida não poderia ser feita com luz branca
se a lâmina tivesse \qty{1}{cm} de espessura?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:5}
\emph{Dubleto do sódio.} (a) Espessura $e$ entre dois desaparecimentos
sucessivos dos anéis, para $\Delta\lambda = \qty{0.597}{nm}$. (b) Um estudante
mede $17$ desaparecimentos ao longo de \qty{5.00}{mm}: $\Delta\lambda$ e sua precisão.
(c) Ordens das duas linhas em $e = \qty{0.29}{mm}$: verifique que diferem de
$\tfrac12$. (d) Por que o contraste nunca é exatamente nulo (as duas linhas têm
intensidades na razão $2 : 1$): contraste mínimo.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:6}
\emph{Índice do ar.} Uma cuba de \qty{10.0}{cm} num dos braços é evacuada e depois
lentamente preenchida com ar a \qty{1}{bar}; passam $94$ franjas ($\lambda = \qty{589}{nm}$).
(a) $n - 1$ para o ar. (b) Franjas que passam para uma variação de \qty{1}{\%} da
pressão; coeficiente de temperatura se $n - 1 \propto \rho$. (c) Por que isso
importa para a estabilidade de qualquer interferômetro no ar? (d) Precisão sobre
$n - 1$ a partir de um décimo de franja.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:7}
\emph{\omterm{prop:b2:scalar-light-model:coherence}{Comprimento de coerência}.} Com uma lâmpada de mercúrio de baixa pressão (linha verde,
\qty{546}{nm}, largura \qty{0.01}{nm}) os anéis se apagam à medida que $e$ cresce. (a) \omterm{prop:b2:scalar-light-model:coherence}{Comprimento
de coerência}; espessura $e$ na qual o contraste se torna desprezível. (b) O mesmo
com uma lâmpada de alta pressão ($\Delta\lambda = \qty{1}{nm}$). (c) Com um laser He--Ne
($\Delta\nu = \qty{1}{MHz}$): ainda se veem anéis com $e = \qty{10}{m}$? (d) O que
limita as franjas em luz branca a $\pm2$ ordens em torno do contato?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:8}
\emph{\omterm{def:b2:michelson:instrument}{Lâmina compensadora}.} Sem $C$, um dos feixes atravessa o vidro do
divisor ($e_g = \qty{5}{mm}$, $n = 1.52$) três vezes e o outro uma só. (a)
\omterm{def:b2:scalar-light-model:path}{Caminho óptico} extra; ele pode ser compensado deslocando um espelho, em
luz monocromática? (b) Por que não em luz branca (dispersão: $n$
varia de $0.01$ no visível)? (c) Que tolerância sobre o
paralelismo de $C$ e $Sp$ preserva a franja branca (tome uma diferença
residual de espessura $< \qty{0.3}{\micro m}$)? (d) Os divisores modernos são
películas finas ou cubos: por que $C$ é então desnecessária?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:michelson:9}
\emph{Michelson--Morley.} Braços de $L = \qty{11}{m}$ (dobrados), $\lambda = \qty{550}{nm}$,
velocidade orbital da Terra $v = \qty{30}{km/s}$. (a) Na teoria do éter a
ida e volta ao longo do braço paralelo a $\vect v$ leva $2L/c(1 - v^2/c^2)$
e a do braço perpendicular $2L/c\sqrt{1 - v^2/c^2}$: diferença, na ordem mais
baixa. (b) Deslocamento de franjas esperado ao girar o aparelho de \qty{90}{\degree}
(os papéis dos braços se trocam). (c) Eles podiam detectar $0.01$ franja: limite
superior sobre $v$. (d) O que diz um resultado nulo, e o que Einstein
concluiu?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:10}
\emph{Espectroscopia de Fourier.} A intensidade no centro dos anéis
vale $I(e) = \int S(\nu)[1 + \cos(4\pi\nu e/c)]\dd\nu$ para uma fonte de espectro
$S(\nu)$. (a) Para uma única linha: $I(e)$; período em $e$. (b) Para uma linha
gaussiana de largura $\Delta\nu$: mostre que as franjas têm uma envoltória gaussiana de
largura $\sim c/2\Delta\nu$ em $e$ (admita que a transformada de Fourier de uma gaussiana
é uma gaussiana). (c) Resolução: duas linhas separadas de $\Delta\nu$ ficam separadas se
a varredura alcança $e_{\max} \approx c/2\Delta\nu$; para $e_{\max} = \qty{10}{cm}$, o
poder de resolução $\lambda/\Delta\lambda$ em \qty{1}{\micro m}. (d) Por que esse método é
preferido no infravermelho (sinal, vantagem "multiplex")?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:11}
\emph{\omterm{rem:b2:michelson:history}{Detector de ondas gravitacionais}.} Braços $L = \qty{4}{km}$, luz reciclada
em $N = 300$ idas e voltas, $\lambda = \qty{1064}{nm}$, a onda muda os braços de
$\Delta L/L = 10^{-21}$ em sentidos opostos. (a) Variação da \omterm{thm:b2:two-wave-interference:formula}{diferença de caminho};
desvio de fase. (b) O instrumento é mantido numa franja escura e mede
a pequena intensidade $I = I_0\sin^2(\Delta\varphi/2)$: variação de intensidade para o
sinal. (c) Com \qty{100}{W} de luz circulante em \qty{1064}{nm}, fótons
por segundo; o ruído de fótons em \qty{1}{ms} ($\sqrt N$) contra o sinal:
ele é detectável? O que se ganha com mais potência? (d) Por que os braços têm
quilômetros de comprimento, e por que os espelhos são suspensos?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:michelson:12}
\emph{Localização.} (a) Mostre que, para a cunha, os dois raios que saem de um
ponto $P$ de $M_2$ rumo a um observador no infinito (incidência $i$) têm a
\omterm{thm:b2:two-wave-interference:formula}{diferença de caminho} $2e(P)\cos i$ mais um termo $\propto\alpha i$ que se anula para
$i = 0$: as franjas ficam sobre os espelhos apenas sob incidência normal. (b)
Uma \omterm{prop:b2:two-wave-interference:spatial}{fonte extensa} ilumina a cunha com incidências até
$i_{\max}$: o que $i_{\max}$ deve satisfazer para que as franjas permaneçam nítidas
($2e\,i_{\max}^2/2 < \lambda/4$ na borda do campo, $e = \qty{20}{\micro m}$)? (c)
Para a lâmina de faces paralelas os anéis não se movem quando o olho se move:
por quê? (d) Uma estudante vê franjas que derivam quando ela move a cabeça:
em que configuração ela está, e o que deve ajustar?
\end{exercise}

\section{Problema: o dubleto do sódio e o índice do ar}

\begin{problem}[{Problema de fim de semana --- uma sessão completa de medidas
num interferômetro de Michelson: achar o contato, ler comprimentos, separar
a linha amarela, pesar o ar}]
\label{pb:b2:michelson:1}

O interferômetro tem um parafuso micrométrico que lê \qty{1}{\micro m}; a
lâmpada de sódio dá $\lambda_1 = \qty{589.0}{nm}$ e $\lambda_2 = \qty{589.6}{nm}$ com
intensidades $2 : 1$; uma lâmpada branca e um laser He--Ne (\qty{632.8}{nm}) também
estão disponíveis.

\medskip
\noindent\textbf{Parte I --- A regulagem.}
\begin{enumerate}
  \item Com o laser, a estudante vê sobre um cartão dois pontos luminosos
        vindos dos dois braços; ela os superpõe com os parafusos de inclinação e vê então
        franjas retilíneas: que configuração é esta, e o que
        seu espaçamento diz sobre a inclinação residual (espaçamento \qty{2}{mm})?
  \item Girando os parafusos de inclinação, as franjas se encurvam e depois se tornam
        anéis: por que aparecem anéis, e onde está $M_1'$ agora?
  \item Ela translada o espelho e conta $158$ anéis engolidos pelo
        centro: deslocamento; em que sentido $e$ está variando?
  \item À medida que ela prossegue, os anéis se espalham e a mancha central cresce
        até preencher o campo: o que ela alcançou? A lâmpada de sódio
        substitui o laser e ela não vê nada de especial: por que não, e
        por que a luz branca mostra nesse ponto uma franja central clara
        ladeada por franjas coloridas?
  \item Ela inclina $M_1$ muito ligeiramente no contato e vê, em luz
        branca, franjas retilíneas coloridas: explique, e diga quantas
        ela pode esperar de cada lado da branca.
  \item Os pontos do laser têm \qty{10}{mm} de diâmetro e ela quer dez anéis
        no campo de uma lente de $f = \qty{20}{cm}$ ($\lambda = \qty{632.8}{nm}$):
        espessura $e$ necessária (use o raio do $k$-ésimo anel).
  \item Por que a \omterm{def:b2:michelson:instrument}{lâmina compensadora} deve ser paralela ao divisor,
        e o que ela veria em luz branca se a compensadora fosse retirada?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O dubleto.}
\begin{enumerate}[resume]
  \item Partindo do contato com a lâmpada de sódio, ela aumenta $e$:
        o contraste dos anéis cai, se anula, volta. Espessura do
        primeiro desaparecimento; do $n$-ésimo.
  \item Ela registra $12$ desaparecimentos entre $e = \qty{0.15}{mm}$ e
        $e = \qty{3.35}{mm}$ (o primeiro e o último lidos no parafuso):
        $\Delta\lambda$; precisão se cada leitura é boa até \qty{5}{\micro m}.
  \item Ordens $p_1$ e $p_2$ das duas linhas no primeiro
        desaparecimento; no primeiro retorno do contraste pleno.
  \item Contraste mínimo num desaparecimento, dada a razão de
        intensidades $2 : 1$ (as franjas da linha mais fraca permanecem).
  \item Quantos desaparecimentos ela pode contar antes que a largura natural
        das linhas (\qty{0.02}{nm} cada) apague tudo?
  \item O que mostraria uma lâmpada com três linhas igualmente espaçadas?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O ar.}
Uma cuba de comprimento $\ell = \qty{5.00}{cm}$ com janelas de vidro é colocada num dos
braços e evacuada; o ar é admitido lentamente enquanto se contam as franjas
do laser.
\begin{enumerate}[resume]
  \item \omterm{thm:b2:two-wave-interference:formula}{Diferença de caminho} óptico criada ao preencher a cuba até
        \qty{1}{atm} se $n - 1 = \qty{2.9e-4}{}$; número de franjas.
  \item Ela conta $46$ franjas: seu valor de $n - 1$.
  \item Ela admite apenas \qty{0.20}{bar}: franjas esperadas ($n - 1 \propto$
        pressão).
  \item A sala esquenta \qty{3}{K} durante a medida, e os
        \qty{40}{cm} de percurso ao ar livre em cada braço mudam de índice de
        $-\qty{1e-6}{}$ por kelvin: deriva das franjas; isso é um problema para a
        Parte II, para a Parte III?
  \item Por que as janelas da cuba não deslocam a medida (elas estão
        lá durante toda a contagem)?
  \item A experiência é repetida com CO$_2$ ($n - 1 = \qty{4.5e-4}{}$):
        franjas para uma cuba cheia; como se poderia identificar o gás só
        pela contagem?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Os limites.}
\begin{enumerate}[resume]
  \item O parafuso é lido a \qty{1}{\micro m}: precisão de um comprimento
        medido por contagem de franjas ao longo de \qty{1}{cm}, comparada à do parafuso
        sozinho.
  \item Uma vibração de \qty{50}{nm} de amplitude a \qty{100}{Hz} sacode um dos
        espelhos: movimento das franjas; o que vê o olho, o que vê uma
        câmera a $25$ quadros por segundo?
  \item A frequência do laser deriva de \qty{1e-8}{} em valor relativo
        durante uma contagem sobre $e = \qty{1}{cm}$: erro em franjas.
  \item Um bloco-padrão de \qty{10}{mm} é certificado a $\lambda/20$ por
        interferometria: precisão absoluta e relativa.
  \item Por que o Michelson é o instrumento de eleição para comprimentos, e
        a rede (próximo capítulo) para espectros?
  \item Resuma as quatro grandezas medidas (um comprimento, a separação
        de um dubleto, um índice, um \omterm{prop:b2:scalar-light-model:coherence}{comprimento de coerência}) e a observação
        de franjas que deu cada uma.
\end{enumerate}
\end{problem}
