\chapter{Ondas eletromagnéticas em plasmas, condutores e dielétricos}\label{ch:b2:waves-in-media}

Uma transmissão em ondas curtas atravessa um oceano ricocheteando na alta
atmosfera, que a reflete como um espelho --- e no entanto a mesma camada deixa
passar sinais de televisão e de satélite. Uma colher de metal é brilhante, uma
tela metálica na porta de um forno de micro-ondas retém as ondas enquanto
você observa a comida através dela, e uma folha de vidro é transparente
à luz e opaca ao ultravioleta. Em cada caso uma onda entra num
material, as cargas do material respondem, e sua resposta muda
a onda: sua celeridade, sua \omterm{def:b2:dispersion-wave-packets:complex}{atenuação} e até se ela se propaga ou
não. Este capítulo trata os três modelos padrão de meio --- um
gás de elétrons livres, um condutor ôhmico e elétrons ligados --- com
o único método que dá conta de todos: acrescentar a corrente induzida às
\omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell} e ler a \omterm{def:b2:dispersion-wave-packets:relation}{relação de dispersão}.

\section{Ondas num plasma}

\begin{definition}[Plasma; o modelo do elétron livre]\label{def:b2:waves-in-media:plasma}
\index{plasma}\index{frequência de plasma}
Um \emph{plasma} é um gás globalmente neutro de elétrons livres (densidade
$n$, massa $m$, carga $-e$) e de íons positivos que, sendo milhares de
vezes mais pesados, são tomados como fixos. A alta atmosfera (ionosfera,
$n \sim 10^{11}$--$10^{12}$ m$^{-3}$), um tubo de descarga, a coroa solar
e --- em frequências ópticas --- os elétrons de condução de um metal
são plasmas. As colisões são desprezadas (o período da onda é curto
comparado com o tempo entre colisões). Sua frequência
característica é a \emph{frequência de plasma}
\[
\omega_p = \sqrt{\frac{ne^2}{\varepsilon_0m}} , \qquad
f_p = \frac{\omega_p}{2\pi} \approx 9\,\text{Hz}\times\sqrt{n\,[\text{m}^{-3}]} :
\]
de \qty{3}{MHz} a \qty{9}{MHz} para a ionosfera, e \qty{2e15}{Hz} (ultravioleta) para
um metal.
\end{definition}

\begin{theorem}[Relação de dispersão de um plasma]\label{thm:b2:waves-in-media:plasmadispersion}
\index{frequência de corte (plasma)}\index{onda evanescente (plasma)}
Para uma onda OPPH transversal $\underline{\vect E} = \underline{\vect E}_0\eu^{\iu(\omega t - kx)}$ num
\omterm{def:b2:waves-in-media:plasma}{plasma}, os elétrons oscilam com $\underline{\vect v} = -e\underline{\vect E}/\iu m\omega$
e transportam a \omterm{def:b2:charges-currents-conduction:densities}{densidade de corrente} $\underline{\vect j} = \underline\gamma\underline{\vect E}$ com
$\underline\gamma = ne^2/\iu m\omega$; as \omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell} dão então
\[
k^2 = \frac{\omega^2 - \omega_p^2}{c^2} .
\]
Acima de $\omega_p$ a onda se propaga, com $v_\varphi = c/\sqrt{1 - \omega_p^2/\omega^2}
> c$ e $v_g = c\sqrt{1 - \omega_p^2/\omega^2} < c$, $v_\varphi v_g = c^2$; abaixo de
$\omega_p$, $k = -\iu\kappa$ com $\kappa = \sqrt{\omega_p^2 - \omega^2}/c$: a onda é
\emph{evanescente}, penetra numa distância $1/\kappa$ e é totalmente refletida
(\cref{ch:b2:wave-interfaces}); o \omterm{def:b2:waves-in-media:plasma}{plasma} é então um espelho. Com $\omega
\gg \omega_p$ os elétrons não conseguem acompanhar e o \omterm{def:b2:waves-in-media:plasma}{plasma} é transparente.
\end{theorem}

\begin{proof}
Equação do movimento $m\,\dd\vect v/\dd t = -e\vect E$ (a força magnética
$ev B \sim evE/c$ é desprezível para $v \ll c$, e o movimento dos íons é
$m/M$ vezes menor): $\iu m\omega\underline{\vect v} = -e\underline{\vect E}$. Então
$\underline{\vect j} = -ne\underline{\vect v}$. O \omterm{def:b2:waves-in-media:plasma}{plasma} permanece neutro para uma
onda transversal ($\vect k\cdot\vect E = 0$ dá $\rho = 0$ por Gauss), de modo que
Maxwell--Ampère diz $-\iu\vect k\wedge\underline{\vect B} = \mu_0\underline\gamma\underline{\vect E}
+ \iu\omega\mu_0\varepsilon_0\underline{\vect E}$ e Maxwell--Faraday $\underline{\vect B} = \vect k
\wedge\underline{\vect E}/\omega$; combinando, $k^2\underline{\vect E} = \mu_0(\iu\omega\underline\gamma -
\omega^2\varepsilon_0)(-1)\underline{\vect E}$, ou seja, $k^2 = \omega^2\mu_0\varepsilon_0 - \iu\omega\mu_0
\underline\gamma = (\omega^2 - ne^2/\varepsilon_0m)/c^2$. As velocidades seguem como no
\cref{ch:b2:dispersion-wave-packets}.
\end{proof}

\begin{remark}[Para onde vai a energia]\label{rem:b2:waves-in-media:plasmaenergy}
A condutividade é imaginária: $\langle\vect j\cdot\vect E\rangle = 0$, os
elétrons tomam energia da onda durante meio ciclo e a devolvem
no meio ciclo seguinte --- um meio sem perdas, no qual a energia da onda
se reparte entre os campos e a energia cinética dos elétrons, e
viaja a $v_g$. As colisões acrescentam uma pequena parte real a $\underline\gamma$ e uma
pequena absorção: a camada D da ionosfera absorve as ondas médias de
dia, e é por isso que as emissoras AM distantes chegam à noite.
\end{remark}

\begin{example}[A ionosfera e o rádio]\label{ex:b2:waves-in-media:ionosphere}
Com $n = \qty{1e12}{m^{-3}}$ na camada F, $f_p = \qty{9}{MHz}$: as transmissões
em ondas curtas abaixo disso são refletidas (e podem saltar em torno da Terra
entre a ionosfera e o solo), ao passo que o FM (\qty{100}{MHz}), a televisão
e os enlaces de satélite atravessam. Uma onda de \qty{3}{MHz} penetra apenas
$1/\kappa = c/\sqrt{\omega_p^2 - \omega^2} = \qty{6}{m}$ na camada antes de
retornar. Uma nave que reentra na atmosfera fica envolvida num
\omterm{def:b2:waves-in-media:plasma}{plasma} de $n \sim 10^{18}$ m$^{-3}$, $f_p \sim \qty{10}{GHz}$: o contato de rádio se
perde por minutos.
\end{example}

\section{Ondas num condutor ôhmico: o efeito pelicular}

\begin{theorem}[Efeito pelicular]\label{thm:b2:waves-in-media:skin}
\index{efeito pelicular}\index{profundidade de penetração}
Num condutor ôhmico ($\vect j = \gamma\vect E$, com $\gamma$ real, em frequências
com $\omega\tau \ll 1$ e $\varepsilon_0\omega \ll \gamma$, ou seja, abaixo de $\sim\qty{1e16}{Hz}$ para o
cobre) a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento} é desprezível e os campos obedecem à
equação de difusão $\Delta\vect E = \mu_0\gamma\,\partial_t\vect E$. Uma onda OPPH que entra
no condutor em $x = 0$ tem
\[
\underline k^2 = -\iu\mu_0\gamma\omega , \qquad \underline k = \frac{1 - \iu}\delta , \qquad
\delta = \sqrt{\frac2{\mu_0\gamma\omega}} ,
\]
de modo que $\underline E = E_0\,\eu^{-x/\delta}\eu^{\iu(\omega t - x/\delta)}$: ela é amortecida
ao longo da \emph{\omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração}} $\delta$, com uma fase que também gira um
radiano por $\delta$ --- um comprimento de onda $2\pi\delta$, uma \omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} $\omega\delta \ll
c$. A corrente $\vect j = \gamma\vect E$ fica confinada a uma camada de espessura
$\delta$ sob a superfície, e $\vect B$ atrasa-se em relação a $\vect E$ de \qty{45}{\degree}, com
$|B| = |E|\sqrt2/\omega\delta$. Para o cobre, $\delta = \qty{9.2}{mm}$ a \qty{50}{Hz},
\qty{0.21}{mm} a \qty{100}{kHz} e \qty{2.1}{\micro m} a \qty{1}{GHz}.
\end{theorem}

\begin{proof}
$\operatorname{\vect{curl}}\vect B = \mu_0\gamma\vect E$ (deslocamento descartado) e
$\operatorname{\vect{curl}}\vect E = -\partial_t\vect B$: tomando o rotacional da segunda,
$-\Delta\vect E = -\mu_0\gamma\partial_t\vect E$ (o condutor é neutro, $\operatorname{div}
\vect E = 0$). Para $\eu^{\iu(\omega t - kx)}$: $k^2 = -\iu\mu_0\gamma\omega$, e
$\sqrt{-\iu} = (1 - \iu)/\sqrt2$ dá $\underline k = (1 - \iu)/\delta$ (a raiz que
decai para dentro do condutor). Faraday: $\underline B = \underline k\underline E/\omega = (1 - \iu)
\underline E/\omega\delta$, de módulo $\sqrt2E/\omega\delta$ e fase $-\pi/4$.
\end{proof}

\begin{example}[Consequências do efeito pelicular]\label{ex:b2:waves-in-media:skinuses}
\index{resistência em alternada}\index{gaiola de Faraday}
(i) Um fio de raio $a \gg \delta$ conduz sua corrente num anel de
espessura $\delta$: sua resistência sobe de $1/\gamma\pi a^2$ para cerca de $1/\gamma
2\pi a\delta$ por metro --- para um fio de cobre de \qty{1}{mm}, quatro vezes o valor
em contínua a \qty{1}{MHz} e cem vezes a \qty{1}{GHz}. As bobinas de alta frequência
usam fio "Litz" (muitos filamentos isolados) ou tubos prateados; os
condutores da rede elétrica são encordoados com uma capa de alumínio. (ii) Uma
caixa metálica atenua uma onda por $\eu^{-d/\delta}$ a cada espessura $d$: uma
caixa de alumínio de \qty{1}{mm} bloqueia \qty{1}{MHz} em \qty{100}{dB}, mas \qty{50}{Hz}
quase nada --- os campos magnéticos de baixa frequência exigem ferro. (iii)
Os submarinos só recebem rádio de ondas muito longas ($\delta = \qty{2.3}{m}$ a
\qty{10}{kHz} na água do mar); as minas são detectadas pelas correntes de Foucault que uma
bobina induz nelas (\cref{ch:b2:maxwell-equations}, com o tempo de
difusão $\mu_0\gamma d^2$).
\end{example}

\section{Ondas num dielétrico: o elétron ligado}

\begin{proposition}[Elétron ligado elasticamente; índice complexo]\label{prop:b2:waves-in-media:lorentz}
\index{dielétrico}\index{polarização (de um meio)}\index{suscetibilidade}\index{índice complexo}\index{modelo de Lorentz}
Num isolante cada elétron está ligado a seu átomo: um deslocamento
$\vect r$ em relação ao equilíbrio chama a força restauradora $-m\omega_0^2\vect r$ e
um amortecimento $-m\Gamma\,\dd\vect r/\dd t$ (a energia irradiada ou entregue à
rede). Excitado pela onda, $m\ddot{\vect r} = -m\omega_0^2\vect r - m\Gamma\dot{\vect r}
- e\vect E$; os elétrons deslocados dão ao meio a \emph{polarização}
(momento dipolar por unidade de volume) $\vect P = -ne\vect r = \varepsilon_0\underline\chi(\omega)
\underline{\vect E}$, com a \emph{suscetibilidade}
\[
\underline\chi(\omega) = \frac{\omega_p^2}{\omega_0^2 - \omega^2 + \iu\Gamma\omega} ,
\qquad \omega_p^2 = \frac{ne^2}{\varepsilon_0m} ,
\]
e a corrente ligada $\vect j = \partial_t\vect P$. As \omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell} dão então
\[
\underline k^2 = \frac{\omega^2}{c^2}\,(1 + \underline\chi) = \frac{\omega^2}{c^2}\,\underline n^2 ,
\qquad \underline n = n' - \iu n'' ,
\]
com $n'$ o índice de refração (\omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} $c/n'$) e $n''$ a
extinção (a intensidade decai como $\eu^{-2n''\omega x/c}$). Bem abaixo de uma
ressonância ($\omega \ll \omega_0$, com $\Gamma$ desprezível), $n'^2 \approx 1 + \omega_p^2/\omega_0^2 +
\omega_p^2\omega^2/\omega_0^4$: o índice \emph{cresce} com a frequência --- dispersão
normal, lei de Cauchy $n \approx A + B/\lambda^2$ --- e a absorção é
desprezível; perto de $\omega_0$ o meio absorve, e logo acima dela $n'$
cai com $\omega$ (dispersão anômala).
\end{proposition}

\begin{proof}
Amplitudes complexas: $(\omega_0^2 - \omega^2 + \iu\Gamma\omega)\underline{\vect r} = -e\underline{\vect E}
/m$; $\underline{\vect P} = -ne\underline{\vect r}$. Então, como no \omterm{def:b2:waves-in-media:plasma}{plasma}, $k^2 = \omega^2\mu_0
\varepsilon_0 - \iu\omega\mu_0\underline\gamma$ com $\underline{\vect j} = \iu\omega\underline{\vect P} = \iu\omega\varepsilon_0
\underline\chi\underline{\vect E}$, ou seja, $\underline\gamma = \iu\omega\varepsilon_0\underline\chi$: $k^2 = (\omega^2/c^2)
(1 + \underline\chi)$. (O \omterm{def:b2:waves-in-media:plasma}{plasma} é o caso $\omega_0 = 0$, $\Gamma = 0$.)
Expansão: $1/(\omega_0^2 - \omega^2) \approx (1 + \omega^2/\omega_0^2)/\omega_0^2$.
\end{proof}

\begin{example}[Vidro, água, ar]\label{ex:b2:waves-in-media:dielectrics}
O vidro tem suas ressonâncias eletrônicas no ultravioleta ($\omega_0 \sim 10^{16}$
rad/s): no visível ele é transparente e dispersivo, $n = 1.51$ a
\qty{700}{nm} e $1.53$ a \qty{400}{nm}, o bastante para um prisma abrir um
arco-íris e para os pulsos de uma fibra se alargarem (\cref{ch:b2:dispersion-wave-packets});
no ultravioleta ele absorve --- ninguém se bronzeia atrás de uma janela.
A água tem, além disso, uma ressonância de rotação de suas moléculas polares perto de
\qty{20}{GHz} (uma relaxação de Debye, e não uma linha estreita): ela absorve
fortemente as micro-ondas, o que aquece a comida e cega o radar na chuva,
e tem $n'' \approx 0$ numa janela visível estreita --- a janela através
da qual os olhos evoluíram. O ar, com $n - 1 = 2.9 \times 10^{-4}$, ainda desvia o
Sol poente em meio grau.
\end{example}

\begin{remark}[Uma equação, três meios]\label{rem:b2:waves-in-media:summary}
Todo meio linear entra nas \omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell} através de sua corrente
induzida $\underline{\vect j} = \underline\gamma(\omega)\underline{\vect E}$, e a \omterm{def:b2:dispersion-wave-packets:relation}{relação de
dispersão} é sempre $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma(\omega)$:
\begin{center}\small
\begin{tabular}{lll}
meio & $\underline\gamma(\omega)$ & $\underline k^2$\\ \hline
\omterm{def:b2:waves-in-media:plasma}{plasma} & $ne^2/\iu m\omega$ (imaginária) & $(\omega^2 - \omega_p^2)/c^2$: corte\\
condutor ôhmico & $\gamma$ (real) & $-\iu\mu_0\gamma\omega$: \omterm{thm:b2:waves-in-media:skin}{efeito pelicular}\\
dielétrico & $\iu\omega\varepsilon_0\underline\chi(\omega)$ & $(\omega^2/c^2)(1 + \underline\chi)$: índice
\end{tabular}
\end{center}
Um metal é os três, um após o outro: ôhmico abaixo de \qty{1e13}{Hz}, um \omterm{def:b2:waves-in-media:plasma}{plasma}
(refletor) no visível e transparente no ultravioleta distante.
\end{remark}

\begin{method}[Ondas num meio]\label{met:b2:waves-in-media:method}
(1) Escreva a equação do movimento das cargas e obtenha $\underline{\vect j} =
\underline\gamma\underline{\vect E}$. (2) Insira em Maxwell--Ampère com o ansatz
OPPH: $\underline k^2 = \omega^2/c^2 - \iu\omega\mu_0\underline\gamma$. (3) Tome a raiz
com $\operatorname{Im}k \le 0$ (decaimento no sentido de propagação), separe
em parte real (propagação, $v_\varphi = \omega/k'$) e imaginária (\omterm{def:b2:dispersion-wave-packets:complex}{atenuação}).
(4) Identifique os regimes em frequência --- propagante,
evanescente, absorvente --- e a frequência característica ($\omega_p$,
$1/\mu_0\gamma\delta^2$, $\omega_0$). (5) Ache $\vect B$ por Faraday e, se necessário,
o fluxo de Poynting e a potência dissipada.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:1}
\omterm{def:b2:waves-in-media:plasma}{Frequências de plasma} para $n = \qty{1e11}{m^{-3}}$ (ionosfera à noite),
\qty{1e12}{m^{-3}} (de dia), \qty{1e18}{m^{-3}} (bainha de reentrada), \qty{8.5e28}{m^{-3}}
(cobre), \qty{1e5}{m^{-3}} (espaço interestelar). Para cada uma, o que
passa: uma emissora AM de \qty{1}{MHz}, uma FM de \qty{100}{MHz}, um
sinal GPS de \qty{1.5}{GHz}, a luz visível?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:2}
\omterm{def:b2:dispersion-wave-packets:complex}{Profundidade de penetração} do cobre ($\gamma = \qty{6.0e7}{S/m}$) a \qty{50}{Hz}, \qty{1}{kHz},
\qty{1}{MHz} e \qty{1}{GHz}; do alumínio (\qty{3.6e7}{S/m}) a \qty{1}{MHz}; da água
do mar (\qty{5}{S/m}) a \qty{10}{kHz} e \qty{1}{GHz}; do solo úmido (\qty{1e-2}{S/m})
a \qty{1}{MHz}. Que frequências de rádio alcançam um submarino a \qty{10}{m}?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:3}
Um vidro obedece a $n = 1.50 + \qty{4.5e3}{nm^2}/\lambda^2$. Índice a \qty{400}{nm},
\qty{550}{nm} e \qty{700}{nm}; \omterm{def:b2:dispersion-wave-packets:relation}{velocidades de fase}; índice de grupo $n_g = n -
\lambda\,\dd n/\dd\lambda$ a \qty{550}{nm}; abertura angular de um feixe branco
refratado no vidro sob incidência de \qty{45}{\degree}.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:waves-in-media:4}
Uma onda de \qty{5}{MHz} encontra uma camada com $n = \qty{2e11}{m^{-3}}$: ela é
refletida? \omterm{def:b2:dispersion-wave-packets:complex}{Profundidade de penetração}; o mesmo a \qty{2}{MHz}; frequência mínima
que atravessa uma camada de $n = \qty{1.5e12}{m^{-3}}$. Por que as faixas de ondas
curtas "abrem" e "fecham" com a hora e com o ciclo solar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:5}
\emph{\omterm{ex:b2:waves-in-media:skinuses}{Resistência em alternada}.} Um fio de cobre de raio $a = \qty{1.0}{mm}$. (a) Resistência
em contínua por metro. (b) A \qty{1}{MHz}: $\delta$; resistência aproximada
por metro com a corrente confinada a um anel de espessura $\delta$;
razão em relação à contínua. (c) A \qty{100}{MHz}. (d) Por que uma bobina de $N$ espiras
desse fio tem um fator de qualidade que deixa de melhorar com $N$ em alta
frequência, e o que é o fio Litz?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:6}
\emph{Blindagem.} Uma caixa de alumínio ($\gamma = \qty{3.6e7}{S/m}$) de \qty{1.0}{mm}
de espessura. \omterm{def:b2:dispersion-wave-packets:complex}{Atenuação} da amplitude do campo ao atravessar a parede, em dB, a
\qty{50}{Hz}, \qty{10}{kHz} e \qty{1}{MHz} (o fator $\eu^{-d/\delta}$, desprezando a reflexão nas
superfícies). Por que o campo magnético da rede é blindado
com ferro (alta permeabilidade) em vez de cobre?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:7}
\emph{Energia num \omterm{def:b2:waves-in-media:plasma}{plasma}.} Para a onda transversal de
\cref{thm:b2:waves-in-media:plasmadispersion}: (a) amplitude da velocidade
dos elétrons e densidade de energia cinética $\tfrac12nm\langle v^2\rangle$. (b)
Densidades médias de energia elétrica e magnética. (c) Mostre que a densidade
total de energia vale $\tfrac12\varepsilon_0E_0^2(1 + \dots)$ --- calcule o parêntese ---
e que $\langle\Pi\rangle$ dividido por ela dá $v_g$. (d) Que fração da
energia está nos elétrons em $\omega = 2\omega_p$?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:8}
\emph{Dispersão dos pulsares.} Pulsos de rádio de um pulsar atravessam uma distância
$L$ de \omterm{def:b2:waves-in-media:plasma}{plasma} interestelar ($n \ll$ qualquer coisa, $\omega \gg \omega_p$). (a) Mostre
que o atraso de grupo vale $t \approx (L/c)(1 + \omega_p^2/2\omega^2)$. (b) O atraso
entre as chegadas em $f_1$ e $f_2$ vale $\Delta t = \dfrac{e^2}{8\pi^2
\varepsilon_0mc}\,nL\Bigl(\dfrac1{f_1^2} - \dfrac1{f_2^2}\Bigr)$: calcule a constante.
(c) Um pulsar a $L = \qty{1}{kpc} = \qty{3.1e19}{m}$ através de $n = \qty{3e4}{m^{-3}}$:
atraso entre \qty{400}{MHz} e \qty{1.4}{GHz}. (d) Os astrônomos medem
$\Delta t$ e conhecem $n$: o que eles obtêm? (A medida de dispersão, a
régua padrão de distâncias para pulsares.)
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:waves-in-media:9}
\emph{Uma linha espectral.} Num gás rarefeito ($\underline n \approx 1 + \underline\chi/2$) o
\omterm{prop:b2:waves-in-media:lorentz}{modelo de Lorentz} dá $n'' = \tfrac12\operatorname{Im}\underline\chi$. (a) Mostre que perto de
$\omega_0$, $n'' \approx \dfrac{\omega_p^2}{4\omega_0}\,\dfrac{\Gamma/2}{(\omega_0 - \omega)^2 + \Gamma^2/4}$, uma
lorentziana de largura total $\Gamma$. (b) Coeficiente de absorção $\alpha = 2n''\omega/c$
no centro. (c) Vapor de sódio, $n = \qty{1e17}{m^{-3}}$, $\lambda_0 = \qty{589}{nm}$,
$\Gamma = \qty{6e7}{s^{-1}}$: $\alpha$ no centro e a espessura que absorve
$90\%$ (ignore o alargamento Doppler, que na verdade alarga a linha
cem vezes). (d) Esboce $n' - 1$ ao longo da linha; onde a
dispersão é anômala?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:10}
\emph{Aquecimento Joule na película.} Uma onda $E_0\eu^{\iu\omega t}$ (segundo $y$)
entra num condutor que preenche $x > 0$. (a) Escreva $\underline E(x, t)$ e deduza
$\underline B(x, t)$ (segundo $z$) por Faraday. (b) \omterm{thm:b2:poynting-vector:poynting}{Vetor de Poynting} médio na
superfície: mostre que $\langle\Pi_x(0)\rangle = E_0^2/2\mu_0\omega\delta$. (c) Potência
Joule média por unidade de área, $\int_0^\infty\tfrac12\gamma|\underline E|^2\dd x$: mostre que ela é igual a
(b). (d) Exprima o resultado como $\tfrac12R_sH_0^2$ com $H_0 = B_0/\mu_0$ o
campo magnético de superfície e $R_s = 1/\gamma\delta$ a \emph{resistência
superficial}; valor para o cobre a \qty{1}{GHz}, e a perda de uma
cavidade de micro-ondas cujas paredes carregam $H_0 = \qty{100}{A/m}$.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:11}
\emph{Por que os metais brilham.} Trate os elétrons de condução da prata ($n =
\qty{5.9e28}{m^{-3}}$) como um \omterm{def:b2:waves-in-media:plasma}{plasma} sem colisões em frequências ópticas.
(a) $\omega_p$ e o comprimento de onda correspondente. (b) A \qty{500}{nm}: $k$ é
imaginário --- \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração}; a onda é totalmente refletida (sem
absorção neste modelo): a prata é um espelho. (c) E a \qty{200}{nm}? Daí
a "transparência no ultravioleta" dos metais alcalinos. (d) A prata real
reflete $95\%$, e o ouro parece amarelo: o que o modelo deixa de fora (pense nas
colisões e nos elétrons ligados cujas ressonâncias caem no
visível, no caso do ouro)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:waves-in-media:12}
\emph{Água no micro-ondas.} A orientação das moléculas polares da água
dá uma suscetibilidade $\underline\chi = \chi_s/(1 + \iu\omega\tau)$ (relaxação de
Debye), com $\chi_s \approx 80$, $\tau = \qty{8}{ps}$, sobre uma constante
$\chi_\infty \approx 4$. (a) Partes real e imaginária de $\underline\varepsilon_r = 1 + \chi_\infty
+ \underline\chi$ a \qty{2.45}{GHz} e a \qty{20}{GHz}. (b) \omterm{prop:b2:waves-in-media:lorentz}{Índice complexo} e o
comprimento de absorção $1/(2n''\omega/c)$ da intensidade a \qty{2.45}{GHz}: a que
profundidade um forno cozinha? (c) Potência absorvida por unidade de volume para uma
amplitude de campo $E_0$ na água: $\tfrac12\omega\varepsilon_0\varepsilon''E_0^2$; valor para
$E_0 = \qty{2}{kV/m}$. (d) Por que se usa \qty{2.45}{GHz} e não os
\qty{20}{GHz} de absorção máxima?
\end{exercise}

\section{Problema: A ionosfera, a porta do forno e o prisma}

\begin{problem}[{Problema de fim de semana --- três materiais vistos por uma onda: o
plasma que encurva os sinais de GPS, o metal que retém as micro-ondas e
o vidro que abre a luz branca}]
\label{pb:b2:waves-in-media:1}

\medskip
\noindent\textbf{Parte I --- A ionosfera e o GPS.}
Densidade eletrônica na camada F, $n = \qty{1e12}{m^{-3}}$; a coluna total de
elétrons ao longo de um trajeto vertical vale $N_T = \int n\,\dd z = \qty{2e17}{m^{-2}}$ de
dia. GPS: $f_1 = \qty{1575}{MHz}$, $f_2 = \qty{1228}{MHz}$.
\begin{enumerate}
  \item \omterm{def:b2:waves-in-media:plasma}{Frequência de plasma} da camada F; o GPS passa? E uma onda
        curta de \qty{5}{MHz}?
  \item Mostre que, para $\omega \gg \omega_p$, a \omterm{prop:b2:dispersion-wave-packets:group}{velocidade de grupo} vale $v_g \approx c(1 -
        \omega_p^2/2\omega^2)$ e a \omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} $c(1 + \omega_p^2/2\omega^2)$.
  \item Atraso de grupo adicional do sinal GPS ao atravessar a camada: mostre que
        $\Delta t = \dfrac{e^2}{8\pi^2\varepsilon_0mc}\,\dfrac{N_T}{f^2}$ e calcule-o para
        $f_1$; o erro de distância $c\,\Delta t$.
  \item O receptor usa as duas frequências: dos dois atrasos ele
        remove o erro. Exprima $N_T$ em função de $\Delta t_1 - \Delta t_2$;
        que precisão sobre a diferença de atrasos dá $N_T$ com $1\%$?
  \item Quantos ciclos extras da portadora representa o atraso da questão 3
        em $f_1$?
  \item A fase da portadora é \emph{adiantada} do mesmo
        tanto que o grupo é atrasado: por quê (sinais das duas
        correções)? Qual delas importa para um receptor que conta
        ciclos da portadora?
  \item À noite $N_T$ cai dez vezes: erro para um receptor de
        frequência única de dia e de noite.
  \item Uma explosão solar eleva $n$ na camada D (\qty{80}{km}) a
        \qty{1e10}{m^{-3}}, com muitas colisões: que serviços de rádio são
        atingidos, e por que o GPS só se degrada um pouco?
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- A porta do forno.}
As paredes da cavidade e a tela da porta são de aço ($\gamma = \qty{1.0e6}{S/m}$);
os furos da tela têm \qty{1.0}{mm} de largura numa chapa de \qty{0.5}{mm} de espessura;
$f = \qty{2.45}{GHz}$, $P = \qty{800}{W}$.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Profundidade de penetração} no aço a \qty{2.45}{GHz}; a parede é "espessa"?
  \item Resistência superficial $R_s = 1/\gamma\delta$; com um campo magnético de
        superfície de amplitude $H_0 = \qty{30}{A/m}$ nas paredes (\qty{1}{m^2}),
        potência perdida nas paredes, $\tfrac12R_sH_0^2$ por unidade de área; fração
        de $P$.
  \item Num furo de largura $a \ll \lambda$ o campo não pode se propagar (o
        furo é um guia de ondas abaixo do corte, \cref{ch:b2:guided-waves}):
        ele decai como $\eu^{-\pi z/a}$ ao longo da espessura $z$. \omterm{def:b2:dispersion-wave-packets:complex}{Atenuação}
        da amplitude ao atravessar a chapa de \qty{0.5}{mm}, em dB.
  \item Por que se enxerga através da tela (comprimento de onda da luz
        contra o tamanho do furo), ao passo que as micro-ondas não saem?
  \item A janela de vidro é um dielétrico com $\underline\varepsilon_r = 4 - 0.02\iu$:
        índice, e o comprimento de \omterm{def:b2:dispersion-wave-packets:complex}{atenuação} da intensidade nele; o
        vidro é aquecido?
  \item Um garfo metálico no forno: o campo em suas pontas fica reforçado;
        com a \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração} da questão 8, estime a \omterm{def:b2:charges-currents-conduction:densities}{densidade de
        corrente} na superfície para $H_0 = \qty{300}{A/m}$ (use $j \approx H_0/\delta$
        na superfície) e o aquecimento local $j^2/\gamma$ --- por que as faíscas?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- O fio em radiofrequência.}
Um fio de cobre de raio $a = \qty{0.5}{mm}$ ($\gamma = \qty{6e7}{S/m}$) numa bobina
de aquecedor por indução de \qty{13.56}{MHz} conduz \qty{20}{A} eficazes.
\begin{enumerate}[resume]
  \item \omterm{def:b2:dispersion-wave-packets:complex}{Profundidade de penetração}; seção condutora efetiva; \omterm{ex:b2:waves-in-media:skinuses}{resistência em alternada} por
        metro contra a de contínua.
  \item Potência dissipada por metro de fio da bobina; temperatura que o
        fio atingiria com resfriamento convectivo $h = \qty{20}{W/(m^2.K)}$.
  \item A bobina induz correntes numa peça de aço ($\gamma =
        \qty{1e6}{S/m}$, $\mu_r \approx 1$ quando quente): \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração} ali; por que
        o aquecimento por indução aquece apenas uma camada superficial, e como isso é
        usado para temperar engrenagens?
  \item Para reduzir a perda da bobina, o fio é trocado por um tubo de cobre
        de \qty{3}{mm} com água por dentro: nova resistência por metro; o
        interior do tubo conduz corrente?
  \item Em que frequência a \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração} no cobre seria igual ao
        raio do fio? Abaixo dela, que aproximação substitui a
        fórmula do "fio espesso"?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- O prisma.}
Um vidro tem $n(\lambda) = 1.500 + \qty{5.0e3}{nm^2}/\lambda^2$, com sua ressonância
em $\lambda_0 = \qty{100}{nm}$.
\begin{enumerate}[resume]
  \item Verifique que a forma de Cauchy segue do \omterm{prop:b2:waves-in-media:lorentz}{modelo de Lorentz} bem
        abaixo da ressonância, e que $\omega_p^2/\omega_0^2 = n^2(\infty) - 1$: valor
        de $\omega_p^2/\omega_0^2$ e do $n$ efetivo deste vidro.
  \item Índice e \omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} a \qty{400}{nm} e \qty{700}{nm}; índice
        de grupo a \qty{550}{nm}; atraso entre as duas cores ao longo de
        \qty{1}{km} desse vidro numa fibra.
  \item Um prisma de \qty{60}{\degree} no desvio mínimo para \qty{550}{nm}:
        desvio (use $n\sin(A/2) = \sin((A + D)/2)$); distância angular
        entre \qty{400}{nm} e \qty{700}{nm} (derive); num anteparo
        a \qty{2}{m}, a largura do espectro.
  \item Por que o vidro é opaco a \qty{200}{nm}, e qual é a sua
        cor se um traço de ferro acrescenta uma ressonância fraca a \qty{700}{nm}?
  \item Na imagem de Lorentz, por que o azul é mais desviado que o vermelho pelo
        prisma?
  \item Numa tabela, dê, para a camada F, o aço, o cobre e
        o vidro em suas frequências de trabalho, a natureza de
        $\underline\gamma$ e o destino da onda.
\end{enumerate}
\end{problem}
