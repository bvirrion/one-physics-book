\chapter{Radiação térmica}\label{ch:b2:thermal-radiation}

Tudo o que é morno brilha. Uma mão, uma parede, um copo de água à
temperatura ambiente irradiam invisivelmente, em comprimentos de onda perto de \qty{10}{\micro m},
algumas centenas de watts por metro quadrado; aqueça um atiçador e a
\qty{600}{\celsius} ele brilha em vermelho-escuro, a \qty{1000}{\celsius} em laranja, a
\qty{2500}{\celsius} --- o filamento de uma lâmpada --- em branco; a superfície do
Sol a \qty{5800}{K} irradia em branco-amarelado, sessenta megawatts por
metro quadrado, e essa luz, após oito minutos de viagem, é o que
mantém a Terra a uma temperatura em que a água é líquida. A radiação é
o terceiro modo de transferência de calor e o único que atravessa o espaço
vazio; é também o lugar em que a física clássica falhou e Planck,
em 1900, teve de introduzir o quantum. Este capítulo enuncia a lei que ele
encontrou, dela extrai os dois resultados que se usam a toda hora ---
o deslocamento de Wien e a lei de Stefan--Boltzmann --- e aplica-os
às trocas entre corpos, ao Sol e à Terra, e ao
modelo de uma camada do \omterm{prop:b2:thermal-radiation:greenhouse}{efeito estufa} que fixa nosso clima.

\section{A radiação térmica e o corpo negro}

\begin{definition}[Radiação térmica, absortividade, corpo negro]\label{def:b2:thermal-radiation:blackbody}
Todo corpo à temperatura $T$ emite radiação eletromagnética a partir da
agitação térmica de suas cargas: sua \emph{radiação
térmica}\index{radiação térmica}. Sua superfície também recebe
radiação do ambiente e absorve uma fração $\alpha$ dela
(a \emph{absortividade}, entre $0$ e $1$, em geral dependente do
comprimento de onda), refletindo ou transmitindo o resto. Um \emph{corpo
negro}\index{corpo negro} absorve tudo ($\alpha = 1$ em todo
comprimento de onda). Um pequeno furo na parede de uma cavidade fechada é o
corpo negro prático: a radiação que entra é absorvida pelas paredes
antes de conseguir achar a saída.
\end{definition}

\begin{proposition}[Radiação de equilíbrio e lei de Kirchhoff]\label{prop:b2:thermal-radiation:kirchhoff}
Dentro de uma cavidade fechada cujas paredes estão a $T$, a radiação está em
equilíbrio com as paredes e é \emph{universal}: isotrópica,
não polarizada, com uma densidade espectral de energia $u_\nu(T)$ (energia por unidade
de volume e de frequência) que depende apenas de $T$ e $\nu$, e não do
material ou da forma da cavidade. A radiação que sai de um pequeno furo
numa cavidade dessas é a \emph{radiação de \omterm{def:b2:thermal-radiation:blackbody}{corpo negro}} a $T$. Uma superfície que
absorve a fração $\alpha_\nu$ da radiação que recebe na
frequência $\nu$ emite, nessa frequência, a fração
$\varepsilon_\nu = \alpha_\nu$ do que um \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} à mesma
temperatura emitiria (sua \emph{emissividade}\index{emissividade}):
\emph{bons absorvedores são bons emissores}.
\end{proposition}

\begin{proof}
Universalidade: ligue duas cavidades à mesma $T$ por um furo com um
filtro que deixe passar uma única frequência; se as densidades diferissem, a energia
fluiria de uma para a outra a temperaturas iguais, violando a segunda
lei. Kirchhoff: coloque a superfície na cavidade à sua própria $T$; em
equilíbrio ela deve emitir em cada frequência exatamente o que absorve,
$\varepsilon_\nu u_\nu = \alpha_\nu u_\nu$.
\end{proof}

\section{A lei de Planck}

\begin{theorem}[Lei de Planck]\label{thm:b2:thermal-radiation:planck}
A densidade espectral de energia da radiação de \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} à temperatura
$T$ vale
\[
u_\nu(T) = \frac{8\pi h\nu^3}{c^3}\,\frac{1}{\eu^{h\nu/k_BT} - 1},
\]
e a potência irradiada por uma superfície negra por unidade de área e por unidade
de comprimento de onda (sua \emph{exitância espectral}\index{exitância espectral}) vale
\[
M_\lambda(T) = \frac{2\pi hc^2}{\lambda^5}\,\frac{1}{\eu^{hc/\lambda k_BT} - 1}.
\]
Dois limites: para $h\nu \ll k_BT$, $u_\nu \approx 8\pi\nu^2 k_BT/c^3$
(Rayleigh--Jeans: cada modo da cavidade carrega a energia clássica
$k_BT$ --- uma lei cuja integral diverge, a ``catástrofe
ultravioleta''); para $h\nu \gg k_BT$, $u_\nu \approx (8\pi h\nu^3/c^3)
\eu^{-h\nu/k_BT}$ (Wien: o fator de Boltzmann de um fóton de energia
$h\nu$, \cref{ch:b2:boltzmann-factor}).
\end{theorem}

\begin{proof}
Admitida: a contagem dos modos da cavidade ($8\pi\nu^2/c^3$ por unidade de
volume e de frequência, duas polarizações) pertence ao volume do terceiro ano,
e a energia média de um modo, $h\nu/(\eu^{h\nu/k_BT} - 1)$ em vez do
clássico $k_BT$, é a hipótese quântica: a energia é trocada com
o modo em quanta $h\nu$, de modo que os modos com $h\nu \gg k_BT$ ficam
congelados. A relação $M_\lambda\,\dd\lambda = (c/4)\,u_\nu\,\dd\nu$ é o
fator geométrico calculado no \cref{thm:b2:thermal-radiation:stefan}.
\end{proof}

\begin{proposition}[Lei do deslocamento de Wien]\label{prop:b2:thermal-radiation:wien}
A \omterm{thm:b2:thermal-radiation:planck}{exitância espectral} $M_\lambda$ tem seu máximo em
\[
\lambda_{\max}\,T = \frac{hc}{4.965\,k_B} = \qty{2.90e-3}{m.K} .
\]
Um corpo a \qty{300}{K} irradia sobretudo perto de \qty{10}{\micro m}; o Sol
(\qty{5800}{K}) perto de \qty{0.5}{\micro m}, no verde, onde o olho é
mais sensível --- não por acaso.
\end{proposition}

\begin{proof}
Com $x = hc/\lambda k_BT$, $M_\lambda \propto x^5/(\eu^x - 1)$; $\dd/\dd x = 0$
dá $5(\eu^x - 1) = x\eu^x$, ou seja, $x = 5(1 - \eu^{-x})$, cuja raiz
não nula é $x = 4.965$.
\end{proof}

\begin{theorem}[Lei de Stefan--Boltzmann]\label{thm:b2:thermal-radiation:stefan}
Uma superfície negra a $T$ irradia, por unidade de área, a potência total
\[
M = \int_0^\infty M_\lambda\,\dd\lambda = \sigma T^4, \qquad
\sigma = \frac{2\pi^5k_B^4}{15h^3c^2} = \qty{5.67e-8}{W/m^2/K^4} ;
\]
a densidade de energia da radiação de equilíbrio vale $u = (4\sigma/c)T^4$
e sua pressão $P = u/3$. Uma superfície cinzenta de \omterm{prop:b2:thermal-radiation:kirchhoff}{emissividade} $\varepsilon$
irradia $\varepsilon\sigma T^4$.
\end{theorem}

\begin{proof}
Integre a densidade de Planck: $u = \int u_\nu\,\dd\nu = (8\pi h/c^3)(k_BT/h)^4
\int_0^\infty x^3\,\dd x/(\eu^x - 1)$ e a integral vale $\pi^4/15$
(admitido da análise): $u = (8\pi^5k_B^4/15h^3c^3)\,T^4$. A potência
que sai de um furo de área unitária na parede da cavidade é $cu/4$: a
radiação é isotrópica, os fótons que se dirigem ao furo dentro do
ângulo sólido $\dd\Omega$ sob o ângulo $\theta$ em relação à sua normal carregam o
fluxo $c\,u\,(\dd\Omega/4\pi)\cos\theta$, e $\int\cos\theta\,\dd\Omega/4\pi$
sobre o hemisfério exterior vale $1/4$. Logo $M = cu/4 = \sigma T^4$.
A pressão, como para todo gás isotrópico de partículas que se movem a $c$, vale
$u/3$.
\end{proof}

\begin{example}[Ordens de grandeza]\label{ex:b2:thermal-radiation:orders}
A \qty{300}{K}: $\sigma T^4 = \qty{460}{W/m^2}$ --- uma pessoa de
\qty{1.7}{m^2} irradia \qty{800}{W}, mas recebe quase outro tanto das
paredes a \qty{293}{K}; a perda líquida, $\sigma A(T^4 - T_0^4) \approx
\qty{100}{W}$, é toda a produção metabólica do corpo, razão pela qual são
necessárias roupas numa sala que parece amena. A superfície do Sol: $\sigma
(5800)^4 = \qty{6.4e7}{W/m^2}$, total $3.9 \times 10^{26}\,\unit{W}$. O
filamento de uma lâmpada a \qty{2800}{K}: \qty{3.5e6}{W/m^2}, \qty{60}{W} de meio
centímetro quadrado de tungstênio. O fundo cósmico, \qty{2.7}{K}:
pico em \qty{1}{mm}, $\qty{3e-6}{W/m^2}$, e $400$ fótons por centímetro
cúbico do universo.
\end{example}

\section{Trocas radiativas}

\begin{proposition}[Troca líquida e coeficiente radiativo]\label{prop:b2:thermal-radiation:exchange}
Um pequeno corpo cinzento (\omterm{prop:b2:thermal-radiation:kirchhoff}{emissividade} $\varepsilon$, área $S$, temperatura $T$)
dentro de um ambiente amplo a $T_0$ perde a potência líquida
\[
P = \varepsilon\sigma S\,(T^4 - T_0^4)
\ \approx\ 4\varepsilon\sigma T_0^3\,S\,(T - T_0) = h_{\text{r}}S(T - T_0)
\quad\text{para } |T - T_0| \ll T_0 ,
\]
onde $h_{\text{r}} = 4\varepsilon\sigma T_0^3 \approx \qty{6}{W/m^2/K}$ à
temperatura ambiente --- comparável à convecção natural: numa sala, a
radiação leva cerca de metade do calor que sai de um corpo ou de um radiador.
Entre duas grandes placas negras paralelas a densidade de fluxo líquida vale
$\sigma(T_1^4 - T_2^4)$; para placas cinzentas de \omterm{prop:b2:thermal-radiation:kirchhoff}{emissividades} $\varepsilon_1$,
$\varepsilon_2$ ela vale $\sigma(T_1^4 - T_2^4)/(1/\varepsilon_1 + 1/\varepsilon_2
- 1)$, e cada fina \emph{blindagem de radiação} interposta a divide
ainda mais --- o princípio da garrafa térmica e da folha de ouro dos
satélites.
\end{proposition}

\begin{proof}
O corpo emite $\varepsilon\sigma T^4$ e absorve a fração $\alpha =
\varepsilon$ da radiação negra $\sigma T_0^4$ que preenche o recinto
(Kirchhoff). Linearize $T^4 - T_0^4 \approx 4T_0^3(T - T_0)$. Para as
placas cinzentas, some as múltiplas reflexões (uma série geométrica).
\end{proof}

\begin{example}[Geada sob um céu limpo, e a câmera térmica]\label{ex:b2:thermal-radiation:frost}
Numa noite limpa o céu irradia como um corpo a cerca de \qty{250}{K}
(a alta atmosfera seca), enquanto o ar junto ao solo está a
\qty{278}{K}. Uma folha ou o teto de um carro ($\varepsilon = 0.95$) equilibra sua
perda radiativa para o céu contra a convecção vinda do ar ($h =
\qty{5}{W/m^2/K}$): $\varepsilon\sigma(T^4 - 250^4) = h(278 - T)$ dá $T
\approx \qty{266}{K}$ --- geada a \qty{-7}{\celsius} sobre a grama enquanto
o termômetro marca $+5$. Uma nuvem (que irradia a \qty{275}{K}) eleva $T$
a \qty{277}{K}: sem geada. Uma câmera térmica mede a radiância perto de
\qty{10}{\micro m} e a converte em temperatura supondo
$\varepsilon \approx 0.95$; um metal polido ($\varepsilon = 0.1$) sobretudo
reflete a sala e aparece à temperatura da sala, qualquer que seja a
sua --- daí a fita preta que os termografistas colam nas peças brilhantes.
\end{example}

\section{O Sol, a Terra e o efeito estufa}

\begin{proposition}[Constante solar e temperatura efetiva]\label{prop:b2:thermal-radiation:earth}
O Sol (raio $R_\odot$, temperatura de superfície $T_\odot$) entrega à
distância $d$ a densidade de fluxo $S = \sigma T_\odot^4 (R_\odot/d)^2 =
\qty{1361}{W/m^2}$ acima da atmosfera terrestre (a \emph{constante
solar}\index{constante solar}). Um planeta de albedo $A$ (a fração
refletida) absorve $(1 - A)S\pi R^2$ e, em regime estacionário, a irradia
de toda a sua superfície $4\pi R^2$ como um \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} à sua \emph{temperatura
efetiva}\index{temperatura efetiva}
\[
T_{\text{e}} = \Big(\frac{(1 - A)S}{4\sigma}\Big)^{1/4} = \qty{255}{K}
\quad\text{para a Terra } (A = 0.30) .
\]
\end{proposition}

\begin{proof}
Conservação da energia: absorvido $= $ emitido, com a \omterm{prop:b2:laser:gain}{seção de choque} $\pi R^2$
para o feixe interceptado e a esfera inteira para a emissão (dia e
noite, distribuídos pela rotação e pela circulação).
\end{proof}

\begin{proposition}[Modelo de efeito estufa de uma camada]\label{prop:b2:thermal-radiation:greenhouse}
Envolva o planeta com uma fina camada de atmosfera transparente
à luz do Sol mas que absorve todo o infravermelho emitido pela superfície e
irradia como um \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} a $T_{\text{a}}$ tanto para cima como para baixo. Em
regime estacionário,
\[
T_{\text{a}} = T_{\text{e}}, \qquad
T_{\text{s}} = 2^{1/4}\,T_{\text{e}} = \qty{303}{K} .
\]
Se a camada absorve apenas a fração $\varepsilon$ do infravermelho,
$T_{\text{s}}^4 = T_{\text{e}}^4/(1 - \varepsilon/2)$; os \qty{288}{K}
observados correspondem a $\varepsilon \approx 0.78$. A superfície é
mais quente que a \omterm{prop:b2:thermal-radiation:earth}{temperatura efetiva} porque recebe, além
do Sol, o infravermelho que a atmosfera devolve para baixo --- o
\emph{efeito estufa}\index{efeito estufa}, de \qty{33}{K} na Terra,
sem o qual os oceanos estariam congelados.
\end{proposition}

\begin{proof}
Balanço da camada: absorve $\sigma T_{\text{s}}^4$, emite $2\sigma
T_{\text{a}}^4$. Balanço do planeta visto do espaço: emite $\sigma
T_{\text{a}}^4 = (1 - A)S/4 = \sigma T_{\text{e}}^4$. Balanço da
superfície: recebe $(1 - A)S/4 + \sigma T_{\text{a}}^4 = 2\sigma T_{\text{e}}^4$,
emite $\sigma T_{\text{s}}^4$. Para a camada parcial, substitua o
infravermelho absorvido e emitido por $\varepsilon\sigma T_{\text{s}}^4$ e
$\varepsilon\sigma T_{\text{a}}^4$ e deixe a fração $1 - \varepsilon$
da radiação da superfície escapar diretamente.
\end{proof}

\begin{remark}[Por que a estufa é seletiva]\label{rem:b2:thermal-radiation:selective}
A luz do Sol tem seu pico em \qty{0.5}{\micro m}, a radiação da Terra em
\qty{10}{\micro m}: os dois espectros quase não se superpõem, de modo que um gás pode ser
transparente ao primeiro e absorvente para o segundo. O nitrogênio e o
oxigênio não absorvem nem um nem outro; o vapor de água, o dióxido de carbono, o metano e o ozônio
têm bandas de vibração no infravermelho e são os gases de \omterm{prop:b2:thermal-radiation:greenhouse}{efeito estufa}. A
janela entre \qty{8}{} e \qty{13}{\micro m} é aquela pela qual a câmera térmica
olha, aquela a que se referem os \qty{250}{K} do céu noturno, e aquela que
as tintas de resfriamento radiativo exploram. O vidro de uma estufa de verdade faz
o mesmo (transparente a \qty{0.5}{\micro m}, opaco a \qty{10}{\micro m}),
mas sobretudo ele bloqueia a convecção --- o nome é um pouco injusto.
\end{remark}

\begin{method}[Estimativas de radiação]\label{meth:b2:thermal-radiation:method}
(1) $\lambda_{\max} = \qty{2900}{\micro m.K}/T$ diz onde está o espectro;
(2) $\varepsilon\sigma T^4$ por unidade de área; (3) trocas: emitido
menos absorvido, linearizado como $h_{\text{r}} = 4\varepsilon\sigma T^3$ perto da
temperatura ambiente; (4) um planeta: $(1 - A)S/4 = \sigma T_{\text{e}}^4$, depois
as camadas; (5) compare com a condução e a convecção --- a radiação
vence em alta temperatura ($T^4$) e no vácuo; (6) atenção às \omterm{prop:b2:thermal-radiation:kirchhoff}{emissividades}:
metais brilhantes \qty{0.05}{}, tintas, pele, água e solo $\approx 0.9$--$0.98$
no infravermelho, seja qual for sua cor visível.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:1}
$\lambda_{\max}$ e $\sigma T^4$ para: o Sol (\qty{5800}{K}), um ser humano
(\qty{310}{K}), o filamento de uma lâmpada (\qty{2800}{K}), o fundo cósmico
(\qty{2.73}{K}), o nitrogênio líquido (\qty{77}{K}), um atiçador em brasa
(\qty{900}{K}). Quais são visíveis?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:2}
A partir de $T_\odot = \qty{5772}{K}$, $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.496e11}{m}$:
a \omterm{prop:b2:thermal-radiation:earth}{constante solar}, a luminosidade do Sol, a potência interceptada pela
Terra ($R = \qty{6.37e6}{m}$), e a mesma dividida por $8 \times 10^9$
pessoas. Compare com o consumo mundial de potência (\qty{2e13}{W}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:3}
Uma pessoa: pele a \qty{306}{K}, $\varepsilon = 0.98$, \qty{1.7}{m^2}, numa
sala a \qty{293}{K}. Potência emitida, potência absorvida, perda líquida; o
coeficiente radiativo $h_{\text{r}}$; compare com os \qty{100}{W} do
metabolismo e conclua a respeito das roupas e de uma sala a \qty{20}{\celsius}
com paredes frias.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:thermal-radiation:4}
Uma lâmpada de \qty{60}{W}: filamento de tungstênio a \qty{2800}{K}, $\varepsilon = 0.35$.
Área irradiante; comprimento de um fio de \qty{50}{\micro m} com essa área;
cerca de $8\%$ da potência é visível: potência luminosa; um LED que dê a
mesma luz com $30\%$ de eficiência.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:5}
\emph{Os dois limites.} (a) Desenvolva o $u_\nu$ de Planck para $h\nu \ll k_BT$ e
identifique a energia por modo. (b) Mostre que a lei de Rayleigh--Jeans
dá uma energia total infinita. (c) A \qty{300}{K}, a frequência em que
$h\nu = k_BT$; conclua que a \omterm{def:b2:thermal-radiation:blackbody}{radiação térmica} de rádio e de micro-ondas é
clássica. (d) No limite de Wien, interprete $\eu^{-h\nu/k_BT}$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:6}
\emph{Wien.} (a) Deduza $x = 5(1 - \eu^{-x})$ e resolva-a numericamente
com três algarismos. (b) Valor de $\lambda_{\max}T$. (c) Mostre que $u_\nu$
tem seu máximo em $h\nu = 2.82\,k_BT$, e que isso não corresponde a
$c/\lambda_{\max}$: por quê? (d) Temperatura de uma estrela cujo espectro
tem pico em \qty{290}{nm}; sua cor ao olho.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:7}
\emph{Stefan.} (a) Efetue a integração da lei de Planck, dado
$\int_0^\infty x^3\dd x/(\eu^x - 1) = \pi^4/15$, e o valor de $\sigma$.
(b) Justifique o fator $c/4$ entre a densidade e a exitância. (c)
Dado $\int_0^\infty x^2\dd x/(\eu^x - 1) = 2.404$, o número de fótons
por unidade de volume a $T$, e a energia média do fóton em unidades de $k_BT$.
(d) Fótons por centímetro cúbico a \qty{300}{K} e a \qty{2.73}{K}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:8}
\emph{Blindagens.} Duas grandes placas negras a $T_1$, $T_2$. (a) Fluxo líquido.
(b) Uma folha negra fina é colocada entre elas: sua temperatura e o
novo fluxo. (c) $n$ folhas. (d) Placas cinzentas de \omterm{prop:b2:thermal-radiation:kirchhoff}{emissividade} $\varepsilon$:
mostre que o fluxo é $\sigma(T_1^4 - T_2^4)/(2/\varepsilon - 1)$ e calcule-o
para superfícies polidas ($\varepsilon = 0.05$) a \qty{300}{K} e
\qty{77}{K} --- a garrafa térmica.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:thermal-radiation:9}
\emph{Geada noturna.} Superfície com $\varepsilon = 0.95$, céu a \qty{250}{K}, ar a
\qty{278}{K} com $h = \qty{5}{W/m^2/K}$. (a) Escreva o balanço e resolva-o
para a temperatura da superfície. (b) Sob uma nuvem a \qty{275}{K}. (c) Por que
o vento impede a geada? (d) Por que um carro estacionado sob uma árvore escapa
dela?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:10}
\emph{Planetas.} (a) \omterm{prop:b2:thermal-radiation:earth}{Temperaturas efetivas} de Vênus ($S = \qty{2600}{W/m^2}$,
$A = 0.75$), da Terra e de Marte ($S = \qty{590}{W/m^2}$, $A = 0.25$). (b) Suas
temperaturas de superfície são \qty{737}{K}, \qty{288}{K}, \qty{215}{K}:
aquecimento de estufa de cada um. (c) No modelo com $n$ camadas opacas,
mostre que $T_{\text{s}}^4 = (n + 1)T_{\text{e}}^4$; quantas camadas para Vênus?
(d) A Lua: $A = 0.12$, sem atmosfera, rotação lenta --- temperatura
do ponto subsolar, e por que o lado noturno cai a \qty{100}{K}.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:11}
\emph{Sensibilidade climática sem \omterm{def:b2:feedback-oscillators:loop}{realimentações}.} (a) Linearize o
balanço planetário: uma pequena forçante extra $\Delta F$ (em \unit{W/m^2}) no
topo da atmosfera eleva a \omterm{prop:b2:thermal-radiation:earth}{temperatura efetiva} de $\Delta T
= \Delta F/4\sigma T_{\text{e}}^3$. Calcule $4\sigma T_{\text{e}}^3$. (b)
Dobrar o dióxido de carbono é uma forçante de cerca de \qty{3.7}{W/m^2}:
$\Delta T$ neste modelo. (c) A camada de mistura do oceano (\qty{50}{m} de
água) armazena o calor: constante de tempo da resposta. (d) Por que as
\omterm{def:b2:feedback-oscillators:loop}{realimentações} do vapor de água e do albedo do gelo tornam a sensibilidade real maior
que a de (b)?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:thermal-radiation:12}
\emph{Pirometria.} (a) No limite de Wien, mostre que a razão das
radiâncias em dois comprimentos de onda, $\lambda_1$ e $\lambda_2$, dá $T$
independentemente de uma \omterm{prop:b2:thermal-radiation:kirchhoff}{emissividade} cinzenta. (b) Uma barra de aço: $M_{0.65}/M_{0.9}
= 0.10$: temperatura. (c) Uma câmera térmica a \qty{10}{\micro m} supõe
$\varepsilon = 0.95$; ela olha para alumínio polido ($\varepsilon = 0.1$)
a \qty{350}{K} numa sala a \qty{293}{K}: no regime linearizado,
que temperatura ela indica? (d) Por que uma superfície pintada ao lado
do alumínio, à mesma \qty{350}{K}, é lida corretamente?
\end{exercise}

\section{Problema: o balanço de energia da Terra e a lâmpada incandescente}

\begin{problem}[{Problema de fim de semana --- corpos negros do Sol ao
filamento}]\label{pb:b2:thermal-radiation:1}
Dados: $\sigma = \qty{5.67e-8}{W/m^2/K^4}$, $\lambda_{\max}T = \qty{2.90e-3}{m.K}$,
$h = \qty{6.63e-34}{J.s}$, $k_B = \qty{1.38e-23}{J/K}$, $c = \qty{3.00e8}{m/s}$;
Sol: $R_\odot = \qty{6.96e8}{m}$, $d = \qty{1.50e11}{m}$; Terra: $R =
\qty{6.37e6}{m}$, albedo $0.30$.

\textbf{Parte I --- O Sol como \omterm{def:b2:thermal-radiation:blackbody}{corpo negro}.}
\begin{enumerate}
  \item O espectro solar tem pico em \qty{0.50}{\micro m}: temperatura
        da superfície.
  \item Exitância da superfície e luminosidade total.
  \item \omterm{prop:b2:thermal-radiation:earth}{Constante solar} na Terra; densidade de energia e pressão da
        luz do Sol ali.
  \item Energia média do fóton ($\approx 2.7\,k_BT$) e número de fótons
        solares por segundo e por metro quadrado na Terra.
  \item O Sol irradia a essa taxa há $4.5 \times 10^9$ anos:
        energia emitida; compare com $Mc^2$ para $M = \qty{2e30}{kg}$.
\end{enumerate}

\textbf{Parte II --- A Terra sem atmosfera.}
\begin{enumerate}[resume]
  \item Potência absorvida pela Terra; \omterm{prop:b2:thermal-radiation:earth}{temperatura efetiva}.
  \item Por que o balanço divide por $4$? Qual seria a
        temperatura no ponto subsolar de uma Terra sem rotação e sem
        atmosfera ($A = 0.30$)?
  \item A Lua ($A = 0.12$): temperatura subsolar; e por que seu lado
        noturno cai a \qty{100}{K} (pense no calor armazenado no
        regolito durante o dia de duas semanas, \qty{1e6}{J/m^2/K} efetivos).
  \item Comprimento de onda do pico de emissão da Terra; compare com o
        do Sol.
  \item Se mais gelo ou mais nuvens elevassem o albedo a $0.35$, de quanto
        cairia a \omterm{prop:b2:thermal-radiation:earth}{temperatura efetiva}?
\end{enumerate}

\textbf{Parte III --- O \omterm{prop:b2:thermal-radiation:greenhouse}{efeito estufa}.}
\begin{enumerate}[resume]
  \item Escreva os três balanços (espaço, camada, superfície) do
        modelo de uma camada e obtenha $T_{\text{s}} = 2^{1/4}T_{\text{e}}$.
  \item Com uma camada que absorve a fração $\varepsilon$ do
        infravermelho: $T_{\text{s}}(\varepsilon)$; o valor de $\varepsilon$
        que dá \qty{288}{K}.
  \item Que gases absorvem, quais não absorvem, e em que banda de
        comprimentos de onda; o que é a janela atmosférica?
  \item O infravermelho devolvido ao solo pela atmosfera,
        em \unit{W/m^2}, no modelo com $\varepsilon = 0.78$; compare com
        a luz do Sol absorvida.
  \item Uma forçante de \qty{3.7}{W/m^2} (dióxido de carbono dobrado):
        variação de temperatura $\Delta T = \Delta F/4\sigma T_{\text{e}}^3$
        sem \omterm{def:b2:feedback-oscillators:loop}{realimentações}.
  \item Capacidade térmica de uma camada de mistura oceânica de \qty{50}{m} por metro
        quadrado e a constante de tempo da resposta.
  \item Por que as nuvens ao mesmo tempo resfriam (albedo) e aquecem (infravermelho) a
        superfície; que efeito vence à noite?
\end{enumerate}

\textbf{Parte IV --- A lâmpada incandescente.}
Uma lâmpada de \qty{60}{W} tem um filamento de tungstênio a \qty{2800}{K},
$\varepsilon = 0.35$; a fração da potência de um \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} que cai
no visível ($0.4$--\qty{0.7}{\micro m}) vale $0.08$ a \qty{2800}{K} e $0.14$
a \qty{3200}{K}.
\begin{enumerate}[resume]
  \item Área irradiante do filamento; comprimento de um fio de \qty{50}{\micro m}
        que a tenha; por que ele é enrolado?
  \item Comprimento de onda de pico; potência visível; eficiência luminosa.
  \item Resistência do filamento em funcionamento a \qty{230}{V}; a
        resistividade do tungstênio é cerca de dez vezes menor à temperatura
        ambiente: corrente ao ligar, e uma consequência.
  \item A \qty{3200}{K}: a área necessária para os mesmos \qty{60}{W} e
        a potência visível; por que não fazer toda lâmpada funcionar a \qty{3200}{K}
        (a taxa de evaporação do tungstênio praticamente dobra a cada
        \qty{50}{K})?
  \item O que faz o ciclo do halogênio, e por que ele permite um filamento
        mais quente?
  \item Para onde vai o resto dos \qty{60}{W}, e por que o vidro da
        lâmpada chega a \qty{150}{\celsius}?
  \item Tempo de resfriamento do filamento depois de desligado: estime-o a partir
        de sua massa (\qty{20}{mg}, $c = \qty{140}{J/kg/K}$) e de sua potência
        irradiada.
  \item Resuma: o que a temperatura de um \omterm{def:b2:thermal-radiation:blackbody}{corpo negro} fixa, e o que
        o filamento, o Sol e a Terra têm em comum.
\end{enumerate}
\end{problem}
