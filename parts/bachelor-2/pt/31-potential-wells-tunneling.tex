\chapter{Poços de potencial, barreiras e efeito túnel}\label{ch:b2:potential-wells-tunneling}

Uma bola numa tigela vai e vem com a energia que lhe deram;
um elétron num átomo, um núcleon num núcleo, um elétron num
cristal de tamanho nanométrico só podem ter certas energias, e a mais baixa
delas não é zero. Uma bola não pode atravessar uma colina mais alta do que sua energia
permite; uma partícula $\alpha$ deixa um núcleo através de uma muralha de Coulomb que
jamais poderia escalar, a agulha de um microscópio de varredura por tunelamento puxa
uma corrente através de um vão de vácuo, e o átomo de nitrogênio da amônia
oscila através do plano de seus três hidrogênios vinte e quatro bilhões
de vezes por segundo. Este capítulo resolve a equação de Schrödinger nos
potenciais mais simples --- poços de paredes infinitas e finitas, um degrau, uma
barreira, um poço duplo --- e neles encontra os dois grandes fatos
quânticos que a mecânica clássica não consegue produzir: a \emph{quantização}
das energias ligadas e o \emph{tunelamento} através de regiões classicamente
proibidas, com sua sensibilidade exponencial à largura e à altura
da barreira, razão pela qual ele pode servir de régua de picômetros e pela qual
os tempos de vida radioativos cobrem trinta ordens de grandeza.

\section{O poço infinito}

\begin{proposition}[Poço quadrado infinito]\label{prop:b2:potential-wells-tunneling:infinite}
Uma partícula confinada a $0 < x < L$ por paredes impenetráveis ($V = 0$
dentro, $V = \infty$ fora, de modo que $\varphi = 0$ nas paredes) tem os
\omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} e as energias
\[
\varphi_n(x) = \sqrt{\frac{2}{L}}\sin\frac{n\pi x}{L}, \qquad
E_n = \frac{n^2h^2}{8mL^2} = n^2E_1, \qquad n = 1, 2, 3, \dots
\]
As energias são \emph{quantizadas}\index{quantização da energia}, crescem
como $n^2$, e a mais baixa não é zero: a \emph{energia de
confinamento}\index{energia de confinamento} $E_1 = h^2/8mL^2$ é o preço da
localização ($\Delta x \sim L$ impõe $\Delta p \sim h/2L$). As
$\varphi_n$ são ortogonais, $\int\varphi_n\varphi_m\dd x = \delta_{nm}$, e qualquer
estado do poço é uma superposição delas.
\end{proposition}

\begin{proof}
Dentro, $\varphi'' = -k^2\varphi$ com $k^2 = 2mE/\hbar^2$: $\varphi = A\sin kx
+ B\cos kx$; $\varphi(0) = 0$ elimina $B$, e $\varphi(L) = 0$ exige $kL = n\pi$
--- as \omterm{prop:b2:waves-on-strings:modes}{ondas estacionárias} de uma corda, com $E = \hbar^2k^2/2m$.
A normalização dá $A = \sqrt{2/L}$; a ortogonalidade é a dos
senos.
\end{proof}

\begin{example}[Ordens de grandeza, e cores]\label{ex:b2:potential-wells-tunneling:orders}
$E_1 = h^2/8mL^2$: um elétron em \qty{1}{nm}, \qty{0.38}{eV}; em
\qty{0.1}{nm} (um átomo), \qty{38}{eV} --- a escala das energias atômicas; um
núcleon em \qty{5}{fm}, \qty{8}{MeV} --- a escala das energias nucleares; uma
bola de gude numa caixa, $10^{-63}\,\unit{J}$ --- ninguém vai notar. Um
nanocristal semicondutor de alguns nanômetros (um \emph{ponto
quântico}) confina seus elétrons: a \omterm{prop:b2:potential-wells-tunneling:infinite}{energia de confinamento} se soma ao
intervalo do cristal, de modo que quanto menor o ponto, mais azul a luz que ele emite ---
o mesmo material brilha em vermelho a \qty{6}{nm} e em verde a \qty{3}{nm},
sintonizado só pelo tamanho.
\end{example}

\section{O poço finito}

\begin{proposition}[Poço quadrado finito]\label{prop:b2:potential-wells-tunneling:finite}
Para $V = 0$ em $|x| < a$ e $V = V_0$ fora, um estado com $0 < E < V_0$
oscila dentro ($k = \sqrt{2mE}/\hbar$) e decai fora ($\kappa =
\sqrt{2m(V_0 - E)}/\hbar$): $\varphi \propto \eu^{-\kappa|x|}$ além das paredes.
Casar $\varphi$ e $\varphi'$ em $x = \pm a$ dá, para os estados pares e
ímpares,
\[
k\tan ka = \kappa \quad\text{(pares)}, \qquad
-k\cot ka = \kappa \quad\text{(ímpares)}, \qquad\text{com}\quad
(ka)^2 + (\kappa a)^2 = \frac{2mV_0a^2}{\hbar^2} \equiv R^2 .
\]
Graficamente, as soluções são as interseções das curvas $\eta =
\xi\tan\xi$, $\eta = -\xi\cot\xi$ com o círculo $\xi^2 + \eta^2 = R^2$
($\xi = ka$, $\eta = \kappa a$): há \emph{sempre ao menos um estado
ligado} (o estado fundamental par), e $1 + \lfloor 2R/\pi\rfloor$ ao todo.
A partícula \emph{penetra} na região proibida ao longo da profundidade
$1/\kappa$, e os níveis ficam abaixo dos do poço infinito de
mesma largura.
\end{proposition}

\begin{proof}
A solução par é $A\cos kx$ dentro e $B\eu^{-\kappa|x|}$ fora; a
continuidade de $\varphi'/\varphi$ em $x = a$ dá $-k\tan ka = -\kappa$. O mesmo
para a ímpar com $\sin$. O círculo é a definição de $k$ e
$\kappa$. Cada ramo de $\xi\tan\xi$ que começa em $\xi = n\pi$ e de
$-\xi\cot\xi$ que começa em $(n + \tfrac12)\pi$ encontra o círculo uma vez se
começar dentro dele: daí a contagem.
\end{proof}

\begin{example}[Um elétron num poço de um nanômetro]\label{ex:b2:potential-wells-tunneling:finite}
$V_0 = \qty{1}{eV}$, largura $2a = \qty{1}{nm}$: $R = a\sqrt{2mV_0}/\hbar = 0.5 \times
10^{-9} \times 5.1 \times 10^9 = 2.6$, logo $1 + \lfloor 2R/\pi\rfloor = 2$ estados
ligados; o estado fundamental fica perto de \qty{0.26}{eV} em vez dos
\qty{0.38}{eV} do poço infinito, e vaza \qty{0.2}{nm} para dentro das paredes
($1/\kappa = \hbar/\sqrt{2m(V_0 - E)}$). Poços desses, crescidos como camadas de
semicondutores de alguns nanômetros de espessura, são o coração dos
diodos laser do \cref{ch:b2:laser}.
\end{example}

\section{Degrau e barreira: o efeito túnel}

\begin{proposition}[Degrau de potencial]\label{prop:b2:potential-wells-tunneling:step}
Uma partícula de energia $E$ que vem de $x < 0$ sobre o degrau $V = V_0$
para $x > 0$:
\begin{itemize}
\item $E > V_0$: $\varphi = \eu^{\iu k_1x} + r\eu^{-\iu k_1x}$ à esquerda, $t\eu^{\iu
      k_2x}$ à direita, com $k_{1,2} = \sqrt{2m(E - V_{0,\text{esq/dir}})}/
      \hbar$; a continuidade de $\varphi$ e $\varphi'$ dá $r = (k_1 - k_2)/(k_1
      + k_2)$ e a probabilidade de reflexão (razão de correntes) $R = r^2$,
      $T = 1 - R = 4k_1k_2/(k_1 + k_2)^2$ --- uma partícula \emph{pode ser
      refletida por um degrau que teria energia para subir}, exatamente como uma onda numa
      corda por uma mudança de impedância (\cref{ch:b2:wave-interfaces});
\item $E < V_0$: à direita $\varphi = t\eu^{-\kappa x}$, $\kappa = \sqrt{2m(V_0
      - E)}/\hbar$ --- uma \emph{onda evanescente}\index{onda evanescente
      (quântica)}; $|r| = 1$ (reflexão total, com um desvio de fase), nenhuma
      corrente entra no degrau, mas a partícula é encontrada ali com uma
      probabilidade que decai ao longo de $1/\kappa$.
\end{itemize}
\end{proposition}

\begin{proof}
Escreva as equações de continuidade em $x = 0$ e resolva; para as correntes
use $j = (|A|^2 - |B|^2)\hbar k/m$ (\cref{ch:b2:schrodinger-wave-functions}).
Para $E < V_0$, $r = (k - \iu\kappa)/(k + \iu\kappa)$, de módulo um.
\end{proof}

\begin{theorem}[Tunelamento através de uma barreira]\label{thm:b2:potential-wells-tunneling:tunnel}
Uma partícula de energia $E < V_0$ que encontra uma barreira retangular de altura
$V_0$ e largura $a$ é transmitida com a probabilidade
\[
T = \Big[1 + \frac{V_0^2\sinh^2(\kappa a)}{4E(V_0 - E)}\Big]^{-1}
\ \approx\ 16\,\frac{E}{V_0}\Big(1 - \frac{E}{V_0}\Big)\eu^{-2\kappa a}
\quad (\kappa a \gg 1), \qquad
\kappa = \frac{\sqrt{2m(V_0 - E)}}{\hbar} .
\]
Este é o \emph{efeito túnel}\index{efeito túnel}: classicamente
impossível, quanticamente exponencialmente pequeno --- e
exponencialmente sensível à largura $a$, à altura $V_0 - E$ e à
massa $m$. Para um elétron \qty{1}{eV} abaixo do topo, $\kappa = \qty{5.1}{nm^{-1}}$:
$T \sim 10^{-2}$ para $a = \qty{0.5}{nm}$, $10^{-4}$ para \qty{1}{nm}, $10^{-9}$ para
\qty{2}{nm}; para um próton, $\kappa$ é $43$ vezes maior e nada passa
com essas larguras.
\end{theorem}

\begin{proof}
$\varphi = \eu^{\iu kx} + r\eu^{-\iu kx}$ para $x < 0$, $A\eu^{\kappa x} + B\eu^{-\kappa x}$
dentro, $t\eu^{\iu kx}$ para $x > a$; quatro condições de continuidade (de $\varphi$
e $\varphi'$ em $0$ e em $a$) para quatro incógnitas; eliminando $A$, $B$, $r$
obtém-se $1/|t|^2 = 1 + (k^2 + \kappa^2)^2\sinh^2(\kappa a)/4k^2\kappa^2$, que é
a fórmula com $k^2 = 2mE/\hbar^2$, $\kappa^2 = 2m(V_0 - E)/\hbar^2$. Para
$\kappa a \gg 1$, $\sinh\kappa a \approx \eu^{\kappa a}/2$. Para uma barreira de
forma arbitrária o expoente se torna $2\int\kappa(x)\dd x$ ao longo da
região proibida (o fator de Gamow, admitido).
\end{proof}

\begin{example}[O microscópio de varredura por tunelamento]\label{ex:b2:potential-wells-tunneling:stm}
Uma ponta metálica afiada é aproximada até $d \approx \qty{0.5}{nm}$ de uma
superfície condutora e aplica-se uma pequena tensão: os elétrons tunelam através
do vão de vácuo, cuja altura de barreira é a função trabalho $\Phi \approx
\qty{4.5}{eV}$, de modo que $\kappa = \sqrt{2m\Phi}/\hbar = \qty{1.1e10}{m^{-1}}$ e
a corrente $I \propto \eu^{-2\kappa d}$ muda por um fator $\eu^{2.2} \approx
10$ a cada \qty{0.1}{nm}. Uma malha de \omterm{def:b2:feedback-oscillators:loop}{realimentação} move a ponta para manter $I$
constante, e a altura da ponta, registrada durante a varredura, desenha a superfície
com uma resolução vertical de um picômetro --- e lateralmente átomo a
átomo, porque o último átomo da ponta, sendo o mais próximo, carrega a maior parte
da corrente (Binnig e Rohrer, 1981).
\end{example}

\begin{example}[Decaimento alfa]\label{ex:b2:potential-wells-tunneling:alpha}
Uma partícula $\alpha$ de \qty{5}{MeV} dentro de um núcleo pesado enfrenta a
barreira de Coulomb da carga restante $Ze$, de dezenas de MeV de altura na
superfície nuclear e que se estende até o raio $b = 2Ze^2/4\pi\varepsilon_0E
\approx \qty{50}{fm}$, onde $V = E$. O fator de Gamow $2\int\kappa\,\dd r$ é
da ordem de $80$: $T \sim \eu^{-80} \sim 10^{-35}$. A partícula bate na parede
umas $10^{21}$ vezes por segundo, de modo que escapa a uma taxa $\sim 10^{-14}
\,\unit{s^{-1}}$: um tempo de vida de um milhão de anos. Como o expoente
varia como $Z/\sqrt E$, uma variação de $\qty{2}{MeV}$ em $E$ desloca o tempo de vida
de vinte ordens de grandeza --- a lei de Geiger--Nuttall, de
microssegundos à idade do universo (Gamow, 1928: a primeira
aplicação do tunelamento).
\end{example}

\section{O poço duplo}

\begin{proposition}[Desdobramento por tunelamento]\label{prop:b2:potential-wells-tunneling:double}
Dois poços idênticos separados por uma barreira: se a barreira fosse
impenetrável cada poço teria o mesmo nível fundamental $E_0$, duas vezes.
O tunelamento os acopla, e os dois estados mais baixos do poço duplo
são as combinações \emph{simétrica} e \emph{antissimétrica},
$\varphi_\pm \approx (\varphi_{\text{L}} \pm \varphi_{\text{R}})/\sqrt2$, com energias
$E_0 \mp \Delta/2$: o nível é \emph{desdobrado} de uma quantidade $\Delta \propto
\eu^{-\kappa a}$ (a amplitude de tunelamento, não seu quadrado). Uma partícula
colocada no poço esquerdo, $\psi = (\varphi_+ + \varphi_-)/\sqrt2$, não é
estacionária: ela oscila entre os poços na frequência
$\nu = \Delta/h$ --- as \emph{oscilações de tunelamento}, os batimentos de dois níveis do
\cref{ch:b2:schrodinger-wave-functions}.
\end{proposition}

\begin{proof}
A combinação simétrica não tem nó sob a barreira e tem uma curvatura
menor, e portanto uma energia menor; a antissimétrica tem um nó e
fica mais alta; o desdobramento é proporcional à superposição das
funções localizadas sob a barreira, $\eu^{-\kappa a}$. O batimento é
a superposição de dois níveis já calculada.
\end{proof}

\begin{example}[A amônia]\label{ex:b2:potential-wells-tunneling:ammonia}
Na NH$_3$ o átomo de nitrogênio fica de um lado do plano dos três
hidrogênios ou do outro --- um poço duplo com uma barreira de cerca de
\qty{0.25}{eV}. O desdobramento por tunelamento do nível fundamental vale $\Delta =
\qty{1e-4}{eV}$: $\nu = \Delta/h = \qty{24}{GHz}$, a frequência de
\emph{inversão}, a transição do primeiro maser e do primeiro relógio
atômico (1949). Troque o hidrogênio por deutério e a molécula, mais pesada,
tunela menos: \qty{1.6}{GHz}. Em pirâmides mais pesadas (PH$_3$, AsH$_3$) o
desdobramento desaba para quilohertz e menos: a molécula fica de seu
lado por anos --- razão pela qual moléculas canhotas e destras existem
afinal, e pela qual o açúcar não racemiza na prateleira.
\end{example}

\begin{method}[Poços e barreiras]\label{meth:b2:potential-wells-tunneling:method}
(1) Escreva $\varphi$ região por região: $\eu^{\pm\iu kx}$ onde $E > V$,
$\eu^{\pm\kappa x}$ onde $E < V$, com $k, \kappa = \sqrt{2m|E - V|}/\hbar$.
(2) Case $\varphi$ e $\varphi'$ em cada fronteira (numa parede infinita,
$\varphi = 0$). (3) Estados ligados: o casamento só tem soluções para
$E$ discretos --- conte-os graficamente. (4) Espalhamento: as correntes
$(|A|^2 - |B|^2)\hbar k/m$ dão $R$ e $T$. (5) Tunelamento: $T \sim
\eu^{-2\kappa a}$; primeiro a ordem de grandeza, depois o prefator. (6)
Poço duplo: desdobramento $\propto\eu^{-\kappa a}$, batimentos em $\Delta/h$.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:1}
Poço infinito: elétron em \qty{1}{nm} ($E_1$, $E_2$, $E_3$, comprimento de onda do
fóton $2 \to 1$); elétron em \qty{0.1}{nm}; núcleon em \qty{5}{fm};
uma bola de gude de \qty{1}{g} em \qty{10}{cm} ($E_1$, e o número quântico para uma
velocidade de \qty{1}{cm/s}).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:2}
Pontos quânticos: um nanocristal de intervalo \qty{1.74}{eV} confina um
par elétron--buraco cuja \omterm{prop:b2:potential-wells-tunneling:infinite}{energia de confinamento} vale $h^2/8m^*L^2$ com
$m^* = 0.1\,m_{\text{e}}$. Comprimento de onda de emissão para $L = 6$, $4$, $3$,
\qty{2.5}{nm}; que cores?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:3}
Tunelamento: elétron com $V_0 - E = \qty{4.5}{eV}$: $\kappa$; $\eu^{-2\kappa a}$
para $a = 0.3$, $0.5$, $1$, \qty{2}{nm}; fator por \qty{0.1}{nm}; o mesmo para
um próton com $a = \qty{0.3}{nm}$.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:potential-wells-tunneling:4}
Degrau: elétron de \qty{2}{eV} sobre um degrau de \qty{1}{eV}: $k_1$, $k_2$,
$R$, $T$. O mesmo para um degrau para baixo de \qty{1}{eV}. Por que isso é impossível
classicamente, e qual é o análogo para a luz?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:5}
\emph{O poço infinito em detalhe.} (a) Deduza os níveis e normalize.
(b) Mostre a ortogonalidade. (c) Para $n = 1$: $\langle x\rangle$, $\Delta x =
L\sqrt{1/12 - 1/2\pi^2}$, $\langle p\rangle = 0$, $\Delta p = \pi\hbar/L$; verifique
Heisenberg. (d) Esboce $|\varphi_n|^2$ para $n$ grande e compare com a
probabilidade clássica de encontrar uma partícula que ricocheteia.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:6}
\emph{Poço finito.} (a) Deduza $k\tan ka = \kappa$ para os estados pares.
(b) Mostre graficamente que há sempre um estado ligado e conte-os para
$R = 1$, $4$, $10$. (c) Elétron, $V_0 = \qty{1}{eV}$, $2a = \qty{1}{nm}$:
$R$, número de estados, \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração} do estado fundamental (tome
$E \approx \qty{0.26}{eV}$). (d) O que acontece com o número de estados ligados
quando $V_0 \to \infty$, e com suas energias?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:7}
\emph{Degrau, $E < V_0$.} (a) Escreva $\varphi$ dos dois lados e encontre $r =
(k - \iu\kappa)/(k + \iu\kappa)$. (b) Mostre que $|r| = 1$ e calcule a fase
de $r$. (c) Mostre que a corrente se anula para $x > 0$ embora $|\varphi|^2 \ne
0$ ali. (d) Elétron \qty{1}{eV} abaixo do topo: \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração};
compare com a \omterm{def:b2:dispersion-wave-packets:complex}{onda evanescente} da \omterm{thm:b2:wave-interfaces:snell}{reflexão total interna}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:8}
\emph{A barreira.} (a) Escreva as quatro condições de continuidade e
deduza $T$. (b) Avalie exatamente e com a aproximação para $E =
V_0/2$, $\kappa a = 3$; e para $\kappa a = 1$. (c) Para $E > V_0$ mostre que $T = [1 +
V_0^2\sin^2(k_2a)/4E(E - V_0)]^{-1}$ e encontre as energias de transmissão
perfeita. (d) Interprete-as (pense na camada antirreflexo do
\cref{ch:b2:wave-interfaces}).
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:potential-wells-tunneling:9}
\emph{Gamow.} Para a barreira de Coulomb, do raio nuclear $R_{\text{n}}$
até $b = 2Ze^2/4\pi\varepsilon_0E$, o expoente vale $G = 2\int\kappa\,\dd r =
(2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b} - \sqrt{(R_{\text{n}}/b)(1 -
R_{\text{n}}/b)}]$. (a) $Z = 90$, $E = \qty{5}{MeV}$, $R_{\text{n}} = \qty{8}{fm}$:
$b$, $G$, $T$. (b) Frequência de choques $v/2R_{\text{n}}$ e o tempo de vida. (c)
Repita para $E = 4$ e \qty{6}{MeV}: razão dos tempos de vida. (d) Por que os
núcleos emitem partículas $\alpha$ em vez de prótons ou de $^{12}$C?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:10}
\emph{Amônia.} Desdobramento $\Delta = \qty{1e-4}{eV}$. (a) Frequência de inversão
e período da oscilação de tunelamento. (b) O átomo de N é
preparado de um lado: escreva $\psi(t)$ e a probabilidade de encontrá-lo
do outro lado. (c) A ND$_3$ tunela a \qty{1.6}{GHz}: deduza $\kappa a$
da razão dos desdobramentos se $\Delta \propto \eu^{-\kappa a}$ e $\kappa
\propto\sqrt m$ ($m_{\text{D}} = 2m_{\text{H}}$, barreira inalterada). (d) Por que
uma molécula quiral pode ficar canhota por anos?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:11}
\emph{O STM.} $\Phi = \qty{4.5}{eV}$, $d = \qty{0.5}{nm}$, tensão \qty{0.1}{V},
$I = \qty{1}{nA}$. (a) $\kappa$, o fator de $I$ por \qty{0.1}{nm} e a
resolução em altura se $I$ é medida com $1\%$ de precisão. (b) Elétrons por segundo
na corrente. (c) O átomo do ápice da ponta está \qty{0.25}{nm} mais perto que seus
vizinhos (afastados lateralmente de \qty{0.25}{nm}, de modo que $d' = \sqrt{d^2 + 0.25^2}$):
fração da corrente que ele carrega. (d) Uma armação de aço de \qty{1}{cm} se dilata
de $\alpha L\Delta T$ com $\alpha = \qty{1e-5}{K^{-1}}$: que estabilidade de
temperatura um picômetro exige, e como os projetos contornam isso?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:potential-wells-tunneling:12}
\emph{Um laser de poço quântico.} Uma camada de \qty{10}{nm} de GaAs (intervalo
\qty{1.42}{eV}, $m_{\text{e}}^* = 0.067\,m_{\text{e}}$, $m_{\text{h}}^* = 0.45\,m_{\text{e}}$)
entre barreiras \qty{0.3}{eV} mais altas. (a) \omterm{prop:b2:potential-wells-tunneling:infinite}{Energias de confinamento} fundamentais
do elétron e do buraco (poço infinito). (b) Energia e comprimento de onda do
fóton do laser. (c) Largura para \qty{780}{nm}. (d) Com a barreira finita,
quantos estados eletrônicos ligados o poço de \qty{10}{nm} comporta?
\end{exercise}

\section{Problema: o STM, a partícula alfa e o maser de amônia}

\begin{problem}[{Problema de fim de semana --- três túneis}]\label{pb:b2:potential-wells-tunneling:1}
Dados: $\hbar = \qty{1.05e-34}{J.s}$, $h = \qty{6.63e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $m_\alpha = \qty{6.64e-27}{kg}$, $e^2/4\pi\varepsilon_0 =
\qty{1.44}{MeV.fm}$, \qty{1}{eV} $= \qty{1.60e-19}{J}$.

\textbf{Parte I --- O microscópio de varredura por tunelamento.}
A ponta e a amostra são de tungstênio ($\Phi = \qty{4.5}{eV}$); vão $d$; tensão
\qty{50}{mV}; a corrente vale $I = I_0\eu^{-2\kappa d}$.
\begin{enumerate}
  \item $\kappa$ para elétrons no nível de Fermi diante da barreira
        de vácuo $\Phi$.
  \item Fator de variação de $I$ para $\Delta d = \qty{0.1}{nm}$; para
        \qty{1}{pm}.
  \item A \omterm{def:b2:feedback-oscillators:loop}{realimentação} mantém $I$ constante a $1\%$: ruído em altura.
  \item $I = \qty{1}{nA}$: elétrons por segundo; tempo médio entre dois
        elétrons, comparado com o ``tempo'' de tunelamento $\sim\hbar/\Phi$.
  \item O átomo do ápice e seus vizinhos a \qty{0.25}{nm} lateralmente,
        com $d = \qty{0.5}{nm}$: parcela do átomo do ápice; por que um único átomo
        basta para imagear átomos.
  \item Um átomo adsorvido de \qty{0.1}{nm} de altura: variação de $I$ no
        modo de altura constante; por que se prefere o modo de corrente constante.
  \item Uma armação metálica de \qty{2}{cm}, $\alpha = \qty{1e-5}{K^{-1}}$: variação de
        temperatura que desloca a ponta de \qty{1}{pm}; como os instrumentos
        lidam com isso (simetria, rapidez, materiais de baixa dilatação)?
  \item Com uma tensão de \qty{1}{V} a barreira deixa de ser retangular:
        esboce-a e diga se $I$ cresce mais depressa ou mais devagar que
        linearmente com $V$.
  \item Num trecho oxidado a função trabalho vale \qty{3}{eV} em vez
        de \qty{4.5}{eV}: de quanto a ponta recua no
        modo de corrente constante sobre uma superfície perfeitamente plana? O que o
        STM imageia, então?
\end{enumerate}

\textbf{Parte II --- Decaimento alfa de um núcleo $Z = 92$, $A = 238$.}
A partícula $\alpha$ ($E = \qty{4.2}{MeV}$) se move num núcleo de raio
$R_{\text{n}} = \qty{8}{fm}$; fora, a energia de Coulomb vale $V(r) =
2(Z - 2)e^2/4\pi\varepsilon_0r$.
\begin{enumerate}[resume]
  \item Altura da barreira em $R_{\text{n}}$; ponto de retorno clássico
        $b$.
  \item Velocidade da $\alpha$ no interior e sua frequência de choques na
        parede.
  \item Expoente de Gamow $G = (2\sqrt{2mE}/\hbar)\,b\,[\arccos\sqrt{R_{\text{n}}/b}
        - \sqrt{(R_{\text{n}}/b)(1 - R_{\text{n}}/b)}]$ e $T = \eu^{-G}$.
  \item Taxa de decaimento e meia-vida; compare com os $4.5 \times
        10^9$ anos medidos.
  \item O mesmo com $E = \qty{5.2}{MeV}$ (outro isótopo): meia-vida;
        a sensibilidade de Geiger--Nuttall.
  \item Tomando $R_{\text{n}} = \qty{9}{fm}$ em vez de \qty{8}{fm}: efeito sobre
        a meia-vida; comente a precisão dessas estimativas.
  \item Por que o tunelamento de um próton (carga $1$, massa $1/4$)
        não é observado a partir do mesmo núcleo, nem o de um $^{12}$C
        (carga $6$)?
  \item A $\alpha$ emerge com os \qty{4.2}{MeV} inteiros, embora tenha
        ``passado por baixo'' de uma muralha de \qty{30}{MeV}: a energia se conserva?
\end{enumerate}

\textbf{Parte III --- O maser de amônia.}
O átomo de N tunela através do plano dos H$_3$; o nível fundamental é desdobrado
de $\Delta = \qty{9.8e-5}{eV}$.
\begin{enumerate}[resume]
  \item Frequência e comprimento de onda da transição de inversão.
  \item Escreva os estados simétrico e antissimétrico em termos de
        ``N em cima'' e ``N embaixo''; qual é o mais baixo, e por quê?
  \item Uma molécula preparada com ``N em cima'': $\psi(t)$, a probabilidade de
        ``embaixo'' no instante $t$, o período.
  \item Razão de Boltzmann entre os dois níveis a \qty{300}{K}: a
        população térmica pode dar um maser? Como se obtém a inversão
        (um campo elétrico seletor de estados)?
  \item Energia do fóton em joules; número de moléculas que devem emitir
        por segundo para uma saída de \qty{1e-10}{W}.
  \item A ND$_3$ tunela a \qty{1.6}{GHz}: deduza $\kappa a$ para a NH$_3$
        supondo $\Delta \propto \eu^{-\kappa a}$ com $\kappa \propto \sqrt{m}$.
  \item O maser foi o primeiro relógio atômico: por que uma frequência de
        tunelamento é um bom relógio e que efeitos a deslocam?
  \item Resuma: a quantização vinda do confinamento, o tunelamento vindo da
        evanescência, e o que fixa os expoentes.
\end{enumerate}
\end{problem}
