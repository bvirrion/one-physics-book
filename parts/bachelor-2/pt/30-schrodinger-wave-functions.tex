\chapter{A equação de Schrödinger e as funções de onda}\label{ch:b2:schrodinger-wave-functions}

Dispare elétrons um a um contra um par de fendas e registre onde cada
um cai: um ponto aqui, um ponto ali, aparentemente ao acaso --- e, após dez
mil pontos, as franjas da experiência de Young, como se cada elétron
tivesse passado pelas duas fendas e interferido consigo mesmo. O último
capítulo do volume do primeiro ano encontrou essa estranheza: a luz vem em
fótons, a matéria tem um comprimento de onda $\lambda = h/p$, e a posição e o
momento não podem ser ambos precisos. O que ele não deu foi a
\emph{equação} --- a regra que diz como evolui a onda associada a uma
partícula, assim como a lei de Newton diz como evolui uma \omterm{def:b2:fluid-kinematics:euler}{trajetória}. Essa
equação é a de Schrödinger (1926), e com ela a descrição quântica
se torna uma teoria calculável: escreve-se uma \emph{\omterm{def:b2:schrodinger-wave-functions:psi}{função de onda}}
$\psi(x,t)$, cujo módulo ao quadrado é a probabilidade de encontrar a
partícula, e resolve-se uma equação diferencial parcial linear. Este
capítulo enuncia a equação, explica o que $\psi$ significa e o que não
significa, encontra seus \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} e os \omterm{prop:b2:schrodinger-wave-functions:free}{pacotes de onda} da
partícula livre, e mostra onde a mecânica de Newton reaparece --- para
tudo o que seja mais pesado que uma molécula.

\section{O que o volume do primeiro ano nos deixou}

\begin{proposition}[As partículas são ondas]\label{prop:b2:schrodinger-wave-functions:recap}
Um fóton de frequência $\nu$ tem a energia $E = h\nu = \hbar\omega$ e o
momento $p = h/\lambda = \hbar k$; uma partícula material de momento $p$
se comporta como uma onda de comprimento de onda $\lambda = h/p$ (de Broglie), com $E =
\hbar\omega$ e $p = \hbar k$ também --- a \omterm{def:b2:diffraction:huygens}{difração} de elétrons por
cristais e a interferometria de nêutrons e mesmo de moléculas o confirmam; e nenhum
estado pode ter uma posição e um momento mais precisos que $\Delta x\,\Delta p
\ge \hbar/2$ (Heisenberg). Com $\hbar = h/2\pi = \qty{1.05e-34}{J.s}$: um
elétron a \qty{100}{eV} tem $\lambda = \qty{0.12}{nm}$, o tamanho de um
átomo; um grão de poeira de \qty{1e-15}{kg} a \qty{1}{mm/s} tem $\lambda =
\qty{7e-16}{m}$, menor que um núcleo --- ele se comporta classicamente.
\end{proposition}

\begin{proof}
Recordado do volume do primeiro ano; a tarefa agora é a dinâmica da
onda.
\end{proof}

\section{A função de onda}

\begin{definition}[Função de onda e regra de Born]\label{def:b2:schrodinger-wave-functions:psi}
O estado de uma partícula que se move ao longo de $x$ é descrito por uma
\emph{função de onda}\index{função de onda} complexa $\psi(x,t)$ tal que
\[
\dd\mathcal P = |\psi(x,t)|^2\,\dd x
\]
é a probabilidade de encontrar a partícula entre $x$ e $x + \dd x$
no instante $t$ (a \emph{regra de Born}\index{regra de Born}); daí a
\emph{normalização} $\int_{-\infty}^{\infty}|\psi|^2\dd x = 1$, e os
valores esperados $\langle x\rangle = \int x|\psi|^2\dd x$, $\Delta x^2 =
\langle x^2\rangle - \langle x\rangle^2$. As funções de onda obedecem ao
\emph{princípio de superposição}\index{princípio de superposição}: se
$\psi_1$ e $\psi_2$ são estados possíveis, também o é $\alpha\psi_1 +
\beta\psi_2$ (normalizado), e as probabilidades contêm então o
termo de interferência $2\operatorname{Re}(\alpha^*\beta\,\psi_1^*\psi_2)$. Um
fator de fase global $\eu^{\iu\theta}$ nada muda; uma fase relativa
entre duas componentes superpostas muda a interferência.
\end{definition}

\begin{remark}[O que $\psi$ não é]\label{rem:b2:schrodinger-wave-functions:not}
$\psi$ não é um campo clássico espalhado no espaço como $\vect E$, nem uma
nuvem de matéria: o elétron é sempre encontrado inteiro, num único ponto.
$\psi$ é uma \emph{amplitude de probabilidade}, o objeto que
interfere; só $|\psi|^2$ é observado, e apenas estatisticamente, ao
repetir a experiência com partículas preparadas de modo idêntico. Os pontos
da fenda dupla são elétrons individuais; as franjas são $|\psi_1 +
\psi_2|^2$.
\end{remark}

\section{A equação de Schrödinger}

\begin{theorem}[Equação de Schrödinger]\label{thm:b2:schrodinger-wave-functions:equation}
Uma partícula de massa $m$ na energia potencial $V(x)$ tem uma \omterm{def:b2:schrodinger-wave-functions:psi}{função
de onda} que obedece a
\[
\iu\hbar\,\frac{\partial\psi}{\partial t} = -\frac{\hbar^2}{2m}\,\frac{\partial^2\psi}{\partial x^2} + V(x)\,\psi .
\]
Ela é linear (superposição), de primeira ordem no tempo (a \omterm{def:b2:schrodinger-wave-functions:psi}{função de onda}
num instante a determina em todos os instantes posteriores) e complexa ($\psi$
não pode ser real: o $\iu$ é essencial). É um \emph{postulado},
justificado por suas consequências.
\end{theorem}

\begin{proof}
Uma motivação, mais que uma demonstração. Uma partícula livre de momento preciso
$p = \hbar k$ e energia $E = \hbar\omega$ deve ser uma onda plana $\psi =
A\eu^{\iu(kx - \omega t)}$, para a qual $\iu\hbar\partial_t\psi = \hbar\omega\psi =
E\psi$ e $-(\hbar^2/2m)\partial_x^2\psi = (\hbar^2k^2/2m)\psi = (p^2/2m)\psi$.
A relação clássica $E = p^2/2m$ é assim reproduzida pela equação
$\iu\hbar\partial_t\psi = -(\hbar^2/2m)\partial_x^2\psi$, e a generalização
natural para $E = p^2/2m + V$ acrescenta $V\psi$. Note que a
equação deve ser linear para valer para superposições de ondas planas, e
de primeira ordem no tempo com um $\iu$ para que $\eu^{-\iu\omega t}$, e não
$\cos\omega t$, seja a dependência temporal --- uma \omterm{thm:b2:waves-on-strings:equation}{equação de onda} real
daria $E^2 \propto p^4$.
\end{proof}

\begin{proposition}[Conservação da probabilidade]\label{prop:b2:schrodinger-wave-functions:current}
A densidade de probabilidade $\rho = |\psi|^2$ e a \emph{corrente de
probabilidade}\index{corrente de probabilidade}
\[
j = \frac{\hbar}{m}\operatorname{Im}\Big(\psi^*\frac{\partial\psi}{\partial x}\Big)
= \frac{\hbar}{2\iu m}\Big(\psi^*\partial_x\psi - \psi\,\partial_x\psi^*\Big)
\]
obedecem a $\partial_t\rho + \partial_xj = 0$: a probabilidade não é criada nem
destruída, e $\int|\psi|^2\dd x$ permanece igual a $1$. Para uma onda plana
$A\eu^{\iu kx}$, $j = |A|^2\hbar k/m = \rho v$: densidade vezes velocidade, como para
qualquer escoamento.
\end{proposition}

\begin{proof}
$\partial_t(\psi^*\psi) = \psi^*\partial_t\psi + \psi\,\partial_t\psi^*$; da
equação e de sua conjugada, $\partial_t\psi = (\iu\hbar/2m)\partial_x^2\psi -
(\iu/\hbar)V\psi$, $\partial_t\psi^* = -(\iu\hbar/2m)\partial_x^2\psi^* +
(\iu/\hbar)V\psi^*$; os termos em $V$ se cancelam e resta $(\iu\hbar/2m)(\psi^*
\partial_x^2\psi - \psi\,\partial_x^2\psi^*) = -\partial_xj$.
\end{proof}

\begin{proposition}[Momento, valores esperados e limite clássico]\label{prop:b2:schrodinger-wave-functions:ehrenfest}
O momento de um estado se obtém pelo operador $\hat p =
-\iu\hbar\,\partial_x$ (que dá $\hbar k$ numa onda plana): $\langle
p\rangle = \int\psi^*(-\iu\hbar\partial_x\psi)\dd x$, e a energia pelo operador $\hat H
= -(\hbar^2/2m)\partial_x^2 + V$. Os valores esperados obedecem
(Ehrenfest) a
\[
\frac{\dd\langle x\rangle}{\dd t} = \frac{\langle p\rangle}{m}, \qquad
\frac{\dd\langle p\rangle}{\dd t} = -\Big\langle\frac{\dd V}{\dd x}\Big\rangle :
\]
o centro do \omterm{prop:b2:schrodinger-wave-functions:free}{pacote de onda} segue a lei de Newton exatamente quando $V$ é
no máximo quadrático, e aproximadamente sempre que o pacote é estreito
frente à escala em que $V$ varia --- o limite clássico,
que vale para qualquer objeto macroscópico porque $\hbar$ é pequeníssimo.
\end{proposition}

\begin{proof}
Derive $\langle x\rangle = \int x|\psi|^2$, use $\partial_t\rho =
-\partial_xj$ e integre por partes: $\dd\langle x\rangle/\dd t = \int j\,\dd x
= \langle p\rangle/m$. A segunda relação é o mesmo cálculo uma
ordem acima (admitido aqui).
\end{proof}

\section{Estados estacionários}

\begin{theorem}[Estados estacionários]\label{thm:b2:schrodinger-wave-functions:stationary}
Se $V$ não depende do tempo, a equação tem soluções da
forma separada
\[
\psi(x,t) = \varphi(x)\,\eu^{-\iu Et/\hbar}, \qquad
-\frac{\hbar^2}{2m}\,\varphi'' + V\varphi = E\varphi
\]
(a \emph{equação de Schrödinger independente do tempo}\index{equação de
Schrödinger!independente do tempo}): os \emph{estados estacionários}\index{estado
estacionário}, de energia $E$ bem definida, cuja densidade de probabilidade
$|\varphi|^2$ não se move. As energias $E$ para as quais $\varphi$ é
aceitável (limitada, normalizável para um estado ligado) são os
\emph{níveis de energia}; a solução geral é uma superposição $\psi =
\sum_nc_n\varphi_n(x)\eu^{-\iu E_nt/\hbar}$, e uma superposição de dois níveis
tem uma densidade que oscila na \emph{frequência de Bohr}\index{frequência
de Bohr} $\nu_{21} = (E_2 - E_1)/h$ --- a frequência da luz que o
átomo emite ou absorve.
\end{theorem}

\begin{proof}
Insira $\varphi(x)f(t)$: $\iu\hbar f'/f = (-\hbar^2\varphi''/2m + V\varphi)/\varphi$,
uma função de $t$ igual a uma função de $x$, logo uma constante $E$; $f
= \eu^{-\iu Et/\hbar}$. Para dois níveis, $|c_1\varphi_1\eu^{-\iu E_1t/\hbar} +
c_2\varphi_2\eu^{-\iu E_2t/\hbar}|^2 = |c_1\varphi_1|^2 + |c_2\varphi_2|^2 +
2\operatorname{Re}(c_1^*c_2\varphi_1^*\varphi_2\eu^{-\iu(E_2 - E_1)t/\hbar})$.
A completude dos \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} é admitida.
\end{proof}

\section{A partícula livre: ondas planas e pacotes de onda}

\begin{proposition}[Partícula livre]\label{prop:b2:schrodinger-wave-functions:free}
Para $V = 0$ os \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} são as ondas planas $\eu^{\iu(kx -
\omega t)}$ com
\[
E = \hbar\omega = \frac{\hbar^2k^2}{2m}, \qquad
v_\varphi = \frac{\omega}{k} = \frac{\hbar k}{2m}, \qquad
v_{\text{g}} = \frac{\dd\omega}{\dd k} = \frac{\hbar k}{m} = \frac{p}{m} :
\]
as ondas de matéria são \emph{dispersivas}; a \omterm{def:b2:dispersion-wave-packets:relation}{velocidade de fase} não tem
significado físico, e a \emph{\omterm{prop:b2:dispersion-wave-packets:group}{velocidade de grupo}} é a velocidade da partícula. Uma onda
plana não é normalizável --- uma partícula de momento perfeitamente preciso está
espalhada por todo o espaço --- de modo que uma partícula real é um \emph{pacote
de onda}\index{pacote de onda}, uma superposição de ondas planas sobre uma banda
$\Delta k$, localizada em $\Delta x \sim 1/\Delta k$, que se move a $v_{\text{g}}$ e
\emph{se alarga}: um pacote gaussiano de largura inicial $\sigma_0$ tem no
instante $t$ a largura
\[
\sigma(t) = \sigma_0\sqrt{1 + \Big(\frac{\hbar t}{2m\sigma_0^2}\Big)^2}, \qquad
\tau = \frac{2m\sigma_0^2}{\hbar} ,
\]
e satisfaz $\Delta x\,\Delta p = \hbar/2$ em $t = 0$ (o mínimo permitido)
e mais depois.
\end{proposition}

\begin{proof}
A \omterm{def:b2:dispersion-wave-packets:relation}{relação de dispersão} decorre da equação; $v_{\text{g}}$ como no
\cref{ch:b2:dispersion-wave-packets}. A fórmula do alargamento vem da
síntese de Fourier do pacote --- cada componente $\eu^{\iu kx}$ adquire
a fase $\eu^{-\iu\hbar k^2t/2m}$, e a integral gaussiana com um
coeficiente complexo dá uma gaussiana de largura crescente; a integral
de Fourier é a ferramenta do volume de matemática do terceiro ano e admitimos o
resultado. Fisicamente: o pacote contém momentos espalhados por $\Delta p
= \hbar/2\sigma_0$, e portanto velocidades espalhadas por $\Delta v = \hbar/2m\sigma_0$,
e após $t$ as componentes mais rápidas ultrapassaram as mais lentas de $\Delta v\,t$
--- a fórmula diz exatamente isso.
\end{proof}

\begin{example}[Quem se alarga]\label{ex:b2:schrodinger-wave-functions:spreading}
Um elétron localizado em $\sigma_0 = \qty{1}{nm}$: $\tau = 2m\sigma_0^2/\hbar
= \qty{1.7e-14}{s}$ --- após um nanossegundo seu pacote tem \qty{60}{\micro m}
de largura: um elétron deixado sozinho não fica quieto; ligado num átomo ele
não se alarga porque o potencial o segura (um \omterm{thm:b2:schrodinger-wave-functions:stationary}{estado estacionário}).
Um grão de poeira de \qty{1e-15}{kg} localizado em \qty{1}{\micro m}: $\tau =
\qty{2e7}{s}$, oito meses; uma bala: mais que a idade do universo.
O alargamento quântico é real, e invisível para tudo o que se possa ver.
\end{example}

\begin{remark}[Energia e tempo]\label{rem:b2:schrodinger-wave-functions:energytime}
Um estado que só existe durante $\Delta t$ --- um átomo excitado de tempo de vida
$\tau$, um trem de onda de comprimento $c\tau$ --- é uma superposição de energias
espalhadas por $\Delta E \sim \hbar/\Delta t$: a relação de Fourier do
\cref{ch:b2:scalar-light-model} entre a duração de um trem e sua
\omterm{prop:b2:scalar-light-model:coherence}{largura espectral}, agora aplicada a $E = \hbar\omega$. Um nível atômico de
tempo de vida \qty{10}{ns} tem uma largura $\Delta E = \qty{7e-8}{eV}$, uma
largura de linha natural de \qty{16}{MHz}; a largura Doppler do \cref{ch:b2:laser}
(\qty{1.5}{GHz}) a esconde num gás, mas não num átomo frio ou num íon em
repouso.
\end{remark}

\begin{method}[Trabalhar com funções de onda]\label{meth:b2:schrodinger-wave-functions:method}
(1) Normalize; as probabilidades são $\int|\psi|^2$ sobre a região. (2)
$V$ independente do tempo: procure $\varphi(x)\eu^{-\iu Et/\hbar}$, resolva a
equação estacionária, imponha as condições de contorno --- daí os níveis.
(3) Superponha \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} para a evolução; dois níveis batem em
$(E_2 - E_1)/h$. (4) Movimento livre: $v_{\text{g}} = p/m$, tempo de alargamento
$2m\sigma_0^2/\hbar$. (5) Correntes: $j = (\hbar/m)\operatorname{Im}(\psi^*\psi')$;
para $A\eu^{\iu kx} + B\eu^{-\iu kx}$, $j = (|A|^2 - |B|^2)\hbar k/m$ (sem termo cruzado).
(6) Limite clássico: Ehrenfest, e $\lambda$ frente ao tamanho do
aparelho.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:1}
Comprimentos de onda de de Broglie: um elétron a \qty{100}{eV}, \qty{10}{keV},
\qty{100}{keV} (use $p = \sqrt{2mE}$ e depois diga por que o último precisa de
correção); um nêutron térmico (\qty{25}{meV}, $m = \qty{1.67e-27}{kg}$); uma
molécula de C$_{60}$ ($m = \qty{1.2e-24}{kg}$) a \qty{200}{m/s}; uma bola de
tênis. Quais deles mostram interferência?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:2}
$\psi(x) = A\eu^{-x^2/2a^2}$: $A$ (use $\int\eu^{-u^2}\dd u = \sqrt\pi$);
$\langle x\rangle$, $\Delta x$; probabilidade de encontrar a partícula em
$|x| < a$ ($\operatorname{erf}(1) = 0.84$) e em $|x| < 2a$ ($0.9995$).
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:3}
Verifique que $\varphi(x)\eu^{-\iu Et/\hbar}$ resolve a equação quando $\varphi$
resolve a estacionária. Frequência do fasor para $E = \qty{1}{eV}$;
para uma superposição de dois níveis separados por \qty{2}{eV}, frequência e
comprimento de onda da oscilação de $|\psi|^2$ --- e da luz
emitida.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:schrodinger-wave-functions:4}
Tempo de alargamento $\tau = 2m\sigma_0^2/\hbar$ e largura após \qty{1}{ns} e
\qty{1}{s}: um elétron com $\sigma_0 = \qty{1}{nm}$ e com $\sigma_0 = \qty{1}{mm}$;
um próton com $\sigma_0 = \qty{1}{nm}$; um grão de \qty{1e-12}{kg} com
$\sigma_0 = \qty{1}{\micro m}$. Por que os objetos ficam onde são postos?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:5}
\emph{Corrente.} (a) Deduza a \omterm{thm:b2:fluid-kinematics:continuity}{equação da continuidade}. (b) $j$ para
$A\eu^{\iu kx}$, para $A\sin kx$, para $A\eu^{\iu kx} + B\eu^{-\iu kx}$. (c) Por que
a ausência de termo cruzado no último caso é a afirmação de que
os fluxos refletido e incidente simplesmente se subtraem? (d) Mostre que para uma
\omterm{def:b2:schrodinger-wave-functions:psi}{função de onda} real $j = 0$: um \omterm{thm:b2:schrodinger-wave-functions:stationary}{estado estacionário} ligado não transporta corrente.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:6}
\emph{Um pacote gaussiano.} $\psi(x,0) = A\eu^{\iu k_0x}\eu^{-x^2/4\sigma^2}$. (a)
$A$, $\langle x\rangle$, $\Delta x$. (b) $\langle p\rangle$ com $\hat p =
-\iu\hbar\partial_x$. (c) Admitindo $\Delta p = \hbar/2\sigma$, verifique
$\Delta x\Delta p = \hbar/2$. (d) Demonstre $\dd\langle x\rangle/\dd t = \langle
p\rangle/m$ a partir da \omterm{thm:b2:fluid-kinematics:continuity}{equação da continuidade}.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:7}
\emph{Elétrons num tubo.} Elétrons acelerados por \qty{100}{V}
percorrem \qty{30}{cm}. (a) Velocidade, comprimento de onda, tempo de voo. (b) \omterm{def:b2:dispersion-wave-packets:relation}{Velocidades de
fase} e de grupo; qual é a do elétron? (c) Um pacote inicialmente de
\qty{10}{\micro m} de largura: tempo de alargamento, largura na chegada. (d) \omterm{def:b2:diffraction:huygens}{Difração}
por uma abertura de \qty{0.3}{mm} no caminho: espalhamento angular e mancha no
anteparo. A imagem de raios é válida?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:8}
\emph{Fases.} (a) Mostre que $\psi$ e $\eu^{\iu\theta}\psi$ dão as mesmas
previsões. (b) Para $\psi = (\psi_1 + \eu^{\iu\alpha}\psi_2)/\sqrt2$, escreva
$|\psi|^2$ e mostre que $\alpha$ importa. (c) Na fenda dupla, o que
são $\psi_1$ e $\psi_2$, e o que fixa sua fase relativa num ponto
do anteparo? (d) Por que detectar por qual fenda o elétron passou
destrói as franjas (qualitativamente)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:schrodinger-wave-functions:9}
\emph{Ehrenfest.} (a) Para $V = \tfrac12m\omega^2x^2$, mostre que $\langle
x\rangle$ obedece exatamente a $\ddot{\langle x\rangle} = -\omega^2\langle x\rangle$.
(b) Para $V = mgx$, $\langle x\rangle$ cai como uma pedra. (c) Para um $V$
geral, quando vale $\langle V'(x)\rangle \approx V'(\langle x\rangle)$? (d) Uma
esfera de \qty{1}{g} numa tigela: estime o tamanho de pacote necessário para
que ela fosse ``quântica'' e compare com um átomo.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:10}
\emph{O vaivém.} Numa caixa $0 < x < L$ com paredes impenetráveis os
\omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários} são $\varphi_n = \sqrt{2/L}\sin(n\pi x/L)$, $E_n =
n^2h^2/8mL^2$ (\cref{ch:b2:potential-wells-tunneling}). Tome $\psi =
(\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/\sqrt2$. (a)
$|\psi|^2$ em função do tempo. (b) Mostre que $\langle x\rangle = L/2 -
(16L/9\pi^2)\cos\omega_{21}t$ (use $\int_0^L x\sin(\pi x/L)\sin(2\pi x/L)\dd x
= -8L^2/9\pi^2$). (c) Elétron, $L = \qty{1}{nm}$: $E_2 - E_1$, a
frequência e o comprimento de onda da oscilação. (d) O que faz uma carga
que oscila (\cref{ch:b2:dipole-radiation})? Conclua sobre as
linhas espectrais.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:11}
\emph{O alargamento, deduzido.} O pacote $\psi(x,0) \propto \eu^{-x^2/4\sigma_0^2}$
é a superposição $\int g(k)\eu^{\iu kx}\dd k$ com $g(k) \propto
\eu^{-\sigma_0^2k^2}$ (admitido: a transformada de Fourier de uma gaussiana é uma
gaussiana, de largura $1/2\sigma_0$ em $k$). (a) Escreva $\psi(x,t)$ dando a cada
componente sua fase $\eu^{-\iu\hbar k^2t/2m}$. (b) Complete o quadrado no
expoente e admita $\int\eu^{-\alpha k^2 + \beta k}\dd k = \sqrt{\pi/\alpha}\,
\eu^{\beta^2/4\alpha}$ para $\alpha$ complexo de parte real positiva: mostre que
$|\psi|^2$ é uma gaussiana de largura $\sigma(t) = \sigma_0\sqrt{1 + (\hbar t/2m
\sigma_0^2)^2}$. (c) Interprete $\tau$ por meio de $\Delta v = \Delta p/m$. (d) Por que
um pacote de fótons no vácuo não se alarga?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:schrodinger-wave-functions:12}
\emph{Energia--tempo.} (a) Um estado que decai como $\eu^{-t/2\tau}\eu^{-\iu E_0t/\hbar}$
(probabilidade $\eu^{-t/\tau}$) é uma superposição de energias: admitindo
que sua distribuição de energia é uma lorentziana de largura total $\Gamma =
\hbar/\tau$, dê $\Gamma$ para $\tau = \qty{10}{ns}$ (átomo), \qty{1e-8}{s},
\qty{1e-23}{s} (uma ressonância hadrônica, em MeV). (b) Largura natural de
linha do nível de \qty{10}{ns} em frequência; compare com a largura Doppler
de \qty{1.5}{GHz}. (c) Como se vê a largura natural (átomos frios,
íons aprisionados)? (d) Relacione com o \omterm{prop:b2:scalar-light-model:coherence}{comprimento de coerência} do trem de onda
emitido.
\end{exercise}

\section{Problema: o elétron num tubo e o alargamento de um pacote de onda}

\begin{problem}[{Problema de fim de semana --- quando um elétron é partícula,
quando é onda}]\label{pb:b2:schrodinger-wave-functions:1}
Dados: $h = \qty{6.63e-34}{J.s}$, $\hbar = \qty{1.05e-34}{J.s}$, $m_{\text{e}} =
\qty{9.11e-31}{kg}$, $e = \qty{1.60e-19}{C}$, $m_{\text{p}} = \qty{1.67e-27}{kg}$.

\textbf{Parte I --- O tubo de raios catódicos.}
Os elétrons saem de um cátodo quente com energia desprezível, são acelerados
por $U = \qty{20}{kV}$, atravessam uma abertura de \qty{0.3}{mm} e atingem um
anteparo a \qty{40}{cm}.
\begin{enumerate}
  \item Velocidade e momento (não relativísticos); comprimento de onda de de Broglie.
  \item A energia cinética vale $4\%$ de $mc^2$: o tratamento não relativístico
        é aceitável para este problema?
  \item Tempo de voo até o anteparo.
  \item \omterm{def:b2:diffraction:huygens}{Difração} pela abertura: espalhamento angular $\sim\lambda/a$ e
        a mancha que ele produz no anteparo; compare com um pixel
        (\qty{0.3}{mm}).
  \item Heisenberg na abertura: $\Delta p_y \gtrsim \hbar/2a$ dá o
        mesmo espalhamento --- verifique.
  \item Um pacote de \qty{0.3}{mm} de largura ao longo do feixe: tempo de alargamento
        $2m\sigma_0^2/\hbar$ e largura no anteparo.
  \item Placas de deflexão aplicam uma força transversal: pelo teorema de Ehrenfest, como
        se move $\langle y\rangle$? Conclua que o tubo é um
        dispositivo clássico.
  \item \omterm{def:b2:dispersion-wave-packets:relation}{Velocidade de fase} da onda do elétron; por que não é paradoxal que
        ela diferir da velocidade do elétron?
\end{enumerate}

\textbf{Parte II --- \omterm{prop:b2:schrodinger-wave-functions:free}{Pacotes de onda}.}
\begin{enumerate}[resume]
  \item Para um pacote gaussiano de largura $\sigma_0$: $\Delta p$ e a
        dispersão de velocidades $\Delta v$.
  \item Tempo de alargamento para um elétron com $\sigma_0 = \qty{0.1}{nm}$,
        \qty{1}{nm}, \qty{1}{\micro m}; para um próton com \qty{1}{nm}; para um
        vírus (\qty{1e-17}{kg}) com \qty{10}{nm}.
  \item Um elétron num átomo de hidrogênio está localizado em \qty{0.1}{nm}
        e não se alarga: por quê?
  \item Um elétron livre a \qty{1}{eV} com $\sigma_0 = \qty{1}{nm}$: que distância
        ele percorre antes que sua largura dobre? Compare com o
        comprimento de onda.
  \item Num metal, os elétrons de condução são ondas de $\lambda \approx
        \qty{0.5}{nm}$ por todo o cristal: em que sentido eles estão
        ``deslocalizados''?
  \item Um microscópio eletrônico a \qty{100}{keV} tem $\lambda \approx
        \qty{4}{pm}$ e, no entanto, resolve apenas \qty{0.1}{nm}: por quê (suas lentes
        magnéticas aceitam um semiângulo de cerca de \qty{10}{mrad})?
  \item Dê o critério que decide se um objeto deve ser
        tratado com uma \omterm{def:b2:schrodinger-wave-functions:psi}{função de onda} ou com Newton.
\end{enumerate}

\textbf{Parte III --- \omterm{thm:b2:schrodinger-wave-functions:stationary}{Estados estacionários} e transições.}
Um elétron está confinado em $0 < x < L = \qty{1}{nm}$; $\varphi_n =
\sqrt{2/L}\sin(n\pi x/L)$, $E_n = n^2h^2/8mL^2$.
\begin{enumerate}[resume]
  \item $E_1$, $E_2$, $E_3$ em elétron-volts; as \omterm{thm:b2:schrodinger-wave-functions:stationary}{frequências de Bohr}
        $\nu_{21}$, $\nu_{31}$ e os comprimentos de onda.
  \item Verifique que $|\varphi_n\eu^{-\iu E_nt/\hbar}|^2$ é independente do tempo;
        onde o elétron é mais provável no estado fundamental?
  \item Para $\psi = (\varphi_1\eu^{-\iu E_1t/\hbar} + \varphi_2\eu^{-\iu E_2t/\hbar})/
        \sqrt2$: $|\psi|^2$ em $t = 0$ e em $t = T_{21}/2$; esboce.
  \item $\langle x\rangle(t)$ (dado $\int_0^Lx\varphi_1\varphi_2\dd x = -16L/9\pi^2$):
        amplitude do vaivém.
  \item O \omterm{thm:b2:dipole-radiation:field}{dipolo oscilante} $e\langle x\rangle$ irradia: em que
        frequência, e o que isso diz sobre as linhas espectrais e
        sobre a estabilidade dos \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários}?
  \item Energia do estado $\psi$: é $E_1$, $E_2$ ou outra coisa?
        O que dá uma medida?
\end{enumerate}

\textbf{Parte IV --- Interpretação.}
\begin{enumerate}[resume]
  \item Na experiência da fenda dupla com um elétron de cada vez, o que é aleatório
        e o que não é?
  \item Escreva a \omterm{prop:b2:schrodinger-wave-functions:current}{corrente de probabilidade} de $A\eu^{\iu kx}$ e relacione-a com
        um feixe de $n$ elétrons por unidade de comprimento à velocidade $v$.
  \item Por que a equação de Schrödinger deve ser complexa e de primeira ordem
        no tempo (duas razões)?
  \item Resuma em cinco linhas: a \omterm{def:b2:schrodinger-wave-functions:psi}{função de onda}, a equação, os
        \omterm{thm:b2:schrodinger-wave-functions:stationary}{estados estacionários}, o pacote, o limite clássico.
\end{enumerate}
\end{problem}
