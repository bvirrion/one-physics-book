\chapter{Fluidos perfeitos: Euler e Bernoulli}\label{ch:b2:euler-bernoulli}

Segure uma folha de papel pela borda e sopre sobre ela: a folha sobe.
Abra o chuveiro e a cortina do banheiro se aproxima. Uma asa sustenta
trezentas toneladas porque o ar passa mais depressa sobre o extradorso
do que sob o intradorso. As três situações dizem a mesma coisa:
\emph{onde um fluido anda mais depressa, sua pressão é menor}. Este
capítulo escreve a segunda lei de Newton para uma \omterm{def:b2:fluid-kinematics:particle}{partícula de fluido}
no modelo mais simples --- o fluido \emph{perfeito}, que sente pressão
mas não atrito --- e dela extrai o teorema de Bernoulli, o balanço de
energia de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}, cujas aplicações vão do medidor de
vazão de uma tubulação ao indicador de velocidade de um avião.

\section{A equação de Euler}

\begin{proposition}[Força de pressão sobre uma partícula de fluido]\label{prop:b2:euler-bernoulli:pressure}
\index{força de pressão (densidade volumétrica)}
A resultante das forças de pressão exercidas pelo fluido circundante
sobre uma partícula de volume $\dd\tau$ vale
\[
\dd\vect F_P = -\operatorname{\vect{grad}}P\,\dd\tau :
\]
a pressão age como uma força volumétrica de densidade $-\operatorname{\vect{grad}}P$,
empurrando das altas para as baixas pressões.
\end{proposition}

\begin{proof}
Na caixa de \cref{thm:b2:fluid-kinematics:continuity}, as faces em
$x$ e em $x + \dd x$ recebem $+P(x)\dd y\dd z$ e $-P(x + \dd x)\dd y\dd z$
ao longo de $x$: resultante $-\partial_xP\,\dd x\dd y\dd z$; o mesmo para $y$ e $z$.
\end{proof}

\begin{definition}[Fluido perfeito]\label{def:b2:euler-bernoulli:perfect}
\index{fluido perfeito}
Um \emph{fluido perfeito} é aquele em que a única força de contato entre
partículas vizinhas é a pressão --- normal a toda superfície, sem
tensão tangencial (viscosa). É uma idealização válida, como o próximo
capítulo explica, longe das paredes e a alto número de Reynolds: a água
numa tubulação longe da parede, o ar em torno de uma asa fora da fina
camada limite, uma onda no mar.
\end{definition}

\begin{theorem}[Equação de Euler]\label{thm:b2:euler-bernoulli:euler}
\index{equação de Euler}
Num referencial galileano, um \omterm{def:b2:euler-bernoulli:perfect}{fluido perfeito} de massa específica $\rho$ submetido
à gravidade obedece, em cada ponto, a
\[
\rho\Bigl(\frac{\partial\vect v}{\partial t} + (\vect v\cdot\operatorname{\vect{grad}})
\vect v\Bigr) = -\operatorname{\vect{grad}}P + \rho\vect g ,
\]
à qual se soma, à direita, qualquer outra densidade de força volumétrica $\vect f_v$ (elétrica, de
inércia num referencial não galileano). Em repouso ela se reduz à
lei da hidrostática $\operatorname{\vect{grad}}P = \rho\vect g$ do
volume do Ano 1.
\end{theorem}

\begin{proof}
Segunda lei de Newton para a partícula de massa $\rho\,\dd\tau$: sua
aceleração é a \omterm{thm:b2:fluid-kinematics:material}{derivada material}
(\cref{thm:b2:fluid-kinematics:material}), e as forças são a resultante
de pressão e o peso $\rho\vect g\,\dd\tau$; divida por $\dd\tau$.
\end{proof}

\begin{example}[Um reservatório que acelera]\label{ex:b2:euler-bernoulli:tank}
A água num reservatório levado por um caminhão que acelera a $\vect a$ está em repouso
no referencial do caminhão; ali, Euler com a força de inércia $-\rho\vect a$
dá $\operatorname{\vect{grad}}P = \rho(\vect g - \vect a)$: a pressão
cresce ao longo da "gravidade efetiva" $\vect g - \vect a$, a superfície
livre (uma isóbara) é perpendicular a ela e se inclina para trás de
um ângulo $\arctan(a/g)$ --- \qty{11}{\degree} para $a = \qty{2}{m/s^2}$. Num
balde que gira a $\Omega$, a força centrífuga $\rho\Omega^2r\,\vect e_r$
dá $P = P_0 + \tfrac12\rho\Omega^2r^2 - \rho gz$ e a superfície livre é
o paraboloide $z = \Omega^2r^2/2g$.
\end{example}

\section{O teorema de Bernoulli}

\begin{theorem}[Bernoulli]\label{thm:b2:euler-bernoulli:bernoulli}
\index{teorema de Bernoulli}\index{pressão de estagnação}\index{pressão dinâmica}
Para um fluido \emph{perfeito} e \emph{incompressível} de massa específica uniforme em
escoamento \emph{permanente}, num referencial galileano com gravidade uniforme, a grandeza
\[
P + \tfrac12\rho v^2 + \rho gz
\]
é constante \emph{ao longo de cada \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}}. Se o escoamento é, além disso,
\emph{irrotacional}, trata-se da mesma constante em todo o fluido. O
termo $\tfrac12\rho v^2$ é a \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{pressão dinâmica}}; $P + \tfrac12\rho v^2$
é a \emph{\omterm{thm:b2:euler-bernoulli:bernoulli}{pressão de estagnação}}, a pressão que o fluido atingiria
se fosse levado ao repouso ao longo de sua \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}.
\end{theorem}

\begin{proof}
Euler em regime permanente com $(\vect v\cdot\operatorname{\vect{grad}})\vect v =
\operatorname{\vect{grad}}(v^2/2) + \vect\omega\wedge\vect v$ e $\rho\vect g =
-\operatorname{\vect{grad}}(\rho gz)$, com $\rho$ uniforme:
\[
\operatorname{\vect{grad}}\bigl(P + \tfrac12\rho v^2 + \rho gz\bigr) = -\rho\,
\vect\omega\wedge\vect v .
\]
O lado direito é perpendicular a $\vect v$, de modo que o gradiente da
grandeza de Bernoulli não tem componente ao longo da \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}: a grandeza
é constante ao longo dela. Se $\vect\omega = \vect 0$ o gradiente se anula
em toda parte.
\end{proof}

\begin{remark}[O que Bernoulli diz e o que não diz]\label{rem:b2:euler-bernoulli:meaning}
Dividido por $\rho$, o teorema diz: a energia mecânica por unidade de
massa, $v^2/2 + gz$, mais $P/\rho$, conserva-se ao longo do caminho de uma partícula.
O termo de pressão é o trabalho realizado \emph{sobre} a partícula pela
pressão do fluido que vem atrás, menos o trabalho que ela realiza sobre o fluido
à frente: o teorema da energia para uma partícula empurrada através de um campo
de pressão sem atrito. Quatro condições: \omterm{def:b2:euler-bernoulli:perfect}{fluido perfeito} (sem perda
viscosa), permanente, incompressível e \emph{ao longo de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}}. Comparar
dois pontos de \omterm{def:b2:fluid-kinematics:euler}{linhas de corrente} diferentes só é permitido em \omterm{def:b2:fluid-kinematics:vorticity}{escoamento
irrotacional}. O teorema não diz que "o ar rápido suga": ele diz que uma partícula que
foi \emph{acelerada} (por uma queda de pressão) está agora a uma pressão
menor --- a causa é o campo de pressão, a celeridade é o efeito.
\end{remark}

\begin{remark}[Fluidos compressíveis]\label{rem:b2:euler-bernoulli:compressible}
Num gás a massa específica varia, mas se o escoamento é permanente, perfeito e
isentrópico a mesma demonstração dá a conservação de $h + v^2/2 + gz$
ao longo de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}, onde $h$ é a entalpia por unidade de massa ($h = c_PT$
para um gás perfeito): a forma usada para bocais e turbinas em
\cref{ch:b2:open-systems}. A celeridades bem abaixo da velocidade do som
($v \lesssim \qty{100}{m/s}$ no ar) a massa específica varia menos de $5\%$
e o teorema incompressível é preciso.
\end{remark}

\section{Aplicações}

\begin{proposition}[O medidor Venturi]\label{prop:b2:euler-bernoulli:venturi}
\index{efeito Venturi}\index{medidor Venturi}
Numa tubulação horizontal que se estreita da seção $S_1$ para $S_2 < S_1$, a
celeridade aumenta ($v_2 = v_1S_1/S_2$, conservação da massa) e a pressão
cai:
\[
P_1 - P_2 = \tfrac12\rho(v_2^2 - v_1^2) = \tfrac12\rho v_1^2\Bigl(\frac{S_1^2}{S_2^2} - 1\Bigr) ,
\qquad
D_V = S_1v_1 = S_1S_2\sqrt{\frac{2(P_1 - P_2)}{\rho(S_1^2 - S_2^2)}} .
\]
Medir a diferença de pressão com um manômetro dá a vazão:
é o \emph{\omterm{prop:b2:euler-bernoulli:venturi}{medidor Venturi}}. O mesmo efeito faz funcionar um carburador, um
pulverizador de tinta, a entrada de ar de um bico de Bunsen e a "sucção" de um
trem que passa.
\end{proposition}

\begin{proof}
Bernoulli ao longo de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente} central a altura constante, com $v_2 =
v_1S_1/S_2$; isole $v_1$.
\end{proof}

\begin{proposition}[O tubo de Pitot]\label{prop:b2:euler-bernoulli:pitot}
\index{tubo de Pitot}
Um tubo voltado para o escoamento, fechado na outra extremidade, leva o fluido ao repouso
em sua boca (um \emph{ponto de estagnação}); uma segunda abertura lateral,
paralela ao escoamento, lê a pressão estática $P$. A diferença é
a \omterm{thm:b2:euler-bernoulli:bernoulli}{pressão dinâmica}:
\[
P_{\text{estag}} - P = \tfrac12\rho v^2 , \qquad v = \sqrt{2(P_{\text{estag}} - P)/\rho} .
\]
Todo avião mede assim sua velocidade em relação ao ar.
\end{proposition}

\begin{proof}
Bernoulli ao longo da \omterm{def:b2:fluid-kinematics:euler}{linha de corrente} que termina no ponto de estagnação, onde
$v = 0$; os furos laterais não perturbam o escoamento, que passa com o
$v$ e o $P$ não perturbados.
\end{proof}

\begin{proposition}[A fórmula de Torricelli]\label{prop:b2:euler-bernoulli:torricelli}
\index{fórmula de Torricelli}\index{escoamento por orifício}
Um grande reservatório aberto esvazia-se por um pequeno furo à profundidade $h$ abaixo da
superfície livre; o jato sai com celeridade
\[
v = \sqrt{2gh} ,
\]
a celeridade de queda livre a partir da superfície. A vazão vale $D_V \approx
\alpha s\sqrt{2gh}$, com $s$ a área do furo e $\alpha \approx 0.6$ o
\emph{coeficiente de contração} de um orifício de bordo vivo (as \omterm{def:b2:fluid-kinematics:euler}{linhas de corrente}
continuam a convergir depois do furo).
\end{proposition}

\begin{proof}
Bernoulli da superfície livre (em repouso, pois o reservatório é grande:
$v_{\text{sup}} = v\,s/S \ll v$; pressão $P_0$) até o jato (pressão $P_0$
logo depois do furo): $P_0 + \rho gh = P_0 + \tfrac12\rho v^2$. O escoamento
é quase permanente: $h$ varia lentamente.
\end{proof}

\begin{example}[Números]\label{ex:b2:euler-bernoulli:numbers}
Um Venturi de \qty{5}{cm} que se estreita a \qty{2.5}{cm} numa tubulação de água, com
$\Delta h = \qty{20}{cm}$ de água (\qty{2.0}{kPa}): $v_1 = \sqrt{2 \times 2000/
(1000 \times 15)} = \qty{0.52}{m/s}$, $D_V = \qty{1.0}{L/s}$. Um avião de linha a
\qty{11}{km} ($\rho = \qty{0.36}{kg/m^3}$) a \qty{250}{m/s}: \omterm{thm:b2:euler-bernoulli:bernoulli}{pressão dinâmica}
\qty{11}{kPa}, $11\%$ da pressão ambiente --- o \omterm{prop:b2:euler-bernoulli:pitot}{tubo de Pitot} ainda
funciona, com uma correção de compressibilidade. Uma caixa d'água \qty{40}{m}
acima das torneiras: \qty{4}{bar} com elas fechadas, e um jato a \qty{28}{m/s} se abertas
por completo (na prática, bem menos: os canos não são perfeitos).
\end{example}

\begin{proposition}[Sustentação de uma asa]\label{prop:b2:euler-bernoulli:lift}
\index{sustentação}\index{teorema de Kutta--Joukowski}\index{efeito Magnus}
No escoamento em torno de uma asa o ar passa mais depressa sobre o
extradorso do que sob o intradorso; a pressão é portanto menor
em cima do que embaixo, e a resultante é a \emph{sustentação} $F_L =
\tfrac12\rho v^2SC_L$ ($S$ a área da asa, $C_L \approx 0.3$--$1.5$ o coeficiente
de sustentação, que depende do perfil e do ângulo de ataque). A
diferença de celeridade equivale a uma \emph{circulação} $\Gamma$ da
velocidade em torno do perfil, e a sustentação por unidade de envergadura vale $F_L' =
\rho v\Gamma$ (Kutta--Joukowski, admitido). Uma bola em rotação arrasta o ar
consigo e é desviada lateralmente do mesmo modo: é o \emph{\omterm{prop:b2:euler-bernoulli:lift}{efeito
Magnus}} de uma bola de tênis cortada ou de uma falta que "encurva".
\end{proposition}

\begin{proof}
Bernoulli nas \omterm{def:b2:fluid-kinematics:euler}{linhas de corrente} logo acima e logo abaixo, que partem das
mesmas condições a montante: $P_{\text{inf}} - P_{\text{sup}} = \tfrac12\rho
(v_{\text{sup}}^2 - v_{\text{inf}}^2)$; integre ao longo da corda. Por que o
escoamento superior é mais rápido --- o bordo de fuga afiado fixa ali o ponto de
estagnação traseiro e determina a circulação (condição de Kutta)
--- escapa ao modelo do \omterm{def:b2:euler-bernoulli:perfect}{fluido perfeito} sozinho; é preciso a camada limite
do próximo capítulo.
\end{proof}

\begin{proposition}[Um escoamento não permanente: o tubo em U]\label{prop:b2:euler-bernoulli:utube}
\index{oscilação em tubo em U}\index{Bernoulli não permanente}
Uma coluna líquida de comprimento total $L$ preenche um tubo em U de seção uniforme;
deslocada de $z$ em relação ao equilíbrio, ela oscila segundo
\[
\ddot z + \frac{2g}{L}\,z = 0 , \qquad T = 2\pi\sqrt{\frac{L}{2g}} ,
\]
como um pêndulo de comprimento $L/2$. Mais geralmente, para um escoamento não permanente mas
irrotacional de potencial $\varphi$, Euler integra-se em
$\rho\,\partial_t\varphi + P + \tfrac12\rho v^2 + \rho gz = C(t)$, a mesma
constante em todo o fluido (\omterm{prop:b2:euler-bernoulli:utube}{Bernoulli não permanente}).
\end{proposition}

\begin{proof}
A celeridade $v = \dot z$ é uniforme ao longo do tubo (incompressível,
seção uniforme). Projete Euler no eixo do tubo e integre ao longo
da coluna de uma superfície livre à outra: os termos de pressão se
cancelam ($P_0$ nas duas extremidades), $\int\rho\,\partial_tv\,\dd s = \rho L\ddot z$, o
termo convectivo $\rho\,\Delta(v^2/2)$ se anula (mesma celeridade nas duas extremidades),
e o termo de gravidade dá $\rho g\cdot2z$. Logo $\rho L\ddot z = -2\rho gz$.
O enunciado geral segue de $\partial_t\vect v = \operatorname{\vect{grad}}
\partial_t\varphi$ para $\vect v = \operatorname{\vect{grad}}\varphi$.
\end{proof}

\begin{example}[Pôr uma tubulação em regime]\label{ex:b2:euler-bernoulli:start}
Uma tubulação horizontal de comprimento $\ell$ leva de um reservatório de carga constante
$h$ a uma saída aberta; a válvula é aberta em $t = 0$. Com $v$ uniforme
ao longo do tubo, Euler integrada da superfície do reservatório até a
saída dá $\ell\,\dd v/\dd t = gh - \tfrac12v^2$, cuja solução é
$v = V\tanh(Vt/2\ell)$ com $V = \sqrt{2gh}$: o escoamento atinge a celeridade de
Torricelli com constante de tempo $2\ell/V$ --- \qty{13}{s} para um conduto forçado de \qty{400}{m}
sob \qty{200}{m} de carga.
\end{example}

\begin{method}[Usar Bernoulli]\label{met:b2:euler-bernoulli:method}
(1) Verifique as quatro condições; decida se o escoamento é irrotacional
(escoamento uniforme a montante em torno de um obstáculo: sim; escoamento numa tubulação com
perfil de velocidades: não --- fique numa só \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}). (2) Escolha dois pontos
de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente} onde o maior número possível de grandezas seja conhecido: uma superfície
livre ($P = P_0$, $v \approx 0$ se a seção é grande), um jato ao
ar livre ($P = P_0$), um ponto de estagnação ($v = 0$). (3) Acrescente a conservação
da massa $vS = $ const. para relacionar as celeridades. (4) Para escoamentos
não permanentes, integre Euler ao longo da linha.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:1}
A água escoa numa tubulação horizontal de diâmetro \qty{8}{cm} a \qty{1.5}{m/s}
e pressão \qty{3.0}{bar}. Ela passa a uma seção de \qty{4}{cm}: celeridade
e pressão ali; depois a uma seção de \qty{2}{cm}: pressão --- o que acontece
se ela resultar negativa?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:2}
O \omterm{prop:b2:euler-bernoulli:pitot}{tubo de Pitot} de um avião a \qty{5}{km} ($\rho = \qty{0.74}{kg/m^3}$)
lê uma \omterm{thm:b2:euler-bernoulli:bernoulli}{pressão dinâmica} de \qty{15}{kPa}. Velocidade verdadeira em relação ao ar. O
indicador é calibrado para a massa específica ao nível do mar (\qty{1.22}{kg/m^3}): que
"velocidade indicada" ele mostra, e por que os pilotos acham essa
diferença útil, e não incômoda?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:3}
Um reservatório contém água com \qty{3.0}{m} de profundidade; um furo de \qty{1.0}{cm^2} é aberto
\qty{1.0}{m} acima do piso. Celeridade de saída; vazão (contração $0.6$);
onde o jato atinge o piso? A que profundidade deveria ser aberto um segundo furo
para alcançar o mesmo ponto?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:euler-bernoulli:4}
Um vento de \qty{30}{m/s} sopra sobre um telhado plano de \qty{100}{m^2}; o ar
sob o telhado está parado, à pressão atmosférica ($\rho = \qty{1.2}{kg/m^3}$).
Diferença de pressão e força resultante sobre o telhado; compare com o peso
do telhado (\qty{50}{kg/m^2}). Para que lado ele é empurrado?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:5}
Um reservatório fechado e retangular de \qty{2.0}{m} de comprimento, meio cheio de água, está sobre um
caminhão. (a) Acelerando a \qty{3.0}{m/s^2}: ângulo da superfície livre
e diferença de altura da água entre a parede traseira e a dianteira.
(b) Diferença de pressão entre os dois cantos do fundo. (c)
Freando a \qty{6}{m/s^2} --- a água alcança a tampa, \qty{0.8}{m} acima
do fundo? (d) O mesmo caminhão fazendo uma curva de raio \qty{40}{m} a
\qty{15}{m/s}: inclinação da superfície.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:6}
Um balde de raio \qty{15}{cm} contendo água com \qty{20}{cm} de profundidade gira
em torno de seu eixo a \qty{2.0}{voltas/s}. (a) Forma e equação da superfície
livre. (b) Diferença de altura entre a borda e o centro. (c) Conservando-se o volume,
altura da água no centro e na borda. (d) A que
frequência de rotação o centro do fundo fica seco? (e) Por que o
mesmo paraboloide dá um excelente espelho de telescópio (uma piscina de
mercúrio em rotação)?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:7}
\emph{Sifão.} Um tubo leva a água de um reservatório por sobre um ponto alto \qty{2.0}{m}
acima da superfície livre até uma saída \qty{3.0}{m} abaixo dela. (a) Celeridade de saída
e, para um tubo de \qty{2}{cm}, vazão. (b) Pressão no ponto alto. (c)
Altura máxima desse ponto para que o sifão funcione (a pressão de vapor
da água, \qty{2.3}{kPa}, não pode ser atingida). (d) Por que o sifão
não funciona com o tubo cheio de ar?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:8}
Um avião leve de \qty{1000}{kg} voa em cruzeiro a \qty{60}{m/s} ao nível do mar com
uma asa de \qty{15}{m^2}. (a) Diferença média de pressão entre as duas
faces da asa; coeficiente de sustentação. (b) Se a celeridade sob a asa é
\qty{58}{m/s}, quanto vale ela sobre a asa (tome as mesmas condições
a montante)? (c) Circulação em torno do perfil pela fórmula de
Kutta--Joukowski, para uma envergadura de \qty{10}{m}. (d) Celeridade de estol se
$C_L$ não pode exceder $1.4$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:euler-bernoulli:9}
\emph{Estenose.} O sangue ($\rho = \qty{1060}{kg/m^3}$) escoa a \qty{0.50}{m/s}
numa artéria de seção \qty{0.50}{cm^2} a \qty{13}{kPa} acima da pressão
atmosférica. Uma placa a estreita para \qty{0.10}{cm^2}. (a) Celeridade e pressão
na estenose. (b) O tecido do lado de fora da artéria está a cerca de
\qty{1}{kPa}: a artéria pode colapsar? (c) Se ela se estreita, a celeridade
sobe ainda mais: explique o descontrole e por que tal estreitamento pode vibrar.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:10}
\emph{Temperatura de estagnação.} Para um gás perfeito em escoamento isentrópico
permanente, $c_PT + v^2/2$ conserva-se ao longo de uma \omterm{def:b2:fluid-kinematics:euler}{linha de corrente}. (a) Mostre que o
ar levado ao repouso no nariz de um avião atinge $T_0 = T(1 +
\tfrac{\gamma - 1}{2}M^2)$, com $M = v/c$ o número de Mach e $c =
\sqrt{\gamma RT/M_{\text{mol}}}$ a velocidade do som. (b) Temperatura no nariz
de um avião de linha a $M = 0.85$ em ar a \qty{220}{K}; de um avião supersônico
a $M = 2$; de uma cápsula de reentrada a $M = 25$ (a fórmula ainda
faz sentido?). (c) Mostre que a fórmula de Pitot incompressível
superestima a celeridade em $M = 0.85$, e por quanto aproximadamente (expanda
$P_0/P = (T_0/T)^{\gamma/(\gamma - 1)}$ até a segunda ordem em $M^2$).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:11}
\emph{Transitório de partida.} Uma tubulação de comprimento $\ell = \qty{50}{m}$ e seção
uniforme leva de um grande reservatório (carga $h = \qty{5.0}{m}$ acima da
saída) a uma válvula na extremidade aberta. A válvula é aberta em $t = 0$. (a)
Integre Euler ao longo do tubo para obter $\ell\,\dd v/\dd t = gh - v^2/2$.
(b) Resolva: $v = V\tanh(Vt/2\ell)$ com $V = \sqrt{2gh}$. (c) Tempo para
atingir $90\%$ da celeridade final. (d) A válvula é agora fechada bruscamente em
\qty{0.5}{s} a partir do escoamento pleno: estime a desaceleração média e a
sobrepressão na válvula a partir de $\rho\ell\,\dd v/\dd t$ --- e por que um fechamento
real produz muito mais (próximos capítulos).
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:euler-bernoulli:12}
\emph{A superfície livre de um vórtice.} A água gira como o \omterm{ex:b2:fluid-kinematics:planeflows}{vórtice pontual}
de \cref{ch:b2:fluid-kinematics}, $v_\theta = \Gamma/2\pi r$, irrotacional,
com superfície livre em $z = 0$ bem longe. (a) Por que Bernoulli pode ser aplicado
entre dois pontos quaisquer? (b) Forma da superfície livre $z(r)$. (c)
Dentro de um núcleo de Rankine ($r < a$, $v_\theta = \Omega r$, rotacional) use
antes Euler no referencial girante, ou diretamente: mostre que $z = z(a) +
\Omega^2(r^2 - a^2)/2g$. (d) Profundidade total do funil para um vórtice de
banheira com $\Gamma = \qty{0.03}{m^2/s}$ e $a = \qty{5}{mm}$; para um tornado
($\Gamma = \qty{3e4}{m^2/s}$, $a = \qty{60}{m}$, no ar: exprima antes o
resultado como uma queda de pressão no centro).
\end{exercise}

\section{Problema: A barragem, o conduto forçado e a turbina}

\begin{problem}[{Problema de fim de semana --- uma usina hidrelétrica do
reservatório ao jato: pressões, celeridades, potências, perdas e o
golpe de aríete}]
\label{pb:b2:euler-bernoulli:1}

Um reservatório retém água com \qty{150}{m} de profundidade atrás de uma barragem; sua superfície livre
está $H = \qty{200}{m}$ acima dos bocais de uma turbina. Um conduto forçado (tubo
de aço) de diâmetro $D = \qty{3.0}{m}$ e comprimento $\ell = \qty{400}{m}$
leva a vazão de projeto $D_V = \qty{60}{m^3/s}$ até a casa de máquinas,
onde um bocal a transforma num jato livre à pressão atmosférica $P_0 =
\qty{1.0}{bar}$ que golpeia as conchas de uma roda Pelton. $\rho =
\qty{1000}{kg/m^3}$, $g = \qty{9.81}{m/s^2}$; o fluido é perfeito salvo
menção em contrário.

\medskip
\noindent\textbf{Parte I --- Em repouso e na vazão de projeto.}
\begin{enumerate}
  \item Pressão na base da barragem e força sobre uma faixa vertical de
        \qty{1}{m} de largura, do topo ao fundo.
  \item Com as válvulas da turbina fechadas, pressão no conduto forçado junto à
        casa de máquinas.
  \item Celeridade no conduto na vazão de projeto; \omterm{thm:b2:euler-bernoulli:bernoulli}{pressão dinâmica} ali;
        pressão na extremidade do conduto junto à casa de máquinas (antes do
        bocal), por Bernoulli desde a superfície do reservatório.
  \item Celeridade do jato livre; seção e diâmetro do bocal.
  \item Potência cinética levada pelo jato; mostre que ela vale $\rho gD_VH$
        e calcule-a.
  \item Massa e energia cinética da água contida no conduto forçado
        na vazão de projeto; tempo de trânsito de uma partícula pelo tubo.
  \item Mostre que a carga $H$ se reparte, na vazão de projeto, entre a
        pressão no fundo do conduto e a energia cinética
        da água, e que o bocal converte a primeira na
        segunda: qual é a pressão logo depois do bocal?
  \item Um \omterm{def:b2:euler-bernoulli:perfect}{fluido perfeito} não tem perda de carga, um real tem: a perda numa
        tubulação vale $\Delta P = \lambda(\ell/D)\,\tfrac12\rho v^2$ com $\lambda
        \approx 0.015$ para aço liso. Perda de carga em metros de coluna d'água e
        fração da potência perdida.
  \item Um engenheiro propõe um conduto de \qty{2}{m} para economizar aço. Refaça
        a questão 7; comente.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- Medir a vazão.}
Um Venturi é embutido no conduto forçado, com garganta de \qty{2.0}{m}.
\begin{enumerate}[resume]
  \item Celeridade na garganta e queda de pressão entre o tubo e a
        garganta na vazão de projeto.
  \item A queda é lida num manômetro de mercúrio ($\rho_{\text{Hg}} =
        \qty{13600}{kg/m^3}$): diferença de altura.
  \item Mostre que a vazão é proporcional à raiz quadrada da
        leitura do manômetro, e dê a leitura na metade da vazão
        de projeto.
  \item Por que as tomadas de pressão devem ficar rentes à parede e não
        avançar para dentro do escoamento?
  \item Um trecho do tubo passa, por razões topográficas, sobre um
        ponto \qty{6}{m} \emph{acima} da superfície do reservatório. Pressão ali
        na vazão de projeto; o que acontece, e o que o projetista deve fazer?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Regimes não permanentes.}
O reservatório tem área superficial $A = \qty{2.0}{km^2}$.
\begin{enumerate}[resume]
  \item Taxa com que o nível do reservatório baixa na vazão de projeto, em
        centímetros por hora.
  \item A turbina é parada e o bocal (seção da questão 4)
        é deixado aberto ao ar: o reservatório esvazia-se por gravidade. Escreva
        o balanço de massa e ache o nível $h(t)$ (escoamento de
        Torricelli quase permanente).
  \item Tempo para o nível baixar \qty{10}{m}; compare com o tempo
        na vazão de projeto constante.
  \item Partida da usina: o conduto está cheio e o bocal é aberto
        em $t = 0$. Tomando a celeridade uniforme ao longo do conduto e
        desprezando a geometria do bocal, integre Euler ao longo do
        tubo para obter $\ell\,\dd v/\dd t = gH - v^2/2$, resolva e dê
        a constante de tempo.
  \item Durante essa partida, no instante em que $v$ vale metade de seu valor
        final, qual é a pressão no fundo do conduto forçado?
        Explique o sinal da diferença em relação ao valor permanente.
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Fechar a válvula.}
\begin{enumerate}[resume]
  \item A válvula da turbina é fechada em $\tau = \qty{10}{s}$; suponha que a
        água do conduto desacelere uniformemente. Usando Euler ao longo
        do tubo, estime a sobrepressão na válvula, em bar.
  \item Num fechamento rápido a água não é incompressível: uma onda de
        pressão sobe pelo tubo a $c = \qty{1400}{m/s}$ (a velocidade do
        som na água dentro de um tubo de aço --- \cref{ch:b2:sound-waves}),
        e a sobrepressão vale $\Delta P = \rho c\,\Delta v$ (Joukowsky).
        Valor para uma parada súbita a partir da vazão de projeto; compare com a
        pressão estática. Quanto tempo a onda leva para chegar ao
        reservatório e voltar, e o que isso diz sobre o significado
        de "rápido"?
  \item Para proteger o conduto instala-se uma \emph{chaminé de equilíbrio} --- um
        poço vertical aberto de seção $A_s = \qty{100}{m^2}$ --- ligada onde um
        túnel horizontal de comprimento $L = \qty{2000}{m}$ e seção $A =
        \qty{7.0}{m^2}$ vindo do reservatório encontra o conduto forçado. Depois de um
        fechamento súbito, a água do túnel continua a se mover e o
        nível $z$ no poço sobe. Escreva a conservação da massa entre
        o túnel e o poço.
  \item Integre Euler ao longo do túnel (\omterm{def:b2:euler-bernoulli:perfect}{fluido perfeito}) para mostrar que
        $z$ obedece a $\ddot z + \dfrac{gA}{LA_s}z = 0$; período da
        oscilação.
  \item Subida máxima do nível no poço depois de um fechamento a partir da
        vazão de projeto (argumento de energia: a energia cinética da água do
        túnel torna-se energia potencial no poço).
  \item Faça o balanço: as cinco pressões encontradas nesta usina (estática na
        barragem, estática na turbina, dinâmica no tubo, golpe de aríete
        e chaminé de equilíbrio) e a lei que dá cada uma.
\end{enumerate}
\end{problem}
