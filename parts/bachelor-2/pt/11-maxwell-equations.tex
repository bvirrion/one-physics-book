\chapter{As equações de Maxwell}\label{ch:b2:maxwell-equations}

Em 1865 James Clerk Maxwell escreveu as leis que Coulomb, Ampère
e Faraday tinham encontrado uma a uma, acrescentou à de Ampère um termo que
nenhuma experiência ainda exigia e descobriu que suas equações admitiam
ondas que viajavam com uma celeridade que ele podia calcular a partir das constantes da
eletricidade e do magnetismo --- a velocidade da luz. Eletricidade,
magnetismo e óptica eram um só assunto. O volume do Ano 1 enunciou as
leis como integrais sobre superfícies e circuitos; este capítulo as escreve como
quatro equações \emph{locais}, válidas em cada ponto e em cada instante,
com os operadores vetoriais que tornam essa escrita possível, e extrai
as primeiras consequências: a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento}, os potenciais, o
comportamento dos campos numa interface e as aproximações sob as
quais os circuitos do volume do Ano 1 são exatos.

\section{Operadores vetoriais}

\begin{definition}[Gradiente, divergente, rotacional, laplaciano]\label{def:b2:maxwell-equations:operators}
\index{gradiente}\index{divergente}\index{rotacional}\index{laplaciano}\index{nabla}
Para um campo escalar $f$ e um campo vetorial $\vect A$, em coordenadas
cartesianas, com o vetor simbólico $\vect\nabla = (\partial_x, \partial_y,
\partial_z)$:
\begin{align*}
\operatorname{\vect{grad}}f &= \vect\nabla f = (\partial_xf,\ \partial_yf,\ \partial_zf) ,\\
\operatorname{div}\vect A &= \vect\nabla\cdot\vect A = \partial_xA_x + \partial_yA_y + \partial_zA_z ,\\
\operatorname{\vect{curl}}\vect A &= \vect\nabla\wedge\vect A = (\partial_yA_z - \partial_zA_y,\ \partial_zA_x - \partial_xA_z,\ \partial_xA_y - \partial_yA_x) ,\\
\Delta f &= \operatorname{div}\operatorname{\vect{grad}}f = \partial_x^2f + \partial_y^2f + \partial_z^2f ,
\end{align*}
e $\Delta\vect A$ é o laplaciano tomado em cada componente cartesiana.
Seus significados independem das coordenadas: $\operatorname{\vect{grad}}f
\cdot\dd\vect l = \dd f$ é a variação de $f$ ao longo de $\dd\vect l$; $\operatorname{div}
\vect A$ é o fluxo de $\vect A$ que sai pela superfície de um volume
pequeno, por unidade de volume; $\operatorname{\vect{curl}}\vect A\cdot\vect n$ é a
circulação de $\vect A$ em torno de uma espira pequena de normal $\vect n$, por unidade de
área.
\end{definition}

\begin{theorem}[Teoremas do divergente e de Stokes]\label{thm:b2:maxwell-equations:theorems}
\index{teorema do divergente}\index{teorema de Ostrogradski}\index{teorema de Stokes}
Para uma superfície fechada $\Sigma$ que limita o volume $V$, e uma curva
fechada orientada $C$ que limita a superfície $S$ (normal $\vect n$ dada pela
regra da mão direita),
\[
\iint_\Sigma\vect A\cdot\vect n\,\dd S = \iiint_V\operatorname{div}\vect A\,\dd\tau ,
\qquad
\oint_C\vect A\cdot\dd\vect l = \iint_S\operatorname{\vect{curl}}\vect A\cdot\vect n\,\dd S .
\]
\end{theorem}
\admitted

\begin{remark}[O que é admitido e por que é crível]\label{rem:b2:maxwell-equations:whyadmitted}
Os dois teoremas são demonstrados no volume de matemática do Ano 3; o
volume de matemática deste ano para em sua versão plana
(Green--Riemann) e nas integrais múltiplas. Seu conteúdo, porém, é
o que usamos no \cref{ch:b2:fluid-kinematics}: preencha o volume com
caixas pequenas (e a superfície da espira com espiras pequenas) --- os fluxos
pelas faces interiores (as circulações pelas arestas interiores)
cancelam-se aos pares, e resta apenas a fronteira. Duas identidades
seguem das definições e são usadas o tempo todo:
$\operatorname{\vect{curl}}\operatorname{\vect{grad}}f = \vect 0$,
$\operatorname{div}\operatorname{\vect{curl}}\vect A = 0$, e
$\operatorname{\vect{curl}}\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}
\operatorname{div}\vect A - \Delta\vect A$. Para campos com simetria, as
fórmulas úteis são: para um campo radial $A_r(r)\vect e_r$, $\operatorname{div}
= \frac1{r^2}\partial_r(r^2A_r)$ (esféricas) ou $\frac1r\partial_r(rA_r)$
(cilíndricas); para um campo azimutal $A_\theta(r)\vect e_\theta$ em torno de um
eixo, $(\operatorname{\vect{curl}})_z = \frac1r\partial_r(rA_\theta)$; e para $f(r)$,
$\Delta f = \frac1{r^2}\partial_r(r^2\partial_rf) = \frac1r\partial_r^2(rf)$ (esféricas),
$\frac1r\partial_r(r\partial_rf)$ (cilíndricas).
\end{remark}

\section{As quatro equações}

\begin{theorem}[Equações de Maxwell]\label{thm:b2:maxwell-equations:maxwell}
\index{equações de Maxwell}\index{equação de Maxwell--Gauss}\index{equação de Maxwell--Faraday}\index{equação de Maxwell--Ampère}\index{corrente de deslocamento}
No vácuo (e na matéria descrita por suas densidades totais de carga e de
corrente $\rho$ e $\vect j$), os campos elétrico e magnético obedecem, em
cada ponto e em cada instante, a
\begin{align*}
\operatorname{div}\vect E &= \frac\rho{\varepsilon_0} &&\text{(Maxwell--Gauss)} , &
\operatorname{\vect{curl}}\vect E &= -\frac{\partial\vect B}{\partial t} &&\text{(Maxwell--Faraday)} ,\\
\operatorname{div}\vect B &= 0 &&\text{(Maxwell--fluxo)} , &
\operatorname{\vect{curl}}\vect B &= \mu_0\vect j + \mu_0\varepsilon_0\frac{\partial\vect E}{\partial t} &&\text{(Maxwell--Ampère)} ,
\end{align*}
com $\varepsilon_0 = \qty{8.85e-12}{F/m}$, $\mu_0 = 4\pi \times \qty{1e-7}{H/m}$ (e
$\varepsilon_0\mu_0 = 1/c^2$). Os campos agem sobre as cargas pela força de
Lorentz $\vect F = q(\vect E + \vect v\wedge\vect B)$. O termo $\varepsilon_0\partial_t\vect E$
é a densidade de \emph{\omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento}} de Maxwell.
\end{theorem}

\begin{proof}[Equivalência com as leis integrais do volume do Ano 1]
Aplique o teorema do \omterm{def:b2:maxwell-equations:operators}{divergente} às duas primeiras: $\iint_\Sigma\vect E\cdot
\vect n\,\dd S = Q_{\text{int}}/\varepsilon_0$ (Gauss), e o fluxo de $\vect B$
por qualquer superfície fechada é nulo. Aplique Stokes às duas últimas:
$\oint_C\vect E\cdot\dd\vect l = -\dd\Phi_B/\dd t$ (Faraday, para uma espira fixa)
e $\oint_C\vect B\cdot\dd\vect l = \mu_0I_{\text{enl}} + \mu_0\varepsilon_0\,\dd\Phi_E/\dd t$
(Ampère, completada). Reciprocamente, as leis integrais, valendo para todo
volume e toda espira, implicam as locais. As equações locais são, portanto,
as leis do Ano 1 mais um termo novo, justificado adiante; elas são tomadas como
os postulados do eletromagnetismo.
\end{proof}

\begin{proposition}[Por que a corrente de deslocamento é necessária]\label{prop:b2:maxwell-equations:displacement}
Tomar o \omterm{def:b2:maxwell-equations:operators}{divergente} da \omterm{thm:b2:maxwell-equations:maxwell}{equação de Maxwell--Ampère} e usar
$\operatorname{div}\operatorname{\vect{curl}} = 0$ e Maxwell--Gauss dá
\[
0 = \mu_0\operatorname{div}\vect j + \mu_0\varepsilon_0\,\partial_t\operatorname{div}\vect E
= \mu_0\Bigl(\operatorname{div}\vect j + \frac{\partial\rho}{\partial t}\Bigr) :
\]
as equações contêm a conservação da carga. Sem o termo de
deslocamento elas exigiriam $\operatorname{div}\vect j = 0$ sempre
--- falso num capacitor que se carrega, onde a corrente de condução para
na placa. Ali, entre as placas, $\varepsilon_0\partial_t\vect E$ transporta
exatamente a corrente que chega pelo fio, e o campo magnético
circula em torno do vão como em torno de um fio.
\end{proposition}

\begin{proof}
O \omterm{def:b2:maxwell-equations:operators}{divergente} dos dois lados, como se disse; para o capacitor, o fluxo de
$\varepsilon_0\partial_t\vect E$ por uma superfície entre as placas vale
$\varepsilon_0S\,\dd E/\dd t = \dd Q/\dd t = I$.
\end{proof}

\begin{example}[O campo dentro de um capacitor que se carrega]\label{ex:b2:maxwell-equations:capacitor}
Placas circulares de raio $R$, separação $d \ll R$, carregadas por uma corrente
$I$: entre elas $E = Q/\varepsilon_0\pi R^2$ é uniforme e cresce a $\dd E/
\dd t = I/\varepsilon_0\pi R^2$. Por simetria $\vect B = B(r)\vect e_\theta$; Ampère--Maxwell
num círculo de raio $r < R$ (nenhuma corrente de condução o atravessa): $2\pi rB
= \mu_0\varepsilon_0\pi r^2\,\dd E/\dd t$, logo $B = \mu_0Ir/2\pi R^2$ --- o campo de uma
corrente uniforme $I$ espalhada pelo disco, contínuo em $r = R$ com o
$\mu_0I/2\pi r$ do fio, do lado de fora. Para $I = \qty{1}{A}$, $R = \qty{5}{cm}$: $B(R) =
\qty{4}{\micro T}$, mensurável; o termo de Maxwell é real.
\end{example}

\begin{remark}[Linearidade; o que as equações dizem]\label{rem:b2:maxwell-equations:linearity}
As equações são lineares nos campos e em suas fontes: os campos se
superpõem. Duas delas (a do fluxo magnético e Maxwell--Faraday) não trazem fontes e
exprimem estrutura --- as linhas de campo elétrico começam e terminam em cargas,
as de campo magnético nunca terminam; as outras duas dizem como cargas e
correntes, e a variação de cada campo, criam os campos. Lidas
juntas, um $\vect B$ variável cria um $\vect E$ que circula, e um $\vect E$
variável cria um $\vect B$ que circula: os dois podem se sustentar sem
carga alguma --- é a onda do
\cref{ch:b2:plane-waves-polarization}, e a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento} é
o que a torna possível.
\end{remark}

\section{Potenciais}

\begin{proposition}[Potenciais escalar e vetor]\label{prop:b2:maxwell-equations:potentials}
\index{potencial vetor}\index{potencial escalar}\index{calibre}\index{calibre de Coulomb}\index{equação de Poisson}
Como $\operatorname{div}\vect B = 0$ e $\operatorname{\vect{curl}}\vect E = -\partial_t
\vect B$, os campos derivam de um \emph{\omterm{prop:b2:maxwell-equations:potentials}{potencial vetor}} $\vect A$ e de um
\emph{\omterm{prop:b2:maxwell-equations:potentials}{potencial escalar}} $V$:
\[
\vect B = \operatorname{\vect{curl}}\vect A , \qquad
\vect E = -\operatorname{\vect{grad}}V - \frac{\partial\vect A}{\partial t} ,
\]
determinados a menos de uma \emph{transformação de calibre} $\vect A \to \vect A +
\operatorname{\vect{grad}}\chi$, $V \to V - \partial_t\chi$, que deixa os campos
inalterados. No \emph{\omterm{prop:b2:maxwell-equations:potentials}{calibre de Coulomb}} $\operatorname{div}\vect A = 0$, a lei de
Gauss torna-se a \emph{\omterm{prop:b2:maxwell-equations:potentials}{equação de Poisson}}
\[
\Delta V = -\frac\rho{\varepsilon_0} ,
\]
e, em regime estacionário, a lei de Ampère torna-se $\Delta\vect A = -\mu_0\vect j$.
Em estática, $V$ é o potencial eletrostático do volume do Ano 1.
\end{proposition}

\begin{proof}
Um campo de \omterm{def:b2:maxwell-equations:operators}{divergente} nulo é um \omterm{def:b2:maxwell-equations:operators}{rotacional} (admitido, como a recíproca
de que um campo de \omterm{def:b2:maxwell-equations:operators}{rotacional} nulo é um \omterm{def:b2:maxwell-equations:operators}{gradiente}, ambas do volume de
matemática do Ano 2 para regiões simplesmente conexas); então $\operatorname{\vect{curl}}
(\vect E + \partial_t\vect A) = \vect 0$, de modo que $\vect E + \partial_t\vect A$ é um \omterm{def:b2:maxwell-equations:operators}{gradiente}.
Calibre: $\operatorname{\vect{curl}}\operatorname{\vect{grad}}\chi = \vect 0$ e os
termos $\partial_t\operatorname{\vect{grad}}\chi$ se cancelam em $\vect E$. Poisson: $\operatorname{div}
\vect E = -\Delta V - \partial_t\operatorname{div}\vect A$. Estática: $\operatorname{\vect{curl}}
\operatorname{\vect{curl}}\vect A = \operatorname{\vect{grad}}\operatorname{div}\vect A - \Delta\vect A
= \mu_0\vect j$.
\end{proof}

\begin{example}[Dois potenciais vetores]\label{ex:b2:maxwell-equations:vectorpot}
Um campo uniforme $\vect B_0$ deriva de $\vect A = \tfrac12\vect B_0\wedge\vect r$
(verifique: $\operatorname{\vect{curl}}(\tfrac12\vect B_0\wedge\vect r) = \vect B_0$, e
$\operatorname{div}\vect A = 0$); dentro de um solenoide infinito isso dá $A_\theta =
Br/2$, e fora $A_\theta = BR^2/2r$ --- um potencial que não se anula
onde o campo se anula, e cuja circulação ao longo de uma espira é o fluxo
de $\vect B$ por ela: $\oint\vect A\cdot\dd\vect l = \Phi_B$ (Stokes). A lei de Faraday
diz então $\vect E = -\partial_t\vect A$ para o campo induzido.
\end{example}

\section{Os campos numa interface}

\begin{proposition}[Relações de passagem]\label{prop:b2:maxwell-equations:boundary}
\index{relações de passagem (campos)}\index{corrente superficial}
Através de uma superfície que carrega a densidade superficial de carga $\sigma$ e a
densidade superficial de corrente $\vect j_s$ (\unit{A/m}), com $\vect n_{12}$ a
normal unitária do meio 1 para o meio 2,
\[
\vect E_2 - \vect E_1 = \frac\sigma{\varepsilon_0}\,\vect n_{12} , \qquad
\vect B_2 - \vect B_1 = \mu_0\,\vect j_s\wedge\vect n_{12} :
\]
a componente tangencial de $\vect E$ e a componente normal de $\vect B$
são contínuas; a componente normal de $\vect E$ salta de $\sigma/\varepsilon_0$,
e a componente tangencial de $\vect B$ de $\mu_0j_s$. Na superfície de um
condutor perfeito ($\vect E = \vect B = \vect 0$ no interior, nos casos do
\cref{ch:b2:wave-interfaces}), $\vect E$ é normal e vale $\sigma\vect n/
\varepsilon_0$, e $\vect B$ é tangencial e vale $\mu_0\vect j_s\wedge\vect n$.
\end{proposition}

\begin{proof}
Componentes normais: as leis de Gauss e do fluxo num cilindro achatado
que atravessa a superfície, de área $\dd S$ e altura tendendo a zero: $(E_{2n}
- E_{1n})\dd S = \sigma\dd S/\varepsilon_0$, $(B_{2n} - B_{1n})\dd S = 0$. Componentes
tangenciais: as leis de Faraday e de Ampère num pequeno retângulo
que atravessa a superfície, de comprimento $\dd l$ e altura tendendo a zero: os
fluxos de $\partial_t\vect B$ e de $\partial_t\vect E$ pelo retângulo se anulam
com sua área, mas uma \omterm{prop:b2:maxwell-equations:boundary}{corrente superficial} $j_s\,\dd l$ o atravessa:
$(E_{2t} - E_{1t})\dd l = 0$, $(B_{2t} - B_{1t})\dd l = \mu_0j_s\dd l$ para a
direção tangencial perpendicular a $\vect j_s$.
\end{proof}

\begin{example}[Plano carregado, lâmina de corrente, solenoide]\label{ex:b2:maxwell-equations:sheets}
Um plano com apenas $\sigma$ no espaço: por simetria $\vect E = \pm E\vect n$ nos
seus dois lados, e o salto $2E = \sigma/\varepsilon_0$ dá $E = \sigma/2\varepsilon_0$ ---
o resultado do Ano 1 em uma linha. Uma lâmina plana de corrente $j_s$: $B =
\mu_0j_s/2$ de cada lado, antiparalelos. Um solenoide longo de $n$ espiras por
metro percorrido por $I$ é uma lâmina cilíndrica de corrente $j_s = nI$: o salto
$\mu_0nI$ através dela, com $B = 0$ do lado de fora, dá $B = \mu_0nI$ no interior.
\end{example}

\section{Regimes quase estacionários}

\begin{definition}[As aproximações quase estacionárias]\label{def:b2:maxwell-equations:arqs}
\index{aproximação quase estacionária}\index{regime quase estacionário}
Um sistema de tamanho $L$ excitado na frequência $f$ é \emph{quase estacionário}
quando $L \ll \lambda = c/f$: o tempo $L/c$ de que os campos precisam para atravessá-lo é
desprezível diante do período, e todos os seus pontos "veem" as
fontes no mesmo instante. Usam-se dois limites:
\begin{itemize}
  \item o regime quase estacionário \emph{magnético} (circuitos com
        correntes, bobinas, transformadores, indução): a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de
        deslocamento} é descartada da lei de Ampère, $\operatorname{\vect{curl}}\vect B
        = \mu_0\vect j$ (de modo que $\operatorname{div}\vect j = 0$ e as correntes circulam em
        circuitos fechados), enquanto a lei de Faraday é mantida por inteiro;
  \item o regime quase estacionário \emph{elétrico} (um capacitor que
        se carrega): o termo de Faraday é descartado, $\vect E$ é o
        campo eletrostático das cargas instantâneas, e a
        \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento} é mantida para achar $\vect B$.
\end{itemize}
\end{definition}

\begin{proof}[Justificativa]
No regime magnético, compare os dois termos da lei de Ampère num
condutor: $\mu_0\varepsilon_0\partial_t E \sim \mu_0\varepsilon_0\omega E$ contra $\mu_0\gamma E$,
uma razão $\varepsilon_0\omega/\gamma = \omega\tau_r \sim 10^{-17}$ a \qty{50}{Hz} no cobre;
fora do condutor a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento} é da ordem de
$\varepsilon_0\omega E$ com $E \sim \omega AB$, ou seja, menor que o efeito da corrente
de condução por $(\omega L/c)^2 = (L/\lambda)^2 \times 4\pi^2$. No regime
elétrico o campo induzido $\omega BL \sim \omega^2\varepsilon_0\mu_0L^2E$ é menor que $E$
pelo mesmo $(L/\lambda)^2$.
\end{proof}

\begin{example}[Circuitos e antenas]\label{ex:b2:maxwell-equations:sizes}
Um circuito de rede a \qty{50}{Hz} ($\lambda = \qty{6000}{km}$) é quase estacionário
até o tamanho de um país --- é por isso que as leis de Kirchhoff e o
transformador funcionam. Um circuito de \qty{10}{cm} é quase estacionário até uns
\qty{100}{MHz}; a \qty{1}{GHz} ($\lambda = \qty{30}{cm}$) seus fios são antenas,
as correntes diferem ao longo de um fio e o circuito irradia: o domínio do
\cref{ch:b2:dipole-radiation} e do projeto de micro-ondas, em que uma trilha numa
placa é uma \omterm{prop:b2:dispersion-wave-packets:coax}{linha de transmissão}.
\end{example}

\begin{method}[Usar as equações locais]\label{met:b2:maxwell-equations:method}
(1) Use as simetrias e invariâncias das fontes para fixar a
direção e as variáveis dos campos (como no volume do Ano 1).
(2) Escolha a forma: integral (Gauss, Ampère numa superfície ou espira
bem escolhida) quando houver simetria bastante; local quando for preciso uma
equação diferencial (Poisson para $V$, a equação de difusão ou a \omterm{thm:b2:waves-on-strings:equation}{equação de onda}
para os campos). (3) Num problema dependente do tempo, decida o regime:
estático, quase estacionário magnético ou elétrico, ou Maxwell completo. (4)
Nas interfaces, use as relações de passagem --- são o único lugar
onde entram as cargas e \omterm{prop:b2:maxwell-equations:boundary}{correntes superficiais}. (5) Verifique
$\operatorname{div}\vect B = 0$ e as dimensões.
\end{method}

\section{Exercícios}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:1}
Calcule o \omterm{def:b2:maxwell-equations:operators}{divergente} e o \omterm{def:b2:maxwell-equations:operators}{rotacional} de (a) $\vect A = (x, y, z)$; (b)
$\vect A = (-y, x, 0)$; (c) $\vect A = (yz, zx, xy)$; (d) $\vect A = \vect r/r^3$
(use a fórmula radial, $r \ne 0$); (e) $\vect A = (0, 0, x^2)$. Quais são
\omterm{def:b2:maxwell-equations:operators}{gradientes}, quais são rotacionais, quais não são nem uma coisa nem outra?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:2}
(a) Verifique as dimensões de cada termo das quatro \omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell}.
(b) Calcule $1/\sqrt{\varepsilon_0\mu_0}$. (c) Razão entre a densidade de \omterm{thm:b2:maxwell-equations:maxwell}{corrente
de deslocamento} e a \omterm{def:b2:charges-currents-conduction:densities}{densidade de corrente} de condução no cobre a \qty{50}{Hz}, na
água do mar ($\gamma = \qty{5}{S/m}$) a \qty{1}{GHz}, no vidro ($\gamma =
\qty{1e-12}{S/m}$, com $\varepsilon_0$ por simplicidade) a \qty{50}{Hz}: qual deles é
"um condutor" em cada caso?
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:3}
Um capacitor de placas circulares ($R = \qty{10}{cm}$, $d = \qty{2}{mm}$) é
carregado por $I = \qty{0.5}{A}$. Taxa de variação de $E$; densidade de \omterm{thm:b2:maxwell-equations:maxwell}{corrente
de deslocamento}; $B$ em $r = \qty{5}{cm}$ e em $r = \qty{20}{cm}$ (no
plano a meia altura entre as placas); compare o último com o campo do
fio de alimentação à mesma distância.
\end{exercise}

\begin{exercise}[$\star$]\label{exo:b2:maxwell-equations:4}
(a) $\vect E = kx\,\vect e_x$ preenche uma região: \omterm{def:b2:charges-currents-conduction:densities}{densidade de carga} ali. (b)
Dentro de uma bola de raio $R$, $\vect E = (\rho_0r/3\varepsilon_0)\vect e_r$: verifique com
o \omterm{def:b2:maxwell-equations:operators}{divergente} radial que $\rho = \rho_0$, e recupere o campo externo
pela lei de Gauss integral. (c) $\vect B = B_0(y/a)\vect e_x$: é um campo
magnético possível? Que \omterm{def:b2:charges-currents-conduction:densities}{densidade de corrente} o sustenta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:5}
\emph{\omterm{prop:b2:maxwell-equations:potentials}{Potenciais vetores}.} (a) Mostre que $\vect A = \tfrac12\vect B_0\wedge\vect r$
dá $\operatorname{\vect{curl}}\vect A = \vect B_0$ e $\operatorname{div}\vect A = 0$.
(b) O mesmo vale para $\vect A' = B_0x\,\vect e_y$: ache a função de calibre $\chi$ com
$\vect A' = \vect A + \operatorname{\vect{grad}}\chi$. (c) Solenoide de raio $R$,
campo $B$ interno: verifique $A_\theta = Br/2$ ($r < R$) e $BR^2/2r$ ($r > R$)
com $(\operatorname{\vect{curl}}\vect A)_z = \frac1r\partial_r(rA_\theta)$; continuidade em $R$.
(d) Circulação de $\vect A$ num círculo de raio $r > R$ e o fluxo de
$\vect B$: o que isso diz sobre a fem induzida numa espira fora de um
solenoide cuja corrente varia?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:6}
\emph{Relações de passagem.} (a) Um condutor em equilíbrio
eletrostático carrega $\sigma$ em sua superfície: campo logo do lado de fora (a partir do
salto, com $\vect E = \vect 0$ no interior); para $\sigma = \qty{1}{\micro C/m^2}$. (b) Dois
planos paralelos com $+\sigma$ e $-\sigma$: campo em toda parte, pelos saltos.
(c) Um solenoide longo ($n = \qty{2000}{m^{-1}}$, $I = \qty{5}{A}$) como lâmina de
corrente: $j_s$, o salto de $B$, e $B$ no interior. (d) Uma bobina toroidal: por que $B$ é
nulo do lado de fora e vale $\mu_0NI/2\pi r$ no interior, pelo mesmo salto?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:7}
\emph{Poisson em uma dimensão.} Entre dois eletrodos planos em $x = 0$
($V = 0$) e $x = d$ ($V = U$), uma \omterm{def:b2:charges-currents-conduction:densities}{densidade de carga} uniforme $\rho_0$ preenche o
vão (uma carga espacial). (a) Resolva a \omterm{prop:b2:maxwell-equations:potentials}{equação de Poisson} para $V(x)$. (b) Campo
$E(x)$; onde ele se anula se $\rho_0 > 0$ e grande? (c) Cargas superficiais nos
dois eletrodos, pela relação de passagem. (d) Com $\rho_0 = 0$
recupere o capacitor; com $U = 0$, esboce $V$.
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:8}
\emph{Quase estacionário ou não.} (a) $\lambda$ a \qty{50}{Hz}, \qty{1}{MHz},
\qty{100}{MHz} e \qty{2.4}{GHz}. (b) Até que frequência um circuito de \qty{10}{cm}
é quase estacionário (critério $L < \lambda/10$)? E um chip de \qty{1}{cm}? E uma
rede elétrica de \qty{1000}{km}? (c) Num capacitor de raio $R = \qty{5}{cm}$ a
\qty{1}{MHz}, estime a razão entre o campo elétrico induzido (pelo
$\vect B$ variável dos exercícios anteriores) e o campo principal: $(\omega R/c)^2
/4$. (d) Por que a frequência da rede de um continente é um problema
quase estacionário, embora a rede tenha \qty{1000}{km} de ponta a ponta?
\end{exercise}

\begin{exercise}[$\star\star$]\label{exo:b2:maxwell-equations:9}
Mostre que a lei de Ampère sem a \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento}, $\operatorname{\vect{curl}}
\vect B = \mu_0\vect j$, contradiz a conservação da carga sempre que $\partial_t\rho
\ne 0$. No capacitor do \cref{exo:b2:maxwell-equations:3}, avalie
$\operatorname{div}\vect j$ com média sobre uma placa de espessura $e = \qty{0.1}{mm}$
(onde a corrente $I$ para), e o $\partial_t\rho$ correspondente.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:10}
\emph{O betatron.} Um campo magnético uniforme $B(t)\vect e_z$ preenche um
cilindro de raio $R$, crescendo a $\dd B/\dd t = \dot B$. (a) Por simetria
$\vect E = E_\theta(r)\vect e_\theta$: use a lei de Faraday integral para achar $E_\theta$
para $r < R$ e $r > R$. (b) Não há carga em parte alguma: comente um
campo elétrico de linhas fechadas. (c) Quanta energia por volta ganha um elétron
num círculo de raio $r_0 < R$? Para $\dot B = \qty{10}{T/s}$,
$r_0 = \qty{0.5}{m}$: energia por volta em eV; voltas necessárias para \qty{20}{MeV}.
(d) Para que o raio da órbita fique constante, a quantidade de movimento $p = eBr_0$ (dada
pela força magnética) deve crescer como o ganho de energia pelo campo induzido
prescreve: mostre a condição $B(r_0) = \tfrac12\langle B\rangle_{\text{disco}}$
--- a condição do betatron.
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:11}
\emph{As ondas de Maxwell.} No vácuo ($\rho = 0$, $\vect j = \vect 0$): (a) tome
o \omterm{def:b2:maxwell-equations:operators}{rotacional} da \omterm{thm:b2:maxwell-equations:maxwell}{equação de Maxwell--Faraday}, use $\operatorname{\vect{curl}}
\operatorname{\vect{curl}} = \operatorname{\vect{grad}}\operatorname{div} - \Delta$ e as demais
equações para obter $\Delta\vect E = \mu_0\varepsilon_0\,\partial_t^2\vect E$. (b) O mesmo para
$\vect B$. (c) Essa é a \omterm{thm:b2:waves-on-strings:equation}{equação de d'Alembert} em três dimensões: qual a celeridade?
Calcule-a a partir de $\varepsilon_0$ e $\mu_0$. (d) Por que esse número convenceu
Maxwell de que a luz é eletromagnética, e que medida de
Weber e Kohlrausch ele usou?
\end{exercise}

\begin{exercise}[$\star\star\star$]\label{exo:b2:maxwell-equations:12}
\emph{Campos num condutor em baixa frequência.} Num condutor ôhmico
($\vect j = \gamma\vect E$) em \omterm{def:b2:maxwell-equations:arqs}{regime quase estacionário} magnético: (a) tome
o \omterm{def:b2:maxwell-equations:operators}{rotacional} da lei de Ampère e use a de Faraday para mostrar que $\Delta\vect B = \mu_0\gamma\,
\partial_t\vect B$; por que é uma equação de difusão? (b) Tempo característico
para $\vect B$ penetrar uma profundidade $d$: $\tau \sim \mu_0\gamma d^2$; números para uma
placa de cobre de \qty{1}{cm} e para o núcleo da Terra ($\gamma \approx \qty{5e5}{S/m}$,
$d \approx \qty{3000}{km}$). (c) Na frequência $f$, a profundidade atingida em um
período: mostre que ela é da ordem de $\delta = \sqrt{2/\mu_0\gamma\omega}$; valor para o cobre
a \qty{50}{Hz} e a \qty{1}{MHz} (a \omterm{def:b2:dispersion-wave-packets:complex}{profundidade de penetração} do
\cref{ch:b2:waves-in-media}). (d) O que (b) implica para as variações
temporais do campo magnético terrestre?
\end{exercise}

\section{Problema: O capacitor, o betatron e os potenciais}

\begin{problem}[{Problema de fim de semana --- as equações de Maxwell em ação num
capacitor que se carrega, no acelerador que funciona só com a lei de Faraday
e nos potenciais que as resolvem}]
\label{pb:b2:maxwell-equations:1}

\medskip
\noindent\textbf{Parte I --- O capacitor que se carrega.}
Placas circulares de raio $R = \qty{10}{cm}$, separação $d = \qty{1.0}{mm}$,
ar; uma corrente $I(t) = I_0\cos\omega t$ com $I_0 = \qty{1.0}{A}$, $f = \qty{1.0}{MHz}$
circula no fio de alimentação ao longo do eixo. Efeitos de borda desprezados.
\begin{enumerate}
  \item Carga $Q(t)$ e campo $E(t)$ entre as placas; amplitude de
        $E$.
  \item Densidade de \omterm{thm:b2:maxwell-equations:maxwell}{corrente de deslocamento} entre as placas; verifique que
        seu fluxo por uma placa é igual a $I(t)$.
  \item Campo magnético $B(r, t)$ entre as placas para $r < R$, pela
        lei integral de Ampère--Maxwell; amplitude em $r = R$.
  \item Campo para $r > R$ (ainda entre os planos das placas):
        compare com o campo do fio ao mesmo $r$; há
        descontinuidade em $r = R$?
  \item O $\vect B$ variável induz, pela lei de Faraday, uma correção
        $\vect E_1$ ao campo elétrico: usando $\operatorname{\vect{curl}}\vect E_1
        = -\partial_t\vect B$ com $\vect E_1 = E_1(r)\vect e_z$ (axial), mostre que
        $E_1(r) - E_1(0) = \dfrac{\mu_0\varepsilon_0r^2}4\,\dfrac{\dd^2E}{\dd t^2}$, e
        estime $|E_1(R) - E_1(0)|/|E|$.
  \item Conclua sobre a validade do tratamento quase estacionário
        elétrico a \qty{1}{MHz}; em que frequência a correção
        chegaria a $10\%$? (A solução exata é uma função de Bessel: o
        capacitor virou uma cavidade.)
  \item Energia: a energia elétrica $\tfrac12\varepsilon_0E^2$ armazenada entre as
        placas, valor máximo; a energia magnética $\int B^2/2\mu_0\,\dd\tau$
        no instante em que $E = 0$; a razão entre elas e sua relação com a
        estimativa da questão 5.
\end{enumerate}

\medskip
\noindent\textbf{Parte II --- O betatron.}
Um eletroímã produz um campo $\vect B = B(r, t)\vect e_z$ simétrico
em relação ao eixo $z$; os elétrons circulam num anel em vácuo de raio $r_0$.
\begin{enumerate}[resume]
  \item Mostre, a partir de Maxwell--Faraday na forma integral, que o
        campo elétrico na órbita vale $E_\theta(r_0) = -\dfrac1{2\pi r_0}
        \dfrac{\dd\Phi}{\dd t}$, com $\Phi$ o fluxo através da órbita.
  \item Equação tangencial do movimento de um elétron (carga $-e$):
        $\dd p/\dd t = eE_\theta$ em módulo; a energia ganha por volta vale
        $e\,|\dd\Phi/\dd t|$.
  \item Equilíbrio radial no círculo: $p = eB(r_0)r_0$. Para que $r_0$
        permaneça constante, mostre que $B(r_0, t) = \Phi(t)/2\pi r_0^2$, ou seja, o
        campo na órbita deve ser metade do campo médio no seu interior.
  \item Por que os elétrons não podem ser acelerados por um campo uniforme?
        Como a condição é realizada na prática (formato das peças
        polares)?
  \item $r_0 = \qty{0.50}{m}$, $B(r_0)$ elevado de $0$ a \qty{0.40}{T} em
        \qty{5}{ms}: quantidade de movimento final; energia final (use $E = pc$ para esses
        elétrons ultrarrelativísticos, $\qty{1}{MeV} = \qty{1.6e-13}{J}$);
        energia por volta e número de voltas; distância percorrida.
  \item Os elétrons irradiam quando acelerados num círculo
        (\cref{ch:b2:dipole-radiation}): isso limita os betatrons a algumas
        centenas de MeV. De onde vem, no fim das contas, a energia dos elétrons,
        e por qual termo das \omterm{thm:b2:maxwell-equations:maxwell}{equações de Maxwell}?
  \item Depois da aceleração os elétrons são lançados contra um alvo e
        produzem raios X: por que o betatron foi a máquina de radioterapia
        dos anos 1950, e o que o substituiu?
\end{enumerate}

\medskip
\noindent\textbf{Parte III --- Potenciais e Poisson.}
\begin{enumerate}[resume]
  \item Escreva $\vect E$ e $\vect B$ em função de $V$ e $\vect A$, e verifique
        que Maxwell--Faraday e a lei do fluxo magnético ficam então automaticamente
        satisfeitas.
  \item Uma bola de raio $a$ carrega a densidade uniforme $\rho_0$: escreva
        a \omterm{prop:b2:maxwell-equations:potentials}{equação de Poisson} em simetria esférica, resolva-a dentro e
        fora ($V \to 0$ no infinito, $V$ e $V'$ contínuos em $a$) e
        recupere os campos do volume do Ano 1.
  \item Energia potencial da bola, $\tfrac12\int\rho V\,\dd\tau$: valor
        $3Q^2/20\pi\varepsilon_0a$; para um núcleo de urânio ($Q = 92e$, $a =
        \qty{7.4}{fm}$), em MeV --- a energia liberada quando ele se parte.
  \item Para o capacitor da Parte I em \omterm{def:b2:maxwell-equations:arqs}{regime quase estacionário}
        elétrico, dê $V$ entre as placas. Mostre que um \omterm{prop:b2:maxwell-equations:potentials}{potencial vetor}
        \emph{axial} $A_z(r, t)\,\vect e_z$, para o qual $(\operatorname{\vect{curl}}
        \vect A)_\theta = -\partial_rA_z$, reproduz o campo $B_\theta$ da
        questão 3, e ache $A_z$ para $r < R$ (tome $A_z(0) = 0$).
  \item Mostre que a correção $\vect E_1$ da questão 5 é exatamente
        $-\partial_t\vect A$ para esse $A_z$, a menos de uma constante.
  \item Calibre: a condição de Coulomb $\operatorname{div}\vect A = 0$ vale para
        o seu $A_z(r)$ --- por quê? Que outro $\vect A$ daria o mesmo $\vect B$?
\end{enumerate}

\medskip
\noindent\textbf{Parte IV --- Interfaces e regimes.}
\begin{enumerate}[resume]
  \item Carga superficial nas placas da Parte I no instante de
        $E$ máximo; verifique a relação de passagem na face interna da
        placa (campo nulo dentro do metal).
  \item A corrente se espalha radialmente na placa, do eixo até
        a borda: densidade superficial de corrente $j_s(r)$ na placa; o
        campo magnético logo fora da face externa da placa, pela
        relação de passagem, comparado com o campo achado na questão 3
        do lado interno.
  \item Classifique: as placas a \qty{1}{MHz} (quase estacionário elétrico?), a espira
        do fio de alimentação de \qty{20}{cm} (quase estacionário magnético?), o mesmo capacitor a
        \qty{1}{GHz}; justifique com $L/\lambda$.
  \item No anel do betatron, por que a \omterm{def:b2:maxwell-equations:arqs}{aproximação quase estacionária}
        magnética é excelente (tamanhos, rampa de \qty{100}{Hz}), embora o
        campo elétrico ali seja essencial?
  \item Resuma, numa tabela, que equação de Maxwell (local ou
        integral) respondeu a cada Parte, e que termo --- \omterm{thm:b2:maxwell-equations:maxwell}{corrente de
        deslocamento}, Faraday, Gauss, salto de passagem --- foi decisivo.
\end{enumerate}
\end{problem}
