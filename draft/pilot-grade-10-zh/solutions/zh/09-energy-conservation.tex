\section*{第 \ref{ch:g10:energy-conservation} 章 --- 能量：形式与守恒}

\begin{solution}{exo:g10:energy-conservation:1}
$E_k = \tfrac12 \times 1200 \times 25^2 = \qty{3.75e5}{J}$。当
速率为 \qty{36.1}{m/s} 时：$\tfrac12 \times 1200 \times 36.1^2 \approx
\qty{7.8e5}{J}$ --- 速率提高 $44\%$，动能为原来的 $(130/90)^2 \approx 2.1$
倍：平方关系在此起作用。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:2}
$\Delta h = \qty{450}{m}$：
$E_p = 72 \times 9.81 \times 450 \approx \qty{3.2e5}{J}
= \num{3.2e5}/\num{3.6e6} \approx \qty{0.088}{kW.h}$，约 
$0.022$ 欧元。攀登一天，电费仅两欧分。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:3}
$\qty{1}{kW.h} = \qty{3.6e6}{J}$;
$\qty{12}{W.h} = 12 \times \qty{3600}{J} \approx \qty{43}{kJ}$;
$\qty{45}{MJ} = 45/3.6 = \qty{12.5}{kW.h}$;
$\qty{10}{MJ} = 10/3.6 \approx \qty{2.8}{kW.h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:4}
$E = 2200 \times 150 = \qty{3.3e5}{J}
= \num{3.3e5}/\num{3.6e6} \approx \qty{0.092}{kW.h}$。人体：
$P = \num{1.0e7}/\num{86400} \approx \qty{116}{W}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:5}
发电机：化学能（骑车人的食物）$\to$ 动能（车轮）$\to$
电能（发电机）$\to$ 辐射能 $+$ 热能（灯）；热量
散失于肌肉、轮胎接触处、发电机和灯。燃气加热器：化学能（燃气）
$\to$ 热能（水），热量随烟气排出。太阳能板：辐射能
（太阳）$\to$ 电能（太阳能板）$\to$ 化学能（电池），热量
散失于太阳能板和充电器。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:6}
消耗的能量为 $2.0 \times 45 = \qty{90}{MJ}$；$\eta = 27/90 = 0.30$。
热量：$90 - 27 = \qty{63}{MJ}$，经
废气和散热器排入周围空气。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:7}
$E_k = E_p = 70 \times 9.81 \times 20 \approx \qty{1.37e4}{J}$;
$v = \sqrt{2 \times 13734/70} = \sqrt{392} \approx \qty{19.8}{m/s}
\approx \qty{71}{km/h}$.
\end{solution}

\begin{solution}{exo:g10:energy-conservation:8}
$41562 - 41236 = \qty{326}{kW.h}$；日均 $326/30 \approx
\qty{10.9}{kW.h}$。电费：$326 \times 0.26 + 12 = 96.76$ 欧元。
平均功率：$\qty{326}{kW.h}/\qty{720}{h} \approx \qty{453}{W}$。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:9}
\emph{1.} 白炽灯：$0.05 \times 60 = \qty{3.0}{W}$；发光二极管：
$\eta = 3/9 \approx 0.33$.
\emph{2.} 全年（$\qty{1460}{h}$）：灯泡 
$60 \times 1460 = \qty{87.6}{kW.h}$，发光二极管 $\qty{13.1}{kW.h}$；
每个灯泡每年节省 $\qty{74.5}{kW.h} \approx 19$ 欧元。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:10}
\emph{1.} $P = \num{1.5e5} \times 9.81 \times 40 \approx
\qty{5.9e7}{W} = \qty{59}{MW}$.
\emph{2.} $0.90 \times 59 \approx \qty{53}{MW}$;
$\num{5.3e7}/525 \approx \num{1.0e5}$ 户家庭。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:11}
$v = \sqrt{2 \times 9.81 \times 0.050} \approx \qty{0.99}{m/s}$
（质量抵消：$\tfrac12 m v^2 = mgh$）。初始
势能 $E_p = 0.20 \times 9.81 \times 0.050 \approx \qty{0.098}{J}$ 最终转化为
空气和支点中的热能：单摆不是
孤立系统，总能量 --- 摆动加热量 --- 恰好
守恒。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:12}
\emph{1.} $\eta = 0.38 \times 0.94 \times 0.85 \approx 0.304$;
消耗的能量为 $12/0.304 \approx \qty{39.5}{W.h}
= 39.5 \times 3600 \approx \qty{142}{kJ}$.
\emph{2.} 每次充电约需煤 $\num{1.42e5}/\num{2.5e7} \approx \qty{5.7}{g}$
，每年约 $\times 365 \approx \qty{2.1}{kg}$。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:13}
\emph{1.} $\num{5e9}/\num{3.6e6} \approx \qty{1400}{kW.h}$;
$1400/12.6 \approx 110$ 天。
\emph{2.} $P = \num{5e9}/\num{1e-4} = \qty{5e13}{W} = \qty{50}{TW}$
--- 相当于人类平均电功率的十七倍，
持续 \qty{100}{\micro s}，地点不可预测。闪电功率很大，
但能量很少，释放得过快而无用：没有
任何转换器能接受这样的尖峰。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:14}
\emph{1.} $E_k = \tfrac12 \times 1300 \times 30.6^2 \approx
\qty{6.1e5}{J}$，转化为刹车盘和刹车片中的
热能（再传给空气）。
\emph{2.} 每次制动约 $0.60 \times \qty{6.1e5}{J} \approx \qty{3.7e5}{J}$，
每千瓦时约 $\num{3.6e6}/\num{3.7e5} \approx 10$ 次制动。
\end{solution}

\begin{solution}{exo:g10:energy-conservation:15}
$E_p = 85 \times 9.81 \times 1200 \approx \qty{1.0}{MJ}$；食物：
$1.0/0.25 = \qty{4.0}{MJ}$，即 $4.0/1.5 \approx 2.7$ 碗
麦片 --- 或 $4.0/45 \approx \qty{0.09}{kg}$ 汽油。整整一个
上午的踏实劳动，一咖啡杯的燃料即可装下：化学能
的密集程度荒谬得惊人。
\end{solution}

\begin{solution}{pb:g10:energy-conservation:1}
\textbf{1.} 冰箱 $0.1 \times 24 = \qty{2.4}{kW.h}$；加热器
$2.4 \times 2.5 = \qty{6.0}{kW.h}$；烤箱 \qty{2.0}{kW.h}；洗衣机
\qty{1.0}{kW.h}；照明 \qty{0.3}{kW.h}；屏幕
\qty{0.9}{kW.h}。
\textbf{2.} 总计 \qty{12.6}{kW.h} $= 12.6 \times 3.6 \approx
\qty{45}{MJ}$：恰好是一千克汽油，精确到焦耳。
\textbf{3.} $\qty{12600}{W.h}/\qty{24}{h} = \qty{525}{W}$ --- 这所房子
的平均功率仅为一只水壶的四分之一。
\textbf{4.} 热水器（\qty{6.0}{kW.h}，将近一半）。在
\qty{1.25}{h} 内：加热器 \qty{3.0}{kW.h}，总计 \qty{9.6}{kW.h}。
\textbf{5.} $15 \times 24 = \qty{0.36}{kW.h}$ 每天，约占
计量总量的 $2.8\%$；全年 $0.36 \times 365 \approx
\qty{131}{kW.h} \approx 33$ 欧元 --- 全都耗在无人照看的灯上。
\textbf{6.} $56604 - 56214 = \qty{390}{kW.h}$；
$390/30 = \qty{13.0}{kW.h}$ 每天：审计所得的 \qty{12.6}{kW.h}
加上待机的 \qty{0.36}{kW.h} --- 得到证实。
\textbf{7.} $390 \times 0.25 + 14 = 111.50$ 欧元。
\textbf{8.} 电网：$0.25/3.6 \approx 0.07$ 欧元每兆焦。
汽油：$1.80/34 \approx 0.05$ 欧元每兆焦。
\textbf{9.} 麦片碗：$0.50/1.5 \approx 0.33$ 欧元每兆焦 ---
是电网的五倍。由便宜到贵依次为汽油、电网，然后是食物，
远远落在后面。
\textbf{10.} $0.20 \times 1000 = \qty{200}{W}$ 每平方米。
\textbf{11.} $200 \times 4.0 = \qty{800}{W.h} = \qty{0.80}{kW.h}$
每平方米每天。
\textbf{12.} $12.6/0.80 = 15.75 \approx \qty{16}{m^2}$：可以装进
\qty{20}{m^2} 的屋顶。
\textbf{13.} $16 \times 0.2 \times 1.5 = \qty{4.8}{kW.h}$：约占
需求的 $38\%$；其余部分须由电网（或在气候较好的月份充好电的
电池）供应。
\textbf{14.} 每平方米每年产出 $0.80 \times 365 \approx
\qty{292}{kW.h}$；回收期 $700/292 \approx 2.4$ 年 ---
相对于 25 年的面板寿命而言很短。
\textbf{15.} $\qty{12.6}{kW.h}/\qty{0.1}{kW} = \qty{126}{h}$：
需要五人以上日夜不停地蹬车（$126/24 = 5.25$）。
\textbf{16.} 食物 $= 12.6/0.25 = \qty{50.4}{kW.h} \approx
\qty{181}{MJ}$，即 $181/1.5 \approx 121$ 碗麦片。
\textbf{17.} $121 \times 0.50 \approx 60$ 欧元，而
电网供能仅需 $12.6 \times 0.25 \approx 3.15$ 欧元：人力发电的
成本约为其 $19$ 倍 --- 尚未计入骑车人的报酬。
\textbf{18.} $5000/45 \approx 110$ 个家庭日。但
$\num{5e9}/\num{1e-4} = \qty{5e13}{W}$ 持续 \qty{100}{\micro s}，
出现于随机地点：没有任何转换器或电池能从这样的水管中取水。
一户住宅需要的是稳定的 \qty{525}{W} 供给，而非瞬时的峰值。
\textbf{19.} 不是焦耳本身 --- 它们守恒且彼此相同。你所付的是
它们的\emph{形式与可用性}：集中、可储存、可随时取用的能量，
其代价是
促成这一切的整条转化链。
\textbf{20.} 一个家庭一天的用量 $= \qty{12.6}{kW.h} \approx
\qty{45}{MJ}$：相当于一千克汽油、十六平方米晴朗的屋顶，
或五名日夜不停蹬车的人。存量是能量
（焦耳、千瓦时）；流量是功率（瓦特）--- 水池与水龙头，
从此不再混淆。
\end{solution}
