\section*{第 \ref{ch:g12:oscillators-and-time} 章——机械振动与时间的测量}

\begin{solution}{exo:g12:oscillators-and-time:1}
(a) $T_0 = 2\pi\sqrt{1.00/9.81} = \qty{2.01}{s}$；
$f_0 = 1/T_0 = \qty{0.50}{Hz}$。(b) $T_0 \propto \sqrt{L}$：周期
变为原来的两倍。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:2}
心脏：$f = 72/60 = \qty{1.2}{Hz}$，$T = \qty{0.83}{s}$。翅膀：
$T = 1/30 = \qty{33}{ms}$。胸部的起伏即为\emph{振幅}。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:3}
$T_0 = 2\pi\sqrt{0.10/25} = \qty{0.40}{s}$，$f_0 = \qty{2.5}{Hz}$。
把 $m$ 加倍时：$T_0 \times \sqrt{2} \approx 1.4$。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:4}
准周期为 \qty{2.0}{s}。各周期的比值 $1.5/2.0 = 1.13/1.5 = 0.85/1.13
= 0.75$；每个周期如此，下一个最大值为 $0.85 \times 0.75 \approx
\qty{0.64}{cm}$。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:5}
单摆：$T_0 \propto 1/\sqrt{g}$，因此乘以 $\times\sqrt{9.81/1.62} =
2.46$（变慢）。弹簧振子：不变——$g$ 不出现在
$2\pi\sqrt{m/k}$ 中。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:6}
(a) $[L/g] = \unit{m}/(\unit{m/s^2}) = \unit{s^2}$；其平方根
为时间。(b) 千克只出现在 $m$ 中：无其他量能将其约去，
故 $m$ 不能进入结果。(c) 永远不能：像 $2\pi$ 这样的纯数，
在量纲分析中不可见。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:7}
$\frac{dx}{dt} = -\frac{2\pi}{T_0}A\sin(2\pi t/T_0 + \varphi)$，故
$\frac{d^2x}{dt^2} = -(2\pi/T_0)^2 x(t)$。要使其与 $-(k/m)x$
对所有 $t$ 相等，当且仅当 $(2\pi/T_0)^2 = k/m$，即
$T_0 = 2\pi\sqrt{m/k}$。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:8}
$E_m = \tfrac12 kA^2 = \tfrac12 \times 40 \times 0.030^2 =
\qty{1.8e-2}{J}$；$v_{\max} = \sqrt{2E_m/m} = \sqrt{0.18} =
\qty{0.42}{m/s}$——在 $x = 0$ 处；在两端点 $x = \pm A$ 处为零。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:9}
(a) $L = gT_0^2/4\pi^2 = \qty{0.994}{m}$。(b) $\Delta T_0/T_0 =
\tfrac12 \times 1.0\% = 0.5\%$：$86400 \times 0.005 \approx
\qty{4.3e2}{s}$——每天慢约七分钟。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:10}
(a) $A \propto \sqrt{E}$：$\sqrt{0.95} \approx 0.975$，下降
$2.5\%$。(b) $0.95^n = 0.5$：$n = \ln 2/\ln(1/0.95) \approx 14$
个周期。(c) 不会：阻尼较轻时，周期与
$T_0$ 相差不到百分之一。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:11}
(a) 在 $f_0$ 处，每次推动都同步到达并增加能量；偏离峰值时，
推动时助时阻。(b) 同步齐步走是一种周期性驱动；
在桥梁的 $f_0$ 附近，振幅将攀升至
共振峰。(c) 鼓在启动过程中
扫过框架的固有频率：产生瞬时共振，随后越过共振峰。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:12}
(a) $T_0 \propto 1/\sqrt{g}$：变慢，相对变化 $\Delta T_0/T_0 = \tfrac12
\times 0.03/9.81 = \num{1.5e-3}$，即
每天 $\approx \qty{132}{s}$。(b) $L \propto g$（$T_0$ 固定时）：缩短 $0.994 \times
0.03/9.81 \approx \qty{3.0}{mm}$。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:13}
$v_{\max} = 2\pi A/T_0 = 2\pi \times 0.040/0.63 = \qty{0.40}{m/s}$；
$a_{\max} = (2\pi/T_0)^2 A = \qty{4.0}{m/s^2}$；$k = m(2\pi/T_0)^2
= \qty{20}{N/m}$；$E_m = \tfrac12 kA^2 = \qty{1.6e-2}{J}$。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:14}
$T_0 = 40.1/20 = \qty{2.005}{s}$；$g = 4\pi^2 L/T_0^2 = 4\pi^2
\times 0.995/4.020 = \qty{9.77}{m/s^2}$。不确定度 $\approx 0.2\%
+ 2 \times 0.5\% = 1.2\%$，即 $\pm\qty{0.12}{m/s^2}$：
区间包含 \qty{9.81}{m/s^2}——结果一致。
\end{solution}

\begin{solution}{exo:g12:oscillators-and-time:15}
单摆 $10/86400 \approx \num{1.2e-4}$；石英 $0.5/\num{2.6e6}
\approx \num{1.9e-7}$；铯 $1/\num{3.2e15} \approx \num{3e-16}$。
一个 \qty{1}{\micro s} 的误差，相当于 $\num{1000} \times \qty{30}{cm} =
\qty{300}{m}$ 的位置偏差：GPS 需要原子钟一类的时钟。
\end{solution}

\begin{solution}{pb:g12:oscillators-and-time:1}
\textbf{1.} 反应误差 $\approx \qty{0.2}{s}$ 被
$50$ 除，对 $T_0$ 的影响约为 \qty{4}{ms}。

\textbf{2.} $T = 99.4/50 = \qty{1.988}{s}$。

\textbf{3.} $L = gT^2/4\pi^2 = 9.81 \times 1.988^2/4\pi^2 \approx
\qty{0.982}{m}$。

\textbf{4.} $L = \qty{0.994}{m}$（秒摆）：将
摆锤下移，使摆长增加 $\approx \qty{1.2}{cm}$。

\textbf{5.} 摆长过短，走快，每天快 $(2.000 -
1.988)/1.988 \times 86400 \approx \qty{5.2e2}{s} \approx 8.7$
分钟。

\textbf{6.} 月球：$\times\sqrt{9.81/1.62} = 2.46$——不可用。
天文台：$\Delta T_0/T_0 = \tfrac12 \times 0.01/9.81 =
\num{5.1e-4}$：每天慢 $\approx \qty{44}{s}$。

\textbf{7.} $\Delta L = L\lambda\,\Delta\theta = 0.994 \times
\num{1.9e-5} \times 5.0 \approx \qty{9.4e-5}{m} \approx
\qty{0.1}{mm}$。

\textbf{8.} $T' = 2\pi\sqrt{L(1 + \lambda\Delta\theta)/g} =
T_0\sqrt{1 + \lambda\Delta\theta} \approx T_0\,(1 + \tfrac12
\lambda\Delta\theta)$。

\textbf{9.} $\Delta T_0/T_0 = \tfrac12 \times \num{1.9e-5} \times
5.0 = \num{4.8e-5}$：$86400 \times \num{4.8e-5} \approx
\qty{4.1}{s}$ 每天，偏慢（杆变长，周期变长）。

\textbf{10.} $\Delta\theta = \qty{-10}{K}$：$\Delta T_0/T_0 =
\num{-9.5e-5}$，每天快 $\approx \qty{8.2}{s}$。

\textbf{11.} 钢与黄铜膨胀不同，两者交替反向悬挂；
黄铜杆使摆锤下降，钢杆则使其上升，
经设计使两者的变化相互抵消，等效摆长——
以及走时——不随季节而变。

\textbf{12.} $280/2.000 = 140$ 个周期。

\textbf{13.} $\theta_m = 2.00^\circ = \qty{0.0349}{rad}$：$E_m =
\tfrac12 \times 1.2 \times 9.81 \times 0.994 \times 0.0349^2 \approx
\qty{7.1e-3}{J}$。

\textbf{14.} 每个周期内振幅缩小为原来的 $2^{-1/140} \approx
0.9951$，能量则乘以 $0.9951^2 \approx 0.990$，即每周期约损失 $1.0\%$
的 $E_m$。

\textbf{15.} 每次推动补充的能量 $\approx 0.010 \times \num{7.1e-3} =
\qty{7.1e-5}{J}$；功率 $\num{7.1e-5}/2.0 \approx \qty{3.5e-5}{W}$
——数十微瓦。

\textbf{16.} 一周共 $604800/2.0 = \num{3.02e5}$ 个周期，
需能量 $\num{3.02e5} \times \num{7.1e-5} \approx \qty{21}{J}$；重物
提供 $Mgh = 2.5 \times 9.81 \times 1.0 = \qty{24.5}{J}$：足够，
效率约为 $21/24.5 \approx 88\%$。

\textbf{17.} $T = 1/\num{32768} = \qty{30.5}{\micro s}$。$\num{32768}
= 2^{15}$：十五级二分频将晶体的信号
变为精确的 \qty{1}{Hz} 滴答。

\textbf{18.} 石英：$0.3/\num{2.6e6} \approx \num{1.2e-7}$。
摆钟：$0.5/86400 \approx \num{5.8e-6}$，即 $\approx
\qty{15}{s}$ 每月。石英的精度约高 $\approx 50$ 倍。

\textbf{19.} $1/\num{3.2e15} \approx \num{3e-16}$：比石英低九个
数量级。GPS 将信号时间精确到纳秒——
光每纳秒传播 \qty{30}{cm}——，故只有原子钟才够用。

\textbf{20.} 修复后，挂钟每月误差保持在 $\pm\qty{15}{s}$ 以内，
相比之下石英仅为每月 $\pm\qty{0.3}{s}$：挂钟留作报时，
走时则交由石英负责——而秒本身
如今由铯原子定义。
\end{solution}
